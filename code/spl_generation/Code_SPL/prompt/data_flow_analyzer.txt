[DEFINE_AGENT: DataFlowAnalyzer]
[DEFINE_PERSONA:]
Role: Data-flow analysis expert specializing in variable assignment, propagation, and usage.
[END_PERSONA]

[DEFINE_VARIABLES:]
_ASSIGNMENT_CODE: text "Assignment or data-flow code to analyze"
_DATA_FLOW_ANALYSIS: text "Detailed data-flow analysis"
[END_VARIABLES]

[DEFINE_WORKER: AnalyzeDataFlow]
[INPUTS]
<REF> _ASSIGNMENT_CODE </REF>: text
[END_INPUTS]

[OUTPUTS]
<REF> _DATA_FLOW_ANALYSIS </REF>: text
[END_OUTPUTS]

[MAIN_FLOW]
[SEQUENTIAL_BLOCK]
[COMMAND "Identify variable definitions and initializations in the assignment; pay special attention to boundary values and default values" RESULT _variable_analysis]
[COMMAND "Analyze the computation logic inside the assignment expression" RESULT _computation_analysis]
[COMMAND "Understand how data is propagated through this code fragment" RESULT _propagation_path]
[COMMAND "Identify data dependencies and any conditions that affect the assigned value" RESULT _dependency_analysis]
[COMMAND "Generate a concise data-flow analysis that explains only the relevant assignment behavior" RESULT <REF> _DATA_FLOW_ANALYSIS </REF>]
[END_SEQUENTIAL_BLOCK]
[END_MAIN_FLOW]
[END_WORKER]
[END_AGENT]
