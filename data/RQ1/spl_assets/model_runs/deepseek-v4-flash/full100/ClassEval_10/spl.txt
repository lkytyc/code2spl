[DEFINE_WORKER: "Counts the zero and one characters in binary_string, computes their proportions relative to the total length, and returns those proportions with the bit length as a dictionary." calculate_binary_info]
    [INPUTS]
        <REF> self.binary_string </REF>: string "The binary string to analyze for zero and one characters. If empty, percentage computation raises ZeroDivisionError."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "Dictionary with keys 'Zeroes' and 'Ones' as float proportions and 'Bit length' as the total character count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the number of '0' characters in self.binary_string. RESULT zeroes_count]
            [COMMAND Count the number of '1' characters in self.binary_string. RESULT ones_count]
            [COMMAND Compute total_length as the length of self.binary_string. RESULT total_length]
            [COMMAND Compute zeroes_percentage as zeroes_count divided by total_length. RESULT zeroes_percentage]
            [COMMAND Compute ones_percentage as ones_count divided by total_length. RESULT ones_percentage]
            [COMMAND Return a dictionary with keys 'Zeroes', 'Ones', and 'Bit length' in that order using the computed values. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.binary_string is empty, so total_length is zero]
        [LOG "Division by zero while computing the percentage values."]
        [THROW ZeroDivisionError "Raised when computing a percentage with total_length equal to zero."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the binary string in self.binary_string to ASCII text by splitting it into 8-bit chunks, converting each chunk to an integer byte value, appending those bytes to a bytearray, and decoding the bytearray as ASCII." convert_to_ascii]
    [INPUTS]
        <REF> self.binary_string </REF>: string "The binary string of '0' and '1' characters to be converted; it is read in sequential 8-character chunks."
    [END_INPUTS]

    [OUTPUTS]
        <REF> ascii_string </REF>: string "The ASCII-decoded string produced by byte_array.decode('ascii')."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize byte_array as a new empty bytearray. RESULT byte_array]
            [COMMAND Loop with index i starting at 0, incrementing by 8, and continuing while i < len(self.binary_string); the loop processes self.binary_string in 8-character chunks. RESULT i]
            [COMMAND Inside the loop, extract the substring byte = self.binary_string[i:i+8]; the final chunk may be shorter than 8 characters if len(self.binary_string) is not a multiple of 8. RESULT byte]
            [COMMAND Inside the loop, convert byte from a base-2 string to an integer using decimal = int(byte, 2), yielding a byte value from 0 to 255 for a full 8-character chunk. RESULT decimal]
            [COMMAND Inside the loop, append decimal to byte_array using byte_array.append(decimal). RESULT byte_array]
            [COMMAND After the loop completes, decode byte_array using ASCII encoding via byte_array.decode('ascii') and return the resulting string. RESULT ascii_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a binary string into UTF-8 text by grouping every 8 bits into bytes and decoding the resulting bytearray." convert_to_utf8]
    [INPUTS]
        <REF> self.binary_string </REF>: str "Binary string containing only '0' and '1' characters, representing UTF-8 encoded bytes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "UTF-8 decoded string produced from the binary string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty bytearray named byte_array to collect byte values. RESULT byte_array]
            [COMMAND Iterate i over range(0, len(self.binary_string), 8) to process the binary string in 8-character chunks; the stop value is exclusive, so a trailing chunk shorter than 8 characters is also processed. RESULT i]
            [COMMAND Within the for loop over i, slice self.binary_string from i to i+8 to obtain the current binary chunk. RESULT byte]
            [COMMAND Within the for loop over i, convert the binary chunk byte to an integer using int(byte, 2). RESULT decimal]
            [COMMAND Within the for loop over i, append decimal to byte_array. RESULT byte_array]
            [COMMAND After the loop completes, decode byte_array using UTF-8 and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The bytes collected in byte_array do not form valid UTF-8 byte sequences during decode.]
        [LOG "Invalid UTF-8 sequence encountered while decoding byte_array."]
        [THROW UnicodeDecodeError "Raised when byte_array contains bytes that are not valid UTF-8."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
