# CLASS ChandrasekharSieve

[DEFINE_WORKER: "Initializes the object by storing the supplied n in self.n and computing self.primes through self.generate_primes()." __init__]
    [INPUTS]
        <REF> n </REF>: unknown "Input value used to initialize self.n and to drive prime generation through self.generate_primes()."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the input value n into self.n. RESULT self.n]
            [COMMAND Call self.generate_primes() and assign the returned prime list to self.primes. RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates all prime numbers from 2 through self.n inclusive using the Sieve of Eratosthenes, returning an empty list when self.n is less than 2." generate_primes]
    [INPUTS]
        <REF> self.n </REF>: int "Inclusive upper bound for prime generation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[int] "List of all prime numbers from 2 through self.n inclusive; empty list when self.n is less than 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a boolean list sieve of length self.n + 1 with every entry set to True. RESULT sieve]
            [COMMAND Set sieve[0] and sieve[1] to False to mark 0 and 1 as non-prime. RESULT sieve]
            [COMMAND Initialize p = 2 as the first prime candidate. RESULT p]
            [COMMAND Start the outer sieve-marking while loop that continues while p * p <= self.n. RESULT p]
            [COMMAND Within the while loop over p, if sieve[p] is True, iterate i over range(p * p, self.n + 1, p) so i takes every multiple of p from p squared through self.n inclusive, and set sieve[i] = False for each such i. RESULT sieve]
            [COMMAND At the end of each iteration of the while loop, increment p by 1. RESULT p]
            [COMMAND Initialize an empty list primes to collect the prime numbers. RESULT primes]
            [COMMAND Loop i from 2 through self.n inclusive using range(2, self.n + 1); if sieve[i] is True, append i to primes. RESULT primes]
            [COMMAND Return primes as the final result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.n < 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately without initializing the sieve or collecting primes. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current value of the instance attribute primes to the caller." get_primes]
    [INPUTS]
        <REF> self </REF>: object "The object instance whose primes attribute is accessed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "The current value stored in self.primes, returned unchanged."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value of the instance attribute primes to the caller. RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
