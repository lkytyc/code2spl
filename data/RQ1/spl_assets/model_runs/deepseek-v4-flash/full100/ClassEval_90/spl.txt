[DEFINE_WORKER: "Extracts the URI scheme from self.url by finding the first '://' and returning the prefix before it, or None if no separator is present." get_scheme]
    [INPUTS]
        <REF> self </REF>: object "Instance with a url string attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string|null "The substring of self.url before the first '://', or None if '://' is not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the index of the first occurrence of '://' in self.url using the string find method and assign the result to scheme_end. RESULT scheme_end]
            [COMMAND When scheme_end is -1, return None because self.url contains no scheme separator. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: scheme_end != -1, meaning self.url contains '://']
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.url[:scheme_end], the substring from the start of self.url up to but excluding the '://' separator, and exit the method. RESULT scheme]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts the host portion from a URL by removing the scheme delimiter and returning the substring before the first slash, or None if no scheme delimiter is present." get_host]
    [INPUTS]
        <REF> self </REF>: object "Object with a url attribute containing the URL string to parse."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string|null "The host portion extracted from self.url, or None if no '://' delimiter is present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first occurrence of the substring '://' in self.url and store its index in scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not -1, slice self.url from scheme_end + 3 to the end and store the result in url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the outer if block where the scheme delimiter exists, find the first occurrence of '/' in url_without_scheme and store its index in host_end. RESULT host_end]
            [COMMAND Within the outer if block, if host_end is not -1, return the substring of url_without_scheme from the beginning up to host_end, excluding the slash. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: scheme_end is -1, meaning self.url does not contain the '://' delimiter.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None to indicate that no host could be extracted. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the outer if block where scheme_end is not -1, host_end is -1 because url_without_scheme contains no '/' character.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the entire url_without_scheme as the host. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the path portion of self.url starting at the first '/' that appears after the '://' scheme delimiter, or None if no scheme delimiter or path slash is present." get_path]
    [INPUTS]
        <REF> self </REF>: object "The object instance whose url attribute contains the URL string to parse."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "The substring of self.url beginning at the first '/' after '://', or None if '://' is absent or no '/' occurs after the scheme delimiter."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search self.url for the first occurrence of the literal string '://' and store its index in scheme_end, or store -1 if it is not found. RESULT scheme_end]
            [COMMAND In the branch where scheme_end is not -1, create url_without_scheme by slicing self.url from index scheme_end + 3 to the end, removing the scheme and '://'. RESULT url_without_scheme]
            [COMMAND In the same scheme_end is not -1 branch, search url_without_scheme for the first occurrence of '/' and store its index in host_end, or store -1 if it is not found. RESULT host_end]
            [COMMAND In the nested branch where scheme_end is not -1 and host_end is not -1, return url_without_scheme[host_end:], which includes the leading '/' and the remaining path, then exit the method. RESULT path_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The delimiter '://' is not found in self.url, so scheme_end is -1.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None because self.url does not contain the '://' scheme delimiter. RESULT none_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The delimiter '://' is found, but url_without_scheme contains no '/' after it, so scheme_end is not -1 and host_end is -1.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None because url_without_scheme does not contain a '/' after the scheme delimiter. RESULT none_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

    [END_INPUTS]

    [OUTPUTS]

        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
