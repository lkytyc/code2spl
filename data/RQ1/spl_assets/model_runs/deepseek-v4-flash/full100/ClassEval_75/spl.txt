# CLASS ShoppingCart

[DEFINE_WORKER: "Initializes a new instance by setting its items attribute to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.items as a new empty dictionary. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Unconditionally stores the given price and quantity under item in self.items, overwriting any existing entry." add_item]
    [INPUTS]
        <REF> item </REF>: object "The dictionary key under which the item data is stored."
        <REF> price </REF>: object "The price value to store in the item entry."
        <REF> quantity </REF>: int "The quantity to store in the item entry; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item is present as a key in self.items. RESULT _item_present]
            [COMMAND Set self.items[item] to the dictionary {'price': price, 'quantity': quantity}. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrements the stored quantity for an item in self.items by the given quantity if the item exists, and does nothing otherwise." remove_item]
    [INPUTS]
        <REF> item </REF>: object "Key to look up in self.items; it must be a hashable dictionary key."
        <REF> quantity </REF>: int "Amount to subtract from the item's quantity; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item exists as a key in self.items. RESULT item_exists]
            [COMMAND If item_exists is true, subtract quantity from the value at self.items[item]['quantity'] and assign the result back to self.items[item]['quantity']. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do nothing and continue. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current items dictionary stored on the view." view_items]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> items </REF>: dict "The stored items dictionary."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the stored items dictionary to the caller. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total price by multiplying each item's quantity by its price and summing the resulting subtotals." total_price]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The total price computed as the sum of quantity times price for all items in self.items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the items collection from self.items and obtain its values view. RESULT item_values]
            [COMMAND For each item in item_values, read item['quantity'] and item['price'], multiply them to compute the item subtotal, and collect all subtotals into a list. RESULT subtotals]
            [COMMAND Sum the subtotals; if the list is empty, treat the sum as 0. RESULT total]
            [COMMAND Return total as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
