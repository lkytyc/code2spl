# CLASS SQLGenerator

[DEFINE_WORKER: "Initializes a new object by storing the provided table name in the instance attribute self.table_name." __init__]
    [INPUTS]
        <REF> table_name </REF>: string "The name of the table to assign to this instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of table_name to the instance attribute self.table_name. RESULT self.table_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a SQL SELECT query for the configured table, selecting all columns when fields is None or a comma-separated column list otherwise, and optionally appends a WHERE clause before terminating with a semicolon." select]
    [INPUTS]
        <REF> fields </REF>: list[str] or None "Optional list of column names to select. If exactly None, the wildcard '*' is used. Otherwise the iterable elements are joined with ', '."
        <REF> condition </REF>: str or None "Optional SQL WHERE condition string. If not exactly None, it is appended to the query as ' WHERE {condition}' even when the string is empty."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The completed SQL SELECT query string ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND At function level, if fields is exactly None, set normalized_fields to '*'; otherwise join fields with ', ' to produce normalized_fields, even if the join result is empty. RESULT normalized_fields]
            [COMMAND Construct sql using the f-string 'SELECT {normalized_fields} FROM {self.table_name}'. RESULT sql]
            [COMMAND If condition is not exactly None, append ' WHERE {condition}' to sql; otherwise leave sql unchanged. RESULT sql]
            [COMMAND Return sql with ';' appended as the final query string. RESULT final_query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds an unescaped SQL INSERT statement by using the data dictionary keys as column names and single-quoted values as row values, interpolating them with self.table_name, and returning the statement with a trailing semicolon." insert]
    [INPUTS]
        <REF> data </REF>: dict "Dictionary containing column names as keys and the corresponding values for one SQL row."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "SQL INSERT statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join the keys of the data dictionary in insertion order with ', ' and assign the resulting column-list string to fields. RESULT fields]
            [COMMAND For each value in data.values(), convert it to a string, wrap it in single quotes without escaping, join all quoted values with ', ', and assign the resulting value-list string to values. RESULT values]
            [COMMAND Build the SQL string by directly interpolating self.table_name, fields, and values into the format INSERT INTO {self.table_name} ({fields}) VALUES ({values}) and assign it to sql. RESULT sql]
            [COMMAND Append a semicolon to sql, assign the resulting complete SQL statement to final_sql, and return final_sql. RESULT final_sql]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a SQL UPDATE statement from a dictionary of column-value pairs and a WHERE condition, returning the statement string with a trailing semicolon." update]
    [INPUTS]
        <REF> data </REF>: dict "Dictionary mapping column names to values for the SET clause."
        <REF> self.table_name </REF>: str "Name of the database table to update."
        <REF> condition </REF>: str "SQL WHERE condition text used in the UPDATE statement."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Final SQL UPDATE statement including the WHERE clause and terminating semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each field and value in data.items(), format the pair as "field = 'value'", then join all formatted pairs with ", " to produce the SET clause. RESULT set_clause]
            [COMMAND Interpolate self.table_name, set_clause, and condition into the SQL template "UPDATE {self.table_name} SET {set_clause} WHERE {condition}". RESULT sql]
            [COMMAND Append ";" to sql and return the resulting string. RESULT final_sql]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a SQL DELETE statement for the object's table using the given condition, appends a semicolon, and returns the final SQL string." delete]
    [INPUTS]
        <REF> self </REF>: object "The object on which delete is called; must provide a table_name attribute containing the target table name."
        <REF> condition </REF>: str "The WHERE clause condition to be embedded verbatim in the DELETE statement."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The final SQL DELETE statement, ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build the SQL DELETE statement by interpolating self.table_name and condition into the f-string template 'DELETE FROM {self.table_name} WHERE {condition}', and assign the resulting string to sql. RESULT sql]
            [COMMAND Append a semicolon to sql and return the resulting SQL string as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a condition string for records where age is less than the supplied age and gender equals 'female', then delegates to self.select with that condition." select_female_under_age]
    [INPUTS]
        <REF> age </REF>: object "The age threshold value interpolated into the condition string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> selection_result </REF>: object "The return value of self.select(condition=condition)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct the condition string by interpolating age into the literal text "age < {age} AND gender = 'female'". RESULT condition]
            [COMMAND Call self.select with condition=condition and return its result. RESULT selection_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds an age BETWEEN condition from min_age and max_age and returns the result of selecting records using that condition." select_by_age_range]
    [INPUTS]
        <REF> min_age </REF>: number "Lower inclusive boundary for the age range."
        <REF> max_age </REF>: number "Upper inclusive boundary for the age range."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "Return value of self.select, usually a collection of matching records."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct the condition string as 'age BETWEEN {str(min_age)} AND {str(max_age)}' using f-string interpolation, without validating whether min_age is less than or equal to max_age. RESULT condition]
            [COMMAND Call self.select with the keyword argument condition set to condition and return its result directly. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
