    [END_INPUTS]

[DEFINE_WORKER: "Appends four random integers from 1 to 9 inclusive to self.nums in a for loop, then asserts that the total length of self.nums is exactly 4." _generate_cards]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each i in range(4), so i takes values 0, 1, 2, and 3, append a random integer from 1 to 9 inclusive, generated by random.randint(1, 9), to self.nums. RESULT self.nums]
            [COMMAND Assert that len(self.nums) is exactly 4. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: len(self.nums) is not exactly 4 after the loop finishes]
        [LOG "Assertion failed: len(self.nums) == 4"]
        [THROW AssertionError "Raised because the length of self.nums is not exactly 4 after appending the four random integers."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resets the nums attribute to an empty list, calls _generate_cards to populate it, and returns the resulting list of cards." get_my_cards]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The generated card list stored in self.nums."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Reset the instance attribute nums to a new empty list. RESULT self.nums]
            [COMMAND Call the internal method _generate_cards() to generate the cards into self.nums. RESULT self.nums]
            [COMMAND Return self.nums as the method result. RESULT self.nums]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the player's cards for 'pass'; otherwise returns evaluate_expression(expression) exactly when the expression's allowed-digit counts match the required counts from self.nums, and returns False otherwise." answer]
    [INPUTS]
        <REF> expression </REF>: str "The expression string to inspect; may be the literal string 'pass'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The result of self.get_my_cards() if expression == 'pass'; the result of self.evaluate_expression(expression) when the digit counts match self.nums exactly; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If expression equals the exact string 'pass', return self.get_my_cards() immediately and stop. RESULT self.get_my_cards()]
            [COMMAND Initialize statistic as an empty dictionary. RESULT statistic]
            [COMMAND For each character c in expression, if c.isdigit() is true and int(c) is in self.nums, set statistic[c] to statistic.get(c, 0) + 1; otherwise do nothing for c. RESULT statistic]
            [COMMAND Copy statistic into nums_used using statistic.copy(). RESULT nums_used]
            [COMMAND For each num in self.nums: if nums_used.get(str(num), -100) is not equal to -100 and nums_used[str(num)] > 0, decrement nums_used[str(num)] by 1; otherwise return False immediately. RESULT nums_used]
            [COMMAND If all(count == 0 for count in nums_used.values()) is true, return self.evaluate_expression(expression); otherwise return False. RESULT final_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

        <REF> expression </REF>: string "The expression string to be evaluated with eval()."

    [OUTPUTS]

    [MAIN_FLOW]
