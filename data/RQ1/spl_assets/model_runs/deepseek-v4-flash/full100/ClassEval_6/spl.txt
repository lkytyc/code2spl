# CLASS AvgPartition

[DEFINE_WORKER: "Initializes a new instance by directly storing the provided list and limit values as instance attributes, without validation, coercion, or boundary checks." __init__]
    [INPUTS]
        <REF> lst </REF>: list "List object to be stored as self.lst using the exact same object reference."
        <REF> limit </REF>: any "Limit value to be stored as self.limit exactly as provided, with no type or boundary validation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the parameter lst directly to the instance attribute self.lst, preserving the exact object reference. RESULT self.lst]
            [COMMAND Assign the parameter limit directly to the instance attribute self.limit, preserving the value exactly as provided. RESULT self.limit]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Partitions the length of the stored list by the stored limit into a quotient and a remainder, then returns both values." setNum]
    [INPUTS]
        <REF> self </REF>: object "Object providing the list attribute 'lst', whose length is partitioned, and the integer attribute 'limit', used as the divisor."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: tuple[int, int] "A tuple (size, remainder) where size = len(self.lst) // self.limit and remainder = len(self.lst) % self.limit. For a positive limit, 0 <= remainder < self.limit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate len(self.lst), divide the result by self.limit using floor division, and assign the quotient to size. RESULT size]
            [COMMAND Evaluate len(self.lst), compute the remainder after dividing by self.limit using the modulo operation, and assign it to remainder. RESULT remainder]
            [COMMAND Return the tuple (size, remainder). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.limit == 0]
        [LOG "ZeroDivisionError is raised when the first assignment attempts floor division of len(self.lst) by zero; the subsequent modulo operation is not reached."]
        [THROW ZeroDivisionError "Raised because self.limit is zero."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the sublist for a given chunk index by splitting self.lst into balanced chunks, with the first remainder chunks containing one extra element." get]
    [INPUTS]
        <REF> index </REF>: int "Zero-based chunk index to retrieve."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The sublist self.lst[start:end] corresponding to the requested chunk."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.setNum() and unpack the returned tuple into size and remainder. RESULT size, remainder]
            [COMMAND Compute start as index * size + min(index, remainder). RESULT start]
            [COMMAND Compute end as start + size. RESULT end]
            [COMMAND If index + 1 <= remainder, increment end by 1; otherwise leave end unchanged. RESULT end]
            [COMMAND Return the slice self.lst[start:end], where end is exclusive. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
