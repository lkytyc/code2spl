[DEFINE_WORKER: "Adds a new username to the users dictionary if it does not already exist, returning True on success and False on duplicate." add_user]
    [INPUTS]
        <REF> username </REF>: str "The username to add to the users dictionary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the username was newly added; False if the username already exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether username is already a key in the self.users dictionary. RESULT membership_check]
            [COMMAND If username is not already present in self.users, assign an empty list as the initial value for that username. RESULT user_entry]
            [COMMAND Return True to indicate that the new user was successfully added. RESULT add_user_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If username is already present in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.users to indicate that the username already exists. RESULT duplicate_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the given username from the users dictionary if it exists and returns True; otherwise returns False without modifying the dictionary." remove_user]
    [INPUTS]
        <REF> username </REF>: str "The username key to look up and remove from self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if the username was found and removed; False if the username was not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists as a key in self.users. RESULT username_exists]
            [COMMAND If username_exists is True, delete self.users[username] and return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username_exists is False]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the else branch, return False without modifying self.users. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Records a message by appending a timestamped message_info dictionary to both sender and receiver message lists after validating both users are registered, and returns True on success or False if either user is invalid." send_message]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the users dictionary that maps user names to lists of message_info records."
        <REF> sender </REF>: str "The username of the message sender; must exist as a key in self.users."
        <REF> receiver </REF>: str "The username of the message receiver; must exist as a key in self.users."
        <REF> message </REF>: str "The content of the message to store in both users' message_info lists."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the message_info record was appended to both users' lists; False if either user was not found in self.users."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Generate the current timestamp as a string formatted as '%Y-%m-%d %H:%M:%S'. RESULT timestamp]
            [COMMAND Create message_info dictionary with sender, receiver, message, and timestamp. RESULT message_info]
            [COMMAND Append message_info to the list stored at self.users[sender]. RESULT sender_message_list]
            [COMMAND Append message_info to the list stored at self.users[receiver]; if receiver equals sender, this appends a second copy to the same list. RESULT receiver_message_list]
            [COMMAND Return True to indicate successful message delivery. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: sender is not in self.users, or receiver is not in self.users (using Python short-circuit OR, evaluated left-to-right)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without appending any message_info or modifying self.users. RESULT false_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the messages associated with the given username, or an empty list if the username is not present in self.users." get_messages]
    [INPUTS]
        <REF> username </REF>: str "The username whose stored messages are requested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> messages </REF>: list "The stored message list for the username, or an empty list if the username is not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether username is not present in self.users. RESULT username_missing]
            [COMMAND If username_missing is true, return an empty list immediately and skip the rest of the method. RESULT []]
            [COMMAND If username_missing is false, return the value stored for username in self.users. RESULT self.users[username]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
