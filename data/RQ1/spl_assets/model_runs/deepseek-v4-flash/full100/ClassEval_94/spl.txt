[DEFINE_WORKER: "Tries to restock the requested item; if the restock call returns a falsey value, it falls back to directly setting the item's inventory entry to the supplied price and quantity." add_item]
    [INPUTS]
        <REF> item_name </REF>: string "The name of the inventory item to add or update."
        <REF> price </REF>: number "The price to store for the item."
        <REF> quantity </REF>: integer "The quantity to store for the item."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.restock_item(item_name, quantity) and store the returned value as restock_result. RESULT restock_result]
            [COMMAND If restock_result is falsey, meaning the restock attempt failed, set self.inventory[item_name] to a dictionary containing price and quantity. RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds the specified amount to the instance balance and returns the updated balance." insert_coin]
    [INPUTS]
        <REF> amount </REF>: numeric "The value to add to the balance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> balance </REF>: numeric "The updated balance after adding the amount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Load the current value of self.balance. RESULT current_balance]
            [COMMAND Add amount to current_balance and assign the result back to self.balance. RESULT self.balance]
            [COMMAND Return the updated self.balance. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to purchase the named item by verifying it exists, has stock, and is affordable; on success it deducts the price, decrements stock, and returns the new balance, otherwise it returns False." purchase_item]
    [INPUTS]
        <REF> self </REF>: object "The instance with inventory (dict) and balance (number) attributes."
        <REF> item_name </REF>: str "The name/key of the item to purchase in self.inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: number_or_boolean "The updated balance after a successful purchase, or False if the item is not in inventory, has no stock, or the balance is insufficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether item_name is a key in self.inventory. RESULT outer_condition]
            [COMMAND If outer_condition is True, retrieve item = self.inventory[item_name]. RESULT item]
            [COMMAND If outer_condition is True, evaluate inner_condition as item['quantity'] > 0 and self.balance >= item['price']; short-circuit after item['quantity'] > 0 when that is False. RESULT inner_condition]
            [COMMAND If outer_condition and inner_condition are True, subtract item['price'] from self.balance. RESULT self.balance]
            [COMMAND If outer_condition and inner_condition are True, decrement item['quantity'] by 1. RESULT item['quantity']]
            [COMMAND If outer_condition and inner_condition are True, return self.balance. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name is not in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.balance or item['quantity']. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: item_name is in self.inventory and inner_condition is False (the inner condition evaluates item['quantity'] > 0 first, and only if True checks self.balance >= item['price'])]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.balance or item['quantity']. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Restocks an existing inventory item by increasing its quantity and returns True, or returns False without changes if the item is not present." restock_item]
    [INPUTS]
        <REF> item_name </REF>: string "Name of the inventory item to restock."
        <REF> quantity </REF>: numeric "Amount to add to the item's current quantity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the item exists and its quantity was increased; False if the item does not exist."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory. RESULT item_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name is present in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.inventory[item_name]['quantity'] by quantity. RESULT self.inventory]
            [COMMAND Return True to indicate that the item was restocked. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: item_name is not present in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that the item was not restocked. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a newline-separated string of inventory item details in the format 'name - $price [quantity]', or False when the inventory is empty." display_items]
    [INPUTS]
        <REF> self </REF>: object "Object instance containing an inventory dictionary attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result_string </REF>: string or boolean "Newline-separated formatted inventory listing, or False when inventory is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the truthiness of self.inventory. RESULT inventory_truthiness]
            [COMMAND On the else path, with self.inventory truthy, initialize an empty list named items. RESULT items]
            [COMMAND For each item_name and item_info in self.inventory.items(), process one inventory entry. RESULT inventory_entry]
            [COMMAND Within the for loop over inventory entries, create the formatted line f"{item_name} - ${item_info['price']} [{item_info['quantity']}]". RESULT formatted_line]
            [COMMAND Within the for loop over inventory entries, append formatted_line to items. RESULT items]
            [COMMAND After the for loop completes, join items with the newline separator "\n" and return the resulting string. RESULT result_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When self.inventory is empty or falsy, making the if branch true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
