        <REF> result </REF>: list "A list of database rows returned by the query; empty list if no student matches the name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection to the database identified by self.database_name and assign the connection object to conn. RESULT conn]

            [COMMAND Set select_query to the SQL string "SELECT * FROM students WHERE name = ?". RESULT select_query]
            [COMMAND Execute select_query on cursor, binding the name parameter to the ? placeholder. RESULT cursor]
            [COMMAND Fetch all remaining rows from the executed cursor and assign them to result. RESULT result]
            [COMMAND Close the SQLite connection conn. RESULT conn]
            [COMMAND Return the fetched rows stored in result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]
