# CLASS SignInSystem

[DEFINE_WORKER: "Initializes a new instance by setting the users attribute to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty dictionary literal {} to the instance attribute self.users. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a new username in the users dictionary with an initial value of False, returning False if the username already exists, otherwise adding it and returning True." add_user]
    [INPUTS]
        <REF> username </REF>: string "The username to register in the users dictionary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username was newly added; False if it already existed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is already a key in self.users. RESULT username_exists]
            [COMMAND If username_exists is false, set self.users[username] = False and return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username_exists is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that the username was not added because it already exists. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True and marks the existing user as active if username is already present in self.users; otherwise returns False." sign_in]
    [INPUTS]
        <REF> username </REF>: string "The username whose presence in self.users is checked and, if present, marked as active."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if username exists in self.users and was updated; False if username does not exist."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is absent from self.users. RESULT is_absent]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_absent is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT sign_in_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: is_absent is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.users[username] to True. RESULT updated_users]
            [COMMAND Return True. RESULT sign_in_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True only when username exists in self.users and its stored value is truthy; otherwise returns False." check_sign_in]
    [INPUTS]
        <REF> username </REF>: string "Username to look up in the self.users mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if username exists in self.users and self.users[username] is truthy; False if the username is missing or the stored value is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is not in self.users. RESULT missing_user]
            [COMMAND If missing_user is true, follow the alternative flow for a missing username. RESULT missing_user_path]
            [COMMAND Within the outer else branch (username exists in self.users), check whether the value stored at self.users[username] is truthy. RESULT stored_value_truthy]
            [COMMAND If stored_value_truthy is true, return True. RESULT return_true_existing]
            [COMMAND If stored_value_truthy is false, follow the alternative flow for a falsy stored value. RESULT falsy_stored_value_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_false_missing]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: username is in self.users and self.users[username] is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_false_existing_falsy]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True when every value in self.users is truthy or when self.users is empty, and False when any value is falsy." all_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "True if all values in self.users are truthy or self.users is empty; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate all(self.users.values()) by checking each value in self.users for truthiness, stopping at the first falsy value due to short-circuiting; treat an empty self.users as making the condition True. RESULT condition_result]
            [COMMAND If condition_result is True, set the return value to True and exit the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: condition_result is False because at least one value in self.users is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the return value to False and exit the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a list of usernames whose signed_in status is falsy in the self.users dictionary." all_not_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> not_signed_in_users </REF>: list "List of usernames whose signed_in value is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize not_signed_in_users as an empty list. RESULT not_signed_in_users]
            [COMMAND For each username and signed_in pair in self.users.items(), iterate over the dictionary entries. RESULT username, signed_in]
            [COMMAND Within the for loop over self.users.items(), if signed_in is falsy, append username to not_signed_in_users. RESULT not_signed_in_users]
            [COMMAND Return not_signed_in_users. RESULT not_signed_in_users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
