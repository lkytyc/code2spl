[DEFINE_WORKER: "Reads an Excel file using openpyxl and returns the active worksheet rows as a list of tuples, or None if any exception occurs." read_excel]
    [INPUTS]
        <REF> file_name </REF>: string "Path to the Excel file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> data </REF>: list of tuples or None "Rows from the active worksheet, each row as a tuple of cell values; None if an exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize data as an empty list. RESULT data]
            [COMMAND Open the Excel workbook located at file_name using openpyxl.load_workbook and assign it to workbook. RESULT workbook]
            [COMMAND Get the active worksheet from workbook and assign it to sheet. RESULT sheet]
            [COMMAND For each row in sheet.iter_rows(values_only=True), append the row tuple to data. RESULT data]
            [COMMAND Close the workbook. RESULT workbook]
            [COMMAND Return data. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during workbook loading, sheet retrieval, row iteration, appending, or workbook closing]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the fallback value. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the given rows of data into an Excel workbook using openpyxl, saves the workbook to the specified file name, and returns 1 on success or 0 on failure." write_excel]
    [INPUTS]
        <REF> data </REF>: iterable of row sequences "Rows of cell values to append to the active worksheet."
        <REF> file_name </REF>: string "Path of the Excel file to save."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "1 if the workbook was created, populated, saved, and closed successfully; 0 if an exception occurred."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the try block, create a new Excel workbook using openpyxl.Workbook() and assign it to workbook. RESULT workbook]
            [COMMAND Within the same try block, get the active worksheet from workbook and assign it to sheet. RESULT sheet]
            [COMMAND Within the same try block, for each row in data, append row to sheet using sheet.append(row). RESULT sheet]
            [COMMAND Within the same try block, after the for loop, save workbook to file_name using workbook.save(file_name). RESULT None]
            [COMMAND Within the same try block, close workbook using workbook.close(). RESULT None]
            [COMMAND Within the same try block, return integer 1 to indicate success. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs inside the try block while creating the workbook, getting the active sheet, appending rows, saving the workbook, or closing the workbook]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return integer 0 to indicate failure. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads an Excel file, transforms each row by appending an uppercase string form of column N when it is not digit-only and the original value otherwise, writes the result to a new '_process.xlsx' file, and returns the write status and new file name, or 0 if the input data is invalid." process_excel_data]
    [INPUTS]
        <REF> N </REF>: int "Index of the column to inspect and transform in each row."
        <REF> save_file_name </REF>: str "Path of the source Excel file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple or int "Returns (success, new_file_name) after writing the processed Excel file, or 0 if data is None or N is out of range."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the Excel file specified by save_file_name and assign the result to data. RESULT data]
            [COMMAND Initialize new_data as an empty list. RESULT new_data]
            [COMMAND For each row in data, create new_row as a shallow copy of the row. RESULT new_row]
            [COMMAND Within the loop over rows, if str(row[N]) is not digit-only, append str(row[N]).upper() to new_row. RESULT new_row]
            [COMMAND Within the loop over rows, if str(row[N]) is digit-only, append the original row[N] to new_row. RESULT new_row]
            [COMMAND Within the loop over rows, append new_row to new_data. RESULT new_data]
            [COMMAND Derive new_file_name by taking everything before the first '.' in save_file_name and appending '_process.xlsx'. RESULT new_file_name]
            [COMMAND Write new_data to new_file_name using self.write_excel, storing the return value as success. RESULT success]
            [COMMAND Return the tuple (success, new_file_name). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: data is None, or N is greater than or equal to the number of columns in the first row of data.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
