# CLASS GomokuGame

[DEFINE_WORKER: "Initializes a square game board with empty spaces and sets the current player to 'X'." __init__]
    [INPUTS]
        <REF> board_size </REF>: int "Number of rows and columns for the square game board."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.board_size to the board_size parameter, preserving the input integer dimension. RESULT self.board_size]
            [COMMAND Build self.board as a nested list comprehension with board_size rows and board_size columns, each cell initialized to a single space ' '; if board_size is 0 or negative, the resulting board is an empty list. RESULT self.board]
            [COMMAND Set self.current_player to the constant string 'X' as the starting player. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Places the current player's mark in the specified board cell if it is empty, toggles the current player after a successful move, and returns True; otherwise returns False." make_move]
    [INPUTS]
        <REF> row </REF>: int "Zero-based row index of the board cell to play."
        <REF> col </REF>: int "Zero-based column index of the board cell to play."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the move was successfully made; False if the cell is not empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.board[row][col] is exactly the single space character ' '. RESULT is_empty]
            [COMMAND If is_empty is true, assign self.current_player to self.board[row][col]. RESULT self.board[row][col]]
            [COMMAND Toggle self.current_player to 'O' if it is currently 'X', otherwise toggle it to 'X'. RESULT self.current_player]
            [COMMAND Return True to indicate the move was successful. RESULT true_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If self.board[row][col] is not exactly the single space character ' ']
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate the move was not made. RESULT false_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the board in row-major order and returns the winner's symbol when it finds five matching pieces along one of the four checked directions, otherwise returns None." check_winner]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string or None "The winning player's symbol if a five-in-a-row is found; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize directions to the list [(0,1), (1,0), (1,1), (1,-1)] representing right, down, down-right, and down-left direction vectors. RESULT directions]
            [COMMAND For each row from 0 to self.board_size - 1 inclusive, enter the outer loop over board rows. RESULT row]
            [COMMAND Within the outer for loop over row, for each col from 0 to self.board_size - 1 inclusive, enter the inner loop over board columns. RESULT col]
            [COMMAND Within the row and col loops, check whether self.board[row][col] is not ' '; if the cell is empty, skip the direction checks for this cell, otherwise continue with the current non-empty cell. RESULT None]
            [COMMAND Within the non-empty cell guard, for each direction in the directions list, process the current direction vector. RESULT direction]
            [COMMAND Within the direction loop, call self._check_five_in_a_row(row, col, direction); if it returns true, immediately return self.board[row][col] as the winning player's symbol. RESULT winner]
            [COMMAND After all row, column, and direction iterations complete without finding a winner, return None. RESULT none_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the starting cell and the next four cells in the given direction all contain the same board symbol, returning False immediately if any checked cell is out of bounds or differs and returning True if all five match." _check_five_in_a_row]
    [INPUTS]
        <REF> row </REF>: int "Zero-based row index of the starting cell."
        <REF> col </REF>: int "Zero-based column index of the starting cell."
        <REF> direction </REF>: tuple[int, int] "Pair (dx, dy) giving the row and column offsets to move in one step from the starting cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the starting cell and the four consecutive cells in the given direction contain the same symbol; False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack the direction pair into dx and dy. RESULT dx, dy]
            [COMMAND Set count to 1, counting the starting cell at (row, col). RESULT count]
            [COMMAND Set symbol to the board value stored at self.board[row][col]. RESULT symbol]
            [COMMAND Start a for loop with i taking values 1 through 4 inclusive, equivalent to range(1, 5). RESULT i]
            [COMMAND Within the for loop over i, compute new_row = row + dx * i. RESULT new_row]
            [COMMAND Within the for loop over i, compute new_col = col + dy * i. RESULT new_col]
            [COMMAND Within the for loop over i, confirm that new_row is between 0 inclusive and self.board_size exclusive and that new_col is between 0 inclusive and self.board_size exclusive. RESULT bounds_ok]
            [COMMAND Within the for loop over i, confirm that self.board[new_row][new_col] equals symbol. RESULT symbol_matches]
            [COMMAND Within the for loop over i, after both bounds_ok and symbol_matches are true, increment count by 1. RESULT count]
            [COMMAND After the for loop completes, return the result of count == 5. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the for loop over i, the bounds check fails because new_row is less than 0, new_row is at least self.board_size, new_col is less than 0, or new_col is at least self.board_size.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the for loop over i, the bounds check succeeds but self.board[new_row][new_col] is not equal to symbol.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
