# CLASS BalancedBrackets

[DEFINE_WORKER: "Initializes the bracket-matching state with an empty stack, the three opening/closing bracket lists, and the expression to validate." __init__]
    [INPUTS]
        <REF> expr </REF>: str "The expression string to be validated for balanced brackets."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.stack to an empty list. RESULT self.stack]
            [COMMAND Set self.left_brackets to the list of opening brackets: '(', '{', '['. RESULT self.left_brackets]
            [COMMAND Set self.right_brackets to the list of closing brackets: ')', '}', ']', aligned by index with self.left_brackets. RESULT self.right_brackets]
            [COMMAND Assign the expr parameter to self.expr. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters self.expr to retain only characters contained in self.left_brackets or self.right_brackets, then assigns the filtered string back to self.expr." clear_expr]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.expr as the input string. RESULT old_expr]
            [COMMAND For each character c in old_expr, test whether c is in self.left_brackets or c is in self.right_brackets. RESULT kept_chars]
            [COMMAND Concatenate all kept characters in their original order using an empty string as the separator. RESULT filtered_expr]
            [COMMAND Assign filtered_expr to self.expr, replacing the previous value. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the brackets in self.expr are balanced and correctly nested by pushing opening brackets onto a stack and validating every closing bracket against the top of the stack, returning False on the first mismatch or if any brackets remain after the loop." check_balanced_brackets]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True if all brackets in self.expr are balanced and properly nested, otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Clear the current expression state by calling self.clear_expr(). RESULT self]
            [COMMAND For each character Brkt in self.expr, begin iterating through the expression. RESULT Brkt]
            [COMMAND Within the for loop over Brkt, test whether Brkt is in self.left_brackets. RESULT is_left_bracket]
            [COMMAND Within the for loop over Brkt, if is_left_bracket is true, push Brkt onto self.stack. RESULT self.stack]
            [COMMAND Within the for loop over Brkt, if is_left_bracket is false, pop the top value from self.stack into Current_Brkt. RESULT Current_Brkt]
            [COMMAND Within the non-left-bracket branch of the for loop, if Current_Brkt is '(' and Brkt is not ')', return False immediately. RESULT False_return]
            [COMMAND Within the non-left-bracket branch of the for loop, if Current_Brkt is '{' and Brkt is not '}', return False immediately. RESULT False_return]
            [COMMAND Within the non-left-bracket branch of the for loop, if Current_Brkt is '[' and Brkt is not ']', return False immediately. RESULT False_return]
            [COMMAND After the for loop completes, if self.stack is non-empty, return False. RESULT False_return]
            [COMMAND After the for loop completes, if self.stack is empty, return True. RESULT True_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
