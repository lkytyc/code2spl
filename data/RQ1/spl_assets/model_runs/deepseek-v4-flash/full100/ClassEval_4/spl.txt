[DEFINE_WORKER: "Updates the score for a specified course in an existing student's record, and does nothing if the student does not exist." add_course_score]
    [INPUTS]
        <REF> name </REF>: string "The key identifying the student in self.students."
        <REF> course </REF>: string "The key identifying the course whose score is to be set."
        <REF> score </REF>: number "The numeric score to assign to the specified course for the student."
    [END_INPUTS]

    [OUTPUTS]
        <REF> None </REF>: null "No return value; self.students is mutated in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether name is exactly an existing key in self.students. RESULT name_in_students]
            [COMMAND If name_in_students is true, set self.students[name]['courses'][course] to score. RESULT course_score_updated]
            [COMMAND Continue with the next statement after the conditional block. RESULT no_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: name is not exactly an existing key in self.students]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the course score assignment and continue with the next statement after the conditional block. RESULT no_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the average grade from a student's course dictionary if the student exists and has at least one course; otherwise returns None." get_gpa]
    [INPUTS]
        <REF> name </REF>: str "The key identifying the student in the students dictionary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float or None "The average of the student's course grade values if the student exists and has courses; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name is a key in self.students and, if so, whether self.students[name]['courses'] is truthy, using short-circuit evaluation so the second check is skipped when name is absent. RESULT has_course_records]
            [COMMAND If has_course_records is true, compute the average of the values in the student's courses dictionary by summing them and dividing by the number of courses, then return that average. RESULT gpa_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: has_course_records is false, meaning the student does not exist or the student's courses dictionary is empty/falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None. RESULT gpa_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects the name of every student in self.students who has at least one course score strictly below 60 and returns that list, adding each qualifying student at most once." get_all_students_with_fail_course]
    [INPUTS]
        <REF> self </REF>: object "Instance containing self.students, a dictionary mapping student names to student records; each student record has a 'courses' dictionary mapping course names to numeric scores."
    [END_INPUTS]

    [OUTPUTS]
        <REF> students </REF>: list "A list of student names with at least one score below 60, with no duplicate names."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named students to collect student names. RESULT students]
            [COMMAND For each name and student record in self.students.items(), enter the outer loop over all students. RESULT name, student]
            [COMMAND Within the outer loop, iterate over course and score pairs in student['courses'].items(), entering the inner loop over all courses for the current student. RESULT course, score]
            [COMMAND Within the inner loop, check whether score is strictly less than 60; a score of 60 is not considered failing. RESULT is_failing]
            [COMMAND If is_failing is true, append name to students and break out of the inner course loop only, then continue with the next student in the outer loop. RESULT students]
            [COMMAND After both loops complete, return the students list. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the average score for a specified course across all students, including only students whose course score is not None, and returns None when no such scores exist." get_course_average]
    [INPUTS]
        <REF> self </REF>: object "Object that owns the students dictionary."
        <REF> course </REF>: object "Course key used to look up scores in each student's courses dictionary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float or None "Average score for the course, or None if no student had a non-None score for the course."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND Initialize count to 0. RESULT count]
            [COMMAND Iterate over each student in self.students.values(). RESULT student]
            [COMMAND For the current student, test whether course is a key in student['courses']. RESULT contains_course]
            [COMMAND For the current student where contains_course is true, retrieve score = student['courses'][course]. RESULT score]
            [COMMAND For the current student where contains_course is true, test whether score is not None. RESULT score_is_valid]
            [COMMAND For the current student where contains_course is true and score_is_valid is true, add score to total. RESULT total]
            [COMMAND For the current student where contains_course is true and score_is_valid is true, increment count by 1. RESULT count]
            [COMMAND After processing all students, if count > 0 return total / count; otherwise return None. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Finds the first student in self.students with a GPA strictly greater than 0, ignoring None GPAs, and returns that student's name, or None if no such student exists." get_top_student]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> top_student </REF>: string or None "Name of the first student encountered with the highest GPA strictly greater than 0, or None if self.students is empty or no GPA is greater than 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize top_student to None. RESULT top_student]
            [COMMAND Initialize top_gpa to 0. RESULT top_gpa]
            [COMMAND Iterate over each key-value pair in self.students.items(), unpacking name and student; student is unused. RESULT name]
            [COMMAND Within the for loop over self.students.items(), compute gpa by calling self.get_gpa(name). RESULT gpa]
            [COMMAND Within the for loop, evaluate the short-circuited condition gpa is not None and gpa > top_gpa; if true, set top_gpa to gpa and top_student to name; if false, leave top_gpa and top_student unchanged. RESULT top_gpa, top_student]
            [COMMAND After the loop completes, return top_student. RESULT top_student]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
