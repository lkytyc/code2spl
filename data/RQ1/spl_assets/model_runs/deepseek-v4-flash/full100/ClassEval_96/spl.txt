[DEFINE_WORKER: "Queries weather data for a city, stores it in self, returns False if the city is not found, and returns the temperature in the requested unit, converting it from the stored unit when necessary." query]
    [INPUTS]
        <REF> weather_list </REF>: dict "Weather data dictionary keyed by city; each city entry contains 'temperature', 'weather', and 'temperature units'."
        <REF> tmp_units </REF>: str "Target temperature unit; defaults to 'celsius'. Only 'celsius' and 'fahrenheit' are handled explicitly."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool | tuple "False if self.city is missing; otherwise a tuple of temperature and weather description, with temperature converted to tmp_units when the stored unit differs and a supported target unit is requested."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign weather_list to self.weather_list. RESULT self.weather_list]
            [COMMAND If self.city is not present in weather_list, return False. RESULT False]
            [COMMAND Set self.temperature to self.weather_list[self.city]['temperature'] and self.weather to self.weather_list[self.city]['weather']. RESULT self.temperature, self.weather]
            [COMMAND If self.weather_list[self.city]['temperature units'] equals tmp_units, return self.temperature and self.weather unchanged. RESULT self.temperature, self.weather]
            [COMMAND If the stored temperature unit differs from tmp_units and tmp_units is 'celsius', return self.fahrenheit_to_celsius() and self.weather. RESULT self.fahrenheit_to_celsius(), self.weather]
            [COMMAND If the stored temperature unit differs from tmp_units and tmp_units is 'fahrenheit', return self.celsius_to_fahrenheit() and self.weather. RESULT self.celsius_to_fahrenheit(), self.weather]
            [COMMAND If the stored temperature unit differs from tmp_units and tmp_units is neither 'celsius' nor 'fahrenheit', fall through without an explicit return. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the Celsius temperature stored in self.temperature to Fahrenheit by multiplying it by 9/5 and adding 32." celsius_to_fahrenheit]
    [INPUTS]
        <REF> self.temperature </REF>: numeric "The Celsius temperature value read from the instance attribute and converted to Fahrenheit."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float "The computed Fahrenheit temperature, equal to (self.temperature * 9/5) + 32."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.temperature from the instance. RESULT self.temperature]
            [COMMAND Multiply self.temperature by 9. RESULT temp_times_9]
            [COMMAND Divide temp_times_9 by 5. RESULT temp_after_division]
            [COMMAND Add 32 to temp_after_division to obtain the Fahrenheit result. RESULT fahrenheit_result]
            [COMMAND Return fahrenheit_result from the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

    [END_INPUTS]
