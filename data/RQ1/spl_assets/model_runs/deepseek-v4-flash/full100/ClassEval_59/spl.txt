    [END_INPUTS]

[DEFINE_WORKER: "Creates a movie record from name, price, parsed HH:MM start/end times, and an n by n zero-filled seat matrix, then appends it to the movie list." add_movie]
    [INPUTS]
        <REF> name </REF>: string "Name of the movie."
        <REF> price </REF>: number "Price of the movie, passed through unchanged."
        <REF> start_time </REF>: string "Start time string in 24-hour HH:MM format."
        <REF> end_time </REF>: string "End time string in 24-hour HH:MM format."
        <REF> n </REF>: integer "Dimension of the square seat matrix; must be non-negative, and 0 produces an empty 2D array."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a movie dictionary by setting name to the input name, price to the input price, start_time to datetime.strptime(start_time, '%H:%M') (a datetime with default date 1900-01-01), end_time to datetime.strptime(end_time, '%H:%M') (a datetime with default date 1900-01-01), and seats to np.zeros((n, n)) (a 2D NumPy array of shape (n, n) filled with zeros). RESULT movie]
            [COMMAND Append the movie dictionary to self.movies. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: start_time or end_time is not a valid time string in the 24-hour HH:MM format.]
        [LOG "datetime.strptime raised ValueError because the time data does not match the required format."]
        [THROW ValueError "Raised when a time string cannot be parsed using the HH:MM format."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: n is negative.]
        [LOG "np.zeros raised ValueError because negative dimensions are not allowed."]
        [THROW ValueError "Raised when the requested seat matrix dimension n is negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: n is not an integer.]
        [LOG "np.zeros raised TypeError because array dimensions must be integers."]
        [THROW TypeError "Raised when the requested seat matrix dimension n is not an integer."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Books all requested seats for the first movie whose name matches, returning 'Booking success.' if every seat is free, 'Booking failed.' if any seat is already booked, and 'Movie not found.' if no movie matches." book_ticket]
    [INPUTS]
        <REF> name </REF>: string "Name of the movie to book tickets for."
        <REF> seats_to_book </REF>: list of [int, int] pairs "Seats to book; each seat is a two-element sequence where seat[0] is the row index and seat[1] is the column index."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Returns 'Booking success.' if all requested seats were available, 'Booking failed.' if any requested seat was already booked, or 'Movie not found.' if no movie has the given name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Loop over each movie in self.movies. RESULT movie]
            [COMMAND For the current movie, check whether movie['name'] == name. RESULT matched_movie]
            [COMMAND For the matched movie, loop over each seat in seats_to_book. RESULT seat]
            [COMMAND For each seat, check whether movie['seats'][seat[0]][seat[1]] == 0. RESULT seat_available]
            [COMMAND If the seat is available, set movie['seats'][seat[0]][seat[1]] = 1 to book it. RESULT updated_seat]
            [COMMAND After all seats in seats_to_book have been processed for the matched movie, return 'Booking success.'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the matched movie's seat loop, a seat is already booked because movie['seats'][seat[0]][seat[1]] != 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 'Booking failed.' immediately; any seats already set to 1 earlier in this request remain booked, and remaining seats are not processed. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The outer loop over self.movies completes without any movie matching movie['name'] == name.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 'Movie not found.'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses start and end time strings and returns the names of movies in self.movies whose scheduled interval lies entirely within that inclusive time window." available_movies]
    [INPUTS]
        <REF> start_time </REF>: string "Start of the query window in 24-hour 'HH:MM' format."
        <REF> end_time </REF>: string "End of the query window in 24-hour 'HH:MM' format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> available_movies </REF>: list of strings "Names of movies whose start and end times fall within the requested inclusive interval."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the start_time string into a datetime object using the 24-hour format '%H:%M' and bind the result to start_time. RESULT start_time]
            [COMMAND Parse the end_time string into a datetime object using the 24-hour format '%H:%M' and bind the result to end_time. RESULT end_time]
            [COMMAND Initialize available_movies as an empty list. RESULT available_movies]
            [COMMAND For each movie in self.movies, in iteration order, evaluate the combined short-circuit condition start_time <= movie['start_time'] and movie['end_time'] <= end_time; if the condition is true, append movie['name'] to available_movies. The boundary comparisons are inclusive, so movies that start exactly at start_time or end exactly at end_time are included. RESULT available_movies]
            [COMMAND Return available_movies. RESULT available_movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: start_time or end_time is not a valid time string in the '%H:%M' 24-hour format, causing datetime.strptime to raise ValueError.]
        [LOG "Invalid time format passed to datetime.strptime; ValueError is propagated to the caller."]
        [THROW ValueError "Raised when a time string does not match the required HH:MM format or contains out-of-range values."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
