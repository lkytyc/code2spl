    [END_OUTPUTS]

[DEFINE_WORKER: "Searches the tickets table for all rows whose customer_name matches the given customer_name and returns the fetched rows." search_tickets_by_customer]
    [INPUTS]
        <REF> customer_name </REF>: string "The customer name used to filter tickets in the database query."
    [END_INPUTS]

    [OUTPUTS]
        <REF> tickets </REF>: list "List of all ticket rows matching customer_name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the parameterized SQL query 'SELECT * FROM tickets WHERE customer_name = ?' on self.cursor, passing customer_name as the parameter. RESULT cursor_execute_result]
            [COMMAND Call fetchall() on self.cursor to retrieve all remaining rows from the executed query and assign the resulting list to tickets. RESULT tickets]
            [COMMAND Return tickets. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
