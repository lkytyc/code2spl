# CLASS BookManagement

[DEFINE_WORKER: "Initializes the inventory instance attribute to a new empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance attribute self.inventory to a new empty dictionary with zero entries. RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a book to the inventory by increasing its existing quantity if the title is already present, or by setting the quantity to the given value if the title is new." add_book]
    [INPUTS]
        <REF> title </REF>: str "The book title used as the key in self.inventory."
        <REF> quantity </REF>: int "The quantity to add for the book; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether title is a key in self.inventory. RESULT title_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title_exists is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the if branch for an existing title, add quantity to the current value stored at self.inventory[title] and store the resulting sum back into self.inventory[title]. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: title_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the else branch for a new title, set self.inventory[title] to quantity. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a specified quantity of a book title from the inventory, guarding against missing or insufficient stock with a raise, and deleting the title when its remaining stock reaches zero." remove_book]
    [INPUTS]
        <REF> title </REF>: string "The book title key to look up in self.inventory."
        <REF> quantity </REF>: integer "The number of copies to subtract from the available stock."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: NoneType "No return value; self.inventory is mutated in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within remove_book, evaluate the guard condition: title not in self.inventory or self.inventory[title] < quantity. Use short-circuit evaluation so the stock lookup occurs only if title exists. RESULT guard_condition]
            [COMMAND Within remove_book, when guard_condition is false, subtract quantity from self.inventory[title] and store the updated stock. RESULT updated_stock]
            [COMMAND Within remove_book, after the subtraction, check whether updated_stock equals 0. RESULT zero_stock]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: zero_stock is true, meaning the remaining stock for title is exactly 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within remove_book, when zero_stock is true, delete the key title from self.inventory. RESULT inventory_after_deletion]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Within remove_book, guard_condition is true: title is not in self.inventory, or self.inventory[title] is strictly less than quantity. A quantity equal to the available stock does not trigger this exception.]
        [LOG "Attempted to remove quantity copies of title from inventory, but title is missing or the stock is insufficient."]
        [THROW False "Raises the value False. In Python, this causes a TypeError because False is not an exception class or instance."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the inventory attribute of the object unchanged to the caller." view_inventory]
    [INPUTS]
        <REF> self </REF>: object "The object instance whose inventory attribute is read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "The value stored in the inventory attribute; if it is a mutable object, the same reference is returned."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Load the value stored in the inventory attribute of the object referenced by self. RESULT inventory_value]
            [COMMAND Return the loaded inventory_value unchanged to the caller. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the inventory quantity for the given book title, or 0 if the title is not present in the inventory." view_book_quantity]
    [INPUTS]
        <REF> title </REF>: str "Title of the book whose inventory quantity is being requested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> quantity </REF>: int "Inventory quantity for the title, or 0 if the title is not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether title is present in self.inventory; if it is absent, follow the alternative_flows early-return path and do not continue to the lookup. RESULT title_present]
            [COMMAND When title is present in self.inventory, retrieve the value stored at self.inventory[title] and return it as the function result. RESULT quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title is not present in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the integer 0 immediately and exit the method. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
