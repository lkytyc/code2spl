# CLASS KappaCalculator

[DEFINE_WORKER: "Computes Cohen's kappa coefficient from a square confusion/agreement matrix by summing diagonal observed agreement, deriving expected agreement from row and column totals, and normalizing by the chance-corrected maximum." kappa]
    [INPUTS]
        <REF> testData </REF>: array-like "Input data to be converted to a NumPy matrix; must be array-like and support matrix indexing along the diagonal."
        <REF> k </REF>: int "Number of classes or categories; the loop range end, exclusive, for summing diagonal entries."
    [END_INPUTS]

    [OUTPUTS]
        <REF> cohens_coefficient </REF>: float "Computed Cohen's kappa coefficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert testData to a NumPy matrix and assign it to dataMat. RESULT dataMat]
            [COMMAND Initialize P0 to 0.0. RESULT P0]
            [COMMAND For each i from 0 to k-1 inclusive, accumulate dataMat[i, i] * 1.0 into P0. RESULT P0]
            [COMMAND Compute xsum as the row sums of dataMat using np.sum(dataMat, axis=1). RESULT xsum]
            [COMMAND Compute ysum as the column sums of dataMat using np.sum(dataMat, axis=0). RESULT ysum]
            [COMMAND Compute sum as the total scalar sum of all entries in dataMat using np.sum(dataMat). RESULT sum]
            [COMMAND Compute Pe as float(ysum * xsum) / sum / sum. RESULT Pe]
            [COMMAND Reassign P0 as float(P0 / sum * 1.0). RESULT P0]
            [COMMAND Compute cohens_coefficient as float((P0 - Pe) / (1 - Pe)). RESULT cohens_coefficient]
            [COMMAND Return cohens_coefficient. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes Fleiss' kappa inter-rater agreement coefficient from a matrix of rating counts by comparing observed agreement with expected agreement." fleiss_kappa]
    [INPUTS]
        <REF> testData </REF>: array-like "N x k matrix of category counts, converted to a float matrix with np.mat."
        <REF> N </REF>: int "Number of subjects, corresponding to the number of rows in testData."
        <REF> k </REF>: int "Number of categories, corresponding to the number of columns in testData."
        <REF> n </REF>: int "Number of raters per subject, used in normalizing row-level agreement."
    [END_INPUTS]

    [OUTPUTS]
        <REF> ans </REF>: float "The resulting Fleiss' kappa coefficient, returned as ans[0, 0]."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert testData to a floating-point matrix named dataMat using np.mat. RESULT dataMat]
            [COMMAND Create a k x 1 column vector of ones named oneMat using np.ones. RESULT oneMat]
            [COMMAND Initialize sum to 0.0. RESULT sum]
            [COMMAND Initialize P0 to 0.0. RESULT P0]
            [COMMAND For each i from 0 to N-1, reset temp to 0.0 at the start of the outer loop. RESULT temp]
            [COMMAND Within the outer loop over i, for each j from 0 to k-1, add dataMat[i, j] to sum and add 1.0 * dataMat[i, j] ** 2 to temp. RESULT temp]
            [COMMAND After the inner loop over j, for each i from 0 to N-1, set temp = (temp - n) / ((n - 1) * n) and add temp to P0. RESULT P0]
            [COMMAND After the outer loop over i, scale P0 by 1.0 and divide by N, assigning the result to P0. RESULT P0]
            [COMMAND Compute ysum as the column-wise sum of dataMat along axis 0 using np.sum. RESULT ysum]
            [COMMAND For each i from 0 to k-1, set ysum[0, i] to (ysum[0, i] / sum) ** 2. RESULT ysum]
            [COMMAND Compute Pe by multiplying ysum by oneMat and then by 1.0. RESULT Pe]
            [COMMAND Compute ans as (P0 - Pe) / (1 - Pe). RESULT ans]
            [COMMAND Return ans[0, 0]. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
