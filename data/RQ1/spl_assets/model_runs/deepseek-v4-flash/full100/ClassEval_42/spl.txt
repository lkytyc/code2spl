# CLASS Hotel

[DEFINE_WORKER: "Initializes a new object with the provided name, room availability mapping, and an empty booked-rooms dictionary." __init__]
    [INPUTS]
        <REF> name </REF>: string "The name identifier to assign to this object."
        <REF> rooms </REF>: dictionary "A mapping of room types to available room counts, such as {'single': 5, 'double': 3}."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.name to the provided name parameter. RESULT self.name]
            [COMMAND Set self.available_rooms to the provided rooms parameter, preserving the room type to available count mapping. RESULT self.available_rooms]
            [COMMAND Initialize self.booked_rooms as an empty dictionary. RESULT self.booked_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to book a requested number of rooms of a given type for a named guest, returning a success message, the remaining available room count, or False depending on room type existence and availability." book_room]
    [INPUTS]
        <REF> room_type </REF>: string "The type of room being booked; must be a key in self.available_rooms."
        <REF> room_number </REF>: integer "The number of rooms requested for the booking."
        <REF> name </REF>: string "The guest name under which the booking is recorded."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string/integer/boolean "Returns 'Success!' when the booking is completed, the remaining integer count of available rooms when the requested number is not available but rooms remain, or False when the room type is unknown or no rooms are available."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If room_type is not a key in self.available_rooms, return False. RESULT room_type_availability_check]
            [COMMAND After the room_type existence check passes, evaluate whether room_number is less than or equal to self.available_rooms[room_type]. RESULT enough_rooms]
            [COMMAND If enough_rooms is true, within the booking branch, if room_type is not already a key in self.booked_rooms, initialize self.booked_rooms[room_type] as an empty dictionary. RESULT booking_dict_ready]
            [COMMAND Within the enough_rooms branch, set self.booked_rooms[room_type][name] to room_number. RESULT booking_recorded]
            [COMMAND Within the enough_rooms branch, subtract room_number from self.available_rooms[room_type]. RESULT available_rooms_updated]
            [COMMAND Within the enough_rooms branch, return the string 'Success!'. RESULT success_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: room_number is greater than self.available_rooms[room_type] and self.available_rooms[room_type] is not zero]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.available_rooms[room_type]. RESULT partial_availability_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: room_number is greater than self.available_rooms[room_type] and self.available_rooms[room_type] is zero]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT zero_availability_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks in a guest for a room type by returning False if the room type is not booked or the requested room number exceeds the existing booking, and otherwise removing the booking when exactly matched or reducing it by the requested number." check_in]
    [INPUTS]
        <REF> room_type </REF>: str "Room category key used to look up the guest's booking in self.booked_rooms."
        <REF> room_number </REF>: int "Number of rooms being checked in or released from the existing booking."
        <REF> name </REF>: str "Guest name whose booking is being updated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean or null "Returns False if room_type is not in self.booked_rooms or if room_number exceeds the stored booking amount; otherwise returns None after removing or reducing the booking."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether room_type is a key in self.booked_rooms.keys(). RESULT room_type_exists]
            [COMMAND If room_type is not in self.booked_rooms.keys(), return False immediately. RESULT false_for_missing_room_type]
            [COMMAND Given that room_type exists, check whether name is a key in self.booked_rooms[room_type]. RESULT name_exists]
            [COMMAND If name is not in self.booked_rooms[room_type], do not modify any booking and fall through to return None. RESULT none_for_missing_name]
            [COMMAND Given that room_type and name both exist, compare room_number with self.booked_rooms[room_type][name]. RESULT comparison_result]
            [COMMAND Given that room_type and name both exist, if room_number is greater than self.booked_rooms[room_type][name], return False immediately. RESULT false_for_excess_room_number]
            [COMMAND Given that room_type and name both exist, if room_number equals self.booked_rooms[room_type][name], remove name from self.booked_rooms[room_type] using pop(name) and fall through to return None. RESULT booking_removed]
            [COMMAND Given that room_type and name both exist, if room_number is less than self.booked_rooms[room_type][name], subtract room_number from self.booked_rooms[room_type][name] and fall through to return None. RESULT booking_reduced]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the available room count for a room type by incrementing the existing count if the room type is already present, otherwise initializing the count to the supplied room number." check_out]
    [INPUTS]
        <REF> room_type </REF>: str "The dictionary key representing the room type to update in self.available_rooms."
        <REF> room_number </REF>: int "The number of rooms to add to the existing count or to use as the initial count."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether room_type is a key in self.available_rooms. RESULT room_type_is_present]
            [COMMAND If room_type_is_present is true, increment self.available_rooms[room_type] by room_number. RESULT self.available_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: room_type is not a key in self.available_rooms (the check in main_flow is false).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.available_rooms[room_type] to room_number. RESULT self.available_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the value stored for a specified room type from the instance's available_rooms mapping." get_available_rooms]
    [INPUTS]
        <REF> room_type </REF>: hashable key (e.g., string) "The key that identifies the room type for which available rooms are requested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "The value associated with room_type in self.available_rooms, typically a collection of available rooms."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate room_type to determine the lookup key. RESULT lookup_key]
            [COMMAND Retrieve the value at the lookup_key position in self.available_rooms. RESULT lookup_value]
            [COMMAND Return lookup_value to the caller as the method's result. RESULT returned_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: room_type is not a key present in self.available_rooms]
        [LOG "KeyError raised because room_type is not present in self.available_rooms."]
        [THROW KeyError "The mapping does not contain the given room_type key."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
