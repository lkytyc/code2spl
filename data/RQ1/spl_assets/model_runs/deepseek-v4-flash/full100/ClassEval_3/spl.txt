[DEFINE_WORKER: "Computes the factorial of n when m is omitted or equals n, otherwise computes the number of arrangements n! / (n - m)!." count]
    [INPUTS]
        <REF> n </REF>: int "Total number of items to arrange."
        <REF> m </REF>: int or None "Number of items selected; defaults to None."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The arrangement count: factorial(n) when m is None or n == m; otherwise factorial(n) // factorial(n - m)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the condition 'm is None or n == m' using short-circuit evaluation. RESULT condition_met]
            [COMMAND Execute the return branch selected by condition_met and obtain its return value. RESULT return_value]
            [COMMAND Return return_value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m is None or n equals m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute ArrangementCalculator.factorial(n) and return it. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: m is not None and n does not equal m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute ArrangementCalculator.factorial(n), compute ArrangementCalculator.factorial(n - m), integer-divide the first factorial by the second, and return the result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total of ArrangementCalculator.count(n, i) for every i from 1 to n inclusive." count_all]
    [INPUTS]
        <REF> n </REF>: int "The number of items; defines the inclusive upper bound for the loop."
    [END_INPUTS]

    [OUTPUTS]
        <REF> total </REF>: int "The accumulated sum of ArrangementCalculator.count(n, i) for i = 1 through n."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND For each integer i from 1 to n inclusive, call ArrangementCalculator.count(n, i) and add the returned value to total. RESULT total]
            [COMMAND Return the accumulated total. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a list of all m-length permutations of self.datas, defaulting m to the length of self.datas when m is not provided." select]
    [INPUTS]
        <REF> self </REF>: object "Instance whose datas attribute provides the sequence of items to permute."
        <REF> m </REF>: integer or null "Optional length of each permutation; if null, the length is set to len(self.datas)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of lists "All permutations of length m from self.datas, with each permutation stored as a list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND For each permutation in itertools.permutations(self.datas, m), convert the permutation to a list and append it to result. RESULT result]
            [COMMAND Return the result list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If m is None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set m to the length of self.datas. RESULT m]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects and concatenates the results of calling self.select(i) for every integer i from 1 to the number of datas entries, inclusive." select_all]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "Combined list of all items returned by self.select(i) for i from 1 to len(self.datas) inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND For i from 1 to len(self.datas) inclusive, iterate over each index in range(1, len(self.datas) + 1). RESULT i]
            [COMMAND Within the for loop over i from 1 to len(self.datas) inclusive, call self.select(i) and extend result with the returned list. RESULT result]
            [COMMAND After the loop completes, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the factorial of n by initializing result to 1 and multiplying it by each integer from 2 through n inclusive, returning 1 when the range is empty." factorial]
    [INPUTS]
        <REF> n </REF>: int "Integer upper bound for the multiplicative range; factorial semantics typically expect a non-negative integer."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Product of 1 and each integer from 2 through n inclusive; equivalent to n! for non-negative n."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to 1. RESULT result]
            [COMMAND For each integer i from 2 up to n inclusive, execute the loop body; if no such i exists, skip the loop body. RESULT i]
            [COMMAND Within the loop for current i, multiply result by i and assign the product back to result. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
