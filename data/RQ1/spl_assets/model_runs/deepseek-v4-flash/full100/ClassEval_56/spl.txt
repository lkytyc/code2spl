# CLASS MetricsCalculator

[DEFINE_WORKER: "Initializes the true positive, false positive, false negative, and true negative counters to zero." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance whose confusion-matrix counters are to be initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.true_positives to the integer constant 0. RESULT self.true_positives]
            [COMMAND Initialize self.false_positives to the integer constant 0. RESULT self.false_positives]
            [COMMAND Initialize self.false_negatives to the integer constant 0. RESULT self.false_negatives]
            [COMMAND Initialize self.true_negatives to the integer constant 0. RESULT self.true_negatives]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates confusion-matrix counters by iterating over paired predicted and true labels and incrementing exactly one of the true positive, false positive, false negative, or true negative counters based on whether each label pair is (1,1), (1,0), (0,1), or (0,0)." update]
    [INPUTS]
        <REF> predicted_labels </REF>: iterable of int "Predicted binary labels, either 0 or 1, in the same order as true_labels."
        <REF> true_labels </REF>: iterable of int "True binary labels, either 0 or 1, in the same order as predicted_labels."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each pair (predicted, true) produced by zip(predicted_labels, true_labels), continuing until the shorter of the two iterables is exhausted, evaluate the following if/elif chain for that pair. RESULT label_pair]
            [COMMAND Within the current iteration of the for loop over zip(predicted_labels, true_labels), if predicted == 1 and true == 1, increment self.true_positives by 1. RESULT self.true_positives]
            [COMMAND Within the current iteration of the for loop over zip(predicted_labels, true_labels), else if predicted == 1 and true == 0, increment self.false_positives by 1. RESULT self.false_positives]
            [COMMAND Within the current iteration of the for loop over zip(predicted_labels, true_labels), else if predicted == 0 and true == 1, increment self.false_negatives by 1. RESULT self.false_negatives]
            [COMMAND Within the current iteration of the for loop over zip(predicted_labels, true_labels), else if predicted == 0 and true == 0, increment self.true_negatives by 1. RESULT self.true_negatives]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the current iteration of the for loop over zip(predicted_labels, true_labels), a pair (predicted, true) does not match any of the four equality conditions.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Continue to the next pair without incrementing any counter. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates internal counters with the given predicted and true labels, then returns precision as true positives divided by all positive predictions, or 0.0 when no positive predictions were made." precision]
    [INPUTS]
        <REF> predicted_labels </REF>: list "Predicted labels for the samples being evaluated."
        <REF> true_labels </REF>: list "Ground truth labels for the samples being evaluated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> precision_value </REF>: float "The computed precision score, or 0.0 if there are no positive predictions."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) to update internal counters. RESULT updated_state]
            [COMMAND Evaluate whether self.true_positives + self.false_positives equals 0. RESULT no_positive_predictions]
            [COMMAND If no_positive_predictions is false, compute precision as self.true_positives divided by (self.true_positives + self.false_positives). RESULT precision_value]
            [COMMAND Return precision_value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.true_positives + self.false_positives == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 as the precision score. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the metric state with predicted and true labels, returns 0.0 when there are no positive ground-truth cases, and otherwise returns the recall (true positive rate)." recall]
    [INPUTS]
        <REF> predicted_labels </REF>: array-like "Predicted labels for the samples being evaluated."
        <REF> true_labels </REF>: array-like "Ground-truth labels corresponding to the predicted labels."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "Recall score: 0.0 if true_positives + false_negatives is zero, otherwise true_positives divided by (true_positives + false_negatives)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Update the metric's internal accumulators by calling self.update(predicted_labels, true_labels). RESULT self]
            [COMMAND Compute the denominator as self.true_positives + self.false_negatives. RESULT denominator]
            [COMMAND Check whether denominator equals zero. RESULT denominator_is_zero]
            [COMMAND If denominator_is_zero is false, compute and return self.true_positives / denominator as the recall value. RESULT recall_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: denominator_is_zero is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 immediately as the recall value to avoid division by zero. RESULT recall_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the F1 score as the harmonic mean of precision and recall after updating internal state, returning 0.0 when the sum of precision and recall is zero." f1_score]
    [INPUTS]
        <REF> predicted_labels </REF>: Iterable "Predicted labels used to compute precision and recall."
        <REF> true_labels </REF>: Iterable "Ground-truth labels used to compute precision and recall."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "F1 score, or 0.0 when precision + recall equals 0.0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) to update internal state; ignore the return value. RESULT none]
            [COMMAND Call self.precision(predicted_labels, true_labels) and assign the returned value to local variable precision. RESULT precision]
            [COMMAND Call self.recall(predicted_labels, true_labels) and assign the returned value to local variable recall. RESULT recall]
            [COMMAND Evaluate whether precision + recall equals 0.0; if so, use the alternative early-return flow, otherwise continue to the final return. RESULT sum_is_zero]
            [COMMAND Compute (2 * precision * recall) / (precision + recall) and return the resulting F1 score. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: precision + recall == 0.0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 immediately as the method's result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the confusion matrix counts from the provided labels and returns the classification accuracy as the ratio of correct predictions to total instances, or 0.0 when no predictions have been accumulated." accuracy]
    [INPUTS]
        <REF> predicted_labels </REF>: array-like "Predicted class labels for the evaluated examples."
        <REF> true_labels </REF>: array-like "Ground truth class labels for the evaluated examples."
    [END_INPUTS]

    [OUTPUTS]
        <REF> accuracy </REF>: float "Returned classification accuracy: the fraction of correct predictions, or 0.0 if the total number of predictions is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update with predicted_labels and true_labels to update the confusion matrix counts. RESULT updated_confusion_matrix]
            [COMMAND Compute total as the sum of self.true_positives, self.true_negatives, self.false_positives, and self.false_negatives. RESULT total]
            [COMMAND When total is not zero, return (self.true_positives + self.true_negatives) / total as the accuracy. RESULT accuracy_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: total == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 immediately without dividing by zero. RESULT 0.0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
