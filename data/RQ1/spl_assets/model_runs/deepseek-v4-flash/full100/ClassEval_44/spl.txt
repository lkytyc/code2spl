    [END_INPUTS]

    [END_OUTPUTS]

        [SEQUENTIAL_BLOCK]

            [COMMAND Set the instance attribute JSON_MARK to the constant string '-JSON-'. RESULT self.JSON_MARK]

[DEFINE_WORKER: "Converts raw HTML into formatted text by replacing pre/blockquote content with a code marker, normalizing list items and paragraphs with punctuation, decoding HTML entities, and applying line-feed normalization." format_line_html_text]
    [INPUTS]
        <REF> html_text </REF>: str or bytes "Raw HTML content to parse and format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: str "Formatted text with code markers, normalized list items, punctuated paragraphs, decoded HTML entities, and normalized line feeds."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse html_text into a BeautifulSoup DOM using the lxml parser and assign the result to soup. RESULT soup]
            [COMMAND Find all pre and blockquote descendants in soup and assign the ResultSet to code_tag. RESULT code_tag]
            [COMMAND For each tag in code_tag, set tag.string to self.CODE_MARK. RESULT soup]
            [COMMAND Find all ul and ol descendants in soup and assign the ResultSet to ul_ol_group. RESULT ul_ol_group]
            [COMMAND For each ul_ol_item in ul_ol_group, collect all li descendants as li_group; for each li_item in li_group, get stripped text as li_item_text; if li_item_text is empty, skip it; if its last character is in string.punctuation, set li_item.string to '[-]' plus li_item_text; otherwise set li_item.string to '[-]' plus li_item_text plus '.'. RESULT soup]
            [COMMAND Find all p descendants in soup and assign the ResultSet to p_group. RESULT p_group]
            [COMMAND For each p_item in p_group, extract stripped text as p_item_text; if p_item_text is truthy: if its last character is in string.punctuation, set p_item.string to p_item_text and continue; otherwise find the next sibling; if next_sibling exists and self.CODE_MARK appears in next_sibling.get_text(), set p_item.string to p_item_text plus ':' and continue; otherwise set p_item.string to p_item_text plus '.'. RESULT soup]
            [COMMAND Extract all text from soup with soup.get_text(), decode HTML entities with gensim.utils.decode_htmlentities, and assign the decoded string to clean_text. RESULT clean_text]
            [COMMAND Return self.__format_line_feed(clean_text). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: html_text is None or len(html_text) == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the input HTML text and, when the code marker is present, extracts and returns the non-empty text contents of all pre and blockquote elements." extract_code_from_html_text]
    [INPUTS]
        <REF> self </REF>: object "Instance providing format_line_html_text and the CODE_MARK constant."
        <REF> html_text </REF>: str "HTML string to inspect and parse for code blocks."
    [END_INPUTS]

    [OUTPUTS]
        <REF> code_list </REF>: list[str] "Non-empty extracted text from pre and blockquote tags, or an empty list when the code marker is missing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.format_line_html_text(html_text) and assign the returned string to text_with_code_tag. RESULT text_with_code_tag]
            [COMMAND Check whether self.CODE_MARK is contained in text_with_code_tag and store the boolean result in has_code_mark. RESULT has_code_mark]
            [COMMAND If has_code_mark is false, return an empty list using the alternative flow; otherwise continue. RESULT has_code_mark]
            [COMMAND Set code_index_start to the integer constant 0. RESULT code_index_start]
            [COMMAND Parse the original html_text with BeautifulSoup(html_text, 'lxml') and assign the parsed document to soup. RESULT soup]
            [COMMAND Call soup.find_all(name=['pre', 'blockquote']) and assign the resulting list of elements to code_tag. RESULT code_tag]
            [COMMAND Count the non-overlapping occurrences of self.CODE_MARK in text_with_code_tag and assign the count to code_count. RESULT code_count]
            [COMMAND Initialize code_list as an empty list. RESULT code_list]
            [COMMAND For code_index from code_index_start inclusive to code_index_start + code_count exclusive, retrieve code_tag[code_index].get_text() as code; if code is truthy, append code to code_list. RESULT code_list]
            [COMMAND Return code_list. RESULT code_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: has_code_mark is false, meaning self.CODE_MARK is not contained in text_with_code_tag]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT []]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
