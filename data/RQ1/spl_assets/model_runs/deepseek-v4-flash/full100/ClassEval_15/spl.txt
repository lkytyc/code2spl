# CLASS BoyerMooreSearch

[DEFINE_WORKER: "Initializes the instance by storing the provided text and pattern and computing their character lengths." __init__]
    [INPUTS]
        <REF> text </REF>: string "The text to store in self.text and from which self.textLen is computed."
        <REF> pattern </REF>: string "The pattern to store in self.pattern and from which self.patLen is computed."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input text value to the instance attribute self.text. RESULT self.text]
            [COMMAND Assign the input pattern value to the instance attribute self.pattern. RESULT self.pattern]
            [COMMAND Compute the length of the input text and store it in the instance attribute self.textLen. RESULT self.textLen]
            [COMMAND Compute the length of the input pattern and store it in the instance attribute self.patLen. RESULT self.patLen]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches self.pattern from right to left for char and returns the index of its rightmost occurrence, or -1 when char is absent." match_in_pattern]
    [INPUTS]
        <REF> self </REF>: object "Object exposing the pattern and its length patLen."
        <REF> char </REF>: string "Character whose rightmost occurrence in self.pattern is to be found."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Rightmost index of char in self.pattern, or -1 if char is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize index i to self.patLen - 1, the rightmost position of self.pattern. RESULT i]
            [COMMAND For each i from self.patLen - 1 down to 0 inclusive, compare char with self.pattern[i]. RESULT char_matches]
            [COMMAND Within the backward loop, if char_matches is true, return i immediately. RESULT i]
            [COMMAND After the loop completes with no match, return -1. RESULT not_found]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the pattern from right to left at a given text alignment and returns the rightmost position in the text where a mismatch occurs, or -1 if all characters match." mismatch_in_text]
    [INPUTS]
        <REF> currentPos </REF>: int "Starting alignment position in the text where the pattern is compared."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Rightmost text index where a mismatch occurs, or -1 if the entire pattern matches at currentPos."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Start with the pattern aligned to the text at currentPos. RESULT alignment]
            [COMMAND Iterate over pattern indices i from self.patLen - 1 down to 0 inclusive, stepping by -1. RESULT i]
            [COMMAND For each i, compare self.pattern[i] with self.text[currentPos + i]. RESULT characters_equal]
            [COMMAND If self.pattern[i] does not equal self.text[currentPos + i], immediately return currentPos + i as the mismatch position. RESULT mismatch_position]
            [COMMAND If the loop completes without finding a mismatch, return -1. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans every possible alignment of the pattern against the text, records starting indices where a full match is found, and returns them; the bad-character skip is attempted on mismatch but has no effect because the for loop overwrites i." bad_character_heuristic]
    [INPUTS]
        <REF> self </REF>: object "State object carrying text, textLen, patLen and the helper methods mismatch_in_text(i) and match_in_pattern(character)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> positions </REF>: list "Starting indices i in self.text where mismatch_in_text(i) returned -1, meaning the pattern matched at that alignment."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize positions as an empty list to collect matching starting indices. RESULT positions]
            [COMMAND For each i in range(self.textLen - self.patLen + 1), a half-open range equivalent to i = 0, 1, ..., self.textLen - self.patLen inclusive; if self.textLen - self.patLen < 0, the range is empty. Compute mismatch_index = self.mismatch_in_text(i). RESULT mismatch_index]
            [COMMAND Use mismatch_index to select one of the two alternative paths: match branch if mismatch_index == -1, otherwise mismatch branch. RESULT mismatch_index]
            [COMMAND After the loop finishes, return positions. RESULT positions]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over i, mismatch_index == -1 (no mismatching character found; the pattern matches at starting index i)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append the current starting index i to positions. RESULT positions]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over i, mismatch_index != -1 (a mismatching character was found at self.text[mismatch_index])]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute match_index = self.match_in_pattern(self.text[mismatch_index]). RESULT match_index]
            [COMMAND Assign i = mismatch_index - match_index. The enclosing for loop overwrites i with the next value from range at the start of the next iteration, so this assignment does not alter the loop traversal. RESULT i]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
