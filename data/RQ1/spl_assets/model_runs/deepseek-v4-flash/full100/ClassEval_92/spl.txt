# CLASS UserLoginDB

[DEFINE_WORKER: "Initializes a SQLite database connection and cursor from the given database name." __init__]
    [INPUTS]
        <REF> db_name </REF>: str "Path or name of the SQLite database file to connect to."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database located at db_name by invoking sqlite3.connect(db_name) and assign the resulting connection object to self.connection. RESULT self.connection]
            [COMMAND Create a cursor for the database by invoking self.connection.cursor() and assign the resulting cursor object to self.cursor. RESULT self.cursor]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a new user with the supplied username and password into the users table and commits the transaction." insert_user]
    [INPUTS]
        <REF> username </REF>: str "The username to insert into the users table."
        <REF> password </REF>: str "The password to insert into the users table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute an INSERT statement on the users table with username and password as bound parameters using the database cursor. RESULT insert_result]
            [COMMAND Commit the current transaction on the database connection to persist the insert. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches for a user by username using a parameterized SELECT query and returns the first matching row or None." search_user_by_username]
    [INPUTS]
        <REF> username </REF>: string "The username to search for in the users table."
    [END_INPUTS]

    [OUTPUTS]
        <REF> user </REF>: tuple or None "The matching user row as a sequence, or None if no matching row exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query 'SELECT * FROM users WHERE username = ?' on self.cursor, binding username as the parameter. RESULT self.cursor]
            [COMMAND Call fetchone() on self.cursor to retrieve the next row from the query result set and assign it to user. RESULT user]
            [COMMAND Return user to the caller. RESULT user]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes a user from the users table by username using a parameterized DELETE query and commits the transaction." delete_user_by_username]
    [INPUTS]
        <REF> username </REF>: string "The username of the user to delete from the users table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the parameterized SQL DELETE statement on self.cursor to remove rows from the users table where the username column equals the input username. RESULT cursor_execution]
            [COMMAND Commit the pending deletion on self.connection to make it permanent in the database. RESULT connection_commit]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates a user login by looking up the supplied username and checking that the stored password at user[1] matches the supplied password." validate_user_login]
    [INPUTS]
        <REF> username </REF>: string "The username used to look up the user record."
        <REF> password </REF>: string "The plaintext password supplied for login validation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "True if a user record is found for username and the password stored at user[1] equals the supplied password; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Look up the user record for the given username by calling search_user_by_username(username). RESULT user]
            [COMMAND Evaluate whether user is not None and the value at user[1] equals the supplied password. RESULT is_valid]
            [COMMAND Return False as the default outcome when the login condition is not satisfied. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_valid is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True as the outcome when the login condition is satisfied. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
