# CLASS CurrencyConverter

[DEFINE_WORKER: "Initializes the exchange-rate converter by assigning self.rates a static lookup table of seven currency conversion factors with USD as the base reference rate." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.rates as a dictionary containing the static currency exchange rates: USD = 1.0 (base reference boundary), EUR = 0.85, GBP = 0.72, JPY = 110.15, CAD = 1.23, AUD = 1.34, CNY = 6.40. RESULT self.rates]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an amount from a source currency to a target currency using stored exchange rates, returning the amount unchanged for identical currencies and False for unsupported currency codes." convert]
    [INPUTS]
        <REF> amount </REF>: number "Monetary amount to convert."
        <REF> from_currency </REF>: string "Source currency code."
        <REF> to_currency </REF>: string "Target currency code."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number | bool "Converted amount in target currency; original amount if currencies are identical; False if either currency is missing from rates."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Look up the exchange rate for from_currency in self.rates and store it as from_rate. RESULT from_rate]
            [COMMAND Look up the exchange rate for to_currency in self.rates and store it as to_rate. RESULT to_rate]
            [COMMAND Compute converted_amount as (amount / from_rate) * to_rate. RESULT converted_amount]
            [COMMAND Return converted_amount. RESULT converted_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: from_currency equals to_currency, checked before rate validation]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return amount unchanged. RESULT amount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: from_currency is not present in self.rates or to_currency is not present in self.rates]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to signal invalid currency codes. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a materialized list of the currency codes that are currently keys in the rates dictionary." get_supported_currencies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: list "A new list of all currency codes currently stored as keys in self.rates."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return a new list containing all current keys of self.rates, preserving dictionary insertion order. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a new exchange rate for a currency to self.rates only if the currency is not already present, returning False when the currency already exists." add_currency_rate]
    [INPUTS]
        <REF> currency </REF>: hashable key "The currency identifier used as the key in self.rates."
        <REF> rate </REF>: object "The exchange rate value to store for the given currency."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean or None "Returns False if currency already exists in self.rates; otherwise returns None after storing the rate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether currency is a key in self.rates. RESULT currency_present]
            [COMMAND If currency_present is false, assign self.rates[currency] = rate. RESULT updated_rates]
            [COMMAND Return None implicitly after the assignment. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency_present is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the exchange rate for an existing currency in the rates dictionary, returning False if the currency is not found." update_currency_rate]
    [INPUTS]
        <REF> currency </REF>: Hashable "Currency identifier used as the key in self.rates."
        <REF> new_rate </REF>: Any "New rate value to store for the currency."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool or None "Returns False if currency is not in self.rates; otherwise returns None implicitly after assigning new_rate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency is a key in self.rates. RESULT currency_exists]
            [COMMAND For the path where currency is present in self.rates, assign new_rate to self.rates[currency]. RESULT self.rates]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency is not a key in self.rates]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.rates. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
