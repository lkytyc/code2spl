# CLASS AreaCalculator

[DEFINE_WORKER: "Initializes the object by storing the supplied radius unchanged in the instance attribute self.radius." __init__]
    [INPUTS]
        <REF> radius </REF>: number "The radius value to assign to self.radius; stored without validation or transformation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the __init__ method, assign the value of the parameter radius to the instance attribute self.radius. RESULT self.radius]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the area of a circle by squaring the instance's radius and multiplying by pi." calculate_circle_area]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> area </REF>: float "The area of the circle, calculated as pi times the radius squared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance attribute self.radius. RESULT radius]
            [COMMAND Compute the square of radius using exponentiation: radius_squared = radius ** 2. RESULT radius_squared]
            [COMMAND Multiply the constant math.pi by radius_squared to compute the area: area = math.pi * radius_squared. RESULT area]
            [COMMAND Return area as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the surface area of a sphere using the instance radius and returns 4πr²." calculate_sphere_area]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float "Surface area of the sphere, equal to 4πr²."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sphere surface area as 4 * math.pi * self.radius ** 2 and return the result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total surface area of a cylinder from the instance radius and the supplied height using 2πr(r + h)." calculate_cylinder_area]
    [INPUTS]
        <REF> height </REF>: numeric "Height of the cylinder."
        <REF> self.radius </REF>: numeric "Radius of the cylinder, read from the instance attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> area </REF>: numeric "Computed total surface area of the cylinder."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the total surface area as 2 * math.pi * self.radius * (self.radius + height), using the radius from self and the height argument. RESULT area]
            [COMMAND Return the computed area as the method result. RESULT area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the area of a circular sector by multiplying the square of the object's radius by the angle in radians and dividing by 2." calculate_sector_area]
    [INPUTS]
        <REF> angle </REF>: numeric "Angle of the circular sector in radians."
    [END_INPUTS]

    [OUTPUTS]
        <REF> sector_area </REF>: numeric "The area of the circular sector calculated as (self.radius ** 2 * angle) / 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the current value of self.radius RESULT radius]
            [COMMAND Compute the square of radius RESULT radius_squared]
            [COMMAND Multiply radius_squared by angle RESULT product]
            [COMMAND Divide product by 2 RESULT sector_area]
            [COMMAND Return sector_area as the result RESULT sector_area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the area of an annulus by taking the difference of the squares of the outer and inner radii and multiplying by pi." calculate_annulus_area]
    [INPUTS]
        <REF> inner_radius </REF>: numeric "Inner radius of the annulus."
        <REF> outer_radius </REF>: numeric "Outer radius of the annulus."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float "Area of the annulus, computed as math.pi * (outer_radius ** 2 - inner_radius ** 2)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the square of outer_radius. RESULT outer_radius_squared]
            [COMMAND Compute the square of inner_radius. RESULT inner_radius_squared]
            [COMMAND Subtract inner_radius_squared from outer_radius_squared to get the squared radius difference. RESULT radius_squared_difference]
            [COMMAND Multiply radius_squared_difference by math.pi to get the annulus area. RESULT annulus_area]
            [COMMAND Return annulus_area as the result of the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
