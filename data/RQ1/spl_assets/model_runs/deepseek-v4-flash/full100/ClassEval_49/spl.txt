# CLASS JobMarketplace

[DEFINE_WORKER: "Initializes a new instance by creating two independent empty list attributes, job_listings and resumes." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance attribute job_listings to an empty list. RESULT self.job_listings]
            [COMMAND Initialize the instance attribute resumes to an empty list. RESULT self.resumes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a job listing dictionary from the job title, company, and requirements, then appends it to the job listings collection." post_job]
    [INPUTS]
        <REF> job_title </REF>: string "The title of the job to be posted."
        <REF> company </REF>: string "The company offering the job."
        <REF> requirements </REF>: list of strings "A list of job requirements."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a dictionary named job with the keys job_title, company, and requirements, using the current values of job_title, company, and requirements in left-to-right order. RESULT job]
            [COMMAND Append the job dictionary to the end of self.job_listings. RESULT self.job_listings]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the first occurrence of the specified job from the job listings list." remove_job]
    [INPUTS]
        <REF> job </REF>: object "The job item to remove from self.job_listings."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove the first element equal to job from self.job_listings, shifting subsequent elements left if the list contains duplicates. RESULT self.job_listings]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: job is not present in self.job_listings]
        [LOG "ValueError: list.remove(job) failed because job was not found in self.job_listings"]
        [THROW ValueError "Raised when job does not exist in self.job_listings."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a resume dictionary from the provided name, skills, and experience fields and appends it to the resumes collection." submit_resume]
    [INPUTS]
        <REF> name </REF>: string "The candidate's name to store in the resume."
        <REF> skills </REF>: array "The candidate's skills to store in the resume."
        <REF> experience </REF>: string "The candidate's experience description to store in the resume."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a dictionary named resume with keys 'name', 'skills', and 'experience' set to the values of the parameters name, skills, and experience. RESULT resume]
            [COMMAND Append the resume dictionary to the resumes list on self. RESULT self.resumes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the first occurrence of the specified resume from the resumes list, mutating self.resumes in place and returning None." withdraw_resume]
    [INPUTS]
        <REF> resume </REF>: object "The resume object to locate and remove from self.resumes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return </REF>: None "The method always returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND In self.resumes, locate the first element equal to resume, remove that element, and shift the remaining elements left to close the gap; ignore the None return value. RESULT self.resumes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: resume is not present in self.resumes]
        [LOG "self.resumes.remove(resume) raises ValueError because resume is not in the list."]
        [THROW ValueError "Raised when resume is not found in self.resumes."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.resumes is None or does not support remove()]
        [LOG "self.resumes.remove(resume) raises AttributeError because self.resumes is not a mutable list."]
        [THROW AttributeError "Raised when self.resumes is not a list or is None."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters self.job_listings and returns job listings whose title contains criteria as a case-insensitive substring or whose requirements list contains criteria as an exact case-insensitive match." search_jobs]
    [INPUTS]
        <REF> criteria </REF>: string "Search term used to filter job listings case-insensitively."
    [END_INPUTS]

    [OUTPUTS]
        <REF> matching_jobs </REF>: list "List of job listings that match the search criteria."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize matching_jobs as an empty list. RESULT matching_jobs]
            [COMMAND For each job_listing in self.job_listings, evaluate the OR condition: if criteria.lower() is a substring of job_listing["job_title"].lower(), accept the listing without building the lowercased requirements list; otherwise, accept it only if criteria.lower() is an exact match for any lowercased requirement in job_listing["requirements"]. When the condition is true, append job_listing to matching_jobs. RESULT matching_jobs]
            [COMMAND After all job listings have been processed, return matching_jobs. RESULT matching_jobs]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects and returns all resumes from self.resumes that satisfy the job's requirements, preserving their original order." get_job_applicants]
    [INPUTS]
        <REF> job </REF>: dict "Job posting containing a 'requirements' key used to evaluate each resume."
    [END_INPUTS]

    [OUTPUTS]
        <REF> applicants </REF>: list "List of resumes that match the job requirements, in their original order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize applicants as an empty list. RESULT applicants]
            [COMMAND Iterate over each resume in self.resumes. RESULT resume]
            [COMMAND Within the for loop over resume, if self.matches_requirements(resume, job['requirements']) returns a truthy value, append the current resume to applicants. RESULT applicants]
            [COMMAND Return the applicants list. RESULT applicants]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether every skill in resume['skills'] is present in the requirements collection, returning False as soon as any skill is missing and True otherwise." matches_requirements]
    [INPUTS]
        <REF> resume </REF>: object/dictionary "Dictionary containing a 'skills' key whose value is a list or sequence of skills to validate."
        <REF> requirements </REF>: collection "Collection of required skills, such as a set or list, used to check membership of each resume skill."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if all resume skills are contained in requirements; False if any resume skill is missing from requirements."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the value of resume["skills"] and store it as the sequence of skills to validate. RESULT skills]
            [COMMAND Iterate over the skills sequence in its natural order, treating each item as the current skill. RESULT current_skill]
            [COMMAND For the current skill, test whether it is absent from the requirements collection. RESULT skill_is_missing]
            [COMMAND If skill_is_missing is true, return False immediately and skip any remaining iterations. RESULT False]
            [COMMAND If iteration completes without any missing skill, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
