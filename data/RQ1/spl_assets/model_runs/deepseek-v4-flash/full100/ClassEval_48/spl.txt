# CLASS IpUtil

[DEFINE_WORKER: "Checks whether a given string is a valid IPv4 address by attempting to parse it with socket.inet_pton, returning True if parsing succeeds and False if a socket.error is raised." is_valid_ipv4]
    [INPUTS]
        <REF> ip_address </REF>: string "IPv4 address string to validate"
    [END_INPUTS]

    [OUTPUTS]
        <REF> is_valid </REF>: boolean "True if ip_address is a valid IPv4 address; False otherwise"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.inet_pton(socket.AF_INET, ip_address) to parse ip_address as an IPv4 address. RESULT parse_result]
            [COMMAND If parsing succeeds without raising socket.error, set is_valid to True and return it. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.inet_pton(socket.AF_INET, ip_address) raises socket.error during parsing]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the socket.error exception, set is_valid to False, and return it. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the given string is a valid IPv6 address by attempting to parse it with socket.inet_pton and returns True only if parsing succeeds." is_valid_ipv6]
    [INPUTS]
        <REF> ip_address </REF>: string "IPv6 address string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if ip_address is a valid IPv6 address; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the try block, attempt to parse ip_address as an IPv6 address using socket.inet_pton with socket.AF_INET6. RESULT conversion_result]
            [COMMAND If the parsing succeeds, stop execution and return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.error is raised while parsing ip_address with socket.inet_pton using socket.AF_INET6.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the socket.error and return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resolves the given IP address to a hostname using reverse DNS lookup, returning None if the lookup fails." get_hostname]
    [INPUTS]
        <REF> ip_address </REF>: str "IP address to resolve to a hostname."
    [END_INPUTS]

    [OUTPUTS]
        <REF> hostname </REF>: str or None "Resolved hostname, or None if no hostname is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the try block, call socket.gethostbyaddr with ip_address to perform a reverse DNS lookup. RESULT lookup_result]
            [COMMAND Extract the first element from the lookup result tuple to get the hostname. RESULT hostname]
            [COMMAND Return hostname. RESULT hostname]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.gethostbyaddr raises socket.herror during the reverse DNS lookup]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the except socket.herror block, return None to indicate that no hostname could be resolved. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
