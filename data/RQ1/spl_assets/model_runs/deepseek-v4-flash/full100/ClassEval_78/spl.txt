# CLASS SplitSentence

[DEFINE_WORKER: "Splits a text string into sentence substrings using a regex that treats periods and question marks followed by whitespace as sentence boundaries while avoiding abbreviation-like patterns." split_sentences]
    [INPUTS]
        <REF> sentences_string </REF>: string "The input text to be split into sentences."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of strings "The list of sentence substrings produced by the regex split."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split `sentences_string` with `re.split` using the pattern `(?<!\w\.\w.)(?<![A-Z][a-z]\.)(?<=\.|\?)\s`, so that whitespace preceded by '.' or '?' is a delimiter except after abbreviation-like sequences such as 'e.g.' or 'Mr.'; assign the resulting list to `sentences`. RESULT sentences]
            [COMMAND Return `sentences`. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts the number of words in a sentence after removing all characters except ASCII letters and whitespace." count_words]
    [INPUTS]
        <REF> sentence </REF>: str "The input text to be sanitized and split into words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_count </REF>: int "The number of words remaining after sanitization."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sanitize sentence by replacing every character that is not an ASCII letter or whitespace with an empty string, using the regex pattern [^a-zA-Z\s]. RESULT sanitized_sentence]
            [COMMAND Split sanitized_sentence on any sequence of whitespace characters to produce a list of words. RESULT words]
            [COMMAND Set return_count to the number of elements in words. RESULT return_count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the maximum word count among all sentences in the input string by splitting the text into sentences, counting the words in each sentence, and keeping the largest count." process_text_file]
    [INPUTS]
        <REF> sentences_string </REF>: string "The input text to be split into sentences and scanned for the maximum word count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> max_count </REF>: integer "The largest number of words found in any single sentence; 0 if the input contains no sentences."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input sentences_string into a list of sentences using self.split_sentences. RESULT sentences]
            [COMMAND Initialize max_count to 0 as the starting maximum word count. RESULT max_count]
            [COMMAND Begin a loop over the sentences list, binding the current element to the variable sentence. RESULT sentence]
            [COMMAND Within the loop over each sentence, compute count by calling self.count_words(sentence). RESULT count]
            [COMMAND Within the loop over each sentence, if count is greater than max_count, set max_count to count; otherwise leave max_count unchanged. RESULT max_count]
            [COMMAND After the loop finishes, return max_count. RESULT max_count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
