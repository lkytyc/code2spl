    [INPUTS]
        <REF> BOARD_SIZE </REF>: int "Board size used to set the self.BOARD_SIZE instance attribute."
        <REF> ICONS </REF>: sequence "Icons used to set the self.ICONS instance attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the BOARD_SIZE parameter to the instance attribute self.BOARD_SIZE. RESULT self.BOARD_SIZE]
            [COMMAND Assign the ICONS parameter to the instance attribute self.ICONS. RESULT self.ICONS]
            [COMMAND With self.BOARD_SIZE and self.ICONS already assigned, call self.create_board() and assign its return value to the instance attribute self.board. RESULT self.board]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a new board of the specified size by filling every cell with a random icon selected from self.ICONS." create_board]
    [INPUTS]
        <REF> self.BOARD_SIZE </REF>: tuple of two integers "Board dimensions given as (rows, columns), where rows = self.BOARD_SIZE[0] and columns = self.BOARD_SIZE[1]."
        <REF> self.ICONS </REF>: list "Non-empty list of icons from which each board cell is randomly chosen."
    [END_INPUTS]

    [OUTPUTS]
        <REF> board </REF>: list[list[str]] "A list of rows where each row contains randomly selected icons from self.ICONS."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.BOARD_SIZE as (rows, columns), where rows = self.BOARD_SIZE[0] and columns = self.BOARD_SIZE[1]. RESULT board_dimensions]
            [COMMAND Build board as an outer list of length rows; for each row, build an inner list of length columns by selecting one random icon from self.ICONS for each column. RESULT board]
            [COMMAND Return board. RESULT board]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a move from pos1 to pos2 is valid by requiring both positions to be within the board bounds, distinct, showing the same icon, and connected by a valid path." is_valid_move]
    [INPUTS]
        <REF> pos1 </REF>: tuple/list of two ints "The first board position as (x, y)."
        <REF> pos2 </REF>: tuple/list of two ints "The second board position as (x, y)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True only if all validity checks pass; False if either position is out of bounds, the positions are the same, the icons differ, or no valid path exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack pos1 into x1 and y1, and pos2 into x2 and y2. RESULT x1, y1, x2, y2]
            [COMMAND Check whether all four coordinates are inside the board bounds, using the short-circuiting condition 0 <= x1 < self.BOARD_SIZE[0] and 0 <= y1 < self.BOARD_SIZE[1] and 0 <= x2 < self.BOARD_SIZE[0] and 0 <= y2 < self.BOARD_SIZE[1]; continue only if the whole condition is true. RESULT bounds_ok]
            [COMMAND After bounds_ok is true, check whether pos1 is different from pos2; continue only if the positions are distinct. RESULT positions_distinct]
            [COMMAND After positions_distinct is true, compare self.board[x1][y1] with self.board[x2][y2] to confirm the two positions contain the same icon; continue only if they are equal. RESULT icons_equal]
            [COMMAND After icons_equal is true, call self.has_path(pos1, pos2) to test whether a valid path connects the two positions; continue only if it returns True. RESULT path_exists]
            [COMMAND All validity checks have passed, so return True. RESULT final_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After unpacking, if any coordinate is outside the board range: x1 < 0 or x1 >= self.BOARD_SIZE[0] or y1 < 0 or y1 >= self.BOARD_SIZE[1] or x2 < 0 or x2 >= self.BOARD_SIZE[0] or y2 < 0 or y2 >= self.BOARD_SIZE[1].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT early_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the board-bounds check succeeds, if pos1 == pos2.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT early_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the same-position check succeeds, if self.board[x1][y1] != self.board[x2][y2].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT early_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the icon-match check succeeds, if self.has_path(pos1, pos2) returns False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT early_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether pos1 and pos2 are connected by an orthogonally adjacent path of board cells with the same value, using an iterative depth-first search stack." has_path]
    [INPUTS]
        <REF> pos1 </REF>: tuple[int, int] "Starting board position as a (row, column) pair."
        <REF> pos2 </REF>: tuple[int, int] "Target board position as a (row, column) pair."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if a path from pos1 to pos2 exists, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize visited as an empty set. RESULT visited]
            [COMMAND Initialize stack as a list containing pos1. RESULT stack]
            [COMMAND While stack is not empty, pop the last element from stack as current_pos. RESULT current_pos]
            [COMMAND Within the while loop, if current_pos equals pos2, return True immediately. RESULT True]
            [COMMAND Within the while loop, if current_pos is already in visited, continue to the next while-loop iteration. RESULT None]
            [COMMAND Within the while loop, after the target and visited checks, add current_pos to visited. RESULT visited]
            [COMMAND Within the while loop, after adding current_pos to visited, unpack current_pos into x and y. RESULT x, y]
            [COMMAND Within the while loop, for each direction (dx, dy) in [(0, 1), (0, -1), (1, 0), (-1, 0)], compute new_x = x + dx and new_y = y + dy. RESULT new_x, new_y]
            [COMMAND Within the neighbor-direction for loop, if 0 <= new_x < self.BOARD_SIZE[0] and 0 <= new_y < self.BOARD_SIZE[1] and (new_x, new_y) is not in visited and self.board[new_x][new_y] equals self.board[x][y], append (new_x, new_y) to stack. RESULT stack]
            [COMMAND After the while loop exits without having returned True, return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Clears the two board cells at positions pos1 and pos2 by writing a space character into each." remove_icons]
    [INPUTS]
        <REF> pos1 </REF>: tuple (row, column) "Coordinates of the first board cell to clear."
        <REF> pos2 </REF>: tuple (row, column) "Coordinates of the second board cell to clear."
    [END_INPUTS]

    [OUTPUTS]
        <REF> None </REF>: NoneType "No value is returned; the method mutates self.board in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack pos1 into x1 and y1. RESULT x1, y1]
            [COMMAND Unpack pos2 into x2 and y2. RESULT x2, y2]
            [COMMAND Write a space character into the existing element self.board[x1][y1]. RESULT self.board[x1][y1]]
            [COMMAND Write a space character into the existing element self.board[x2][y2]. RESULT self.board[x2][y2]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True only if every cell in self.board is a space; otherwise it returns False as soon as the first non-space icon is encountered." is_game_over]
    [INPUTS]
        <REF> self </REF>: object "Game object exposing a board attribute, where board is an iterable of rows and each row is an iterable of icon characters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if all cells on the board are spaces; False if any cell contains a non-space icon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board in order. RESULT row]
            [COMMAND Within the current row, evaluate any(icon != ' ' for icon in row), scanning icons from left to right and short-circuiting at the first non-space icon. RESULT has_non_space_icon]
            [COMMAND If the complete for loop finishes without encountering a non-space icon, return True. RESULT final_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: has_non_space_icon is True for the current row]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, exiting the method without inspecting any remaining rows or cells. RESULT early_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
