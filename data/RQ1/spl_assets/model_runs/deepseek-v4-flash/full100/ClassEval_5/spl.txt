# CLASS AutomaticGuitarSimulator

[DEFINE_WORKER: "Initializes the object by storing the input text in the play_text attribute." __init__]
    [INPUTS]
        <REF> text </REF>: text "The text value to store in the instance attribute play_text."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the text parameter to the instance attribute self.play_text. RESULT self.play_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses self.play_text into a list of chord/tune dictionaries, using each space-delimited segment's leading alphabetic characters as the Chord and the remainder as the Tune, with optional display callback." interpret]
    [INPUTS]
        <REF> self </REF>: object "Object whose play_text attribute is parsed and whose display method may be called."
        <REF> display </REF>: bool "If true, calls self.display(play_chord, play_value) for every parsed segment; defaults to False."
    [END_INPUTS]

    [OUTPUTS]
        <REF> play_list </REF>: list[dict[str, str]] "List of dictionaries with keys Chord and Tune, or an empty list when self.play_text is empty or whitespace-only."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Strip leading and trailing whitespace from self.play_text and check whether the stripped result is empty. RESULT play_text_stripped]
            [COMMAND Initialize play_list as an empty list. RESULT play_list]
            [COMMAND Split self.play_text by literal single space characters into play_segs; consecutive spaces produce empty segments. RESULT play_segs]
            [COMMAND For each play_seg in play_segs, set pos to 0. RESULT pos]
            [COMMAND Within the loop over play_seg, iterate over each character ele in play_seg; while ele.isalpha() is true, increment pos and continue; stop scanning at the first non-alphabetic character. RESULT pos]
            [COMMAND Within the loop over play_seg, set play_chord to play_seg[0:pos], the leading alphabetic substring. RESULT play_chord]
            [COMMAND Within the loop over play_seg, set play_value to play_seg[pos:], the remaining substring after the leading alphabetic prefix. RESULT play_value]
            [COMMAND Within the loop over play_seg, append {"Chord": play_chord, "Tune": play_value} to play_list. RESULT play_list]
            [COMMAND Within the loop over play_seg, if display is truthy, call self.display(play_chord, play_value). RESULT display_output]
            [COMMAND After the loop over all play_segs, return play_list. RESULT play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.play_text.strip() is empty, meaning self.play_text is empty or contains only whitespace]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT empty_play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates and returns a string that reports normal guitar playing using a chord key and a tune value." display]
    [INPUTS]
        <REF> key </REF>: Any "Chord identifier to be converted to a string and inserted into the Chord field."
        <REF> value </REF>: Any "Tune value to be converted to a string and inserted into the Play Tune field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "Formatted string: 'Normal Guitar Playing -- Chord: <key>, Play Tune: <value>'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a formatted message by replacing the first '%s' placeholder in the literal string 'Normal Guitar Playing -- Chord: %s, Play Tune: %s' with str(key) and the second '%s' placeholder with str(value). RESULT formatted_message]
            [COMMAND Return formatted_message as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
