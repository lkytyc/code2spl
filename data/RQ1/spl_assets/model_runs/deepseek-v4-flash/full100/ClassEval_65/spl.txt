# CLASS NumberWordFormatter

[DEFINE_WORKER: "Initializes five numeric lookup tables for converting numbers to English words and scale suffixes." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.NUMBER as a 10-element unit-name lookup table: index 0 is an empty string, indices 1 through 9 map to 'ONE' through 'NINE'. RESULT self.NUMBER]
            [COMMAND Initialize self.NUMBER_TEEN as a 10-element teen-name lookup table: indices 0 through 9 map to 'TEN' through 'NINETEEN'. RESULT self.NUMBER_TEEN]
            [COMMAND Initialize self.NUMBER_TEN as a 9-element tens-name lookup table: indices 0 through 8 map to 'TEN' through 'NINETY'. RESULT self.NUMBER_TEN]
            [COMMAND Initialize self.NUMBER_MORE as a 4-element scale-name lookup table: index 0 is empty, indices 1 through 3 map to 'THOUSAND', 'MILLION', 'BILLION'. RESULT self.NUMBER_MORE]
            [COMMAND Initialize self.NUMBER_SUFFIX as a 16-element suffix lookup table: indices 0, 1, 3, 6, 9, 12, and 15 map to 'k', 'w', 'm', 'b', 't', 'p', and 'e' respectively, while all other indices are empty placeholder strings. RESULT self.NUMBER_SUFFIX]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats a non-None input by converting it to a string and delegating to self.format_string, or returns an empty string when the input is None." format]
    [INPUTS]
        <REF> x </REF>: object "Value to format; may be None."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted string for non-None x, or an empty string if x is None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether x is None using identity comparison. RESULT is_none]
            [COMMAND If x is not None, convert x to a string with str(x). RESULT x_str]
            [COMMAND If x is not None, call self.format_string(x_str) and return the result. RESULT formatted_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: x is None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string. RESULT empty_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats a dot-separated decimal string into an English check-amount phrase by converting the integer part with trans_three/parse_more and cents with trans_two, returning 'ZERO ONLY' when the integer part is blank." format_string]
    [INPUTS]
        <REF> x </REF>: string "Input decimal amount string; only the first two dot-separated components are used."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "English amount phrase ending in 'ONLY', or 'ZERO ONLY' if the integer part is blank."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split x on '.' and pad the resulting list with '' so it has at least two elements; take the first two elements as lstr and rstr. RESULT lstr, rstr]
            [COMMAND Reverse lstr and store the result as lstrrev. RESULT lstrrev]
            [COMMAND Initialize a as a list of 5 empty strings. RESULT a]
            [COMMAND If len(lstrrev) % 3 == 1, append '00' to lstrrev; else if len(lstrrev) % 3 == 2, append '0' to lstrrev; otherwise leave lstrrev unchanged. RESULT lstrrev]
            [COMMAND Initialize lm as an empty string. RESULT lm]
            [COMMAND For each i from 0 to len(lstrrev) // 3 - 1, set a[i] to the reverse of lstrrev[3*i:3*i+3]; if a[i] != '000', prepend self.trans_three(a[i]) + ' ' + self.parse_more(i) + ' ' to lm; otherwise append self.trans_three(a[i]) to lm. RESULT lm]
            [COMMAND If rstr is truthy, set xs to 'AND CENTS ' concatenated with self.trans_two(rstr) and a trailing space; otherwise set xs to an empty string. RESULT xs]
            [COMMAND If lm.strip() is empty, return 'ZERO ONLY'; otherwise return the string formed by concatenating lm.strip(), a space, xs, and the suffix 'ONLY'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a numeric string from 0 to 99 into its English word representation by zero-padding it to two digits and branching on the tens and ones digits." trans_two]
    [INPUTS]
        <REF> s </REF>: str "Numeric string representing an integer from 0 to 99."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: str "English word representation of the numeric value described by s."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Zero-pad the input numeric string s to at least two characters with leading zeros by executing s = s.zfill(2); use this zero-padded value for all later checks. RESULT s]
            [COMMAND Evaluate the if/elif/else chain on the zero-padded string s in order and execute the first matching branch; each branch returns immediately, so later branches are skipped. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The first character of s is '0' (s[0] == '0').]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.NUMBER[int(s[-1])], using the last character of s as the index into the NUMBER word list. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The first character of s is '1' (s[0] == '1'), reached after the first condition failed.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.NUMBER_TEEN[int(s) - 10], converting the entire zero-padded string to an integer and subtracting 10. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The second character of s is '0' (s[1] == '0'), reached after the first two conditions failed.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.NUMBER_TEN[int(s[0]) - 1], using the tens digit minus 1 as the index into the NUMBER_TEN word list. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: None of the previous conditions match: the first character is neither '0' nor '1', and the second character is not '0'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.NUMBER_TEN[int(s[0]) - 1] + " " + self.NUMBER[int(s[-1])], constructing the tens word and the ones word separated by a space. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a three-digit numeric string into its English word equivalent, delegating leading-zero cases to trans_two and using 'HUNDRED' with 'AND' only when a nonzero hundred part is followed by a nonzero two-digit part." trans_three]
    [INPUTS]
        <REF> s </REF>: string "A three-character numeric string representing a number from 000 to 999."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The English phrase for the number; leading-zero numbers use the two-digit conversion, exact hundreds include only the hundred word, and general three-digit numbers include 'AND' before the two-digit portion."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the first character of s is '0'. RESULT has_leading_zero]
            [COMMAND If has_leading_zero is true, return the result of calling trans_two with the substring s[1:]. RESULT leading_zero_result]
            [COMMAND If has_leading_zero is false, check whether the substring s[1:] equals '00'. RESULT is_exact_hundred]
            [COMMAND If has_leading_zero is false and is_exact_hundred is true, return the value from the NUMBER dictionary for int(s[0]) followed by ' HUNDRED'. RESULT exact_hundred_result]
            [COMMAND If has_leading_zero is false and is_exact_hundred is false, return the value from the NUMBER dictionary for int(s[0]) followed by ' HUNDRED AND ' and the result of calling trans_two with s[1:]. RESULT general_three_digit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the element at index or key i from the NUMBER_MORE collection of the current object." parse_more]
    [INPUTS]
        <REF> self </REF>: object with NUMBER_MORE collection attribute "The object containing the NUMBER_MORE collection to read."
        <REF> i </REF>: valid index or key for NUMBER_MORE "The index or key identifying the element to return from the NUMBER_MORE collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: element type of NUMBER_MORE "The value stored at index or key i in the NUMBER_MORE collection."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the NUMBER_MORE attribute on self. RESULT number_more_collection]
            [COMMAND Read the element at index or key i from number_more_collection. RESULT return_value]
            [COMMAND Return return_value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
