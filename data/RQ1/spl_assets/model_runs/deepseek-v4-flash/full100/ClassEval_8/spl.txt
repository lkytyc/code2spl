# CLASS BankAccount

[DEFINE_WORKER: "Initializes a new object instance by setting its balance attribute to the provided balance value, defaulting to 0 when omitted." __init__]
    [INPUTS]
        <REF> self </REF>: object "The object instance being initialized."
        <REF> balance </REF>: number "Initial balance value to store in self.balance; defaults to 0 when omitted."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of the balance parameter to the instance attribute self.balance. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deposits a non-negative amount into the account balance, raises ValueError for negative deposits, and returns the updated balance." deposit]
    [INPUTS]
        <REF> amount </REF>: numeric "Amount to deposit; must be non-negative."
    [END_INPUTS]

    [OUTPUTS]
        <REF> balance </REF>: numeric "Updated account balance after the deposit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether amount is less than 0; if so, follow the ValueError exception flow. RESULT amount]
            [COMMAND Add amount to self.balance and store the updated value in self.balance. RESULT self.balance]
            [COMMAND Return the updated self.balance value. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount is less than 0]
        [LOG "Invalid amount"]
        [THROW ValueError "Raised when the deposit amount is negative."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the withdrawal amount is non-negative and not greater than the current balance, then subtracts it from the balance and returns the updated balance." withdraw]
    [INPUTS]
        <REF> self </REF>: object "The account instance whose balance is read, checked, and updated."
        <REF> amount </REF>: numeric "The amount to withdraw; must be non-negative and less than or equal to the current balance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> updated_balance </REF>: numeric "The value of self.balance after subtracting amount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether amount is strictly less than 0; if so, stop normal execution and use the exception flow for an invalid amount. RESULT negative_amount_check]
            [COMMAND Check whether amount is strictly greater than self.balance; if so, stop normal execution and use the exception flow for insufficient balance. RESULT insufficient_balance_check]
            [COMMAND Subtract amount from self.balance by executing self.balance -= amount and keep the updated value as self.balance. RESULT self.balance]
            [COMMAND Return the updated self.balance to the caller. RESULT updated_balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount is strictly less than 0]
        [LOG "Rejected withdrawal because amount is negative; raising ValueError with message 'Invalid amount'."]
        [THROW ValueError "Raised when the withdrawal amount is negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: amount is strictly greater than self.balance]
        [LOG "Rejected withdrawal because amount exceeds the current balance; raising ValueError with message 'Insufficient balance.'."]
        [THROW ValueError "Raised when the withdrawal amount is greater than the current balance."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current balance value stored on the object instance." view_balance]
    [INPUTS]
        <REF> self </REF>: object "The current object instance whose balance attribute is accessed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: unknown "The value of the balance attribute of the current object instance."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value of the balance attribute from the current object instance. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Transfers a given amount from this account to another account by first withdrawing it from the source account and then depositing the same amount into the destination account." transfer]
    [INPUTS]
        <REF> self </REF>: Account "Source account from which the amount is withdrawn."
        <REF> other_account </REF>: Account "Destination account into which the amount is deposited."
        <REF> amount </REF>: Numeric "Amount to transfer; the same value is passed unchanged to withdraw and deposit."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Withdraw the specified amount from this account by calling self.withdraw(amount). If this call raises, the method exits before the deposit step. RESULT _]
            [COMMAND Deposit the same amount into the destination account by calling other_account.deposit(amount). RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
