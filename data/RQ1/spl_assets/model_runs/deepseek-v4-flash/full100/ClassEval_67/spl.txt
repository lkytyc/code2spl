# CLASS Order

[DEFINE_WORKER: "Initializes the order data structures by setting menu and selected_dishes to empty lists and sales to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.menu to an empty list; menu entries are expected to be dicts with keys dish, price, and count. RESULT self.menu]
            [COMMAND Initialize self.selected_dishes to an empty list; selected_dish entries are expected to be dicts with keys dish, count, and price. RESULT self.selected_dishes]
            [COMMAND Initialize self.sales to an empty dictionary. RESULT self.sales]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a dish to selected_dishes after decrementing the matching menu item's count when sufficient stock exists, or returns False immediately if a matching menu item has insufficient count." add_dish]
    [INPUTS]
        <REF> self </REF>: object "The object whose menu list and selected_dishes list are accessed."
        <REF> dish </REF>: dict "Dictionary containing the dish name under the key 'dish' and the requested count under the key 'count'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the dish was appended to selected_dishes; False if a matching menu_dish had insufficient count and caused an early return."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Start a for loop over each menu_dish in self.menu. RESULT menu_dish]
            [COMMAND Within the for loop, compare dish['dish'] with menu_dish['dish'] to determine whether the dish names match. RESULT dish_name_matches]
            [COMMAND If dish_name_matches is false, continue to the next iteration of the for loop. RESULT next_menu_dish]
            [COMMAND Inside the matching-name branch, if menu_dish['count'] is greater than or equal to dish['count'], subtract dish['count'] from menu_dish['count'] and break out of the for loop. RESULT updated_menu_dish]
            [COMMAND After the for loop ends, append dish to self.selected_dishes. RESULT selected_dishes]
            [COMMAND Return True from add_dish. RESULT true_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over self.menu, dish_name_matches is true and menu_dish['count'] is less than dish['count'].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False from add_dish immediately without modifying selected_dishes or continuing the loop. RESULT false_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total cost of selected dishes by summing each dish's price times its count times the corresponding sales multiplier." calculate_total]
    [INPUTS]
        <REF> self </REF>: object "Object exposing selected_dishes, an iterable of dish dictionaries, and sales, a mapping from dish name to a numeric multiplier."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The accumulated total computed as the sum over each dish in self.selected_dishes of dish['price'] * dish['count'] * self.sales[dish['dish']]."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to the constant integer 0. RESULT total]
            [COMMAND For each dish in self.selected_dishes, in iteration order, run the accumulation block. RESULT dish]
            [COMMAND Within the loop over dish, use dish['dish'] as the key to look up the sales multiplier in self.sales. RESULT sales_multiplier]
            [COMMAND Within the loop over dish, compute dish['price'] * dish['count'] * sales_multiplier and add the product to total. RESULT total]
            [COMMAND Return total as the final result. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total price of the selected dishes, clears the selection, and returns the total; if no dishes are selected, returns False without changing the selection." checkout]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric or boolean "Returns the calculated total after clearing selected_dishes, or False if selected_dishes was empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND After confirming selected_dishes is not empty, calculate total = self.calculate_total(). RESULT total]
            [COMMAND Reset self.selected_dishes to a new empty list. RESULT selected_dishes]
            [COMMAND Return total. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When len(self.selected_dishes) == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without clearing selected_dishes. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
