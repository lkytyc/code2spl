[DEFINE_WORKER: "Returns the longest word in a lowercased, punctuation-free tokenized sentence that is present in self.word_list, preserving the first occurrence among equal-length longest words." find_longest_word]
    [INPUTS]
        <REF> sentence </REF>: string "Input sentence to search."
        <REF> self.word_list </REF>: iterable of strings "Allowed words; membership determines which tokens are eligible."
    [END_INPUTS]

    [OUTPUTS]
        <REF> longest_word </REF>: string "Longest eligible word found; empty string if no eligible word exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize longest_word to the empty string. RESULT longest_word]
            [COMMAND Convert sentence to lowercase by assigning sentence = sentence.lower(). RESULT sentence]
            [COMMAND Remove punctuation from sentence by assigning sentence = re.sub('[%s]' % re.escape(string.punctuation), '', sentence). RESULT sentence]
            [COMMAND Split sentence on the literal space character only, assigning sentence = re.split(' ', sentence). RESULT sentence]
            [COMMAND For each word in sentence, if word is in self.word_list and len(word) > len(longest_word), assign longest_word = word (strict greater-than keeps the first word among equal lengths). RESULT longest_word]
            [COMMAND Return longest_word. RESULT longest_word]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
