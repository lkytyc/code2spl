[DEFINE_WORKER: "Advances the playlist selection from the current song to the next song when possible, returning True on success and False when no current song exists or the current song is already the last item." switch_song]
    [INPUTS]
        <REF> self </REF>: object "Object containing current_song and playlist state."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if current_song was advanced to the next playlist item; False if there is no current song or the current song is already the last playlist item."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the outer condition `if self.current_song`; if `self.current_song` is falsy, take the alternative flow that returns False. RESULT outer_condition]
            [COMMAND Inside the outer `if self.current_song` true branch, compute `current_index = self.playlist.index(self.current_song)`. RESULT current_index]
            [COMMAND Inside the outer `if self.current_song` true branch, evaluate `current_index < len(self.playlist) - 1`. RESULT is_not_last]
            [COMMAND When `is_not_last` is true, set `self.current_song = self.playlist[current_index + 1]` and return True. RESULT advance_success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: `self.current_song` is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without changing `self.current_song`. RESULT return_false_no_current]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: `self.current_song` is truthy and `current_index` is not less than `len(self.playlist) - 1`, meaning the current song is already the last playlist entry]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without changing `self.current_song`. RESULT return_false_last]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: `self.current_song` is truthy but is not present in `self.playlist`]
        [LOG "`self.playlist.index(self.current_song)` raises ValueError because the current song cannot be found in the playlist."]
        [THROW ValueError "Raised when the current song is not an element of self.playlist."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Moves the current song to the previous position in the playlist and returns True when a current song exists and is not already first; otherwise returns False without changing the current song." previous_song]
    [INPUTS]
        <REF> self </REF>: object "Object containing a playlist list and a current_song attribute; when current_song is truthy, it is expected to be present in playlist."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if self.current_song was changed to the previous playlist item; False if there is no current song or the current song is already first."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.current_song is truthy and take the outer if branch when it is. RESULT current_song_present]
            [COMMAND Within the truthy self.current_song branch, compute current_index = self.playlist.index(self.current_song). RESULT current_index]
            [COMMAND Evaluate whether current_index > 0 and take the inner if branch when it is. RESULT has_previous_song]
            [COMMAND Within the current_index > 0 branch, assign self.current_song = self.playlist[current_index - 1]. RESULT updated_current_song]
            [COMMAND Return True after assigning the previous song. RESULT return_value_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.current_song is falsy (None, empty, or otherwise false), so the outer else branch is taken.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without changing self.current_song. RESULT return_value_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.current_song is truthy but current_index == 0, so the inner else branch is taken.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the truthy self.current_song branch and the current_index == 0 branch, return False without changing self.current_song. RESULT return_value_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.current_song is truthy but is not present in self.playlist, so self.playlist.index(self.current_song) raises ValueError.]
        [LOG "ValueError: self.current_song is not in self.playlist"]
        [THROW ValueError "self.playlist.index(self.current_song) raises ValueError because the current song is not present in self.playlist."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
