# CLASS BitStatusUtil

[DEFINE_WORKER: "Adds a status bit mask to an existing status bit mask after validating that both inputs are valid, and returns the union of the two masks." add]
    [INPUTS]
        <REF> states </REF>: int "Existing bit status mask to which the status is added."
        <REF> stat </REF>: int "Status bit or bit mask to add to states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> combined_status </REF>: int "Bitwise OR of states and stat, containing every bit set in either input."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate both input bit masks by calling BitStatusUtil.check([states, stat]); if validation fails, the method exits through the exception flow. RESULT validation_result]
            [COMMAND Compute the bitwise OR of states and stat and return the resulting combined bit mask. RESULT combined_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: BitStatusUtil.check([states, stat]) fails because states or stat is negative or otherwise invalid.]
        [LOG "Invalid bit status value: states and stat must be valid non-negative bit masks."]
        [THROW IllegalArgumentException "Thrown by BitStatusUtil.check when an input bit status value is invalid."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns whether all status bits in stat are present in the states bitmask, after validating both inputs." has]
    [INPUTS]
        <REF> states </REF>: int "Current bitmask of enabled status bits."
        <REF> stat </REF>: int "Bitmask of status bits whose presence is being tested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if every bit set in stat is also set in states; false otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check([states, stat]) to validate the two bitmask arguments. RESULT validation_result]
            [COMMAND Return the boolean result of the expression (states & stat) == stat. RESULT has_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a specified bit flag from a bitmask if present, after validating both operands, and returns the updated bitmask; otherwise returns the original bitmask." remove]
    [INPUTS]
        <REF> states </REF>: int "The bitmask from which the flag should be removed."
        <REF> stat </REF>: int "The bit flag to remove from the bitmask."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The updated bitmask if the flag was present, otherwise the original bitmask."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate states and stat by invoking BitStatusUtil.check with a list containing both values. RESULT validation_result]
            [COMMAND Determine whether states contains stat by invoking BitStatusUtil.has(states, stat). RESULT has_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: BitStatusUtil.has(states, stat) is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the bitwise XOR of states and stat to clear the flag, then return that value. RESULT updated_states]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: BitStatusUtil.has(states, stat) is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the original states value without modification. RESULT states]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that every element in the input iterable is a non-negative even integer, raising ValueError for negative or odd elements." check]
    [INPUTS]
        <REF> args </REF>: iterable of integers "Iterable of integer values to validate; each element must be non-negative and even."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each element arg in args, in order. RESULT arg]
            [COMMAND For the current arg, check whether arg is less than 0. RESULT negative_check]
            [COMMAND Within the iteration, if arg is less than 0, raise ValueError with a message containing the formatted arg value followed by ' must be greater than or equal to 0'. RESULT negative_error_raised]
            [COMMAND For the current arg that did not fail the negative check, check whether arg is odd, meaning not divisible by 2. RESULT even_check]
            [COMMAND Within the iteration, if arg is odd, raise ValueError with a message containing the formatted arg value followed by ' not even'. RESULT odd_error_raised]
            [COMMAND After all elements have been processed without raising an exception, finish the loop and return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: An element arg in args is less than 0.]
        [LOG "The thrown ValueError message contains the formatted failing arg value followed by ' must be greater than or equal to 0'."]
        [THROW ValueError "Raised when an element in args is negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: An element arg in args is odd, meaning not divisible by 2.]
        [LOG "The thrown ValueError message contains the formatted failing arg value followed by ' not even'."]
        [THROW ValueError "Raised when an element in args is odd."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
