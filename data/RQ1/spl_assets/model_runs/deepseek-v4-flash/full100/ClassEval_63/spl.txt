# CLASS NLPDataProcessor2

[DEFINE_WORKER: "Converts each string in a list to lowercase, removes non-English and non-whitespace characters, splits the result into words, and returns a list of word lists." process_data]
    [INPUTS]
        <REF> string_list </REF>: list of strings "List of strings to be processed into word lists."
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list of lists of strings "A list where each element is the list of words extracted from the corresponding input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named words_list to collect the word lists. RESULT words_list]
            [COMMAND For each string in string_list, convert the string to lowercase and assign the result to processed_string. RESULT processed_string]
            [COMMAND Within the for loop over string in string_list, replace every character in processed_string that is not an ASCII English letter or whitespace (using regex [^a-zA-Z\s]) with an empty string, and assign the cleaned result to processed_string. RESULT processed_string]
            [COMMAND Within the for loop over string in string_list, split processed_string on whitespace using split() and assign the resulting list to words. RESULT words]
            [COMMAND Within the for loop over string in string_list, append words to words_list. RESULT words_list]
            [COMMAND Return words_list. RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts word frequencies across all word collections in words_list and returns a dictionary of the five most frequent words sorted by descending frequency." calculate_word_frequency]
    [INPUTS]
        <REF> words_list </REF>: Iterable[Iterable[str]] "A collection of word collections; each inner collection supplies the words whose counts are accumulated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> top_5_word_frequency </REF>: Dict[str, int] "A dictionary mapping the five most frequent words to their counts, in descending count order, with fewer than five entries if fewer than five unique words were counted."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty Counter named word_frequency to accumulate counts for every encountered word. RESULT word_frequency]
            [COMMAND For each element words in words_list, update word_frequency using Counter.update(words), adding counts for every word in that element. RESULT word_frequency]
            [COMMAND Build sorted_word_frequency by sorting word_frequency.items() by frequency value in descending order, preserving original insertion order for equal frequencies, and converting the sorted pairs back into a dictionary. RESULT sorted_word_frequency]
            [COMMAND Build top_5_word_frequency by taking the first 5 items from sorted_word_frequency.items(), or all items if fewer than 5 exist, and converting those items into a new dictionary. RESULT top_5_word_frequency]
            [COMMAND Return top_5_word_frequency as the result of the method. RESULT top_5_word_frequency]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes a list of strings into a list of words and then returns a dictionary of word frequencies." process]
    [INPUTS]
        <REF> string_list </REF>: list "List of input strings to be processed into words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> word_frequency_dict </REF>: dict "Dictionary mapping each word to its frequency count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Process the input string_list into a words_list by calling self.process_data(string_list). RESULT words_list]
            [COMMAND Compute word_frequency_dict from words_list by calling self.calculate_word_frequency(words_list). RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict as the final output. RESULT word_frequency_dict]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
