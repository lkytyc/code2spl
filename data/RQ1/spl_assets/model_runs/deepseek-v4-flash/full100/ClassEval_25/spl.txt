# CLASS CookiesUtil

[DEFINE_WORKER: "Initializes the object by storing the cookies file path and setting the initial cookies value to None." __init__]
    [INPUTS]
        <REF> cookies_file </REF>: str "Path to the file from which cookies will be loaded."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.cookies_file to the value of the cookies_file parameter. RESULT self.cookies_file]
            [COMMAND Initialize self.cookies to None. RESULT self.cookies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Stores the value of reponse['cookies'] into self.cookies and then calls self._save_cookies() to persist the cookie state." get_cookies]
    [INPUTS]
        <REF> reponse </REF>: mapping "Object that supports key-based access and is expected to contain the key 'cookies'."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the value associated with the key 'cookies' from the reponse mapping and assign it to self.cookies. RESULT self.cookies]
            [COMMAND Call self._save_cookies() to persist the updated cookie state. RESULT self._save_cookies()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Loads cookies from a JSON file and returns them, or returns an empty dictionary if the file does not exist." load_cookies]
    [INPUTS]
        <REF> self </REF>: object "The object instance containing the cookies_file attribute which specifies the path to the JSON file."
    [END_INPUTS]

    [OUTPUTS]
        <REF> cookies_data </REF>: dict "The parsed JSON data from the cookies file, or an empty dictionary if the file is not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the cookies file specified by self.cookies_file in read mode. RESULT file]
            [COMMAND Load JSON data from the opened file using json.load and assign it to cookies_data. RESULT cookies_data]
            [COMMAND Return cookies_data. RESULT cookies_data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cookies file specified by self.cookies_file is not found, raising FileNotFoundError.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty dictionary as a fallback. RESULT {}]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Saves self.cookies as JSON to self.cookies_file, returning True on success and False if any exception occurs." _save_cookies]
    [INPUTS]
        <REF> self.cookies_file </REF>: str "File path used as the destination for the JSON cookie data."
        <REF> self.cookies </REF>: object "JSON-serializable cookie data to be written to the file."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if the cookies were written successfully; False if an exception occurred."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open self.cookies_file in write mode ('w') using a with statement, truncating the file if it already exists, and bind the opened file object to file. RESULT file]
            [COMMAND Within the with statement, write self.cookies to file as JSON using json.dump with no additional formatting options. RESULT json_dump_done]
            [COMMAND After the with statement completes successfully, return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If a BaseException is raised while opening self.cookies_file or while serializing self.cookies to file, the bare except block is entered.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the raised exception and return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sets the 'cookies' key of the request to a semicolon-separated string of 'key=value' pairs built from self.cookies." set_cookies]
    [INPUTS]
        <REF> self </REF>: object with cookies attribute "The object whose cookies attribute is expected to be a dict-like collection of cookie key-value pairs."
        <REF> request </REF>: mutable dict-like object "The request object that supports item assignment; its 'cookies' key receives the serialized cookie string."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Get the collection of cookie key-value pairs from self.cookies.items(). RESULT cookie_pairs]
            [COMMAND For each key and value in cookie_pairs, format the string f'{key}={value}' and collect all formatted cookie strings in a list. RESULT formatted_cookie_strings]
            [COMMAND Join all formatted cookie strings in formatted_cookie_strings with '; ' to produce a single cookie header string. RESULT cookie_header]
            [COMMAND Assign cookie_header to request['cookies']. RESULT request['cookies']]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
