# CLASS ArgumentParser

[DEFINE_WORKER: "Initializes the instance attributes arguments, required, and types as empty collections." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.arguments to an empty dictionary. RESULT self.arguments]
            [COMMAND Initialize self.required to an empty set. RESULT self.required]
            [COMMAND Initialize self.types to an empty dictionary. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses the tokens after the first token in command_string into long and short options, populates self.arguments with converted values or True flags, and returns whether all required arguments are present." parse_arguments]
    [INPUTS]
        <REF> command_string </REF>: str "Raw command string whose first token is the command name and is ignored; remaining tokens are parsed as options."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple(bool, set|None) "Returns (False, missing_args) if any required argument is missing; otherwise returns (True, None). The method also mutates self.arguments with parsed options."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split command_string on whitespace and, if the resulting list is non-empty, discard its first token; store the remaining tokens as args. RESULT args]
            [COMMAND For each index i from 0 to len(args)-1, assign arg = args[i]. RESULT arg]
            [COMMAND Within the loop over i, if arg starts with '--', strip the leading '--' from arg and split the remainder by '=' to form key_value. RESULT key_value]
            [COMMAND Within the loop over i, if arg starts with '--' and key_value has exactly two elements, set self.arguments[key_value[0]] to self._convert_type(key_value[0], key_value[1]). RESULT self.arguments]
            [COMMAND Within the loop over i, if arg starts with '--' and key_value does not have exactly two elements, set self.arguments[key_value[0]] to True. RESULT self.arguments]
            [COMMAND Within the loop over i, if arg does not start with '--' but starts with '-', set key = arg[1:]. RESULT key]
            [COMMAND Within the loop over i, for a short option, if i + 1 < len(args) (short-circuit) and args[i + 1] does not start with '-', set self.arguments[key] to self._convert_type(key, args[i + 1]). RESULT self.arguments]
            [COMMAND Within the loop over i, for a short option, if the next-token condition is false, set self.arguments[key] to True. RESULT self.arguments]
            [COMMAND Within the loop over i, if arg does not start with '-', perform no action for that token. RESULT None]
            [COMMAND After the loop, compute missing_args = self.required - set(self.arguments.keys()). RESULT missing_args]
            [COMMAND If missing_args is empty, return (True, None). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: missing_args is non-empty after parsing.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return (False, missing_args). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the value associated with the given key from the arguments dictionary, or None if the key is not present." get_argument]
    [INPUTS]
        <REF> key </REF>: object "The key whose associated argument value is to be returned."
    [END_INPUTS]

    [OUTPUTS]
        <REF> value </REF>: object "The value associated with key in the arguments dictionary, or None if key is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call get on the current instance's arguments dictionary with the supplied key and return the resulting value, which is None when the key is absent. RESULT value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers an argument name with an optional required flag and a type, storing the required marker in self.required and the type mapping in self.types." add_argument]
    [INPUTS]
        <REF> arg </REF>: object "The argument name to register."
        <REF> required </REF>: bool "Whether to mark the argument as required; defaults to False."
        <REF> arg_type </REF>: type "The type to associate with the argument; defaults to str."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If required is True, add arg to the required set. RESULT self.required]
            [COMMAND Register arg_type as the type of arg by setting self.types[arg] to arg_type. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the provided value using the type associated with the arg key, preserving the original value if the type lookup fails or conversion raises ValueError." _convert_type]
    [INPUTS]
        <REF> arg </REF>: string "Key used to look up the target type in self.types."
        <REF> value </REF>: any "Value to be converted to the registered type."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The converted value if conversion succeeds; otherwise the original value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Look up self.types[arg] to get the target type for the given argument. RESULT target_type]
            [COMMAND Call target_type(value) to convert the value to the target type. RESULT converted_value]
            [COMMAND Return converted_value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Looking up self.types[arg] raises KeyError, or calling target_type(value) raises ValueError.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the ValueError or KeyError and keep the original value. RESULT exception]
            [COMMAND Return the original value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
