# CLASS DataStatistics

[DEFINE_WORKER: "Computes the arithmetic mean of the numeric values in data and rounds the result to two decimal places." mean]
    [INPUTS]
        <REF> data </REF>: iterable of numeric values "External input iterable of numeric values; must be non-empty or a ZeroDivisionError occurs."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The mean of data rounded to 2 decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sum of all elements in data using sum(data). RESULT total]
            [COMMAND Compute the number of elements in data using len(data). RESULT count]
            [COMMAND Compute the arithmetic mean by dividing total by count. RESULT average]
            [COMMAND Round the average to 2 decimal places using Python's round(average, 2). RESULT rounded_average]
            [COMMAND Return rounded_average as the result of the method. RESULT mean_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the median of an iterable of values by sorting them and returning the middle element for odd length or the rounded average of the two middle elements for even length." median]
    [INPUTS]
        <REF> self </REF>: object "The object instance on which the median method is invoked."
        <REF> data </REF>: iterable "The iterable of values whose median is to be computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float or input element type "The median value. For even-length data, the average of the two middle sorted values rounded to 2 decimal places. For odd-length data, the middle sorted value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new list sorted_data by sorting the input data. RESULT sorted_data]
            [COMMAND Set n to the length of sorted_data. RESULT n]
            [COMMAND Evaluate whether n is even using n % 2 == 0 and select the corresponding median branch. RESULT parity_check]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n % 2 == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the even-length branch, compute middle as n // 2. RESULT middle]
            [COMMAND Within the even-length branch, return round((sorted_data[middle - 1] + sorted_data[middle]) / 2, 2). RESULT median_even]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: n % 2 == 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the odd-length branch, compute middle as n // 2. RESULT middle]
            [COMMAND Within the odd-length branch, return sorted_data[middle]. RESULT median_odd]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts occurrences of each element in the input iterable and returns all elements with the maximum frequency." mode]
    [INPUTS]
        <REF> data </REF>: iterable "Input iterable whose mode values are to be computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> mode </REF>: list "List of all distinct elements that occur with the maximum frequency."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the frequency of every element in data using collections.Counter. RESULT counter]
            [COMMAND Compute the maximum frequency by taking the maximum of counter.values(). RESULT mode_count]
            [COMMAND Build the list of elements x for each (x, count) in counter.items() where count equals mode_count. RESULT mode]
            [COMMAND Return mode as the result of the method. RESULT mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is empty, so counter has no values]
        [LOG "max() is called on an empty counter.values(), which raises ValueError"]
        [THROW ValueError "Cannot compute the mode of an empty iterable because there is no maximum frequency."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
