# CLASS TextFileProcessor

[DEFINE_WORKER: "Initializes a new instance by storing the provided file_path argument in the instance attribute self.file_path." __init__]
    [INPUTS]
        <REF> file_path </REF>: string "The file path to be stored on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the file_path parameter to the instance attribute self.file_path. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens the file specified by self.file_path, parses its contents as JSON, and returns the parsed data." read_file_as_json]
    [INPUTS]
        <REF> self </REF>: object "The instance whose file_path attribute is used to locate the JSON file."
    [END_INPUTS]

    [OUTPUTS]
        <REF> data </REF>: object "The parsed JSON content from the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.file_path in read mode using a with statement and assign the file object to file. RESULT file]
            [COMMAND Within the with block, parse the contents of file as JSON using json.load and assign the parsed data to data. RESULT data]
            [COMMAND After the with block exits, closing file automatically, return data. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads the entire content of the file located at self.file_path and returns it as a string." read_file]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> file_contents </REF>: str "The full text content of the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.file_path in read mode using a context manager so it is closed automatically. RESULT file]
            [COMMAND Read the entire contents from the opened file and return them. RESULT file_contents]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the provided content to the file located at self.file_path by opening it in write mode and using file.write." write_file]
    [INPUTS]
        <REF> content </REF>: str "The text content to write to the file."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.file_path with mode 'w' using a with statement and assign the file object to file. RESULT file]
            [COMMAND Within the with block, write content to the opened file by calling file.write(content). RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a file, removes every non-alphabetic character from its content, writes the filtered content back to the file, and returns the filtered string." process_file]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> content </REF>: str "The filtered alphabetic content that was written to the file and returned by the method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.read_file() and assign the returned raw file content to the variable content. RESULT content]
            [COMMAND Build a new string from the characters in content for which char.isalpha() returns True, concatenated in their original order, and assign that filtered string back to content. RESULT content]
            [COMMAND Call self.write_file(content) to write the filtered content to the file, discarding any return value. RESULT void]
            [COMMAND Return the filtered content string. RESULT content]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
