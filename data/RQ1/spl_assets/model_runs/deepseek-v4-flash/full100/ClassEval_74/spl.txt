[DEFINE_WORKER: "Adds an address to the whitelist only if it is not already present, returning False for duplicates and otherwise returning the updated whitelist." add_white_list]
    [INPUTS]
        <REF> addr </REF>: unknown "The address value to be checked and added to the whitelist."
        <REF> self.white_list </REF>: list "The existing whitelist that may be mutated by appending addr."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or list "Returns False if addr is already in white_list; otherwise returns the updated white_list after appending addr."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is contained in self.white_list. RESULT is_whitelisted]
            [COMMAND If is_whitelisted is False, append addr to self.white_list. RESULT self.white_list]
            [COMMAND Return the updated self.white_list. RESULT self.white_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_whitelisted is True (addr is already present in self.white_list)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a given address from the white list only when it is present, returning False when the address is absent and returning the updated white list when removal occurs." del_white_list]
    [INPUTS]
        <REF> addr </REF>: object "The address to look for in self.white_list and remove if it is present."
        <REF> white_list </REF>: container "The current white list stored on self; it is mutated by removing addr only when addr is present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean or container "Returns False if addr is not in self.white_list; otherwise returns self.white_list after removing addr."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether addr is absent from self.white_list using the membership test addr not in self.white_list. RESULT absent]
            [COMMAND If absent is True, return False immediately without modifying self.white_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: absent is False (addr is in self.white_list)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.white_list.remove(addr) to remove addr from self.white_list. RESULT self.white_list]
            [COMMAND Return the mutated self.white_list after the removal. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates incoming info, enforces the address whitelist, stores accepted content in self.receive_struct, and returns content or a rejection value." recv]
    [INPUTS]
        <REF> info </REF>: dict "Incoming message data; expected to be a dict with keys 'addr' and 'content'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "Returns -1 if info is invalid, False if addr is not whitelisted, otherwise returns self.receive_struct['content']."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate that info is a dict and contains both keys 'addr' and 'content'; if validation fails, follow the invalid_info alternative flow. RESULT validation_result]
            [COMMAND Assign addr = info['addr'] and content = info['content']. RESULT addr_content]
            [COMMAND Check whether addr is in self.white_list; if it is not, follow the non_whitelisted alternative flow. RESULT whitelist_check]
            [COMMAND If addr is in self.white_list, set self.receive_struct = {'addr': addr, 'content': content}. RESULT self.receive_struct]
            [COMMAND Within that same else branch, return self.receive_struct['content']. RESULT returned_content]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dict, or 'addr' is not in info, or 'content' is not in info]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1. RESULT invalid_info_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: addr is not in self.white_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT non_whitelisted_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that info is a dictionary containing both 'addr' and 'content', returns an error string if not, otherwise stores the corresponding values in self.send_struct." send]
    [INPUTS]
        <REF> info </REF>: object "Expected to be a dictionary with keys 'addr' and 'content'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> method_result </REF>: string or None "Returns 'info structure is not correct' if validation fails; otherwise returns None after assigning self.send_struct."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the short-circuit validation: check whether info is not a dict, then whether 'addr' is absent from info, then whether 'content' is absent from info; the result is true only for the invalid branch and false when all checks pass. RESULT validation_failed]
            [COMMAND Because validation_failed is false, read info['addr'] and info['content'] and build a new dictionary literal with keys 'addr' and 'content'. RESULT new_send_struct]
            [COMMAND Assign new_send_struct to the object attribute self.send_struct. RESULT self.send_struct]
            [COMMAND Return None implicitly after the assignment. RESULT method_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dictionary, or info lacks an 'addr' key, or info lacks a 'content' key, evaluated with short-circuit logic.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the string 'info structure is not correct' as the method result. RESULT method_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Dispatches on the type parameter to return the corresponding send or receive struct, or False when the type is unrecognized." show]
    [INPUTS]
        <REF> self </REF>: object "The object instance providing send_struct and receive_struct."
        <REF> type </REF>: string "The selector value compared against 'send' and 'receive'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: send_struct | receive_struct | boolean "The value returned by the method: self.send_struct, self.receive_struct, or False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If type equals "send", return self.send_struct and exit the method immediately. RESULT return_value]
            [COMMAND Otherwise, if type equals "receive", return self.receive_struct and exit the method immediately. RESULT return_value]
            [COMMAND Otherwise, return False and exit the method immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
