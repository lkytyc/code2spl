# CLASS VectorUtil

[DEFINE_WORKER: "Computes the cosine similarity between two vectors by normalizing each input to a unit vector and returning their dot product." similarity]
    [INPUTS]
        <REF> vector_1 </REF>: vector "First input vector."
        <REF> vector_2 </REF>: vector "Second input vector."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: scalar "Dot product of the unit-normalized input vectors, representing their cosine similarity."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Normalize vector_1 to a unit vector using matutils.unitvec. RESULT unit_vector_1]
            [COMMAND Normalize vector_2 to a unit vector using matutils.unitvec. RESULT unit_vector_2]
            [COMMAND Compute the dot product of unit_vector_1 and unit_vector_2. RESULT similarity_score]
            [COMMAND Return similarity_score. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the cosine similarity between a single vector and each vector in a matrix by dividing their dot products by the product of their Euclidean norms." cosine_similarities]
    [INPUTS]
        <REF> vector_1 </REF>: 1-D numeric array "Reference vector whose cosine similarity to each row of vectors_all is computed."
        <REF> vectors_all </REF>: 2-D numeric array "Matrix of vectors; each row is compared with vector_1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> similarities </REF>: 1-D numeric array "Cosine similarity of vector_1 with each row of vectors_all."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the Euclidean norm of vector_1 as a scalar. RESULT norm]
            [COMMAND Compute the Euclidean norm of each row in vectors_all along axis 1 to produce an array of norms. RESULT all_norms]
            [COMMAND Compute the dot product of vectors_all with vector_1 to produce an array of dot products. RESULT dot_products]
            [COMMAND Divide dot_products elementwise by the product of norm and all_norms to compute similarities; if norm or any all_norms element is zero, division by zero occurs. RESULT similarities]
            [COMMAND Return similarities. RESULT similarities]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the cosine similarity between the mean vectors of two input lists by normalizing each mean vector and taking the dot product; raises ZeroDivisionError if either list is empty." n_similarity]
    [INPUTS]
        <REF> vector_list_1 </REF>: list of numeric vectors "First collection of vectors used to compute a mean vector."
        <REF> vector_list_2 </REF>: list of numeric vectors "Second collection of vectors used to compute a mean vector."
    [END_INPUTS]

    [OUTPUTS]
        <REF> similarity_score </REF>: float "Cosine similarity between the mean vectors, in the range [-1, 1]."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert vector_list_1 to a NumPy array. RESULT array_1]
            [COMMAND Compute the element-wise mean of array_1 along axis 0. RESULT centroid_1]
            [COMMAND Normalize centroid_1 to unit length using unitvec. RESULT unit_vector_1]
            [COMMAND Convert vector_list_2 to a NumPy array. RESULT array_2]
            [COMMAND Compute the element-wise mean of array_2 along axis 0. RESULT centroid_2]
            [COMMAND Normalize centroid_2 to unit length using unitvec. RESULT unit_vector_2]
            [COMMAND Compute the dot product of unit_vector_1 and unit_vector_2. RESULT similarity_score]
            [COMMAND Return similarity_score as the function result. RESULT similarity_score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Either vector_list_1 or vector_list_2 is empty; the condition not (len(vector_list_1) and len(vector_list_2)) is true.]
        [LOG "At least one of the passed list is empty."]
        [THROW ZeroDivisionError "Raised to prevent division by zero when an input list has no elements."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes a dictionary mapping original keys to smoothed logarithmic inverse document frequency (IDF) weights from their occurrence counts and the total number of documents." compute_idf_weight_dict]
    [INPUTS]
        <REF> total_num </REF>: scalar "Total number of documents, used for smoothing in the IDF formula."
        <REF> number_dict </REF>: dict "Dictionary mapping original keys to their occurrence counts."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "Dictionary mapping original keys to their smoothed IDF weights."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize index_2_key_map as an empty dict, index as 0, and count_list as an empty list. RESULT index_2_key_map, index, count_list]
            [COMMAND Iterate over number_dict.items(); for each key, count, store the current key in index_2_key_map at the current index, append count to count_list, and increment index by 1. RESULT index_2_key_map, count_list, index]
            [COMMAND Convert count_list into a NumPy array and assign it to a. RESULT a]
            [COMMAND Apply the smoothed IDF formula by replacing a with np.log((total_num + 1) / (a + 1)). RESULT a]
            [COMMAND Initialize result as an empty dict. RESULT result]
            [COMMAND Iterate over enumerate(a); for each index, w, retrieve the original key using index_2_key_map[index] and assign result[key] = w. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
