# CLASS JSONProcessor

[DEFINE_WORKER: "Reads and parses a JSON file from file_path, returning the parsed data on success, 0 if the file does not exist, and -1 if an exception occurs during access or parsing." read_json]
    [INPUTS]
        <REF> file_path </REF>: str "Path to the JSON file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object|int "Parsed JSON data on success; 0 if file_path does not exist; -1 if an exception occurs while opening or parsing the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether file_path exists by calling os.path.exists(file_path). RESULT path_exists]
            [COMMAND If path_exists is True, inside a try block, open file_path in read mode using with open(file_path, 'r') as file. RESULT file]
            [COMMAND Inside the same try block and within the with block, parse the JSON content from file by calling json.load(file). RESULT data]
            [COMMAND Return data. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: path_exists is False]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the integer 0. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: An exception is raised while opening file_path or parsing its JSON content.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the integer -1. RESULT -1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes JSON-serializable data to a file and returns 1 on success or -1 if any exception occurs." write_json]
    [INPUTS]
        <REF> data </REF>: json-serializable object "The data to be serialized and written to the file."
        <REF> file_path </REF>: string "Path to the file where the JSON data will be written."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Returns 1 on successful write, or -1 if an exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at file_path in write mode using a with statement. RESULT file]
            [COMMAND Serialize data to JSON and write it to the file using json.dump. RESULT write_result]
            [COMMAND Return 1 to indicate successful write. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception is raised while opening the file, during the with-context execution, or while serializing data with json.dump.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 to indicate failure. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a JSON file, returns 0 if the read result is 0 or -1, otherwise deletes the specified key from the parsed data when present, writes the updated data back to the file, and returns 1 on success or 0 if the key is absent." process_json]
    [INPUTS]
        <REF> file_path </REF>: string "Path to the JSON file to read and potentially rewrite."
        <REF> remove_key </REF>: string "Key to search for and remove from the parsed JSON data."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Returns 1 if the key was removed and the file was written; returns 0 if data was 0 or -1, or if the key was not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.read_json(file_path) and assign the returned parsed data to data. RESULT data]
            [COMMAND Verify that data is not 0 and is not -1; if it is 0 or -1, use the corresponding alternative flow. RESULT data_guard]
            [COMMAND Verify that remove_key is present in data; if it is not, use the corresponding alternative flow. RESULT key_present]
            [COMMAND Delete data[remove_key] from data. RESULT data]
            [COMMAND Call self.write_json(data, file_path) to persist the updated data. RESULT write_result]
            [COMMAND Return 1 to indicate that the key was removed and the data was persisted. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: data equals 0 or data equals -1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without modifying the file. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: data is neither 0 nor -1, and remove_key is not present in data]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without modifying the file. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
