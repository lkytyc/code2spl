[DEFINE_WORKER: "Computes the Pearson correlation coefficient between two equal-length numeric datasets, returning 0 when the denominator is zero." correlation_coefficient]
    [INPUTS]
        <REF> data1 </REF>: list of numbers "First numeric dataset."
        <REF> data2 </REF>: list of numbers "Second numeric dataset with the same length as data1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> correlation_coefficient </REF>: number "The Pearson correlation coefficient, or 0 when the denominator is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set n to the length of data1. RESULT n]
            [COMMAND Compute mean1 as the sum of data1 divided by n. RESULT mean1]
            [COMMAND Compute mean2 as the sum of data2 divided by n. RESULT mean2]
            [COMMAND For i from 0 to n-1, accumulate the product of (data1[i] - mean1) and (data2[i] - mean2), and set numerator to the accumulated sum. RESULT numerator]
            [COMMAND For i from 0 to n-1, accumulate the square of (data1[i] - mean1), then multiply its square root by the square root of the analogous sum of squares for data2; set denominator to that product. RESULT denominator]
            [COMMAND If denominator is not equal to 0, return the result of numerator divided by denominator. RESULT correlation_coefficient]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: denominator is equal to 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT correlation_coefficient]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the sample skewness of a numeric dataset, returning 0 when the standard deviation is zero." skewness]
    [INPUTS]
        <REF> data </REF>: list "Numeric data values used to compute skewness."
    [END_INPUTS]

    [OUTPUTS]
        <REF> skewness </REF>: float "The computed skewness value, or 0 if the standard deviation is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the number of elements in data. RESULT n]
            [COMMAND Compute mean as the sum of data divided by n. RESULT mean]
            [COMMAND Compute variance as the sum over each x in data of (x - mean) squared, divided by n. RESULT variance]
            [COMMAND Compute std_deviation as the square root of variance using math.sqrt. RESULT std_deviation]
            [COMMAND Compute skewness as follows: if std_deviation is not equal to 0, evaluate sum((x - mean) ** 3 for x in data) * n / ((n - 1) * (n - 2) * std_deviation ** 3); otherwise assign skewness to 0. RESULT skewness]
            [COMMAND Return skewness. RESULT skewness]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the population excess kurtosis of a numeric dataset by subtracting 3 from the normalized fourth central moment, returning math.nan if the population standard deviation is zero." kurtosis]
    [INPUTS]
        <REF> data </REF>: iterable "Non-empty iterable of numeric values representing the dataset."
    [END_INPUTS]

    [OUTPUTS]
        <REF> kurtosis_value </REF>: float "Excess kurtosis of the input data, or math.nan when the population standard deviation is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set n to the number of elements in data. RESULT n]
            [COMMAND Set mean to the sum of data divided by n. RESULT mean]
            [COMMAND Set std_dev to the square root of the sum of the squared differences between each data element and mean, divided by n. RESULT std_dev]
            [COMMAND Create centered_data as a list in which each element is the corresponding data element minus mean. RESULT centered_data]
            [COMMAND Set fourth_moment to the sum of each centered_data element raised to the fourth power, divided by n. RESULT fourth_moment]
            [COMMAND Set kurtosis_value to fourth_moment divided by std_dev raised to the fourth power, minus 3. RESULT kurtosis_value]
            [COMMAND Return kurtosis_value. RESULT kurtosis_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After computing std_dev, if std_dev is exactly equal to 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return math.nan. RESULT math.nan]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the probability density function of a normal distribution with mean mu and standard deviation sigma for each element in data." pdf]
    [INPUTS]
        <REF> data </REF>: iterable of numeric values "The input values at which to evaluate the normal PDF."
        <REF> mu </REF>: numeric "The mean of the normal distribution."
        <REF> sigma </REF>: numeric "The standard deviation of the normal distribution; must be nonzero."
    [END_INPUTS]

    [OUTPUTS]
        <REF> pdf_values </REF>: list of numeric values "The computed normal density values for each element in data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND In a list comprehension over data, for each x in data compute 1 / (sigma * math.sqrt(2 * math.pi)) * math.exp(-0.5 * ((x - mu) / sigma) ** 2) and collect each result in pdf_values. RESULT pdf_values]
            [COMMAND Return pdf_values. RESULT pdf_values]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
