# CLASS PushBoxGame

[DEFINE_WORKER: "Initializes the game state by storing the given map, resetting the player position, targets, boxes, target count, and game-over flag, and then invoking init_game() to perform any remaining setup." __init__]
    [INPUTS]
        <REF> map </REF>: object "The game map object to be stored as self.map."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the incoming map reference in self.map. RESULT self.map]
            [COMMAND Set self.player_row to 0. RESULT self.player_row]
            [COMMAND Set self.player_col to 0. RESULT self.player_col]
            [COMMAND Initialize self.targets as an empty list. RESULT self.targets]
            [COMMAND Initialize self.boxes as an empty list. RESULT self.boxes]
            [COMMAND Set self.target_count to 0. RESULT self.target_count]
            [COMMAND Set self.is_game_over to False. RESULT self.is_game_over]
            [COMMAND Call self.init_game() to perform additional game initialization. RESULT self]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the 2D game map in row-major order and initializes the player position, target list and target count, and box list from cells marked O, G, and X respectively." init_game]
    [INPUTS]
        <REF> self </REF>: object "Game instance containing self.map as a list of strings or lists; the attributes self.targets, self.target_count, and self.boxes must already exist before this method is called."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row index row from 0 through len(self.map)-1. RESULT row_index]
            [COMMAND Within the outer row loop, iterate over each column index col from 0 through len(self.map[row])-1. RESULT col_index]
            [COMMAND Within the nested row and column loops, inspect the current cell self.map[row][col] and branch on its character value. RESULT current_cell]
            [COMMAND Within the nested row and column loops, if self.map[row][col] == "O", set self.player_row = row and self.player_col = col, overwriting any previously stored player position. RESULT player_position]
            [COMMAND Within the nested row and column loops, else if self.map[row][col] == "G", append the tuple (row, col) to self.targets and increment self.target_count by 1. RESULT target_recorded]
            [COMMAND Within the nested row and column loops, else if self.map[row][col] == "X", append the tuple (row, col) to self.boxes. RESULT box_recorded]
            [COMMAND Within the nested row and column loops, if none of the previous conditions matched, ignore the current cell. RESULT ignored_cell]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts how many boxes are on target cells and, if the count equals the total target count, sets the game-over flag; returns the game-over flag." check_win]
    [INPUTS]
        <REF> self </REF>: object "Game state containing self.boxes, self.targets, self.target_count, and self.is_game_over."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self.is_game_over </REF>: bool "True when all required boxes are on targets, or the previous game-over state otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize box_on_target_count to 0. RESULT box_on_target_count]
            [COMMAND For each occurrence of box in self.boxes, if box is contained in self.targets, increment box_on_target_count by 1. RESULT box_on_target_count]
            [COMMAND If box_on_target_count equals self.target_count, set self.is_game_over to True. RESULT self.is_game_over]
            [COMMAND Return the current value of self.is_game_over. RESULT self.is_game_over]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to move the player one step in the requested direction, pushes a box when possible, and returns the win-check result." move]
    [INPUTS]
        <REF> direction </REF>: string "Movement direction: 'w' (up), 's' (down), 'a' (left), or 'd' (right)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> check_win_result </REF>: bool "Result returned by self.check_win() after the movement attempt."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set new_player_row to self.player_row. RESULT new_player_row]
            [COMMAND Set new_player_col to self.player_col. RESULT new_player_col]
            [COMMAND If direction equals 'w', decrement new_player_row by 1. RESULT new_player_row]
            [COMMAND Else if direction equals 's', increment new_player_row by 1. RESULT new_player_row]
            [COMMAND Else if direction equals 'a', decrement new_player_col by 1. RESULT new_player_col]
            [COMMAND Else if direction equals 'd', increment new_player_col by 1. RESULT new_player_col]
            [COMMAND If self.map[new_player_row][new_player_col] equals '#', make no state changes and skip to the final return. RESULT state_unchanged]
            [COMMAND If self.map[new_player_row][new_player_col] is not '#', continue to the movement and box-push logic. RESULT target_not_wall]
            [COMMAND If (new_player_row, new_player_col) is in self.boxes, set new_box_row to new_player_row plus (new_player_row minus self.player_row). RESULT new_box_row]
            [COMMAND Within the same box-present branch, set new_box_col to new_player_col plus (new_player_col minus self.player_col). RESULT new_box_col]
            [COMMAND Within the box-present branch, if self.map[new_box_row][new_box_col] equals '#', make no box or player position changes. RESULT state_unchanged]
            [COMMAND Within the box-present branch, if self.map[new_box_row][new_box_col] is not '#', remove (new_player_row, new_player_col) from self.boxes, append (new_box_row, new_box_col) to self.boxes, set self.player_row to new_player_row, and set self.player_col to new_player_col. RESULT box_and_player_updated]
            [COMMAND If (new_player_row, new_player_col) is not in self.boxes, set self.player_row to new_player_row and self.player_col to new_player_col. RESULT player_position_updated]
            [COMMAND Return self.check_win(). RESULT check_win_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
