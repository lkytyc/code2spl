# CLASS Classroom

[DEFINE_WORKER: "Initializes a new instance by storing the provided id and creating an empty courses list." __init__]
    [INPUTS]
        <REF> id </REF>: any "Identifier value assigned to the instance attribute self.id."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of parameter id to the instance attribute self.id. RESULT self.id]
            [COMMAND Create a new empty list and assign it to the instance attribute self.courses, ensuring each instance gets its own independent list. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a course to the courses list only if it is not already present, preventing duplicate entries." add_course]
    [INPUTS]
        <REF> course </REF>: unknown "The course object to be added to self.courses."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether course is not already present in self.courses. RESULT is_absent]
            [COMMAND If is_absent is true, append course to self.courses. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the first occurrence of the specified course from self.courses if it is present; otherwise, it makes no changes." remove_course]
    [INPUTS]
        <REF> course </REF>: object "The course value or object to remove from self.courses."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether course is present in self.courses. RESULT course_present]
            [COMMAND If course_present is True, remove the first occurrence of course from self.courses. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns False if the given HH:MM time falls within any course's inclusive start-to-end interval, otherwise returns True." is_free_at]
    [INPUTS]
        <REF> check_time </REF>: string "Time string in HH:MM format representing the time to check for availability."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if check_time is not inside any course interval; False if check_time is inside at least one course interval, with boundaries inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse check_time from a string in HH:MM format into a datetime object using datetime.strptime, and assign the parsed datetime back to check_time. RESULT check_time]
            [COMMAND Iterate over each course in self.courses in collection order. RESULT course]
            [COMMAND Within the for loop over course, parse course['start_time'] and course['end_time'] with datetime.strptime using '%H:%M', and evaluate whether parsed_start_time <= check_time <= parsed_end_time. RESULT overlap_detected]
            [COMMAND After the for loop completes without overlap_detected being true for any course, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over course, overlap_detected is true, meaning check_time falls inside the course interval including both start and end boundaries.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, exiting the method and skipping all remaining courses. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: During the initial parsing step, check_time is not a string in HH:MM format.]
        [LOG "datetime.strptime raised a ValueError because check_time does not match '%H:%M'."]
        [THROW ValueError "Raised when the input time string is not a valid HH:MM time."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: During the loop, a course's start_time or end_time is not a string in HH:MM format.]
        [LOG "datetime.strptime raised a ValueError because a course time does not match '%H:%M'."]
        [THROW ValueError "Raised when a course time string is not a valid HH:MM time."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether either endpoint of a proposed course time interval falls inside any existing course interval, treating touching endpoints as conflicts." check_course_conflict]
    [INPUTS]
        <REF> self </REF>: object with courses iterable "The object containing 'courses', an iterable of existing course dictionaries with 'start_time' and 'end_time' HH:MM strings."
        <REF> new_course </REF>: dict "The proposed course with 'start_time' and 'end_time' HH:MM strings to be checked for conflicts."
    [END_INPUTS]

    [OUTPUTS]
        <REF> flag </REF>: bool "True if neither new_start_time nor new_end_time lies within any existing course interval under inclusive boundaries; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse new_course['start_time'] using datetime.strptime with format '%H:%M', defaulting the date to 1900-01-01, and assign the resulting datetime object to new_start_time. RESULT new_start_time]
            [COMMAND Parse new_course['end_time'] using datetime.strptime with format '%H:%M', defaulting the date to 1900-01-01, and assign the resulting datetime object to new_end_time. RESULT new_end_time]
            [COMMAND Initialize flag to True. RESULT flag]
            [COMMAND Iterate over each course in self.courses. RESULT course]
            [COMMAND Within the loop over self.courses, parse course['start_time'] using datetime.strptime with format '%H:%M', defaulting the date to 1900-01-01, and assign the resulting datetime object to start_time. RESULT start_time]
            [COMMAND Within the loop over self.courses, parse course['end_time'] using datetime.strptime with format '%H:%M', defaulting the date to 1900-01-01, and assign the resulting datetime object to end_time. RESULT end_time]
            [COMMAND Within the loop over self.courses, if start_time <= new_start_time and end_time >= new_start_time, set flag to False. RESULT flag]
            [COMMAND Within the loop over self.courses, if start_time <= new_end_time and end_time >= new_end_time, set flag to False. RESULT flag]
            [COMMAND Return the final value of flag. RESULT returned_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
