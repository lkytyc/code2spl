# CLASS Warehouse

[DEFINE_WORKER: "Initializes a new instance by setting self.inventory and self.orders to independent empty dictionaries." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance being initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.inventory as an empty dictionary mapping product IDs to Product objects, and initialize self.orders as a separate empty dictionary. RESULT self]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a new product to the inventory with the given name and quantity, or increases the quantity of an existing product by the given amount." add_product]
    [INPUTS]
        <REF> product_id </REF>: string or hashable identifier "Identifier of the product to add or update."
        <REF> name </REF>: string "Name of the product, used only when creating a new inventory entry."
        <REF> quantity </REF>: number "Quantity to set for a new product or add to an existing product's quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is not present as a key in self.inventory. RESULT is_new_product]
            [COMMAND If is_new_product is true, set self.inventory[product_id] to {'name': name, 'quantity': quantity}. RESULT self.inventory[product_id]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_new_product is false, meaning product_id already exists in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Add quantity to the existing value stored in self.inventory[product_id]['quantity']. RESULT self.inventory[product_id]['quantity']]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the quantity of an existing product in the inventory by adding the specified quantity to the product's stored quantity." update_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: object "Identifier of the product whose quantity is to be updated."
        <REF> quantity </REF>: number "Value to add to the product's current quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT self.inventory]
            [COMMAND If product_id is present in self.inventory, increment self.inventory[product_id]['quantity'] by quantity. RESULT updated_inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: product_id is not in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do not modify self.inventory and continue execution without taking any action. RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the quantity for a given product_id from self.inventory, or False when the product_id is not present." get_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: hashable "Identifier of the product to look up in self.inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or boolean "The quantity stored at self.inventory[product_id]['quantity'] if product_id exists, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is present as a key in self.inventory. RESULT is_present]
            [COMMAND If is_present is true, access self.inventory[product_id]['quantity'] and return that value. RESULT quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_present is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT false_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates an order and marks it as Shipped when sufficient product quantity is available; otherwise returns False without changing stock." create_order]
    [INPUTS]
        <REF> order_id </REF>: string "Identifier for the order, used as the key in self.orders."
        <REF> product_id </REF>: string "Identifier for the product whose stock is checked and updated."
        <REF> quantity </REF>: integer "Number of product units requested for the order."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None or False "Returns None when the order is successfully created; returns False when stock is insufficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_product_quantity(product_id) and compare the result with quantity using >= (inclusive boundary) to determine whether available stock is sufficient. RESULT stock_sufficient]
            [COMMAND After confirming sufficient stock, call self.update_product_quantity(product_id, -quantity) to reduce the product quantity by the requested quantity. RESULT stock_updated]
            [COMMAND Within the sufficient-stock branch, store a new order dictionary in self.orders[order_id] with keys 'product_id', 'quantity', and 'status' set to product_id, quantity, and 'Shipped'. RESULT created_order]
            [COMMAND After the success branch completes, return None implicitly because no explicit return exists in that branch. RESULT implicit_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.get_product_quantity(product_id) is less than quantity, meaning insufficient stock is available.]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the else branch, return False immediately without updating stock or creating an order. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the status of an existing order if the given order_id is found in self.orders, otherwise returns False." change_order_status]
    [INPUTS]
        <REF> order_id </REF>: hashable key "Identifier used to look up the order in self.orders."
        <REF> status </REF>: any "New status value to assign to the matched order."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None or False "Returns None after a successful status update; returns False if order_id is not present in self.orders."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT order_exists]
            [COMMAND When order_id exists in self.orders, assign status to self.orders[order_id]['status']. RESULT self.orders[order_id]['status']]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_id is not in self.orders]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately to indicate that the order was not found. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Retrieves the status of an order by order_id, or returns False when no matching order exists in self.orders." track_order]
    [INPUTS]
        <REF> order_id </REF>: hashable key (string or identifier) "Identifier used to look up an order in the self.orders dictionary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> status </REF>: value stored in order record under 'status' "The 'status' value for the order when order_id is found in self.orders."
        <REF> False </REF>: boolean "Returned when order_id does not exist in self.orders."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id is a key in the self.orders dictionary RESULT order_exists]
            [COMMAND If order_exists is true, retrieve and return self.orders[order_id]['status'] RESULT status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If order_exists is false, meaning order_id is not found in self.orders]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False as the sentinel value for an unknown order RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
