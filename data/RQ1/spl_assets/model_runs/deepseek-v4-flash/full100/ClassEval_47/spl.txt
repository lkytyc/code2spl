[DEFINE_WORKER: "Validates that self.ip_address is a valid IPv4 address with exactly four dot-separated octets, each being a digit string whose integer value is between 0 and 255 inclusive." is_valid]
    [INPUTS]
        <REF> self.ip_address </REF>: str "The IP address string to be validated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> is_valid </REF>: bool "True if self.ip_address has exactly four octets and each octet is numeric and in the inclusive range 0 to 255; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address on the '.' character and assign the resulting list to octets. RESULT octets]
            [COMMAND Check whether the length of octets is exactly 4; if not, take the length_mismatch alternative flow. RESULT length_ok]
            [COMMAND For each octet in octets, use short-circuit evaluation to check if octet.isdigit() is False, or int(octet) is less than 0, or int(octet) is greater than 255; if any of those is true, take the invalid_octet alternative flow. RESULT all_octets_valid]
            [COMMAND Return True because the address has exactly four octets and every octet passes the inclusive 0 to 255 numeric range validation. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The list octets does not contain exactly 4 elements.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately because the IP address does not have exactly four dot-separated octets. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over octets, for the current octet: octet.isdigit() is False, or int(octet) is less than 0, or int(octet) is greater than 255.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately because the current octet is not a valid digit string or its integer value is outside the inclusive range 0 to 255. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the IP address is valid and, if so, splits it into octet strings; otherwise returns an empty list." get_octets]
    [INPUTS]
        <REF> self </REF>: object "Instance providing ip_address as a string and is_valid() to validate it."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of octet strings from self.ip_address when valid, or empty list when invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate self.is_valid() and store its boolean result. RESULT validity_result]
            [COMMAND If validity_result is true, split self.ip_address on '.' and return the resulting list of octets. RESULT octets]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If validity_result is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list as the method result. RESULT empty_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a dotted binary string representation of the object's octets when the object is valid, and returns an empty string otherwise." get_binary]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Dotted binary representation of all octets when the object is valid; an empty string when the object is not valid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate self.is_valid(). If it is false, follow the invalid-case alternative flow; otherwise continue with the valid-case flow. RESULT is_valid]
            [COMMAND In the valid-case flow, initialize an empty list named binary_octets. RESULT binary_octets]
            [COMMAND In the valid-case flow, iterate over each octet returned by self.get_octets(). RESULT octet]
            [COMMAND Within the for loop over octet in self.get_octets(), convert the current octet to an integer and format it as an 8-bit binary string using format(int(octet), '08b'). RESULT binary_string]
            [COMMAND Within the for loop over octet in self.get_octets(), append binary_string to binary_octets. RESULT binary_octets]
            [COMMAND After the for loop completes, join the elements of binary_octets with '.' as the delimiter. RESULT joined_binary]
            [COMMAND Return joined_binary. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() evaluates to false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string ''. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
