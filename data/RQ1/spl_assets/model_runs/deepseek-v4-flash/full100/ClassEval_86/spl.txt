# CLASS TicTacToe

[DEFINE_WORKER: "Initializes the Tic-Tac-Toe board as a 3 by N grid of space characters and sets the starting player to X." __init__]
    [INPUTS]
        <REF> N </REF>: int "Number of columns for the board; defaults to 3."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Bind the optional parameter N, using the default value 3 when N is not supplied. RESULT N]
            [COMMAND Create a new list of exactly 3 rows, where each row is an independent list of N space characters, and assign it to self.board. RESULT self.board]
            [COMMAND Set self.current_player to the string 'X'. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to place the current player's mark in an empty cell of the board, switches the active player, and returns True on success; returns False if the cell is already occupied." make_move]
    [INPUTS]
        <REF> row </REF>: int "Row index of the target board cell."
        <REF> col </REF>: int "Column index of the target board cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was made and the current player was switched; False if the cell was not empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether the board cell at row and col contains a single space character, meaning it is empty. RESULT cell_is_empty]
            [COMMAND If cell_is_empty is true, set the board cell at row and col to the current player's mark. RESULT board_updated]
            [COMMAND If cell_is_empty is true, toggle current_player to O if it is currently X, and to X otherwise. RESULT current_player_updated]
            [COMMAND If cell_is_empty is true, return True to indicate that the move was successful. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: cell_is_empty is false, meaning the board cell at row and col is not empty.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that the move was rejected because the cell is already occupied. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the Tic-Tac-Toe board for a winning line in rows, columns, or diagonals and returns the winning symbol, or None if no winner exists." check_winner]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "The winning player's symbol if a row, column, or diagonal contains three identical non-blank values; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each row in self.board, check whether row[0], row[1], and row[2] are equal and not equal to a space. If they are, return row[0] immediately and skip all remaining checks. RESULT winner_row]
            [COMMAND If the row check did not return, for each column index col from 0 to 2 inclusive, check whether self.board[0][col], self.board[1][col], and self.board[2][col] are equal and not equal to a space. If they are, return self.board[0][col] immediately and skip all remaining checks. RESULT winner_column]
            [COMMAND If the column check did not return, check whether self.board[0][0], self.board[1][1], and self.board[2][2] are equal and not equal to a space. If they are, return self.board[0][0] immediately and skip all remaining checks. RESULT winner_main_diagonal]
            [COMMAND If the main diagonal check did not return, check whether self.board[0][2], self.board[1][1], and self.board[2][0] are equal and not equal to a space. If they are, return self.board[0][2] immediately. RESULT winner_anti_diagonal]
            [COMMAND If no winning row, column, or diagonal was found, return None. RESULT no_winner]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True if no row of the board contains a space, and False as soon as any row contains a space." is_board_full]
    [INPUTS]
        <REF> self </REF>: object "The board object that owns the board attribute being inspected."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the board is full because every row lacks spaces; False when any row contains a space."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board, assigning the current element to row. RESULT row]
            [COMMAND Within the for loop over row in self.board, check whether the string ' ' is present in row. RESULT contains_space]
            [COMMAND Within the for loop over row in self.board, if contains_space is true, immediately exit the loop and the function by returning False. RESULT early_return_false]
            [COMMAND After the for loop over row in self.board has finished without finding a space in any row, return True. RESULT final_return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
