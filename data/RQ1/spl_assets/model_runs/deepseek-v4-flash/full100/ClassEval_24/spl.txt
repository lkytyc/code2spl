[DEFINE_WORKER: "Computes the sum of two complex numbers by adding their real and imaginary components separately and returning a new complex result." add]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex operand whose .real and .imag attributes are read."
        <REF> c2 </REF>: complex "Second complex operand whose .real and .imag attributes are read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: complex "The complex number constructed from the summed real and imaginary parts, equivalent to c1 + c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the real components of c1 and c2 and compute their sum, storing it in real. RESULT real]
            [COMMAND Read the imaginary components of c1 and c2 and compute their sum, storing it in imaginary. RESULT imaginary]
            [COMMAND Construct a new complex number answer from real as its real part and imaginary as its imaginary part. RESULT answer]
            [COMMAND Return answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the complex difference of two complex numbers by subtracting their real and imaginary parts separately." subtract]
    [INPUTS]
        <REF> c1 </REF>: complex "The complex number from which to subtract."
        <REF> c2 </REF>: complex "The complex number to subtract."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The complex result of c1 minus c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Subtract the real part of c2 from the real part of c1 and store the result as real. RESULT real]
            [COMMAND Subtract the imaginary part of c2 from the imaginary part of c1 and store the result as imaginary. RESULT imaginary]
            [COMMAND Construct a complex number from real and imaginary and return it as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiplies two complex numbers by computing the real part as c1.real * c2.real - c1.imag * c2.imag and the imaginary part as c1.real * c2.imag + c1.imag * c2.real, then returns a new complex number formed from those components." multiply]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex number whose real and imag fields are used in the multiplication."
        <REF> c2 </REF>: complex "Second complex number whose real and imag fields are used in the multiplication."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "Complex product of c1 and c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the real part of the product as c1.real * c2.real - c1.imag * c2.imag and assign it to real. RESULT real]
            [COMMAND Compute the imaginary part of the product as c1.real * c2.imag + c1.imag * c2.real and assign it to imaginary. RESULT imaginary]
            [COMMAND Construct a complex number from real and imaginary and return it as the result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the quotient of two complex numbers by multiplying the dividend by the conjugate of the divisor and dividing by the squared magnitude of the divisor." divide]
    [INPUTS]
        <REF> c1 </REF>: complex "The dividend complex number."
        <REF> c2 </REF>: complex "The divisor complex number."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The quotient c1 / c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the denominator as the squared magnitude of c2: c2.real**2 + c2.imag**2. RESULT denominator]
            [COMMAND Compute the real part of the quotient as (c1.real * c2.real + c1.imag * c2.imag) / denominator. RESULT real]
            [COMMAND Compute the imaginary part of the quotient as (c1.imag * c2.real - c1.real * c2.imag) / denominator. RESULT imaginary]
            [COMMAND Return the complex number formed from real and imaginary. RESULT quotient]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: denominator == 0, which occurs when c2.real == 0 and c2.imag == 0]
        [LOG "Complex division attempted with zero denominator."]
        [THROW ZeroDivisionError "Division by zero in complex division."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
