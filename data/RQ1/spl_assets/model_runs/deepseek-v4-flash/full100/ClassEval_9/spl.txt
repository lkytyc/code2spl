# CLASS BigNumCalculator

[DEFINE_WORKER: "Adds two decimal integer strings by zero-padding them to equal length, summing digits from right to left with carry, and returning the sum as a decimal string." add]
    [INPUTS]
        <REF> num1 </REF>: string "First decimal number represented as a string."
        <REF> num2 </REF>: string "Second decimal number represented as a string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "Decimal string representing the sum of num1 and num2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute max_length as max(len(num1), len(num2)). RESULT max_length]
            [COMMAND Pad num1 with leading zeros to width max_length using zfill. RESULT num1]
            [COMMAND Pad num2 with leading zeros to width max_length using zfill. RESULT num2]
            [COMMAND Initialize carry to 0. RESULT carry]
            [COMMAND Initialize result to an empty list. RESULT result]
            [COMMAND For i from max_length - 1 down to 0 inclusive, compute digit_sum as int(num1[i]) + int(num2[i]) + carry. RESULT digit_sum]
            [COMMAND In the same loop iteration for i, set carry to digit_sum // 10. RESULT carry]
            [COMMAND In the same loop iteration for i, set digit to digit_sum % 10. RESULT digit]
            [COMMAND In the same loop iteration for i, insert str(digit) at position 0 of result. RESULT result]
            [COMMAND After the digit loop, check whether carry is greater than 0; if it is, execute the final-carry alternative flow. RESULT carry]
            [COMMAND Return the string formed by joining the elements of result with an empty separator. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: carry > 0 after the digit loop]
        [SEQUENTIAL_BLOCK]
            [COMMAND Insert str(carry) at position 0 of result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Subtracts two non-negative decimal integers represented as strings using digit-by-digit borrowing and returns the signed decimal result as a string." subtract]
    [INPUTS]
        <REF> num1 </REF>: string "Non-negative decimal integer as a string."
        <REF> num2 </REF>: string "Non-negative decimal integer as a string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The signed decimal result of subtracting num2 from num1, as a string without leading zeros."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If len(num1) < len(num2), swap num1 and num2 and set negative to True; else if len(num1) > len(num2), set negative to False; else, when the lengths are equal, if num1 < num2 then swap num1 and num2 and set negative to True; otherwise set negative to False. RESULT num1, num2, negative]
            [COMMAND Set max_length to the maximum of len(num1) and len(num2). RESULT max_length]
            [COMMAND Zero-pad num1 and num2 to length max_length using left padding. RESULT num1, num2]
            [COMMAND Initialize borrow to 0 and result to an empty list. RESULT borrow, result]
            [COMMAND For each i from max_length - 1 down to 0 inclusive, compute digit_diff = int(num1[i]) - int(num2[i]) - borrow; if digit_diff < 0, add 10 to digit_diff and set borrow to 1, otherwise set borrow to 0; then insert str(digit_diff) at the beginning of result. RESULT borrow, result]
            [COMMAND While len(result) > 1 and result[0] == '0', remove the first element of result. RESULT result]
            [COMMAND If negative is True, insert '-' at the beginning of result. RESULT result]
            [COMMAND Return the string obtained by joining all elements of result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiplies two non-negative integers represented as decimal strings using digit-by-digit long multiplication with carry handling, then returns the product as a decimal string without leading zeros." multiply]
    [INPUTS]
        <REF> num1 </REF>: string "String of decimal digits representing a non-negative integer."
        <REF> num2 </REF>: string "String of decimal digits representing a non-negative integer."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Product of num1 and num2 as a decimal string without leading zeros, or '0' if the product is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign len1 = len(num1) and len2 = len(num2). RESULT len1_len2]
            [COMMAND Initialize result = [0] * (len1 + len2), a list of zeros with length len1 + len2. RESULT result]
            [COMMAND Start outer loop: for i in range(len1 - 1, -1, -1), so i visits each index from len1 - 1 down to 0 inclusive. RESULT i]
            [COMMAND Within the outer loop over i, start inner loop: for j in range(len2 - 1, -1, -1), so j visits each index from len2 - 1 down to 0 inclusive. RESULT j]
            [COMMAND Within the outer loop over i and the inner loop over j, compute mul = int(num1[i]) * int(num2[j]). RESULT mul]
            [COMMAND Within the outer loop over i and the inner loop over j, set p1 = i + j and p2 = i + j + 1. RESULT p1_p2]
            [COMMAND Within the outer loop over i and the inner loop over j, compute total = mul + result[p2]. RESULT total]
            [COMMAND Within the outer loop over i and the inner loop over j, update result[p1] += total // 10 and set result[p2] = total % 10. RESULT result]
            [COMMAND After both loops finish, set start = 0. RESULT start]
            [COMMAND While start < len(result) - 1 and, with short-circuit evaluation, result[start] == 0, increment start by 1 to skip leading zeros but preserve at least one digit. RESULT start]
            [COMMAND Return ''.join(map(str, result[start:])), converting the remaining digit list to a decimal string. RESULT product]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
