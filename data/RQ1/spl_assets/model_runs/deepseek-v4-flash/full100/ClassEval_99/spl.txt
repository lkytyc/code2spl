# CLASS ZipFileProcessor

[DEFINE_WORKER: "Initializes a new instance by assigning the provided file_name to the instance attribute self.file_name." __init__]
    [INPUTS]
        <REF> file_name </REF>: string "The file name to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Accept the file_name parameter for the constructor. RESULT file_name]
            [COMMAND Assign the file_name parameter to the instance attribute self.file_name. RESULT self.file_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads the zip file at self.file_name and returns the opened ZipFile object, or None if opening fails." read_zip_file]
    [INPUTS]
        <REF> self.file_name </REF>: string "Path to the zip file to be opened"
    [END_INPUTS]

    [OUTPUTS]
        <REF> zip_file </REF>: zipfile.ZipFile "Opened zip file object returned on success"
        <REF> None </REF>: NoneType "Returned when opening the zip file fails"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the zip file at self.file_name in read mode using zipfile.ZipFile. RESULT zip_file]
            [COMMAND Return the successfully opened zip_file object immediately. RESULT zip_file]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If zipfile.ZipFile raises any exception while attempting to open the file]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the fallback result. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts all contents of a ZIP archive referenced by self.file_name to the given output_path, returning True on success and False on any exception." extract_all]
    [INPUTS]
        <REF> output_path </REF>: string "Destination directory path where the ZIP archive contents will be extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if extraction succeeds, False if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the ZIP archive at self.file_name in read mode within a try block using zipfile.ZipFile, assigning it to zip_file. RESULT zip_file]
            [COMMAND Extract all contents from zip_file to output_path using zip_file.extractall(output_path). RESULT None]
            [COMMAND Return True after successful extraction. RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening self.file_name as a ZIP archive or extracting its contents to output_path.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate failure. RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts a specified file from a zip archive to a target directory, returning True on success and False if any exception occurs." extract_file]
    [INPUTS]
        <REF> file_name </REF>: str "Name or path of the file inside the zip archive to extract."
        <REF> output_path </REF>: str "Destination directory where the file should be extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the file was extracted successfully, False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip archive located at self.file_name in read mode and assign it to zip_file. RESULT zip_file]
            [COMMAND Extract the file named file_name from zip_file into output_path. RESULT extract_status]
            [COMMAND Return True after the extraction completes successfully. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip archive or extracting the specified file.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception and return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a ZIP archive containing the given files and returns True on success or False if any error occurs." create_zip_file]
    [INPUTS]
        <REF> files </REF>: list "List of file paths to add to the ZIP archive."
        <REF> output_file_name </REF>: str "Path of the ZIP file to create."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the ZIP archive was created successfully; False if an exception occurred."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a new ZIP archive at output_file_name in write mode. RESULT zip_file]
            [COMMAND For each file in files, write the file into the zip_file. RESULT None]
            [COMMAND After all files have been written, exit the with block and close the ZIP archive. RESULT None]
            [COMMAND Return True after successful creation of the ZIP archive. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception is raised while opening the ZIP archive, writing files, or closing the archive.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
