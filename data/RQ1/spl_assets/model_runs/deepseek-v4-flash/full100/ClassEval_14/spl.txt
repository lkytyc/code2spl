# CLASS BookManagementDB

[DEFINE_WORKER: "Initializes the object by opening the SQLite database, creating a cursor, and calling create_table to ensure the table schema exists." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "Path to the SQLite database file, ':memory:' for an in-memory database, or None for a temporary database."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the SQLite database identified by db_name using sqlite3.connect and store the resulting connection object in self.connection. RESULT self.connection]
            [COMMAND Create a database cursor by calling self.connection.cursor() and store it in self.cursor. RESULT self.cursor]
            [COMMAND Call self.create_table() to ensure the required table schema is created. RESULT self.create_table()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: If sqlite3.connect(db_name) cannot open or create the database due to an invalid path, permission error, or other database error.]
        [LOG "Failed to connect to SQLite database using the provided db_name."]
        [THROW sqlite3.Error "Raised when the database connection cannot be established from db_name."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates the books table in the connected database if it does not already exist, defining id, title, author, and available columns, then commits the transaction." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL statement CREATE TABLE IF NOT EXISTS books (id INTEGER PRIMARY KEY, title TEXT, author TEXT, available INTEGER) using self.cursor.execute(). RESULT database_schema_change_staged]
            [COMMAND Commit the current transaction using self.connection.commit() to make the table creation durable. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a new book record with the given title and author and an available flag set to 1, then commits the transaction." add_book]
    [INPUTS]
        <REF> title </REF>: string "Title of the book to insert into the books table."
        <REF> author </REF>: string "Author of the book to insert into the books table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL INSERT statement on self.cursor, binding title to the title column, author to the author column, and setting available to the constant 1. RESULT none]
            [COMMAND Commit the transaction on self.connection to persist the inserted row. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes the book with the given ID from the books table and commits the transaction." remove_book]
    [INPUTS]
        <REF> book_id </REF>: integer "ID of the book to remove from the books table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute SQL DELETE statement 'DELETE FROM books WHERE id = ?' on self.cursor, binding book_id as the single SQL parameter and ignoring the returned cursor. RESULT execute_effect]
            [COMMAND Commit the transaction on self.connection to persist the deletion. RESULT commit_effect]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as unavailable by setting its available column to 0 and committing the transaction." borrow_book]
    [INPUTS]
        <REF> book_id </REF>: integer "Identifier of the book to mark as unavailable."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute an UPDATE statement on the books table to set available to 0 for the row whose id equals the bound parameter book_id. RESULT update_executed]
            [COMMAND Commit the current database transaction to make the availability change persistent. RESULT commit_done]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks the book identified by book_id as available in the database and commits the transaction to persist the change." return_book]
    [INPUTS]
        <REF> book_id </REF>: book identifier "Identifier of the book to mark as available."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute SQL UPDATE statement on the books table to set available = 1 where id equals book_id, binding book_id as the query parameter. RESULT cursor_execution]
            [COMMAND Commit the current transaction on the database connection to persist the update. RESULT connection_commit]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a query to select all books from the database cursor and returns all fetched rows." search_books]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> books </REF>: list "A list of tuples representing all rows from the books table, or an empty list if no rows are returned."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query 'SELECT * FROM books' using self.cursor. RESULT query_executed]
            [COMMAND Retrieve all remaining rows from the cursor result set by calling self.cursor.fetchall() and assign the returned list to books. RESULT books]
            [COMMAND Return the books list as the method's result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
