    [END_OUTPUTS]

[DEFINE_WORKER: "Builds an n by n Minesweeper board by placing k randomly located mines and incrementing all non-mine neighboring cells for each placed mine." generate_mine_sweeper_map]
    [INPUTS]
        <REF> self.n </REF>: int "Side length of the square board."
        <REF> self.k </REF>: int "Number of mines to place; the placement loop runs exactly self.k times."
    [END_INPUTS]

    [OUTPUTS]
        <REF> arr </REF>: list[list[int|str]] "A self.n by self.n list of lists containing 'X' at mined cells and the adjacent mine count at non-mined cells."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize arr as a square 2D list with self.n rows, where every row contains self.n integers initialized to 0. RESULT arr]
            [COMMAND Repeat the following self.k times: for each num in range(self.k), where num is unused, choose x = random.randint(0, self.n - 1) and y = random.randint(0, self.n - 1), then set arr[y][x] = 'X'. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 0 and x <= self.n - 2) and (y >= 0 and y <= self.n - 1) and arr[y][x + 1] != 'X', increment arr[y][x + 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 1 and x <= self.n - 1) and (y >= 0 and y <= self.n - 1) and arr[y][x - 1] != 'X', increment arr[y][x - 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 1 and x <= self.n - 1) and (y >= 1 and y <= self.n - 1) and arr[y - 1][x - 1] != 'X', increment arr[y - 1][x - 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 0 and x <= self.n - 2) and (y >= 1 and y <= self.n - 1) and arr[y - 1][x + 1] != 'X', increment arr[y - 1][x + 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 0 and x <= self.n - 1) and (y >= 1 and y <= self.n - 1) and arr[y - 1][x] != 'X', increment arr[y - 1][x] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 0 and x <= self.n - 2) and (y >= 0 and y <= self.n - 2) and arr[y + 1][x + 1] != 'X', increment arr[y + 1][x + 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 1 and x <= self.n - 1) and (y >= 0 and y <= self.n - 2) and arr[y + 1][x - 1] != 'X', increment arr[y + 1][x - 1] by 1. RESULT arr]
            [COMMAND In the same loop iteration after placing the mine at (x, y), if (x >= 0 and x <= self.n - 1) and (y >= 0 and y <= self.n - 2) and arr[y + 1][x] != 'X', increment arr[y + 1][x] by 1. RESULT arr]
            [COMMAND Return arr as the final Minesweeper map. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a square self.n by self.n game board filled with '-' characters in every cell and returns it." generate_playerMap]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: list of list of str "A self.n by self.n grid where each cell is initialized to the string '-'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a two-dimensional list arr by evaluating the list comprehension [['-' for row in range(self.n)] for column in range(self.n)], giving an outer list with self.n inner lists, each inner list containing self.n occurrences of '-'. RESULT arr]
            [COMMAND Return arr as the generated player map. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns False if any cell in map is hidden ('-') and is not a mine ('X') in self.minesweeper_map; otherwise returns True after checking all cells." check_won]
    [INPUTS]
        <REF> map </REF>: 2D list of characters "The board to inspect; '-' represents a hidden cell and any other character represents a revealed cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if every hidden cell in map corresponds to 'X' in self.minesweeper_map; False if any hidden cell is not 'X'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each integer i such that 0 <= i < self.n, visit row i of map. RESULT i]
            [COMMAND Within the outer loop over i, for each integer j such that 0 <= j < self.n, visit cell (i, j). RESULT j]
            [COMMAND Within the nested loops over i and j, evaluate the condition with short-circuit semantics: first test whether map[i][j] equals '-'; if it does not equal '-', the combined condition is false without evaluating self.minesweeper_map[i][j] != 'X'. If it does equal '-', evaluate self.minesweeper_map[i][j] != 'X' and use that value as the combined condition result. RESULT hidden_non_mine_cell]
            [COMMAND After the nested loops complete without triggering the alternative flow, return True. RESULT check_won_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: At the current cell, hidden_non_mine_cell is true, meaning map[i][j] == '-' and self.minesweeper_map[i][j] != 'X'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, stopping all further loops and bypassing the final return True. RESULT check_won_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reveals the selected Minesweeper cell, updates the score, and returns False on a mine, True on a winning move, or the updated player map when the move is safe and not winning." sweep]
    [INPUTS]
        <REF> self </REF>: object "The Minesweeper game instance, providing minesweeper_map, player_map, score, and check_won."
        <REF> x </REF>: int "Row index of the selected cell."
        <REF> y </REF>: int "Column index of the selected cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or player_map "Returns False if the selected cell is a mine, True if the move wins the game, or the updated player_map otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.minesweeper_map[x][y] equals the mine marker 'X'. RESULT is_mine]
            [COMMAND If is_mine is True, exit the method by returning False. RESULT mine_return]
            [COMMAND When is_mine is False, reveal the selected cell by assigning self.minesweeper_map[x][y] to self.player_map[x][y]. RESULT revealed_player_map]
            [COMMAND After revealing the cell, increment self.score by 1. RESULT updated_score]
            [COMMAND Evaluate whether self.check_won(self.player_map) equals True. RESULT has_won]
            [COMMAND If has_won is True, exit the method by returning True. RESULT win_return]
            [COMMAND If has_won is False, return self.player_map as the final result. RESULT final_player_map]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
