    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.segments as an empty list. RESULT self.segments]
            [COMMAND Initialize self.with_end_tag as False. RESULT self.with_end_tag]
        [END_SEQUENTIAL_BLOCK]

[DEFINE_WORKER: "Adds a segment to the collection after transforming it with fix_path." add]
    [INPUTS]
        <REF> segment </REF>: object "Segment to be transformed by fix_path and appended to self.segments."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

[DEFINE_WORKER: "Parses a non-empty path by noting whether it originally ended with '/', normalizing it with fix_path, splitting it into slash-delimited segments, URL-decoding each segment, and appending the decoded segments to self.segments." parse]
    [INPUTS]
        <REF> path </REF>: string "Path string to parse; if falsy, parsing is skipped."
        <REF> charset </REF>: string "Character encoding used for URL-decoding each path segment."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether path is truthy; if path is falsy, skip all remaining parsing steps. RESULT path]
            [COMMAND Within the outer if path truthy, check whether path ends with '/' and, if so, set self.with_end_tag to True. RESULT self.with_end_tag]
            [COMMAND Within the outer if path truthy, call self.fix_path(path) and assign the normalized result back to path. RESULT path]
            [COMMAND Within the outer if path truthy, check whether the normalized path is truthy; if it is, split path on every '/' character into a list named split, preserving empty segments from leading, trailing, or consecutive slashes. RESULT split]
            [COMMAND Within the outer if path truthy and the inner if normalized path truthy, iterate over each segment seg in split in order. RESULT seg]
            [COMMAND Within the outer if path truthy, the inner if normalized path truthy, and the for loop over split, URL-decode seg using charset and append the decoded result to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalizes an input path by returning an empty string for falsy inputs and stripping all leading and trailing forward slashes otherwise." fix_path]
    [INPUTS]
        <REF> path </REF>: str "Input path to be normalized. Falsy values such as an empty string or None cause an early empty-string return."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The normalized path with all leading and trailing forward slashes removed, or an empty string if the input path is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND When path is not falsy, remove all leading and trailing forward slash characters from path by calling path.strip('/') and assign the result to segment_str. RESULT segment_str]
            [COMMAND Return segment_str. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: path evaluates to false, for example it is an empty string or None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
