# CLASS NLPDataProcessor

[DEFINE_WORKER: "Constructs and returns a list of lowercase English stop words: 'a', 'an', and 'the'." construct_stop_word_list]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: list "List containing the lowercase stop words 'a', 'an', and 'the'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a local variable named stop_word_list and assign it a new list containing the string literals 'a', 'an', and 'the' in that order. RESULT stop_word_list]
            [COMMAND Return the value of stop_word_list as the method's result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Splits each string in string_list on whitespace, attempts to remove stop words by mutating each split word list during iteration, and returns the resulting list of word lists." remove_stop_words]
    [INPUTS]
        <REF> self </REF>: object "The method receiver instance."
        <REF> string_list </REF>: list of strings "The list of strings to split and filter."
        <REF> stop_word_list </REF>: collection of strings "The stop words to remove from each split string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: list of list of strings "The list of word lists after stop-word removal; some stop words may remain because the source list is mutated during iteration."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize answer as an empty list. RESULT answer]
            [COMMAND For each string in string_list, enter the outer loop body with the current element bound to string. RESULT string]
            [COMMAND Within that outer loop, split string using default whitespace splitting into string_split. RESULT string_split]
            [COMMAND Within the outer loop over string, enter an inner loop over each word in the live string_split, binding the current element to word. RESULT word]
            [COMMAND If word is in stop_word_list, remove the first occurrence of word from string_split; because the inner loop is iterating over the same list, this mutation may cause the next element to be skipped. RESULT string_split]
            [COMMAND After the inner loop finishes for the current string, append the current string_split to answer. RESULT answer]
            [COMMAND After the outer loop finishes, return answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a stop word list and uses it to filter stop words out of the provided string list, returning the filtered list." process]
    [INPUTS]
        <REF> string_list </REF>: list of strings "The list of strings to be processed and filtered using the stop word list."
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list of strings "The resulting list after stop words have been removed from string_list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.construct_stop_word_list() and assign the returned stop word list to stop_word_list. RESULT stop_word_list]
            [COMMAND Call self.remove_stop_words(string_list, stop_word_list) and assign the returned filtered word list to words_list. RESULT words_list]
            [COMMAND Return words_list from the method. RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
