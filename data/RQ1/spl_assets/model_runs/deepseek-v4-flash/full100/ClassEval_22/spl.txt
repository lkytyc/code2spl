# CLASS ClassRegistrationSystem

[DEFINE_WORKER: "Initializes a new instance by setting students to an empty list and students_registration_classes to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.students as a new empty list. RESULT self.students]
            [COMMAND Initialize self.students_registration_classes as a new empty dictionary. RESULT self.students_registration_classes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a student by adding it to the students list only if it is not already present, returning 1 on success and 0 on duplicate." register_student]
    [INPUTS]
        <REF> student </REF>: object "The student to add to self.students."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 0 if the student was already present, or 1 if the student was successfully added."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether student is present in self.students. RESULT is_present]
            [COMMAND If student is present in self.students, return 0. RESULT duplicate_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If student is not present in self.students]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append student to self.students. RESULT updated_students]
            [COMMAND Return 1. RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a student for a class by appending the class name to the student's list of classes, creating the list first if the student is not already registered." register_class]
    [INPUTS]
        <REF> student_name </REF>: string "The name of the student being registered."
        <REF> class_name </REF>: string "The name of the class to register the student for."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of strings "The updated list of class names associated with the student."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether student_name is already a key in self.students_registration_classes. RESULT key_exists]
            [COMMAND If key_exists is true, append class_name to the existing list stored at self.students_registration_classes[student_name]. RESULT updated_list]
            [COMMAND If key_exists is false, assign a new list containing class_name to self.students_registration_classes[student_name]. RESULT updated_list]
            [COMMAND Return the list stored at self.students_registration_classes[student_name]. RESULT registration_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects and returns the names of all students whose major field exactly matches the provided major value, preserving the original order of self.students." get_students_by_major]
    [INPUTS]
        <REF> major </REF>: string "The exact major value to compare against each student's 'major' field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> student_list </REF>: list of strings "The names of students whose major equals major, in the order they appear in self.students."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize student_list to an empty list. RESULT student_list]
            [COMMAND Iterate over each student in self.students in original order. RESULT student]
            [COMMAND Within the for loop over self.students, check whether student["major"] equals major using an exact, case-sensitive comparison. RESULT is_major_match]
            [COMMAND Within the for loop over self.students, if is_major_match is true, append student["name"] to student_list. RESULT student_list]
            [COMMAND After the for loop completes, return student_list. RESULT student_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects the distinct majors from self.students in the order they are first encountered and returns them as a list." get_all_major]
    [INPUTS]
        <REF> self </REF>: object "The object instance containing a students attribute with an iterable of student dictionaries."
    [END_INPUTS]

    [OUTPUTS]
        <REF> major_list </REF>: list "The unique major values from self.students, preserving the order of first occurrence."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize major_list as an empty list. RESULT major_list]
            [COMMAND Iterate over each student in self.students. RESULT student]
            [COMMAND For each student, evaluate whether student['major'] is not already contained in major_list. RESULT condition_result]
            [COMMAND If condition_result is true, append student['major'] to major_list. RESULT major_list]
            [COMMAND After the loop completes, return major_list. RESULT major_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters students by major, aggregates the registered classes of matching students, and returns the most frequently occurring class." get_most_popular_class_in_major]
    [INPUTS]
        <REF> major </REF>: string "The major value used to filter students in self.students."
    [END_INPUTS]

    [OUTPUTS]
        <REF> most_popular_class </REF>: object "The class label with the highest frequency among the aggregated registered classes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize class_list to an empty list. RESULT class_list]
            [COMMAND For each student in self.students, evaluate whether student["major"] is equal to major. RESULT student]
            [COMMAND Within the for loop, when student["major"] equals major, extend class_list with self.students_registration_classes[student["name"]]. RESULT class_list]
            [COMMAND After the loop, compute most_popular_class as the element with the highest frequency in class_list using max(set(class_list), key=class_list.count), with tie-breaking determined by set iteration order. RESULT most_popular_class]
            [COMMAND Return most_popular_class. RESULT most_popular_class]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: class_list is empty when max(set(class_list), key=class_list.count) is evaluated]
        [LOG "ValueError: max() arg is an empty sequence"]
        [THROW ValueError "Raised because max() requires a non-empty iterable and no registered classes were collected for the matching major."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
