    [END_INPUTS]

[DEFINE_WORKER: "Reads a CSV file, treats the first row as the title, and returns that title along with all remaining rows as data." read_csv]
    [INPUTS]
        <REF> file_name </REF>: string "Path to the CSV file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A pair (title, data) where title is the header row of the CSV file and data is a list of all remaining rows."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize data as an empty list. RESULT data]
            [COMMAND Open the file named file_name in read mode, explicitly using mode 'r', and keep it open as file within a context manager. RESULT file]
            [COMMAND Within the opened file, create a csv.reader object from file and assign it to reader. RESULT reader]
            [COMMAND Within the opened file and using reader, retrieve the first row from reader and assign it to title. RESULT title]
            [COMMAND Within the opened file and using reader, iterate over each remaining row in reader and append the row to data. RESULT data]
            [COMMAND Return title and data as a pair. RESULT title, data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the given data rows to a CSV file and returns 1 on success or 0 if an exception occurs." write_csv]
    [INPUTS]
        <REF> data </REF>: iterable of row sequences "The data rows to write into the CSV file."
        <REF> file_name </REF>: string "Path to the CSV file to create or overwrite."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 1 if the CSV write succeeds, or 0 if an exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file specified by file_name in write mode with newline='' and assign the file object. RESULT file]
            [COMMAND Create a CSV writer for the opened file. RESULT writer]
            [COMMAND Write all rows in data to the CSV file using writer.writerows(data). RESULT write_operation]
            [COMMAND Return 1 to indicate successful CSV writing. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception occurs while opening the file or writing the CSV data.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 to indicate failure. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a CSV file, extracts the values in column N, uppercases them, and writes the result to a new CSV file." process_csv_data]
    [INPUTS]
        <REF> N </REF>: int "Index of the column to extract from each data row."
        <REF> save_file_name </REF>: str "Path to the input CSV file."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "Result returned by self.write_csv after writing the processed data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the CSV file using self.read_csv with save_file_name and unpack the returned values into title and data. RESULT title, data]
            [COMMAND Extract element at index N from each row in data into a new list column_data. RESULT column_data]
            [COMMAND Apply .upper() to every element in column_data and replace column_data with the resulting list. RESULT column_data]
            [COMMAND Create new_data as a two-element list containing title and the final column_data. RESULT new_data]
            [COMMAND Build output_file_name by taking the part of save_file_name before the first dot and appending '_process.csv'. RESULT output_file_name]
            [COMMAND Call self.write_csv with new_data and output_file_name, and return its result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
