# CLASS FitnessTracker

[DEFINE_WORKER: "Initializes the object's height, weight, age, sex, and BMI standard range attributes from the supplied constructor parameters." __init__]
    [INPUTS]
        <REF> height </REF>: untyped "Height value assigned to the corresponding instance attribute."
        <REF> weight </REF>: untyped "Weight value assigned to the corresponding instance attribute."
        <REF> age </REF>: untyped "Age value assigned to the corresponding instance attribute."
        <REF> sex </REF>: untyped "Sex value assigned to the corresponding instance attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> None </REF>: NoneType "The constructor returns None after initializing the instance attributes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter height to the instance attribute self.height. RESULT self.height]
            [COMMAND Assign the input parameter weight to the instance attribute self.weight. RESULT self.weight]
            [COMMAND Assign the input parameter age to the instance attribute self.age. RESULT self.age]
            [COMMAND Assign the input parameter sex to the instance attribute self.sex. RESULT self.sex]
            [COMMAND Initialize the instance attribute self.BMI_std with the constant nested list of dictionaries: male range [20, 25] and female range [19, 24]. RESULT self.BMI_std]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the body mass index by dividing the object's weight by the square of its height." get_BMI]
    [INPUTS]
        <REF> self </REF>: object "Object that exposes the numeric attributes self.weight and self.height."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The body mass index computed as self.weight divided by self.height squared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.weight from the object. RESULT weight]
            [COMMAND Read self.height from the object. RESULT height]
            [COMMAND Compute the square of height as height ** 2. RESULT height_squared]
            [COMMAND Compute weight divided by height_squared using float division. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.weight or self.height attribute is missing on self]
        [LOG "Required attribute missing while accessing weight or height."]
        [THROW AttributeError "Raised when the object does not provide the weight or height attribute."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.height equals 0]
        [LOG "Division by zero while computing BMI."]
        [THROW ZeroDivisionError "Raised because BMI division uses a zero height."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Classifies the subject's BMI as above, below, or within the sex-specific reference range, returning 1, -1, or 0 respectively." condition_judge]
    [INPUTS]
        <REF> self </REF>: object "Object providing the sex, BMI_std reference ranges, and get_BMI() method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return </REF>: int "1 if BMI is above the sex-specific reference range; -1 if below; 0 if within the inclusive range."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the subject's BMI by calling self.get_BMI() and assign the result to BMI. RESULT BMI]
            [COMMAND Select the sex-specific reference range: if self.sex equals "male", set BMI_range to self.BMI_std[0]["male"]; otherwise set BMI_range to self.BMI_std[1]["female"]. RESULT BMI_range]
            [COMMAND Classify BMI against BMI_range: if BMI is strictly greater than BMI_range[1], return 1; else if BMI is strictly less than BMI_range[0], return -1; otherwise, meaning BMI is within the inclusive range, return 0. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates daily calorie intake by computing the basal metabolic rate from sex, weight, height, and age, then applying a lifestyle multiplier selected from condition_judge() results." calculate_calorie_intake]
    [INPUTS]
        <REF> self </REF>: object "The object instance providing sex, weight, height, age, and condition_judge()."
    [END_INPUTS]

    [OUTPUTS]
        <REF> calorie_intake </REF>: float "The daily calorie intake calculated as BMR times the selected lifestyle activity multiplier."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If self.sex is exactly 'male', compute BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age + 5; otherwise compute BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age - 161. RESULT BMR]
            [COMMAND Call self.condition_judge() and store the returned value. RESULT condition_result]
            [COMMAND If condition_result is exactly 1, set calorie_intake = BMR * 1.2. RESULT calorie_intake]
            [COMMAND If condition_result is not exactly 1, call self.condition_judge() again and store the returned value. RESULT condition_result_2]
            [COMMAND If condition_result_2 is exactly -1, set calorie_intake = BMR * 1.6. RESULT calorie_intake]
            [COMMAND If condition_result is not exactly 1 and condition_result_2 is not exactly -1, set calorie_intake = BMR * 1.4. RESULT calorie_intake]
            [COMMAND Return calorie_intake. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
