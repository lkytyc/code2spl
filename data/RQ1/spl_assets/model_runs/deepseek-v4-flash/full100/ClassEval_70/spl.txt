# CLASS PersonRequest

[DEFINE_WORKER: "Initializes a new object by validating the provided name, sex, and phoneNumber values and assigning them to the corresponding instance attributes." __init__]
    [INPUTS]
        <REF> name </REF>: str "The name value to validate and assign to self.name."
        <REF> sex </REF>: str "The sex value to validate and assign to self.sex."
        <REF> phoneNumber </REF>: str "The phone number value to validate and assign to self.phoneNumber."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate the name parameter by calling _validate_name(name) and assign the result to self.name. RESULT self.name]
            [COMMAND Validate the sex parameter by calling _validate_sex(sex) and assign the result to self.sex. RESULT self.sex]
            [COMMAND Validate the phoneNumber parameter by calling _validate_phoneNumber(phoneNumber) and assign the result to self.phoneNumber. RESULT self.phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates a name by rejecting falsy values and names longer than 33 characters, returning the original name otherwise." _validate_name]
    [INPUTS]
        <REF> self </REF>: object "Bound method receiver; not used by the validation logic."
        <REF> name </REF>: str "Input name to validate; may be None or an empty string in practice despite the declared return type."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "The original name if it is truthy and no longer than 33 characters; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name is truthy; if it is not, divert to the falsy-name alternative flow. RESULT name]
            [COMMAND Check whether len(name) is less than or equal to 33; if it is greater, divert to the too-long-name alternative flow. RESULT name]
            [COMMAND Return the validated name unchanged. RESULT name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: name is falsy, for example None, empty string, or another value that evaluates to False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: name is truthy and len(name) > 33.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the sex value is exactly one of 'Man', 'Woman', or 'UGM', returning it unchanged when valid and None otherwise." _validate_sex]
    [INPUTS]
        <REF> sex </REF>: str "The value to validate against the allowed list ['Man', 'Woman', 'UGM']."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | None "Returns the original sex string if it is exactly 'Man', 'Woman', or 'UGM'; otherwise returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether sex is exactly one of the allowed values ["Man", "Woman", "UGM"] using a case-sensitive exact equality comparison. RESULT is_valid]
            [COMMAND If is_valid is true, return sex unchanged as the validated result. RESULT sex]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_valid is false, meaning sex is not in ["Man", "Woman", "UGM"]]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately as the guard-clause result. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that phoneNumber is a non-empty string of exactly 11 digits, returning it unchanged if valid and None otherwise." _validate_phoneNumber]
    [INPUTS]
        <REF> phoneNumber </REF>: str "Phone number string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "The original phoneNumber if it is non-empty and consists of exactly 11 digits; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether phoneNumber is falsy, including None, empty string, or other falsy value. If it is falsy, return None; otherwise continue. RESULT falsy_check]
            [COMMAND Evaluate whether len(phoneNumber) != 11 or not phoneNumber.isdigit(), using short-circuit evaluation. If true, return None; otherwise continue. RESULT validity_check]
            [COMMAND Return phoneNumber unchanged as the validated phone number. RESULT phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: phoneNumber is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len(phoneNumber) != 11 or not phoneNumber.isdigit()]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
