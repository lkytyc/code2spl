[DEFINE_WORKER: "Performs piecewise linear interpolation on the given x/y data to estimate y values for each coordinate in x_interp, returning a list of estimates for coordinates that fall inside the data range." interpolate_1d]
    [INPUTS]
        <REF> x </REF>: array[number] "Independent coordinate values; the code expects them to be sorted in ascending order, len(x) >= 2, and no duplicate consecutive values."
        <REF> y </REF>: array[number] "Dependent values paired with x; y must contain at least as many elements as x."
        <REF> x_interp </REF>: array[number] "Query coordinates for which interpolated y values are requested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> y_interp </REF>: array[number] "Interpolated y values, one for each x_interp value that falls inside an interval [x[i], x[i+1]]. x_interp values outside the range of x are not included in the returned list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize y_interp as an empty list. RESULT y_interp]
            [COMMAND For each xi in x_interp, process it in order. RESULT xi]
            [COMMAND For the current xi, loop over segment indexes i from 0 to len(x)-2 inclusive, in increasing order. RESULT i]
            [COMMAND Within the current xi and segment index i, evaluate whether x[i] <= xi <= x[i+1]. The comparison is inclusive of both endpoints. If true, select this i as the matching interval and do not consider later i values; because segments are checked left to right, the first matching interval is used when xi lies on a shared boundary. RESULT i]
            [COMMAND Using the selected interval i, compute yi = y[i] + (y[i+1] - y[i]) * (xi - x[i]) / (x[i+1] - x[i]). RESULT yi]
            [COMMAND Append yi to y_interp and break out of the inner segment loop for the current xi. RESULT y_interp]
            [COMMAND If no segment i satisfies the inclusive condition for the current xi, do not append any value for that xi and continue with the next xi. RESULT y_interp]
            [COMMAND After all xi values in x_interp have been processed, return y_interp. RESULT y_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Performs bilinear interpolation on a 2D grid for each (x, y) query point and returns the interpolated values for points that fall inside the grid." interpolate_2d]
    [INPUTS]
        <REF> x </REF>: List[float] "Grid x-coordinates, assumed to be sorted; used to define x intervals."
        <REF> y </REF>: List[float] "Grid y-coordinates, assumed to be sorted; used to define y intervals."
        <REF> z </REF>: List[List[float]] "2D grid values where z[i][j] corresponds to (x[i], y[j])."
        <REF> x_interp </REF>: Sequence[float] "x-coordinates of the query points, paired element-wise with y_interp."
        <REF> y_interp </REF>: Sequence[float] "y-coordinates of the query points, paired element-wise with x_interp."
    [END_INPUTS]

    [OUTPUTS]
        <REF> z_interp </REF>: List[float] "Interpolated z values, one for each query point that lies within the grid; points outside the grid are omitted."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize z_interp as an empty list to accumulate interpolated values. RESULT z_interp]
            [COMMAND For each pair (xi, yi) from zip(x_interp, y_interp), process one interpolation point. RESULT xi_yi_pair]
            [COMMAND Within the outer loop over (xi, yi), iterate i from 0 to len(x)-2 to search for the first x interval containing xi. RESULT i]
            [COMMAND Within the outer loop over (xi, yi) and the i loop, test whether x[i] <= xi <= x[i+1]. RESULT x_interval_matched]
            [COMMAND Within the i loop when the x-interval condition is true, iterate j from 0 to len(y)-2 to search for the first y interval containing yi. RESULT j]
            [COMMAND Within the i loop and j loop, test whether y[j] <= yi <= y[j+1]. RESULT y_interval_matched]
            [COMMAND Within the i loop and j loop when both interval conditions are true, read z00 = z[i][j], z01 = z[i][j+1], z10 = z[i+1][j], and z11 = z[i+1][j+1]. RESULT corner_values]
            [COMMAND Within the matched grid cell, compute zi using the bilinear interpolation formula: (z00*(x[i+1]-xi)*(y[j+1]-yi) + z10*(xi-x[i])*(y[j+1]-yi) + z01*(x[i+1]-xi)*(yi-y[j]) + z11*(xi-x[i])*(yi-y[j])) / ((x[i+1]-x[i])*(y[j+1]-y[j])). RESULT zi]
            [COMMAND Within the matched grid cell, append zi to z_interp and break out of the j loop. RESULT z_interp]
            [COMMAND After exiting the j loop for a matched x interval, break out of the i loop; this terminates the x interval search even if no y interval matched. RESULT i_loop_terminated]
            [COMMAND After processing all (xi, yi) pairs, return z_interp. RESULT z_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: For a query point (xi, yi), xi is not contained in any inclusive interval x[i]..x[i+1] for i = 0..len(x)-2.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip appending a value for this (xi, yi) pair and continue to the next pair in zip(x_interp, y_interp). RESULT z_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: For a query point (xi, yi), xi is contained in an inclusive x interval, but yi is not contained in any inclusive interval y[j]..y[j+1] for j = 0..len(y)-2 within that x interval.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Because the matched x interval still triggers the break out of the i loop, do not append a value for this (xi, yi) pair and continue to the next pair. RESULT z_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
