# CLASS BoyerMooreSearch

[DEFINE_WORKER: "Initialize the object by storing the provided text and pattern and computing their lengths at construction time." __init__]
    [INPUTS]
        <REF> text </REF>: object "Input text value to store and to measure with len(...)."
        <REF> pattern </REF>: object "Input pattern value to store and to measure with len(...)."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter text directly to self.text and assign the input parameter pattern directly to self.pattern. RESULT self.text,self.pattern]
            [COMMAND Compute len(text) and assign it to self.textLen, and compute len(pattern) and assign it to self.patLen. RESULT self.textLen,self.patLen]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches backward through the pattern and returns the rightmost index of the input character, or -1 if the character does not appear." match_in_pattern]
    [INPUTS]
        <REF> char </REF>: character "The character to search for in the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The rightmost index where the character is found in the pattern, or -1 if no match exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate index i backward from self.patLen - 1 down to 0 inclusive using step -1. RESULT i]
            [COMMAND Within the backward loop over i, compare char with self.pattern[i]. RESULT is_match]
            [COMMAND If char equals self.pattern[i] within the loop, return i immediately as the result. RESULT result]
            [COMMAND After the loop completes with no match found at any index, return -1. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The character matches self.pattern[i] at some iteration during the backward scan.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the current index i immediately without checking any earlier indices. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the pattern against the text from right to left at the given alignment and returns the text index of the first backward-found mismatch, or -1 if all positions match." mismatch_in_text]
    [INPUTS]
        <REF> currentPos </REF>: integer "Starting alignment index in self.text where self.pattern is compared."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The index in self.text of the first mismatch encountered during backward scanning, or -1 if no mismatch is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate i in reverse from self.patLen - 1 down to 0 inclusive using range(self.patLen - 1, -1, -1). RESULT i]
            [COMMAND Within the reverse loop for each i, compare self.pattern[i] with self.text[currentPos + i]. RESULT comparison_result]
            [COMMAND Within the reverse loop, if self.pattern[i] is not equal to self.text[currentPos + i], immediately return currentPos + i as the mismatch index in self.text. RESULT result]
            [COMMAND If the loop completes without any mismatch at any index from self.patLen - 1 down to 0, return -1. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A mismatch is found at some index i during the reverse scan.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return currentPos + i immediately without checking any smaller indices. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scan all valid pattern start positions in the text, collect positions with no mismatch, and otherwise compute a bad-character-style shift value that does not affect the actual Python for-loop progression." bad_character_heuristic]
    [INPUTS]
        <REF> self </REF>: object "Instance containing text, pattern, their lengths, and helper methods used to detect mismatches and locate characters in the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> positions </REF>: list "List of starting indices where the pattern fully matches the text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named positions to store successful match start indices. RESULT positions]
            [COMMAND Iterate i over every integer from 0 inclusive to self.textLen - self.patLen inclusive by using range(self.textLen - self.patLen + 1), covering every valid text start position where the pattern can fit. RESULT loop_i]
            [COMMAND Within the for loop for each current i, call self.mismatch_in_text(i) to determine whether the pattern mismatches the text at that start position, where -1 means no mismatch and any other value is a mismatch index in the text. RESULT mismatch_index]
            [COMMAND Within the for loop, if mismatch_index equals -1, append the current i to positions because the pattern fully matches at that start position. RESULT positions]
            [COMMAND Within the same for loop, otherwise when mismatch_index is not -1, read the mismatching text character as self.text[mismatch_index] and call self.match_in_pattern on that character to get match_index. RESULT match_index]
            [COMMAND Within that mismatch branch, assign i = mismatch_index - match_index as a computed bad-character-style shift candidate for the current iteration only. RESULT i]
            [COMMAND Continue through the for loop using the next value produced by range regardless of the reassignment to i, because in Python this inner assignment does not alter future loop iteration values. RESULT for_loop_progression]
            [COMMAND After the loop finishes, return positions. RESULT positions]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
