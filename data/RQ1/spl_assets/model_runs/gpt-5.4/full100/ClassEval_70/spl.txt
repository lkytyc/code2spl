        <REF> sex </REF>: str "Input sex value to validate before storing in the instance."
        <REF> phoneNumber </REF>: str "Input phone number value to validate before storing in the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self._validate_name with the input name and assign the returned validated value to self.name. RESULT self.name]
            [COMMAND Call self._validate_sex with the input sex and assign the returned validated value to self.sex. RESULT self.sex]
            [COMMAND Call self._validate_phoneNumber with the input phoneNumber and assign the returned validated value to self.phoneNumber. RESULT self.phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[DEFINE_WORKER: "Validate a name by returning None for falsy values or values longer than 33 characters, otherwise return the original name unchanged." _validate_name]
    [INPUTS]
        <REF> self </REF>: object "Instance context for the method."
        <REF> name </REF>: str "Name value to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str_or_none "Returns None when the name is falsy or has length greater than 33; otherwise returns the original name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name is falsy. RESULT is_name_falsy]
            [COMMAND If name is falsy, return None immediately as an early exit. RESULT result]
            [COMMAND If name is truthy, check whether its length is greater than 33. RESULT is_name_too_long]
            [COMMAND If the length of name is greater than 33, return None immediately as an early exit. RESULT result]
            [COMMAND If name is truthy and its length is 33 or less, return the original name unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: name is falsy, including cases such as None or an empty string.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately without performing the length check. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: name is truthy but len(name) is greater than 33, meaning 34 or more characters.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that the input sex value is one of the allowed strings and return None for invalid values or the original value for valid ones." _validate_sex]
    [INPUTS]
        <REF> sex </REF>: str "The sex value to validate against the allowed strings Man, Woman, and UGM."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The original sex value if it matches one of the allowed strings, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether sex is not one of the exact allowed strings "Man", "Woman", or "UGM". RESULT validation_check]
            [COMMAND If validation_check is true, return None immediately as an early exit from the method. RESULT result]
            [COMMAND If validation_check is false, return the original sex value unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate a phone number by returning it only when it is present, exactly 11 characters long, and contains only digits; otherwise return None." _validate_phoneNumber]
    [INPUTS]
        <REF> phoneNumber </REF>: str "Phone number string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | None "The original phone number if it passes validation, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether phoneNumber is truthy before any further validation. RESULT phoneNumber_presence_checked]
            [COMMAND If phoneNumber is falsy, return None immediately as an early exit. RESULT return_none_when_missing]
            [COMMAND If phoneNumber is present, evaluate whether its length is not equal to 11 or whether it contains any non-digit characters, preserving the short-circuit OR logic. RESULT phoneNumber_format_validation]
            [COMMAND If either the length is not exactly 11 or phoneNumber.isdigit() is false, return None immediately. RESULT return_none_when_invalid]
            [COMMAND If phoneNumber passed all prior checks, return the original phoneNumber unchanged. RESULT validated_phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: phoneNumber is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None without performing length or digit validation. RESULT return_none_when_missing]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: phoneNumber is present but its length is not exactly 11 or it contains non-digit characters.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the validation failure result. RESULT return_none_when_invalid]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
