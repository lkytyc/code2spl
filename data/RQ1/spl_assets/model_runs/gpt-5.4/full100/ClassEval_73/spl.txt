# CLASS RPGCharacter

[DEFINE_WORKER: "Initializes a new instance by copying the provided name, hp, attack_power, defense, and level values into instance attributes and setting experience to 0." __init__]
    [INPUTS]
        <REF> name </REF>: data "Value assigned directly to the instance name attribute."
        <REF> hp </REF>: data "Value assigned directly to the instance hp attribute."
        <REF> attack_power </REF>: data "Value assigned directly to the instance attack_power attribute."
        <REF> defense </REF>: data "Value assigned directly to the instance defense attribute."
        <REF> level </REF>: data "Value assigned directly to the instance level attribute; defaults to 1 when not provided by the caller."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input name value directly to self.name without transformation or validation. RESULT self.name]
            [COMMAND Assign the input hp value directly to self.hp without transformation or validation. RESULT self.hp]
            [COMMAND Assign the input attack_power value directly to self.attack_power without transformation or validation. RESULT self.attack_power]
            [COMMAND Assign the input defense value directly to self.defense without transformation or validation. RESULT self.defense]
            [COMMAND Assign the input level value directly to self.level without transformation or validation, using the method default of 1 if no level argument was provided by the caller. RESULT self.level]
            [COMMAND Initialize self.exp to the constant 0. RESULT self.exp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes damage from the attacker’s power and the target’s defense, enforcing a minimum damage of 1, then subtracts that damage from the target’s hit points." attack]
    [INPUTS]
        <REF> self </REF>: object "The attacker object providing attack_power."
        <REF> other_character </REF>: object "The target character object providing defense and hp."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.attack_power and other_character.defense, then compute damage as the larger of their difference and 1. RESULT damage]
            [COMMAND Subtract the computed damage from other_character.hp. RESULT other_character.hp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Heals the object by increasing hp by 10, capping hp at 100, and returning the updated hp." heal]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the hp field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The updated hp value after healing and capping."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.hp by 10. RESULT self.hp]
            [COMMAND If self.hp is greater than 100, set self.hp to 100. RESULT self.hp]
            [COMMAND Return self.hp. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Repeatedly adds experience until the input amount is exhausted, leveling up each time the accumulated experience reaches or exceeds the current threshold." gain_exp]
    [INPUTS]
        <REF> amount </REF>: int "Experience points to add."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND While amount is not equal to 0, repeatedly process the remaining experience gain. RESULT loop]
            [COMMAND Within the while loop, if self.exp + amount is greater than or equal to self.level * 100, consume only the experience needed to reach the threshold. RESULT threshold_check]
            [COMMAND In that threshold-reached branch, subtract (self.level * 100 - self.exp) from amount and then call self.level_up(). RESULT leveled_up]
            [COMMAND Within the while loop, if self.exp + amount is less than self.level * 100, add the full amount to self.exp and set amount to 0 to end the loop. RESULT exp_added]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Increase the character's level and core stats only when the current level is below 100, then return the current level and updated combat attributes." level_up]
    [INPUTS]
        <REF> self </REF>: object "Instance containing level, experience, health, attack power, and defense attributes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the current level, hp, attack_power, and defense after any applicable level-up changes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.level is strictly less than 100. RESULT level_below_max]
            [COMMAND If self.level is strictly less than 100, increase self.level by 1. RESULT updated_level]
            [COMMAND If self.level is strictly less than 100, set self.exp to 0. RESULT reset_exp]
            [COMMAND If self.level is strictly less than 100, increase self.hp by 20. RESULT updated_hp]
            [COMMAND If self.level is strictly less than 100, increase self.attack_power by 5. RESULT updated_attack_power]
            [COMMAND If self.level is strictly less than 100, increase self.defense by 5. RESULT updated_defense]
            [COMMAND Return a tuple of self.level, self.hp, self.attack_power, and self.defense. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.level is greater than or equal to 100]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do not modify self.level, self.exp, self.hp, self.attack_power, or self.defense. RESULT state_unchanged]
            [COMMAND Return a tuple of the existing self.level, self.hp, self.attack_power, and self.defense values. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns whether the object's hp value is greater than zero." is_alive]
    [INPUTS]
        <REF> self </REF>: object "The instance whose hp attribute is checked."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when self.hp is greater than 0, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.hp. RESULT hp_value]
            [COMMAND Compare self.hp against 0 using a greater-than check. RESULT is_positive]
            [COMMAND Return the boolean comparison result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
