# CLASS ShoppingCart

[DEFINE_WORKER: "Initializes the instance by creating an empty dictionary for items." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary to self.items as an instance attribute initialization. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Set self.items[item] to a dictionary containing the provided price and quantity, whether or not the item already exists." add_item]
    [INPUTS]
        <REF> self </REF>: object "Instance that contains the items mapping."
        <REF> item </REF>: data_type "Key identifying the item to add or update in self.items."
        <REF> price </REF>: data_type "Price value to store for the item."
        <REF> quantity </REF>: data_type "Quantity value to store for the item; defaults to 1 when not provided."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item is already a key in self.items. RESULT item_exists]
            [COMMAND If item exists in self.items, assign self.items[item] to a dictionary with keys 'price' mapped to price and 'quantity' mapped to quantity. RESULT updated_existing_item_entry]
            [COMMAND If item does not exist in self.items, assign self.items[item] to a dictionary with keys 'price' mapped to price and 'quantity' mapped to quantity. RESULT created_item_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item is already present in self.items]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace the existing self.items[item] entry with a dictionary containing the provided price and quantity. RESULT updated_existing_item_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: item is not present in self.items]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create self.items[item] with a dictionary containing the provided price and quantity. RESULT created_item_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrease the stored quantity for a given item in self.items by the specified amount when the item exists, otherwise leave the collection unchanged." remove_item]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the items collection."
        <REF> item </REF>: any "Key identifying the item to update in self.items."
        <REF> quantity </REF>: number "Amount to subtract from the item's stored quantity; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item exists as a key in self.items. RESULT item_exists]
            [COMMAND If item_exists is true, subtract quantity from self.items[item]['quantity']. RESULT updated_quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item does not exist as a key in self.items]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do nothing and leave self.items unchanged. RESULT no_change]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the current value of the object's items field without modification." view_items]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "The current value stored in self.items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the items field from the current object instance. RESULT self.items]
            [COMMAND Return the retrieved items value directly to the caller without modification. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total price of all items by summing each item's quantity multiplied by its price." total_price]
    [INPUTS]
        <REF> self </REF>: object "Instance containing an items collection whose values are item dictionaries."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The sum of quantity times price for every item in self.items.values(); returns 0 when there are no items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over all item values in self.items.values(). RESULT items]
            [COMMAND For each item, read item['quantity'] and item['price']. RESULT quantity_and_price]
            [COMMAND Multiply the quantity by the price for each item to obtain that item's subtotal. RESULT subtotal]
            [COMMAND Sum all item subtotals and return the total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: An item is missing the 'quantity' or 'price' key, or either value is not numeric.]
        [LOG "The multiplication or summation cannot be completed because an item value is invalid."]
        [THROW RuntimeError "Propagation of the underlying runtime error such as KeyError or TypeError."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
