# CLASS TimeUtils

[DEFINE_WORKER: "Initializes the instance by storing the current local date and time in the datetime attribute." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call datetime.datetime.now() to obtain the current local date and time. RESULT current_datetime]
            [COMMAND Assign the obtained datetime value to self.datetime. RESULT self.datetime]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the current datetime using a fixed hour-minute-second pattern and returns the resulting time string." get_current_time]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The current time formatted as HH:MM:SS."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the fixed format pattern "%H:%M:%S" to the local variable format. RESULT format]
            [COMMAND Use self.datetime.strftime(format) to format the current datetime with the assigned pattern and return the formatted string. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the current datetime using the fixed date pattern YYYY-MM-DD and returns the resulting string." get_current_date]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The current date formatted as a YYYY-MM-DD string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constant string "%Y-%m-%d" to the local variable format. RESULT format]
            [COMMAND Call self.datetime.strftime with format to produce the formatted current date string. RESULT formatted_date]
            [COMMAND Return the formatted date string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a given number of seconds to the current datetime and returns the result formatted as an H:M:S string." add_seconds]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the current datetime value."
        <REF> seconds </REF>: number "Number of seconds to add to the current datetime."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The updated time formatted as hours, minutes, and seconds."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute a new datetime by adding a timedelta of the given seconds to self.datetime. RESULT new_datetime]
            [COMMAND Set the output format string to %H:%M:%S. RESULT format]
            [COMMAND Return new_datetime formatted with the time pattern. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses an input timestamp string into a datetime object using the fixed format %Y-%m-%d %H:%M:%S." string_to_datetime]
    [INPUTS]
        <REF> string </REF>: str "Timestamp text that must match the format %Y-%m-%d %H:%M:%S."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: datetime "A datetime object parsed from the input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Receive the input string parameter. RESULT string]
            [COMMAND Parse string with datetime.datetime.strptime using the fixed format %Y-%m-%d %H:%M:%S. RESULT datetime_value]
            [COMMAND Return the parsed datetime value. RESULT datetime_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the input string does not exactly match the required format.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to parse the input string with datetime.datetime.strptime using the fixed format %Y-%m-%d %H:%M:%S. RESULT parse_error]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the given datetime value into a string using the fixed pattern YYYY-MM-DD HH:MM:SS." datetime_to_string]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> datetime </REF>: datetime-like object "Value to format as a date-time string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted date-time string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call strftime on the input datetime value with the fixed format string "%Y-%m-%d %H:%M:%S". RESULT formatted_datetime_string]
            [COMMAND Return the formatted date-time string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert two time strings to datetime values and return the rounded difference between them in minutes." get_minutes]
    [INPUTS]
        <REF> string_time1 </REF>: string "The first time value as a string."
        <REF> string_time2 </REF>: string "The second time value as a string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The rounded minute difference between the second time and the first time."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert string_time1 to a datetime value using self.string_to_datetime and store it in time1. RESULT time1]
            [COMMAND Convert string_time2 to a datetime value using self.string_to_datetime and store it in time2. RESULT time2]
            [COMMAND Compute the time difference time2 minus time1, take the seconds component, divide it by 60, round the result, and return it. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats a datetime built from the given year, month, day, hour, minute, and second into a string using the fixed pattern %Y-%m-%d %H:%M:%S." get_format_time]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> year </REF>: int "Year component for the datetime value."
        <REF> month </REF>: int "Month component for the datetime value."
        <REF> day </REF>: int "Day component for the datetime value."
        <REF> hour </REF>: int "Hour component for the datetime value."
        <REF> minute </REF>: int "Minute component for the datetime value."
        <REF> second </REF>: int "Second component for the datetime value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The formatted datetime string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the fixed format string to %Y-%m-%d %H:%M:%S. RESULT format]
            [COMMAND Construct a datetime value from year, month, day, hour, minute, and second. RESULT time_item]
            [COMMAND Convert the datetime value to a string using the fixed format and return it. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
