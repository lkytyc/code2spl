# CLASS ExpressionCalculator

[DEFINE_WORKER: "Initialize the instance with an empty postfix stack and a fixed operator-priority table." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create and store an empty deque in self.postfix_stack. RESULT self.postfix_stack]
            [COMMAND Assign the literal priority list [0, 3, 2, 1, -1, 1, 0, 2] to self.operat_priority. RESULT self.operat_priority]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Evaluate a postfix expression by normalizing negative markers, applying stack-based operator calculations, and returning the final numeric result as a float." calculate]
    [INPUTS]
        <REF> self </REF>: object "Instance providing prepare, transform, postfix_stack, is_operator, and _calculate behavior."
        <REF> expression </REF>: string "Input expression to transform and evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The evaluated expression result."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Prepare the internal postfix state by passing the transformed expression to self.prepare(self.transform(expression)). RESULT postfix_stack_ready]
            [COMMAND Initialize an empty deque named result_stack. RESULT result_stack]
            [COMMAND Reverse self.postfix_stack so tokens can be processed in pop order. RESULT self.postfix_stack]
            [COMMAND While self.postfix_stack is not empty, pop the next token into current_op. RESULT current_op]
            [COMMAND Within the loop, if current_op is not an operator, replace every '~' with '-' and append the token to result_stack. RESULT result_stack]
            [COMMAND Within the loop, if current_op is an operator, pop second_value and first_value from result_stack. RESULT first_value_and_second_value]
            [COMMAND Within the operator branch, normalize first_value and second_value by replacing every '~' with '-'. RESULT normalized_operands]
            [COMMAND Within the operator branch, compute temp_result by calling self._calculate(first_value, second_value, current_op). RESULT temp_result]
            [COMMAND Within the operator branch, convert temp_result to a string and append it to result_stack. RESULT result_stack]
            [COMMAND After the loop finishes, join result_stack items with '*' and evaluate the expression, then return the float of that value. RESULT final_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert an input expression into postfix components by scanning characters, accumulating operand substrings, and using an operator stack with sentinel and parenthesis handling." prepare]
    [INPUTS]
        <REF> expression </REF>: iterable "Input expression whose elements are materialized into a list for left-to-right parsing."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize op_stack as a deque containing the single sentinel string ',' to act as the base stack boundary. RESULT op_stack]
            [COMMAND Convert expression into a list named arr so each element can be accessed by index during parsing. RESULT arr]
            [COMMAND Initialize current_index to 0 as the starting index of the current operand span. RESULT current_index]
            [COMMAND Initialize count to 0 as the number of consecutive non-operator elements currently accumulated. RESULT count]
            [COMMAND Iterate left to right over arr with enumerate, binding i to the current index and current_op to the current element. RESULT loop_iteration]
            [COMMAND Within the loop, when self.is_operator(current_op) is true, treat the current element as an operator or delimiter and begin operator-processing logic. RESULT operator_branch]
            [COMMAND Within the operator branch, if count > 0, append the pending operand substring formed by joining arr[current_index: current_index + count] onto self.postfix_stack. RESULT pending_operand_appended]
            [COMMAND Within the operator branch after any pending operand append, read the top operator from op_stack using op_stack[-1] and store it in peek_op. RESULT peek_op]
            [COMMAND Within the operator branch, if current_op == ')', repeatedly pop operators from op_stack and append their string forms to self.postfix_stack while op_stack[-1] != '('. RESULT popped_until_open_paren]
            [COMMAND Still within the current_op == ')' path, pop the matching '(' from op_stack without appending it after the loop terminates. RESULT open_paren_removed]
            [COMMAND Within the operator branch, otherwise when current_op is not ')', repeatedly pop and append the top of op_stack while all three conditions hold simultaneously: current_op != '(', peek_op != ',', and self.compare(current_op, peek_op) is true. RESULT precedence_pops]
            [COMMAND Within that same non-')' path, after each pop update peek_op to the new top of op_stack using op_stack[-1] before reevaluating the while condition. RESULT peek_op_refreshed]
            [COMMAND Within the non-')' path after the precedence loop finishes, push current_op onto op_stack. RESULT operator_pushed]
            [COMMAND After completing either operator subpath, reset count to 0 because the current operand span has ended at this operator position. RESULT count_reset]
            [COMMAND After completing either operator subpath, set current_index to i + 1 so the next operand span starts immediately after the operator. RESULT current_index_advanced]
            [COMMAND Within the loop, when self.is_operator(current_op) is false, increment count by 1 to extend the current contiguous non-operator span. RESULT count_incremented]
            [COMMAND After the loop finishes, if count > 1 or if count == 1 and self.is_operator(arr[current_index]) is false, append the trailing operand substring formed by joining arr[current_index: current_index + count] onto self.postfix_stack. RESULT trailing_operand_appended]
            [COMMAND After trailing operand handling, while op_stack[-1] != ',', pop each remaining operator from op_stack, convert it to string, and append it to self.postfix_stack. RESULT remaining_operators_flushed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return whether the input character is one of the predefined operator or parenthesis symbols." is_operator]
    [INPUTS]
        <REF> c </REF>: character "The input character to test for membership in the operator symbol set."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the input character is one of '+', '-', '*', '/', '(', ')', or '%'; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether input character c is a member of the fixed set of symbols {'+', '-', '*', '/', '(', ')', '%'}. RESULT membership_check]
            [COMMAND Return the boolean result of that membership check. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalize '%' to '/' for both input operator characters, then return whether the normalized peek operator priority is greater than or equal to the normalized cur operator priority using the priority table indexed by character code minus 40." compare]
    [INPUTS]
        <REF> cur </REF>: string "The current operator character to normalize if it is '%' and then compare by priority."
        <REF> peek </REF>: string "The next operator character to normalize if it is '%' and then compare by priority."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the priority value for the normalized peek character is greater than or equal to the priority value for the normalized cur character; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If cur is exactly '%', replace cur with '/'. RESULT cur]
            [COMMAND If peek is exactly '%', replace peek with '/'. RESULT peek]
            [COMMAND Compute the priority-table index for peek as ord(peek) minus 40 and retrieve the corresponding priority from self.operat_priority. RESULT peek_priority]
            [COMMAND Compute the priority-table index for cur as ord(cur) minus 40 and retrieve the corresponding priority from self.operat_priority. RESULT cur_priority]
            [COMMAND Return whether peek_priority is greater than or equal to cur_priority. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return a Decimal arithmetic result for two input values based on a supported operator, or raise an error for an unsupported operator." _calculate]
    [INPUTS]
        <REF> first_value </REF>: any "The first operand value to be converted to Decimal before calculation."
        <REF> second_value </REF>: any "The second operand value to be converted to Decimal before calculation."
        <REF> current_op </REF>: string "The operator selector expected to be one of '+', '-', '*', '/', or '%'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: Decimal "The Decimal result of applying the selected arithmetic operator to the two input values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether current_op is exactly '+'. RESULT is_addition]
            [COMMAND If current_op is exactly '+', convert first_value and second_value to Decimal, add them, and return the sum immediately. RESULT result]
            [COMMAND Check whether current_op is exactly '-'. RESULT is_subtraction]
            [COMMAND If current_op is exactly '-', convert first_value and second_value to Decimal, subtract the second from the first, and return the difference immediately. RESULT result]
            [COMMAND Check whether current_op is exactly '*'. RESULT is_multiplication]
            [COMMAND If current_op is exactly '*', convert first_value and second_value to Decimal, multiply them, and return the product immediately. RESULT result]
            [COMMAND Check whether current_op is exactly '/'. RESULT is_division]
            [COMMAND If current_op is exactly '/', convert first_value and second_value to Decimal, divide the first by the second, and return the quotient immediately. RESULT result]
            [COMMAND Check whether current_op is exactly '%'. RESULT is_modulo]
            [COMMAND If current_op is exactly '%', convert first_value and second_value to Decimal, compute the remainder of the first modulo the second, and return the modulo result immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: current_op is not exactly one of '+', '-', '*', '/', or '%'.]
        [LOG "Unexpected operator: {current_op}"]
        [THROW ValueError "Raised when the operator is unsupported."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalizes an expression string by removing whitespace and a trailing equals sign, marks unary minus occurrences with a special token, and rewrites a leading negated parenthesized expression into a zero-prefixed subtraction form before returning the final string." transform]
    [INPUTS]
        <REF> expression </REF>: string "Expression text to normalize and transform."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The transformed expression string after normalization and unary-minus handling."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove all whitespace characters from expression using a global regular-expression substitution, and store the normalized string back into expression. RESULT expression]
            [COMMAND Remove a trailing '=' from expression if and only if one exists at the end of the string, and store the result back into expression. RESULT expression]
            [COMMAND Convert the final expression string into a character list and store it in arr. RESULT arr]
            [COMMAND Iterate left to right over arr with both index i and current character c from enumerate(arr). RESULT iteration_over_arr]
            [COMMAND Within the loop, process only iterations where c equals '-'; for all other characters, leave arr unchanged for that iteration. RESULT minus_filter]
            [COMMAND Within the loop when c equals '-' and i equals 0, replace arr[i] with '~' to mark a leading unary minus. RESULT arr]
            [COMMAND Within the loop when c equals '-' and i is greater than 0, read the previous character arr[i - 1] into prev_c. RESULT prev_c]
            [COMMAND Within the loop when c equals '-' and i is greater than 0, if prev_c is one of '+', '-', '*', '/', '(', 'E', or 'e', replace arr[i] with '~'; otherwise leave arr[i] unchanged. RESULT arr]
            [COMMAND After the loop, evaluate whether arr[0] equals '~' and, only if that is true, evaluate whether len(arr) is greater than 1 and arr[1] equals '(' using short-circuit logic. RESULT leading_negated_parenthesized_check]
            [COMMAND If the post-loop condition is true, change arr[0] from '~' to '-', join arr into a string, prefix the joined string with '0', and return that value immediately. RESULT result]
            [COMMAND Otherwise, join arr into a string without further modification and return that value immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After the loop, arr[0] is '~', len(arr) is greater than 1, and arr[1] is '('.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set arr[0] to '-'. RESULT arr]
            [COMMAND Join arr into a string, prefix it with '0', and return the resulting string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the loop, the condition arr[0] == '~' and (len(arr) > 1 and arr[1] == '(') is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join arr as it currently stands and return the resulting string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
