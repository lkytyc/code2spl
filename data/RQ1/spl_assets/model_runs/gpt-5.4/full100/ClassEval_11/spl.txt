# CLASS BitStatusUtil

[DEFINE_WORKER: "Validate the two input bit-status values and return a new value that merges stat into states using a bitwise OR without clearing existing bits." add]
    [INPUTS]
        <REF> states </REF>: value "Existing bit-status value or bit-mask accumulator to be updated."
        <REF> stat </REF>: value "Bit-status flag or bit-mask value to merge into states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: value "Combined bit-status value with all bits from states and stat set in the returned result."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check with [states, stat] to validate both input values before performing the merge. RESULT validated_inputs]
            [COMMAND Compute the bitwise OR of states and stat so that every bit set in either operand is set in the returned value. RESULT result]
            [COMMAND Return the combined bit-status result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether all bits set in stat are also set in states using a bitmask comparison." has]
    [INPUTS]
        <REF> states </REF>: integer "Bit state value to test against."
        <REF> stat </REF>: integer "Bit mask that must be fully contained in states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when every bit in stat is present in states; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check with states and stat to perform the input validation or side-effect-free check step. RESULT checked_inputs]
            [COMMAND Compute the bitwise AND of states and stat, then compare the result for equality with stat. RESULT is_contained]
            [COMMAND Return the equality result as the method output. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate the input bit-status arguments and remove the specified status bit from states only if that bit is currently present, otherwise return states unchanged." remove]
    [INPUTS]
        <REF> states </REF>: unknown "The current bitmask or status set from which a status may be removed."
        <REF> stat </REF>: unknown "The bit or status flag to validate and potentially remove from states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: unknown "The resulting states value after conditionally removing stat, or the original states if stat is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate the input arguments by calling BitStatusUtil.check with the list [states, stat] before any further processing. RESULT validated_inputs]
            [COMMAND Evaluate whether BitStatusUtil.has(states, stat) is true to determine if the target status bit is currently present in states. RESULT has_stat]
            [COMMAND If has_stat is true, return the value of states XOR stat immediately to remove or toggle off that present status bit. RESULT states_without_stat]
            [COMMAND If has_stat is false, return the original states unchanged. RESULT states]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that each value in args is a non-negative even number, raising a ValueError immediately on the first invalid element." check]
    [INPUTS]
        <REF> args </REF>: list "Collection of values to validate in iteration order."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each element in args in order using the loop variable arg. RESULT arg]
            [COMMAND Within the loop for the current arg, evaluate whether arg is less than 0. RESULT is_negative]
            [COMMAND Within the same loop iteration, if arg is not less than 0, evaluate whether arg modulo 2 is not equal to 0. RESULT is_not_even]
            [COMMAND If neither exception condition is triggered for the current arg, continue to the next element in args until the loop completes. RESULT validation_complete]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Within the loop, the current arg is less than 0.]
        [LOG "arg must be greater than or equal to 0"]
        [THROW ValueError "Raise a ValueError immediately indicating that the current value is below the inclusive lower bound of 0."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: Within the loop, the current arg is not less than 0 and arg modulo 2 is not equal to 0.]
        [LOG "arg not even"]
        [THROW ValueError "Raise a ValueError immediately indicating that the current value is not evenly divisible by 2."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
