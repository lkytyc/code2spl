# CLASS Thermostat

[DEFINE_WORKER: "Initialize the object by storing the provided current temperature, target temperature, and mode on the instance." __init__]
    [INPUTS]
        <REF> current_temperature </REF>: unknown "Initial current temperature value to store."
        <REF> target_temperature </REF>: unknown "Target temperature value to store."
        <REF> mode </REF>: unknown "Mode value to store."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input current_temperature directly to the instance field self.current_temperature. RESULT self.current_temperature]
            [COMMAND Assign the input target_temperature directly to the instance field self.target_temperature. RESULT self.target_temperature]
            [COMMAND Assign the input mode directly to the instance field self.mode. RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current target temperature stored on the instance." get_target_temperature]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: unknown "The value of self.target_temperature."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the instance's target_temperature attribute as the method result. RESULT self.target_temperature]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sets the object's target temperature to the provided temperature value." set_target_temperature]
    [INPUTS]
        <REF> self </REF>: object "The instance whose target temperature is being updated."
        <REF> temperature </REF>: any "The new temperature value to assign."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]

        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the current mode stored on the instance." get_mode]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the mode attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The value of the instance's mode attribute."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the mode attribute from self and return it immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate the requested mode against the allowed values heat and cool, then store it on the object or return False for invalid input." set_mode]
    [INPUTS]
        <REF> mode </REF>: any "Requested operating mode to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False when the mode is invalid; otherwise the mode is assigned to the object and no explicit value is returned."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether mode is one of the allowed values heat or cool. RESULT mode_validity]
            [COMMAND If mode is valid, assign mode to self.mode. RESULT self.mode]
            [COMMAND If mode is not valid, return False immediately. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: mode is not equal to heat or cool]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.mode. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Set the mode to heat when the current temperature is below the target temperature; otherwise set it to cool." auto_set_mode]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the current temperature is strictly less than the target temperature. RESULT temperature_comparison]
            [COMMAND If the current temperature is below the target, set the mode to heat; otherwise set the mode to cool. RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the current temperature requires the system to already be in cooling or heating mode, and otherwise switches mode automatically before returning false." auto_check_conflict]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether current_temperature is greater than target_temperature. RESULT temperature_is_above_target]
            [COMMAND If current_temperature is greater than target_temperature, check whether mode equals 'cool'. RESULT cooling_branch_result]
            [COMMAND If current_temperature is greater than target_temperature and mode is 'cool', return True immediately. RESULT True]
            [COMMAND If current_temperature is greater than target_temperature and mode is not 'cool', call auto_set_mode() and return False. RESULT False]
            [COMMAND If current_temperature is not greater than target_temperature, check whether mode equals 'heat'. RESULT heating_branch_result]
            [COMMAND If current_temperature is not greater than target_temperature and mode is 'heat', return True immediately. RESULT True]
            [COMMAND If current_temperature is not greater than target_temperature and mode is not 'heat', call auto_set_mode() and return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Automatically sets the operating mode, then moves the current temperature one unit at a time toward the target temperature while counting the number of steps taken." simulate_operation]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call auto_set_mode on the current object to determine the operating mode before any temperature adjustment. RESULT self.auto_set_mode()]
            [COMMAND Initialize the step counter use_time to 0 as the starting time boundary. RESULT use_time]
            [COMMAND If self.mode equals 'heat', repeatedly increase current_temperature by 1 while it is still less than target_temperature, and increase use_time by 1 for each increment. RESULT heat_adjustment]
            [COMMAND Otherwise, repeatedly decrease current_temperature by 1 while it is still greater than target_temperature, and increase use_time by 1 for each decrement. RESULT cool_adjustment]
            [COMMAND Return the final use_time value after the temperature reaches the target boundary. RESULT use_time]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
