# CLASS AutomaticGuitarSimulator

[DEFINE_WORKER: "Initialize the instance by storing the input text in the play_text attribute." __init__]
    [INPUTS]
        <REF> text </REF>: text "Input text to store on the instance during initialization."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input text to the instance attribute self.play_text. RESULT self.play_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parse the stored play text into a list of dictionaries by splitting on literal spaces and separating each segment into a leading alphabetic chord prefix and the remaining tune suffix, with optional display of each parsed pair." interpret]
    [INPUTS]
        <REF> display </REF>: boolean "Whether to call the instance display method for each parsed chord and tune pair."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "An empty list when the stripped play text is empty, otherwise a list of dictionaries containing keys Chord and Tune for each space-split segment."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.play_text.strip() is empty after removing leading and trailing whitespace. RESULT is_blank_play_text]
            [COMMAND If the stripped play text is empty, return an empty list immediately. RESULT result]
            [COMMAND Otherwise initialize an empty list named play_list to accumulate parsed segment dictionaries. RESULT play_list]
            [COMMAND Split the original self.play_text on the literal single-space separator " " to produce play_segs, preserving empty segments created by consecutive spaces. RESULT play_segs]
            [COMMAND Within the loop over each play_seg in play_segs, initialize pos to 0 before scanning the segment. RESULT pos]
            [COMMAND Within the inner loop over each character ele in the current play_seg, if ele.isalpha() is true then increment pos by 1 and continue scanning the next character; otherwise stop the inner loop immediately at the first non-alphabetic character. RESULT leading_alpha_length]
            [COMMAND After the character scan for the current play_seg, set play_chord to the slice play_seg[0:pos], which is the leading alphabetic prefix and may be empty. RESULT play_chord]
            [COMMAND Set play_value to the slice play_seg[pos:], which is the remainder beginning at the first non-alphabetic character or an empty string when the segment is entirely alphabetic. RESULT play_value]
            [COMMAND Append a dictionary {'Chord': play_chord, 'Tune': play_value} to play_list for the current segment, including empty strings when the segment is empty or begins with a non-alphabetic character. RESULT play_list]
            [COMMAND If display is truthy for the current segment, call self.display(play_chord, play_value). RESULT display_side_effect]
            [COMMAND After all segments have been processed, return play_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The stripped self.play_text is empty or contains only whitespace.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list without splitting or parsing any segments. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A segment is empty because splitting on the literal single-space separator produced an empty string.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the inner character scan because there are no characters, leaving pos as 0. RESULT pos]
            [COMMAND Create play_chord as an empty string and play_value as an empty string, then append {'Chord': '', 'Tune': ''} to play_list. RESULT play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A segment begins with a non-alphabetic character.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Stop the inner scan immediately with pos remaining 0. RESULT pos]
            [COMMAND Use an empty string for play_chord and the full segment for play_value, then append the parsed dictionary to play_list. RESULT play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A segment contains only alphabetic characters.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Finish the inner scan without breaking so that pos equals the full segment length. RESULT pos]
            [COMMAND Use the full segment as play_chord and an empty string as play_value, then append the parsed dictionary to play_list. RESULT play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a formatted string describing normal guitar playing by inserting the provided chord key and tune value into a fixed message template." display]
    [INPUTS]
        <REF> key </REF>: any "The chord value inserted into the 'Chord' portion of the output string."
        <REF> value </REF>: any "The tune value inserted into the 'Play Tune' portion of the output string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A formatted string in the form 'Normal Guitar Playing -- Chord: <key>, Play Tune: <value>'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Format the constant message template "Normal Guitar Playing -- Chord: %s, Play Tune: %s" using Python percent-style string substitution with key as the first %s value and value as the second %s value, converting each to string form as needed. RESULT result]
            [COMMAND Return the formatted string directly without any intermediate processing or conditional logic. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
