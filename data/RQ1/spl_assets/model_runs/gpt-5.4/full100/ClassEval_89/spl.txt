# CLASS TwentyFourPointGame

[DEFINE_WORKER: "Initializes the instance by creating an empty list for nums." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new empty list and assign it to self.nums. RESULT self.nums]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates four random integers from 1 to 9 and appends them to self.nums, then verifies that exactly four cards were added." _generate_cards]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate four times over the range from 0 to 3 inclusive. RESULT loop_index]
            [COMMAND On each iteration, append a random integer between 1 and 9 inclusive to self.nums. RESULT self.nums]
            [COMMAND After the loop completes, assert that the length of self.nums is exactly 4. RESULT assertion_passed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Initializes the card list, generates cards, and returns the populated list." get_my_cards]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.nums to an empty list. RESULT self.nums]
            [COMMAND Call self._generate_cards() to populate the card list. RESULT generated_cards]
            [COMMAND Return self.nums. RESULT self.nums]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the player's cards for the special input 'pass', otherwise validates that the expression uses exactly the numbers in self.nums and, if valid, returns the evaluated expression result." answer]
    [INPUTS]
        <REF> expression </REF>: text "Expression string to validate and possibly evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "Returns self.get_my_cards() when expression is 'pass', returns self.evaluate_expression(expression) when the expression uses exactly the available numbers, otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If expression is exactly 'pass', return self.get_my_cards() immediately. RESULT result]
            [COMMAND Initialize statistic as an empty dictionary to count eligible digit characters found in expression. RESULT statistic]
            [COMMAND For each character c in expression, if c.isdigit() is true and int(c) is contained in self.nums, increment statistic[c] by 1 using statistic.get(c, 0) + 1; otherwise do nothing for that character. RESULT statistic]
            [COMMAND Create nums_used as a shallow copy of statistic. RESULT nums_used]
            [COMMAND For each num in self.nums, check whether nums_used.get(str(num), -100) is not equal to -100 and nums_used[str(num)] is greater than 0; if both conditions are true, decrement nums_used[str(num)] by 1, otherwise return False immediately. RESULT nums_used_or_false]
            [COMMAND After the loop completes, if all(count == 0 for count in nums_used.values()) == True, return self.evaluate_expression(expression); otherwise return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Evaluates the input expression and returns true only when its evaluated result equals 24, otherwise returning false including when evaluation raises any exception." evaluate_expression]
    [INPUTS]
        <REF> expression </REF>: text "Expression string to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the evaluated expression equals 24; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to evaluate the input expression using eval(expression). RESULT evaluated_value]
            [COMMAND If the evaluation succeeds and evaluated_value equals 24, return true. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The evaluation succeeds and evaluated_value does not equal 24.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Evaluating expression raises any exception.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
