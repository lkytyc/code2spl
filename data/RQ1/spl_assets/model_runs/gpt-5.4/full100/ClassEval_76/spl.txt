# CLASS SignInSystem

[DEFINE_WORKER: "Initializes the instance user registry as an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty dictionary to self.users as the initial user storage. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a username to the user registry only if it is not already present, returning False for duplicates and True after insertion." add_user]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users mapping."
        <REF> username </REF>: any "Username key to check and add."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False if the username already exists; True if it was added successfully."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is already present in self.users. RESULT exists]
            [COMMAND If username exists in self.users, return False immediately. RESULT result]
            [COMMAND Otherwise, store username in self.users with value False. RESULT self.users[username]]
            [COMMAND Return True immediately after inserting the new username. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username exists in the user map, returns False if it does not, and otherwise marks the user as signed in and returns True." sign_in]
    [INPUTS]
        <REF> username </REF>: any "Username to look up in self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False when the username is missing; True after marking an existing user as signed in."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is present in self.users. RESULT is_known_user]
            [COMMAND If username is not present in self.users, return False immediately. RESULT False]
            [COMMAND Otherwise, set self.users[username] to True to mark the user as signed in. RESULT self.users[username]]
            [COMMAND Return True to indicate successful sign-in. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username exists in the user store and returns the truthiness of its stored value, or False if the username is missing." check_sign_in]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users mapping."
        <REF> username </REF>: any "Username key to verify in the users mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the username exists and its stored value is truthy; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the username is absent from self.users. RESULT is_missing]
            [COMMAND If the username is absent from self.users, return False immediately. RESULT False]
            [COMMAND Otherwise, read self.users[username] and evaluate its truthiness. RESULT user_value_truthy]
            [COMMAND If self.users[username] is truthy, return True. RESULT True]
            [COMMAND If self.users[username] is falsy, return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The username key is not present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without checking any stored user value. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The username key is present in self.users but the stored value is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False after evaluating the stored value. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True when every user value in self.users is truthy, otherwise returns False." all_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Boolean indicating whether every user is signed in."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether all values in self.users are truthy using all(self.users.values()). RESULT all_users_signed_in]
            [COMMAND If all_users_signed_in is true, return True. RESULT True]
            [COMMAND Otherwise, return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect and return the usernames of all users whose signed-in status is falsy." all_not_signed_in]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a users mapping from usernames to signed-in status values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of usernames whose signed-in status evaluated to falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to store usernames of users who are not signed in. RESULT not_signed_in_users]
            [COMMAND Iterate over each username and signed_in status in self.users.items(). RESULT for-loop]
            [COMMAND Within the loop, if signed_in is falsy, append the current username to not_signed_in_users. RESULT not_signed_in_users]
            [COMMAND Return the completed not_signed_in_users list. RESULT not_signed_in_users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
