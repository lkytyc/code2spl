# CLASS TextFileProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided file path on the object." __init__]
    [INPUTS]
        <REF> file_path </REF>: any "The file path value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming file_path value directly to the instance attribute self.file_path. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens the file at self.file_path, parses its JSON contents, and returns the resulting data object." read_file_as_json]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file_path attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> data </REF>: object "Parsed JSON data loaded from the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file located at self.file_path in read mode. RESULT file]
            [COMMAND Load JSON content from the opened file into data. RESULT data]
            [COMMAND Return the parsed data object. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads the contents of the file at the stored file path and returns the entire text." read_file]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file_path attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Full contents of the file as a string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file located at self.file_path in read mode using a context manager. RESULT file]
            [COMMAND Read all text from the opened file. RESULT result]
            [COMMAND Return the read text. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the given content string to the file at the instance's file path, overwriting any existing file contents." write_file]
    [INPUTS]
        <REF> content </REF>: string "Text to write to the file."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.file_path in write mode, which truncates any existing content. RESULT file]
            [COMMAND Write the provided content into the opened file. RESULT file.write(content)]
            [COMMAND Close the file when the context manager exits. RESULT file]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads file content, removes all non-letter characters, writes the cleaned text back to the file, and returns the filtered string." process_file]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The file content after removing all non-alphabetic characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current file content from the input source. RESULT content]
            [COMMAND Reassign content to a new string containing only the alphabetic characters from the previous content by filtering each character with isalpha() and joining the kept characters. RESULT content]
            [COMMAND Write the filtered content back to the file. RESULT written_content]
            [COMMAND Return the filtered content. RESULT content]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
