# CLASS UserLoginDB

[DEFINE_WORKER: "Initializes the database connection and creates a cursor for the new SQLite connection." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "Database name or path passed to SQLite connect."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection using the provided database name and store it in self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection and store it in self.cursor. RESULT self.cursor]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert a new row into the users table using the provided username and password, then commit the transaction." insert_user]
    [INPUTS]
        <REF> username </REF>: data_type "Username value to insert into the users.username column."
        <REF> password </REF>: data_type "Password value to insert into the users.password column."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute an SQL INSERT statement on the users table with two positional placeholders, binding username to the username column and password to the password column. RESULT insert_executed]
            [COMMAND Commit the database connection to persist the inserted user row. RESULT transaction_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a parameterized query to find a user row by username and returns the first matching result or None if no row is found." search_user_by_username]
    [INPUTS]
        <REF> username </REF>: string "The username value used to search for a matching row in the users table."
    [END_INPUTS]

    [OUTPUTS]
        <REF> user </REF>: row_or_none "The first row returned by the query, or None if no matching user exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL query on self.cursor to select all columns from the users table where username equals the provided username value, using a single bound parameter tuple containing username. RESULT query_executed]
            [COMMAND Fetch the next available row from self.cursor after the query execution and assign it directly to user without transformation. RESULT user]
            [COMMAND Return user. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the executed query yields no matching row or no row is available to fetch.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the fetchone() result of self.cursor as None to user. RESULT user]
            [COMMAND Return user as None. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete rows from the users table that match the provided username and commit the database transaction." delete_user_by_username]
    [INPUTS]
        <REF> username </REF>: data_type "The username value bound to the DELETE statement condition."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL DELETE statement against the users table using the constant query "DELETE FROM users WHERE username = ?" and bind the provided username as the single placeholder value. RESULT delete_execution]
            [COMMAND Commit the current database transaction after the DELETE execution to persist any resulting row deletions. RESULT transaction_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a user exists for the given username and whether the stored password matches the provided password, returning a boolean result." validate_user_login]
    [INPUTS]
        <REF> self </REF>: object "Instance providing access to the user lookup method."
        <REF> username </REF>: string "Username used to look up a user record."
        <REF> password </REF>: string "Password to compare against the stored password."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when a user is found and the stored password matches; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the provided username to look up a user record through the instance's user search method. RESULT user]
            [COMMAND If the user record exists and the second element of the record exactly equals the provided password, return true immediately. RESULT True]
            [COMMAND If no matching user exists or the password comparison fails, return false. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
