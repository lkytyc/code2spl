# CLASS DocFileHandler

[DEFINE_WORKER: "Initialize the instance by storing the provided file path in the file_path attribute without modification." __init__]
    [INPUTS]
        <REF> file_path </REF>: data_type "The path value provided to initialize the instance attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter file_path directly to the instance attribute self.file_path without transformation or defaulting. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads all paragraphs from the document at self.file_path and returns them joined into a single newline-separated string." read_text]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the file_path to the document."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The combined text content of all document paragraphs separated by newline characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a document object by loading the file at self.file_path. RESULT doc]
            [COMMAND Initialize an empty list to collect paragraph text. RESULT text]
            [COMMAND For each paragraph in doc.paragraphs, append the paragraph's text to text. RESULT text]
            [COMMAND Join the collected text items with newline characters and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a document, writes the given text with the specified font size and alignment, saves it to self.file_path, and returns whether the operation succeeded." write_text]
    [INPUTS]
        <REF> content </REF>: text "The text content to write into the document."
        <REF> font_size </REF>: number "The font size to apply to the text, defaulting to 12."
        <REF> alignment </REF>: text "The text alignment label to convert and apply to the paragraph, defaulting to 'left'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True if the document is created, formatted, and saved successfully; otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Start a try block for the document creation and save process. RESULT try_context]
            [COMMAND Create a new Document instance. RESULT doc]
            [COMMAND Add a paragraph to the document. RESULT paragraph]
            [COMMAND Add a run containing content to the paragraph. RESULT run]
            [COMMAND Get the font object from the run. RESULT font]
            [COMMAND Set the font size to Pt(font_size). RESULT font_size_applied]
            [COMMAND Call self._get_alignment_value(alignment) to convert the alignment label into a paragraph alignment value. RESULT alignment_value]
            [COMMAND Set the paragraph alignment to alignment_value. RESULT alignment_applied]
            [COMMAND Save the document to self.file_path. RESULT document_saved]
            [COMMAND Return True after the document is saved successfully. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs at any step inside the try block.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception without additional handling details. RESULT caught_exception]
            [COMMAND Return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a Word document from the stored file path, adds a heading at the requested level, saves the document, and returns success or failure." add_heading]
    [INPUTS]
        <REF> heading </REF>: string "Text to add as the heading."
        <REF> level </REF>: integer "Heading level, defaulting to 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the heading was added and the document saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open the Word document at self.file_path. RESULT doc]
            [COMMAND Add the provided heading to the document at the given level. RESULT doc]
            [COMMAND Save the updated document back to self.file_path. RESULT saved_document]
            [COMMAND Return True after successfully completing the update. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any error occurs while opening, modifying, or saving the document, catch the exception and stop processing.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate the heading update failed. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a table in a Word document from a 2D data array, writes each cell as text, saves the document, and returns success unless any error occurs." add_table]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file path for the target document."
        <REF> data </REF>: list "Two-dimensional table data to insert into the document."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the table was added and saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open the document at self.file_path using a Document instance. RESULT doc]
            [COMMAND Create a table with rows equal to len(data) and columns equal to len(data[0]). RESULT table]
            [COMMAND Within the outer loop over each row index i and row value, and the inner loop over each cell index j and cell_value, set table.cell(i, j).text to the string form of cell_value. RESULT table]
            [COMMAND Save the document back to self.file_path. RESULT saved_document]
            [COMMAND Return True to indicate the operation succeeded. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during document creation, table creation, cell population, or saving, catch it and return False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without re-raising the error. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the paragraph alignment constant for a given alignment name, defaulting to left alignment when the name is missing or unrecognized." _get_alignment_value]
    [INPUTS]
        <REF> alignment </REF>: string "Requested alignment name to resolve into a paragraph alignment constant."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: alignment constant "The resolved WD_PARAGRAPH_ALIGNMENT value, defaulting to left alignment."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a lookup dictionary that maps 'left', 'center', and 'right' to their corresponding WD_PARAGRAPH_ALIGNMENT constants. RESULT alignment_options]
            [COMMAND Convert the input alignment name to lowercase and use it to retrieve the matching alignment constant from the lookup dictionary. RESULT alignment_value]
            [COMMAND If the alignment name is not found in the lookup dictionary, return WD_PARAGRAPH_ALIGNMENT.LEFT as the default value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
