# CLASS VendingMachine

[DEFINE_WORKER: "Initialize the object by resetting inventory to an empty dictionary and balance to zero." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.inventory to an empty dictionary, overwriting any previous inventory contents. RESULT self.inventory]
            [COMMAND Set self.balance to the integer 0, overwriting any previous balance value. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to restock an existing inventory item and, if that fails, creates or overwrites the item entry with the provided price and quantity." add_item]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the inventory and restock behavior."
        <REF> item_name </REF>: string "The name of the inventory item to restock or add."
        <REF> price </REF>: number "The price to store for the item if a new inventory entry is created."
        <REF> quantity </REF>: number "The quantity to use when attempting restock and when creating a new inventory entry."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.restock_item with item_name and quantity, and evaluate the truthiness of its return value. RESULT restock_result]
            [COMMAND Negate restock_result to determine whether the item was not restocked successfully. RESULT should_create_item]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If should_create_item is true because self.restock_item(item_name, quantity) returned a falsy value.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.inventory[item_name] to a dictionary containing price mapped to price and quantity mapped to quantity. RESULT self.inventory[item_name]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds the input amount to the object's balance and returns the updated balance." insert_coin]
    [INPUTS]
        <REF> amount </REF>: data_type "The value to add to the current object balance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: data_type "The updated balance after adding the input amount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Add the input amount to self.balance and store the updated value back into self.balance. RESULT self.balance]
            [COMMAND Return the current value of self.balance. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempt to purchase a named inventory item by verifying existence, positive stock, and sufficient balance, then decrement stock, deduct the price, and return the updated balance or return False otherwise." purchase_item]
    [INPUTS]
        <REF> item_name </REF>: string "The name of the item to purchase from the inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number_or_boolean "Returns the updated balance after a successful purchase, or False if the item does not exist, is out of stock, or the balance is insufficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory. RESULT item_exists]
            [COMMAND If item_exists is true, retrieve self.inventory[item_name] and store it as item. RESULT item]
            [COMMAND Within the branch where item exists, check whether item['quantity'] is strictly greater than 0 and self.balance is greater than or equal to item['price']; both conditions must be true. RESULT purchase_allowed]
            [COMMAND Within the branch where purchase_allowed is true, subtract item['price'] from self.balance. RESULT updated_balance_state]
            [COMMAND Within the branch where purchase_allowed is true, decrement item['quantity'] by 1. RESULT updated_item_quantity]
            [COMMAND Within the branch where purchase_allowed is true, return the updated self.balance immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name does not exist in self.inventory.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: item exists, but item['quantity'] is less than or equal to 0 or self.balance is less than item['price'].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Restocks an existing inventory item by increasing its quantity and returns whether the update succeeded." restock_item]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
        <REF> item_name </REF>: string "Name of the inventory item to restock."
        <REF> quantity </REF>: number "Amount to add to the item's existing quantity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the item existed and was restocked; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory. RESULT exists]
            [COMMAND If item_name exists, increase self.inventory[item_name]['quantity'] by quantity. RESULT updated_quantity]
            [COMMAND If item_name exists, return True immediately after updating the quantity. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name is not present in self.inventory.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without changing the inventory. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return False when the inventory is empty or falsy; otherwise build and return a newline-joined text listing each inventory item's name, price, and quantity." display_items]
    [INPUTS]
        <REF> self.inventory </REF>: mapping "Inventory mapping whose keys are item names and whose values contain at least 'price' and 'quantity' fields."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean_or_string "Returns False if the inventory is empty or falsy; otherwise returns a newline-separated string of formatted inventory entries."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.inventory is empty or otherwise falsy. RESULT inventory_is_empty]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.inventory is empty or falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.inventory is not empty]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named items to collect formatted inventory lines. RESULT items]
            [COMMAND Iterate over each item_name and item_info pair from self.inventory.items(). RESULT inventory_entries]
            [COMMAND Within the loop over self.inventory.items(), append the formatted string "{item_name} - ${item_info['price']} [{item_info['quantity']}]" to items for the current entry. RESULT items]
            [COMMAND After processing all inventory entries, join items using the newline character "\n" and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
