# CLASS DecryptionUtils

        <REF> key </REF>: any "The value to assign to the instance key attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided key value to the instance key attribute. RESULT self.key]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypts a Caesar-shifted ciphertext by shifting alphabetic characters backward while preserving case and leaving all non-letter characters unchanged." caesar_decipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> ciphertext </REF>: string "Input text to decrypt."
        <REF> shift </REF>: integer "Number of positions to shift letters backward."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decrypted plaintext."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize plaintext as an empty string. RESULT plaintext]
            [COMMAND For each character in ciphertext, check whether the character is alphabetic. RESULT char]
            [COMMAND Within the loop, if the character is not alphabetic, append it unchanged to plaintext. RESULT plaintext]
            [COMMAND Within the loop, if the character is alphabetic, determine whether it is uppercase or lowercase and set ascii_offset to 65 for uppercase or 97 for lowercase. RESULT ascii_offset]
            [COMMAND Within the loop, compute the shifted character by converting the letter to its ASCII code, subtracting ascii_offset and shift, applying modulo 26, then adding ascii_offset back and converting to a character. RESULT shifted_char]
            [COMMAND Within the loop, append the shifted character to plaintext. RESULT plaintext]
            [COMMAND Return plaintext after all characters have been processed. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypt the input ciphertext with a repeating Vigenere-style key while preserving nonletters and original letter case." vigenere_decipher]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the secret key in self.key."
        <REF> ciphertext </REF>: string "Text to decrypt."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decrypted text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize decrypted_text as an empty string and key_index as 0. RESULT decrypted_text, key_index]
            [COMMAND For each character in ciphertext, check whether the character is alphabetic. RESULT char]
            [COMMAND Within the ciphertext loop, when the character is alphabetic, compute the key shift from self.key at key_index modulo the key length using the lowercase key character offset from 'a'. RESULT shift]
            [COMMAND Within the ciphertext loop, when the character is alphabetic, decrypt the character by shifting its lowercase code point backward by the computed shift modulo 26 and converting it back to a lowercase letter. RESULT decrypted_char]
            [COMMAND Within the ciphertext loop, when the character is alphabetic, append the decrypted letter in uppercase if the original character was uppercase; otherwise append it in lowercase, then increment key_index by 1. RESULT decrypted_text, key_index]
            [COMMAND Within the ciphertext loop, when the character is not alphabetic, append it unchanged to decrypted_text. RESULT decrypted_text]
            [COMMAND Return decrypted_text after processing every character. RESULT decrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypt a rail-fence-ciphered string by reconstructing the zigzag rail grid from the encrypted text and then reading it back along the traversal path to produce the plaintext." rail_fence_decipher]
    [INPUTS]
        <REF> self </REF>: object "Instance context passed to the method and not otherwise used in the decryption logic."
        <REF> encrypted_text </REF>: string "The cipher text whose characters are arranged into the reconstructed rail fence."
        <REF> rails </REF>: integer "The number of rail rows to use when reconstructing the zigzag traversal."
    [END_INPUTS]

    [OUTPUTS]
        <REF> plain_text </REF>: string "The decrypted plaintext reconstructed by following the zigzag path through the filled fence."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize fence as a 2D list with rails rows and len(encrypted_text) columns, filling every cell with '\n', then set direction to -1 and set row and col to 0. RESULT fence, direction, row, col]
            [COMMAND For each iteration from 0 to len(encrypted_text) - 1, if row equals 0 or row equals rails - 1 then negate direction, then mark fence[row][col] as an empty string '', increment col by 1, and increment row by direction to trace the zigzag positions that will hold characters. RESULT fence with zigzag positions marked]
            [COMMAND Initialize index to 0. RESULT index]
            [COMMAND For each rail index i from 0 to rails - 1, and within that for each column index j from 0 to len(encrypted_text) - 1, if fence[i][j] equals '' then assign fence[i][j] = encrypted_text[index] and increment index by 1 so that encrypted characters fill only the previously marked zigzag cells in row-major scan order. RESULT filled fence]
            [COMMAND Initialize plain_text as an empty string, reset direction to -1, and reset row and col to 0. RESULT plain_text, direction, row, col]
            [COMMAND For each iteration from 0 to len(encrypted_text) - 1, if row equals 0 or row equals rails - 1 then negate direction, then append fence[row][col] to plain_text, increment col by 1, and increment row by direction to read the plaintext along the same zigzag traversal path. RESULT plain_text]
            [COMMAND Return plain_text. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
