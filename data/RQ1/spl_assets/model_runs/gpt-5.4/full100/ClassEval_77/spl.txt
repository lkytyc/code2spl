# CLASS Snake

[DEFINE_WORKER: "Initialize a game state with default length, centered starting position, zero score, copied screen settings, and the initial food position." __init__]
    [INPUTS]
        <REF> SCREEN_WIDTH </REF>: number "Screen width used to store dimensions and compute the starting horizontal position."
        <REF> SCREEN_HEIGHT </REF>: number "Screen height used to store dimensions and compute the starting vertical position."
        <REF> BLOCK_SIZE </REF>: number "Block size stored on the instance without modification."
        <REF> food_position </REF>: any "Initial food position copied directly into the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: none "This initializer does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance length to 1 as the default starting size. RESULT self.length]
            [COMMAND Copy SCREEN_WIDTH into the instance screen width field. RESULT self.SCREEN_WIDTH]
            [COMMAND Copy SCREEN_HEIGHT into the instance screen height field. RESULT self.SCREEN_HEIGHT]
            [COMMAND Copy BLOCK_SIZE into the instance block size field. RESULT self.BLOCK_SIZE]
            [COMMAND Initialize the instance positions list with one centered position computed as SCREEN_WIDTH divided by 2 and SCREEN_HEIGHT divided by 2. RESULT self.positions]
            [COMMAND Set the instance score to 0. RESULT self.score]
            [COMMAND Copy food_position into the instance food position field. RESULT self.food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Update the snake head position by applying the movement direction with screen wraparound, then handle food consumption, self-collision reset, or list growth and trimming." move]
    [INPUTS]
        <REF> self </REF>: object "Instance containing positions, food state, size limits, and game configuration."
        <REF> direction </REF>: tuple "Two-value movement vector unpacked into x and y components."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current head position from self.positions[0] and unpack direction into x and y. RESULT cur, x, y]
            [COMMAND Compute the next position by adding x multiplied by self.BLOCK_SIZE to the current x coordinate and wrapping it by self.SCREEN_WIDTH, and by adding y multiplied by self.BLOCK_SIZE to the current y coordinate and wrapping it by self.SCREEN_HEIGHT. RESULT new]
            [COMMAND If new equals self.food_position, call self.eat_food(). RESULT food_consumed]
            [COMMAND If self.positions has more than 2 elements and new is found in self.positions[2:], reset the state by calling self.reset(). RESULT reset]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If self.positions has 2 or fewer elements, or new is not in self.positions[2:], continue with normal movement handling.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Insert new at the front of self.positions. RESULT self.positions]
            [COMMAND If the updated self.positions length is greater than self.length, remove the last position with pop(). RESULT self.positions]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Repeatedly generates a random grid-aligned food position until it no longer collides with any occupied position." random_food_position]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.food_position is present in self.positions. RESULT collision_check]
            [COMMAND While self.food_position is present in self.positions, generate a new random grid-aligned position for self.food_position using an x coordinate from 0 to self.SCREEN_WIDTH // self.BLOCK_SIZE - 1 multiplied by self.BLOCK_SIZE and a y coordinate from 0 to self.SCREEN_HEIGHT // self.BLOCK_SIZE - 1 multiplied by self.BLOCK_SIZE. RESULT self.food_position]
            [COMMAND Exit the loop once self.food_position is no longer present in self.positions. RESULT updated_food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reinitialize the game state to the starting length, centered snake position, zero score, and a new food position." reset]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the length to 1. RESULT self.length]
            [COMMAND Initialize the positions list with a single tuple at the screen center using SCREEN_WIDTH / 2 and SCREEN_HEIGHT / 2. RESULT self.positions]
            [COMMAND Set the score to 0. RESULT self.score]
            [COMMAND Call random_food_position to place food for the reset game state. RESULT self.random_food_position()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Increases the snake length and score, then spawns a new food position." eat_food]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increment self.length by 1. RESULT length_updated]
            [COMMAND Increment self.score by 100. RESULT score_updated]
            [COMMAND Call self.random_food_position() to generate the next food location. RESULT food_repositioned]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
