# CLASS CombinationCalculator

    [INPUTS]
        <REF> datas </REF>: List[str] "List of string data to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input datas list to the instance attribute self.datas. RESULT self.datas]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns 1 for the binomial base cases where m is 0 or n equals m, otherwise computes n choose m using factorials and integer division." count]
    [INPUTS]
        <REF> n </REF>: int "Non-negative integer total count."
        <REF> m </REF>: int "Non-negative integer selected count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The computed binomial coefficient value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether m is 0 or n equals m. RESULT condition_result]
            [COMMAND If the condition is true, return 1 immediately. RESULT 1]
            [COMMAND Otherwise compute math.factorial(n) divided by math.factorial(n - m) multiplied by math.factorial(m) using integer floor division. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m is 0 or n equals m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 1 as the binomial base case. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: n, m, or n - m is not valid for math.factorial, such as a negative value or non-integer input]
        [LOG "math.factorial cannot evaluate the provided arguments."]
        [THROW ValueError "Raised by factorial for invalid arguments."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns False for inputs outside 0 to 63, otherwise returns a bitmask with the lowest n bits set, except that n equals 63 returns positive infinity." count_all]
    [INPUTS]
        <REF> n </REF>: int "Integer input that is validated against the inclusive range 0 to 63."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int or bool or float "False when n is out of range, otherwise the computed bitmask or positive infinity at n برابر 63."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether n is less than 0 or greater than 63. RESULT range_check]
            [COMMAND If n is outside the inclusive range 0 to 63, return False immediately and stop. RESULT False]
            [COMMAND If n is within the inclusive range 0 to 63, compute (1 << n) - 1 when n is not 63, producing a mask with the lowest n bits set. RESULT mask_value]
            [COMMAND If n equals 63, return positive infinity instead of the bitmask value. RESULT float_inf]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build and return the list of combinations produced by the internal selection routine for size m." select]
    [INPUTS]
        <REF> m </REF>: int "The size of the combination elements to generate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "A list of generated string combinations."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Call the internal selection routine starting at index 0 with a length-m placeholder list filled with None, an initial position of 0, and the result accumulator. RESULT result]
            [COMMAND Return the accumulated result list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect all rows returned by selecting each 1-based entry in self.datas and return them as a single aggregated list." select_all]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the datas collection and the select method used to retrieve rows by 1-based index."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "A flattened list containing all rows returned by calling select for every index from 1 through len(self.datas), inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Iterate i over the integer range from 1 to len(self.datas), inclusive, by using range(1, len(self.datas) + 1). RESULT i]
            [COMMAND Within the loop for each current i, call self.select(i) and extend result with all elements returned from that call. RESULT result]
            [COMMAND After completing the loop, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build all valid selections recursively by filling resultList, stopping early to append a copied complete result when the required count exceeds the available length." _select]
    [INPUTS]
        <REF> dataIndex </REF>: int "Starting index in self.datas for the current selection step."
        <REF> resultList </REF>: List[str] "Mutable working list holding the current partial selection."
        <REF> resultIndex </REF>: int "Current position in resultList to fill."
        <REF> result </REF>: List[List[str]] "Accumulator for completed selections."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "Updated accumulator containing copied completed selections."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute resultLen as the length of resultList. RESULT resultLen]
            [COMMAND Compute resultCount as resultIndex plus 1. RESULT resultCount]
            [COMMAND If resultCount is greater than resultLen, append a copy of resultList to result and return immediately. RESULT result]
            [COMMAND Within the for loop over i from dataIndex up to but not including len(self.datas) plus resultCount minus resultLen, assign self.datas[i] into resultList[resultIndex]. RESULT resultList]
            [COMMAND Within the same for loop, recursively call <SPL>_select</SPL> with i plus 1, resultList, resultIndex plus 1, and result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
