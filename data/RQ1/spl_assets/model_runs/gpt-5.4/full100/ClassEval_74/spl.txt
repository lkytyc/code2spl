[DEFINE_WORKER: "Checks whether an address is already in the whitelist, returns False for duplicates, and otherwise appends the address and returns the updated whitelist." add_white_list]
    [INPUTS]
        <REF> addr </REF>: object "The address value to check for membership in the whitelist and append if it is not already present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool_or_list "Returns False if the address is already in the whitelist; otherwise returns the updated whitelist after appending the address."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is already present in self.white_list. RESULT addr_in_white_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: addr is already present in self.white_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.white_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: addr is not present in self.white_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append addr to self.white_list. RESULT updated_white_list_state]
            [COMMAND Return self.white_list after the append operation. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes an address from the whitelist when present and returns the updated whitelist, otherwise returns False." del_white_list]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the whitelist collection."
        <REF> addr </REF>: any "Address value to remove if it exists in the whitelist."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or collection "False if the address is absent; otherwise the updated whitelist after removal."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is present in self.white_list. RESULT membership_check]
            [COMMAND If addr is not present, return False immediately. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: addr is present in self.white_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove addr from self.white_list. RESULT self.white_list]
            [COMMAND Return the updated self.white_list. RESULT self.white_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates an input mapping, rejects malformed or non-whitelisted senders, stores the received address and content, and returns the content on success." recv]
    [INPUTS]
        <REF> info </REF>: dict "Input mapping expected to contain the keys 'addr' and 'content'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int|bool|any "Returns -1 for invalid input structure, False for non-whitelisted addresses, or the received content for allowed addresses."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether info is a dictionary and contains both required keys 'addr' and 'content'; if not, return -1 immediately. RESULT early_reject]
            [COMMAND Extract addr from info['addr'] and content from info['content']. RESULT addr, content]
            [COMMAND If addr is not present in self.white_list, return False without modifying state. RESULT rejected]
            [COMMAND Otherwise, store {'addr': addr, 'content': content} in self.receive_struct and return self.receive_struct['content']. RESULT receive_struct]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dict, or info is missing 'addr', or info is missing 'content'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 as a malformed-input signal. RESULT invalid_input]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: addr is not in self.white_list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False as a whitelist rejection. RESULT blocked]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that the input is a dictionary containing addr and content, then store those two fields in self.send_struct without modification." send]
    [INPUTS]
        <REF> info </REF>: dict "Input dictionary expected to contain addr and content."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Returns an error message string when the input structure is invalid; otherwise updates internal state and returns nothing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether info is a dictionary and contains both addr and content keys. RESULT validation_passed]
            [COMMAND If the validation fails, return the message "info structure is not correct" immediately. RESULT error_message]
            [COMMAND Create a new dictionary for self.send_struct with addr copied from info["addr"] and content copied from info["content"]. RESULT self.send_struct]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the matching stored structure for the given type value, using send and receive cases with False as the fallback." show]
    [INPUTS]
        <REF> self </REF>: object "Instance holding the send_struct and receive_struct attributes"
        <REF> type </REF>: string "Selector value used to choose which structure to return"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object|boolean "send_struct for send, receive_struct for receive, otherwise False"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether type equals "send". RESULT condition_send]
            [COMMAND If the send condition is true, return self.send_struct immediately. RESULT return_self_send_struct]
            [COMMAND Otherwise check whether type equals "receive". RESULT condition_receive]
            [COMMAND If the receive condition is true, return self.receive_struct immediately. RESULT return_self_receive_struct]
            [COMMAND If neither condition matches, return False. RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
