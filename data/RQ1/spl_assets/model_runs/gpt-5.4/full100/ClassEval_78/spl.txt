# CLASS SplitSentence

[DEFINE_WORKER: "Split an input string into sentences at whitespace following periods or question marks, while avoiding splits inside common abbreviation-like patterns." split_sentences]
    [INPUTS]
        <REF> sentences_string </REF>: string "Input text to be split into sentences."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of strings "List of sentence fragments produced by the regex split."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the fixed regular expression pattern to split sentences_string at whitespace that follows either a period or a question mark, while preserving abbreviation-like patterns protected by negative lookbehinds. RESULT sentences]
            [COMMAND Return the resulting list of sentence fragments. RESULT sentences]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes non-letter and non-whitespace characters from the input sentence, splits the cleaned text into words, and returns the word count." count_words]
    [INPUTS]
        <REF> sentence </REF>: string "Input text to sanitize and count words in."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The number of whitespace-separated words after filtering punctuation and other non-letter characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove every character from sentence except English letters and whitespace using the regular expression r'[^a-zA-Z\s]'. RESULT sentence]
            [COMMAND Split the cleaned sentence on whitespace to produce the words list. RESULT words]
            [COMMAND Return the number of items in words. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the maximum word count among all sentences produced from the input text." process_text_file]
    [INPUTS]
        <REF> sentences_string </REF>: string "Input text that is split into sentences and analyzed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The largest word count found in any sentence, or 0 when no sentence exceeds the initial default."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input string into sentences using the internal sentence splitter. RESULT sentences]
            [COMMAND Initialize the running maximum word count to 0 as the boundary default. RESULT max_count]
            [COMMAND Within the loop over each sentence in sentences, count the words in the current sentence. RESULT count]
            [COMMAND Within the same loop, if the current word count is strictly greater than max_count, update max_count to that count. RESULT max_count]
            [COMMAND After processing all sentences, return max_count. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
