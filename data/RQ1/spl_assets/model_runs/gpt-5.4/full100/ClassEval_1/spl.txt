# CLASS AreaCalculator

[DEFINE_WORKER: "Initialize the instance by storing the provided radius value directly on the object without validation or transformation." __init__]
    [INPUTS]
        <REF> radius </REF>: any "The value to assign to the instance radius attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter radius directly to the instance attribute self.radius. RESULT self.radius]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the circle area by multiplying pi by the square of the object's radius." calculate_circle_area]
    [INPUTS]
        <REF> self </REF>: object "Current object instance providing the radius attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "Area of the circle computed as math.pi multiplied by self.radius squared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the radius value from self.radius. RESULT radius]
            [COMMAND Square the radius value as radius ** 2. RESULT squared_radius]
            [COMMAND Multiply math.pi by squared_radius to compute the circle area. RESULT area]
            [COMMAND Return the computed area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes and returns the surface area of a sphere using the object's radius and the formula 4πr²." calculate_sphere_area]
    [INPUTS]
        <REF> self </REF>: object "Object instance providing the radius attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed surface area of the sphere."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance attribute self.radius. RESULT radius]
            [COMMAND Compute the square of radius. RESULT radius_squared]
            [COMMAND Multiply radius_squared by math.pi. RESULT pi_radius_squared]
            [COMMAND Multiply pi_radius_squared by 4. RESULT sphere_area]
            [COMMAND Return sphere_area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the cylinder surface area expression using the instance radius, the given height, and pi." calculate_cylinder_area]
    [INPUTS]
        <REF> self.radius </REF>: numeric "The cylinder radius stored on the instance and used both as a multiplicative factor and within the sum with height."
        <REF> height </REF>: numeric "The cylinder height provided to the method and added to the radius before multiplication."
        <REF> math.pi </REF>: numeric constant "The mathematical constant pi read from the math module."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The computed value of 2 * math.pi * self.radius * (self.radius + height)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.radius and the provided height, then compute the sum self.radius + height. RESULT radius_plus_height]
            [COMMAND Multiply 2 by math.pi, then multiply by self.radius, and then multiply by radius_plus_height to form 2 * math.pi * self.radius * (self.radius + height). RESULT cylinder_area]
            [COMMAND Return the computed cylinder_area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the sector area by squaring the object's radius, multiplying by the input angle, and dividing by 2." calculate_sector_area]
    [INPUTS]
        <REF> self.radius </REF>: numeric "The current radius value read from the object state."
        <REF> angle </REF>: numeric "The angle value used to scale the squared radius."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The computed sector area value equal to self.radius squared times angle divided by 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the radius value from the object state as self.radius. RESULT radius]
            [COMMAND Compute the square of the radius as self.radius ** 2. RESULT squared_radius]
            [COMMAND Multiply the squared radius by angle. RESULT scaled_area]
            [COMMAND Divide the multiplied result by the constant 2. RESULT result]
            [COMMAND Return the computed result to the caller. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate and return the area of an annulus by multiplying pi by the difference between the squares of the outer and inner radii." calculate_annulus_area]
    [INPUTS]
        <REF> self </REF>: object "Instance context passed to the method and not used in the calculation."
        <REF> inner_radius </REF>: number "The inner radius whose square is subtracted from the outer radius square."
        <REF> outer_radius </REF>: number "The outer radius whose square is used as the outer area term."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed annulus area equal to pi times the difference between the squared outer and inner radii."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Square outer_radius to compute the outer radius area term base. RESULT outer_radius_squared]
            [COMMAND Square inner_radius to compute the inner radius area term base. RESULT inner_radius_squared]
            [COMMAND Subtract inner_radius_squared from outer_radius_squared to get the annulus radial area difference. RESULT area_difference]
            [COMMAND Multiply area_difference by math.pi and return the computed annulus area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
