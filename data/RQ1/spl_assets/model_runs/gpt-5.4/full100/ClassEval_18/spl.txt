# CLASS CamelCaseMap

[DEFINE_WORKER: "Initialize the instance by assigning a new empty dictionary to self._data." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance being initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary to the instance attribute self._data, unconditionally initializing or resetting its stored data state. RESULT self._data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert a string key to camel case and otherwise return the key unchanged." _convert_key]
    [INPUTS]
        <REF> key </REF>: any "The input key to inspect and possibly transform."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The camel-cased key if the input is a string; otherwise the original key unchanged."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether key is an instance of str. RESULT is_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If key is a string.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self._to_camel_case with key and return the converted value immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If key is not a string.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return key unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert an underscore-delimited string into camelCase by keeping the first segment unchanged and title-casing each subsequent segment before concatenation." _to_camel_case]
    [INPUTS]
        <REF> key </REF>: string "Input text expected to support splitting on underscores."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The camelCase-style string formed from the underscore-separated parts of the input."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input key on the literal underscore character '_' and store the resulting ordered list of segments in parts. RESULT parts]
            [COMMAND Take the first segment parts[0] unchanged, then for each segment in parts[1:], convert it to title case and join all converted segments into one string. RESULT suffix]
            [COMMAND Concatenate the unchanged first segment with the joined title-cased suffix segments and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The input key contains no underscore characters.]
        [SEQUENTIAL_BLOCK]
            [COMMAND After splitting, use the only segment as parts[0], the remaining slice parts[1:] is empty, and return the original key unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The input key is empty or contains leading, trailing, or repeated underscores.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Preserve the split behavior that produces empty-string segments, including an initial empty first segment or empty later segments, then apply the same concatenation logic with title-casing on segments in parts[1:]. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
