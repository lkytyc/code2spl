# CLASS IpUtil

[DEFINE_WORKER: "Checks whether a given string is a valid IPv4 address by attempting an IPv4-specific socket parse and returning a boolean based on whether a socket error occurs." is_valid_ipv4]
    [INPUTS]
        <REF> ip_address </REF>: string "The candidate IP address string to validate as IPv4."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the input can be parsed as a valid IPv4 address; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to parse ip_address as an IPv4 address using socket.inet_pton with the socket.AF_INET address family. RESULT ipv4_parse_attempt]
            [COMMAND If the IPv4 parse attempt succeeds without raising socket.error, return True immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The call to socket.inet_pton with socket.AF_INET and ip_address raises socket.error.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the socket.error raised during IPv4 parsing. RESULT caught_socket_error]
            [COMMAND Return False as the validation result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate whether the input string is a syntactically valid IPv6 address by attempting an IPv6 socket conversion and returning a boolean result." is_valid_ipv6]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address string to validate as an IPv6 address."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the input can be parsed as an IPv6 address, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to parse the input ip_address as an IPv6 address by calling socket.inet_pton with socket.AF_INET6 and ip_address. RESULT ipv6_parse_attempt]
            [COMMAND If the IPv6 parse attempt succeeds without raising an exception, return True immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The call to socket.inet_pton with socket.AF_INET6 and ip_address raises socket.error.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the socket.error from the IPv6 parse attempt and return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resolve an IP address to its primary hostname using reverse DNS and return null-equivalent None when the lookup raises a socket.herror." get_hostname]
    [INPUTS]
        <REF> ip_address </REF>: string "IP address to resolve to a hostname via reverse DNS lookup"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string_or_none "Primary hostname for the IP address if reverse lookup succeeds, otherwise None"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to perform a reverse DNS lookup for the given ip_address using socket.gethostbyaddr(ip_address). RESULT host_lookup_result]
            [COMMAND From the successful lookup result, take the element at index 0 as the primary hostname. RESULT hostname]
            [COMMAND Return the primary hostname. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The reverse DNS lookup raises socket.herror.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None instead of a hostname. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
