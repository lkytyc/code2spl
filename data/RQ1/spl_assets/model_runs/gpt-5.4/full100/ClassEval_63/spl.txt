# CLASS NLPDataProcessor2

[DEFINE_WORKER: "Process each input string by lowercasing it, removing all non-English-letter characters except whitespace, splitting it into words, and returning the resulting list of word lists." process_data]
    [INPUTS]
        <REF> string_list </REF>: list "A list of strings to clean and tokenize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list "A list where each element is the whitespace-split word list produced from the corresponding cleaned input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize words_list as an empty list. RESULT words_list]
            [COMMAND For each string in string_list, convert the string to lowercase, remove every character that is not an English letter A-Z or a-z and not whitespace using the regular expression [^a-zA-Z\s], and store the cleaned text in processed_string. RESULT processed_string]
            [COMMAND Within the loop for each string in string_list, split processed_string on whitespace into a list of words. RESULT words]
            [COMMAND Within the loop for each string in string_list, append words to words_list. RESULT words_list]
            [COMMAND After all elements in string_list have been processed with no iteration beyond the last element, return words_list. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a word-frequency map from a list of word collections, sorts the counts in descending order, and returns the top five entries." calculate_word_frequency]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> words_list </REF>: iterable "A sequence of word collections to count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "A dictionary containing up to the five most frequent words and their counts."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty Counter to accumulate word counts. RESULT word_frequency]
            [COMMAND For each words collection in words_list, update the Counter with that collection. RESULT word_frequency]
            [COMMAND Sort word_frequency items by count in descending order and convert the sorted pairs into a dictionary. RESULT sorted_word_frequency]
            [COMMAND Take the first 5 items from sorted_word_frequency and convert them into a dictionary. RESULT top_5_word_frequency]
            [COMMAND Return top_5_word_frequency. RESULT top_5_word_frequency]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes the input string list into a word collection and then computes and returns a word-frequency mapping from that processed data." process]
    [INPUTS]
        <REF> string_list </REF>: data "Incoming string-based collection or source data to be processed into words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> word_frequency_dict </REF>: mapping "Word-frequency dictionary computed from the processed words list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.process_data with string_list to transform the input data into a processed word collection. RESULT words_list]
            [COMMAND Call self.calculate_word_frequency with words_list to compute the frequency mapping for the processed words. RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: string_list is empty, null-like, malformed, or contains unexpected content, and self.process_data handles such input without raising an exception.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Process string_list through self.process_data and use whatever resulting words_list that function produces for the given boundary case. RESULT words_list]
            [COMMAND Compute word frequencies from the resulting words_list using self.calculate_word_frequency. RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: words_list is empty or otherwise reflects a boundary-case result from preprocessing, and self.calculate_word_frequency handles it without raising an exception.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the frequency mapping from words_list using self.calculate_word_frequency, which may produce an empty mapping for an empty words_list. RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
