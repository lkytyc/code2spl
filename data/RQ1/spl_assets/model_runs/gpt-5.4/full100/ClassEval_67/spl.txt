# CLASS Order

[DEFINE_WORKER: "Initialize a new instance with empty menu, empty selected dish list, and empty sales records containers." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance attribute self.menu as an empty list to hold menu item dictionaries with dish, price, and count fields. RESULT self.menu]
            [COMMAND Initialize the instance attribute self.selected_dishes as an empty list to hold selected dish dictionaries with dish, count, and price fields. RESULT self.selected_dishes]
            [COMMAND Initialize the instance attribute self.sales as an empty dictionary to hold sales aggregation or sales record mappings. RESULT self.sales]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks the menu for a matching dish, rejects the request if the available count is too low, otherwise decreases the stored count and adds the dish to the selected list." add_dish]
    [INPUTS]
        <REF> dish </REF>: dict "A dish record containing at least a dish name and a requested count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the dish is accepted and added to the selected list; False when the requested count exceeds available menu stock."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each menu_dish in self.menu and compare dish["dish"] with menu_dish["dish"]. RESULT matched_menu_dish_search]
            [COMMAND When a matching menu_dish is found, if menu_dish["count"] is less than dish["count"], return False immediately. RESULT false_return]
            [COMMAND When a matching menu_dish is found and menu_dish["count"] is at least dish["count"], subtract dish["count"] from menu_dish["count"] and break out of the loop. RESULT updated_menu_dish]
            [COMMAND After the loop completes, append dish to self.selected_dishes and return True. RESULT true_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the total sales-adjusted cost of all selected dishes by summing each dish's price multiplied by its count and the sales multiplier for that dish." calculate_total]
    [INPUTS]
        <REF> self.selected_dishes </REF>: list "A collection of dish records, where each record provides at least 'price', 'count', and 'dish' keys."
        <REF> self.sales </REF>: mapping "A mapping from dish name to the sales multiplier used in the total calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> total </REF>: number "The accumulated total of price times count times sales multiplier across all selected dishes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to the numeric constant 0 as the accumulator before processing any selected dishes. RESULT total]
            [COMMAND Iterate over each dish record in self.selected_dishes. RESULT dish]
            [COMMAND Within the loop for the current dish record, read dish["price"], dish["count"], and dish["dish"], then use dish["dish"] as the key to read the corresponding multiplier from self.sales. RESULT current_dish_factors]
            [COMMAND Within the loop for the current dish record, compute dish["price"] * dish["count"] * self.sales[dish["dish"]]. RESULT dish_total]
            [COMMAND Within the loop for the current dish record, add the computed dish_total to total using cumulative assignment. RESULT total]
            [COMMAND After all dish records have been processed, return total. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return False when no dishes are selected; otherwise compute the current total, clear the selected dishes list, and return the computed total." checkout]
    [INPUTS]
        <REF> self </REF>: object "The instance whose selected dishes are checked, totaled, and then cleared."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool_or_calculated_total "Returns False if there are no selected dishes; otherwise returns the computed total before clearing the selected dishes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the length of self.selected_dishes is equal to 0. RESULT is_empty]
            [COMMAND If the selected dishes list is not empty, call self.calculate_total() and store the returned value in total. RESULT total]
            [COMMAND After computing total, assign a new empty list to self.selected_dishes to reset the selection state. RESULT self.selected_dishes]
            [COMMAND Return total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The length of self.selected_dishes is equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without calculating a total or modifying self.selected_dishes. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
