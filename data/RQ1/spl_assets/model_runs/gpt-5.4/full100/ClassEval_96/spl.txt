# CLASS WeatherSystem

[DEFINE_WORKER: "Initializes instance state by setting temperature and weather to None, storing the provided city, and creating an empty weather list dictionary." __init__]
    [INPUTS]
        <REF> city </REF>: any "City value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "No return value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set instance attribute self.temperature to None. RESULT self.temperature]
            [COMMAND Set instance attribute self.weather to None. RESULT self.weather]
            [COMMAND Assign the input city value to instance attribute self.city. RESULT self.city]
            [COMMAND Initialize instance attribute self.weather_list as an empty dictionary. RESULT self.weather_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Stores the provided weather data, validates that the current city exists, and returns either the current or unit-converted temperature together with the city's weather." query]
    [INPUTS]
        <REF> weather_list </REF>: mapping "Weather data keyed by city, including temperature, weather, and temperature units."
        <REF> tmp_units </REF>: string "Requested temperature unit, defaulting to 'celsius'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean_or_tuple_or_none "Returns False if the current city is absent; otherwise returns a tuple of temperature and weather when units already match or when conversion is performed; if units differ and tmp_units is neither 'celsius' nor 'fahrenheit', execution falls through with no explicit return."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input weather_list directly to self.weather_list. RESULT self.weather_list]
            [COMMAND Check whether self.city is not present in the input weather_list. RESULT city_missing]
            [COMMAND If city_missing is true, return False immediately. RESULT False]
            [COMMAND If city_missing is false, read self.weather_list[self.city]['temperature'] and assign it to self.temperature. RESULT self.temperature]
            [COMMAND If city_missing is false, read self.weather_list[self.city]['weather'] and assign it to self.weather. RESULT self.weather]
            [COMMAND Compare self.weather_list[self.city]['temperature units'] against tmp_units to determine whether the stored unit is different from the requested unit. RESULT units_differ]
            [COMMAND If units_differ is false, return the tuple (self.temperature, self.weather). RESULT temperature_and_weather]
            [COMMAND If units_differ is true and tmp_units equals 'celsius', call self.fahrenheit_to_celsius() and return the tuple (converted temperature, self.weather). RESULT converted_celsius_and_weather]
            [COMMAND If units_differ is true and tmp_units equals 'fahrenheit', call self.celsius_to_fahrenheit() and return the tuple (converted temperature, self.weather). RESULT converted_fahrenheit_and_weather]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.city is not present in weather_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without reading city-specific weather fields. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.weather_list[self.city]['temperature units'] equals tmp_units]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the existing self.temperature together with self.weather without conversion. RESULT temperature_and_weather]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.weather_list[self.city]['temperature units'] does not equal tmp_units and tmp_units equals 'celsius']
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the stored temperature by calling self.fahrenheit_to_celsius(). RESULT converted_temperature]
            [COMMAND Return the tuple of converted_temperature and self.weather. RESULT converted_celsius_and_weather]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.weather_list[self.city]['temperature units'] does not equal tmp_units and tmp_units equals 'fahrenheit']
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the stored temperature by calling self.celsius_to_fahrenheit(). RESULT converted_temperature]
            [COMMAND Return the tuple of converted_temperature and self.weather. RESULT converted_fahrenheit_and_weather]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.weather_list[self.city]['temperature units'] does not equal tmp_units and tmp_units is neither 'celsius' nor 'fahrenheit']
        [SEQUENTIAL_BLOCK]
            [COMMAND Perform no conversion branch and fall through with no explicit return from this method body. RESULT implicit_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Assign the provided city value directly to the instance attribute city." set_city]
    [INPUTS]
        <REF> self </REF>: object "Instance whose city attribute will be updated."
        <REF> city </REF>: any "Value to store in the city attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input city value directly to self.city. RESULT self.city]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the instance temperature from Celsius to Fahrenheit using the standard linear formula." celsius_to_fahrenheit]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the temperature value in Celsius via self.temperature"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The temperature converted to Fahrenheit"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the temperature value from self.temperature. RESULT temperature_celsius]
            [COMMAND Multiply the Celsius temperature by 9/5. RESULT scaled_temperature]
            [COMMAND Add 32 to the scaled value to convert it to Fahrenheit. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert the instance temperature from Fahrenheit to Celsius by subtracting 32 and multiplying the result by 5/9." fahrenheit_to_celsius]
    [INPUTS]
        <REF> self.temperature </REF>: numeric "The instance temperature value in degrees Fahrenheit used for the conversion."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The converted temperature value in degrees Celsius."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.temperature from the instance. RESULT temperature_fahrenheit]
            [COMMAND Subtract 32 from temperature_fahrenheit. RESULT fahrenheit_offset]
            [COMMAND Multiply fahrenheit_offset by 5/9 to compute the Celsius temperature. RESULT celsius_value]
            [COMMAND Return celsius_value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
