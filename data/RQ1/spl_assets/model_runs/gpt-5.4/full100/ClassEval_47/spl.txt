# CLASS IPAddress

[DEFINE_WORKER: "Initialize the instance by storing the provided ip_address parameter directly into the object's ip_address attribute." __init__]
    [INPUTS]
        <REF> ip_address </REF>: data "The input value to assign directly to the instance ip_address attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided ip_address value directly to self.ip_address without validation, transformation, or parsing. RESULT self.ip_address]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that self.ip_address is an IPv4-style dotted string by splitting it into exactly four octets and ensuring each octet is a digit-only value in the inclusive range 0 to 255." is_valid]
    [INPUTS]
        <REF> self.ip_address </REF>: string "The IP address string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True when the IP address contains exactly four dot-separated octets and every octet is a digit-only number from 0 through 255 inclusive; otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address on the literal '.' character and store the resulting list of substrings in octets. RESULT octets]
            [COMMAND Check whether the number of elements in octets is not equal to 4. RESULT octet_count_invalid]
            [COMMAND If octet_count_invalid is true, return False immediately. RESULT result]
            [COMMAND Iterate over each octet in octets in order. RESULT octet]
            [COMMAND Within the loop for each octet, evaluate whether octet is not digit-only, or if digit-only whether int(octet) is less than 0, or if still needed whether int(octet) is greater than 255, using short-circuit OR semantics. RESULT octet_invalid]
            [COMMAND Within the loop, if octet_invalid is true, return False immediately. RESULT result]
            [COMMAND After the loop completes without any early return and the octet count check has passed, return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The split operation produces a list whose length is not exactly 4.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without validating individual octet contents. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During iteration, any octet is empty, contains any non-digit characters, or represents an integer outside the inclusive range 0 to 255.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False at the first invalid octet and stop further iteration. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the IP address split into dot-separated octets when the object is valid, otherwise return an empty list." get_octets]
    [INPUTS]
        <REF> self </REF>: object "Object that provides validity checking and an ip_address string attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of dot-separated IP address components if valid, or an empty list if not valid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.is_valid() returns true. RESULT is_valid]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address using '.' as the separator and return the resulting list immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the object's octets as dot-separated 8-bit binary strings when the value is valid, otherwise return an empty string." get_binary]
    [INPUTS]
        <REF> self </REF>: object "Object that provides validity checking and access to octet values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A dot-separated string of 8-bit binary octets for a valid value, or an empty string for an invalid value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.is_valid() returns true. RESULT is_valid]
            [COMMAND If the value is valid, initialize an empty list named binary_octets to collect binary octet strings. RESULT binary_octets]
            [COMMAND If the value is valid, iterate over each octet returned by self.get_octets(). RESULT octet]
            [COMMAND Within the valid-case loop, convert the current octet to an integer and format it as an 8-character binary string using the format specifier '08b'. RESULT binary_octet]
            [COMMAND Within the valid-case loop, append the formatted 8-bit binary string to binary_octets. RESULT binary_octets]
            [COMMAND After all octets have been processed in the valid case, join binary_octets with '.' and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
