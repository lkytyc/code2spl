# CLASS EncryptionUtils

[DEFINE_WORKER: "Initializes the object by storing the provided key value on the instance." __init__]
    [INPUTS]
        <REF> key </REF>: any "Value to assign to the instance key attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input key to the instance attribute self.key. RESULT self.key]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypt the input text with a Caesar cipher by shifting alphabetic characters by the given amount while leaving non-letters unchanged." caesar_cipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> plaintext </REF>: string "Text to encrypt."
        <REF> shift </REF>: integer "Number of positions to shift each alphabetic character."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The transformed ciphertext."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty ciphertext string. RESULT ciphertext]
            [COMMAND For each character in plaintext, process it in sequence. RESULT char]
            [COMMAND Within the loop, if the character is alphabetic, determine whether it is uppercase or lowercase. RESULT case_check]
            [COMMAND Within the alphabetic branch, use ASCII offset 65 for uppercase letters and 97 for lowercase letters. RESULT ascii_offset]
            [COMMAND Within the alphabetic branch, compute the shifted character with wraparound using modulo 26 and append it to ciphertext. RESULT shifted_char]
            [COMMAND Within the loop, if the character is not alphabetic, append it unchanged to ciphertext. RESULT ciphertext]
            [COMMAND After all characters are processed, return ciphertext. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypt the input text with a repeating Vigenere key while preserving non-alphabetic characters in place and retaining each letter's original case." vigenere_cipher]
    [INPUTS]
        <REF> plain_text </REF>: text "The plaintext string to encrypt character by character."
    [END_INPUTS]

    [OUTPUTS]
        <REF> encrypted_text </REF>: text "The Vigenere-encrypted string with non-letter characters unchanged and letter case preserved."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize encrypted_text to the empty string "". RESULT encrypted_text]
            [COMMAND Initialize key_index to 0 so key consumption starts from the first key character. RESULT key_index]
            [COMMAND Iterate left to right over each char in plain_text. RESULT char]
            [COMMAND Within the loop, check whether char.isalpha() is true. RESULT is_alpha]
            [COMMAND Within the loop when char is alphabetic, compute shift as ord(self.key[key_index % len(self.key)].lower()) - ord('a') so the repeating key character maps to the range 0 through 25. RESULT shift]
            [COMMAND Within the loop when char is alphabetic, compute encrypted_char as chr((ord(char.lower()) - ord('a') + shift) % 26 + ord('a')) so the lowercase plaintext letter is shifted with modulo 26 wraparound. RESULT encrypted_char]
            [COMMAND Within the loop when char is alphabetic, append encrypted_char.upper() to encrypted_text if char.isupper() is true; otherwise append encrypted_char unchanged. RESULT encrypted_text]
            [COMMAND Within the loop when char is alphabetic, increment key_index by 1 so the key advances only after encrypting a letter. RESULT key_index]
            [COMMAND Within the loop when char is not alphabetic, append char unchanged to encrypted_text and leave key_index unchanged. RESULT encrypted_text]
            [COMMAND After all characters have been processed, return encrypted_text. RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The current character in the loop is not alphabetic.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append the original non-alphabetic character to encrypted_text without modification. RESULT encrypted_text]
            [COMMAND Do not increment key_index so non-letter characters do not consume key characters. RESULT key_index]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The current character in the loop is alphabetic and char.isupper() is true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert encrypted_char to uppercase before appending it to encrypted_text. RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The current character in the loop is alphabetic and char.isupper() is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append encrypted_char in lowercase form to encrypted_text. RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypts the input text by placing characters into a rail-fence zigzag grid and then reading the grid row by row." rail_fence_cipher]
    [INPUTS]
        <REF> plain_text </REF>: string "Text to encrypt."
        <REF> rails </REF>: integer "Number of rails used for the zigzag pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Encrypted rail-fence cipher text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a 2D fence grid with rails rows and len(plain_text) columns, filling every cell with newline characters. RESULT fence]
            [COMMAND Initialize the traversal direction to -1 and set row and col to 0 to start at the top-left position. RESULT direction, row, col]
            [COMMAND Within the loop over each character in plain_text, if the current row is at the top rail 0 or the bottom rail rails - 1, reverse direction by negating the current direction. RESULT direction]
            [COMMAND Within the same loop, place the current character into fence[row][col], then increment col by 1 and update row by adding the current direction. RESULT fence, row, col]
            [COMMAND Initialize encrypted_text as an empty string. RESULT encrypted_text]
            [COMMAND Within the outer loop over i from 0 to rails - 1, and the inner loop over j from 0 to len(plain_text) - 1, append fence[i][j] to encrypted_text only when the cell is not a newline character. RESULT encrypted_text]
            [COMMAND Return encrypted_text as the final encrypted output. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
