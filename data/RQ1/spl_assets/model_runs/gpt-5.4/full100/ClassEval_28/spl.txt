# CLASS DatabaseProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided database name on the object without modification." __init__]
    [INPUTS]
        <REF> database_name </REF>: data_type "Incoming database name value to store as an instance attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input value database_name directly to the instance attribute self.database_name without validation, transformation, or default handling. RESULT self.database_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Connect to the SQLite database, build a CREATE TABLE IF NOT EXISTS statement from the provided table and column names, execute it, commit the schema change, and close the connection." create_table]
    [INPUTS]
        <REF> table_name </REF>: string "Name of the table to create if it does not already exist."
        <REF> key1 </REF>: string "Name of the first dynamic column, created with TEXT type."
        <REF> key2 </REF>: string "Name of the second dynamic column, created with INTEGER type."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite database connection using self.database_name. RESULT conn]
            [COMMAND Create a cursor from the opened database connection. RESULT cursor]
            [COMMAND Construct the SQL statement "CREATE TABLE IF NOT EXISTS {table_name} (id INTEGER PRIMARY KEY, {key1} TEXT, {key2} INTEGER)" by directly interpolating table_name, key1, and key2 into the table and column identifier positions. RESULT create_table_query]
            [COMMAND Execute the constructed CREATE TABLE IF NOT EXISTS statement with the cursor. RESULT table_creation_executed]
            [COMMAND Commit the database transaction to persist the schema change. RESULT transaction_committed]
            [COMMAND Close the database connection. RESULT connection_closed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts each record from the input data into the specified database table using the connection tied to the instance database name." insert_into_database]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database_name attribute."
        <REF> table_name </REF>: string "Target database table name used in the INSERT statement."
        <REF> data </REF>: iterable "Collection of items containing name and age fields to insert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "Does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection using self.database_name. RESULT conn]
            [COMMAND Create a database cursor from the connection. RESULT cursor]
            [COMMAND For each item in data, build an INSERT statement for the given table_name with name and age placeholders. RESULT insert_query]
            [COMMAND For each item in data, execute the INSERT statement with item['name'] and item['age'] as bound values. RESULT row_inserted]
            [COMMAND Commit all pending database changes. RESULT committed]
            [COMMAND Close the database connection. RESULT closed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches a SQLite table by name using a parameterized query and returns all matching rows, or None when no rows are found." search_database]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the SQLite database name."
        <REF> table_name </REF>: string "Name of the table to query."
        <REF> name </REF>: string "Value used to filter rows by the name column."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list or None "All matching rows when the query returns rows; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using self.database_name and store the connection in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Build a SQL query that selects all columns from the given table_name where name equals a bound parameter, and store it in select_query. RESULT select_query]
            [COMMAND Execute select_query on cursor with name supplied as the single bound parameter. RESULT executed_cursor]
            [COMMAND Fetch all remaining rows from cursor and store them in result. RESULT result]
            [COMMAND If result is truthy, return result immediately. RESULT result]
            [COMMAND Otherwise return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When fetchall returns an empty list, result is falsy and the method takes the None branch.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None instead of any rows. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete a row from the specified SQLite table by matching the given name, then commit and close the database connection." delete_from_database]
    [INPUTS]
        <REF> table_name </REF>: string "Name of the database table to delete from."
        <REF> name </REF>: string "Value matched against the name column for deletion."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using the configured database name and create a cursor from that connection. RESULT conn, cursor]
            [COMMAND Build a parameterized DELETE statement for the given table name with a name filter using a bound placeholder. RESULT delete_query]
            [COMMAND Execute the DELETE statement with the provided name as the bound parameter. RESULT delete_result]
            [COMMAND Commit the transaction so the deletion is saved to the database. RESULT commit_result]
            [COMMAND Close the database connection. RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
