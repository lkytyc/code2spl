# CLASS CSVProcessor

[DEFINE_WORKER: "Initialize the object without performing any additional setup." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create the object and perform no initialization beyond the default constructor behavior. RESULT self]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a CSV file, stores all data rows after the header, and returns the header and collected rows." read_csv]
    [INPUTS]
        <REF> file_name </REF>: string "Path to the CSV file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the header row and a list of data rows."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named data to store CSV rows. RESULT data]
            [COMMAND Open the file at file_name in read mode as file. RESULT file]
            [COMMAND Create a CSV reader from file. RESULT reader]
            [COMMAND Read the first row from reader as the header title. RESULT title]
            [COMMAND For each remaining row in reader, append the row to data. RESULT data]
            [COMMAND Return title and data. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes rows of data to a CSV file and returns 1 on success or 0 if any exception occurs." write_csv]
    [INPUTS]
        <REF> data </REF>: list "Rows of data to write to the CSV file."
        <REF> file_name </REF>: string "Path to the output CSV file."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "1 if the CSV write succeeds, otherwise 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open file_name in write mode with newline set to an empty string. RESULT file]
            [COMMAND Create a CSV writer for the opened file. RESULT writer]
            [COMMAND Write all rows from data to the CSV file using the writer. RESULT written_rows]
            [COMMAND Close the file automatically when the with-block ends. RESULT file_closed]
            [COMMAND Return 1 to indicate success. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs while opening the file or writing the CSV data.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Handle the exception by skipping normal completion. RESULT exception_handled]
            [COMMAND Return 0 to indicate failure. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads CSV data, extracts the Nth column from each row, uppercases those values, and writes the result to a new CSV file with a _process suffix." process_csv_data]
    [INPUTS]
        <REF> N </REF>: integer "Zero-based column index to extract from each row."
        <REF> save_file_name </REF>: string "Input CSV file name and base for the output file name."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "Return value from writing the processed CSV data to the output file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the CSV data from save_file_name and unpack the returned title and data values. RESULT title, data]
            [COMMAND Build a list containing the Nth element from each row in data. RESULT column_data]
            [COMMAND Replace column_data with a new list where every element is converted to uppercase. RESULT column_data]
            [COMMAND Create new_data as a two-item list containing title and the transformed column_data. RESULT new_data]
            [COMMAND Write new_data to a file named by removing the original extension from save_file_name and appending _process.csv. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
