# CLASS JSONProcessor

[DEFINE_WORKER: "Checks whether a file exists, returns 0 if it does not, otherwise attempts to load and return JSON content from the file and returns -1 if reading or parsing fails." read_json]
    [INPUTS]
        <REF> self </REF>: object "Instance context for the method."
        <REF> file_path </REF>: string "Path to the JSON file to check and read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object_or_integer "Returns 0 when the file does not exist, the parsed JSON data when loading succeeds, or -1 if an error occurs while opening or parsing the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the path given by file_path exists in the filesystem. RESULT path_exists]
            [COMMAND If path_exists is false, return 0 immediately as an early exit. RESULT result]
            [COMMAND If path_exists is true, open the file at file_path in read mode. RESULT file_handle]
            [COMMAND Load JSON data from the opened file handle. RESULT data]
            [COMMAND Return the loaded JSON data. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The path given by file_path does not exist.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without attempting to open or parse the file. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: An error occurs while opening the file or while loading JSON from it.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 as the fallback result for any exception caught by the try-except block. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Write the provided data as JSON to the specified file path and return 1 on success or -1 if any exception occurs." write_json]
    [INPUTS]
        <REF> data </REF>: any "Data to serialize and write as JSON content."
        <REF> file_path </REF>: string "Path of the file to open for writing the JSON output."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 1 when the JSON data is written successfully, or -1 if any exception occurs during opening or writing the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the file at file_path in write mode as file within a try block. RESULT file]
            [COMMAND Within the try block and using the opened file handle, serialize data and write it to the file as JSON. RESULT write_status]
            [COMMAND If opening the file and writing the JSON both complete without exception, return 1. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the file in write mode or while serializing or writing the JSON data.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception without rethrowing it. RESULT caught_exception]
            [COMMAND Return -1 as the failure indicator. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Load JSON data from a file, return 0 immediately if the read result is the sentinel value 0 or -1, otherwise remove a specified key if present, persist the updated data, and return 1 on success or 0 if no key was removed." process_json]
    [INPUTS]
        <REF> file_path </REF>: text "Path to the JSON file to read and potentially update."
        <REF> remove_key </REF>: text "Key to remove from the loaded JSON mapping if it exists."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 1 if the key is found, removed, and the updated JSON is written back; otherwise returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read JSON data from file_path by calling self.read_json(file_path) and store the returned value in data. RESULT data]
            [COMMAND Check whether data equals 0 or data equals -1 using short-circuit OR; if either comparison is true, return 0 immediately without attempting key removal or writing the file. RESULT early_return_on_invalid_read]
            [COMMAND If remove_key exists as a key in data, delete data[remove_key]. RESULT data_after_key_removal]
            [COMMAND Within the branch where remove_key exists in data, write the modified data back to file_path by calling self.write_json(data, file_path). RESULT write_completed]
            [COMMAND Within the branch where remove_key exists in data, return 1 immediately after writing the updated JSON. RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The value returned from self.read_json(file_path) is 0 or -1.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately and do not check membership, delete any key, or write any file content. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: remove_key does not exist in data after the read result is confirmed to be neither 0 nor -1.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 without modifying data and without calling self.write_json(data, file_path). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
