# CLASS CookiesUtil

[DEFINE_WORKER: "Initialize the instance by storing the provided cookies file reference and setting the cookies state to an uninitialized None value." __init__]
    [INPUTS]
        <REF> cookies_file </REF>: any "The value to store as the instance's cookies file reference without validation or transformation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input value cookies_file directly to the instance attribute self.cookies_file without modification or validation. RESULT self.cookies_file]
            [COMMAND Set the instance attribute self.cookies to None as the default uninitialized cookies value, overwriting any prior value. RESULT self.cookies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the cookies entry from the response object, store it on the instance, and then save the cookies." get_cookies]
    [INPUTS]
        <REF> self </REF>: object "Instance receiving the cookies value."
        <REF> reponse </REF>: mapping "Response-like object expected to contain a 'cookies' key."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value from reponse['cookies'] to the instance attribute self.cookies. RESULT self.cookies]
            [COMMAND Call self._save_cookies() to persist the updated cookies on the instance. RESULT _save_cookies_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Loads cookie data from the configured cookies file as JSON and returns an empty dictionary if the file does not exist." load_cookies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "The loaded cookie data, or an empty dictionary if the file is missing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.cookies_file in read mode and parse its contents as JSON. RESULT cookies_data]
            [COMMAND Return the parsed cookies_data. RESULT cookies_data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the cookies file is not found when opening self.cookies_file.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty dictionary instead of cookie data. RESULT {}]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Save the current cookies dictionary to the configured cookies file as JSON and return success or failure." _save_cookies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the cookies are saved successfully, otherwise False"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Enter a try block to attempt saving the cookies to disk. RESULT try_block]
            [COMMAND Open the file at self.cookies_file in write mode as file. RESULT file]
            [COMMAND Serialize self.cookies into the open file using json.dump. RESULT written_cookies]
            [COMMAND Return True after the cookies are successfully written. RESULT true_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during file opening or JSON writing.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception and suppress it. RESULT caught_exception]
            [COMMAND Return False to indicate the save operation failed. RESULT false_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build the request cookies header as a semicolon-separated string from the object's stored cookie key-value pairs." set_cookies]
    [INPUTS]
        <REF> self </REF>: object "Object containing a cookies mapping."
        <REF> request </REF>: mapping "Request object that receives the cookies header string under the 'cookies' key."
    [END_INPUTS]

    [OUTPUTS]
        <REF> request </REF>: mapping "The same request object updated with a 'cookies' string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read all key-value pairs from self.cookies. RESULT cookie_items]
            [COMMAND Format each cookie pair as 'key=value'. RESULT formatted_cookies]
            [COMMAND Join the formatted cookie strings with '; ' as the separator. RESULT cookies_header]
            [COMMAND Assign the resulting string to request['cookies']. RESULT request]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
