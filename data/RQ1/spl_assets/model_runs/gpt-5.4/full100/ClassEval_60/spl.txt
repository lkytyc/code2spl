# CLASS MovieTicketDB

[DEFINE_WORKER: "Initialize the instance by opening a SQLite connection from the provided database name, creating a cursor from that connection, and invoking table creation." __init__]
    [INPUTS]
        <REF> db_name </REF>: text "Database name or path passed to sqlite3.connect to open the SQLite database."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using db_name by calling sqlite3.connect(db_name) and assign the returned connection object to self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection by calling self.connection.cursor() and assign the returned cursor object to self.cursor. RESULT self.cursor]
            [COMMAND Invoke self.create_table() after self.connection and self.cursor have been initialized. RESULT table_creation_invoked]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create the tickets database table if it does not already exist and commit the schema change to the database connection." create_table]
    [INPUTS]
        <REF> self </REF>: object "Object instance that provides an initialized database cursor at self.cursor and the associated database connection at self.connection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use self.cursor to execute a constant SQL DDL statement that creates the tickets table only if it does not already exist, with columns id INTEGER PRIMARY KEY, movie_name TEXT, theater_name TEXT, seat_number TEXT, and customer_name TEXT. RESULT tickets_table_creation_attempted]
            [COMMAND Commit the current database transaction through self.connection so the schema change is persisted. RESULT schema_change_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The tickets table already exists in the database.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the CREATE TABLE IF NOT EXISTS statement without creating or modifying the existing tickets table schema. RESULT no_schema_creation_needed]
            [COMMAND Commit the transaction through self.connection. RESULT transaction_committed_without_schema_change]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert a ticket record into the tickets table using the provided movie, theater, seat, and customer values, then commit the transaction." insert_ticket]
    [INPUTS]
        <REF> movie_name </REF>: data "Movie name value to insert into the ticket record."
        <REF> theater_name </REF>: data "Theater name value to insert into the ticket record."
        <REF> seat_number </REF>: data "Seat number value to insert into the ticket record."
        <REF> customer_name </REF>: data "Customer name value to insert into the ticket record."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL INSERT into the tickets table for the columns movie_name, theater_name, seat_number, and customer_name using the current values of movie_name, theater_name, seat_number, and customer_name in that positional order. RESULT insert_execution]
            [COMMAND Commit the active database transaction on self.connection to persist the inserted ticket record. RESULT commit_complete]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Execute a parameterized database query to retrieve all tickets whose customer_name matches the provided customer name and return the full result set." search_tickets_by_customer]
    [INPUTS]
        <REF> customer_name </REF>: string "The customer name value used as the parameter in the ticket lookup query."
    [END_INPUTS]

    [OUTPUTS]
        <REF> tickets </REF>: list "The complete list of rows fetched from the query result, typically empty if no matching tickets are found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL query on self.cursor to select all columns from the tickets table where customer_name equals the provided customer_name value, using customer_name as the single bound parameter. RESULT query_execution]
            [COMMAND Fetch all remaining rows from the current result set of self.cursor and assign the complete fetched collection to tickets. RESULT tickets]
            [COMMAND Return tickets. RESULT tickets]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete the ticket record matching the provided ticket ID from the database and commit the transaction." delete_ticket]
    [INPUTS]
        <REF> ticket_id </REF>: data_type "Identifier of the ticket record to delete from the tickets table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL DELETE statement on the tickets table where id equals the provided ticket_id. RESULT delete_operation]
            [COMMAND Commit the database transaction to persist the deletion. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
