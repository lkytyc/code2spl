# CLASS DataStatistics2

[DEFINE_WORKER: "Initialize the instance by converting the input data into a NumPy array and storing it in self.data." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance being initialized."
        <REF> data </REF>: any "Input data to be converted into a NumPy array."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the input data into a NumPy array using np.array(data). RESULT array_data]
            [COMMAND Assign the converted NumPy array to the instance attribute self.data. RESULT self.data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the sum of self.data using NumPy." get_sum]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the data attribute to be summed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or array-like scalar "The NumPy sum of self.data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.data from the current object. RESULT data]
            [COMMAND Compute the sum of data with np.sum. RESULT sum_result]
            [COMMAND Return sum_result to the caller. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the minimum value from self.data using NumPy." get_min]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the data attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or array scalar "The minimum value present in self.data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access self.data as the source collection to inspect. RESULT data]
            [COMMAND Compute the minimum value of data using np.min. RESULT min_value]
            [COMMAND Return min_value immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the maximum value in self.data using NumPy." get_max]
    [INPUTS]
        <REF> self.data </REF>: array-like "The data collection to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric or comparable value "The maximum element in self.data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.data from the current instance. RESULT data]
            [COMMAND Compute the maximum value of data with np.max(data). RESULT max_value]
            [COMMAND Return max_value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.data is empty or not valid for np.max.]
        [LOG "np.max(self.data) cannot determine a maximum for the provided data."]
        [THROW ValueError "Raised when the input is empty or invalid for maximum computation."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the variance of self.data and return it rounded to two decimal places." get_variance]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the data attribute used for variance calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The variance of self.data rounded to two decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the data values from self.data. RESULT data]
            [COMMAND Compute the variance of data using np.var. RESULT variance]
            [COMMAND Round the computed variance to 2 decimal places and return the rounded value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the standard deviation of self.data rounded to two decimal places." get_std_deviation]
    [INPUTS]
        <REF> self </REF>: object "Object containing the data sequence in self.data."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The standard deviation of self.data rounded to 2 decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the standard deviation of self.data using np.std. RESULT std_value]
            [COMMAND Round the computed standard deviation to 2 decimal places. RESULT rounded_std]
            [COMMAND Return the rounded standard deviation value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the correlation coefficient matrix computed from self.data with columns treated as variables." get_correlation]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the data attribute used for correlation computation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: ndarray "Correlation coefficient matrix produced by numpy.corrcoef."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.data as the input dataset for correlation computation. RESULT data]
            [COMMAND Call numpy.corrcoef on data with rowvar set to False so that columns are treated as variables. RESULT correlation_matrix]
            [COMMAND Return correlation_matrix as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
