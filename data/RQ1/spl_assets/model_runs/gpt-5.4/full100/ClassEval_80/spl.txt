# CLASS SQLQueryBuilder

[DEFINE_WORKER: "Build a SQL SELECT query from the table name, optional column list, and optional WHERE conditions." select]
    [INPUTS]
        <REF> table </REF>: string "Table name to select from."
        <REF> columns </REF>: string or iterable "Columns to select, defaulting to all columns with '*'."
        <REF> where </REF>: dict or null "Optional mapping of column names to values for equality filters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Rendered SQL query string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If columns is not '*', join the column names with ', ' and replace columns with the resulting string. RESULT columns]
            [COMMAND Create the base SQL string in the form SELECT {columns} FROM {table}. RESULT query]
            [COMMAND If where is provided, append a WHERE clause using AND between equality predicates built as {key}='{value}' for each mapping entry. RESULT query]
            [COMMAND Return the final SQL query string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an SQL INSERT statement from a table name and a data dictionary by joining the dictionary keys as column names and quoting each value." insert]
    [INPUTS]
        <REF> table </REF>: string "Target table name to insert into."
        <REF> data </REF>: dict "Dictionary of column names and corresponding values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted SQL INSERT statement."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join all keys from data with comma and space to form the column list. RESULT keys]
            [COMMAND Join all values from data with comma and space after wrapping each value in single quotes to form the value list. RESULT values]
            [COMMAND Return an INSERT statement using table, keys, and values in the form INSERT INTO {table} ({keys}) VALUES ({values}). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a DELETE SQL statement for the given table and optionally append equality filters joined by AND." delete]
    [INPUTS]
        <REF> table </REF>: string "Target table name to delete from."
        <REF> where </REF>: map "Optional mapping of column names to values used as exact-match filters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Rendered SQL DELETE query."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQL string starting with DELETE FROM followed by the table name. RESULT query]
            [COMMAND If where is provided and is truthy, append a WHERE clause built from each key-value pair formatted as key='value' and joined with AND. RESULT query]
            [COMMAND Return the final SQL query string. RESULT query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an SQL UPDATE statement string from a table name, a data mapping, and an optional WHERE clause." update]
    [INPUTS]
        <REF> table </REF>: any "Table name to update."
        <REF> data </REF>: mapping "Column-value pairs to place in the SET clause."
        <REF> where </REF>: mapping or null "Optional column-value pairs to place in the WHERE clause."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Constructed SQL UPDATE query string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create the SET clause string by joining each data item as key='value' with ', '. RESULT update_str]
            [COMMAND Build the base query string as UPDATE {table} SET {update_str}. RESULT query]
            [COMMAND If a where mapping is provided, append a WHERE clause joined by ' AND ' with each condition formatted as key='value'. RESULT query]
            [COMMAND Return the final query string. RESULT query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
