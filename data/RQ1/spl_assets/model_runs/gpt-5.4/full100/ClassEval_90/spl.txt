# CLASS URLHandler

[DEFINE_WORKER: "Initializes the instance URL field from the provided url argument." __init__]
    [INPUTS]
        <REF> url </REF>: any "The URL value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self </REF>: object "The initialized instance with url assigned."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming url value directly to self.url. RESULT self.url]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the URL scheme prefix before '://' when present, otherwise return None." get_scheme]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the url string to inspect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The URL scheme prefix before '://', or None when no scheme separator is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first occurrence of "://" in self.url and store its index in scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not -1, return the substring of self.url from the start up to scheme_end. RESULT self.url[:scheme_end]]
            [COMMAND If scheme_end is -1, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The substring "://" is not found in self.url, so find() returns -1.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None instead of a substring. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the host portion of the URL by removing the scheme and returning the substring before the first slash, or the remaining string if no slash exists." get_host]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the URL string in self.url."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The host segment of the URL, or null if the URL does not contain a scheme delimiter."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the index of the substring "://" in self.url and store it as scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not -1, take the substring of self.url starting 3 characters after the scheme delimiter and store it as url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the scheme-present path, find the first "/" in url_without_scheme and store the index as host_end. RESULT host_end]
            [COMMAND Within the scheme-present path, if host_end is not -1, return the substring of url_without_scheme from the beginning up to host_end, excluding the slash. RESULT result]
            [COMMAND Within the scheme-present path, if host_end is -1, return the entire url_without_scheme as the host. RESULT result]
            [COMMAND If scheme_end is -1, return null because no scheme delimiter was found. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A scheme delimiter exists but no slash appears after it.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the remainder of the URL after "://" as the host value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the path portion of self.url by removing the scheme and host and returning the substring starting at the first slash after the host, or return None if either delimiter is missing." get_path]
    [INPUTS]
        <REF> self.url </REF>: string "The URL string to inspect for a scheme delimiter and a path after the host."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string_or_none "The path substring beginning with the first slash after the host when both the scheme delimiter and path separator exist; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the starting index of the substring "://" in self.url and store the integer result in scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not equal to -1, take the substring of self.url starting at index scheme_end + 3 to remove the scheme and its delimiter, and store it in url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the condition that scheme_end is not equal to -1, find the starting index of the first "/" in url_without_scheme and store the integer result in host_end. RESULT host_end]
            [COMMAND Within the nested condition that scheme_end is not equal to -1 and host_end is not equal to -1, return the substring of url_without_scheme starting at index host_end. RESULT return_path]
            [COMMAND If no earlier return occurred, return None. RESULT return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: scheme_end is equal to -1, meaning self.url does not contain the substring "://".]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip host and path extraction and fall through to the default return value of None. RESULT return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: scheme_end is not equal to -1 but host_end is equal to -1, meaning there is no "/" after the scheme-stripped host portion.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do not return a path substring and fall through to the default return value of None. RESULT return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract query parameters from self.url into a dictionary by parsing the substring between ? and #, returning None when no query marker exists." get_query_params]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict or null "A dictionary of parsed query parameters when a query delimiter exists, or None when no query delimiter is present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first '?' in self.url and store its index in query_start, using -1 when '?' is absent. RESULT query_start]
            [COMMAND Find the first '#' in self.url and store its index in fragment_start, using -1 when '#' is absent. RESULT fragment_start]
            [COMMAND If query_start is not -1, extract the substring of self.url from query_start + 1 up to fragment_start and store it in query_string. RESULT query_string]
            [COMMAND If query_start is not -1, initialize params as an empty dictionary. RESULT params]
            [COMMAND Within the outer condition query_start != -1, if query_string length is greater than 0, split query_string on '&' into param_pairs. RESULT param_pairs]
            [COMMAND Within the outer condition query_start != -1 and the inner condition len(query_string) > 0, iterate over each pair in param_pairs. RESULT pair]
            [COMMAND Within the outer condition query_start != -1, for each pair, split it on '=' into key_value. RESULT key_value]
            [COMMAND Within the outer condition query_start != -1, if key_value length is exactly 2, unpack it into key and value and assign params[key] = value. RESULT params]
            [COMMAND Within the outer condition query_start != -1, return params after processing all valid pairs, even if params is still empty. RESULT params]
            [COMMAND If query_start is -1, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the substring after the first '#' in self.url and return None when no fragment delimiter is present." get_fragment]
    [INPUTS]
        <REF> self.url </REF>: string-like "The URL value to inspect for a fragment delimiter using the find operation and slicing."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The substring after the first '#' character if one exists; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute fragment_start by calling self.url.find("#") to locate the first occurrence of the fragment delimiter, where a not-found result is represented by -1. RESULT fragment_start]
            [COMMAND Check whether fragment_start is not equal to -1, meaning a '#' character exists in self.url. RESULT has_fragment]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: fragment_start is not equal to -1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the substring of self.url starting at index fragment_start + 1 and continuing to the end, thereby excluding the '#' character itself. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: fragment_start is equal to -1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None because no fragment delimiter was found in self.url. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
