[DEFINE_WORKER: "Compute mean reciprocal-rank-style scores for either a single tuple or a list of tuple inputs by applying reciprocal position weights and selecting the first strictly positive weighted value." mrr]

    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A pair containing the aggregate score and a list of per-item scores."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether data is exactly of type list or exactly of type tuple using strict type equality. RESULT type_validation]
            [COMMAND Check whether len(data) equals 0, and if so return 0.0 and [0.0] immediately. RESULT empty_input_return]
            [COMMAND If data is exactly of type tuple, unpack it into sub_list and total_num, convert sub_list to a NumPy array, and continue with the single-item computation path. RESULT tuple_inputs]
            [COMMAND Within the tuple path, if total_num equals 0 exactly, return 0.0 and [0.0] immediately. RESULT tuple_zero_total_return]
            [COMMAND Within the tuple path when total_num is not 0, build ranking_array as 1.0 divided elementwise by NumPy array [1, 2, ..., len(sub_list)], then compute mr_np as sub_list multiplied elementwise by ranking_array. RESULT tuple_weighted_values]
            [COMMAND Within the tuple path after computing mr_np, initialize mr to 0.0, iterate through mr_np from left to right, set mr to the first value strictly greater than 0, and break immediately when that first positive value is found. RESULT tuple_first_positive_mr]
            [COMMAND Within the tuple path, if no positive value is found then keep mr as 0.0, and return mr together with [mr]. RESULT tuple_return]
            [COMMAND If data is exactly of type list, initialize separate_result as an empty list and continue with the multi-item computation path. RESULT list_init]
            [COMMAND Within the list path, iterate over each (sub_list, total_num) tuple in data, and for each item convert sub_list to a NumPy array. RESULT list_item_inputs]
            [COMMAND Within the list loop for each item, if total_num equals 0 exactly, set mr to 0.0 for that item. RESULT list_zero_total_mr]
            [COMMAND Within the list loop for each item when total_num is not 0, build ranking_array as 1.0 divided elementwise by NumPy array [1, 2, ..., len(sub_list)], then compute mr_np as sub_list multiplied elementwise by ranking_array. RESULT list_weighted_values]
            [COMMAND Within the list loop after computing mr_np for an item, initialize mr to 0.0, iterate through mr_np from left to right, set mr to the first value strictly greater than 0, and break immediately when that first positive value is found. RESULT list_first_positive_mr]
            [COMMAND Within the list loop for each item, if no positive value is found then keep mr as 0.0, and append mr to separate_result. RESULT list_append_result]
            [COMMAND After finishing the full list loop, return the NumPy mean of separate_result together with separate_result. RESULT list_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is neither exactly of type list nor exactly of type tuple]

[DEFINE_WORKER: "Computes an average-precision-like score for either one tuple input or a list of tuple inputs, returning the score and per-item results." map]
    [INPUTS]
        <REF> data </REF>: list or tuple "Either a single tuple of (sub_list, total_num) or a list of such tuples."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "Returns the average score and either a single-item list for tuple input or the list of per-item scores for list input."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate that data is either a list or a tuple; otherwise raise an exception with the required input-type message. RESULT validation_status]
            [COMMAND If data is empty, return 0.0 and [0.0] immediately. RESULT empty_result]
            [COMMAND If data is a tuple, unpack it into sub_list and total_num, convert sub_list to a NumPy array, and if total_num is 0 return 0.0 and [0.0] immediately. RESULT tuple_case_result]
            [COMMAND Within the tuple case when total_num is not 0, build reciprocal position weights from 1 through the length of sub_list, assign increasing ranks to nonzero elements while assigning 0 to zero elements, compute ap as the weighted sum divided by total_num, and return ap and [ap]. RESULT tuple_ap_result]
            [COMMAND If data is a list, initialize an empty separate_result list and iterate over each (sub_list, total_num) pair in order. RESULT separate_result]
            [COMMAND Within the list case for each pair, convert sub_list to a NumPy array, set ap to 0.0 when total_num is 0, and otherwise compute ap using the same reciprocal-weight and nonzero-ranking procedure as in the tuple case. RESULT ap]
            [COMMAND Within the list case after processing each pair, append ap to separate_result, and after the loop return the mean of separate_result together with separate_result itself. RESULT list_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is neither a list nor a tuple]
        [LOG "the input must be a tuple([0,...,1,...],int) or a iteration of list of tuple"]
        [THROW Exception "Signals that the input type is invalid."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
