# CLASS StudentDatabaseProcessor

    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[DEFINE_WORKER: "Create the students table in the configured SQLite database if it does not already exist, then commit and close the connection." create_student_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using self.database_name and store the connection in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Assign the fixed multiline SQL CREATE TABLE IF NOT EXISTS statement for the students table to create_table_query. RESULT create_table_query]
            [COMMAND Execute create_table_query with cursor to create the students table if it does not already exist. RESULT cursor.execute(create_table_query)]
            [COMMAND Commit the transaction on conn. RESULT conn.commit()]
            [COMMAND Close conn. RESULT conn.close()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert a student record into the students table using a parameterized SQLite query and then commit and close the database connection." insert_student]
    [INPUTS]
        <REF> student_data </REF>: dict "Dictionary containing student fields: name, age, gender, and grade."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Assign a parameterized multi-line INSERT statement for the students table with columns name, age, gender, and grade to insert_query. RESULT insert_query]
            [COMMAND Execute insert_query through cursor with the values student_data['name'], student_data['age'], student_data['gender'], and student_data['grade'] bound to the four placeholders. RESULT insert_result]
            [COMMAND Commit the transaction on conn. RESULT commit_result]
            [COMMAND Close conn. RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search for students by exact name in the database and return all matching rows." search_student_by_name]
    [INPUTS]
        <REF> name </REF>: any "The student name used as the SQL parameter."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "All rows matching the given name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using self.database_name. RESULT conn]
            [COMMAND Create a cursor from the open connection. RESULT cursor]
            [COMMAND Set the SQL query to select all columns from students where name equals the parameter placeholder. RESULT select_query]
            [COMMAND Execute the SQL query with the provided name bound as the parameter. RESULT query_execution]
            [COMMAND Fetch all rows produced by the executed query. RESULT result]
            [COMMAND Close the database connection. RESULT conn]
            [COMMAND Return the fetched rows. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete the student record whose name matches the provided name from the SQLite students table." delete_student_by_name]

    [OUTPUTS]
    [END_OUTPUTS]

    [END_MAIN_FLOW]
