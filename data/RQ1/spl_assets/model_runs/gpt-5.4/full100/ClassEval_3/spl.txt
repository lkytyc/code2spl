# CLASS ArrangementCalculator

[DEFINE_WORKER: "Initialize the instance by storing the provided datas argument on the object's datas attribute." __init__]
    [INPUTS]
        <REF> datas </REF>: any "Input value to assign to the instance datas attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter datas directly to self.datas without modification. RESULT self.datas]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the arrangement count by returning n! when m is omitted or equal to n, otherwise returning n! divided by (n - m)! using integer division." count]
    [INPUTS]
        <REF> n </REF>: number "The total number of items."
        <REF> m </REF>: number or null "The number of selected items; if omitted or null, the full arrangement of n items is used."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed arrangement count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether m is None; if true, short-circuit the condition and return ArrangementCalculator.factorial(n) immediately. RESULT result]
            [COMMAND If m is not None, evaluate whether n equals m; if true, return ArrangementCalculator.factorial(n) immediately. RESULT result]
            [COMMAND If m is not None and n does not equal m, return ArrangementCalculator.factorial(n) // ArrangementCalculator.factorial(n - m) using integer division. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m is None or n equals m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return ArrangementCalculator.factorial(n) as the final result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: m is not None and n does not equal m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the factorial of n and the factorial of n minus m, then return their integer quotient as the final result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the total number of arrangements for n by summing ArrangementCalculator.count(n, i) for every i from 1 through n inclusive." count_all]
    [INPUTS]
        <REF> n </REF>: number "The upper bound and primary input used both as the first argument to the count operation and to define the inclusive loop range from 1 to n."
    [END_INPUTS]

    [OUTPUTS]
        <REF> total </REF>: number "The accumulated sum of ArrangementCalculator.count(n, i) over all i from 1 through n inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND For each i from 1 through n inclusive, compute ArrangementCalculator.count(n, i) and add the result to total. RESULT total]
            [COMMAND Return total after completing the full inclusive iteration. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate and return a list of all permutations of self.datas with length m, defaulting m to the full length of self.datas when no value is provided." select]
    [INPUTS]
        <REF> self </REF>: object "Object instance that provides the datas collection."
        <REF> m </REF>: optional integer "Desired permutation length; if None, use the length of self.datas."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of permutations, where each permutation is converted to a list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If m is None, set m to the length of self.datas. RESULT m]
            [COMMAND Initialize result as a new empty list. RESULT result]
            [COMMAND For each permutation produced by itertools.permutations(self.datas, m), convert the permutation to a list and append it to result. RESULT result]
            [COMMAND Return result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect and return the combined results of selecting every index from 1 through the length of datas." select_all]
    [INPUTS]
        <REF> self </REF>: object "Instance containing datas and the select method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list containing the concatenated selections for all indices from 1 to len(datas)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Iterate i from 1 through len(self.datas) inclusive. RESULT i]
            [COMMAND Within the loop, extend result with the items returned by self.select(i). RESULT result]
            [COMMAND After processing all indices, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the factorial of the input integer by multiplying all integers from 2 through n into an accumulator initialized to 1." factorial]
    [INPUTS]
        <REF> n </REF>: integer "The upper bound of the factorial computation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The product of all integers from 2 through n, or 1 when n is less than 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to 1. RESULT result]
            [COMMAND For each integer i from 2 through n inclusive, multiply result by i and store the updated product back into result. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
