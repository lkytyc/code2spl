# CLASS BinaryDataProcessor

[DEFINE_WORKER: "Initialize the instance by storing the provided binary string and then invoking the cleanup routine to remove non-binary characters." __init__]
    [INPUTS]
        <REF> binary_string </REF>: text "Input string intended to contain binary characters for the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter binary_string to the instance attribute self.binary_string. RESULT self.binary_string]
            [COMMAND Invoke self.clean_non_binary_chars() to sanitize the stored binary string by removing any non-binary characters. RESULT sanitized_self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filter the current binary_string in place so that only '0' and '1' characters remain." clean_non_binary_chars]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the binary_string field to be sanitized."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: none "No explicit return value; updates self.binary_string in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.binary_string. RESULT current_binary_string]
            [COMMAND Iterate over each character in current_binary_string and keep only characters that are either '0' or '1'. RESULT filtered_characters]
            [COMMAND Join the retained characters into a new string in their original order. RESULT filtered_binary_string]
            [COMMAND Assign filtered_binary_string back to self.binary_string. RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts zero and one bits in the stored binary string, computes their proportions relative to total length, and returns those values with the bit length." calculate_binary_info]
    [INPUTS]
        <REF> self.binary_string </REF>: str "Binary string used as the source for all counts and length calculations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "A dictionary containing the zero proportion, one proportion, and total bit length."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the number of '0' characters in self.binary_string. RESULT zeroes_count]
            [COMMAND Count the number of '1' characters in self.binary_string. RESULT ones_count]
            [COMMAND Compute the total length of self.binary_string. RESULT total_length]
            [COMMAND Divide zeroes_count by total_length to compute the zero proportion. RESULT zeroes_percentage]
            [COMMAND Divide ones_count by total_length to compute the one proportion. RESULT ones_percentage]
            [COMMAND Return a dictionary containing Zeroes, Ones, and Bit length mapped to the computed proportions and total length. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: total_length is 0 when dividing by total_length.]
        [LOG "Division by zero occurs while computing the percentage values."]
        [THROW ZeroDivisionError "Raised when the binary string is empty and percentage normalization is attempted."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert the instance binary string into ASCII text by reading it in 8-bit chunks, converting each chunk to a byte value, collecting the bytes, and decoding the resulting byte array as ASCII." convert_to_ascii]
    [INPUTS]
        <REF> self.binary_string </REF>: string "Binary string source whose characters are processed in 8-character slices as base-2 byte representations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "ASCII-decoded text produced from the collected byte values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize byte_array as a new empty mutable byte array. RESULT byte_array]
            [COMMAND For i from 0 up to but not including the length of self.binary_string, advancing by 8 each iteration, extract the slice self.binary_string[i:i+8] into byte. RESULT byte]
            [COMMAND Within the loop for each extracted byte slice, convert byte from a base-2 string to its integer value and store it in decimal. RESULT decimal]
            [COMMAND Within the same loop iteration, append decimal to byte_array. RESULT byte_array]
            [COMMAND After all 8-character slices have been processed, decode byte_array using ASCII and return the decoded string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: A processed slice is not a valid base-2 string for int(byte, 2), such as when it contains characters other than 0 or 1 or is empty.]
        [LOG "Binary slice conversion to integer failed during base-2 parsing."]
        [THROW ValueError "Raised when a binary slice cannot be parsed as a base-2 integer."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The collected byte sequence cannot be decoded with ASCII because at least one byte value is outside the ASCII decoding range.]
        [LOG "ASCII decoding of the byte array failed."]
        [THROW UnicodeDecodeError "Raised when byte_array.decode('ascii') cannot decode the produced bytes as ASCII text."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert the instance binary string into UTF-8 text by grouping it into 8-bit chunks, turning each chunk into a byte value, collecting those bytes in a bytearray, and decoding the result." convert_to_utf8]
    [INPUTS]
        <REF> self.binary_string </REF>: string "A binary string whose characters are expected to represent byte data in base 2, processed in 8-character chunks."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The UTF-8 decoded text produced from the accumulated byte values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize byte_array as a new empty bytearray. RESULT byte_array]
            [COMMAND For i in range from 0 inclusive to len(self.binary_string) exclusive with step 8, extract self.binary_string[i:i+8] into byte. RESULT byte]
            [COMMAND Within the loop for each current byte slice, convert byte from base 2 to an integer and store it in decimal. RESULT decimal]
            [COMMAND Within the same loop iteration, append decimal to byte_array. RESULT byte_array]
            [COMMAND After all iterations complete, decode byte_array using UTF-8 and return the decoded string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The length of self.binary_string is not a multiple of 8.]
        [SEQUENTIAL_BLOCK]
            [COMMAND On the final loop iteration, extract a trailing slice shorter than 8 characters using self.binary_string[i:i+8]. RESULT byte]
            [COMMAND Convert that non-empty shorter binary slice from base 2 to an integer and append it to byte_array exactly like any other iteration. RESULT byte_array]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any extracted byte slice contains characters that are not valid for base-2 conversion when int(byte, 2) is executed.]
        [LOG "Base-2 conversion failed for a binary string slice while converting self.binary_string into bytes."]
        [THROW ValueError "Raised when a sliced chunk cannot be interpreted as a valid binary integer."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The accumulated byte_array cannot be decoded as UTF-8 when byte_array.decode('utf-8') is executed.]
        [LOG "UTF-8 decoding failed for the byte sequence produced from self.binary_string."]
        [THROW UnicodeDecodeError "Raised when the generated byte sequence is not valid UTF-8."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
