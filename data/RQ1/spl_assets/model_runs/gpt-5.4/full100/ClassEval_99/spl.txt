# CLASS ZipFileProcessor

[DEFINE_WORKER: "Initialize the instance by storing the provided file name on the object." __init__]
    [INPUTS]
        <REF> file_name </REF>: text "The file name value to assign to the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input file name to the instance attribute self.file_name. RESULT self.file_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens the file path stored in self.file_name as a ZIP archive in read mode and returns the archive object, or returns None if any exception occurs." read_zip_file]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file_name attribute with the target ZIP file path."
    [END_INPUTS]

    [OUTPUTS]
        <REF> zip_file_or_none </REF>: ZipFile_or_None "The opened ZIP file object in read mode when successful, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the file at self.file_name as a ZIP archive in read mode and store the opened archive object. RESULT zip_file]
            [COMMAND Return the opened ZIP archive object. RESULT zip_file]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening self.file_name as a ZIP archive in read mode.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None instead of a ZIP archive object. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract all contents from the zip file stored in self.file_name to the given output path and return whether the operation succeeded." extract_all]
    [INPUTS]
        <REF> output_path </REF>: path "Destination path where all zip contents should be extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns true if all zip contents are extracted successfully; otherwise returns false if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the zip file at self.file_name in read mode using a managed context. RESULT zip_file]
            [COMMAND If the zip file is opened successfully, extract all archive contents to output_path. RESULT extraction_completed]
            [COMMAND After successful extraction, return true. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip file or extracting its contents.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to extract a named file from the instance zip archive into the given output path and returns whether the extraction succeeded." extract_file]
    [INPUTS]
        <REF> file_name </REF>: string "Name or path of the file inside the zip archive to extract."
        <REF> output_path </REF>: string "Destination directory or path where the file should be extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the file is extracted successfully; otherwise False if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip archive located at self.file_name in read mode within a managed context. RESULT zip_file]
            [COMMAND Within the opened zip archive context, extract the archive member named file_name to output_path. RESULT extraction_completed]
            [COMMAND After successful extraction and context exit, return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip archive or extracting the requested file.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a zip archive at the given output path by writing each file from the input collection and returns whether the operation succeeded." create_zip_file]
    [INPUTS]
        <REF> files </REF>: list "A collection of file paths or file entries to add to the zip archive."
        <REF> output_file_name </REF>: string "The output path and name for the zip archive to create."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the zip file is created and all input files are written successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a zip archive at output_file_name in write mode within a protected try block. RESULT zip_file]
            [COMMAND Within the opened zip archive context, iterate over each file in files. RESULT current_file]
            [COMMAND Within the loop for each current_file, write the file into the zip archive. RESULT file_written]
            [COMMAND After all files are written and the zip archive context closes successfully, return True. RESULT result_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip archive, iterating through files, writing a file, or closing the zip archive context.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception without rethrowing it. RESULT caught_exception]
            [COMMAND Return False. RESULT result_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
