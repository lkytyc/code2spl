# CLASS BalancedBrackets

[DEFINE_WORKER: "Initialize a new instance by creating an empty stack, defining the opening and closing bracket lists, and storing the input expression." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance being initialized."
        <REF> expr </REF>: text "The expression value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance stack attribute as an empty list. RESULT self.stack]
            [COMMAND Assign the instance left_brackets attribute to a list containing the three opening bracket strings "(", "{", and "[" in that order. RESULT self.left_brackets]
            [COMMAND Assign the instance right_brackets attribute to a list containing the three closing bracket strings ")", "}", and "]" in that order. RESULT self.right_brackets]
            [COMMAND Store the input expr value directly in the instance expr attribute without modification. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filter self.expr so it retains only characters that belong to self.left_brackets or self.right_brackets, preserving their original order." clear_expr]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each character in the current self.expr and keep only characters for which membership in self.left_brackets is true or membership in self.right_brackets is true, using short-circuit OR logic for the inclusion test. RESULT filtered_characters]
            [COMMAND Join the kept characters into a single string with an empty-string separator, producing an empty string if no characters were kept or if the original self.expr was empty. RESULT filtered_expr]
            [COMMAND Assign the resulting filtered string back to self.expr. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the current expression has properly balanced and correctly matched brackets by resetting state, scanning the expression with a stack, and returning False on the first mismatch or leftover opening bracket." check_balanced_brackets]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the expression, bracket definitions, stack state, and helper methods used during validation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True when all brackets in the expression are balanced and correctly matched; otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.clear_expr() before validation begins to reset or normalize the expression-related state used by this method. RESULT expression_state_cleared]
            [COMMAND Iterate through each Brkt in self.expr from left to right. RESULT current_bracket_sequence_processed]
            [COMMAND Within the loop for each Brkt, if Brkt is in self.left_brackets, append Brkt to self.stack as an unmatched opening bracket. RESULT opening_bracket_pushed]
            [COMMAND Within the same loop, otherwise treat Brkt as a closing bracket and pop the most recent opening bracket from self.stack into Current_Brkt. RESULT current_brkt_popped_from_stack]
            [COMMAND Within the closing-bracket branch, if Current_Brkt equals "(", require Brkt to equal ")"; otherwise return False immediately. RESULT parenthesis_pair_validated_or_false_returned]
            [COMMAND Within the closing-bracket branch, if Current_Brkt equals "{", require Brkt to equal "}"; otherwise return False immediately. RESULT brace_pair_validated_or_false_returned]
            [COMMAND Within the closing-bracket branch, if Current_Brkt equals "[", require Brkt to equal "]"; otherwise return False immediately. RESULT square_bracket_pair_validated_or_false_returned]
            [COMMAND After the loop completes, check whether self.stack still contains any unmatched opening brackets. RESULT remaining_stack_checked]
            [COMMAND If self.stack is not empty after processing all characters, return False. RESULT false_returned_for_leftover_opening_brackets]
            [COMMAND Return True when the full expression has been processed without any mismatch and no unmatched opening brackets remain in self.stack. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A closing bracket does not match the required closing counterpart for the popped opening bracket.]
        [SEQUENTIAL_BLOCK]
            [COMMAND During loop processing of a closing bracket, if Current_Brkt is "(" and Brkt is not ")", return False immediately. RESULT result]
            [COMMAND During loop processing of a closing bracket, if Current_Brkt is "{" and Brkt is not "}", return False immediately. RESULT result]
            [COMMAND During loop processing of a closing bracket, if Current_Brkt is "[" and Brkt is not "]", return False immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: One or more opening brackets remain unmatched after all characters in self.expr have been processed.]
        [SEQUENTIAL_BLOCK]
            [COMMAND After the loop, if self.stack contains any remaining items, return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
