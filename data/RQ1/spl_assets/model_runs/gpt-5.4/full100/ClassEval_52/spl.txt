# CLASS Lemmatization

[DEFINE_WORKER: "Initialize the instance by creating and storing a default WordNetLemmatizer object on the lemmatizer attribute." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance whose lemmatizer attribute is being initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new WordNetLemmatizer instance using its no-argument constructor and assign it to self.lemmatizer for persistent reuse on this object. RESULT self.lemmatizer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalize a sentence by removing punctuation, tokenize and POS-tag its words, then lemmatize each token using POS-specific rules and return the resulting list." lemmatize_sentence]
    [INPUTS]
        <REF> sentence </REF>: string "Input sentence text to clean, tokenize, tag, and lemmatize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> lemmatized_words </REF>: list "List of lemmatized tokens produced from the cleaned and tagged sentence, in original token order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect lemmatized output tokens. RESULT lemmatized_words]
            [COMMAND Remove punctuation from the input sentence by reassigning sentence to the result of self.remove_punctuation(sentence). RESULT sentence]
            [COMMAND Tokenize the cleaned sentence into words using word_tokenize(sentence). RESULT words]
            [COMMAND Assign part-of-speech tags to the token list using pos_tag(words). RESULT tagged_words]
            [COMMAND For each (word, tag) pair in tagged_words, if tag starts with 'V', lemmatize the word with self.lemmatizer.lemmatize(word, pos='v'); else if tag starts with 'J', lemmatize it with self.lemmatizer.lemmatize(word, pos='a'); else if tag starts with 'R', lemmatize it with self.lemmatizer.lemmatize(word, pos='r'); otherwise lemmatize it with self.lemmatizer.lemmatize(word). RESULT lemmatized_word]
            [COMMAND Within the same loop iteration for each tagged word, append the computed lemmatized_word to lemmatized_words. RESULT lemmatized_words]
            [COMMAND Return the accumulated lemmatized_words list after all tagged words have been processed. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If tagged_words is empty, the loop executes zero times.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip all per-token lemmatization and appending because there are no tagged word pairs to process. RESULT lemmatized_words]
            [COMMAND Return the still-empty lemmatized_words list. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes punctuation from an input sentence, tokenizes the cleaned text, tags each token with its part of speech, and returns the list of POS tag labels." get_pos_tag]
    [INPUTS]
        <REF> sentence </REF>: text "The input sentence to clean, tokenize, and part-of-speech tag."
    [END_INPUTS]

    [OUTPUTS]
        <REF> pos_tags </REF>: list "A list containing the part-of-speech tag string for each token in the cleaned sentence."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to accumulate part-of-speech tags. RESULT pos_tags]
            [COMMAND Remove punctuation from the input sentence by calling self.remove_punctuation on the current sentence value and overwrite sentence with the cleaned text. RESULT sentence]
            [COMMAND Tokenize the cleaned sentence into word tokens using word_tokenize. RESULT words]
            [COMMAND Assign part-of-speech tags to the token list using pos_tag. RESULT tagged_words]
            [COMMAND For each tagged_word in tagged_words, append the element at index 1, which is the part-of-speech tag, to pos_tags. RESULT pos_tags]
            [COMMAND Return the accumulated list of part-of-speech tags. RESULT pos_tags]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove all punctuation characters defined by string.punctuation from the input sentence and return the resulting string." remove_punctuation]
    [INPUTS]
        <REF> sentence </REF>: string "Input text to process by removing punctuation characters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The input sentence with every character in string.punctuation removed and all other characters preserved."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the punctuation character set from string.punctuation and create a translation table with str.maketrans('', '', string.punctuation) that marks those characters for deletion without any character substitution. RESULT translation_table]
            [COMMAND Apply sentence.translate(translation_table) to remove all punctuation characters from the input sentence. RESULT result]
            [COMMAND Return the transformed string result. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
