# CLASS CurrencyConverter

[DEFINE_WORKER: "Initialize the instance currency rate mapping by assigning self.rates to a fixed dictionary of hardcoded currency-to-rate constants." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.rates to a new dictionary containing the fixed mappings 'USD' to 1.0, 'EUR' to 0.85, 'GBP' to 0.72, 'JPY' to 110.15, 'CAD' to 1.23, 'AUD' to 1.34, and 'CNY' to 6.40. RESULT self.rates]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert an amount from one currency to another using stored exchange rates, returning the original amount for identical currencies and False when either currency rate is unavailable." convert]
    [INPUTS]
        <REF> amount </REF>: number "The monetary amount to convert."
        <REF> from_currency </REF>: string "The source currency code to convert from."
        <REF> to_currency </REF>: string "The target currency code to convert to."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number_or_boolean "The original amount if both currencies are the same, the converted amount if both rates exist, or False if either currency is missing from the rates mapping."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether from_currency is different from to_currency before performing any rate lookup or conversion. RESULT currency_comparison]
            [COMMAND When from_currency and to_currency are different, verify with short-circuit logic whether from_currency is absent from self.rates or to_currency is absent from self.rates. RESULT rate_availability_check]
            [COMMAND When both currencies exist in self.rates, read self.rates[from_currency] into from_rate. RESULT from_rate]
            [COMMAND When both currencies exist in self.rates, read self.rates[to_currency] into to_rate. RESULT to_rate]
            [COMMAND When both rates have been read, compute converted_amount as (amount / from_rate) * to_rate. RESULT converted_amount]
            [COMMAND Return converted_amount. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: from_currency is equal to to_currency.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return amount unchanged without reading self.rates. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: from_currency is not in self.rates or to_currency is not in self.rates.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the list of currency codes currently present as keys in self.rates." get_supported_currencies]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the rates mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A new list of the keys from self.rates in their current mapping order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the keys from self.rates. RESULT rate_keys_view]
            [COMMAND Convert the keys view into a new list. RESULT supported_currencies]
            [COMMAND Return the resulting list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a currency rate to the stored rates map unless that currency already exists, in which case it returns false without changing anything." add_currency_rate]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the rates mapping."
        <REF> currency </REF>: key "Currency identifier used as the map key."
        <REF> rate </REF>: value "Rate value to store for the given currency."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns false if the currency already exists; otherwise stores the rate and returns null implicitly."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given currency is already present in self.rates. RESULT currency_exists]
            [COMMAND If the currency is already present, return false immediately without modifying self.rates. RESULT false]
            [COMMAND Otherwise, store the provided rate in self.rates under the given currency key. RESULT self.rates[currency]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The currency key is already present in self.rates.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the assignment to preserve the existing stored rate. RESULT no_change]
            [COMMAND Return false to signal that the rate was not added. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Update the stored exchange rate for an existing currency and return False when the currency is not present." update_currency_rate]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the rates mapping."
        <REF> currency </REF>: key "Currency identifier used to locate the rate entry."
        <REF> new_rate </REF>: value "Replacement rate value to store."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False if the currency does not exist; otherwise the method updates the rate and returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given currency exists in self.rates. RESULT currency_exists]
            [COMMAND If the currency does not exist in self.rates, return False immediately. RESULT early_return_false]
            [COMMAND Within the existing currency path, assign new_rate to self.rates[currency]. RESULT updated_rate_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
