    [OUTPUTS]
    [END_OUTPUTS]

[DEFINE_WORKER: "Creates a job listing dictionary from the provided title, company, and requirements, then appends it to the instance's job listings collection." post_job]
    [INPUTS]
        <REF> self </REF>: object "The instance that holds the job_listings collection."
        <REF> job_title </REF>: data_type "The job title value to store in the new job listing."
        <REF> company </REF>: data_type "The company value to store in the new job listing."
        <REF> requirements </REF>: list "The requirements value to store in the new job listing, indicated by the comment as a list such as ['requirement1', 'requirement2']."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a dictionary named job with fixed keys "job_title", "company", and "requirements", assigning them directly from the current values of job_title, company, and requirements without transformation or validation. RESULT job]
            [COMMAND Append the job dictionary to self.job_listings. RESULT self.job_listings]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove the first occurrence of the specified job from the instance's job_listings collection in place." remove_job]
    [INPUTS]
        <REF> self </REF>: object "Instance that owns the job_listings attribute to be mutated."
        <REF> job </REF>: object "Job value to locate and remove from self.job_listings."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "No value is returned; the effect is the in-place modification of self.job_listings."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the job_listings attribute from self as the mutable target collection. RESULT self.job_listings]
            [COMMAND Remove the first element in self.job_listings that is equal to job, mutating the collection in place. RESULT self.job_listings]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The value job is not present in self.job_listings when the removal is attempted.]
        [LOG "Removal cannot complete because no element equal to job exists in self.job_listings."]
        [THROW ValueError "Raised when attempting to remove a value that is not contained in the collection."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The object self does not provide a usable job_listings attribute, or self.job_listings does not support the remove operation.]
        [LOG "Removal cannot be performed because the target collection is missing, uninitialized, or incompatible with the remove method call."]
        [THROW AttributeError "Raised when self lacks the required job_listings attribute or the accessed target does not provide the expected operation."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search all job listings and return those whose job title contains the criteria case-insensitively or whose requirements contain an exact case-insensitive match to the criteria." search_jobs]
    [INPUTS]
        <REF> criteria </REF>: string "The search text used to match against each job listing's title and requirements."
    [END_INPUTS]

    [OUTPUTS]
        <REF> matching_jobs </REF>: list "A list of job listing objects that satisfy the case-insensitive title substring match or exact requirement membership match."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize matching_jobs as an empty list accumulator. RESULT matching_jobs]
            [COMMAND Iterate through each job_listing in self.job_listings sequentially. RESULT job_listing]
            [COMMAND Within the loop for the current job_listing, evaluate whether criteria.lower() is a substring of job_listing["job_title"].lower(); if that is true, treat the overall condition as true and do not evaluate the requirements condition because the or logic short-circuits. RESULT title_match_condition]
            [COMMAND Within the same loop iteration, only if the title substring condition is false, build a lowercase list from job_listing["requirements"] by applying lower() to each requirement r, then evaluate whether criteria.lower() is an exact member of that lowercase requirements list. RESULT requirements_match_condition]
            [COMMAND Within the same loop iteration, if either the title substring condition is true or the exact lowercase requirements membership condition is true, append the current job_listing to matching_jobs. RESULT matching_jobs]
            [COMMAND After all job listings have been processed, return matching_jobs. RESULT matching_jobs]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect and return all resumes from self.resumes that satisfy the given job's requirements." get_job_applicants]
    [INPUTS]
        <REF> job </REF>: object "Job data containing a required 'requirements' field used to test each resume."
    [END_INPUTS]

    [OUTPUTS]
        <REF> applicants </REF>: list "A list of resumes from self.resumes that match job['requirements'], preserving iteration order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize applicants as an empty list to represent the default result when no resumes match or when self.resumes is empty. RESULT applicants]
            [COMMAND Iterate over each resume in self.resumes sequentially; if self.matches_requirements(resume, job["requirements"]) evaluates to true for the current resume, append that resume to applicants, otherwise do nothing and continue to the next resume. RESULT applicants]
            [COMMAND Return the applicants list after all resumes in self.resumes have been processed. RESULT applicants]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether every skill listed in the resume is contained in the requirements collection and returns false immediately on the first missing skill, otherwise returns true." matches_requirements]
    [INPUTS]
        <REF> resume </REF>: object "Input record containing a 'skills' collection to validate."
        <REF> requirements </REF>: collection "Collection of allowed or required skills used for membership checks."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "False if any skill in resume['skills'] is not in requirements; otherwise true."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each skill in resume["skills"] one item at a time. RESULT skill]
            [COMMAND Within the loop for the current skill, check whether the skill is not contained in requirements. RESULT skill_missing_from_requirements]
            [COMMAND If the current skill is not contained in requirements, return false immediately and stop further iteration. RESULT result_false]
            [COMMAND After the loop completes without any early return, return true. RESULT result_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During iteration, at least one skill from resume["skills"] is absent from requirements.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false immediately when the first missing skill is encountered. RESULT result_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
