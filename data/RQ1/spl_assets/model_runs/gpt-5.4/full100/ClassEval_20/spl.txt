# CLASS Chat

[DEFINE_WORKER: "Initialize the instance by setting the users attribute to a new empty dictionary." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance whose users attribute is initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary to self.users, overwriting any previous value and establishing an empty mapping as the initial state. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username already exists in the user collection, and if not, adds it with an empty list and returns a success flag." add_user]
    [INPUTS]
        <REF> self </REF>: object "Object instance containing the users collection."
        <REF> username </REF>: string "Username to check for existence and add if missing."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns false if the username already exists; otherwise adds the username with an empty list and returns true."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists in self.users. RESULT username_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username exists in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false immediately without modifying self.users. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: username does not exist in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Add username to self.users and initialize its value to an empty list. RESULT self.users[username]]
            [COMMAND Return true immediately after adding the new user. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove a username from the user collection if it exists, and return whether the deletion succeeded." remove_user]
    [INPUTS]
        <REF> self </REF>: object "The object holding the users collection."
        <REF> username </REF>: any "The key to look up and delete from self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the username was found and deleted; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists in self.users. RESULT membership_check]
            [COMMAND If username exists in self.users, delete self.users[username] and return True immediately. RESULT true]
            [COMMAND If username does not exist in self.users, return False immediately. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sends a message from a sender to a receiver if both users exist, records the message with a current timestamp, stores it for both users, and returns success." send_message]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users collection."
        <REF> sender </REF>: any "Sender identifier."
        <REF> receiver </REF>: any "Receiver identifier."
        <REF> message </REF>: any "Message content."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the message is stored for both users; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the sender or receiver is missing from the users collection. RESULT validation_result]
            [COMMAND If either user is missing, return False immediately. RESULT False]
            [COMMAND Compute the current timestamp as a string using datetime.now().strftime("%Y-%m-%d %H:%M:%S"). RESULT timestamp]
            [COMMAND Create a message_info dictionary containing sender, receiver, message, and timestamp. RESULT message_info]
            [COMMAND Append the message_info dictionary to the sender's message list. RESULT sender_messages_updated]
            [COMMAND Append the same message_info dictionary to the receiver's message list. RESULT receiver_messages_updated]
            [COMMAND Return True to indicate the message was successfully sent and stored. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Sender or receiver does not exist in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without creating a timestamp or storing any message. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the stored messages for a username, or return an empty list if the username is not present in self.users." get_messages]
    [INPUTS]
        <REF> self </REF>: object "Object instance that contains the users collection."
        <REF> username </REF>: data_type "Username key to check and use for retrieving stored messages."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "An empty list if the username is absent from self.users; otherwise the value stored in self.users for that username."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists in self.users. RESULT username_exists]
            [COMMAND If username exists in self.users, return the value stored at self.users[username] without modification. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not present in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately as an early exit. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
