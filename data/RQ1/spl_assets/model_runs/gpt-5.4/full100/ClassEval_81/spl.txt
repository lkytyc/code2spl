# CLASS Statistics3

[DEFINE_WORKER: "Sort the input data and return the median value, using the middle element for odd lengths and the average of the two middle elements for even lengths." median]
    [INPUTS]
        <REF> data </REF>: sequence "Input collection of numeric values to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The median value computed from the input data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the input data into a new collection and store it as sorted_data. RESULT sorted_data]
            [COMMAND Compute the length of sorted_data and store it as n. RESULT n]
            [COMMAND Check whether n is odd by evaluating n % 2 == 1. RESULT is_odd]
            [COMMAND If n is odd, return the element at index n // 2 from sorted_data as the median. RESULT result]
            [COMMAND If n is even, return the average of the two middle elements at indexes n // 2 - 1 and n // 2 from sorted_data. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n % 2 == 1 is false, so the sorted data length is even.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the average of the two middle elements at indexes n // 2 - 1 and n // 2 from sorted_data. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Count each distinct value in the input data, then return all values tied for the highest frequency." mode]
    [INPUTS]
        <REF> data </REF>: iterable "Input collection of values to analyze."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of all values that occur most frequently."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty dictionary named counts to store frequency totals for each value. RESULT counts]
            [COMMAND Iterate through each value in data and increment that value's count in counts, starting from 0 when the value is not yet present. RESULT counts]
            [COMMAND Compute max_count as the maximum frequency among counts values. RESULT max_count]
            [COMMAND Build mode_values by selecting every value from counts whose count equals max_count. RESULT mode_values]
            [COMMAND Return mode_values. RESULT mode_values]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data contains no values, so counts is empty when max_count is computed.]
        [LOG "max() cannot be applied to an empty sequence of counts."]
        [THROW ValueError "Raised when attempting to compute the maximum frequency from an empty collection."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the Pearson correlation coefficient between two sequences, returning None when the standard deviation product is zero." correlation]
    [INPUTS]
        <REF> x </REF>: sequence "First numeric sequence used to compute the mean, covariance, and variance term."
        <REF> y </REF>: sequence "Second numeric sequence used to compute the mean, covariance, and variance term."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or null "The correlation coefficient, or None if the denominator is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of x. RESULT n]
            [COMMAND Compute mean_x as the sum of x divided by n. RESULT mean_x]
            [COMMAND Compute mean_y as the sum of y divided by n. RESULT mean_y]
            [COMMAND Compute numerator as the sum of (xi - mean_x) * (yi - mean_y) for each paired xi, yi from zip(x, y). RESULT numerator]
            [COMMAND Compute denominator as the square root of the product of the sum of squared deviations of x from mean_x and the sum of squared deviations of y from mean_y. RESULT denominator]
            [COMMAND If denominator equals 0, return None immediately. RESULT None]
            [COMMAND Otherwise, return numerator divided by denominator. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If x and y have different lengths, zip(x, y) truncates to the shorter sequence when computing the numerator.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use only paired elements up to the shorter input length when summing covariance terms. RESULT numerator]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If n equals 0, the mean calculations divide by zero before reaching the denominator check.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempting to compute mean_x or mean_y with n equal to 0 causes a division by zero error. RESULT error]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the arithmetic mean of a non-empty sequence, or None when the input sequence is empty." mean]
    [INPUTS]
        <REF> data </REF>: sequence "Input collection of numeric values to average."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or null "The arithmetic mean of the input values, or None if the input is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the input data length is exactly 0. RESULT is_empty]
            [COMMAND If the input data is empty, return None immediately as an early exit. RESULT None]
            [COMMAND Otherwise, compute the total of all values in data and divide it by the number of values. RESULT mean_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a square matrix of pairwise column correlations for the input data." correlation_matrix]
    [INPUTS]
        <REF> data </REF>: list "Two-dimensional data where each inner row is indexed by column position."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of list "A square correlation matrix containing the correlation for every pair of columns."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize matrix as an empty list. RESULT matrix]
            [COMMAND For each column index i from 0 to len(data[0]) - 1, create a new empty row for the current output row. RESULT row]
            [COMMAND Within the outer loop over i, for each column index j from 0 to len(data[0]) - 1, build column1 from the i-th element of every row in data. RESULT column1]
            [COMMAND Within the outer loop over i, for each column index j from 0 to len(data[0]) - 1, build column2 from the j-th element of every row in data. RESULT column2]
            [COMMAND Within the nested loops over i and j, compute the correlation of column1 and column2 using Statistics3.correlation. RESULT correlation]
            [COMMAND Append the computed correlation to the current row. RESULT row]
            [COMMAND After finishing the inner loop for a given i, append the completed row to matrix. RESULT matrix]
            [COMMAND Return the completed matrix. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the sample standard deviation of the input data by returning None for fewer than two items, then calculating the mean, sample variance using n - 1, and the square root of that variance." standard_deviation]
    [INPUTS]
        <REF> data </REF>: collection "The input sequence of numeric values whose sample standard deviation is to be computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number_or_null "The sample standard deviation of the input data, or None when the input contains fewer than two values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the number of elements in data using len(data). RESULT n]
            [COMMAND If n is not less than 2, compute the mean of data using Statistics3.mean(data). RESULT mean_value]
            [COMMAND If n is not less than 2, compute the sample variance as the sum over each x in data of (x - mean_value) squared, divided by n - 1. RESULT variance]
            [COMMAND If n is not less than 2, return the square root of variance using math.sqrt(variance). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n is less than 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately without computing the mean, variance, or square root. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the mean and standard deviation of the input data, return None when the standard deviation is missing or zero, otherwise return the z-score for each value." z_score]
    [INPUTS]
        <REF> data </REF>: sequence of numbers "Input values to normalize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of numbers or null "A list of z-scores for the input values, or null when normalization is not possible."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the mean of the input data using Statistics3.mean. RESULT mean]
            [COMMAND Compute the standard deviation of the input data using Statistics3.standard_deviation. RESULT std_deviation]
            [COMMAND Check whether the standard deviation is None or equal to 0. RESULT condition]
            [COMMAND If the standard deviation is None or 0, return None immediately. RESULT result]
            [COMMAND Otherwise, compute and return a list of z-scores for each x in data using (x - mean) / std_deviation. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The standard deviation is None or equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None without computing normalized values. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
