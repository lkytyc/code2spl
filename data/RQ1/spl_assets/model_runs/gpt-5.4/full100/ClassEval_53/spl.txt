# CLASS LongestWord

[DEFINE_WORKER: "Initialize the instance by setting self.word_list to a new empty list." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance whose word_list attribute is being initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a fresh empty list to the instance attribute self.word_list. RESULT self.word_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Append the provided word to the end of the object's existing word list." add_word]
    [INPUTS]
        <REF> self </REF>: object "Object instance that owns the existing word_list collection."
        <REF> word </REF>: any "Value or object reference to append to the end of self.word_list."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append the current value or reference of word to the end of self.word_list, mutating the existing list in place without transforming word. RESULT self.word_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalize an input sentence by lowercasing and removing punctuation, then return the first longest token that appears in self.word_list, or an empty string if none qualify." find_longest_word]
    [INPUTS]
        <REF> sentence </REF>: string "The input sentence to normalize, sanitize, and split into candidate words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> longest_word </REF>: string "The longest word from the processed sentence that is present in self.word_list, or an empty string if no matching word is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize longest_word to the empty string as the default result when no qualifying word is found. RESULT longest_word]
            [COMMAND Convert sentence to lowercase so later matching against self.word_list is case-insensitive relative to the transformed input. RESULT sentence]
            [COMMAND Remove all punctuation characters from sentence by replacing every character matched by the regex pattern built from re.escape(string.punctuation) with an empty string. RESULT sentence]
            [COMMAND Split sentence on the single-space delimiter ' ' to transform it from a string into a list of tokens. RESULT sentence]
            [COMMAND Iterate over each word in the token list sentence in order. RESULT word]
            [COMMAND Within the loop over each word, check whether word is in self.word_list and, only if that membership test is true, check whether len(word) is strictly greater than len(longest_word). RESULT is_qualifying_longer_word]
            [COMMAND Within the same loop iteration, if both conditions are true, assign longest_word = word; because the comparison is strict '>', equal-length matches do not replace the current value, so the first longest qualifying word is preserved. RESULT longest_word]
            [COMMAND After all tokens have been processed, return longest_word. RESULT longest_word]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
