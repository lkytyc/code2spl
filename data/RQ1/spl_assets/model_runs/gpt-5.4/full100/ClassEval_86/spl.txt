# CLASS TicTacToe

[DEFINE_WORKER: "Initializes a 3-row game board with N columns filled with spaces and sets the current player to X." __init__]
    [INPUTS]
        <REF> N </REF>: int "Number of columns used to build each board row; defaults to 3."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create the board as a list containing 3 separate rows, where each row is a new list of length N filled with the space character. RESULT self.board]
            [COMMAND Set the current player to the constant string X. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to place the current player's mark on an empty board cell, switches players, and returns whether the move succeeded." make_move]
    [INPUTS]
        <REF> row </REF>: int "Row index of the target board cell."
        <REF> col </REF>: int "Column index of the target board cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was placed on an empty cell, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.board[row][col] is exactly the space character ' '. RESULT is_empty]
            [COMMAND If the cell is empty, set self.board[row][col] to self.current_player. RESULT board_updated]
            [COMMAND If the cell is empty, toggle self.current_player from 'X' to 'O' or from 'O' to 'X'. RESULT current_player_updated]
            [COMMAND If the cell is empty, return True immediately. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When self.board[row][col] is not exactly the space character ' '.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without changing the board or current_player. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Check a 3x3 board for any winning row, column, or diagonal and return the winning symbol, or None if no winner exists." check_winner]
    [INPUTS]
        <REF> self </REF>: object "Object containing a 3x3 board in self.board."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The winning symbol, or None if there is no winner."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each row in self.board, and if the three entries in the current row are all equal and not a space, return that row's symbol immediately. RESULT row winner]
            [COMMAND Iterate through columns 0 to 2, and if the three entries in the current column are all equal and not a space, return that column's symbol immediately. RESULT column winner]
            [COMMAND Check the main diagonal from top-left to bottom-right, and if all three entries are equal and not a space, return the top-left symbol. RESULT main_diagonal_winner]
            [COMMAND Check the anti-diagonal from top-right to bottom-left, and if all three entries are equal and not a space, return the top-right symbol. RESULT anti_diagonal_winner]
            [COMMAND Return None when no row, column, or diagonal contains three matching non-space symbols. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether every row on the board contains no space characters and returns False immediately when any empty cell is found." is_board_full]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the board to inspect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the board contains no spaces in any row; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each row in self.board. RESULT row]
            [COMMAND Within the loop over each row, check whether the row contains a space character ' '. RESULT contains_space]
            [COMMAND If any row contains a space character, return False immediately and stop scanning further rows. RESULT false]
            [COMMAND If the loop completes without finding any space characters, return True. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
