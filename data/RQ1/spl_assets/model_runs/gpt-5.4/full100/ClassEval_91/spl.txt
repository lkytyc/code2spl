# CLASS UrlPath

[DEFINE_WORKER: "Initializes the instance with an empty segments list and disables the end tag flag." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance segments attribute to an empty list. RESULT self.segments]
            [COMMAND Set the instance with_end_tag attribute to false. RESULT self.with_end_tag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Appends a path-normalized version of the given segment to the segment list." add]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the segments collection and path normalization logic."
        <REF> segment </REF>: any "Segment value to normalize and append."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the normalized form of the input segment by calling self.fix_path(segment). RESULT normalized_segment]
            [COMMAND Append normalized_segment to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses a path into URL-decoded segments, marking trailing-slash paths and normalizing the path before splitting and storing each segment." parse]
    [INPUTS]
        <REF> path </REF>: string "Input path to normalize and split into segments."
        <REF> charset </REF>: string "Character set used to URL-decode each path segment."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: void "No return value; updates object state by setting the trailing-slash flag and appending decoded segments."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If path is truthy, continue parsing; otherwise skip all path processing. RESULT path]
            [COMMAND Within the outer path check, if the original path ends with '/' set self.with_end_tag to True. RESULT self.with_end_tag]
            [COMMAND Within the outer path check, normalize the path by calling self.fix_path(path). RESULT path]
            [COMMAND Within the outer path check, if the normalized path is truthy, split it on '/' into segments. RESULT split]
            [COMMAND Within the split segments loop, URL-decode each segment using urllib.parse.unquote(seg, encoding=charset). RESULT decoded_seg]
            [COMMAND Within the split segments loop, append each decoded segment to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns an empty string for a falsy path input, otherwise returns the path with all leading and trailing forward slashes removed while preserving interior slashes." fix_path]
    [INPUTS]
        <REF> path </REF>: data "Input path value to normalize by trimming boundary forward slashes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "An empty string when the input path is falsy, otherwise the input path with leading and trailing '/' characters removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether path is falsy. RESULT path_is_falsy]
            [COMMAND If path is not falsy, assign segment_str to the result of stripping all leading and trailing '/' characters from path, preserving any interior slashes. RESULT segment_str]
            [COMMAND Return segment_str. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: path is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
