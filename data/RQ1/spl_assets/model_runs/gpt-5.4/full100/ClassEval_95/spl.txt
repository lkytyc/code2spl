# CLASS Warehouse

[DEFINE_WORKER: "Initialize the instance with empty inventory and order dictionaries as default container state." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary to self.inventory to initialize the product inventory mapping with no entries. RESULT self.inventory]
            [COMMAND Assign a new empty dictionary to self.orders to initialize the order storage mapping with no entries. RESULT self.orders]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Add a new product record to the inventory when the product ID is absent, otherwise increase the existing product's quantity by the provided amount." add_product]
    [INPUTS]
        <REF> product_id </REF>: any "The identifier used as the key for the product in the inventory."
        <REF> name </REF>: string "The product name to store when creating a new inventory entry."
        <REF> quantity </REF>: number "The quantity to assign to a new product or add to an existing product's quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is absent from self.inventory using a dictionary-key membership test. RESULT product_absent]
            [COMMAND If product_absent is true, create self.inventory[product_id] as a new record with name set to name and quantity set to quantity. RESULT inventory_entry_created]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: product_id is already present in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.inventory[product_id]['quantity'] by quantity without changing the existing name field. RESULT inventory_quantity_updated]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Update the stored quantity for an existing product in the inventory by adding the provided quantity when the product ID is present." update_product_quantity]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
        <REF> product_id </REF>: data_type "Identifier of the product to update in the inventory."
        <REF> quantity </REF>: data_type "Amount to add to the product's existing quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT product_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: product_id exists in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.inventory[product_id]['quantity'] by quantity using an in-place addition update. RESULT updated_inventory_quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: product_id does not exist in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do nothing and leave the inventory unchanged. RESULT inventory_unchanged]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the quantity for a given product ID from the inventory when the product exists, otherwise return False." get_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: any "The product identifier to look up in the inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: quantity_or_boolean "The product quantity if the product ID exists in the inventory; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT product_exists]
            [COMMAND If product_exists is true, return self.inventory[product_id]['quantity'] immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: product_id does not exist as a key in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately to indicate that the product was not found. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the requested product quantity is available, and if so decreases inventory and stores the order as shipped; otherwise returns False." create_order]
    [INPUTS]
        <REF> order_id </REF>: any "Identifier used as the key for the order record."
        <REF> product_id </REF>: any "Identifier of the product being ordered."
        <REF> quantity </REF>: number "Requested quantity to fulfill from inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean_or_implicit_none "Returns False when available stock is less than the requested quantity; otherwise completes the inventory update and order creation without an explicit return value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.get_product_quantity(product_id) is greater than or equal to quantity, using an inclusive boundary where equality is sufficient. RESULT stock_is_sufficient]
            [COMMAND If stock_is_sufficient is true, call self.update_product_quantity(product_id, -quantity) to decrease the stored product quantity by the requested amount. RESULT inventory_updated]
            [COMMAND If stock_is_sufficient is true, store a new order record in self.orders[order_id] with product_id set to product_id, quantity set to quantity, and status set to 'Shipped'. RESULT order_record_stored]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Available stock is less than the requested quantity, meaning self.get_product_quantity(product_id) < quantity.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without updating inventory and without creating an order record. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates an existing order's status in self.orders and returns False when the order_id is missing." change_order_status]
    [INPUTS]
        <REF> order_id </REF>: any "Order key to look up in self.orders."
        <REF> status </REF>: any "New status value to store for the order."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns False if the order does not exist; otherwise no explicit value is returned."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT order_exists]
            [COMMAND If order_id exists, update self.orders[order_id]['status'] to status. RESULT updated_order]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_id is not present in self.orders]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying any order data. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the given order ID exists in the orders collection and returns its status if found, otherwise returns False." track_order]
    [INPUTS]
        <REF> order_id </REF>: any "Identifier of the order to look up in self.orders."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The order status value when the order ID exists, or False when it does not exist."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT order_exists]
            [COMMAND If order_exists is true, return self.orders[order_id]['status'] immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_id does not exist in self.orders]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
