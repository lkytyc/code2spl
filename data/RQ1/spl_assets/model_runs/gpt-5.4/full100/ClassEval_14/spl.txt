# CLASS BookManagementDB

[DEFINE_WORKER: "Initialize the instance by opening a SQLite connection from the provided database name, creating a cursor from that connection, and then invoking create_table on the instance." __init__]
    [INPUTS]
        <REF> db_name </REF>: text "Database name or identifier used to open the SQLite connection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection using db_name and assign the returned connection object to self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection by calling its cursor method and assign the returned cursor object to self.cursor. RESULT self.cursor]
            [COMMAND Invoke self.create_table after self.connection and self.cursor have been initialized. RESULT table_initialization_completed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create the books table with id, title, author, and available columns if it does not already exist, then commit the schema change to the database." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a SQL CREATE TABLE IF NOT EXISTS statement through self.cursor to ensure the books table exists with columns id INTEGER PRIMARY KEY, title TEXT, author TEXT, and available INTEGER. RESULT books_table_creation_applied_or_skipped]
            [COMMAND Commit the current database transaction through self.connection to persist the table creation operation. RESULT schema_change_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a new book record with the given title and author, marks it available by default, and commits the transaction." add_book]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database cursor and connection."
        <REF> title </REF>: string "Book title to insert."
        <REF> author </REF>: string "Book author to insert."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute an INSERT statement on the books table using the provided title and author, and set available to the constant value 1. RESULT inserted_row]
            [COMMAND Commit the database transaction after the insert completes. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes the book row whose id matches the input book_id from the books table and then commits the database transaction." remove_book]
    [INPUTS]
        <REF> book_id </REF>: data_type "Identifier value used as the bound parameter in the DELETE statement to select the book row to remove."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL DELETE statement on the books table with condition id = ? and bind the input book_id as the single parameter. RESULT delete_operation]
            [COMMAND Commit the current database transaction to persist the deletion effect. RESULT committed_transaction]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If book_id is None, invalid for the database comparison, or does not match any row in books.id.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the same parameterized SQL DELETE statement with the provided book_id value, which may result in zero rows being deleted. RESULT delete_operation_with_no_matching_row]
            [COMMAND Commit the current database transaction even if no row was deleted. RESULT committed_no_op_transaction]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks the book with the given identifier as unavailable in the database and commits the update to persistent storage." borrow_book]
    [INPUTS]
        <REF> book_id </REF>: data_type "Identifier of the book record to update in the books table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a database update that sets the available column to the numeric constant 0 in the books table for the row whose id equals the provided book_id, passing book_id unchanged as a bound SQL parameter. RESULT database_update_applied]
            [COMMAND Commit the current database transaction so the update is persisted beyond the transactional buffer. RESULT transaction_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The provided book_id does not match any existing row in the books table.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the same update statement, resulting in zero rows being modified while still proceeding to commit the transaction. RESULT no_matching_row_updated]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks the book with the given ID as available in the books table by setting available to 1 and then commits the database transaction." return_book]
    [INPUTS]
        <REF> book_id </REF>: unknown "Identifier of the book row to update in the books table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a database UPDATE statement on the books table to set available = 1 for the row whose id equals the provided book_id, using book_id as a bound SQL parameter without transformation. RESULT update_execution]
            [COMMAND Commit the current database transaction to persist the availability change. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a database query to retrieve all rows from the books table, stores the fetched result set in a local variable, and returns it." search_books]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the database cursor used to execute the query and fetch results."
    [END_INPUTS]

    [OUTPUTS]
        <REF> books </REF>: list "All rows returned by the query on the books table, typically as a list of row objects or tuples, with an empty list as the common no-result case."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query `SELECT * FROM books` using `self.cursor` to request all columns and all rows from the `books` table. RESULT query_executed]
            [COMMAND Fetch all remaining rows from the current result set on `self.cursor` and store them in `books`. RESULT books]
            [COMMAND Return `books`. RESULT returned_books]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
