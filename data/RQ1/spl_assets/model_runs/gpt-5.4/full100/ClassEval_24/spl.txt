[DEFINE_WORKER: "Add two complex-like inputs by summing their real and imaginary components and returning a new complex value built from those sums." add]
    [INPUTS]
        <REF> c1 </REF>: complex-like object "First input providing real and imag components."
        <REF> c2 </REF>: complex-like object "Second input providing real and imag components."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: complex "A new complex value whose real part is c1.real + c2.real and whose imaginary part is c1.imag + c2.imag."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the summed real component as c1.real + c2.real. RESULT real]
            [COMMAND Compute the summed imaginary component as c1.imag + c2.imag. RESULT imaginary]
            [COMMAND Construct a new complex value from the computed real and imaginary components. RESULT answer]
            [COMMAND Return the constructed complex result. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a new complex number whose real and imaginary parts are the pairwise differences of the input complex numbers." subtract]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex number operand."
        <REF> c2 </REF>: complex "Second complex number operand."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "Complex number formed by subtracting c2 from c1 component-wise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the real part by subtracting c2.real from c1.real. RESULT real]
            [COMMAND Compute the imaginary part by subtracting c2.imag from c1.imag. RESULT imaginary]
            [COMMAND Return a complex number constructed from the computed real and imaginary parts. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the product of two complex numbers from their real and imaginary parts and returns the resulting complex value." multiply]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex operand."
        <REF> c2 </REF>: complex "Second complex operand."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The complex product of c1 and c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the real and imaginary components from c1 and c2. RESULT c1.real, c1.imag, c2.real, c2.imag]
            [COMMAND Compute the real part as c1.real * c2.real - c1.imag * c2.imag. RESULT real]
            [COMMAND Compute the imaginary part as c1.real * c2.imag + c1.imag * c2.real. RESULT imaginary]
            [COMMAND Return a complex number built from the computed real and imaginary parts. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Divides one complex number by another by computing the quotient from their real and imaginary parts and returning the resulting complex value." divide]
    [INPUTS]
        <REF> c1 </REF>: complex "Dividend complex number whose real and imaginary parts are read."
        <REF> c2 </REF>: complex "Divisor complex number whose real and imaginary parts are read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The complex quotient of c1 divided by c2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute denominator as c2.real squared plus c2.imag squared. RESULT denominator]
            [COMMAND Compute the real part of the quotient as (c1.real * c2.real + c1.imag * c2.imag) divided by denominator. RESULT real]
            [COMMAND Compute the imaginary part of the quotient as (c1.imag * c2.real - c1.real * c2.imag) divided by denominator. RESULT imaginary]
            [COMMAND Return a complex number built from real and imaginary. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Denominator is zero, causing division by zero during the quotient calculations.]
        [LOG "Division by zero occurs because c2.real squared plus c2.imag squared equals zero."]
        [THROW ZeroDivisionError "Raised when the divisor complex number has zero magnitude."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
