# CLASS BlackjackGame

[DEFINE_WORKER: "Initialize a new instance by creating the deck and setting both the player and dealer hands to empty lists." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.create_deck() and assign its return value to the instance variable self.deck. RESULT self.deck]
            [COMMAND Assign an empty list to the instance variable self.player_hand. RESULT self.player_hand]
            [COMMAND Assign an empty list to the instance variable self.dealer_hand. RESULT self.dealer_hand]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a standard 52-card deck by combining each rank with each suit, then shuffles and returns the deck." create_deck]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> deck </REF>: list of strings "A shuffled list of 52 card codes formed from rank-suit combinations."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named deck. RESULT deck]
            [COMMAND Define suits as the four suit symbols S, C, D, and H. RESULT suits]
            [COMMAND Define ranks as the thirteen rank symbols A through K, including 10. RESULT ranks]
            [COMMAND Within the outer for loop over each suit in suits, within the inner for loop over each rank in ranks, append the concatenation of rank and suit to deck. RESULT deck]
            [COMMAND Shuffle deck in place using random.shuffle. RESULT deck]
            [COMMAND Return deck. RESULT deck]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate a blackjack hand total by summing card ranks, treating face cards as 10 and aces as 11 before reducing aces as needed to avoid busting over 21." calculate_hand_value]
    [INPUTS]
        <REF> hand </REF>: list "List of card strings whose rank is all characters except the final suit character."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The computed blackjack hand value after ace adjustment."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the running total to 0 and initialize the ace counter to 0. RESULT value, num_aces]
            [COMMAND Within the loop over each card in hand, extract the rank by removing the final character from the card string. RESULT rank]
            [COMMAND For each extracted rank, if it is numeric, add its integer value to the running total. RESULT value]
            [COMMAND Otherwise, if the rank is J, Q, or K, add 10 to the running total. RESULT value]
            [COMMAND Otherwise, if the rank is A, add 11 to the running total and increment the ace counter by 1. RESULT value, num_aces]
            [COMMAND After all cards are processed, while the total is greater than 21 and at least one ace remains adjustable, subtract 10 from the total and decrement the ace counter by 1. RESULT value, num_aces]
            [COMMAND Return the final adjusted total. RESULT value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether the player or dealer wins by calculating both hand values and applying bust, comparison, and tie-breaking rules around the threshold of 21." check_winner]
    [INPUTS]
        <REF> player_hand </REF>: data "The player's hand to evaluate."
        <REF> dealer_hand </REF>: data "The dealer's hand to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A winner message indicating either 'Player wins' or 'Dealer wins'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute player_value by calling self.calculate_hand_value with player_hand. RESULT player_value]
            [COMMAND Compute dealer_value by calling self.calculate_hand_value with dealer_hand. RESULT dealer_value]
            [COMMAND Check whether both player_value and dealer_value are strictly greater than 21. RESULT both_over_21]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: both player_value and dealer_value are strictly greater than 21]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the both-over-21 branch, check whether player_value is less than or equal to dealer_value. RESULT player_value_less_or_equal_dealer_value]
            [COMMAND If player_value is less than or equal to dealer_value, return 'Player wins' immediately. RESULT result]
            [COMMAND Otherwise, return 'Dealer wins' immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: both_over_21 is false and player_value is strictly greater than 21]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 'Dealer wins' immediately because only the player is over 21. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: both_over_21 is false, player_value is not strictly greater than 21, and dealer_value is strictly greater than 21]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 'Player wins' immediately because only the dealer is over 21. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: neither player_value nor dealer_value is strictly greater than 21]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the neither-over-21 branch, check whether player_value is less than or equal to dealer_value. RESULT player_value_less_or_equal_dealer_value]
            [COMMAND If player_value is less than or equal to dealer_value, return 'Dealer wins' immediately. RESULT result]
            [COMMAND Otherwise, return 'Player wins' immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
