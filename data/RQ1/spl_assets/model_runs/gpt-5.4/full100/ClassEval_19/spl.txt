# CLASS ChandrasekharSieve

[DEFINE_WORKER: "Initialize the instance by storing the provided n value and then computing and storing primes using the object's prime-generation method." __init__]
    [INPUTS]
        <REF> self </REF>: object "The instance being initialized."
        <REF> n </REF>: data_type "The input value assigned directly to the instance attribute self.n and used indirectly to compute self.primes."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input value n directly to the instance attribute self.n without applying validation or transformation. RESULT self.n]
            [COMMAND Call self.generate_primes() after self.n has been assigned, and store the returned value in the instance attribute self.primes. RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates all prime numbers from 2 through self.n using a boolean sieve and returns them as a list." generate_primes]
    [INPUTS]
        <REF> self.n </REF>: integer "Upper bound for prime generation, inclusive."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[int] "A list of all prime numbers from 2 to self.n, or an empty list when self.n is less than 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.n is less than 2. RESULT boundary_check]
            [COMMAND If self.n is less than 2, return an empty list immediately. RESULT []]
            [COMMAND Create a boolean sieve list of length self.n + 1 with every entry initialized to True. RESULT sieve]
            [COMMAND Set sieve[0] and sieve[1] to False to exclude 0 and 1 from prime consideration. RESULT sieve]
            [COMMAND Initialize the candidate factor p to 2. RESULT p]
            [COMMAND While p * p is less than or equal to self.n, inspect the current candidate p. RESULT while_loop]
            [COMMAND Within the while loop, if sieve[p] is True, mark every multiple of p from p * p through self.n inclusive, stepping by p, as False in the sieve. RESULT sieve]
            [COMMAND Within the while loop, increment p by 1 after each iteration regardless of whether sieve[p] was True. RESULT p]
            [COMMAND Create an empty list named primes. RESULT primes]
            [COMMAND For each i from 2 through self.n inclusive, if sieve[i] is True, append i to primes. RESULT primes]
            [COMMAND Return the primes list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the current value stored in the instance attribute self.primes without modifying it." get_primes]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the primes attribute to be read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The current value of self.primes as stored on the instance."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of the instance attribute self.primes and return it directly to the caller without any transformation or additional processing. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
