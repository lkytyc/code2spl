# CLASS ExcelProcessor

    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Enter the constructor and perform no operation. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[DEFINE_WORKER: "Loads an Excel workbook from the given file, reads all rows from the active sheet into a list of row tuples, and returns None if any error occurs." read_excel]
    [INPUTS]
        <REF> file_name </REF>: text "Path or name of the Excel file to load."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list_or_none "A list containing each row from the active worksheet as value-only row data, or None if loading or reading fails."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect worksheet rows. RESULT data]
            [COMMAND Attempt to load the Excel workbook from file_name. RESULT workbook]
            [COMMAND From the loaded workbook, select the active worksheet. RESULT sheet]
            [COMMAND Within the iteration over all rows returned by sheet.iter_rows with values_only set to True, append each row value sequence to data. RESULT data]
            [COMMAND Close the workbook after all rows have been read. RESULT workbook_closed]
            [COMMAND Return the collected row data list. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while loading the workbook, accessing the active sheet, iterating rows, appending data, or closing the workbook.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the fallback result. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create an Excel workbook, write each row from the input data into the active sheet, save it to the specified file name, and return 1 on success or 0 if any error occurs." write_excel]
    [INPUTS]
        <REF> data </REF>: list "Row data to append to the active worksheet, where each item represents one row."
        <REF> file_name </REF>: string "Target file path or file name for saving the Excel workbook."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 1 if the workbook is written and closed successfully, or 0 if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Start a protected execution block for creating and writing an Excel workbook. RESULT protected_write_context]
            [COMMAND Create a new Excel workbook instance using openpyxl.Workbook(). RESULT workbook]
            [COMMAND Get the active worksheet from the workbook. RESULT sheet]
            [COMMAND For each row in data, append that row to the active worksheet. RESULT populated_sheet]
            [COMMAND Save the workbook to file_name. RESULT saved_workbook]
            [COMMAND Close the workbook after saving. RESULT closed_workbook]
            [COMMAND Return the integer 1 to indicate successful completion. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs during workbook creation, worksheet access, row appending, saving, or closing.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception without rethrowing it. RESULT caught_exception]
            [COMMAND Return the integer 0 to indicate failure. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Read spreadsheet data from a file, return 0 when the data is missing or the target column is out of range, otherwise append a transformed copy of the target column to each row, save the result to a new Excel file, and return the write result with the new file name." process_excel_data]
    [INPUTS]
        <REF> self </REF>: object "Instance providing read_excel and write_excel methods."
        <REF> N </REF>: int "Target column index to inspect and append from each row."
        <REF> save_file_name </REF>: str "Input Excel file name or path used for reading and for deriving the output file name."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple_or_int "Returns 0 if input data is missing or the target index is invalid; otherwise returns the write status and the generated output file name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call read_excel with save_file_name to load the spreadsheet data. RESULT data]
            [COMMAND If data is None or N is greater than or equal to the length of the first row, stop and return 0. RESULT 0]
            [COMMAND Initialize an empty list to collect transformed rows. RESULT new_data]
            [COMMAND Within the outer loop over each row in data, create a shallow copy of the current row. RESULT new_row]
            [COMMAND Within the outer loop over each row in data, if the value at column N is not composed only of digits, append its uppercase string form to new_row; otherwise append the original value at column N. RESULT new_row]
            [COMMAND Within the outer loop over each row in data, append the completed new_row to new_data. RESULT new_data]
            [COMMAND Derive the output file name by taking save_file_name before the first dot and appending _process.xlsx. RESULT new_file_name]
            [COMMAND Write new_data to new_file_name using write_excel and store the write result. RESULT success]
            [COMMAND Return the write result together with the generated output file name. RESULT success, new_file_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
