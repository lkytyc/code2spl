# CLASS DiscountStrategy

[DEFINE_WORKER: "Initialize the object by storing the customer, cart, and optional promotion, then cache the computed total." __init__]
    [INPUTS]
        <REF> customer </REF>: unknown "Customer value to store on the instance."
        <REF> cart </REF>: unknown "Cart value to store on the instance."
        <REF> promotion </REF>: unknown "Optional promotion value to store on the instance; defaults to none."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming customer value to the instance customer attribute. RESULT self.customer]
            [COMMAND Assign the incoming cart value to the instance cart attribute. RESULT self.cart]
            [COMMAND Assign the incoming promotion value to the instance promotion attribute. RESULT self.promotion]
            [COMMAND Compute the current total by calling the instance total method and store the result in the private total cache attribute. RESULT self.__total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the cart total by summing each item's quantity multiplied by its price, stores it on the object, and returns that total." total]
    [INPUTS]
        <REF> self </REF>: object "Instance containing cart data and the total field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed total of all cart items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each item in self.cart and compute the product of item['quantity'] and item['price'] for every item. RESULT line_item_values]
            [COMMAND Sum all computed line item values using sum with its implicit zero initial value. RESULT total_value]
            [COMMAND Assign the computed total value to self.__total. RESULT self.__total]
            [COMMAND Return self.__total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the amount due by subtracting an optional promotion discount from the total, using zero discount when no promotion exists." due]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the total amount and optional promotion callable."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The total amount due after applying any discount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.promotion is None. RESULT promotion_absent]
            [COMMAND If self.promotion is None, set discount to 0; otherwise call self.promotion(self) and store the returned value in discount. RESULT discount]
            [COMMAND Return self.__total minus discount. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.promotion is None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set discount to 0. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.promotion is not None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.promotion(self) and use the returned value as the discount. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a 5% discount on the order total when the customer's fidelity score is at least 1000, otherwise returns 0." FidelityPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object providing customer fidelity data and a total() method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "Discount amount, either 5% of the order total or 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read order.customer['fidelity'] and compare it against the threshold 1000 using an inclusive greater-than-or-equal condition. RESULT fidelity_check]
            [COMMAND If order.customer['fidelity'] is at least 1000, call order.total() and multiply the result by 0.05. RESULT discount]
            [COMMAND If order.customer['fidelity'] is below 1000, return 0 instead of any discount. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total discount for an order by adding 10% of the price of each cart item whose quantity is at least 20." BulkItemPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object containing a cart collection of items with quantity and price fields."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The total discount amount accumulated from qualifying cart items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize discount to 0. RESULT discount]
            [COMMAND Within the loop over order.cart, inspect each item one by one. RESULT item]
            [COMMAND For each item, if item['quantity'] is at least 20, add item['quantity'] * item['price'] * 0.1 to discount; otherwise, leave discount unchanged. RESULT discount]
            [COMMAND After all cart items are processed, return discount. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate a 7% discount based on the order total when the cart contains at least 10 distinct products, otherwise return 0." LargeOrderPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object providing cart contents and a total() method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "Discount amount equal to 7% of the order total when there are at least 10 distinct products in the cart, otherwise 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read order.cart and build a set of item['product'] values to obtain the distinct products present in the cart. RESULT distinct_products]
            [COMMAND Count the number of distinct products by taking the length of the distinct product set. RESULT distinct_product_count]
            [COMMAND If distinct_product_count is greater than or equal to 10, call order.total() and multiply the returned total by 0.07. RESULT discount_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the number of distinct products in the cart is less than 10.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the constant value 0 without using order.total(). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If the number of distinct products in the cart is greater than or equal to 10.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the computed discount amount, where exactly 10 distinct products also qualifies for the calculation. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
