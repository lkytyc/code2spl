# CLASS MovieBookingSystem

[DEFINE_WORKER: "Initialize the instance by assigning the movies attribute to a new empty list." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.movies to a newly created empty list so the instance starts with no stored movies. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create a movie record from the provided name, price, parsed start and end times, and an n by n zero-initialized seat matrix, then append it to the movies collection on the instance." add_movie]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the movies collection to update."
        <REF> name </REF>: data "Movie name stored directly in the movie record."
        <REF> price </REF>: data "Movie price stored directly in the movie record."
        <REF> start_time </REF>: string "Start time string expected to match the %H:%M format for parsing."
        <REF> end_time </REF>: string "End time string expected to match the %H:%M format for parsing."
        <REF> n </REF>: integer "Dimension used to create an n by n zero-initialized seats matrix."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self.movies </REF>: collection "Updated movies collection with the new movie record appended."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new dictionary named movie with name set directly from name, price set directly from price, start_time set to datetime.strptime(start_time, '%H:%M'), end_time set to datetime.strptime(end_time, '%H:%M'), and seats set to np.zeros((n, n)). RESULT movie]
            [COMMAND Append movie to self.movies. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Finds a movie by name and attempts to book each requested seat in order, returning success, failure on the first unavailable seat, or not found if no movie matches." book_ticket]
    [INPUTS]
        <REF> name </REF>: text "The target movie name to search for in self.movies."
        <REF> seats_to_book </REF>: list "A list of seat coordinate pairs where each seat is accessed as seat[0] and seat[1]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: text "Returns 'Booking success.', 'Booking failed.', or 'Movie not found.' depending on the search and seat availability."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each movie in self.movies. RESULT current_movie]
            [COMMAND Within the loop over self.movies, check whether current_movie['name'] == name. RESULT movie_name_matches]
            [COMMAND When current_movie['name'] == name, iterate through each seat in seats_to_book. RESULT current_seat]
            [COMMAND Within the matched movie and for each current_seat, check whether current_movie['seats'][current_seat[0]][current_seat[1]] == 0. RESULT seat_is_available]
            [COMMAND When current_movie['seats'][current_seat[0]][current_seat[1]] == 0, set current_movie['seats'][current_seat[0]][current_seat[1]] = 1 before continuing to the next requested seat. RESULT seat_marked_booked]
            [COMMAND After all seats in seats_to_book have been processed for the matched movie without failure, return "Booking success." immediately. RESULT result]
            [COMMAND If the loop over self.movies completes without finding any movie whose name equals name, return "Movie not found.". RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the matched movie, any requested seat has current_movie['seats'][current_seat[0]][current_seat[1]] != 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return "Booking failed." immediately at the first unavailable seat without rolling back any earlier seat assignments already changed from 0 to 1 in this call. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parse the input time strings into datetime values and return the names of movies whose start and end times fall fully within the inclusive requested time window." available_movies]
    [INPUTS]
        <REF> start_time </REF>: string "Requested lower time bound in HH:MM format, which is parsed and then used as the inclusive earliest allowed movie start time."
        <REF> end_time </REF>: string "Requested upper time bound in HH:MM format, which is parsed and then used as the inclusive latest allowed movie end time."
        <REF> self.movies </REF>: list "Collection of movie records to scan, where each record is expected to provide name, start_time, and end_time fields."
    [END_INPUTS]

    [OUTPUTS]
        <REF> available_movies </REF>: list "List of movie names whose start_time is greater than or equal to the parsed start_time and whose end_time is less than or equal to the parsed end_time."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the input start_time string using the exact format %H:%M and overwrite start_time with the resulting datetime value. RESULT start_time]
            [COMMAND Parse the input end_time string using the exact format %H:%M and overwrite end_time with the resulting datetime value. RESULT end_time]
            [COMMAND Initialize available_movies as an empty list to collect matching movie names. RESULT available_movies]
            [COMMAND Iterate over each movie in self.movies. RESULT movie]
            [COMMAND Within the loop for each movie, check the inclusive containment condition start_time <= movie['start_time'] and movie['end_time'] <= end_time using short-circuit AND logic. RESULT movie_within_window]
            [COMMAND Within the same loop, if the condition is true, append movie['name'] to available_movies. RESULT available_movies]
            [COMMAND After all movies have been evaluated, return available_movies. RESULT available_movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The input start_time string cannot be parsed by datetime.strptime with the exact format %H:%M or contains an invalid time value.]
        [LOG "Parsing start_time fails before the method can continue, so no movie filtering result is returned."]
        [THROW ValueError "Raised when start_time is missing, malformed, or not a valid HH:MM time string for parsing."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The input end_time string cannot be parsed by datetime.strptime with the exact format %H:%M or contains an invalid time value.]
        [LOG "Parsing end_time fails before iteration over movies can complete, so no movie filtering result is returned."]
        [THROW ValueError "Raised when end_time is missing, malformed, or not a valid HH:MM time string for parsing."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
