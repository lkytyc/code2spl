# CLASS SQLGenerator

[DEFINE_WORKER: "Initializes the instance by storing the provided table name on the object." __init__]
    [INPUTS]
        <REF> table_name </REF>: any "The table name value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided table_name value directly to self.table_name. RESULT self.table_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a SQL SELECT statement from an optional field list and optional condition, defaulting to all fields when none are provided." select]
    [INPUTS]
        <REF> self </REF>: object "Object providing the table_name used in the query."
        <REF> fields </REF>: optional iterable of strings "Field names to select; if None, all fields are selected."
        <REF> condition </REF>: optional string "SQL WHERE clause condition appended when provided."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A complete SQL SELECT statement terminated with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If fields is None, set fields to "*"; otherwise join the field names with ", " into a single string. RESULT fields]
            [COMMAND Create the base SQL string as SELECT followed by fields and FROM followed by self.table_name. RESULT sql]
            [COMMAND If condition is not None, append a WHERE clause with the condition to sql. RESULT sql]
            [COMMAND Return sql with a trailing semicolon. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an SQL INSERT statement from a mapping by joining its keys as column names and its values as single-quoted values, then append a semicolon." insert]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name."
        <REF> data </REF>: mapping "Input mapping whose keys become column names and whose values become inserted values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "SQL INSERT statement terminated with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join the keys of data with comma and space separators to produce the fields string. RESULT fields]
            [COMMAND Join the values of data with comma and space separators after wrapping each value in single quotes to produce the values string. RESULT values]
            [COMMAND Format an SQL INSERT statement using self.table_name, fields, and values. RESULT sql]
            [COMMAND Append a semicolon to the SQL string and return the final statement. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an SQL UPDATE statement from the provided field-value pairs, table name, and condition, then return it with a trailing semicolon." update]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name."
        <REF> data </REF>: dict "Mapping of fields to updated values."
        <REF> condition </REF>: string "SQL WHERE clause condition."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Complete SQL UPDATE statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each field-value pair in data and format it as field = 'value'. RESULT formatted_pairs]
            [COMMAND Join the formatted pairs with ', ' to produce the SET clause. RESULT set_clause]
            [COMMAND Construct an SQL UPDATE statement using self.table_name, the SET clause, and the provided condition in the WHERE clause. RESULT sql]
            [COMMAND Append a semicolon to the SQL string and return the final query. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If data is empty, the SET clause becomes empty and the generated SQL statement is incomplete or invalid.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join no field-value pairs, producing an empty SET clause. RESULT set_clause]
            [COMMAND Construct the UPDATE statement with the empty SET clause and the provided condition. RESULT sql]
            [COMMAND Append a semicolon to the SQL string and return the final query. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a SQL DELETE statement for the current table using the given condition and return it with a trailing semicolon." delete]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name."
        <REF> condition </REF>: string "SQL condition used in the WHERE clause."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Complete DELETE SQL statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a DELETE SQL string using the current object's table name and the provided condition. RESULT sql]
            [COMMAND Return the constructed SQL string with a semicolon appended. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a filter condition for females younger than the given age and pass it to the select operation." select_female_under_age]
    [INPUTS]
        <REF> age </REF>: unknown "Age threshold used in the less-than comparison."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: unknown "The result returned by the select call."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a condition string that filters records where age is less than the given age and gender equals female. RESULT condition]
            [COMMAND Call select with the constructed condition string and return its result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an age range condition string from the given minimum and maximum ages and pass it to the select method." select_by_age_range]
    [INPUTS]
        <REF> self </REF>: object "Instance used to invoke the select operation."
        <REF> min_age </REF>: number "Inclusive lower age boundary."
        <REF> max_age </REF>: number "Inclusive upper age boundary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "The value returned by self.select using the constructed condition."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct the condition string using the inclusive BETWEEN syntax with min_age as the lower bound and max_age as the upper bound. RESULT condition]
            [COMMAND Call self.select with the constructed condition as the condition argument. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
