# CLASS BigNumCalculator

[DEFINE_WORKER: "Add two numbers represented as strings by left-padding them to equal length, summing digits from right to left with carry propagation, and returning the resulting string." add]
    [INPUTS]
        <REF> num1 </REF>: string "The first numeric string to add."
        <REF> num2 </REF>: string "The second numeric string to add."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decimal string produced by adding the two input numeric strings."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute max_length as the larger of len(num1) and len(num2). RESULT max_length]
            [COMMAND Reassign num1 to num1.zfill(max_length) so it is left-padded with zeros to length max_length. RESULT num1]
            [COMMAND Reassign num2 to num2.zfill(max_length) so it is left-padded with zeros to length max_length. RESULT num2]
            [COMMAND Initialize carry to 0. RESULT carry]
            [COMMAND Initialize result as an empty list to accumulate output digits. RESULT result]
            [COMMAND For i from max_length - 1 down to 0 inclusive with step -1, compute digit_sum as int(num1[i]) + int(num2[i]) + carry, update carry to digit_sum // 10, compute digit as digit_sum % 10, and insert str(digit) at index 0 of result. RESULT result_and_carry_after_loop]
            [COMMAND If carry > 0 after the loop, insert str(carry) at index 0 of result. RESULT result]
            [COMMAND Join all elements of result into a single string and return it. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Subtract two non-negative integer strings by normalizing their lengths, performing digit-by-digit subtraction with borrowing, removing leading zeros, and prefixing a minus sign when the result is negative." subtract]
    [INPUTS]
        <REF> num1 </REF>: string "First numeric string operand."
        <REF> num2 </REF>: string "Second numeric string operand."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decimal subtraction result as a string, with a leading minus sign if needed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compare the lengths of num1 and num2, and if num1 is shorter, swap the two values and mark the result as negative. RESULT negative]
            [COMMAND If num1 is longer than num2, keep the values as they are and mark the result as non-negative. RESULT negative]
            [COMMAND If num1 and num2 have equal length, compare them lexicographically; if num1 is smaller, swap the two values and mark the result as negative, otherwise mark the result as non-negative. RESULT negative]
            [COMMAND Compute max_length as the greater of the lengths of num1 and num2. RESULT max_length]
            [COMMAND Pad num1 on the left with zeros to reach max_length. RESULT num1]
            [COMMAND Pad num2 on the left with zeros to reach max_length. RESULT num2]
            [COMMAND Initialize borrow to 0 and result to an empty list. RESULT borrow,result]
            [COMMAND For each index i from max_length - 1 down to 0 inclusive, compute digit_diff as int(num1[i]) - int(num2[i]) - borrow. RESULT digit_diff]
            [COMMAND Within the backward digit loop, if digit_diff is negative, add 10 to digit_diff and set borrow to 1. RESULT borrow]
            [COMMAND Within the backward digit loop, otherwise set borrow to 0. RESULT borrow]
            [COMMAND Within the backward digit loop, insert the current digit_diff string at the front of result. RESULT result]
            [COMMAND After the digit loop, while result has more than one element and the first element is '0', remove the first element. RESULT result]
            [COMMAND If negative is true, insert '-' at the front of result. RESULT result]
            [COMMAND Join the elements of result into a single string and return it. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiplies two numeric strings by simulating digit-by-digit long multiplication into an integer array and then returns the array contents as a string after skipping leading zeros." multiply]
    [INPUTS]
        <REF> num1 </REF>: string "The first numeric string whose characters are expected to be individually convertible to digits."
        <REF> num2 </REF>: string "The second numeric string whose characters are expected to be individually convertible to digits."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result_string </REF>: string "The decimal string formed from the multiplication result array after removing leading zeros while preserving at least one digit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read num1 and num2, compute len1 = len(num1) and len2 = len(num2), and allocate result as a zero-filled list of length len1 + len2. RESULT len1_len2_result]
            [COMMAND For i from len1 - 1 down to 0 inclusive, iterate backward over every digit position in num1. RESULT outer_loop_i]
            [COMMAND Within the outer loop over i, for j from len2 - 1 down to 0 inclusive, iterate backward over every digit position in num2. RESULT inner_loop_j]
            [COMMAND Within the nested loops for the current i and j, convert num1[i] and num2[j] to integers and compute mul as their product. RESULT mul]
            [COMMAND Within the nested loops for the current i and j, compute p1 = i + j and p2 = i + j + 1 as the carry and digit positions in result. RESULT p1_p2]
            [COMMAND Within the nested loops for the current i and j, compute total = mul + result[p2] to accumulate the new product with any previously stored value at position p2. RESULT total]
            [COMMAND Within the nested loops for the current i and j, add total // 10 to result[p1] and assign total % 10 to result[p2]. RESULT updated_result]
            [COMMAND Initialize start to 0. RESULT start]
            [COMMAND While start < len(result) - 1 and result[start] == 0, increment start by 1 to skip leading zeros without removing the final remaining digit. RESULT trimmed_start]
            [COMMAND Return the string produced by joining the decimal string form of each element in result[start:]. RESULT result_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
