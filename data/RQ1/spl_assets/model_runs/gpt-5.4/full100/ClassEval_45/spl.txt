# CLASS ImageProcessor

[DEFINE_WORKER: "Initialize the instance by setting the image attribute to None as its default unset state." __init__]
    [INPUTS]
        <REF> self </REF>: object "The current instance being initialized."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the instance attribute image to the constant value None, overwriting any previous value and establishing the default unset state. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Open an image from the provided path and store the resulting image object in the instance attribute self.image." load_image]
    [INPUTS]
        <REF> self </REF>: object "The current instance whose image state is updated."
        <REF> image_path </REF>: path_or_file_reference "The path or file-like source identifying the image to open."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call Image.open with image_path to obtain an image object. RESULT opened_image]
            [COMMAND Assign the opened image object to the instance attribute self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Save the current image to the provided path only if `self.image` is truthy." save_image]
    [INPUTS]
        <REF> save_path </REF>: data_type "The destination path or save target passed to the image object's save method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the truthiness of `self.image`. RESULT image_exists_check]
            [COMMAND If `self.image` is truthy, call `self.image.save(save_path)` to save the image to the provided target. RESULT save_operation]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: `self.image` is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the save operation and continue without performing any action. RESULT no_op]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resize the instance image to the given width and height only when the current image attribute is truthy, and store the resized image back into the instance." resize_image]
    [INPUTS]
        <REF> width </REF>: number "Target image width passed to the resize operation."
        <REF> height </REF>: number "Target image height passed to the resize operation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the truthiness of self.image before attempting any resize operation. RESULT image_exists]
            [COMMAND If self.image is truthy, call self.image.resize with the size tuple (width, height) and assign the returned resized image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.image is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the resize operation and leave self.image unchanged. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Rotate the stored image by the given degrees and save the rotated result back only when an image is currently present and truthy." rotate_image]
    [INPUTS]
        <REF> degrees </REF>: data_type "The rotation amount passed to the image rotate operation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.image is truthy. RESULT image_exists_check]
            [COMMAND If self.image is truthy, call self.image.rotate(degrees) and assign the returned rotated image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.image is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the rotation and leave self.image unchanged. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adjust the current image's brightness by the given factor when an image is present, updating the stored image in place." adjust_brightness]
    [INPUTS]
        <REF> factor </REF>: parameter "Brightness adjustment factor passed to the image enhancer."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is truthy before performing any brightness adjustment. RESULT image_presence_check]
            [COMMAND When self.image is truthy, create a brightness enhancer from the current self.image using ImageEnhance.Brightness(self.image). RESULT enhancer]
            [COMMAND When self.image is truthy, apply enhancer.enhance(factor) and assign the enhanced image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.image is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do nothing and leave self.image unchanged. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
