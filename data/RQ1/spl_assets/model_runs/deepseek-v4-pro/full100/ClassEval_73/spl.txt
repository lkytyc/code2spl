# CLASS RPGCharacter

[DEFINE_WORKER: "Initializes a new instance by assigning the provided name, hp, attack_power, defense, and level values to the corresponding instance attributes and setting exp to 0." __init__]
    [INPUTS]
        <REF> name </REF>: string "Character name assigned to self.name."
        <REF> hp </REF>: number "Health points assigned to self.hp."
        <REF> attack_power </REF>: number "Attack power assigned to self.attack_power."
        <REF> defense </REF>: number "Defense value assigned to self.defense."
        <REF> level </REF>: number "Character level assigned to self.level; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter name to the instance attribute self.name. RESULT self.name]
            [COMMAND Assign the input parameter hp to the instance attribute self.hp. RESULT self.hp]
            [COMMAND Assign the input parameter attack_power to the instance attribute self.attack_power. RESULT self.attack_power]
            [COMMAND Assign the input parameter defense to the instance attribute self.defense. RESULT self.defense]
            [COMMAND Assign the input parameter level to the instance attribute self.level. RESULT self.level]
            [COMMAND Set the instance attribute self.exp to the constant 0. RESULT self.exp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates damage as the attacker's attack power minus the defender's defense with a minimum of 1, then reduces the defender's hit points by that damage." attack]
    [INPUTS]
        <REF> self </REF>: object "The attacking character with an attack_power attribute."
        <REF> other_character </REF>: object "The defending character with defense and hp attributes."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute damage as the maximum of self.attack_power minus other_character.defense and 1, ensuring damage is at least 1. RESULT damage]
            [COMMAND Subtract the computed damage from other_character.hp. RESULT other_character.hp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Heal the target by adding 10 HP, clamp the HP to a maximum of 100, and return the resulting HP." heal]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "The resulting HP value after adding 10 and clamping to a maximum of 100."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.hp by 10. RESULT self.hp]
            [COMMAND If self.hp is strictly greater than 100, set self.hp to 100. RESULT self.hp]
            [COMMAND Return the current value of self.hp. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes an experience amount by repeatedly applying exactly enough experience to trigger level-ups until the remaining amount is smaller than the next level threshold, then adds that leftover to current experience." gain_exp]
    [INPUTS]
        <REF> amount </REF>: integer "Experience amount to distribute; it is reduced by level-up costs and any remaining part is added to current experience."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Accept the incoming amount as the remaining experience to distribute. RESULT amount]
            [COMMAND While amount is not equal to 0, execute the appropriate branch below, then re-evaluate the loop condition. RESULT loop_condition]
            [COMMAND When the loop exits because amount equals 0, finish the method. RESULT completion_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, when self.exp + amount is greater than or equal to self.level * 100.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Reduce amount by the remaining experience needed to reach the current level threshold: self.level * 100 - self.exp. RESULT amount]
            [COMMAND Call self.level_up() to apply the level-up and update current level/experience. RESULT level_up_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, when self.exp + amount is less than self.level * 100.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Add the entire remaining amount to self.exp. RESULT self.exp]
            [COMMAND Set amount to 0 so the while loop exits after this iteration. RESULT amount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Levels up the character if its level is below 100 by incrementing level, resetting exp, and increasing hp, attack_power, and defense, then returns the current attributes." level_up]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the current level, hp, attack_power, and defense after attempting a level-up."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.level is less than 100 and store the boolean result as can_level_up. RESULT can_level_up]
            [COMMAND If can_level_up is true, increment self.level by 1. RESULT self.level]
            [COMMAND If can_level_up is true, set self.exp to 0. RESULT self.exp]
            [COMMAND If can_level_up is true, increase self.hp by 20. RESULT self.hp]
            [COMMAND If can_level_up is true, increase self.attack_power by 5. RESULT self.attack_power]
            [COMMAND If can_level_up is true, increase self.defense by 5. RESULT self.defense]
            [COMMAND Return a tuple containing self.level, self.hp, self.attack_power, and self.defense in that order. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the instance is alive by evaluating whether its hp attribute is greater than 0." is_alive]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> alive_status </REF>: boolean "True when hp is greater than 0; False when hp is 0 or negative."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance attribute hp and return the boolean result of the comparison hp > 0. RESULT alive_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
