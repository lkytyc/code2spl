# CLASS ExpressionCalculator

[DEFINE_WORKER: "Initializes the expression parser state by setting up an empty postfix stack and a constant operator precedence table." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the instance attribute postfix_stack to a new empty deque object. RESULT self.postfix_stack]
            [COMMAND Assign the instance attribute operat_priority to the constant list [0, 3, 2, 1, -1, 1, 0, 2]. RESULT self.operat_priority]

[DEFINE_WORKER: "Evaluates a postfix expression by preparing its transformed input, processing tokens with a stack-based RPN evaluator, and returning the floating-point product of the remaining stack values." calculate]
    [INPUTS]
        <REF> expression </REF>: string "The expression to be transformed and prepared before postfix evaluation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> calculated_result </REF>: float "The floating-point result of multiplying the operands remaining on the result stack after postfix evaluation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Transform the input expression by calling self.transform(expression), then pass the transformed result to self.prepare. RESULT prepared_expression]
            [COMMAND Initialize result_stack as an empty deque. RESULT result_stack]
            [COMMAND Reverse self.postfix_stack in place. RESULT postfix_stack]
            [COMMAND While self.postfix_stack is not empty, pop the next token from self.postfix_stack into current_op. RESULT current_op]
            [COMMAND Within the while loop, if self.is_operator(current_op) is false, replace every occurrence of '~' in current_op with '-' and append the normalized current_op to result_stack. RESULT result_stack]
            [COMMAND Within the while loop, in the else branch when self.is_operator(current_op) is true, pop the top value from result_stack into second_value, then pop the next top value from result_stack into first_value. RESULT second_value_first_value]
            [COMMAND Still within the operator branch, replace every occurrence of '~' in first_value and in second_value with '-'. RESULT first_value_second_value]
            [COMMAND Still within the operator branch, compute temp_result by calling self._calculate(first_value, second_value, current_op). RESULT temp_result]
            [COMMAND Still within the operator branch, append str(temp_result) to result_stack. RESULT result_stack]
            [COMMAND After the while loop completes, join all items in result_stack with '*' and evaluate the resulting multiplication expression, returning its float value. RESULT calculated_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an infix expression into postfix notation by scanning characters, buffering operands, and using a comma-sentinel operator stack to handle precedence, associativity, and parentheses." prepare]
    [INPUTS]
        <REF> expression </REF>: str "Infix expression to convert into postfix notation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize op_stack as a deque with the sentinel ',' so the stack is never empty and the lowest stopping condition is available. RESULT op_stack]
            [COMMAND Convert expression into arr by applying list(expression), producing a single-character token list. RESULT arr]
            [COMMAND Set current_index to 0 and count to 0, representing the start of the current operand and its length. RESULT current_index_count]
            [COMMAND For each current_op at index i from enumerate(arr), start the left-to-right scan loop. RESULT scan_loop]
            [COMMAND Within the scan loop, if self.is_operator(current_op) is true, take the operator branch; otherwise take the operand branch. RESULT operator_branch_check]
            [COMMAND Within the operator branch, if count > 0, append the joined string from arr[current_index: current_index + count] to self.postfix_stack to flush the pending operand. RESULT flush_pending_operand]
            [COMMAND Within the operator branch, set peek_op to op_stack[-1] after the pending operand flush. RESULT peek_op]
            [COMMAND Within the operator branch, if current_op equals ')', while op_stack[-1] is not '(' append str(op_stack.pop()) to self.postfix_stack, then pop the matching '(' from op_stack without appending it. RESULT right_parenthesis_handling]
            [COMMAND Within the operator branch, otherwise while current_op is not '(' and peek_op is not ',' and self.compare(current_op, peek_op) is true, append str(op_stack.pop()) to self.postfix_stack and update peek_op to op_stack[-1]; after the loop append current_op to op_stack. RESULT operator_handling]
            [COMMAND Within the operator branch, reset count to 0 and set current_index to i + 1 after processing the operator. RESULT reset_after_operator]
            [COMMAND Within the scan loop, if self.is_operator(current_op) is false, increment count by 1 to buffer the current operand character. RESULT buffer_operand_character]
            [COMMAND After the scan loop, if count > 1 or (count == 1 and not self.is_operator(arr[current_index])) with short-circuit evaluation, append the joined string from arr[current_index: current_index + count] to self.postfix_stack as the final pending operand. RESULT final_operand_flush]
            [COMMAND After the final operand flush check, while op_stack[-1] is not equal to ',', append str(op_stack.pop()) to self.postfix_stack to drain remaining operators. RESULT drain_operator_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if c is one of '+', '-', '*', '/', '(', ')', or '%'; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True if c belongs to the set {'+', '-', '*', '/', '(', ')', '%'}, and return False otherwise. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compares operator precedence between cur and peek after normalizing '%' to '/', returning true when peek's priority is greater than or equal to cur's priority." compare]
    [INPUTS]
        <REF> cur </REF>: string "Current operator character whose priority is compared."
        <REF> peek </REF>: string "Next operator character whose priority is compared against cur."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the priority of peek is greater than or equal to the priority of cur after normalizing '%' to '/'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If cur is exactly '%', assign '/' to cur. RESULT cur]
            [COMMAND If peek is exactly '%', assign '/' to peek. RESULT peek]
            [COMMAND Compute peek_priority as self.operat_priority[ord(peek) - 40]. RESULT peek_priority]
            [COMMAND Compute cur_priority as self.operat_priority[ord(cur) - 40]. RESULT cur_priority]
            [COMMAND Return the boolean result of peek_priority >= cur_priority. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Selects one of five Decimal arithmetic operations based on current_op and returns the Decimal result, raising ValueError for an unsupported operator." _calculate]
    [INPUTS]
        <REF> first_value </REF>: Decimal|number|string "Left operand converted to Decimal."
        <REF> second_value </REF>: Decimal|number|string "Right operand converted to Decimal."
        <REF> current_op </REF>: string "Arithmetic operator; supported values are '+', '-', '*', '/', and '%'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decimal_result </REF>: Decimal "The Decimal result of the selected arithmetic operation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate current_op against the supported operator branches in order, selecting the first match. RESULT selected_branch]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: current_op equals '+'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert first_value to Decimal and store it as first_decimal. RESULT first_decimal]
            [COMMAND Convert second_value to Decimal and store it as second_decimal. RESULT second_decimal]
            [COMMAND Return the Decimal result of first_decimal plus second_decimal. RESULT decimal_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_op equals '-'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert first_value to Decimal and store it as first_decimal. RESULT first_decimal]
            [COMMAND Convert second_value to Decimal and store it as second_decimal. RESULT second_decimal]
            [COMMAND Return the Decimal result of first_decimal minus second_decimal. RESULT decimal_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_op equals '*'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert first_value to Decimal and store it as first_decimal. RESULT first_decimal]
            [COMMAND Convert second_value to Decimal and store it as second_decimal. RESULT second_decimal]
            [COMMAND Return the Decimal result of first_decimal multiplied by second_decimal. RESULT decimal_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_op equals '/'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert first_value to Decimal and store it as first_decimal. RESULT first_decimal]
            [COMMAND Convert second_value to Decimal and store it as second_decimal. RESULT second_decimal]
            [COMMAND Return the Decimal result of first_decimal divided by second_decimal. RESULT decimal_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_op equals '%'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert first_value to Decimal and store it as first_decimal. RESULT first_decimal]
            [COMMAND Convert second_value to Decimal and store it as second_decimal. RESULT second_decimal]
            [COMMAND Return the Decimal result of first_decimal modulo second_decimal. RESULT decimal_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: current_op is not one of '+', '-', '*', '/', or '%'.]
        [LOG "Unexpected operator: {current_op}"]
        [THROW ValueError "Raised when current_op is not one of the supported arithmetic operators."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalizes an arithmetic expression by removing whitespace and a trailing equals sign, rewriting unary minus signs as '~', and converting a leading '~(' to '0-(' before returning the transformed string." transform]
    [INPUTS]
        <REF> expression </REF>: str "Input arithmetic expression string that may contain whitespace, a trailing equals sign, operators, parentheses, and minus signs."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Normalized expression string after whitespace removal, trailing '=' removal, unary minus rewriting to '~', and leading '~(' to '0-(' conversion."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove all whitespace characters from expression using re.sub(r"\s+", "", expression) and assign the result back to expression; if expression is not a string or bytes-like object compatible with re.sub, Python raises TypeError. RESULT expression]
            [COMMAND Remove exactly one trailing '=' character, if present, from expression using re.sub(r"=$", "", expression) and assign the result back to expression. RESULT expression]
            [COMMAND Convert the updated expression string into a list of its individual characters and assign it to arr. RESULT arr]
            [COMMAND Within the for loop over each index i and character c in enumerate(arr), if c equals '-' and i equals 0, set arr[i] to '~'. RESULT arr]
            [COMMAND Within the same for loop over each index i and character c in enumerate(arr), if c equals '-' and i is not 0, read prev_c from the already-modified arr at index i-1; if prev_c is one of '+', '-', '*', '/', '(', 'E', or 'e', set arr[i] to '~'. RESULT arr]
            [COMMAND After the loop, evaluate the final short-circuit condition from left to right: if arr is empty, accessing arr[0] raises IndexError; otherwise, if arr[0] is '~' and len(arr) is greater than 1 and arr[1] is '(', then set arr[0] to '-' and return the string "0" concatenated with the string formed by joining arr with an empty separator. RESULT return_value]
            [COMMAND Otherwise, return the string formed by joining arr with an empty separator. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
