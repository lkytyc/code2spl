# CLASS JSONProcessor

[DEFINE_WORKER: "Reads a JSON file from file_path and returns the parsed JSON object; returns 0 if the file does not exist and -1 if an exception occurs while opening or parsing the file." read_json]
    [INPUTS]
        <REF> file_path </REF>: string "Path to the JSON file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object | int "Parsed JSON object when the file exists and is valid; integer 0 when file_path does not exist; integer -1 when an exception occurs during opening or parsing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether file_path exists using os.path.exists(file_path). RESULT file_exists]
            [COMMAND If file_path exists, open file_path in read mode ('r') as file. RESULT file]
            [COMMAND Within the open file context, load JSON content from file using json.load(file) into data. RESULT data]
            [COMMAND Return data. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: file_path does not exist according to os.path.exists(file_path).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return integer 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: An exception of any type is raised while opening file_path or parsing its JSON content with json.load.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return integer -1. RESULT return_-1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the given data as a JSON document to the specified file path and returns 1 on success or -1 if any error occurs." write_json]
    [INPUTS]
        <REF> data </REF>: object "Data to serialize and write as JSON"
        <REF> file_path </REF>: string "Destination file path for the JSON output"
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Returns 1 on success and -1 if any exception occurs"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at file_path in write mode ('w') and obtain the file handle RESULT file]
            [COMMAND Serialize data to JSON and write it to the file using json.dump(data, file) RESULT dump_result]
            [COMMAND If the open and write operations complete without exception, return 1 RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs while opening the file or writing JSON data]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception and return -1 RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads JSON data from a file, removes a specified key if present, writes the updated data back to the same file, and returns an integer status indicating success or failure." process_json]
    [INPUTS]
        <REF> file_path </REF>: string "Path to the JSON file to read and, on successful removal, write."
        <REF> remove_key </REF>: string "Key to remove from the JSON data if it is present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Returns 1 if the key was removed and the file was written; returns 0 if the read result was 0 or -1 or if the key was not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.read_json(file_path) and assign the returned value to data. RESULT data]
            [COMMAND If data equals 0 or data equals -1, return 0 immediately. RESULT return_0]
            [COMMAND If data is not 0 or -1 and remove_key is present in data, delete data[remove_key]. RESULT data]
            [COMMAND Within the same remove_key-present branch, call self.write_json(data, file_path) to write the updated data back to the file. RESULT write_result]
            [COMMAND Within the same remove_key-present branch, return 1. RESULT return_1]
            [COMMAND If data is not 0 or -1 and remove_key is not present in data, return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
