# CLASS PDFHandler

[DEFINE_WORKER: "Stores the provided file paths and initializes an ordered list of PDF reader objects by applying PyPDF2.PdfReader to each path in the input iterable." __init__]
    [INPUTS]
        <REF> filepaths </REF>: Iterable[str] "An iterable of file paths that will be stored directly and used to create one PyPDF2.PdfReader object per path."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the parameter filepaths directly to self.filepaths without copying or transforming it. RESULT self.filepaths]
            [COMMAND Within the list comprehension, for each element fp in the filepaths iterable, create PyPDF2.PdfReader(fp), collect the resulting reader objects in the same order as the input iterable, and assign the new list to self.readers. If filepaths is an empty iterable, self.readers becomes an empty list; if filepaths is a single string, it is iterated character-by-character; no None handling is performed. RESULT self.readers]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Merges all pages from each PDF reader in self.readers into a single PDF written to output_filepath and returns a confirmation message." merge_pdfs]
    [INPUTS]
        <REF> output_filepath </REF>: string "File path where the merged PDF will be written."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "Confirmation message containing the output file path."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new PyPDF2.PdfWriter instance with no constructor arguments and assign it to pdf_writer. RESULT pdf_writer]
            [COMMAND For each reader in self.readers, iterate over the readers in their existing order. RESULT reader]
            [COMMAND Within the outer loop over reader, for each page_num from 0 to len(reader.pages) - 1 inclusive, retrieve page = reader.pages[page_num] and append it to pdf_writer using pdf_writer.add_page(page). RESULT pdf_writer]
            [COMMAND Using a with block, open output_filepath in binary write mode as out and write the contents of pdf_writer to out. RESULT out]
            [COMMAND Return the string f"Merged PDFs saved at {output_filepath}". RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts text from every page of every reader in self.readers and returns the collected page texts in deterministic order." extract_text_from_pdfs]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> pdf_texts </REF>: List[str] "A list of extracted page texts, ordered by reader order and ascending page index."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named pdf_texts. RESULT pdf_texts]
            [COMMAND For each reader in self.readers, in the order they appear, execute the following page loop. If self.readers is empty, skip the loop body. RESULT reader]
            [COMMAND Within the outer loop over reader, for each page_num from 0 to len(reader.pages)-1 inclusive, in ascending order, set page = reader.pages[page_num]. If reader.pages is empty, skip the inner loop body. RESULT page]
            [COMMAND Within the same inner loop, append page.extract_text() to pdf_texts. RESULT pdf_texts]
            [COMMAND After all readers and pages have been processed, return pdf_texts. RESULT pdf_texts]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
