# CLASS Warehouse

[DEFINE_WORKER: "Initializes a new instance with an empty inventory dictionary for mapping product IDs to products and an empty orders dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance attribute inventory to a new empty dictionary intended to map Product ID to Product. RESULT self.inventory]
            [COMMAND Set the instance attribute orders to a new empty dictionary. RESULT self.orders]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the inventory by creating a new entry for a missing product or incrementing the existing product's quantity." add_product]
    [INPUTS]
        <REF> product_id </REF>: string "Identifier of the product to add or update."
        <REF> name </REF>: string "Product name stored when creating a new inventory entry."
        <REF> quantity </REF>: number "Quantity to add to the product's inventory."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is present in self.inventory. RESULT product_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: product_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.inventory[product_id] to a dictionary containing name and quantity. RESULT new_inventory_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: product_exists is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increment self.inventory[product_id]['quantity'] by quantity. RESULT updated_inventory_quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the quantity of an existing product in the inventory by adding the given amount, doing nothing if the product is not present." update_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: any "Identifier used as a key to look up the product in the inventory."
        <REF> quantity </REF>: number "Amount to add to the product's current quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT product_exists]
            [COMMAND If product_exists is true, read self.inventory[product_id]['quantity'], add quantity to it, and assign the updated value back to self.inventory[product_id]['quantity']. RESULT updated_quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the quantity for product_id from self.inventory if it exists, otherwise return False." get_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: any "Identifier used as the key to look up the product in self.inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: number|boolean "The quantity from self.inventory[product_id]['quantity'] when found, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is present as a key in self.inventory. RESULT is_present]
            [COMMAND If is_present is true, retrieve self.inventory[product_id]['quantity'] and return it as return_value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If is_present is false, meaning product_id is not a key in self.inventory.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False as return_value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates an order by validating that the available product quantity is greater than or equal to the requested quantity, decrementing inventory, recording the order as Shipped, and returning False when stock is insufficient." create_order]
    [INPUTS]
        <REF> order_id </REF>: any "Identifier used as the key for the created order record"
        <REF> product_id </REF>: any "Identifier of the product whose quantity is checked and updated"
        <REF> quantity </REF>: integer "Requested product quantity to order"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean_or_none "False when the available quantity is less than the requested quantity; otherwise None after the order is successfully recorded with status Shipped"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_product_quantity(product_id) and store the returned amount as available_quantity RESULT available_quantity]
            [COMMAND If available_quantity >= quantity, proceed with the sufficient-stock branch; otherwise, trigger the insufficient-stock alternative flow RESULT stock_check]
            [COMMAND Within the sufficient-stock branch, call self.update_product_quantity(product_id, -quantity) to decrement product inventory by the requested quantity RESULT inventory_update_result]
            [COMMAND Within the sufficient-stock branch, assign self.orders[order_id] to a dictionary containing product_id, quantity, and status 'Shipped' RESULT order_record]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The available product quantity is less than the requested quantity (available_quantity < quantity)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without updating product inventory or recording the order RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the status of an existing order identified by order_id in self.orders; returns False if the order_id is not present." change_order_status]
    [INPUTS]
        <REF> order_id </REF>: any "Identifier used as a key to look up the order in self.orders."
        <REF> status </REF>: any "New status value assigned to self.orders[order_id]['status'] when the order exists."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean_or_none "Returns False when order_id is missing; otherwise returns None implicitly after a successful status update."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT order_exists]
            [COMMAND If order_exists is true, set self.orders[order_id]['status'] to status and complete without an explicit return, yielding an implicit None return. RESULT success_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If order_id is not present as a key in self.orders.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, ending the method without updating any order. RESULT false_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the status of the order if the order ID exists; otherwise return False." track_order]
    [INPUTS]
        <REF> order_id </REF>: Any "The order identifier to look up in self.orders."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: Any "The status value from self.orders[order_id] if the order exists; otherwise boolean False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists in self.orders. RESULT order_exists]
            [COMMAND When order_exists is true, return self.orders[order_id]['status']. RESULT status_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT not_found_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
