# CLASS NumberConverter

[DEFINE_WORKER: "Converts a decimal integer to a binary string using Python's bin() function and removes the '0b' prefix with slice [2:]." decimal_to_binary]
    [INPUTS]
        <REF> decimal_num </REF>: int "The decimal integer to convert. Python's bin() accepts integer-like values; negative inputs produce '-0b...' before slicing."
    [END_INPUTS]

    [OUTPUTS]
        <REF> binary_num </REF>: string "The resulting string after slicing off the first two characters from bin(decimal_num); for zero this is '0', for positive integers this is the binary digit string without leading zeros, and for negative integers this may begin with 'b' instead of a negative sign due to the fixed [2:] slice."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call bin(decimal_num) to convert decimal_num to its binary string representation, then apply slice [2:] to remove the first two characters and assign the result to binary_num. RESULT binary_num]
            [COMMAND Return binary_num. RESULT binary_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a binary string, bytes, or bytearray input into its decimal integer equivalent by parsing it with base 2." binary_to_decimal]
    [INPUTS]
        <REF> binary_num </REF>: str|bytes|bytearray "The binary input to convert; it must contain valid base-2 digits, optionally with surrounding whitespace, a leading sign, or a 0b/0B prefix."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decimal_num </REF>: int "The decimal integer parsed from binary_num."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse binary_num as a base-2 integer using int with radix 2 and assign the returned integer to decimal_num. RESULT decimal_num]
            [COMMAND Return decimal_num as the method result. RESULT decimal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: binary_num is not a string, bytes, or bytearray.]
        [LOG "int(binary_num, 2) raises TypeError; decimal_num is not assigned and retains any previous value."]
        [THROW TypeError "binary_num must be a string, bytes, or bytearray for base-2 conversion."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: binary_num contains invalid base-2 content, such as an empty string or digits other than 0 and 1 after optional whitespace, sign, or 0b/0B prefix.]
        [LOG "int(binary_num, 2) raises ValueError; decimal_num is not assigned and retains any previous value."]
        [THROW ValueError "binary_num must contain valid base-2 digits for conversion."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an integer decimal number to an octal string representation after removing Python's '0o' prefix." decimal_to_octal]
    [INPUTS]
        <REF> decimal_num </REF>: int "The decimal integer to convert to octal."
    [END_INPUTS]

    [OUTPUTS]
        <REF> octal_num </REF>: str "The octal string representation of decimal_num with the '0o' prefix removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert decimal_num to an octal string using oct(decimal_num), then slice the string from index 2 using [2:] to remove the first two characters and assign the result to octal_num; for non-negative decimal_num this strips the '0o' prefix, while for negative decimal_num it strips '-0' and leaves strings like 'o10'. RESULT octal_num]
            [COMMAND Return octal_num as the converted octal string. RESULT octal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an octal string input into its decimal integer equivalent using base 8." octal_to_decimal]
    [INPUTS]
        <REF> octal_num </REF>: str "The octal string representation to convert to decimal."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decimal_num </REF>: int "The decimal integer equivalent of the input octal string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert octal_num from base 8 to a decimal integer and assign the result to decimal_num. RESULT decimal_num]
            [COMMAND Return decimal_num as the method result. RESULT decimal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a decimal number to a hexadecimal string by applying hex() and slicing off the first two characters, which removes the 0x prefix for non-negative inputs." decimal_to_hex]
    [INPUTS]
        <REF> decimal_num </REF>: int "The decimal integer value to convert to hexadecimal."
    [END_INPUTS]

    [OUTPUTS]
        <REF> hex_num </REF>: str "The resulting hexadecimal string after slicing the first two characters from hex(decimal_num)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute hex(decimal_num), remove the first two characters from that hexadecimal string using [2:], and assign the result to hex_num. RESULT hex_num]
            [COMMAND Return the value of hex_num. RESULT hex_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a hexadecimal string, bytes, or bytearray value to its decimal integer equivalent using base 16." hex_to_decimal]
    [INPUTS]
        <REF> hex_num </REF>: string/bytes/bytearray "A hexadecimal value to convert to a decimal integer."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decimal_num </REF>: integer "The decimal integer equivalent of the input hexadecimal value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert hex_num from base 16 to an integer and assign the result to decimal_num. RESULT decimal_num]
            [COMMAND Return the value of decimal_num. RESULT decimal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
