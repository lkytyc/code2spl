    [END_INPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]

[DEFINE_WORKER: "Normalizes an input segment using fix_path and appends the normalized result to the instance's segments list." add]
    [INPUTS]
        <REF> segment </REF>: object "Input value to normalize and append to self.segments."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.fix_path with segment as the argument to normalize or fix the input segment. RESULT normalized_segment]
            [COMMAND Append normalized_segment to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses a path string by optionally recording whether it ends with '/', normalizing it via fix_path, splitting it on '/', percent-decoding each segment using the given charset, and appending the decoded segments to self.segments." parse]
    [INPUTS]
        <REF> path </REF>: str "The path string to parse; if falsy, parsing is skipped."
        <REF> charset </REF>: str "The character encoding used to percent-decode each path segment."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If the initial path is truthy and path.endswith('/') is true, set self.with_end_tag to True. RESULT self.with_end_tag]
            [COMMAND If the initial path is truthy, reassign path to the result of self.fix_path(path). RESULT path]
            [COMMAND After reassignment, if path is truthy, split path on '/' using path.split('/') and assign the resulting list to split. RESULT split]
            [COMMAND Within the truthy reassigned path block, for each seg in split in order, decode seg using urllib.parse.unquote(seg, encoding=charset) and assign the decoded value to decoded_seg. RESULT decoded_seg]
            [COMMAND Within the same for loop iteration, append decoded_seg to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return an empty string when the input path is falsy, otherwise return the path with leading and trailing slash characters removed." fix_path]
    [INPUTS]
        <REF> path </REF>: str "The input path string to normalize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: str "An empty string if the input is falsy; otherwise the input path with leading and trailing slash characters removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the input parameter `path` to determine whether it is truthy or falsy. RESULT path_evaluation]
            [COMMAND When `path` is truthy, assign to `segment_str` the result of `path.strip('/')`, which removes all leading and trailing slash characters from `path` while preserving internal slashes and treating the argument as a character set. RESULT segment_str]
            [COMMAND Return `segment_str`. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If `path` is falsy, such as an empty string or None.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
