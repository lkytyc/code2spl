[DEFINE_WORKER: "Filters self.events to return only events whose event['date'] calendar date matches the provided date's calendar date." get_events]
    [INPUTS]
        <REF> date </REF>: datetime/date "The target date whose calendar date is compared against each event's date."
    [END_INPUTS]

    [OUTPUTS]
        <REF> events_on_date </REF>: list "A list of events from self.events whose calendar date matches the target date."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize events_on_date as an empty list. RESULT events_on_date]
            [COMMAND For each event in self.events, if event['date'].date() equals date.date(), append event to events_on_date. RESULT events_on_date]
            [COMMAND Return events_on_date. RESULT events_on_date]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True if the requested interval from start_time to end_time does not overlap any existing event in self.events, and False otherwise." is_available]
    [INPUTS]
        <REF> start_time </REF>: datetime "Start time of the interval to check for availability."
        <REF> end_time </REF>: datetime "End time of the interval to check for availability."
    [END_INPUTS]

    [OUTPUTS]
        <REF> availability_result </REF>: boolean "True when the requested interval is available and does not overlap any existing event; False when an overlap is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each event in self.events, compare start_time to event['end_time'] using strict less-than. RESULT starts_before_event_ends]
            [COMMAND Within the for loop over self.events, only when starts_before_event_ends is true, compare end_time to event['start_time'] using strict greater-than. RESULT ends_after_event_starts]
            [COMMAND Within the for loop over self.events, set overlap to true only when starts_before_event_ends is true and ends_after_event_starts is true; otherwise set overlap to false. RESULT overlap]
            [COMMAND After completing the for loop without an overlap, return True. RESULT availability_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over self.events, overlap is true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT availability_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Get all available 60-minute time slots for the given date by checking availability from midnight until 23:59 and returning slots where self.is_available is true." get_available_slots]
    [INPUTS]
        <REF> date </REF>: date "The date for which available time slots are computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> available_slots </REF>: list<tuple<datetime, datetime>> "List of available 60-minute time slots represented as (start_time, slot_end_time) tuples."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize available_slots to an empty list. RESULT available_slots]
            [COMMAND Set start_time to datetime(date.year, date.month, date.day, 0, 0), representing midnight at hour 0, minute 0, second 0, and microsecond 0. RESULT start_time]
            [COMMAND Set end_time to datetime(date.year, date.month, date.day, 23, 59), representing 23:59:00, not the final second of the day. RESULT end_time]
            [COMMAND Repeat the following nested loop-body steps while start_time is strictly less than end_time. RESULT result]
            [COMMAND Within the while loop, set slot_end_time to start_time plus timedelta(minutes=60); do not clamp slot_end_time to end_time, so the final generated slot may extend beyond end_time. RESULT slot_end_time]
            [COMMAND Within the while loop, evaluate the alternative flow condition for self.is_available(start_time, slot_end_time); execute its steps only if the condition is true. RESULT result]
            [COMMAND Within the while loop, unconditionally increment start_time by timedelta(minutes=60). RESULT start_time]
            [COMMAND After the while loop terminates when start_time is no longer strictly less than end_time, return available_slots. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_available(start_time, slot_end_time) returns true within the while loop.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the while loop, append the tuple (start_time, slot_end_time) to available_slots. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect up to num_events events from self.events whose start_time is at or after the current datetime, preserving source order and stopping early when the exact target count is reached." get_upcoming_events]
    [INPUTS]
        <REF> num_events </REF>: int "Target number of upcoming events to collect; the loop breaks only when len(upcoming_events) exactly equals num_events."
    [END_INPUTS]

    [OUTPUTS]
        <REF> upcoming_events </REF>: list "List of event dictionaries from self.events with start_time >= now, collected in source order up to the target count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set now to the current datetime returned by datetime.now(). RESULT now]
            [COMMAND Initialize upcoming_events as an empty list. RESULT upcoming_events]
            [COMMAND Begin a for loop over each event in self.events in source order. RESULT event_loop]
            [COMMAND Within the for loop over event in self.events, if event['start_time'] >= now (inclusive boundary, including equality with now), append event to upcoming_events. RESULT upcoming_events]
            [COMMAND Within the same for loop iteration, after the previous if statement is evaluated regardless of whether it appended an event, if len(upcoming_events) == num_events exactly, break out of the for loop immediately. RESULT loop_break]
            [COMMAND After the for loop ends normally or is exited by break, return upcoming_events. RESULT return_upcoming_events]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
