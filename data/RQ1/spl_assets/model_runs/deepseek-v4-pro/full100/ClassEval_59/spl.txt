# CLASS MovieBookingSystem

[DEFINE_WORKER: "Initializes the object by assigning the movies attribute to an empty list." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.movies to a new empty list with zero elements. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a movie record with the supplied name and price, parsed start and end times, and an n by n zero seat matrix to the instance's movies list." add_movie]
    [INPUTS]
        <REF> name </REF>: any "The movie name to store directly in the movie record; no type or range constraints are applied."
        <REF> price </REF>: any "The price to store directly in the movie record; no type or range constraints are applied."
        <REF> start_time </REF>: string "Start time string in '%H:%M' format; must be compatible with datetime.strptime or ValueError is raised."
        <REF> end_time </REF>: string "End time string in '%H:%M' format; must be compatible with datetime.strptime or ValueError is raised."
        <REF> n </REF>: integer "Seat matrix dimension; must be a non-negative integer for np.zeros((n, n)). n=0 produces an empty array of shape (0, 0); n<0 raises ValueError; a non-integer may raise TypeError."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse start_time into a datetime object using the exact format '%H:%M', where the date defaults to 1900-01-01 because no date is provided. RESULT parsed_start_time]
            [COMMAND Parse end_time into a datetime object using the exact format '%H:%M', where the date defaults to 1900-01-01 because no date is provided. RESULT parsed_end_time]
            [COMMAND Create an n by n two-dimensional NumPy array of zeros with default dtype float64. RESULT seats_array]
            [COMMAND Construct a movie dictionary with keys 'name', 'price', 'start_time', 'end_time', and 'seats', assigning name directly, price directly, parsed_start_time to 'start_time', parsed_end_time to 'end_time', and seats_array to 'seats'. RESULT movie]
            [COMMAND Append the constructed movie dictionary to the instance attribute self.movies. RESULT updated_movies_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Books the requested seats for the first movie matching the given name by marking available seats as occupied and returning a success, failure, or movie-not-found message." book_ticket]
    [INPUTS]
        <REF> name </REF>: string "The movie name to search for."
        <REF> seats_to_book </REF>: list "A list of seat coordinates, where each seat is accessed by seat[0] and seat[1] within movie['seats']."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Returns 'Booking success.' when all requested seats are available, 'Booking failed.' when any requested seat is already occupied, or 'Movie not found.' when no movie matches the requested name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Start iterating over each movie in self.movies. RESULT current_movie]
            [COMMAND Within the outer for loop over self.movies, compare current_movie['name'] with name; if it does not match, continue to the next movie. RESULT matched_movie]
            [COMMAND For the first matching movie, start iterating over each seat in seats_to_book. RESULT current_seat]
            [COMMAND Within the inner for loop over seats_to_book, check current movie['seats'][current_seat[0]][current_seat[1]]; if the value equals 0, set it to 1. RESULT booking_update]
            [COMMAND After all requested seats have been successfully processed for the first matching movie, return the string 'Booking success.'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the inner for loop over seats_to_book, if the selected seat value is not 0, meaning the seat is already occupied.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the string 'Booking failed.' immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After completing the outer for loop over self.movies without finding any movie whose name matches name.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the string 'Movie not found.'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the names of movies whose entire showing interval fits within the inclusive time window after parsing the provided time strings as HH:MM." available_movies]
    [INPUTS]
        <REF> start_time </REF>: string "Start time string in HH:MM 24-hour format."
        <REF> end_time </REF>: string "End time string in HH:MM 24-hour format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> available_movies </REF>: list "List of movie names whose showing intervals fall completely within the inclusive window."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse start_time from a string into a datetime object using the format '%H:%M'. RESULT start_time]
            [COMMAND Parse end_time from a string into a datetime object using the format '%H:%M'. RESULT end_time]
            [COMMAND Initialize available_movies as an empty list. RESULT available_movies]
            [COMMAND For each movie in self.movies, in original order: if start_time <= movie['start_time'] is true, then check movie['end_time'] <= end_time; if both inclusive comparisons are true, append movie['name'] to available_movies. RESULT available_movies]
            [COMMAND Return available_movies. RESULT available_movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
