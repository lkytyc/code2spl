# CLASS DataStatistics4

[DEFINE_WORKER: "Compute the Pearson correlation coefficient between two numeric lists and return 0 when the denominator is zero." correlation_coefficient]
    [INPUTS]
        <REF> data1 </REF>: list[float] "First list of numeric values; its length defines the number of paired elements used."
        <REF> data2 </REF>: list[float] "Second list of numeric values; it should have the same length as data1 for correct paired computation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float "Pearson correlation coefficient between data1 and data2, or 0 if the denominator equals 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of data1. RESULT n]
            [COMMAND Compute mean1 as sum(data1) divided by n. RESULT mean1]
            [COMMAND Compute mean2 as sum(data2) divided by n. RESULT mean2]
            [COMMAND Compute numerator as the sum over each i from 0 to n-1 inclusive of (data1[i] - mean1) multiplied by (data2[i] - mean2). RESULT numerator]
            [COMMAND Compute denominator as math.sqrt(sum over each i from 0 to n-1 inclusive of (data1[i] - mean1) squared) multiplied by math.sqrt(sum over each i from 0 to n-1 inclusive of (data2[i] - mean2) squared). RESULT denominator]
            [COMMAND If denominator is not equal to 0, return numerator divided by denominator. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: denominator equals 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 instead of dividing by zero. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the skewness of a list of numeric values as the third central moment scaled by n divided by ((n - 1) * (n - 2) * std_deviation^3), returning 0 when the population standard deviation is zero." skewness]
    [INPUTS]
        <REF> data </REF>: list[float] "Input list of numeric values for which skewness is calculated. An empty list causes a ZeroDivisionError because n is 0."
    [END_INPUTS]

    [OUTPUTS]
        <REF> skewness </REF>: float "The computed skewness value, or 0 when the population standard deviation is zero. If n is 2 and std_deviation is not 0, the denominator contains a factor of 0 and a ZeroDivisionError occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of data. RESULT n]
            [COMMAND Compute mean as sum(data) divided by n. RESULT mean]
            [COMMAND Compute variance as sum((x - mean) ** 2 for x in data) divided by n. RESULT variance]
            [COMMAND Compute std_deviation as math.sqrt(variance). RESULT std_deviation]
            [COMMAND If std_deviation != 0, compute skewness as sum((x - mean) ** 3 for x in data) * n / ((n - 1) * (n - 2) * std_deviation ** 3); otherwise set skewness to 0. RESULT skewness]
            [COMMAND Return skewness. RESULT skewness]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the excess kurtosis of a numeric dataset using the population standard deviation and returns math.nan when the standard deviation is zero." kurtosis]
    [INPUTS]
        <REF> data </REF>: list[float] "Input numeric dataset for which excess kurtosis is calculated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "Excess kurtosis value of the dataset, or math.nan when the population standard deviation is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign n as the length of data. RESULT n]
            [COMMAND Compute mean as sum(data) divided by n. RESULT mean]
            [COMMAND Compute std_dev as the square root of sum((x - mean) ** 2 for x in data) divided by n. RESULT std_dev]
            [COMMAND When std_dev is not equal to 0, compute centered_data as a list of x - mean for each x in data. RESULT centered_data]
            [COMMAND When std_dev is not equal to 0, compute fourth_moment as sum(x ** 4 for x in centered_data) divided by n. RESULT fourth_moment]
            [COMMAND When std_dev is not equal to 0, compute kurtosis_value as (fourth_moment / (std_dev ** 4)) - 3. RESULT kurtosis_value]
            [COMMAND When std_dev is not equal to 0, return kurtosis_value. RESULT kurtosis_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: std_dev equals 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return math.nan. RESULT math.nan]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the Gaussian probability density function value for each element in data using the supplied mean mu and standard deviation sigma." pdf]
    [INPUTS]
        <REF> data </REF>: iterable of numeric values "Collection of numeric values at which the PDF is evaluated; the output order and length match this collection."
        <REF> mu </REF>: numeric "Mean parameter of the normal distribution."
        <REF> sigma </REF>: numeric "Standard deviation parameter of the normal distribution; should be positive and non-zero for valid PDF values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> pdf_values </REF>: list of numeric values "List of calculated Gaussian PDF values corresponding to each element in data, preserving the original order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each x in data, compute 1 / (sigma * sqrt(2 * math.pi)) * exp(-0.5 * ((x - mu) / sigma) ** 2) and collect the results in order into the list pdf_values. RESULT pdf_values]
            [COMMAND Return the computed list pdf_values. RESULT pdf_values]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
