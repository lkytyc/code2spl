# CLASS IpUtil

[DEFINE_WORKER: "Determines whether the given IP address string is a valid IPv4 address by attempting to parse it with socket.inet_pton and returning True only if parsing succeeds." is_valid_ipv4]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if ip_address is a valid IPv4 address; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the try block, call socket.inet_pton(socket.AF_INET, ip_address) to attempt to parse the input as an IPv4 address. RESULT validation_result]
            [COMMAND If the socket.inet_pton call does not raise socket.error, return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.inet_pton(socket.AF_INET, ip_address) raises socket.error.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False because the address is not a valid IPv4 address. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether the given IP address string is a valid IPv6 address using socket.inet_pton and returns True on success or False on socket.error." is_valid_ipv6]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address string to validate as IPv6."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if ip_address is a valid IPv6 address; False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.inet_pton with socket.AF_INET6 and ip_address to validate the IPv6 address. RESULT validation_result]
            [COMMAND If the call succeeds, return True to indicate that ip_address is a valid IPv6 address. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.inet_pton raises socket.error because ip_address is not a valid IPv6 address.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that ip_address is not a valid IPv6 address. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resolves the primary hostname for a given IP address and returns it, or returns None if socket.gethostbyaddr fails with socket.herror." get_hostname]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address to resolve to a hostname."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string|null "The resolved hostname, or None if resolution fails with socket.herror."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the try block, call socket.gethostbyaddr(ip_address) and take the first element from the returned tuple as the hostname. RESULT hostname]
            [COMMAND Return hostname as the successful result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If socket.gethostbyaddr(ip_address) raises socket.herror inside the try block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the fallback result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
