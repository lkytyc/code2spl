# CLASS ArgumentParser

[DEFINE_WORKER: "Initializes a new instance by setting self.arguments and self.types to empty dictionaries and self.required to an empty set." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty dictionary to the instance attribute self.arguments. RESULT self.arguments]
            [COMMAND Create an empty set by calling the built-in set constructor and assign it to the instance attribute self.required. RESULT self.required]
            [COMMAND Assign an empty dictionary to the instance attribute self.types. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses command_string by dropping the first token, then stores long options (--key=value or --flag) and short options (-key value or -key) in self.arguments using _convert_type or True, and returns a boolean success indicator plus missing required arguments." parse_arguments]
    [INPUTS]
        <REF> command_string </REF>: string "The raw command string whose first token is the command name and whose remaining tokens are arguments to parse."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: tuple "Returns (False, missing_args) when required arguments are missing; otherwise returns (True, None)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split command_string on whitespace, drop the first token, and assign the remaining token list to args. RESULT args]
            [COMMAND For each index i from 0 to len(args) - 1, assign args[i] to arg. RESULT arg]
            [COMMAND Within the for loop over i, if arg starts with '--', remove the leading '--' and split the remainder on '=' into key_value. RESULT key_value]
            [COMMAND Within the long-option branch, if len(key_value) equals 2, assign self.arguments[key_value[0]] = self._convert_type(key_value[0], key_value[1]). RESULT self.arguments[key_value[0]]]
            [COMMAND Within the long-option branch, otherwise (len(key_value) is not 2, including no '=' or multiple '=' signs), assign self.arguments[key_value[0]] = True. RESULT self.arguments[key_value[0]]]
            [COMMAND Within the for loop over i, else if arg starts with '-' and is not a long option, set key = arg[1:]. RESULT key]
            [COMMAND Within the short-option branch, if i + 1 is less than len(args) and args[i + 1] does not start with '-', assign self.arguments[key] = self._convert_type(key, args[i + 1]); do not skip the consumed next argument in the for-loop indexing. RESULT self.arguments[key]]
            [COMMAND Within the short-option branch, otherwise assign self.arguments[key] = True. RESULT self.arguments[key]]
            [COMMAND Within the for loop over i, if arg does not start with '-', leave arg unchanged and continue the loop without storing anything. RESULT arg]
            [COMMAND After the for loop, compute missing_args = self.required - set(self.arguments.keys()). RESULT missing_args]
            [COMMAND If missing_args is empty, return True and None. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: missing_args is non-empty after required arguments are compared with parsed argument keys.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False and missing_args. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the value associated with the given key from the instance's arguments mapping, or None if the key is absent." get_argument]
    [INPUTS]
        <REF> self </REF>: object "The current instance that owns the arguments mapping."
        <REF> key </REF>: any "The key to look up in the arguments mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The value stored under key in self.arguments, or None if the key is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the arguments mapping attribute on self. RESULT arguments_map]
            [COMMAND Return the value from arguments_map for key using a mapping get operation with default value None. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers an argument by adding it to the required set when required is true and always recording its type in the types mapping." add_argument]
    [INPUTS]
        <REF> arg </REF>: str "The argument name or key to register."
        <REF> required </REF>: bool "If true, the argument is added to the required set."
        <REF> arg_type </REF>: type "The type to associate with the argument."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If required is True, add arg to self.required. RESULT self.required]
            [COMMAND Store arg_type into self.types under key arg, overwriting any existing value or inserting a new key-value pair. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert a value by retrieving the converter associated with arg from self.types and calling it with value, falling back to the original value when a ValueError or KeyError occurs." _convert_type]
    [INPUTS]
        <REF> arg </REF>: object "Key used to select the converter from self.types."
        <REF> value </REF>: object "Value to convert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "Converted value, or the original value if conversion raises ValueError or KeyError."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the converter callable from self.types using arg as the key. RESULT converter]
            [COMMAND Call the converter with value. RESULT converted_value]
            [COMMAND Return converted_value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Retrieving the converter from self.types or calling the converter raises ValueError or KeyError.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the original value unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
