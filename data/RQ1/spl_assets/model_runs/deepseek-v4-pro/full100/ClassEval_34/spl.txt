# CLASS DocFileHandler

        <REF> file_path </REF>: string "The file path value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter file_path to the instance attribute self.file_path. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a document from self.file_path and returns the paragraph texts joined by newline characters." read_text]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "All paragraph texts joined with newline characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Document object from self.file_path by calling Document(self.file_path). RESULT doc]
            [COMMAND Initialize an empty list named text. RESULT text]
            [COMMAND For each paragraph in doc.paragraphs, append paragraph.text to text. RESULT text]
            [COMMAND Return the string formed by joining all elements in text with newline characters. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the provided text content into a new document paragraph with a specified font size and alignment, saves the document, and returns True on success or False if any error occurs." write_text]
    [INPUTS]
        <REF> content </REF>: str "Text content to write into the document paragraph."
        <REF> font_size </REF>: number "Font size in points to apply to the run; default is 12."
        <REF> alignment </REF>: str "Paragraph alignment identifier resolved to an alignment value; default is 'left'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if the document is created, configured, and saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the try block, create a new Document instance. RESULT doc]
            [COMMAND Within the try block, add a new paragraph to the document. RESULT paragraph]
            [COMMAND Within the try block, add a run to the paragraph with the provided content text. RESULT run]
            [COMMAND Within the try block, set the run font size to Pt(font_size). RESULT font]
            [COMMAND Within the try block, resolve the alignment identifier by calling _get_alignment_value(alignment). RESULT alignment_value]
            [COMMAND Within the try block, assign the resolved alignment value to the paragraph alignment. RESULT paragraph.alignment]
            [COMMAND Within the try block, save the document to self.file_path. RESULT doc]
            [COMMAND Return True after the document saves successfully. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs during document creation, paragraph or run addition, font or alignment assignment, or document saving.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False due to the caught exception. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a heading with the specified text and level to the DOCX document at self.file_path, saves the document, and returns True, returning False if any exception occurs." add_heading]
    [INPUTS]
        <REF> heading </REF>: str "The heading text to add to the document."
        <REF> level </REF>: int "The heading level; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if the heading was added and the document was saved successfully; False if any exception occurred."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the DOCX document at self.file_path by creating a Document instance. RESULT doc]
            [COMMAND Add a heading using the given heading text and level to the opened document. RESULT heading_paragraph]
            [COMMAND Save the modified document back to self.file_path. RESULT save_result]
            [COMMAND Return True to indicate successful completion. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs while opening the document, adding the heading, or saving the document.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate failure. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a table to a document from the provided two-dimensional data and saves the document, returning True on success and False if any error occurs." add_table]
    [INPUTS]
        <REF> data </REF>: list[list] "Two-dimensional data used to create the table rows and cells."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the table is added and the document is saved successfully, False if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the Document at self.file_path and assign it to doc. RESULT doc]
            [COMMAND Add a table to doc with rows equal to len(data) and cols equal to len(data[0]) and assign it to table. RESULT table]
            [COMMAND Within the outer for loop over i and row in enumerate(data), for each j and cell_value in enumerate(row), set table.cell(i, j).text to str(cell_value). RESULT table]
            [COMMAND Save doc to self.file_path. RESULT save_result]
            [COMMAND Return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the document, adding the table, writing cell values, or saving the document.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resolves a paragraph alignment name to its corresponding WD_PARAGRAPH_ALIGNMENT constant, defaulting to left alignment when the name is not recognized." _get_alignment_value]
    [INPUTS]
        <REF> alignment </REF>: str "Alignment name to resolve, such as 'left', 'center', or 'right'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: WD_PARAGRAPH_ALIGNMENT "The matching paragraph alignment constant, or WD_PARAGRAPH_ALIGNMENT.LEFT if the input does not match a known alignment name."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Define alignment_options as a dictionary mapping 'left' to WD_PARAGRAPH_ALIGNMENT.LEFT, 'center' to WD_PARAGRAPH_ALIGNMENT.CENTER, and 'right' to WD_PARAGRAPH_ALIGNMENT.RIGHT. RESULT alignment_options]
            [COMMAND Compute lowercased_alignment by calling alignment.lower(). RESULT lowercased_alignment]
            [COMMAND Return alignment_options.get(lowercased_alignment, WD_PARAGRAPH_ALIGNMENT.LEFT), returning the mapped alignment for 'left', 'center', or 'right' and defaulting to WD_PARAGRAPH_ALIGNMENT.LEFT for unrecognized values. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
