# CLASS ShoppingCart

[DEFINE_WORKER: "Initializes the instance by setting its items attribute to a new empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a fresh empty dictionary to the instance attribute self.items. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds or updates an item entry in the items dictionary with the given price and quantity." add_item]
    [INPUTS]
        <REF> item </REF>: object "The key in the items dictionary to add or update."
        <REF> price </REF>: number "The price value to store for the item."
        <REF> quantity </REF>: integer "The quantity value to store for the item; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.items[item] to a new dictionary containing the keys 'price' and 'quantity' with values price and quantity respectively, overwriting any existing entry or creating a new one. RESULT self.items[item]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrements the quantity of a given item in self.items when the item exists, and does nothing when the item is missing." remove_item]
    [INPUTS]
        <REF> item </REF>: object "The key of the item to look up in self.items."
        <REF> quantity </REF>: number "The amount to subtract from the item's current quantity; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item exists as a key in self.items. RESULT item_exists]
            [COMMAND If item_exists is true, decrement self.items[item]['quantity'] by quantity. RESULT updated_quantity]
            [COMMAND If item_exists is false, take no action. RESULT no_change]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current value of the instance attribute `items` to the caller." view_items]
    [INPUTS]
        <REF> self </REF>: object "The current instance whose `items` attribute is returned."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "The object currently referenced by the instance attribute `items`."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the current value of the instance attribute `items` to the caller. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total price by summing quantity times price for every item in self.items." total_price]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> total </REF>: float "Sum of quantity multiplied by price for all items in self.items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each item in self.items.values(), compute the product of item['quantity'] and item['price'], sum all products with an empty collection yielding 0, and return the total as a float. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
