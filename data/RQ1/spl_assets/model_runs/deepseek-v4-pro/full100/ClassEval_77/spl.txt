# CLASS Snake

[DEFINE_WORKER: "Initializes the snake instance by setting its default length and score, copying screen and block configuration, computing the initial centered position, and storing the provided food position." __init__]
    [INPUTS]
        <REF> SCREEN_WIDTH </REF>: number "External screen width used to compute the initial centered position."
        <REF> SCREEN_HEIGHT </REF>: number "External screen height used to compute the initial centered position."
        <REF> BLOCK_SIZE </REF>: number "External block size stored unchanged on the instance."
        <REF> food_position </REF>: object "External food position propagated directly into instance state."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the literal constant 1 to the instance attribute length as the initial snake length. RESULT self.length]
            [COMMAND Copy the parameter SCREEN_WIDTH into the instance attribute SCREEN_WIDTH. RESULT self.SCREEN_WIDTH]
            [COMMAND Copy the parameter SCREEN_HEIGHT into the instance attribute SCREEN_HEIGHT. RESULT self.SCREEN_HEIGHT]
            [COMMAND Copy the parameter BLOCK_SIZE into the instance attribute BLOCK_SIZE. RESULT self.BLOCK_SIZE]
            [COMMAND Compute the initial centered position as ((SCREEN_WIDTH / 2), (SCREEN_HEIGHT / 2)) using true division by 2, wrap that tuple in a list, and assign it to the instance attribute positions. RESULT self.positions]
            [COMMAND Assign the literal constant 0 to the instance attribute score as the initial score. RESULT self.score]
            [COMMAND Store the parameter food_position directly into the instance attribute food_position. RESULT self.food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Moves the snake by computing a new wrapped head position from direction, eating food when applicable, then either resetting on self-collision or inserting the new head and trimming the tail to the configured length." move]
    [INPUTS]
        <REF> direction </REF>: iterable of two numbers "Movement direction vector; must unpack into exactly two components x and y."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set cur to the current head position, which is the first element of self.positions. RESULT cur]
            [COMMAND Unpack direction into components x and y. RESULT x, y]
            [COMMAND Compute new as the 2-tuple ((cur[0] + (x * self.BLOCK_SIZE)) % self.SCREEN_WIDTH, (cur[1] + (y * self.BLOCK_SIZE)) % self.SCREEN_HEIGHT). RESULT new]
            [COMMAND If new equals self.food_position, call self.eat_food(). RESULT none]
            [COMMAND Evaluate the collision condition: len(self.positions) > 2 and new in self.positions[2:]. RESULT collision_condition]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(self.positions) > 2 and new in self.positions[2:]]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.reset(). RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: not (len(self.positions) > 2 and new in self.positions[2:])]
        [SEQUENTIAL_BLOCK]
            [COMMAND Insert new at index 0 of self.positions. RESULT self.positions]
            [COMMAND Within the else branch of the collision check, if len(self.positions) > self.length, pop the last element from self.positions. RESULT self.positions]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Randomly places food on an unoccupied grid-aligned position by repeatedly generating a new random position while the current position is already occupied." random_food_position]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND While self.food_position is in self.positions, assign self.food_position to a tuple whose x-coordinate is random.randint(0, self.SCREEN_WIDTH // self.BLOCK_SIZE - 1) * self.BLOCK_SIZE and whose y-coordinate is random.randint(0, self.SCREEN_HEIGHT // self.BLOCK_SIZE - 1) * self.BLOCK_SIZE, using inclusive randint bounds, then re-evaluate the while condition. RESULT self.food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resets the game state by setting length to 1, score to 0, centering the positions list on the screen midpoint, and generating a new random food position." reset]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.length to the constant integer value 1. RESULT self.length]
            [COMMAND Set self.positions to a list containing one tuple whose first element is self.SCREEN_WIDTH divided by 2 and whose second element is self.SCREEN_HEIGHT divided by 2. RESULT self.positions]
            [COMMAND Set self.score to the constant integer value 0. RESULT self.score]
            [COMMAND Call self.random_food_position() to generate a new random food position. RESULT self.random_food_position_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes a food consumption event by increasing the current object's length by one, adding 100 points to its score, and generating a new random food position." eat_food]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase the current object's length attribute by 1. RESULT self.length]
            [COMMAND Increase the current object's score attribute by 100. RESULT self.score]
            [COMMAND Call the method random_food_position on the current object without capturing any returned value. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
