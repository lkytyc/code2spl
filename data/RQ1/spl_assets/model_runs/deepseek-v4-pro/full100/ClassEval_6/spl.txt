# CLASS AvgPartition

[DEFINE_WORKER: "Initializes the instance by storing the supplied list reference and limit value without transformation or boundary enforcement." __init__]
    [INPUTS]
        <REF> lst </REF>: list "The list or iterable whose reference is assigned directly to the instance attribute self.lst."
        <REF> limit </REF>: number "The limit value assigned directly to the instance attribute self.limit without validation or defaulting."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the instance attribute self.lst to the same reference as the parameter lst, without copying. RESULT self.lst]
            [COMMAND Assign the instance attribute self.limit to the value of the parameter limit, without transformation or boundary enforcement. RESULT self.limit]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the number of full groups and leftover elements when dividing the length of self.lst by self.limit." setNum]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "Tuple of two integers: size, the number of full groups of size self.limit, and remainder, the number of leftover elements."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current length of self.lst and store it as lst_len. RESULT lst_len]
            [COMMAND Compute size as the integer quotient of lst_len divided by self.limit; if self.limit is zero, this step raises ZeroDivisionError. RESULT size]
            [COMMAND Compute remainder as the modulo of lst_len divided by self.limit. RESULT remainder]
            [COMMAND Return size and remainder as a tuple. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the slice of self.lst for the given zero-based index, distributing remainder extra elements among the first remainder partitions." get]
    [INPUTS]
        <REF> index </REF>: int "Zero-based index of the partition to retrieve."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The slice of self.lst corresponding to the computed partition."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.setNum() to obtain the base partition size and the number of leading partitions that receive one extra element. RESULT size, remainder]
            [COMMAND Compute start as index * size + min(index, remainder), where min(index, remainder) accounts for extra elements already distributed to earlier partitions. RESULT start]
            [COMMAND Compute end as start + size. RESULT end]
            [COMMAND If index + 1 <= remainder, increment end by 1 so the current partition receives one extra element. RESULT end]
            [COMMAND Return self.lst[start:end] as the result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
