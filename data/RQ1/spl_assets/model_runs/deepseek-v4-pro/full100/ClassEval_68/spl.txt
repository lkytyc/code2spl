# CLASS PageUtil

[DEFINE_WORKER: "Initializes pagination state by storing the input data and page size, then computing total_items from the data length and total_pages via ceiling division." __init__]
    [INPUTS]
        <REF> data </REF>: collection "Collection object used to compute total_items via len(data) and stored on self.data."
        <REF> page_size </REF>: integer "Positive integer page size used to compute total_pages."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.data to the input parameter data. RESULT self.data]
            [COMMAND Assign self.page_size to the input parameter page_size. RESULT self.page_size]
            [COMMAND Compute self.total_items as len(data). RESULT self.total_items]
            [COMMAND Compute self.total_pages using the formula (self.total_items + page_size - 1) // page_size, performing integer floor division after adding page_size - 1. RESULT self.total_pages]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a page of data by computing zero-based start and end indices from a 1-based page number, or return an empty list for an out-of-range page number." get_page]
    [INPUTS]
        <REF> page_number </REF>: int "1-based page number of the page to retrieve."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The slice of self.data from start_index inclusive to end_index exclusive for the requested page, or an empty list when page_number is less than 1 or greater than self.total_pages."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND After the page_number validity check passes, compute start_index as (page_number - 1) * self.page_size. RESULT start_index]
            [COMMAND After the page_number validity check passes, compute end_index as start_index + self.page_size, producing the exclusive upper bound. RESULT end_index]
            [COMMAND Return the slice self.data[start_index:end_index] with end_index exclusive. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: page_number is less than 1 or greater than self.total_pages]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Retrieves one page of paginated data and its associated metadata when the requested page number is within range, returning an empty dictionary otherwise." get_page_info]
    [INPUTS]
        <REF> page_number </REF>: integer "1-based page number to fetch"
    [END_INPUTS]

    [OUTPUTS]
        <REF> page_info </REF>: dictionary "Pagination metadata and current page data when the page number is valid; otherwise an empty dictionary."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For a page_number that satisfies 1 <= page_number <= self.total_pages, compute start_index = (page_number - 1) * self.page_size. RESULT start_index]
            [COMMAND Compute end_index = min(start_index + self.page_size, self.total_items). RESULT end_index]
            [COMMAND Extract page_data = self.data[start_index:end_index]. RESULT page_data]
            [COMMAND Initialize page_info as a dictionary. RESULT page_info]
            [COMMAND Set page_info['current_page'] to page_number. RESULT page_info.current_page]
            [COMMAND Set page_info['per_page'] to self.page_size. RESULT page_info.per_page]
            [COMMAND Set page_info['total_pages'] to self.total_pages. RESULT page_info.total_pages]
            [COMMAND Set page_info['total_items'] to self.total_items. RESULT page_info.total_items]
            [COMMAND Set page_info['has_previous'] based on whether page_number > 1. RESULT page_info.has_previous]
            [COMMAND Set page_info['has_next'] based on whether page_number < self.total_pages. RESULT page_info.has_next]
            [COMMAND Set page_info['data'] to page_data. RESULT page_info.data]
            [COMMAND Return page_info. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: page_number < 1 or page_number > self.total_pages]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty dictionary. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search the instance data for items whose string representation contains the given keyword, count the matching results, compute the total number of pages using page_size, and return a dictionary containing the keyword, totals, and matching results." search]
    [INPUTS]
        <REF> keyword </REF>: string "Search term used to filter self.data by checking whether it is contained in str(item) for each item."
    [END_INPUTS]

    [OUTPUTS]
        <REF> search_info </REF>: dict "Dictionary containing the keyword, total_results, total_pages, and the filtered results list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build results as a new list by iterating over self.data and keeping each item where keyword is contained in str(item). RESULT results]
            [COMMAND Set num_results to len(results). RESULT num_results]
            [COMMAND Set num_pages to the ceiling division of num_results by self.page_size using the formula (num_results + self.page_size - 1) // self.page_size. RESULT num_pages]
            [COMMAND Create search_info as a dictionary with keyword set to keyword, total_results set to num_results, total_pages set to num_pages, and results set to the same results list object by reference. RESULT search_info]
            [COMMAND Return search_info. RESULT search_info]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
