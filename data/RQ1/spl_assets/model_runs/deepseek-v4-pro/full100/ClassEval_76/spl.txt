# CLASS SignInSystem

[DEFINE_WORKER: "Initializes the instance by creating a new empty dictionary and assigning it to the users attribute." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance attribute self.users to a newly created empty dictionary, discarding any previous value. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a username to the users mapping with an initial value of False and returns True if it was not already present, otherwise returns False." add_user]
    [INPUTS]
        <REF> username </REF>: str "Username key to add to the users mapping when it is not already present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the username was added to users; False if the username already existed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is already present as a key in self.users. RESULT username_exists]
            [COMMAND Within the false branch when username_exists is false, set self.users[username] to False. RESULT user_added]
            [COMMAND Within the false branch, return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username_exists is true, meaning username is already present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the true branch, return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Signs in a user by returning false when the username is not present in self.users, otherwise marking the existing user entry as true and returning true." sign_in]
    [INPUTS]
        <REF> username </REF>: string "Username to check and sign in."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "False when the username does not exist; true when an existing user was marked signed in."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether username is not present in self.users. RESULT username_missing]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When username_missing is true because username is not in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When username_missing is false because username is present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the value of self.users[username] to True. RESULT updated_user]
            [COMMAND Return True immediately. RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username exists in self.users and has a truthy stored value, returning True only when both conditions hold and False otherwise." check_sign_in]
    [INPUTS]
        <REF> username </REF>: string "Username to check in self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if username exists in self.users and the associated value is truthy; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check if username is not in self.users. RESULT is_missing]
            [COMMAND If username is not in self.users, return False immediately. RESULT result]
            [COMMAND Otherwise, within the outer else branch where username exists in self.users, check if self.users[username] is truthy. RESULT is_truthy]
            [COMMAND If self.users[username] is truthy, return True. RESULT result]
            [COMMAND Otherwise, return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True when every value in self.users is truthy or when self.users is empty, and False when any value is falsy." all_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if all values in self.users are truthy or if self.users is empty; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether all values in self.users are truthy by applying the all() built-in to self.users.values(), short-circuiting at the first falsy value and treating an empty self.users as truthy. RESULT all_users_signed_in]
            [COMMAND If all_users_signed_in is true, return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If all_users_signed_in is false, meaning at least one value in self.users is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects and returns the usernames of all users whose signed_in status is falsy from self.users." all_not_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> not_signed_in_users </REF>: list "List of usernames whose signed_in value is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list not_signed_in_users to hold usernames of users who are not signed in. RESULT not_signed_in_users]
            [COMMAND For each username, signed_in pair in self.users.items() in insertion order, if signed_in is falsy, append username to not_signed_in_users. RESULT not_signed_in_users]
            [COMMAND Return the not_signed_in_users list. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
