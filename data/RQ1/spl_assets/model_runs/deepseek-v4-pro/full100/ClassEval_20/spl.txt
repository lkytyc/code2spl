    [INPUTS]
    [END_INPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]

            [COMMAND Assign a newly created empty dictionary to the instance attribute self.users, clearing any previous mapping and leaving it with no keys. RESULT self.users]

[DEFINE_WORKER: "Adds a new username to the users collection when it is not already present, returning True on success and False if the username already exists." add_user]
    [INPUTS]
        <REF> username </REF>: str "The username to add to the users collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if the username was added, False if the username already existed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username already exists as a key in self.users. RESULT username_exists]
            [COMMAND If username does not already exist, add username to self.users with an empty list as its value. RESULT new_user_entry]
            [COMMAND Return True to indicate that the new user was created. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username already exists in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that the user was not added. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove a username from the users collection and return True if the username was present, otherwise return False." remove_user]
    [INPUTS]
        <REF> username </REF>: str "Username to remove from the users collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the username was present and removed; False if the username was not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists in self.users. RESULT exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: exists is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Delete username from self.users. RESULT deleted]
            [COMMAND Return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.users. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sends a message by validating that both sender and receiver exist in self.users, then appends a timestamped message record to both users' message histories and returns True, or returns False if either user is unknown." send_message]
    [INPUTS]
        <REF> sender </REF>: string "Identifier of the sender user."
        <REF> receiver </REF>: string "Identifier of the receiver user."
        <REF> message </REF>: string "Message content to be sent."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the message is appended to both users; False when the sender or receiver is not present in self.users."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate that sender is present in self.users and receiver is present in self.users. RESULT users_valid]
            [COMMAND Obtain the current timestamp as a string using datetime.now().strftime("%Y-%m-%d %H:%M:%S"). RESULT timestamp]
            [COMMAND Construct a message_info dictionary with keys sender, receiver, message, and timestamp. RESULT message_info]
            [COMMAND Append message_info to self.users[sender]. RESULT sender_history_updated]
            [COMMAND Append message_info to self.users[receiver]. RESULT receiver_history_updated]
            [COMMAND Return True. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Either sender is not present in self.users or receiver is not present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Retrieves the messages associated with a username from the users mapping, returning an empty list when the username is not present." get_messages]
    [INPUTS]
        <REF> username </REF>: string "The username key used to look up the user's messages in self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The list of messages associated with the given username, or an empty list if the username is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists as a key in the self.users mapping. RESULT username_exists]
            [COMMAND If username_exists is true, return the value stored at self.users[username]. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not present as a key in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
