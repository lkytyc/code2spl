# CLASS CurrencyConverter

[DEFINE_WORKER: "Initializes the instance's currency exchange rate dictionary with seven fixed rates relative to USD." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.rates to a new dictionary containing the fixed currency conversion rates: 'USD': 1.0, 'EUR': 0.85, 'GBP': 0.72, 'JPY': 110.15, 'CAD': 1.23, 'AUD': 1.34, 'CNY': 6.40. RESULT self.rates]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an amount from one currency to another using exchange rates from self.rates, returning the original amount when the currencies match and False when either currency rate is unavailable." convert]
    [INPUTS]
        <REF> amount </REF>: number "The monetary amount to convert."
        <REF> from_currency </REF>: string "The source currency code."
        <REF> to_currency </REF>: string "The target currency code."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If from_currency equals to_currency, return amount immediately without converting. RESULT amount]
            [COMMAND If from_currency is not present in self.rates or to_currency is not present in self.rates, return False immediately. RESULT False]
            [COMMAND Retrieve from_rate from self.rates using from_currency as the key. RESULT from_rate]
            [COMMAND Retrieve to_rate from self.rates using to_currency as the key. RESULT to_rate]
            [COMMAND Compute converted_amount as (amount / from_rate) * to_rate. RESULT converted_amount]
            [COMMAND Return converted_amount. RESULT converted_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a new list containing the keys of the object's rates dictionary in current insertion order." get_supported_currencies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> currency_list </REF>: list "A new list of keys from self.rates; returns an empty list when self.rates is empty. Requires self.rates to be an initialized dictionary-like object."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return a new list created from the keys of self.rates, preserving the dictionary's current insertion order. RESULT currency_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a currency exchange rate to the self.rates mapping only when the currency is not already present, and returns False for duplicate currency keys." add_currency_rate]
    [INPUTS]
        <REF> currency </REF>: hashable key "Currency identifier used as a key in self.rates."
        <REF> rate </REF>: value "Exchange rate value to store for the currency when it is not already present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool|None "Returns False when the currency already exists in self.rates; otherwise returns None after adding the rate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency is already present as a key in self.rates. RESULT currency_exists]
            [COMMAND If currency is not present in self.rates, assign rate to self.rates[currency]. RESULT self.rates[currency]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency is already present as a key in self.rates]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.rates. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the rate for an existing currency in self.rates and returns False when the currency is not present." update_currency_rate]
    [INPUTS]
        <REF> currency </REF>: string "Currency key to look up in self.rates."
        <REF> new_rate </REF>: number "New rate value to store for the currency."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency exists in self.rates. RESULT currency_exists]
            [COMMAND When currency exists in self.rates, assign new_rate to self.rates[currency]. RESULT self.rates[currency]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency not in self.rates]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
