[DEFINE_WORKER: "Adds a new employee record to the employees mapping when the given employee_id is not already present, returning True on successful insertion and False when the id already exists." add_employee]
    [INPUTS]
        <REF> employee_id </REF>: string "Unique key used to identify the employee in the employees mapping."
        <REF> name </REF>: string "Employee name to store in the record."
        <REF> position </REF>: string "Employee position to store in the record."
        <REF> department </REF>: string "Employee department to store in the record."
        <REF> salary </REF>: number "Employee salary to store in the record."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the new employee record was added, False if the employee_id already existed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id already exists in self.employees. RESULT employee_exists]
            [COMMAND If employee_exists is false, create a new employee record dictionary containing name, position, department, and salary, and assign it to self.employees[employee_id]. RESULT employee_record]
            [COMMAND Return True to indicate the employee was added successfully. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_exists is true, meaning employee_id is already present in self.employees]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.employees. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes an employee by ID from the employees mapping and returns True if the ID was present, otherwise returns False." remove_employee]
    [INPUTS]
        <REF> employee_id </REF>: any "The employee identifier to remove from the employees mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the employee was found and removed; False when the employee ID was not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id exists as a key in self.employees. RESULT employee_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_id exists as a key in self.employees]
        [SEQUENTIAL_BLOCK]
            [COMMAND Delete self.employees[employee_id]. RESULT delete_result]
            [COMMAND Return True immediately. RESULT return_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: employee_id does not exist as a key in self.employees]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates an employee record with the provided employee_info fields after verifying the employee exists and all provided keys already exist, returning True on success and False on failure." update_employee]
    [INPUTS]
        <REF> employee_id </REF>: int "Identifier used to look up the employee record."
        <REF> employee_info </REF>: dict "Mapping of employee fields whose values will be assigned to the matching employee keys."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the employee exists, all provided keys exist, and the update is applied; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the employee record for employee_id by calling self.get_employee(employee_id) and assign the returned value to employee. RESULT employee]
            [COMMAND If employee is equal to False, follow the invalid-employee alternative flow and return False; otherwise, enter the else branch and iterate over each key and value in employee_info.items(). RESULT validation_started]
            [COMMAND Within the else branch, for each key and value from employee_info.items(), if key is not in employee, follow the missing-key alternative flow and return False; if all keys are present, continue after the first loop. RESULT keys_validated]
            [COMMAND Within the else branch, after successful validation, for each key and value in employee_info.items(), assign employee[key] = value. RESULT employee_updated]
            [COMMAND Return True to indicate the employee was found, all keys were valid, and the update was applied. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee equals False, including values that compare equal to False such as 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying employee. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the first loop over employee_info.items(), the current key is not present in employee]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying employee. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Looks up an employee by ID and returns the associated record, or False when the ID is not present." get_employee]
    [INPUTS]
        <REF> employee_id </REF>: object "Identifier used as the lookup key in self.employees."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object or boolean "The employee record when employee_id exists in self.employees; otherwise the boolean False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Load the employee mapping from self.employees. RESULT employee_mapping]
            [COMMAND Check whether employee_id exists as a key in employee_mapping. RESULT employee_exists]
            [COMMAND If employee_exists is true, return employee_mapping[employee_id]. RESULT employee_record]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the boolean False. RESULT not_found]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts self.employees into a dictionary keyed by employee ID, where each value is a dictionary containing employee_ID and all employee_info fields, or returns an empty dictionary when self.employees is falsy." list_employees]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> employee_data </REF>: dict "A dictionary keyed by employee_id, where each value is a dictionary containing the employee_ID and all employee_info fields; empty if self.employees is falsy or empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize employee_data as an empty dictionary. RESULT employee_data]
            [COMMAND Check whether self.employees is truthy and non-empty; if it is falsy, empty, or None, skip the nested employee processing block and continue to the final return. RESULT guard_check]
            [COMMAND If self.employees is truthy and non-empty, for each employee_id and employee_info from self.employees.items(): initialize employee_details as a new empty dictionary. RESULT employee_details]
            [COMMAND Within the if self.employees guard and the outer for loop over employee_id and employee_info from self.employees.items(), set employee_details["employee_ID"] to the current employee_id. RESULT employee_details]
            [COMMAND Within the if self.employees guard and the outer for loop over employee_id and employee_info from self.employees.items(), for each key and value from employee_info.items(), set employee_details[key] to value; if employee_info contains the key "employee_ID", that value overwrites the employee_ID set from employee_id. RESULT employee_details]
            [COMMAND Within the if self.employees guard and the outer for loop over employee_id and employee_info from self.employees.items(), store employee_details in employee_data under the key employee_id. RESULT employee_data]
            [COMMAND Return employee_data. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.employees is falsy, empty, or None.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Leave employee_data unchanged as an empty dictionary and proceed to the final return. RESULT employee_data]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
