# CLASS MetricsCalculator

[DEFINE_WORKER: "Initialize four classification metric counters to zero by setting true positives, false positives, false negatives, and true negatives on the instance." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance attribute true_positives to the integer literal 0. RESULT self.true_positives]
            [COMMAND Set the instance attribute false_positives to the integer literal 0. RESULT self.false_positives]
            [COMMAND Set the instance attribute false_negatives to the integer literal 0. RESULT self.false_negatives]
            [COMMAND Set the instance attribute true_negatives to the integer literal 0. RESULT self.true_negatives]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the confusion matrix counters on self by iterating over paired predicted and true binary labels and incrementing the matching true positive, false positive, false negative, or true negative counter for each pair." update]
    [INPUTS]
        <REF> predicted_labels </REF>: Iterable[int] "Predicted binary labels to compare element-wise with true labels."
        <REF> true_labels </REF>: Iterable[int] "Ground-truth binary labels to compare element-wise with predicted labels."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each predicted and true pair obtained from zip(predicted_labels, true_labels), evaluate the following if/elif conditions sequentially for the current pair and then continue with the next pair; iteration stops at the shorter input length. RESULT current_pair]
            [COMMAND Within the for loop, if predicted == 1 and true == 1, increment self.true_positives by 1. RESULT true_positives_incremented]
            [COMMAND Within the for loop, else if predicted == 1 and true == 0, increment self.false_positives by 1. RESULT false_positives_incremented]
            [COMMAND Within the for loop, else if predicted == 0 and true == 1, increment self.false_negatives by 1. RESULT false_negatives_incremented]
            [COMMAND Within the for loop, else if predicted == 0 and true == 0, increment self.true_negatives by 1. RESULT true_negatives_incremented]
            [COMMAND Within the for loop, if none of the four conditions match because a value is not exactly 0 or 1, do nothing for this pair and continue to the next pair. RESULT no_update_for_non_binary_pair]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates internal metric counters with the predicted and true labels, then returns precision as true positives divided by the sum of true positives and false positives, or 0.0 when the denominator is zero." precision]
    [INPUTS]
        <REF> predicted_labels </REF>: any "Predicted labels used to update internal metric counters."
        <REF> true_labels </REF>: any "True labels used to update internal metric counters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> precision_result </REF>: float "Computed precision value; returns 0.0 when the sum of true positives and false positives is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) to update internal true positive and false positive counters. RESULT self]
            [COMMAND After the zero-denominator guard, return self.true_positives / (self.true_positives + self.false_positives) as the precision result. RESULT precision_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.true_positives + self.false_positives == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 as the precision result. RESULT precision_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the recall metric by updating internal true positive and false negative counts with the provided predicted and true labels, returning 0.0 when the sum of true positives and false negatives is zero and otherwise returning true positives divided by that sum." recall]
    [INPUTS]
        <REF> predicted_labels </REF>: list "Predicted labels used to update the internal metric counters."
        <REF> true_labels </REF>: list "Ground-truth labels used to update the internal metric counters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> recall_value </REF>: float "Recall score computed as true positives divided by the sum of true positives and false negatives, or 0.0 when the denominator is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) and discard any return value. RESULT discarded_update_result]
            [COMMAND When the denominator guard condition is false, compute and return self.true_positives / (self.true_positives + self.false_negatives) as the recall value. RESULT recall_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.true_positives + self.false_negatives == 0 after calling self.update.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 as the recall value to avoid division by zero. RESULT recall_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the F1 score from predicted and true labels by updating internal statistics, deriving precision and recall, and returning the harmonic mean with a zero-denominator safeguard." f1_score]
    [INPUTS]
        <REF> predicted_labels </REF>: list "Predicted labels to evaluate."
        <REF> true_labels </REF>: list "Ground truth labels to compare against."
    [END_INPUTS]

    [OUTPUTS]
        <REF> f1_score </REF>: float "The computed F1 score, or 0.0 when precision plus recall is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) to update internal statistics before computing metrics. RESULT update_result]
            [COMMAND Assign precision to the return value of self.precision(predicted_labels, true_labels). RESULT precision]
            [COMMAND Assign recall to the return value of self.recall(predicted_labels, true_labels). RESULT recall]
            [COMMAND Check whether precision + recall equals 0.0; if true, follow the alternative flow for zero denominator and return 0.0. RESULT is_zero_denominator]
            [COMMAND Return (2 * precision * recall) / (precision + recall). RESULT f1_score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: precision + recall == 0.0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0. RESULT f1_score]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes classifier accuracy after updating confusion matrix counts, returning 0.0 when the total count is zero and otherwise returning the ratio of correct predictions to total predictions." accuracy]
    [INPUTS]
        <REF> predicted_labels </REF>: list "Predicted labels for the samples used to update the confusion matrix counts."
        <REF> true_labels </REF>: list "Ground truth labels for the samples used to update the confusion matrix counts."
    [END_INPUTS]

    [OUTPUTS]
        <REF> accuracy </REF>: float "Classification accuracy as a float between 0.0 and 1.0, or 0.0 when no samples are counted."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update(predicted_labels, true_labels) to update the true positive, true negative, false positive, and false negative counts. RESULT none]
            [COMMAND Compute total = self.true_positives + self.true_negatives + self.false_positives + self.false_negatives. RESULT total]
            [COMMAND If total is not equal to 0, return (self.true_positives + self.true_negatives) / total. RESULT accuracy]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: total equals 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0. RESULT accuracy]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
