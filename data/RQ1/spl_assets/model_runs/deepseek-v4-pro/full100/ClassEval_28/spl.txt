# CLASS DatabaseProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided database_name parameter into the instance attribute self.database_name." __init__]
    [INPUTS]
        <REF> database_name </REF>: text "The database name value assigned to the instance attribute self.database_name."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of the parameter database_name to the instance attribute self.database_name. RESULT self.database_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a SQLite table if it does not exist with an integer primary key, one text column, and one integer column, then commits and closes the connection." create_table]
    [INPUTS]
        <REF> self </REF>: object "The instance on which the method is called, providing the database_name attribute."
        <REF> table_name </REF>: string "The name of the table to create, interpolated directly into the SQL statement."
        <REF> key1 </REF>: string "The name of the first column to create with TEXT type, interpolated directly into the SQL statement."
        <REF> key2 </REF>: string "The name of the second column to create with INTEGER type, interpolated directly into the SQL statement."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Establish a SQLite connection to self.database_name using sqlite3.connect and assign it to conn. RESULT conn]
            [COMMAND Create a cursor from conn using conn.cursor() and assign it to cursor. RESULT cursor]
            [COMMAND Build create_table_query by directly interpolating table_name, key1, and key2 without quoting or escaping into the string "CREATE TABLE IF NOT EXISTS {table_name} (id INTEGER PRIMARY KEY, {key1} TEXT, {key2} INTEGER)" and assign it to create_table_query. RESULT create_table_query]
            [COMMAND Execute create_table_query using cursor.execute(create_table_query). RESULT execute_result]
            [COMMAND Commit the transaction on conn using conn.commit(). RESULT commit_result]
            [COMMAND Close the connection conn using conn.close(). RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens an SQLite database, inserts each record from data into the target table's name and age columns using a parameterized INSERT statement, commits the transaction, and closes the connection." insert_into_database]
    [INPUTS]
        <REF> table_name </REF>: str "Name of the SQLite table into which records will be inserted."
        <REF> data </REF>: iterable of dict "Collection of records where each record provides name and age keys used as bound INSERT parameters; if empty, the loop body is skipped."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database identified by self.database_name by calling sqlite3.connect(self.database_name); this uses SQLite default behavior for special database names such as ':memory:' or an empty string, and assigns the returned connection to conn. RESULT conn]
            [COMMAND Create a cursor object from conn by calling conn.cursor() and assign it to cursor. RESULT cursor]
            [COMMAND For each item in data, within the for loop, build the INSERT query string by interpolating table_name into the SQL template: f"INSERT INTO {table_name} (name, age) VALUES (?, ?)", and assign it to insert_query. RESULT insert_query]
            [COMMAND Within the same for loop over each item in data, execute the parameterized INSERT on cursor with the current item's name and age as bound parameters: cursor.execute(insert_query, (item['name'], item['age'])). RESULT execution_result]
            [COMMAND After the for loop over data completes without an exception, commit the transaction on conn by calling conn.commit(). RESULT commit_status]
            [COMMAND After committing, close the database connection by calling conn.close(). RESULT close_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search a SQLite table by name and return all matching rows, or None if no rows match." search_database]
    [INPUTS]
        <REF> table_name </REF>: string "Name of the SQLite table to query; it is interpolated directly into the SELECT statement."
        <REF> name </REF>: string "Bound parameter value used to match the name column in the WHERE clause."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list|None "All rows returned by the query when result is truthy, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using the value of self.database_name and assign the connection object to conn. RESULT conn]
            [COMMAND Create a cursor from conn and assign it to cursor. RESULT cursor]
            [COMMAND Build the SELECT query string by embedding table_name into the SQL text SELECT * FROM {table_name} WHERE name = ? and assign it to select_query. RESULT select_query]
            [COMMAND Execute select_query on cursor using the bound parameter tuple (name,) so the cursor holds the active result set. RESULT cursor]
            [COMMAND Fetch all rows from cursor and assign the returned rows to result. RESULT result]
            [COMMAND If result is truthy, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: result is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes a row from a SQLite table by matching the name column, using a parameterized DELETE statement and committing the change." delete_from_database]
    [INPUTS]
        <REF> table_name </REF>: str "SQL table name to delete from; interpolated directly into SQL and must be trusted or sanitized."
        <REF> name </REF>: str "Value matched against the name column in the WHERE clause as a bound parameter."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection to the database path stored in self.database_name. RESULT conn]
            [COMMAND Create a cursor from the open connection. RESULT cursor]
            [COMMAND Build the DELETE SQL statement by interpolating table_name into "DELETE FROM {table_name} WHERE name = ?". RESULT delete_query]
            [COMMAND Execute delete_query on the cursor with name bound as the WHERE parameter. RESULT result]
            [COMMAND Commit the transaction on the connection. RESULT result]
            [COMMAND Close the connection. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
