# CLASS MinesweeperGame

[DEFINE_WORKER: "Initializes the Minesweeper game state by storing the board size and mine count, generating the mine and player maps, and resetting the score to zero." __init__]
    [INPUTS]
        <REF> n </REF>: int "Positive integer representing the board size."
        <REF> k </REF>: int "Non-negative integer representing the number of mines to place, typically less than the total number of cells."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constructor parameter n to the instance attribute self.n. RESULT self.n]
            [COMMAND Assign the constructor parameter k to the instance attribute self.k. RESULT self.k]
            [COMMAND Call self.generate_mine_sweeper_map() and assign its return value to self.minesweeper_map. RESULT self.minesweeper_map]
            [COMMAND Call self.generate_playerMap() and assign its return value to self.player_map. RESULT self.player_map]
            [COMMAND Assign the literal constant 0 to self.score. RESULT self.score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates an n by n minesweeper board with k randomly placed mines and increments adjacent non-mine cells to count neighboring mines." generate_mine_sweeper_map]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> arr </REF>: list[list] "An n by n minesweeper board containing integer mine counts and 'X' mine markers."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an n by n two-dimensional list arr where n is self.n, initializing every cell to integer 0. RESULT arr]
            [COMMAND For each num from 0 through self.k - 1 inclusive, execute the following mine placement and neighbor update steps. RESULT num]
            [COMMAND Within the loop over num, randomly choose x from the inclusive range 0 to self.n - 1. RESULT x]
            [COMMAND Within the loop over num, randomly choose y from the inclusive range 0 to self.n - 1. RESULT y]
            [COMMAND Within the loop over num, set arr[y][x] to the mine marker 'X'. RESULT arr]
            [COMMAND Within the loop over num, if x >= 0 and x <= self.n - 2 and y >= 0 and y <= self.n - 1, then if arr[y][x + 1] is not 'X', increment arr[y][x + 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 1 and x <= self.n - 1 and y >= 0 and y <= self.n - 1, then if arr[y][x - 1] is not 'X', increment arr[y][x - 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 1 and x <= self.n - 1 and y >= 1 and y <= self.n - 1, then if arr[y - 1][x - 1] is not 'X', increment arr[y - 1][x - 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 0 and x <= self.n - 2 and y >= 1 and y <= self.n - 1, then if arr[y - 1][x + 1] is not 'X', increment arr[y - 1][x + 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 0 and x <= self.n - 1 and y >= 1 and y <= self.n - 1, then if arr[y - 1][x] is not 'X', increment arr[y - 1][x] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 0 and x <= self.n - 2 and y >= 0 and y <= self.n - 2, then if arr[y + 1][x + 1] is not 'X', increment arr[y + 1][x + 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 1 and x <= self.n - 1 and y >= 0 and y <= self.n - 2, then if arr[y + 1][x - 1] is not 'X', increment arr[y + 1][x - 1] by 1. RESULT arr]
            [COMMAND Within the loop over num, if x >= 0 and x <= self.n - 1 and y >= 0 and y <= self.n - 2, then if arr[y + 1][x] is not 'X', increment arr[y + 1][x] by 1. RESULT arr]
            [COMMAND Return arr. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates an n by n player map grid as a list of lists filled with '-' characters, returning an empty list when self.n is less than or equal to 0." generate_playerMap]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> arr </REF>: list[list[str]] "An n by n list of lists filled with '-'; empty list when self.n is less than or equal to 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the value of self.n. RESULT n_value]
            [COMMAND When self.n is greater than 0, create arr as a list of self.n row lists, each row list containing self.n '-' elements. RESULT arr]
            [COMMAND Return arr. RESULT arr_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.n is less than or equal to 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set arr to an empty list. RESULT arr]
            [COMMAND Return arr. RESULT arr_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether all hidden cells are mines by scanning the visible board and returns False if any hidden cell is not a mine, otherwise True." check_won]
    [INPUTS]
        <REF> map </REF>: list[list[str]] "Visible Minesweeper board where '-' represents a hidden cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns True if every hidden cell is a mine, and False if any hidden cell is not a mine."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate i over each row index from 0 to self.n - 1 inclusive. RESULT row_index]
            [COMMAND Within the outer loop over i, iterate j over each column index from 0 to self.n - 1 inclusive. RESULT column_index]
            [COMMAND Within the nested loops over i and j, if map[i][j] equals '-' and self.minesweeper_map[i][j] does not equal 'X', immediately return False. RESULT return_false]
            [COMMAND After both loops complete without having returned False, return True. RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes a single Minesweeper move by returning False if the selected cell contains a mine; otherwise it reveals the cell, increments the score, and returns True on a win or the updated player map otherwise." sweep]
    [INPUTS]
        <REF> x </REF>: int "Row index of the selected cell."
        <REF> y </REF>: int "Column index of the selected cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool | player_map "False if the selected cell is a mine; True if the move wins the game; otherwise the updated player map."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the selected cell is a mine by evaluating whether self.minesweeper_map[x][y] equals 'X'. RESULT is_mine]
            [COMMAND If is_mine is false, assign self.player_map[x][y] to the value of self.minesweeper_map[x][y]. RESULT player_map_update]
            [COMMAND If is_mine is false, increment self.score by 1. RESULT score_update]
            [COMMAND If is_mine is false, evaluate whether the result of self.check_won(self.player_map) is exactly True. RESULT won]
            [COMMAND If is_mine is false and won is not exactly True, return self.player_map. RESULT safe_move_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The selected cell is a mine.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT mine_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The selected cell is safe and the result of self.check_won(self.player_map) is exactly True.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True. RESULT win_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
