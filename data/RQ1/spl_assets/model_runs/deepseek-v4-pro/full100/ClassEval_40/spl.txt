# CLASS FitnessTracker

[DEFINE_WORKER: "Initialize a BMI-related object by assigning height, weight, age, sex, and standard normal BMI boundary ranges for males and females." __init__]
    [INPUTS]
        <REF> height </REF>: float "Body height value used to initialize the instance attribute."
        <REF> weight </REF>: float "Body weight value used to initialize the instance attribute."
        <REF> age </REF>: int "Age value used to initialize the instance attribute."
        <REF> sex </REF>: string "Sex marker used to initialize the instance attribute and relevant for BMI boundary classification."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the instance attribute self.height directly from the input parameter height. RESULT self.height]
            [COMMAND Assign the instance attribute self.weight directly from the input parameter weight. RESULT self.weight]
            [COMMAND Assign the instance attribute self.age directly from the input parameter age. RESULT self.age]
            [COMMAND Assign the instance attribute self.sex directly from the input parameter sex. RESULT self.sex]
            [COMMAND Assign self.BMI_std to a constant list containing two dictionaries: one maps male to the inclusive normal BMI range with lower bound 20 and upper bound 25, and one maps female to the inclusive normal BMI range with lower bound 19 and upper bound 24. RESULT self.BMI_std]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes and returns the body mass index value by dividing self.weight by the square of self.height." get_BMI]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> bmi </REF>: float "The computed BMI value from self.weight divided by self.height squared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute self.weight divided by self.height squared and return the resulting quotient. RESULT bmi]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Classifies a person's BMI as too fat, too thin, or normal by comparing the value returned by self.get_BMI() against sex-specific BMI standard ranges from self.BMI_std." condition_judge]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_code </REF>: int "Returned classification code: 1 means too fat, -1 means too thin, and 0 means normal."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_BMI() and assign the returned value to BMI. RESULT BMI]
            [COMMAND If self.sex is exactly "male", assign BMI_range to self.BMI_std[0]["male"]; else assign BMI_range to self.BMI_std[1]["female"]. RESULT BMI_range]
            [COMMAND If BMI is strictly greater than BMI_range[1], return 1 indicating too fat. RESULT return_code]
            [COMMAND Else if BMI is strictly less than BMI_range[0], return -1 indicating too thin. RESULT return_code]
            [COMMAND Else return 0 indicating normal. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates daily calorie intake by applying the Mifflin-St Jeor BMR formula based on sex, weight, height, and age, then multiplying by an activity factor determined by condition_judge." calculate_calorie_intake]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> calorie_intake </REF>: number "Estimated daily calorie intake based on calculated BMR and activity-level multiplier."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If self.sex equals exactly "male" (case-sensitive), compute BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age + 5; otherwise compute BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age - 161. Assign the computed value to BMR. RESULT BMR]
            [COMMAND Call self.condition_judge(). If this first call returns 1, set calorie_intake = BMR * 1.2. RESULT calorie_intake]
            [COMMAND If the first call to self.condition_judge() did not return 1, call self.condition_judge() a second time. If this second call returns -1, set calorie_intake = BMR * 1.6. RESULT calorie_intake]
            [COMMAND If the first call to self.condition_judge() did not return 1 and the second call did not return -1, set calorie_intake = BMR * 1.4. RESULT calorie_intake]
            [COMMAND Return calorie_intake. RESULT calorie_intake]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
