[DEFINE_WORKER: "Evaluate an arithmetic expression string with support for +, -, *, /, ^ and parentheses by using a shunting-yard parser with an operand stack, operator stack, and numeric buffer, and return the final numeric result or None if no operand was produced." calculate]
    [INPUTS]
        <REF> expression </REF>: string "Arithmetic expression string to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float | None "The evaluated numeric result, or None if operand_stack is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize operand_stack as an empty list. RESULT operand_stack]
            [COMMAND Initialize operator_stack as an empty list. RESULT operator_stack]
            [COMMAND Initialize num_buffer as an empty string. RESULT num_buffer]
            [COMMAND For each character char in expression, process it from left to right. RESULT char]
            [COMMAND Within the for loop over char, if char is a digit or char equals '.', append char to num_buffer. RESULT num_buffer]
            [COMMAND Within the for loop over char, in the else branch when char is neither a digit nor '.', if num_buffer is non-empty, append float(num_buffer) to operand_stack and then reset num_buffer to an empty string. RESULT operand_stack, num_buffer]
            [COMMAND Within the for loop over char, in the same else branch after the buffer flush check, if char is one of '+', '-', '*', '/', '^', while operator_stack is non-empty and operator_stack[-1] is not '(' and self.precedence(operator_stack[-1]) is greater than or equal to self.precedence(char), evaluated left-to-right with short-circuit, call self.apply_operator(operand_stack, operator_stack) and assign the returned pair back to operand_stack and operator_stack. RESULT operand_stack, operator_stack]
            [COMMAND Within the for loop over char, in the same operator branch, after the while loop, append char to operator_stack. RESULT operator_stack]
            [COMMAND Within the for loop over char, in the same else branch, else if char equals '(', append char to operator_stack. RESULT operator_stack]
            [COMMAND Within the for loop over char, in the same else branch, else if char equals ')', while operator_stack is non-empty and operator_stack[-1] is not '(', evaluated left-to-right with short-circuit, call self.apply_operator(operand_stack, operator_stack) and assign the returned pair back to operand_stack and operator_stack. RESULT operand_stack, operator_stack]
            [COMMAND Within the for loop over char, in the same ')' branch, after the while loop, pop the top '(' from operator_stack. RESULT operator_stack]
            [COMMAND After the for loop, if num_buffer is non-empty, append float(num_buffer) to operand_stack. RESULT operand_stack]
            [COMMAND While operator_stack is non-empty, call self.apply_operator(operand_stack, operator_stack) and assign the returned pair back to operand_stack and operator_stack. RESULT operand_stack, operator_stack]
            [COMMAND If operand_stack is non-empty, return operand_stack[-1]; otherwise return None. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the arithmetic precedence level of a given operator using a predefined dictionary, defaulting to 0 for unknown operators." precedence]
    [INPUTS]
        <REF> operator </REF>: string "The operator symbol whose precedence is looked up."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The integer precedence level for the operator, or 0 if the operator is not in the predefined precedence dictionary."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Define a constant dictionary `precedences` mapping '+' and '-' to 1, '*' and '/' to 2, and '^' to 3. RESULT precedences]
            [COMMAND Return the value from `precedences` for the input `operator`, defaulting to 0 when the operator is not present. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Pops an operator from operator_stack, applies the corresponding operator function from self.operators to the top two operands from operand_stack, pushes the result back, and returns the updated operand and operator stacks." apply_operator]
    [INPUTS]
        <REF> self </REF>: object "The current evaluator instance that owns the operators mapping."
        <REF> operand_stack </REF>: list "Stack of operands from which the top two values are popped and to which the computed result is appended."
        <REF> operator_stack </REF>: list "Stack of operators from which the top value is popped."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the updated operand_stack and operator_stack."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop the top element from operator_stack and store it as operator. RESULT operator]
            [COMMAND Evaluate whether operator equals '^'; both the if and else branches contain identical subsequent operations. RESULT operator_is_caret]
            [COMMAND Within either branch, pop the top element from operand_stack and store it as operand2. RESULT operand2]
            [COMMAND Within either branch, pop the next top element from operand_stack and store it as operand1. RESULT operand1]
            [COMMAND Within either branch, compute result by calling self.operators[operator](operand1, operand2). RESULT result]
            [COMMAND Within either branch, append result to operand_stack. RESULT operand_stack]
            [COMMAND Return operand_stack and operator_stack. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
