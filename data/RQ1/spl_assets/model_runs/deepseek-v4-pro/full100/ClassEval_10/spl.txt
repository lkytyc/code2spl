    [END_INPUTS]

[DEFINE_WORKER: "Calculates the proportion of '0' and '1' characters and the total length of the binary string stored in self.binary_string." calculate_binary_info]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: dict "Dictionary containing 'Zeroes' as the proportion of zero characters, 'Ones' as the proportion of one characters, and 'Bit length' as the total string length."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the number of '0' characters in self.binary_string and store the count as zeroes_count. RESULT zeroes_count]
            [COMMAND Count the number of '1' characters in self.binary_string and store the count as ones_count. RESULT ones_count]
            [COMMAND Compute the length of self.binary_string and store it as total_length. RESULT total_length]
            [COMMAND Compute zeroes_percentage as zeroes_count divided by total_length using true division; if total_length equals 0, this division raises ZeroDivisionError. RESULT zeroes_percentage]
            [COMMAND Compute ones_percentage as ones_count divided by total_length using true division; if total_length equals 0, this division raises ZeroDivisionError. RESULT ones_percentage]
            [COMMAND Return a dictionary with keys 'Zeroes', 'Ones', and 'Bit length' mapped to zeroes_percentage, ones_percentage, and total_length, respectively. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the instance binary string into an ASCII string by processing it in 8-bit chunks, converting each chunk from base 2 to an integer, appending the integer to a bytearray, and decoding the bytearray as ASCII." convert_to_ascii]
    [INPUTS]
        <REF> binary_string </REF>: string "Instance attribute containing the binary string to convert; it is read in chunks of up to 8 characters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decoded_ascii </REF>: string "The ASCII string produced by decoding the bytearray built from each converted binary chunk."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty bytearray and assign it to byte_array. RESULT byte_array]
            [COMMAND For each index i starting at 0, continuing while i is less than len(self.binary_string), and increasing by 8 after each iteration, slice self.binary_string[i:i+8] and assign the resulting substring to byte; on the final iteration the slice may be shorter than 8 characters. RESULT byte]
            [COMMAND Within the same for loop over i, convert byte from a base-2 string to a decimal integer and assign the result to decimal; byte may be shorter than 8 characters only on the final iteration and is converted without left-padding. RESULT decimal]
            [COMMAND Within the same for loop over i, append decimal to byte_array. RESULT byte_array]
            [COMMAND Decode byte_array using ASCII encoding and return the decoded string. RESULT decoded_ascii]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert the instance attribute binary_string into a UTF-8 decoded string by splitting it into 8-bit chunks, converting each chunk to an integer byte value, and appending it to a bytearray." convert_to_utf8]
    [INPUTS]
        <REF> binary_string </REF>: string "Instance attribute containing a binary string whose 8-bit chunks represent UTF-8 encoded bytes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decoded_text </REF>: string "The UTF-8 decoded string returned after converting the binary chunks to bytes and decoding them."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize byte_array as a new empty bytearray. RESULT byte_array]
            [COMMAND Start a definite for loop over i using range(0, len(self.binary_string), 8), where stop len(self.binary_string) is exclusive and is evaluated once when range is created; if the length is 0, skip the loop body. RESULT i]
            [COMMAND Within the for loop over i, assign byte to self.binary_string[i:i+8]; if the remaining string is shorter than 8 characters, the slice returns the remaining characters. RESULT byte]
            [COMMAND Within the for loop over i, convert byte from base 2 to an integer. RESULT decimal]
            [COMMAND Within the for loop over i, append decimal to byte_array. RESULT byte_array]
            [COMMAND After the loop finishes, decode byte_array using UTF-8 and return the resulting string. RESULT decoded_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
