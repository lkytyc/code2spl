[DEFINE_WORKER: "Converts recognized number words and ordinal words in a hyphen-normalized text string into numeric string representations while preserving unrecognized words and spacing." text2int]
    [INPUTS]
        <REF> textnum </REF>: string "Input text containing words that may include number words, ordinal words, and ordinary words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> curstring </REF>: string "The converted string where recognized numeric and ordinal words are replaced with their numeric values and non-number words remain unchanged."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign to textnum the result of replacing every hyphen '-' in the input textnum with a space. RESULT textnum]
            [COMMAND Initialize current to 0, result to 0, curstring to the empty string, and onnumber to False. RESULT current, result, curstring, onnumber]
            [COMMAND For each word in textnum.split(), evaluate whether word is a key in self.ordinal_words. RESULT word_in_ordinal_words]
            [COMMAND Within the outer for loop, if word is in self.ordinal_words, set scale to 1 and increment to self.ordinal_words[word]; then update current to current * scale + increment and set onnumber to True. RESULT scale, increment, current, onnumber]
            [COMMAND Within the outer for loop, if word is not in self.ordinal_words, iterate over each ending and replacement in self.ordinal_endings; for each ending, if the current word ends with that ending, replace that suffix with replacement and assign the result back to word, continuing through all endings. RESULT word]
            [COMMAND Within the outer for loop and the not-in-ordinal_words branch, after the inner ordinal-ending loop, if the current word is not in self.numwords and onnumber is True, append repr(result + current) plus a single space to curstring. RESULT curstring]
            [COMMAND Still within the not-in-numwords branch, append the current word plus a single space to curstring, then reset result and current to 0 and set onnumber to False. RESULT curstring, result, current, onnumber]
            [COMMAND Still within the outer for loop and the not-in-ordinal_words branch, if the current word is in self.numwords, set scale and increment to self.numwords[word], update current to current * scale + increment, and if scale is greater than 100 add current to result and reset current to 0; then set onnumber to True. RESULT scale, increment, current, result, onnumber]
            [COMMAND After the outer for loop over textnum.split() finishes, if onnumber is True, append repr(result + current) to curstring. RESULT curstring]
            [COMMAND Return curstring as the method result. RESULT curstring]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates an input string by replacing hyphens with spaces, splitting it into words, accepting known ordinal words, normalizing other words through ordinal endings, and returning false immediately if a normalized word is not in numwords." is_valid_input]
    [INPUTS]
        <REF> textnum </REF>: string "The input text to validate after hyphen replacement and ordinal or number word normalization."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "Returns true if every whitespace-separated token is either a known ordinal word or can be normalized to a known number word; otherwise returns false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace every '-' character in textnum with a space and assign the resulting string back to textnum. RESULT textnum]
            [COMMAND Split textnum on whitespace into an ordered sequence of word tokens. RESULT tokens]
            [COMMAND For each word in tokens, check whether word is present in self.ordinal_words. RESULT is_ordinal_word]
            [COMMAND If is_ordinal_word is true, skip the remaining ordinal-ending normalization and numwords validation for that word and continue to the next token. RESULT word_skip]
            [COMMAND If is_ordinal_word is false, for each ending and replacement pair in self.ordinal_endings in order, check whether the current word ends with ending; if it does, replace the matched suffix with replacement and continue checking subsequent endings against the modified word because no break is used. RESULT normalized_word]
            [COMMAND After the inner ending-replacement loop for a non-ordinal word, check whether normalized_word is not present in self.numwords. RESULT invalid_word_found]
            [COMMAND After all word tokens have been processed without triggering an early return, return true. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: For any word that is not in self.ordinal_words, after all matching ordinal-ending replacements have been applied, the normalized word is not found in self.numwords.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false immediately, terminating validation and skipping all remaining word tokens. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
