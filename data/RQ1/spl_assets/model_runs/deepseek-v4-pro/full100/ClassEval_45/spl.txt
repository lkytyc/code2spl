# CLASS ImageProcessor

[DEFINE_WORKER: "Initializes the instance by setting the image attribute to None." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign None to the instance attribute self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Loads an image from the given path using Image.open and stores the returned image object in the instance attribute self.image." load_image]
    [INPUTS]
        <REF> image_path </REF>: str "Path-like object pointing to a readable, supported image file."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call Image.open(image_path) to open the image file; if image_path is valid, readable, and points to a supported image format, assign the returned image object to self.image; otherwise the call raises before assignment and self.image remains unchanged or undefined. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Saves the current image object to the specified path if it is set." save_image]
    [INPUTS]
        <REF> save_path </REF>: string "Filesystem path where the image will be saved."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is truthy. RESULT image_exists]
            [COMMAND If image_exists is true, save self.image to save_path. RESULT save_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resizes the current image to the specified width and height when an image is present." resize_image]
    [INPUTS]
        <REF> width </REF>: int "Target width for the image resize operation."
        <REF> height </REF>: int "Target height for the image resize operation."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check the if condition self.image to determine whether an image is present and truthy. RESULT image_present]
            [COMMAND If image_present is true, within the if self.image branch, call self.image.resize((width, height)) and assign the returned image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Rotates the current image by the specified number of degrees if an image is set." rotate_image]
    [INPUTS]
        <REF> degrees </REF>: number "The angle in degrees by which to rotate the image."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If self.image is truthy, assign self.image to the result of calling self.image.rotate(degrees). RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adjusts the brightness of the current image by the specified factor when an image is present." adjust_brightness]
    [INPUTS]
        <REF> factor </REF>: float "Brightness adjustment factor applied to the image via ImageEnhance.Brightness.enhance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If self.image is truthy, create an ImageEnhance.Brightness enhancer from self.image. RESULT enhancer]
            [COMMAND If self.image is truthy, call enhancer.enhance(factor) and assign the returned image to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.image is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do not adjust brightness; leave self.image unchanged. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
