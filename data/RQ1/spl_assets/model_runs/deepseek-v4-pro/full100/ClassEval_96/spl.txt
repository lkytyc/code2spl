[DEFINE_WORKER: "Query a city's weather and return the temperature in the requested units, converting between Celsius and Fahrenheit when needed, or False if the city is missing." query]
    [INPUTS]
        <REF> weather_list </REF>: dict "Mapping of city names to weather records containing temperature, weather, and temperature units."
        <REF> tmp_units </REF>: string "Target temperature unit, either 'celsius' or 'fahrenheit'; defaults to 'celsius'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool|tuple|null "False if the city is not found; a tuple of temperature and weather if units match or conversion is supported; null if units differ and tmp_units is not celsius or fahrenheit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the parameter weather_list to self.weather_list. RESULT self.weather_list]
            [COMMAND If self.city is not a key in weather_list, return False immediately and skip all remaining steps. RESULT $return]
            [COMMAND Within the else branch when self.city exists in weather_list, assign self.temperature to self.weather_list[self.city]['temperature']. RESULT self.temperature]
            [COMMAND Within the same else branch, assign self.weather to self.weather_list[self.city]['weather']. RESULT self.weather]
            [COMMAND If self.weather_list[self.city]['temperature units'] equals tmp_units, return self.temperature and self.weather. RESULT $return]
            [COMMAND If self.weather_list[self.city]['temperature units'] differs from tmp_units and tmp_units is 'celsius', call self.fahrenheit_to_celsius() and return its result together with self.weather. RESULT $return]
            [COMMAND If self.weather_list[self.city]['temperature units'] differs from tmp_units and tmp_units is 'fahrenheit', call self.celsius_to_fahrenheit() and return its result together with self.weather. RESULT $return]
            [COMMAND If self.weather_list[self.city]['temperature units'] differs from tmp_units and tmp_units is neither 'celsius' nor 'fahrenheit', return None implicitly because the inner if/elif has no matching branch. RESULT $return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the Celsius temperature stored in self.temperature to Fahrenheit by applying the formula ((self.temperature * 9) / 5) + 32 and returning the result." celsius_to_fahrenheit]
    [INPUTS]
        <REF> temperature </REF>: number "The object attribute self.temperature representing the Celsius temperature to convert; it must be numeric."
    [END_INPUTS]

    [OUTPUTS]
        <REF> fahrenheit_result </REF>: number "The converted Fahrenheit temperature as a numeric result, using true division so the value is a float."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value of the expression ((self.temperature * 9) / 5) + 32, evaluating left-to-right with true division. RESULT fahrenheit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a Fahrenheit temperature stored in self.temperature to Celsius using the formula (temperature - 32) * 5 / 9 and returns the result as a float." fahrenheit_to_celsius]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> celsius_value </REF>: float "The Fahrenheit temperature converted to Celsius."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current Fahrenheit temperature from self.temperature. RESULT temperature_f]
            [COMMAND Subtract 32 from temperature_f. RESULT fahrenheit_offset]
            [COMMAND Multiply fahrenheit_offset by 5. RESULT scaled_value]
            [COMMAND Divide scaled_value by 9 using floating-point division to produce the Celsius temperature. RESULT celsius_value]
            [COMMAND Return celsius_value as the method result. RESULT method_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]
