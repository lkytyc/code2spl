[DEFINE_WORKER: "Sorts the input numeric dataset and returns its median, using the middle element for odd-length data and the average of the two middle elements for even-length data." median]
    [INPUTS]
        <REF> data </REF>: iterable of numbers "Non-empty input dataset of comparable numeric values for which the median is computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> median_result </REF>: number "The median of the input dataset: the middle element for an odd number of elements, or the average of the two middle elements for an even number of elements."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the input data in ascending order and assign the resulting list to sorted_data. RESULT sorted_data]
            [COMMAND Compute the length of sorted_data and assign it to n. RESULT n]
            [COMMAND If n % 2 == 1, assign sorted_data[n // 2] to median_result and return it. RESULT median_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When n % 2 == 0, meaning n is even]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign (sorted_data[n // 2 - 1] + sorted_data[n // 2]) / 2 to median_result and return it. RESULT median_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the mode values of the input data by counting frequencies, finding the maximum frequency, and returning all values that have that maximum frequency." mode]
    [INPUTS]
        <REF> data </REF>: list "Input collection of hashable values whose frequencies are counted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> mode_values </REF>: list "List of values that have the maximum frequency in the input data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty dictionary named counts to store the frequency of each value. RESULT counts]
            [COMMAND For each value in the input data, set counts[value] to counts.get(value, 0) + 1, incrementing the frequency count for that value. RESULT counts]
            [COMMAND Compute max_count as the maximum count among all frequency values in counts using max(counts.values()). RESULT max_count]
            [COMMAND Create mode_values as a list comprehension over counts.items(), including each value whose count equals max_count. RESULT mode_values]
            [COMMAND Return mode_values as the list of mode values. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: If data is empty, then counts is empty and max(counts.values()) raises ValueError, so mode_values is not assigned and execution stops.]
        [LOG "Cannot compute mode of an empty collection because max() is called on an empty sequence."]
        [THROW ValueError "max() arg is an empty sequence"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the correlation-like coefficient numerator divided by denominator for x and y, returning None when the denominator is zero." correlation]
    [INPUTS]
        <REF> x </REF>: sequence of numeric values "Input sequence; must be sized, non-empty, and contain numeric values."
        <REF> y </REF>: sequence of numeric values "Input sequence; must be sized, non-empty, and contain numeric values; should have the same length as x for consistent behavior."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: float or None "The computed ratio numerator / denominator if denominator is not zero; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as len(x). RESULT n]
            [COMMAND Compute mean_x as sum(x) divided by n. RESULT mean_x]
            [COMMAND Compute mean_y as sum(y) divided by n, where n is the length of x from the prior step. RESULT mean_y]
            [COMMAND Compute numerator as the sum over each paired (xi, yi) from zip(x, y) of (xi - mean_x) * (yi - mean_y); only the shorter paired prefix is used if x and y differ in length. RESULT numerator]
            [COMMAND Compute denominator as the square root of (sum((xi - mean_x)**2 for xi in x) * sum((yi - mean_y)**2 for yi in y)). RESULT denominator]
            [COMMAND If denominator is not zero, return numerator divided by denominator. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: denominator equals zero]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the arithmetic mean of the input data, or None when the data is empty." mean]
    [INPUTS]
        <REF> data </REF>: Collection[Number] "Input collection of numeric values whose mean is computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: Number | None "Arithmetic mean of data, or None if data is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute len(data) to determine whether the input collection is empty. RESULT data_length]
            [COMMAND If data is not empty, compute the arithmetic mean as sum(data) divided by len(data). RESULT mean_result]
            [COMMAND Return mean_result as the function result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(data) == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes an n-by-n column-wise correlation matrix for a numeric table by calculating the correlation between every pair of columns." correlation_matrix]
    [INPUTS]
        <REF> data </REF>: list[list[float]] "A non-empty table of numeric rows where each row represents an observation and each column represents a variable; all rows are assumed to have the same length as the first row."
    [END_INPUTS]

    [OUTPUTS]
        <REF> matrix </REF>: list[list[float]] "An n-by-n matrix where matrix[i][j] is the correlation between column i and column j."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize matrix to an empty list. RESULT matrix]
            [COMMAND Let n be len(data[0]), the number of columns based on the first row of data. RESULT n]
            [COMMAND For each i from 0 to n-1 inclusive, do the following. RESULT i]
            [COMMAND Within the outer for loop over i, initialize row to an empty list. RESULT row]
            [COMMAND Within the outer for loop over i, for each j from 0 to n-1 inclusive, do the following. RESULT j]
            [COMMAND Within the inner for loop over j, build column1 by taking the element at index i from each row in data. RESULT column1]
            [COMMAND Within the inner for loop over j, build column2 by taking the element at index j from each row in data. RESULT column2]
            [COMMAND Within the inner for loop over j, compute correlation = Statistics3.correlation(column1, column2). RESULT correlation]
            [COMMAND Within the inner for loop over j, append correlation to row. RESULT row]
            [COMMAND Within the outer for loop over i, after the inner loop over j completes, append row to matrix. RESULT matrix]
            [COMMAND Return matrix. RESULT matrix]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the sample standard deviation of a numeric dataset, returning None if the dataset has fewer than two values." standard_deviation]
    [INPUTS]
        <REF> data </REF>: sequence of numeric values "Re-iterable dataset of numeric values for which the sample standard deviation is computed; must support len and multiple iterations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or None "The sample standard deviation, computed as the square root of the sample variance when data has at least two values; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of data using len(data). RESULT n]
            [COMMAND Compute mean_value by calling Statistics3.mean(data). RESULT mean_value]
            [COMMAND Compute variance as the sum of squared differences between each element x in data and mean_value, divided by (n - 1). RESULT variance]
            [COMMAND Compute the sample standard deviation as the square root of variance using math.sqrt(variance), and return it. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n < 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes z-score normalization for the input data using its mean and standard deviation, returning None when the standard deviation is None or zero." z_score]
    [INPUTS]
        <REF> data </REF>: list[float] "Input collection of numeric values for which z-scores are computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[float] or None "A list of z-scores computed as (x - mean) / std_deviation for each x in data, or None if the standard deviation is None or zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the arithmetic mean of data using Statistics3.mean(data) and assign the result to mean. RESULT mean]
            [COMMAND Compute the standard deviation of data using Statistics3.standard_deviation(data) and assign the result to std_deviation. RESULT std_deviation]
            [COMMAND If std_deviation is not None and std_deviation is not equal to 0, compute a list of z-scores by applying (x - mean) / std_deviation to each x in data. RESULT z_scores]
            [COMMAND Return the computed z_scores list from the function. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: std_deviation is None or std_deviation == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately from the function without computing z-scores. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
