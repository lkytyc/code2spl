# CLASS DiscountStrategy

[DEFINE_WORKER: "Initializes an instance with customer, cart, and optional promotion references, then caches the computed total from the total method." __init__]
    [INPUTS]
        <REF> customer </REF>: object "Customer information or object assigned directly to self.customer."
        <REF> cart </REF>: object "Cart data assigned directly to self.cart."
        <REF> promotion </REF>: object "Optional promotion data assigned directly to self.promotion; defaults to None."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the customer parameter to the instance attribute self.customer without transformation. RESULT self.customer]
            [COMMAND Assign the cart parameter to the instance attribute self.cart without transformation. RESULT self.cart]
            [COMMAND Assign the promotion parameter to the instance attribute self.promotion without transformation. RESULT self.promotion]
            [COMMAND Call the instance method self.total() after the previous attributes are assigned and store its returned value in the name-mangled private instance attribute self.__total. RESULT self.__total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total value of all cart items by summing each item's quantity multiplied by its price and returns that total." total]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed total sum of quantity multiplied by price for all items in self.cart."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sum of item['quantity'] multiplied by item['price'] for each item in self.cart, using 0 as the initial accumulator; if self.cart is empty, the sum is 0. Assign the computed sum to self.__total. RESULT self.__total]
            [COMMAND Return the value stored in self.__total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the amount due by subtracting a promotion discount from the total, defaulting the discount to zero when no promotion is configured." due]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> due_amount </REF>: number "The amount due after subtracting the promotion discount from self.__total."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Inside the if branch where self.promotion is None, set discount to 0. RESULT discount]
            [COMMAND Return self.__total minus discount. RESULT due_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.promotion is not None]
        [SEQUENTIAL_BLOCK]
            [COMMAND Inside the else branch where self.promotion is not None, call self.promotion(self) and assign the returned value to discount. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates a promotional discount equal to 5% of the order total when the customer's fidelity score is at least 1000, otherwise returns 0." FidelityPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object with a total() method and a customer dictionary containing a fidelity score."
    [END_INPUTS]

    [OUTPUTS]
        <REF> discount </REF>: number "The computed promotional discount: order.total() * 0.05 when fidelity >= 1000; otherwise 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the customer fidelity value from order.customer['fidelity']. RESULT fidelity]
            [COMMAND Evaluate whether fidelity is greater than or equal to 1000. RESULT is_eligible]
            [COMMAND If is_eligible is true, compute order.total() * 0.05. RESULT discount]
            [COMMAND If is_eligible is true, return discount. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: fidelity < 1000]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the literal value 0. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total 10% bulk discount for all cart items whose quantity is at least 20." BulkItemPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object containing a cart list of items, where each item has quantity and price."
    [END_INPUTS]

    [OUTPUTS]
        <REF> discount </REF>: number "Total accumulated discount amount returned by the method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize discount to 0. RESULT discount]
            [COMMAND For each item in order.cart, if item['quantity'] >= 20, add item['quantity'] * item['price'] * 0.1 to discount; otherwise do nothing. RESULT discount]
            [COMMAND Return discount. RESULT discount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates a promotional amount: 7 percent of order.total() when the order contains at least 10 distinct products, otherwise 0." LargeOrderPromo]
    [INPUTS]
        <REF> order </REF>: object "An order object that provides a cart collection and a total method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> promo_amount </REF>: number "The promotional amount, equal to order.total() * 0.07 when the order has at least 10 distinct products, otherwise 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Collect the distinct product identifiers from order.cart into a set by extracting item['product'] from each item in order.cart. RESULT distinct_products]
            [COMMAND Count the number of elements in distinct_products. RESULT distinct_product_count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: distinct_product_count is greater than or equal to 10]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return order.total() * 0.07. RESULT promo_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: distinct_product_count is less than 10]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT promo_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
