    [END_INPUTS]

    [OUTPUTS]

        [SEQUENTIAL_BLOCK]

    [END_MAIN_FLOW]

[DEFINE_WORKER: "Adds a specified number of seconds to the instance's datetime and returns the resulting time formatted as HH:MM:SS." add_seconds]
    [INPUTS]
        <REF> seconds </REF>: number "Number of seconds to add to self.datetime; may be positive, negative, or zero."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The resulting time formatted as HH:MM:SS."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a datetime.timedelta duration from the supplied seconds. RESULT delta]
            [COMMAND Add delta to self.datetime to produce a new datetime. RESULT new_datetime]
            [COMMAND Set format to the string literal "%H:%M:%S". RESULT format]
            [COMMAND Return new_datetime.strftime(format). RESULT formatted_time]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses a string into a datetime.datetime object using the exact format '%Y-%m-%d %H:%M:%S'." string_to_datetime]
    [INPUTS]
        <REF> string </REF>: str "Input string to parse into a datetime.datetime object."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: datetime.datetime "The datetime.datetime object parsed from the input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call datetime.datetime.strptime on string with the exact format '%Y-%m-%d %H:%M:%S' and store the parsed datetime in parsed_datetime. RESULT parsed_datetime]
            [COMMAND Return parsed_datetime. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

    [END_INPUTS]

    [OUTPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the result of calling strftime on the datetime parameter with the format string %Y-%m-%d %H:%M:%S. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts two time strings to datetime objects and returns the rounded number of minutes from the first time to the second time based on the timedelta seconds field." get_minutes]
    [INPUTS]
        <REF> string_time1 </REF>: string "A string representation of the first time to be converted by self.string_to_datetime."
        <REF> string_time2 </REF>: string "A string representation of the second time to be converted by self.string_to_datetime."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Rounded number of minutes from time1 to time2, computed from the seconds field of the datetime difference."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.string_to_datetime with string_time1 as the argument and assign the resulting datetime object to time1. RESULT time1]
            [COMMAND Call self.string_to_datetime with string_time2 as the argument and assign the resulting datetime object to time2. RESULT time2]
            [COMMAND Compute the difference time2 - time1, read the difference's seconds field exactly as seconds, divide that value by 60, round the result to the nearest integer, and return it. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a datetime from the supplied year, month, day, hour, minute, and second and returns it formatted as '%Y-%m-%d %H:%M:%S'." get_format_time]
    [INPUTS]
        <REF> year </REF>: int "Year value for datetime construction."
        <REF> month </REF>: int "Month value for datetime construction."
        <REF> day </REF>: int "Day value for datetime construction."
        <REF> hour </REF>: int "Hour value for datetime construction."
        <REF> minute </REF>: int "Minute value for datetime construction."
        <REF> second </REF>: int "Second value for datetime construction."
    [END_INPUTS]

    [OUTPUTS]
        <REF> formatted_time </REF>: str "The datetime formatted as a string using '%Y-%m-%d %H:%M:%S'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the static format string "%Y-%m-%d %H:%M:%S" to format. RESULT format]
            [COMMAND Construct a datetime object from year, month, day, hour, minute, and second via datetime.datetime(year, month, day, hour, minute, second). RESULT time_item]
            [COMMAND Return the string produced by time_item.strftime(format). RESULT formatted_time]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
