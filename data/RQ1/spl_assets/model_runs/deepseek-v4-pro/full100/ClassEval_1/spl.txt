# CLASS AreaCalculator

[DEFINE_WORKER: "Initializes the instance by directly assigning the provided radius argument to the instance attribute self.radius without transformation or validation." __init__]
    [INPUTS]
        <REF> radius </REF>: Any "The radius value to store on the instance attribute self.radius."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of the radius parameter to the instance attribute self.radius. RESULT self.radius]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the area of a circle using the instance's radius and math.pi." calculate_circle_area]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> area </REF>: float "The computed circle area, equal to pi times the square of self.radius."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the result of multiplying math.pi by self.radius squared, where self.radius is squared before multiplication. RESULT area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate and return the surface area of a sphere from self.radius using 4 * math.pi * radius squared." calculate_sphere_area]
    [INPUTS]
        <REF> self.radius </REF>: number "Radius of the sphere; must be numeric and already initialized."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: number "Sphere surface area computed as 4 * math.pi * self.radius ** 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance attribute self.radius as the sphere radius. RESULT radius]
            [COMMAND Compute radius squared using radius ** 2. RESULT radius_squared]
            [COMMAND Multiply 4 * math.pi by radius_squared to compute the sphere surface area. RESULT surface_area]
            [COMMAND Return surface_area as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the surface area of a cylinder using the instance radius and the supplied height." calculate_cylinder_area]
    [INPUTS]
        <REF> self </REF>: object "The current instance containing the radius attribute."
        <REF> height </REF>: number "The height of the cylinder."
    [END_INPUTS]

    [OUTPUTS]
        <REF> area </REF>: number "The calculated cylinder surface area."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the radius attribute from self. RESULT radius]
            [COMMAND Read the supplied height parameter. RESULT height]
            [COMMAND Compute the sum of radius and height. RESULT radius_plus_height]
            [COMMAND Compute 2 * math.pi * radius. RESULT double_pi_radius]
            [COMMAND Multiply double_pi_radius by radius_plus_height to obtain the cylinder surface area. RESULT area]
            [COMMAND Return the computed area. RESULT return_area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the sector area from the instance radius and the given angle as (radius squared times angle) divided by 2." calculate_sector_area]
    [INPUTS]
        <REF> angle </REF>: numeric "Angle value used to scale the squared radius in the sector area calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: numeric "Computed sector area equal to (self.radius ** 2 * angle) / 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sector area using the expression self.radius ** 2 * angle / 2, which evaluates as ((self.radius ** 2) * angle) / 2, and return the resulting numeric value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes and returns the area of an annulus from inner and outer radii using the formula math.pi * (outer_radius ** 2 - inner_radius ** 2)." calculate_annulus_area]
    [INPUTS]
        <REF> inner_radius </REF>: float "Inner radius of the annulus"
        <REF> outer_radius </REF>: float "Outer radius of the annulus"
    [END_INPUTS]

    [OUTPUTS]
        <REF> area </REF>: float "Calculated annulus area; may be negative if inner_radius is greater than outer_radius"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the annulus area by multiplying math.pi by the difference between the square of outer_radius and the square of inner_radius, then return the computed value. RESULT area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
