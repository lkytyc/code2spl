# CLASS DataStatistics

[DEFINE_WORKER: "Computes the arithmetic mean of the input data and returns the result rounded to two decimal places." mean]
    [INPUTS]
        <REF> data </REF>: iterable of numeric values "A non-empty sized iterable collection of numeric or addable values whose average is computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The mean of data rounded to two decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sum of all elements in data. RESULT total_sum]
            [COMMAND Compute the number of elements in data. RESULT data_count]
            [COMMAND Divide total_sum by data_count to obtain the average. RESULT average]
            [COMMAND Round average to 2 decimal places. RESULT rounded_average]
            [COMMAND Return rounded_average. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the median of an input dataset by sorting it and returning the middle element for odd lengths or the average of the two middle elements rounded to 2 decimal places for even lengths." median]
    [INPUTS]
        <REF> data </REF>: iterable "Input iterable of comparable elements whose median is computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> median </REF>: number "The median value: middle element for odd length, or rounded average of the two middle elements for even length."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the input data in ascending order and store the result in sorted_data. RESULT sorted_data]
            [COMMAND Set n to the number of elements in sorted_data. RESULT n]
            [COMMAND If n is even, set middle to the integer quotient of n divided by 2, compute the average of sorted_data[middle - 1] and sorted_data[middle], round the average to 2 decimal places, and return it. RESULT median]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n % 2 != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set middle to the integer quotient of n divided by 2. RESULT middle]
            [COMMAND Return sorted_data[middle]. RESULT median]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the mode(s) of the input data by counting element frequencies and returning all values that share the maximum frequency." mode]
    [INPUTS]
        <REF> data </REF>: iterable "Input iterable whose elements are counted to determine the mode."
    [END_INPUTS]

    [OUTPUTS]
        <REF> mode </REF>: list "List of all values in data that have the highest frequency; if only one value has the highest frequency, the list contains that single value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a frequency mapping counter from data using Counter(data). RESULT counter]
            [COMMAND Compute mode_count as the maximum value among counter.values(). If data is empty, counter.values() is empty and this step raises ValueError. RESULT mode_count]
            [COMMAND Within the list comprehension over counter.items(), collect each key x where count equals mode_count. RESULT mode]
            [COMMAND Return mode. RESULT mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
