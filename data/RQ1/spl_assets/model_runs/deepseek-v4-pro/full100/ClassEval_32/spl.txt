[DEFINE_WORKER: "Decrypts a Caesar cipher by shifting each alphabetic character backward by the given shift within its ASCII case range while leaving non-alphabetic characters unchanged." caesar_decipher]
    [INPUTS]
        <REF> ciphertext </REF>: str "The encrypted text to be decrypted."
        <REF> shift </REF>: int "The number of positions to shift each alphabetic character backward."
    [END_INPUTS]

    [OUTPUTS]
        <REF> plaintext </REF>: str "The resulting plaintext after applying the Caesar shift to all alphabetic characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize plaintext as an empty string. RESULT plaintext]
            [COMMAND For each character char in ciphertext, processed from first to last, execute exactly one of the alternative_flows branch paths based on the current char conditions. RESULT char]
            [COMMAND After all characters in ciphertext have been processed, return plaintext. RESULT plaintext]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: For the current character char in the outer for loop over ciphertext, char.isalpha() is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append char to plaintext unchanged. RESULT plaintext]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: For the current character char in the outer for loop over ciphertext, char.isalpha() is true and char.isupper() is true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set ascii_offset to 65. RESULT ascii_offset]
            [COMMAND Compute shifted_char as chr((ord(char) - ascii_offset - shift) % 26 + ascii_offset). RESULT shifted_char]
            [COMMAND Append shifted_char to plaintext. RESULT plaintext]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: For the current character char in the outer for loop over ciphertext, char.isalpha() is true and char.isupper() is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set ascii_offset to 97. RESULT ascii_offset]
            [COMMAND Compute shifted_char as chr((ord(char) - ascii_offset - shift) % 26 + ascii_offset). RESULT shifted_char]
            [COMMAND Append shifted_char to plaintext. RESULT plaintext]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypts ciphertext using a repeating-key Vigenere cipher, preserving non-alphabetic characters and letter case while advancing the key only for alphabetic characters." vigenere_decipher]
    [INPUTS]
        <REF> ciphertext </REF>: str "The encrypted text to decrypt."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decrypted_text </REF>: str "The decrypted plaintext string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize decrypted_text to an empty string. RESULT decrypted_text]
            [COMMAND Initialize key_index to integer 0. RESULT key_index]
            [COMMAND For each char in ciphertext, process characters sequentially from first to last. RESULT char]
            [COMMAND Within the for loop over ciphertext, if char.isalpha() is true, compute shift as ord(self.key[key_index % len(self.key)].lower()) - ord('a'). RESULT shift]
            [COMMAND Still inside the if-char.isalpha() true branch, compute decrypted_char as chr((ord(char.lower()) - ord('a') - shift) % 26 + ord('a')). RESULT decrypted_char]
            [COMMAND Still inside the if-char.isalpha() true branch, append decrypted_char.upper() to decrypted_text if char.isupper() is true, otherwise append decrypted_char. RESULT decrypted_text]
            [COMMAND Still inside the if-char.isalpha() true branch, increment key_index by 1. RESULT key_index]
            [COMMAND After processing all characters in ciphertext, return decrypted_text. RESULT decrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over ciphertext, when char.isalpha() is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append char unchanged to decrypted_text and do not increment key_index. RESULT decrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypts a rail fence ciphertext by marking zigzag rail positions in an empty grid, filling marked cells row-major with ciphertext characters, and reading the zigzag path to reconstruct plaintext." rail_fence_decipher]
    [INPUTS]
        <REF> encrypted_text </REF>: string "The ciphertext string to be decrypted."
        <REF> rails </REF>: int "The number of rails used in the rail fence encryption pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> plain_text </REF>: string "The decrypted plaintext string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize fence as a 2D list with rails rows and len(encrypted_text) columns, setting every element to the newline character. RESULT fence]
            [COMMAND Set direction to -1, row to 0, and col to 0. RESULT movement_state]
            [COMMAND For each column position from 0 through len(encrypted_text)-1 inclusive: check if row equals 0 or row equals rails - 1 using left-to-right short-circuit OR; if true, replace direction with -direction; then unconditionally set fence[row][col] to the empty string, increment col by 1, and add direction to row. RESULT fence]
            [COMMAND Set index to 0. RESULT index]
            [COMMAND For each row i from 0 through rails - 1 inclusive: for each column j from 0 through len(encrypted_text)-1 inclusive: if fence[i][j] equals the empty string, assign encrypted_text[index] to fence[i][j], then increment index by 1. RESULT fence]
            [COMMAND Initialize plain_text to an empty string, reset direction to -1, and reset row and col to 0. RESULT plain_text_state]
            [COMMAND For each column position from 0 through len(encrypted_text)-1 inclusive: check if row equals 0 or row equals rails - 1 using left-to-right short-circuit OR; if true, replace direction with -direction; then unconditionally append fence[row][col] to plain_text, increment col by 1, and add direction to row. RESULT plain_text]
            [COMMAND Return plain_text. RESULT plain_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
