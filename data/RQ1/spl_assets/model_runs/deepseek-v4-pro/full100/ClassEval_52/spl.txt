[DEFINE_WORKER: "Lemmatizes an input sentence by removing punctuation, tokenizing, part-of-speech tagging, and applying part-of-speech-aware lemmatization with a default fallback before returning the list of lemmatized words." lemmatize_sentence]
    [INPUTS]
        <REF> sentence </REF>: string "The input sentence to be cleaned, tokenized, tagged, and lemmatized."
    [END_INPUTS]

    [OUTPUTS]
        <REF> lemmatized_words </REF>: list "The ordered list of lemmatized words produced from the input sentence."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize lemmatized_words as an empty list to collect lemmatized tokens. RESULT lemmatized_words]
            [COMMAND Call self.remove_punctuation with the current sentence and assign the cleaned result back to sentence. RESULT sentence]
            [COMMAND Call word_tokenize on the cleaned sentence and assign the resulting token list to words. RESULT words]
            [COMMAND Call pos_tag on words and assign the resulting tagged token list to tagged_words. RESULT tagged_words]
            [COMMAND For each word and tag in tagged_words, apply the first matching branch exactly once: if tag starts with uppercase 'V', set lemmatized_word to self.lemmatizer.lemmatize(word, pos='v'); elif tag starts with uppercase 'J', set lemmatized_word to self.lemmatizer.lemmatize(word, pos='a'); elif tag starts with uppercase 'R', set lemmatized_word to self.lemmatizer.lemmatize(word, pos='r'); otherwise set lemmatized_word to self.lemmatizer.lemmatize(word). Then append lemmatized_word to lemmatized_words. RESULT lemmatized_words]
            [COMMAND Return the accumulated lemmatized_words list. RESULT lemmatized_words]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Cleans the input sentence by removing punctuation, tokenizes it, assigns part-of-speech tags, and returns the list of POS tags." get_pos_tag]
    [INPUTS]
        <REF> sentence </REF>: str "Input sentence to clean, tokenize, and POS-tag."
    [END_INPUTS]

    [OUTPUTS]
        <REF> pos_tags </REF>: list[str] "List of part-of-speech tags extracted from the tagged tokens."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize pos_tags to an empty list. RESULT pos_tags]
            [COMMAND Set sentence to the result of self.remove_punctuation(sentence); this may leave sentence empty. RESULT sentence]
            [COMMAND Set words to word_tokenize(sentence). RESULT words]
            [COMMAND Set tagged_words to pos_tag(words). RESULT tagged_words]
            [COMMAND For each tagged_word in tagged_words, append tagged_word[1] to pos_tags. RESULT pos_tags]
            [COMMAND Return pos_tags. RESULT pos_tags]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes all punctuation characters from the input string and returns the cleaned string." remove_punctuation]
    [INPUTS]
        <REF> sentence </REF>: str "Input string from which punctuation characters are removed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> cleaned_sentence </REF>: str "A copy of the input string with all punctuation characters removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a translation table that performs no character replacements and deletes every character in string.punctuation. RESULT translation_table]
            [COMMAND Apply the translation table to sentence using the translate method to remove all punctuation characters. RESULT cleaned_sentence]
            [COMMAND Return the cleaned sentence. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
