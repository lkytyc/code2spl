# CLASS BookManagement

[DEFINE_WORKER: "Initializes the object by setting its inventory attribute to a new empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary literal to the attribute inventory on self. RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the inventory by increasing the quantity for an existing title or adding a new title with the given quantity." add_book]
    [INPUTS]
        <REF> title </REF>: string "The title of the book to add or update."
        <REF> quantity </REF>: integer "The quantity to add; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If title is in self.inventory, increase self.inventory[title] by quantity. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title is not in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.inventory[title] to quantity. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a requested quantity of a book from inventory and deletes the title entry when its stock reaches zero, raising an error if the title is missing or stock is insufficient." remove_book]
    [INPUTS]
        <REF> title </REF>: str "Book title to remove from inventory."
        <REF> quantity </REF>: int "Number of copies to remove."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check the inventory guard condition: title must be present in self.inventory and self.inventory[title] must be greater than or equal to quantity; if either part fails, follow the exception flow. RESULT availability_check]
            [COMMAND Decrease the stock count self.inventory[title] by quantity. RESULT remaining_stock]
            [COMMAND After decrementing, if self.inventory[title] is exactly 0, delete the entry for title from self.inventory. RESULT zero_stock_cleanup]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: title is not in self.inventory or self.inventory[title] is less than quantity; the right-hand quantity check is evaluated only if the title exists due to short-circuit evaluation.]
        [LOG "Insufficient inventory: title is missing or available stock is less than requested quantity; raise False."]
        [THROW False "Conceptually raised when the requested title is absent or the available quantity is strictly less than the requested quantity; the original code uses raise False."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the inventory attribute of the current instance without modification." view_inventory]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "The exact object or value currently stored in self.inventory, returned as a reference or value without copy or transformation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the current value of the instance attribute `inventory` from `self` directly, without copying or transforming it. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the inventory quantity for a given title, defaulting to 0 when the title is not present." view_book_quantity]
    [INPUTS]
        <REF> self </REF>: object "Current instance that owns the inventory mapping."
        <REF> title </REF>: string "Title key whose inventory quantity is requested."
    [END_INPUTS]

    [OUTPUTS]
        <REF> quantity </REF>: integer "The stored inventory quantity for the title, or 0 if the title is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check if title is not present as a key in self.inventory. RESULT title_missing]
            [COMMAND If title_missing is false, return the value stored in self.inventory under the key title. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title_missing is true, meaning title is not present in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately and exit the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
