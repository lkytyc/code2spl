# CLASS BalancedBrackets

[DEFINE_WORKER: "Initializes the instance by resetting the bracket stack to empty, defining the opening and closing bracket character sets, and storing the supplied expression." __init__]
    [INPUTS]
        <REF> expr </REF>: text "Expression or text passed into the constructor and stored as self.expr."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.stack to a new empty list to serve as the bracket matching stack. RESULT self.stack]
            [COMMAND Set self.left_brackets to the opening bracket characters '(' , '{' , and '['. RESULT self.left_brackets]
            [COMMAND Set self.right_brackets to the closing bracket characters ')' , '}' , and ']'. RESULT self.right_brackets]
            [COMMAND Assign self.expr to reference the same value as the supplied expr argument. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters self.expr to retain only characters that are present in either self.left_brackets or self.right_brackets." clear_expr]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign to self.expr the result of joining every character c from the old self.expr where c is in self.left_brackets or c is in self.right_brackets, preserving original character order and duplicates; for each character, the self.right_brackets membership test is evaluated only when the self.left_brackets membership test is false. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the brackets in self.expr are balanced by clearing previous expression parsing state, scanning each bracket to push opening brackets onto a stack and matching closing brackets against the stack in LIFO order, returning False on any mismatch or leftover opening bracket and True otherwise." check_balanced_brackets]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> balanced </REF>: boolean "True when every closing bracket matches the top of the stack and no opening brackets remain after scanning; False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.clear_expr() to clear any previous expression parsing state. RESULT clear_status]
            [COMMAND For each Brkt in self.expr, evaluate whether Brkt is in self.left_brackets. RESULT bracket_check]
            [COMMAND Within the for loop over self.expr, if Brkt is in self.left_brackets, append Brkt to self.stack. RESULT push_result]
            [COMMAND Within the for loop over self.expr, in the else branch where Brkt is not in self.left_brackets, pop the top value from self.stack into Current_Brkt. RESULT popped_bracket]
            [COMMAND Within the for loop over self.expr, in the else branch, if Current_Brkt is '(' and Brkt is not ')', return False. RESULT paren_mismatch_result]
            [COMMAND Within the for loop over self.expr, in the else branch, if Current_Brkt is '{' and Brkt is not '}', return False. RESULT brace_mismatch_result]
            [COMMAND Within the for loop over self.expr, in the else branch, if Current_Brkt is '[' and Brkt is not ']', return False. RESULT square_mismatch_result]
            [COMMAND After the for loop completes, if self.stack is not empty, return False. RESULT leftover_stack_result]
            [COMMAND Return True because no mismatch occurred and self.stack is empty. RESULT balanced_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
