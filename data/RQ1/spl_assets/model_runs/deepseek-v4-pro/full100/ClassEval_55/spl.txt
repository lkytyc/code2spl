# CLASS Manacher

    [INPUTS]
        <REF> input_string </REF>: string "The string value to assign to the instance attribute input_string."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input_string parameter to the instance attribute self.input_string. RESULT self.input_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Recursively counts how many symmetric character pairs around a center match before reaching a string boundary or encountering a mismatch." palindromic_length]
    [INPUTS]
        <REF> center </REF>: int "Center index around which palindrome expansion is checked."
        <REF> diff </REF>: int "Current expansion distance from the center."
        <REF> string </REF>: str "Input string whose mirrored characters are compared."
    [END_INPUTS]

    [OUTPUTS]
        <REF> palindromic_length_result </REF>: int "Number of valid symmetric character pairs found before a boundary or mismatch; returns 0 when the current expansion is invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the early-exit guard using left-to-right short-circuit evaluation: center - diff == -1, center + diff == len(string), or string[center - diff] != string[center + diff]. RESULT early_exit_triggered]
            [COMMAND If early_exit_triggered is false, return 1 plus palindromic_length(center, diff + 1, string). RESULT palindromic_length_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: early_exit_triggered is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT palindromic_length_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a pipe-separated version of the input string, scans each index as an odd-length palindrome center, and returns the longest palindromic substring with pipe separators removed." palindromic_string]
    [INPUTS]
        <REF> self.input_string </REF>: string "The input string whose longest palindromic substring is to be found."
    [END_INPUTS]

    [OUTPUTS]
        <REF> output_string </REF>: string "The longest palindromic substring found after removing pipe separator characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize max_length to 0, new_input_string to an empty string, and output_string to an empty string. RESULT max_length, new_input_string, output_string]
            [COMMAND For each character i in self.input_string[:len(self.input_string) - 1], append i followed by "|" to new_input_string; then append the final character self.input_string[-1] to new_input_string. RESULT new_input_string]
            [COMMAND For each index i from 0 through len(new_input_string) - 1, compute length = self.palindromic_length(i, 1, new_input_string); if max_length < length, set max_length = length and start = i. RESULT max_length, start]
            [COMMAND For each character i in the slice new_input_string[start - max_length:start + max_length + 1], with inclusive start at start - max_length and exclusive stop at start + max_length + 1, if i != "|", append i to output_string. RESULT output_string]
            [COMMAND Return output_string. RESULT output_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
