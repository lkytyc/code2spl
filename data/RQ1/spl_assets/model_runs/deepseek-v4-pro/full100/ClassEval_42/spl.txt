# CLASS Hotel

[DEFINE_WORKER: "Initializes an instance by storing the provided name and available rooms dictionary, and creating an empty booked rooms dictionary." __init__]
    [INPUTS]
        <REF> name </REF>: any "The name value to assign directly to the instance, with no validation applied."
        <REF> rooms </REF>: dict "A dictionary mapping room types to available counts, assigned directly to the instance without copying."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the caller-supplied `name` value directly to the instance attribute `self.name`. RESULT self.name]
            [COMMAND Assign the caller-supplied `rooms` dictionary directly to the instance attribute `self.available_rooms`, preserving the original reference. RESULT self.available_rooms]
            [COMMAND Create a new empty dictionary and assign it to the instance attribute `self.booked_rooms`. RESULT self.booked_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Books a requested number of rooms for a guest by validating the room type and available count, updating the booked and available room dictionaries, and returning 'Success!', the remaining available count, or False depending on availability." book_room]
    [INPUTS]
        <REF> room_type </REF>: string "The type of room to book."
        <REF> room_number </REF>: integer "The number of rooms requested for the booking."
        <REF> name </REF>: string "The guest name associated with the booking."
    [END_INPUTS]

    [OUTPUTS]
        <REF> booking_result </REF>: string|integer|boolean "Returns 'Success!' when the booking succeeds, an integer remaining room count when requested rooms exceed availability but some rooms remain, or False when the room type is unavailable or no rooms are available."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether room_type exists as a key in self.available_rooms.keys(). RESULT room_type_available]
            [COMMAND If room_type_available is false, return False immediately. RESULT return_false_unavailable_type]
            [COMMAND Check whether room_number is less than or equal to self.available_rooms[room_type]. RESULT sufficient_rooms]
            [COMMAND If sufficient_rooms is false, follow the alternative flow matching the insufficiency condition. RESULT insufficient_availability_branch]
            [COMMAND Within the outer if branch where sufficient_rooms is true, check whether room_type is not present in self.booked_rooms.keys(). RESULT missing_booking_dict]
            [COMMAND If missing_booking_dict is true, initialize self.booked_rooms[room_type] to an empty dictionary. RESULT booking_dict_initialized]
            [COMMAND Within the outer if branch where sufficient_rooms is true, assign self.booked_rooms[room_type][name] to room_number. RESULT booking_recorded]
            [COMMAND Within the outer if branch where sufficient_rooms is true, subtract room_number from self.available_rooms[room_type]. RESULT available_rooms_updated]
            [COMMAND Return "Success!". RESULT return_success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If room_number is greater than self.available_rooms[room_type] and self.available_rooms[room_type] is not 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.available_rooms[room_type] as the number of available rooms. RESULT return_remaining_available]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If room_number is greater than self.available_rooms[room_type] and self.available_rooms[room_type] is 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_false_no_available_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks in a guest by validating the requested room type and existing booking, then returns False for invalid or overbooked requests, removes the booking for an exact full check-in, or reduces the booked room count for a partial check-in." check_in]
    [INPUTS]
        <REF> room_type </REF>: str "The room type to check in."
        <REF> room_number </REF>: int "The number of rooms requested for check-in."
        <REF> name </REF>: str "The guest name associated with the booking."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool_or_none "Returns False if the room type is unknown or if room_number exceeds the booked count; otherwise returns None after updating or removing the booking."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If room_type is not in self.booked_rooms.keys(), return False immediately. RESULT invalid_room_type_early_return]
            [COMMAND Check whether name is present in self.booked_rooms[room_type]. RESULT name_booking_exists]
            [COMMAND If name_booking_exists is false, return None without modifying self.booked_rooms. RESULT missing_name_no_modification]
            [COMMAND Within the outer if block for name_booking_exists, compare room_number to self.booked_rooms[room_type][name]. RESULT room_count_comparison]
            [COMMAND If room_number is greater than self.booked_rooms[room_type][name], return False immediately. RESULT overbooking_early_return]
            [COMMAND If room_number is equal to self.booked_rooms[room_type][name], remove the full booking by calling self.booked_rooms[room_type].pop(name). RESULT full_check_in_removal]
            [COMMAND If room_number is less than self.booked_rooms[room_type][name], reduce the booked count by setting self.booked_rooms[room_type][name] to self.booked_rooms[room_type][name] minus room_number. RESULT partial_check_in_reduction]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the available rooms dictionary by increasing the count for an existing room type or initializing a new room type with the given room number." check_out]
    [INPUTS]
        <REF> room_type </REF>: string "The room type key to update in the available rooms dictionary."
        <REF> room_number </REF>: integer "The number of rooms to add to an existing room type or to initialize a new room type with."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether room_type exists as a key in self.available_rooms. RESULT room_type_exists]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: room_type_exists is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.available_rooms[room_type] by room_number. RESULT updated_available_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: room_type_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.available_rooms[room_type] to room_number. RESULT initialized_available_rooms]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the available rooms value associated with the given room type from the instance's available_rooms mapping." get_available_rooms]
    [INPUTS]
        <REF> room_type </REF>: object "Key used to look up available rooms in the available_rooms mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "Value stored in available_rooms under the given room_type key."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value from self.available_rooms for the key room_type. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
