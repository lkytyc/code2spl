[DEFINE_WORKER: "Validates that mode is either 'heat' or 'cool', assigns it to self.mode when valid, and returns False otherwise." set_mode]
    [INPUTS]
        <REF> mode </REF>: string "The mode value to validate; valid values are 'heat' and 'cool'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "Returns False when mode is not one of the allowed values; otherwise the method completes without an explicit return value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether mode is exactly 'heat' or exactly 'cool' using inclusive membership in ['heat', 'cool']. RESULT is_valid_mode]
            [COMMAND If is_valid_mode is true, assign mode to self.mode. RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: mode is not in ['heat', 'cool']]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without assigning self.mode. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sets self.mode to 'heat' when the current temperature is below the target temperature; otherwise sets it to 'cool'." auto_set_mode]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.current_temperature is less than self.target_temperature. RESULT current_below_target]
            [COMMAND If current_below_target is true, set self.mode to 'heat'. RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: current_below_target is false, meaning self.current_temperature is greater than or equal to self.target_temperature including equality.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.mode to 'cool'. RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether the current mode already matches the required heating or cooling action based on current_temperature relative to target_temperature, returning True when it matches and otherwise calling auto_set_mode and returning False." auto_check_conflict]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> check_result </REF>: bool "True when the current mode already matches the required heating or cooling state; False when auto_set_mode is called and the mode is changed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.current_temperature is strictly greater than self.target_temperature; treat equality as not greater and use this to select the outer branch. RESULT outer_temperature_branch]
            [COMMAND Within the outer true branch (self.current_temperature > self.target_temperature), evaluate whether self.mode is exactly equal to the string 'cool'. RESULT inner_cool_mode_branch]
            [COMMAND Within the outer true branch and inner true branch (self.mode == 'cool'), return True immediately and do not call self.auto_set_mode. RESULT return_true_hot_cool_path]
            [COMMAND Within the outer true branch and inner false branch (self.mode != 'cool'), call self.auto_set_mode(), then return False immediately. RESULT return_false_hot_non_cool_path]
            [COMMAND Within the outer false branch (self.current_temperature <= self.target_temperature), evaluate whether self.mode is exactly equal to the string 'heat'. RESULT inner_heat_mode_branch]
            [COMMAND Within the outer false branch and inner true branch (self.mode == 'heat'), return True immediately and do not call self.auto_set_mode. RESULT return_true_not_hot_heat_path]
            [COMMAND Within the outer false branch and inner false branch (self.mode != 'heat'), call self.auto_set_mode(), then return False immediately. RESULT return_false_not_hot_non_heat_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Simulates heating or cooling operation by adjusting current temperature toward the target temperature in unit steps and returns the total number of adjustment steps taken." simulate_operation]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> use_time </REF>: integer "Total number of unit temperature adjustment steps performed during the simulation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.auto_set_mode() to automatically set the operation mode. RESULT self.mode]
            [COMMAND Initialize use_time to 0. RESULT use_time]
            [COMMAND If self.mode equals 'heat', while self.current_temperature is strictly less than self.target_temperature, increment self.current_temperature by 1 and increment use_time by 1. RESULT use_time]
            [COMMAND Return use_time. RESULT use_time]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.mode is not equal to 'heat'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND While self.current_temperature is strictly greater than self.target_temperature, decrement self.current_temperature by 1 and increment use_time by 1. RESULT use_time]
            [COMMAND Return use_time. RESULT use_time]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
