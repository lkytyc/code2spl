# CLASS Order

[DEFINE_WORKER: "Initializes an instance by setting menu and selected_dishes to empty lists and sales to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty list to self.menu, intended to store dish records with keys dish, price, and count. RESULT self.menu]
            [COMMAND Assign an empty list to self.selected_dishes, intended to store selected dish records with keys dish, count, and price. RESULT self.selected_dishes]
            [COMMAND Assign an empty dictionary to self.sales, with no initial sales records. RESULT self.sales]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deducts the requested count from the first matching menu dish if stock is sufficient, appends the dish to selected_dishes, and returns True; returns False if stock is insufficient." add_dish]
    [INPUTS]
        <REF> dish </REF>: object "The dish to add, containing keys dish and count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the dish was appended successfully; False when the matching menu dish has insufficient stock."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over self.menu and compare dish["dish"] with each menu_dish["dish"] to find the first matching menu item. RESULT matching_menu_dish]
            [COMMAND Within the for loop over self.menu, when a matching menu_dish is found and menu_dish["count"] is greater than or equal to dish["count"], subtract dish["count"] from menu_dish["count"] and break out of the loop. RESULT consume_stock]
            [COMMAND After the for loop completes or breaks without an early return, append dish to self.selected_dishes. RESULT append_dish]
            [COMMAND Return True. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the for loop over self.menu, a matching menu_dish is found and menu_dish["count"] is less than dish["count"].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.menu or self.selected_dishes. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate the total cost of selected dishes by summing each dish's price multiplied by its count and its sales rate." calculate_total]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> total </REF>: number "The total cost of selected dishes after applying each dish's price, count, and sales rate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND For each dish in self.selected_dishes, compute dish['price'] * dish['count'] * self.sales[dish['dish']] and add the product to total. RESULT total]
            [COMMAND Return the final accumulated total as the method result. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks out the selected dishes by returning False when none are selected, otherwise calculating the total, clearing the selected dishes, and returning the total." checkout]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean or numeric "False when selected_dishes is empty; otherwise the total returned by calculate_total() after selected_dishes is cleared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND When self.selected_dishes is not empty, call self.calculate_total() and assign the returned value to total. RESULT total]
            [COMMAND Assign self.selected_dishes to a new empty list. RESULT self.selected_dishes]
            [COMMAND Return total. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Length of self.selected_dishes is zero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
