# CLASS ClassRegistrationSystem

[DEFINE_WORKER: "Initializes instance state by setting students to a new empty list and students_registration_classes to a new empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.students to a newly allocated empty list. RESULT self.students]
            [COMMAND Initialize self.students_registration_classes to a newly allocated empty dictionary. RESULT self.students_registration_classes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a student by adding them to the internal student collection only if they are not already present, returning 1 for a successful addition and 0 for a duplicate." register_student]
    [INPUTS]
        <REF> student </REF>: object "The student to register."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_code </REF>: int "Returns 1 if the student was appended to the collection, or 0 if the student already existed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether student is already present in self.students. RESULT already_exists]
            [COMMAND If already_exists is false, append student to self.students. RESULT append_result]
            [COMMAND If already_exists is false, return 1. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: already_exists is true because student is already present in self.students.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 without modifying self.students. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a student for a class by appending the class to the student's existing registration list or creating a new list if the student is not yet registered, then returns the updated registration list." register_class]
    [INPUTS]
        <REF> student_name </REF>: string "Identifier of the student to register."
        <REF> class_name </REF>: string "Identifier of the class to add to the student's registration list."
    [END_INPUTS]

    [OUTPUTS]
        <REF> registration_list </REF>: list "The updated list of class names registered for the student."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether student_name exists as a key in self.students_registration_classes. RESULT student_exists]
            [COMMAND If student_exists is true, append class_name to self.students_registration_classes[student_name]; otherwise, assign self.students_registration_classes[student_name] to a new list containing only class_name. RESULT updated_registration_list]
            [COMMAND Return self.students_registration_classes[student_name]. RESULT registration_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects and returns the names of students whose major exactly equals the given major parameter, preserving the original student iteration order." get_students_by_major]
    [INPUTS]
        <REF> major </REF>: string "Target major value used to filter students by exact equality."
    [END_INPUTS]

    [OUTPUTS]
        <REF> student_list </REF>: list "A list of student names whose major equals the given major parameter; empty if no matching students exist."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named student_list. RESULT student_list]
            [COMMAND Iterate over each student in self.students in the original order. RESULT student]
            [COMMAND Within the iteration over self.students, if student["major"] is exactly equal to major, append student["name"] to student_list. RESULT student_list]
            [COMMAND Return the accumulated student_list. RESULT student_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds and returns a list of unique student majors from self.students, preserving first-seen order and removing duplicates." get_all_major]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> major_list </REF>: list "A list of unique student majors in first-seen order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named major_list. RESULT major_list]
            [COMMAND For each student in self.students, execute the nested logic in the following step. RESULT student]
            [COMMAND Within the for loop over self.students, if student["major"] is not present in major_list, append student["major"] to major_list. RESULT major_list]
            [COMMAND After the loop completes, return major_list. RESULT major_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the most frequently registered class among students whose major equals the given major." get_most_popular_class_in_major]
    [INPUTS]
        <REF> major </REF>: string "The major value used to filter students."
    [END_INPUTS]

    [OUTPUTS]
        <REF> most_popular_class </REF>: string "The class with the highest registration count among matching students."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize class_list as an empty list. RESULT class_list]
            [COMMAND For each student in self.students in iteration order, if student["major"] exactly equals major, extend class_list with all elements from self.students_registration_classes[student["name"]]; missing "major" or "name" keys raise KeyError. RESULT class_list]
            [COMMAND After the loop completes, create unique_classes as set(class_list). RESULT unique_classes]
            [COMMAND Compute most_popular_class as max(unique_classes, key=class_list.count). If class_list is empty, max raises ValueError. If multiple classes tie for the highest count, max returns the first one encountered in the iteration order of unique_classes. RESULT most_popular_class]
            [COMMAND Return most_popular_class. RESULT most_popular_class]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
