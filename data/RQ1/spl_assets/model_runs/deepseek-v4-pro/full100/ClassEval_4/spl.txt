[DEFINE_WORKER: "Adds or updates a student record in self.students keyed by name, storing the provided grade and major and initializing an empty courses dictionary." add_student]
    [INPUTS]
        <REF> name </REF>: any "Student name used as the key in self.students and stored in the student record."
        <REF> grade </REF>: any "Grade value stored in the student record."
        <REF> major </REF>: any "Major value stored in the student record."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign to self.students[name] a newly created dictionary with 'name' set to name, 'grade' set to grade, 'major' set to major, and 'courses' set to a new empty dictionary. This creates a new entry if name is absent and replaces the existing entry if name already exists. RESULT assigned_student_record]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates or adds a course score for an existing student in the self.students dictionary." add_course_score]
    [INPUTS]
        <REF> name </REF>: str "Student name used as a key in self.students."
        <REF> course </REF>: str "Course name used as a key in the student's courses mapping."
        <REF> score </REF>: number "Score value to assign to the course."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether name exists as a key in self.students. RESULT name_exists]
            [COMMAND If name_exists is true, assign score to self.students[name]['courses'][course]. RESULT updated_course_score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: name_exists is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the assignment and leave self.students unchanged. RESULT no_update]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the average course score for a student if the student exists and has a non-empty courses collection, otherwise returns None." get_gpa]
    [INPUTS]
        <REF> name </REF>: string "The name of the student to look up in self.students."
    [END_INPUTS]

    [OUTPUTS]
        <REF> gpa_result </REF>: float or None "The average of the student's course scores when available, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name exists as a key in self.students. RESULT student_exists]
            [COMMAND If student_exists is true, check whether self.students[name]['courses'] is truthy. RESULT courses_truthy]
            [COMMAND If both checks are true, compute the average course score by summing the values in self.students[name]['courses'] and dividing by the number of course entries, then return that average. RESULT gpa_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: name is not in self.students, or self.students[name]['courses'] is falsy such as empty.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None. RESULT gpa_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a list of student names that have at least one course score strictly below 60." get_all_students_with_fail_course]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> students </REF>: list "A list containing the names of students who have at least one course score below 60."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named students to collect the names of students who have a failing course score. RESULT students]
            [COMMAND For each name and student entry in self.students.items(), execute the following course-score check for that student. RESULT name, student]
            [COMMAND Within the outer loop over name and student, for each course and score entry in student['courses'].items(), execute the following failing-score check; if student['courses'] is empty, skip this inner loop for the current student. RESULT course, score]
            [COMMAND Within the inner loop over course and score, if score is strictly less than 60, append name to the students list and break out of the inner course loop only; continue the outer loop with the next student. RESULT students]
            [COMMAND Return the students list after the outer loop completes; if no student has a failing score, return an empty list. RESULT students]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the average of all non-None scores for a given course across all student records, returning None when no valid scores are found." get_course_average]
    [INPUTS]
        <REF> course </REF>: string "The course identifier used to look up whether a student has a score for that course."
    [END_INPUTS]

    [OUTPUTS]
        <REF> average </REF>: float or None "The arithmetic mean of all valid non-None course scores, or None if there are no valid scores."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0 and count to 0. RESULT total_count_init]
            [COMMAND For each student in self.students.values(), execute the loop body. RESULT student]
            [COMMAND Within the loop over self.students.values(), if course is not in student['courses'], skip the current student and continue with the next student. RESULT skip_student]
            [COMMAND Within the loop over self.students.values(), if course is in student['courses'], retrieve score as student['courses'][course]. RESULT score]
            [COMMAND Within the loop over self.students.values(), if score is not None, add score to total and increment count by 1. RESULT valid_score_accumulation]
            [COMMAND After the loop completes, if count is greater than 0, return total divided by count. RESULT average]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After the loop completes, count is 0 because no valid non-None course score was found.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None. RESULT none_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the name of the first student encountered with the highest GPA greater than 0, ignoring students with None GPAs, or return None if no such GPA exists." get_top_student]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> top_student </REF>: string|null "The name of the student with the highest strictly positive GPA greater than 0, or None if no such student is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize top_student to None. RESULT top_student]
            [COMMAND Initialize top_gpa to 0. RESULT top_gpa]
            [COMMAND For each name and student pair in self.students.items() in dictionary iteration order, call self.get_gpa(name) and store the result in gpa. RESULT gpa]
            [COMMAND Within the for loop over name and student in self.students.items(), if gpa is not None and gpa > top_gpa, set top_gpa to gpa and set top_student to name. RESULT top_gpa]
            [COMMAND Return top_student. RESULT top_student]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
