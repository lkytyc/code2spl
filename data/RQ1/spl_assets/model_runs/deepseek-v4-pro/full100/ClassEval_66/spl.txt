[DEFINE_WORKER: "Decode HTML numeric character references, both decimal and hexadecimal, from the input string into their corresponding Unicode characters, returning the accumulated decoded string." replace]
    [INPUTS]
        <REF> string </REF>: str "Input string that may contain numeric character references such as &#...; or &#x...;."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The joined string of decoded characters and ordinary characters collected before the loop exits; unprocessed trailing characters are not appended."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize `out` to an empty list to collect decoded output characters. RESULT out]
            [COMMAND Set `pos` to 0, the starting index for scanning the input string. RESULT pos]
            [COMMAND Set `length` to `len(string)`, the total number of characters in the input. RESULT length]
            [COMMAND Enter an outer while loop with condition `pos < length - 2`; processing stops when fewer than three characters remain from the current position. RESULT _outer_loop]
            [COMMAND Within the outer while loop, check with short-circuit evaluation whether `string[pos] == '&'` and `string[pos + 1] == '#'`. RESULT _is_entity_start]
            [COMMAND If the current two characters are not `&#`, append `string[pos]` to `out`. RESULT out]
            [COMMAND After appending an ordinary character, increment `pos` by 1 and continue the outer while loop. RESULT pos]
            [COMMAND If the current two characters are `&#`, set `start` to `pos + 2`. RESULT start]
            [COMMAND Still in the entity branch, set `is_hex` to False. RESULT is_hex]
            [COMMAND Still in the entity branch, set `first_char` to `string[start]`. RESULT first_char]
            [COMMAND Still in the entity branch, if `first_char` is `x` or `X`, increment `start` by 1. RESULT start]
            [COMMAND Still in the entity branch, if `first_char` is `x` or `X`, set `is_hex` to True. RESULT is_hex]
            [COMMAND Still in the entity branch, if `start == length`, return `''.join(out)` immediately. RESULT _early_return]
            [COMMAND Set `end` to `start` before scanning the numeric part. RESULT end]
            [COMMAND Enter an inner while loop that continues while `end < length` and `self.is_hex_char(string[end])` is true, incrementing `end` by 1 each iteration. RESULT end]
            [COMMAND After the inner while loop, check with short-circuit evaluation whether `end < length` and `string[end] == ';'`. RESULT _has_semicolon]
            [COMMAND Inside the entity branch, if no semicolon is present after the numeric part, append `string[pos]` to `out`. RESULT out]
            [COMMAND After appending the current `&` for a missing semicolon, increment `pos` by 1 and continue the outer while loop. RESULT pos]
            [COMMAND Inside the entity branch, if a semicolon is present, attempt to convert `string[start:end]` to an integer using base 16 when `is_hex` is True, otherwise base 10, and assign the result to `entity_value`. RESULT entity_value]
            [COMMAND If the integer conversion attempt raises an exception, return `''.join(out)` immediately. RESULT _early_return]
            [COMMAND On successful conversion, append `chr(entity_value)` to `out`. RESULT out]
            [COMMAND After appending the converted character, set `pos` to `end + 1` and continue the outer while loop. RESULT pos]
            [COMMAND When the outer while loop condition `pos < length - 2` becomes false, return `''.join(out)` without appending any unprocessed trailing characters. RESULT _final_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether an input string is a hexadecimal digit or a case-insensitive hexadecimal letter (a-f or A-F) by first testing for a digit and then testing whether the lowercased value falls between 'a' and 'f'." is_hex_char]
    [INPUTS]
        <REF> char </REF>: str "The string to test as a hexadecimal digit or hexadecimal letter; the code does not enforce a single-character length."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "True if char is a digit according to Python's str.isdigit() or its lowercased form is lexicographically between 'a' and 'f' for the chained comparison, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Receive the input parameter `char`. RESULT char]
            [COMMAND Evaluate `char.isdigit()`. If it is True, the `or` expression short-circuits and the method returns True immediately without evaluating `char.lower()` or the hexadecimal letter comparison. RESULT return_value]
            [COMMAND If `char.isdigit()` is False, compute the lowercased string `char.lower()` and evaluate the chained comparison `'a' <= char.lower() <= 'f'`, treating it as the conjunction `('a' <= char.lower()) and (char.lower() <= 'f')`. RESULT hex_letter_check]
            [COMMAND Return True if the hexadecimal letter comparison is True, otherwise return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
