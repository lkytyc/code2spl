# CLASS SQLQueryBuilder

[DEFINE_WORKER: "Builds a SQL SELECT query string from a table name, an optional column list or '*', and optional equality WHERE conditions." select]
    [INPUTS]
        <REF> table </REF>: string "Name of the target table."
        <REF> columns </REF>: string or iterable "Column selection; the string '*' or an iterable of column names."
        <REF> where </REF>: dict or None "Optional mapping of column names to values used to build equality WHERE conditions."
    [END_INPUTS]

    [OUTPUTS]
        <REF> query </REF>: string "The generated SQL SELECT query string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If columns is not equal to the string '*', replace columns with the result of joining the columns iterable using ', '. RESULT columns]
            [COMMAND Construct query as the string 'SELECT {columns} FROM {table}' using the current columns and table values converted to strings. RESULT query]
            [COMMAND If where is truthy, append ' WHERE ' plus the string formed by joining each key='value' item from where.items() with ' AND '. RESULT query]
            [COMMAND Return query. RESULT query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds an SQL INSERT statement by formatting dictionary keys as comma-separated columns and dictionary values as comma-separated single-quoted literals for a given table." insert]
    [INPUTS]
        <REF> table </REF>: str "The target SQL table name."
        <REF> data </REF>: dict "A dictionary-like object whose keys are column names and whose values are row values to insert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> insert_statement </REF>: str "The generated SQL INSERT statement string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Extract the dictionary keys from data using data.keys(). RESULT key_view]
            [COMMAND Join the extracted keys into a single comma-separated string using ', ' as the separator. If data is empty, the result is an empty string; if any key is not a string, the join operation raises TypeError. RESULT keys]
            [COMMAND Extract the dictionary values from data using data.values(). RESULT value_view]
            [COMMAND For each value v in value_view, format it as a single-quoted string by surrounding v with single quotes without escaping any single quotes inside v. RESULT quoted_values]
            [COMMAND Join the quoted values into a single comma-separated string using ', ' as the separator. If data is empty, the result is an empty string. RESULT values]
            [COMMAND Return an INSERT statement string composed from the given table, the joined keys as the column list, and the joined quoted values as the value list. RESULT insert_statement]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a DELETE SQL statement from a table name and optional WHERE equality conditions." delete]
    [INPUTS]
        <REF> table </REF>: str "Target table name for the DELETE statement."
        <REF> where </REF>: dict "Optional dictionary of column names to values used as equality filters in the WHERE clause; defaults to None."
    [END_INPUTS]

    [OUTPUTS]
        <REF> query </REF>: str "The constructed DELETE SQL statement."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign query to the DELETE FROM statement formed by interpolating table into the SQL text. RESULT query]
            [COMMAND If where is truthy, append ' WHERE ' plus the conjunction of key='value' pairs from where.items() joined by ' AND ' to query. RESULT query]
            [COMMAND Return query. RESULT query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds and returns a SQL UPDATE statement from a table and data mapping, optionally adding a WHERE clause from a where mapping." update]
    [INPUTS]
        <REF> table </REF>: str "Target table name inserted directly into the SQL UPDATE statement."
        <REF> data </REF>: dict "Mapping of column names to values used to construct the SET clause; values are formatted directly without escaping."
        <REF> where </REF>: dict "Optional mapping of column names to values used to construct the WHERE clause. Defaults to None. If falsy, no WHERE clause is added."
    [END_INPUTS]

    [OUTPUTS]
        <REF> query </REF>: str "Generated SQL UPDATE statement."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct update_str by formatting each data item in iteration order as key='value' and joining the fragments with ', '; if data is empty, update_str is an empty string. RESULT update_str]
            [COMMAND Build query as 'UPDATE {table} SET {update_str}' using the provided table and update_str. RESULT query]
            [COMMAND If where is truthy, append ' WHERE ' plus the fragments from formatting each where item in iteration order as key='value' joined by ' AND ' to query. RESULT query]
            [COMMAND Return query. RESULT query]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
