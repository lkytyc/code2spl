# CLASS BookManagementDB

[DEFINE_WORKER: "Initialize a SQLite database connection and cursor from db_name, then call create_table to set up the database schema." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "SQLite database file name or ':memory:' for an in-memory database."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open an SQLite database using db_name and assign the resulting connection object to self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection and assign it to self.cursor. RESULT self.cursor]
            [COMMAND Call self.create_table() to create the required database table schema. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates the books table if it does not already exist with columns id, title, author, and available, then commits the transaction." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.cursor.execute with the SQL DDL statement CREATE TABLE IF NOT EXISTS books (id INTEGER PRIMARY KEY, title TEXT, author TEXT, available INTEGER). RESULT cursor_execute_result]
            [COMMAND Call self.connection.commit to persist the current transaction. RESULT connection_commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the books table already exists when self.cursor.execute runs]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the CREATE TABLE schema change and continue to the commit step as a no-op for table creation. RESULT no_op_create_table]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert a new book record with the given title and author, mark it as available, and commit the database transaction." add_book]
    [INPUTS]
        <REF> title </REF>: str "Book title to insert into the books table."
        <REF> author </REF>: str "Book author to insert into the books table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized INSERT statement on self.cursor that inserts the provided title, the provided author, and the constant availability value 1 into the books table, binding title to the first placeholder and author to the second placeholder. RESULT insert_execution]
            [COMMAND After the INSERT statement completes successfully, call self.connection.commit() to persist the pending transaction. RESULT commit_result]
            [COMMAND Return None implicitly after the commit completes. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes a book record by its ID from the books table and commits the transaction." remove_book]
    [INPUTS]
        <REF> book_id </REF>: any "Identifier of the book row to delete."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.cursor.execute with the SQL text 'DELETE FROM books WHERE id = ?' and the parameter tuple (book_id,). RESULT delete_result]
            [COMMAND Call self.connection.commit() to persist the deletion in the current transaction. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as unavailable by setting its available column to 0 and commits the database transaction." borrow_book]
    [INPUTS]
        <REF> book_id </REF>: int "Identifier of the book record to mark as unavailable."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL UPDATE on self.cursor, setting available to 0 for the row where id equals the provided book_id. RESULT update_execution]
            [COMMAND Commit the current database transaction on self.connection. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as available by updating its available column to 1 in the database and committing the transaction." return_book]
    [INPUTS]
        <REF> book_id </REF>: int "Identifier of the book to mark as available."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL UPDATE statement through self.cursor to set available = 1 in the books table for the row whose id equals book_id, binding book_id as the query parameter; if no row matches, no update occurs. RESULT cursor_update_result]
            [COMMAND Commit the current database transaction through self.connection to persist the update. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a SELECT query on the books table and returns all fetched book rows." search_books]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> books </REF>: list "All remaining rows fetched from the cursor's current result set, normally a list of row tuples; empty list if no rows remain."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query 'SELECT * FROM books' on self.cursor. RESULT self.cursor]
            [COMMAND Assign books to the result of calling self.cursor.fetchall(), which returns all remaining rows from the cursor's current result set, normally as a list of row tuples or an empty list if no rows remain. RESULT books]
            [COMMAND Return books. RESULT books]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
