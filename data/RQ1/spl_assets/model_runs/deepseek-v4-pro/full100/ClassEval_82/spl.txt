# CLASS StockPortfolioTracker

[DEFINE_WORKER: "Initializes a new instance by setting portfolio to an empty list and cash_balance to the provided cash balance." __init__]
    [INPUTS]
        <REF> cash_balance </REF>: any "Initial cash balance value assigned to the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new empty list and assign it to the instance portfolio attribute. RESULT self.portfolio]
            [COMMAND Assign the current cash_balance parameter value or reference to the instance cash_balance attribute. RESULT self.cash_balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a stock to the portfolio by increasing the quantity of an existing portfolio entry with the same name or appending the stock as a new entry if no matching name is found." add_stock]
    [INPUTS]
        <REF> stock </REF>: dict "The stock to add, expected to contain name and quantity fields."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each portfolio entry pf in self.portfolio, execute the matching process. RESULT pf]
            [COMMAND Within the loop over self.portfolio, compare pf['name'] with stock['name']. RESULT name_match]
            [COMMAND Within the loop over self.portfolio, if name_match is false, continue to the next portfolio entry. RESULT next_pf]
            [COMMAND After the loop over self.portfolio completes without an early return, append stock to self.portfolio. RESULT appended_stock]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: pf['name'] equals stock['name'] for the current portfolio entry inside the loop over self.portfolio]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase pf['quantity'] by stock['quantity'] for the matched portfolio entry. RESULT updated_quantity]
            [COMMAND Return immediately from add_stock without appending stock to the portfolio. RESULT return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes or reduces a matching stock entry from the portfolio and returns whether the operation succeeded." remove_stock]
    [INPUTS]
        <REF> stock </REF>: dict "The stock to remove, containing at least name and quantity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if a matching portfolio entry with sufficient quantity was decremented or removed, False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each portfolio entry pf in self.portfolio in order. RESULT pf]
            [COMMAND Within the loop, check whether pf['name'] equals stock['name']. RESULT name_matches]
            [COMMAND Only if name_matches is true, check whether pf['quantity'] is greater than or equal to stock['quantity']. RESULT quantity_sufficient]
            [COMMAND If both name_matches and quantity_sufficient are true, proceed with the matching entry; otherwise continue to the next portfolio entry. RESULT matched]
            [COMMAND Within the matching entry, decrement pf['quantity'] by stock['quantity']. RESULT updated_quantity]
            [COMMAND Within the matching entry, check whether pf['quantity'] now equals 0. RESULT is_depleted]
            [COMMAND If is_depleted is true, remove pf from self.portfolio. RESULT removed]
            [COMMAND Return True immediately, exiting the loop and method. RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After the loop completes without finding any portfolio entry whose name equals stock['name'] and whose quantity is greater than or equal to stock['quantity'].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT failure]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a stock purchase by adding the stock and deducting its total cost from the cash balance when the cost is within available cash, otherwise returns False." buy_stock]
    [INPUTS]
        <REF> stock </REF>: dict "Stock mapping with price and quantity fields used to compute the total purchase cost."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True if the stock was purchased and cash was deducted; returns False if available cash was insufficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the total purchase cost as stock['price'] multiplied by stock['quantity']. RESULT total_cost]
            [COMMAND Evaluate the if condition: total_cost > self.cash_balance; when true, follow the insufficient cash alternative flow. RESULT cash_check]
            [COMMAND In the else branch, call self.add_stock(stock) to add the stock. RESULT add_result]
            [COMMAND In the else branch, subtract total_cost from self.cash_balance. RESULT cash_balance]
            [COMMAND In the else branch, return True. RESULT return_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: stock['price'] * stock['quantity'] > self.cash_balance]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying cash balance or stock holdings. RESULT return_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes stock and, if removal succeeds, adds the sale proceeds to cash balance and returns True; otherwise returns False." sell_stock]
    [INPUTS]
        <REF> stock </REF>: dict "Stock data containing price and quantity fields used to calculate the sale proceeds"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the stock was removed and the sale proceeds were added to cash_balance; False when remove_stock returns False"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.remove_stock(stock), compare the returned value to False using explicit equality, and store the comparison result as removal_failed_flag. RESULT removal_failed_flag]
            [COMMAND When removal_failed_flag is False, add the product of stock['price'] and stock['quantity'] to self.cash_balance. RESULT updated_cash_balance]
            [COMMAND Return True after the cash_balance has been updated. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When removal_failed_flag is True]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without updating cash_balance. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total portfolio value by summing the cash balance and the market value of all stock holdings." calculate_portfolio_value]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> total_value </REF>: number "Total portfolio value equal to cash balance plus the sum of price times quantity for each holding."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set total_value to the current value of self.cash_balance. RESULT total_value]
            [COMMAND For each stock in self.portfolio, multiply the value stored under the 'price' key by the value stored under the 'quantity' key, then add the product to total_value. RESULT total_value]
            [COMMAND Return total_value. RESULT total_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes each stock value in the portfolio, builds a summary list of name-value records, calculates the total portfolio value, and returns both as a tuple." get_portfolio_summary]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> portfolio_value </REF>: unknown "The total portfolio value returned by self.calculate_portfolio_value()."
        <REF> summary </REF>: list "A list of dictionaries containing each stock's name and computed value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize summary as an empty list. RESULT summary]
            [COMMAND For each stock in self.portfolio, iterate in natural order until the iterable is exhausted; within the loop, compute value by calling self.get_stock_value(stock), then append a dictionary with key 'name' set to stock['name'] and key 'value' set to value to summary. If self.portfolio is empty, skip the loop body and leave summary as an empty list. RESULT summary]
            [COMMAND Compute portfolio_value by calling self.calculate_portfolio_value(). RESULT portfolio_value]
            [COMMAND Return portfolio_value and summary as a tuple. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total stock value by multiplying the stock's price by its quantity." get_stock_value]
    [INPUTS]
        <REF> stock </REF>: mapping "A mapping containing 'price' and 'quantity' keys with numeric values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stock_value </REF>: number "The product of the stock's price and quantity."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the product of the stock's 'price' value and the stock's 'quantity' value. RESULT stock_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
