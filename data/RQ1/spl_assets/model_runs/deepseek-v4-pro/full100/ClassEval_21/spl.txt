[DEFINE_WORKER: "Determines whether a given HH:MM time falls within any course interval in self.courses using inclusive start and end boundaries, returning False if a conflict exists and True otherwise." is_free_at]
    [INPUTS]
        <REF> check_time </REF>: str "Time string in HH:MM format to check for course conflicts."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False if check_time lies within any course interval inclusive; True if no course contains check_time."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the input check_time string using datetime.strptime with format '%H:%M' and assign the resulting datetime object back to check_time. Valid values are zero-padded HH:MM strings from 00:00 through 23:59. RESULT check_time]
            [COMMAND Within the for loop over each course in self.courses in order, parse course['start_time'] using datetime.strptime with format '%H:%M'. If the parsed start_time <= check_time, parse course['end_time'] using the same format and then check whether check_time <= parsed end_time. If both inclusive comparisons are true, set result to False and return immediately, stopping the loop. RESULT result]
            [COMMAND If the for loop over self.courses completes without setting result to False, set result to True and return it. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a new course conflicts with any existing course by parsing start and end times in '%H:%M' format and returning False if any existing course interval contains the new course start or end time." check_course_conflict]
    [INPUTS]
        <REF> new_course </REF>: dict "Dictionary containing 'start_time' and 'end_time' string values in '%H:%M' format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> flag </REF>: bool "Returns True if no existing course interval contains the new course start or end time; otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse new_course['start_time'] using datetime.strptime with format '%H:%M' and assign the resulting datetime object to new_start_time. RESULT new_start_time]
            [COMMAND Parse new_course['end_time'] using datetime.strptime with format '%H:%M' and assign the resulting datetime object to new_end_time. RESULT new_end_time]
            [COMMAND Set flag to True. RESULT flag]
            [COMMAND For each course in self.courses, parse course['start_time'] using datetime.strptime with format '%H:%M' and assign the resulting datetime object to start_time. RESULT start_time]
            [COMMAND Within the for loop over self.courses, parse course['end_time'] using datetime.strptime with format '%H:%M' and assign the resulting datetime object to end_time. RESULT end_time]
            [COMMAND Within the for loop over self.courses, if start_time <= new_start_time and end_time >= new_start_time, set flag to False. RESULT flag]
            [COMMAND Within the for loop over self.courses, if start_time <= new_end_time and end_time >= new_end_time, set flag to False. RESULT flag]
            [COMMAND Return flag. RESULT flag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
