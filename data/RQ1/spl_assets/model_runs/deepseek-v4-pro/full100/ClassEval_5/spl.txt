# CLASS AutomaticGuitarSimulator

[DEFINE_WORKER: "Initializes the instance by assigning the provided text argument to the play_text attribute." __init__]
    [INPUTS]
        <REF> text </REF>: str "Text value to store as the play_text attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter text to the instance attribute self.play_text. RESULT self.play_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses space-separated segments from play_text, splitting each segment into Chord and Tune at the first non-alphabetic character and returning a list of Chord/Tune dictionaries, optionally displaying each pair and returning an empty list when play_text is blank." interpret]
    [INPUTS]
        <REF> play_text </REF>: str "Instance text to parse; segments are separated by single spaces."
        <REF> display </REF>: bool "When true, self.display is called for each parsed chord and tune pair; defaults to false."
    [END_INPUTS]

    [OUTPUTS]
        <REF> play_list </REF>: list[dict] "List of dictionaries with keys Chord and Tune parsed from play_text; empty list when play_text is blank or whitespace-only."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize play_list to an empty list. RESULT play_list]
            [COMMAND Split self.play_text by a single space and assign the resulting segments to play_segs. RESULT play_segs]
            [COMMAND For each play_seg in play_segs, set pos to 0 for the current segment. RESULT pos]
            [COMMAND Within the loop over play_seg, for each ele in play_seg until the first non-alphabetic character or end of segment, if ele.isalpha() is true, increment pos by 1 and continue; otherwise break out of the character loop. RESULT pos]
            [COMMAND Within the loop over play_seg, set play_chord to play_seg[0:pos]. RESULT play_chord]
            [COMMAND Within the loop over play_seg, set play_value to play_seg[pos:]. RESULT play_value]
            [COMMAND Within the loop over play_seg, append a dictionary with keys Chord and Tune set to play_chord and play_value to play_list. RESULT play_list]
            [COMMAND Within the loop over play_seg, if display is true, call self.display(play_chord, play_value). RESULT display_result]
            [COMMAND After processing all play_seg items, return play_list. RESULT play_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.play_text.strip() is empty or whitespace-only]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately without parsing or displaying anything. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats and returns a normal guitar playing message using the given chord key and tune value." display]
    [INPUTS]
        <REF> key </REF>: string "The chord key substituted into the first string placeholder."
        <REF> value </REF>: string "The tune value substituted into the second string placeholder."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The formatted message string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Format the string "Normal Guitar Playing -- Chord: %s, Play Tune: %s" by substituting key for the first placeholder and value for the second placeholder. RESULT formatted_message]
            [COMMAND Return the formatted message. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
