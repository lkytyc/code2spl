# CLASS MovieTicketDB

[DEFINE_WORKER: "Initializes a SQLite database connection and cursor, then calls self.create_table() to create the table." __init__]
    [INPUTS]
        <REF> db_name </REF>: string or path-like object "The SQLite database name or path; may be a filesystem path, ':memory:', or an empty string for a temporary database."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open or create the SQLite database identified by db_name using sqlite3.connect and assign the returned connection object to self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection using self.connection.cursor() and assign it to self.cursor. RESULT self.cursor]
            [COMMAND Call self.create_table() to create the table using the initialized connection and cursor. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create the tickets table if it does not already exist and commit the current transaction." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the fixed SQL DDL statement 'CREATE TABLE IF NOT EXISTS tickets (id INTEGER PRIMARY KEY, movie_name TEXT, theater_name TEXT, seat_number TEXT, customer_name TEXT)' using self.cursor.execute. RESULT _execute_result]
            [COMMAND Commit the current transaction using self.connection.commit. RESULT _commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a ticket record using the supplied movie, theater, seat, and customer values and commits the database transaction." insert_ticket]
    [INPUTS]
        <REF> movie_name </REF>: string "Movie name to store in the ticket record."
        <REF> theater_name </REF>: string "Theater name to store in the ticket record."
        <REF> seat_number </REF>: string "Seat number to store in the ticket record."
        <REF> customer_name </REF>: string "Customer name to store in the ticket record."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.cursor.execute with the INSERT INTO tickets (movie_name, theater_name, seat_number, customer_name) VALUES (?, ?, ?, ?) SQL and the tuple (movie_name, theater_name, seat_number, customer_name); the return value is not used. RESULT execute_result]
            [COMMAND If the execute call succeeded, call self.connection.commit() to persist the pending transaction; if execute raised an exception, this commit step is not reached. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches the tickets table for all rows whose customer_name matches the provided customer_name and returns the fetched ticket rows." search_tickets_by_customer]
    [INPUTS]
        <REF> customer_name </REF>: string "Customer name used as the equality filter for the tickets query."
    [END_INPUTS]

    [OUTPUTS]
        <REF> tickets </REF>: list "All rows from the tickets table where customer_name equals the supplied parameter; may be an empty list if no matching tickets exist."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.cursor.execute with the parameterized SQL query 'SELECT * FROM tickets WHERE customer_name = ?' and bind customer_name as the query parameter. RESULT execute_result]
            [COMMAND Fetch all rows from the current cursor result set and assign them to tickets. If no rows remain, tickets receives an empty sequence. RESULT tickets]
            [COMMAND Return the tickets list. RESULT tickets]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete the ticket with the given ticket_id from the tickets table and commit the database transaction." delete_ticket]
    [INPUTS]
        <REF> ticket_id </REF>: int "Identifier of the ticket row to delete."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the DELETE statement on self.cursor using the SQL text 'DELETE FROM tickets WHERE id = ?' and bind ticket_id as the sole parameter in a single-element tuple. RESULT execute_result]
            [COMMAND After the DELETE statement executes without raising an exception, commit the current database transaction on self.connection. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
