# CLASS SplitSentence

[DEFINE_WORKER: "Splits an input string into sentence segments by splitting on whitespace immediately preceded by a period or question mark while avoiding common abbreviation patterns." split_sentences]
    [INPUTS]
        <REF> sentences_string </REF>: string "The input text to be split into sentences."
    [END_INPUTS]

    [OUTPUTS]
        <REF> sentences </REF>: list "The list of sentence segments produced by the regex split."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.split on sentences_string using the regular expression r'(?<!\w\.\w.)(?<![A-Z][a-z]\.)(?<=\.|\?)\s' to split on whitespace preceded by a period or question mark while negative lookbehinds prevent splitting after abbreviation-like patterns such as Mr., Dr., Mrs., e.g., or U.S.; assign the returned list to sentences. If the regex does not match, re.split returns a list containing the original sentences_string; if sentences_string is empty, it returns a list containing one empty string; a trailing period or question mark followed by whitespace may produce an empty final element. RESULT sentences]
            [COMMAND Return the list sentences as the method result. RESULT sentences]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Cleans a sentence by removing all non-ASCII-letter and non-whitespace characters, splits the cleaned text into words, and returns the number of words." count_words]
    [INPUTS]
        <REF> sentence </REF>: string "The input sentence to clean and count words from."
    [END_INPUTS]

    [OUTPUTS]
        <REF> word_count </REF>: integer "The number of words in the cleaned sentence."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove every character from sentence that is not an ASCII letter or whitespace using regex substitution r'[^a-zA-Z\s]' with empty replacement and store the cleaned string back into sentence. RESULT sentence]
            [COMMAND Split the cleaned sentence on whitespace runs to produce a list of word tokens. RESULT words]
            [COMMAND Return the length of words. RESULT word_count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Splits the input text into sentences, counts the words in each sentence, and returns the maximum word count found." process_text_file]
    [INPUTS]
        <REF> sentences_string </REF>: text "The input text containing sentences to split and analyze."
    [END_INPUTS]

    [OUTPUTS]
        <REF> max_count </REF>: integer "The maximum word count among all sentences; returns 0 when there are no sentences."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.split_sentences with sentences_string and store the returned collection in sentences. RESULT sentences]
            [COMMAND Initialize max_count to the integer literal 0. RESULT max_count]
            [COMMAND Start a for loop over each sentence in sentences. RESULT sentence]
            [COMMAND Within the for loop over sentences, call self.count_words with sentence and store the returned value in count. RESULT count]
            [COMMAND Within the for loop over sentences, if count is strictly greater than max_count, set max_count to count. RESULT max_count]
            [COMMAND Return the final max_count value. RESULT max_count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
