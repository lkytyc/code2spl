# CLASS RegexUtils

[DEFINE_WORKER: "Return True if the regular expression pattern matches the beginning of text, otherwise return False." match]
    [INPUTS]
        <REF> pattern </REF>: str "Regular expression pattern to match at the start of text."
        <REF> text </REF>: str "Input string to match against the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when re.match returns a truthy match object; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.match(pattern, text) and assign the returned match object or None to ans. If pattern is an invalid regular expression, re.error propagates and execution stops before assignment. RESULT ans]
            [COMMAND Evaluate ans in a boolean context. If ans is truthy, return True; otherwise, return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns all non-overlapping matches for a regular expression pattern in the provided text using re.findall." findall]
    [INPUTS]
        <REF> pattern </REF>: str|re.Pattern "Regular expression pattern or compiled pattern object used to search the text."
        <REF> text </REF>: str|bytes "Input string or bytes object to search for matches."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of non-overlapping matches, captured group strings, or tuples of captured strings when capturing groups are present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.findall with the provided pattern and text to compute a list of non-overlapping matches or captured groups. RESULT result]
            [COMMAND Return the computed result list to the caller. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Splits the input text using the regular expression pattern via re.split with default maxsplit 0 and returns the resulting list." split]
    [INPUTS]
        <REF> pattern </REF>: str "Regular expression pattern used to split the text."
        <REF> text </REF>: str "The input string to be split."
    [END_INPUTS]

    [OUTPUTS]
        <REF> split_result </REF>: list "List of substrings produced by splitting text on the regular expression pattern."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the imported regex module re to split the input text at every match of the regular expression pattern with no limit on the number of splits, then return the resulting list. RESULT split_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a copy of text with all non-overlapping matches of the regular expression pattern replaced by replacement." sub]
    [INPUTS]
        <REF> pattern </REF>: str "The regular expression pattern to search for in text."
        <REF> replacement </REF>: str or callable "The replacement string or callable invoked for each match."
        <REF> text </REF>: str or bytes "The input string or bytes-like object to search and replace within."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or bytes "A copy of text with all non-overlapping matches of pattern replaced by replacement."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the result of re.sub(pattern, replacement, text), which replaces all non-overlapping matches of pattern in text with replacement using default count=0 and flags=0; if replacement is callable, invoke it for each match, otherwise use it as a replacement template. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates and returns a raw string regular expression pattern for matching standard email addresses." generate_email_pattern]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> pattern </REF>: string "Raw string regular expression pattern used to match email addresses."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the raw string regular expression pattern "\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b" to variable pattern. RESULT pattern]
            [COMMAND Return the current value of variable pattern. RESULT pattern]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a constant raw regular expression string that matches a phone-like sequence of three digits, a hyphen, three digits, a hyphen, and four digits bounded by word boundaries." generate_phone_number_pattern]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> pattern </REF>: str "The raw regular expression string r'\b\d{3}-\d{3}-\d{4}\b'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constant raw regular expression string r'\b\d{3}-\d{3}-\d{4}\b' to the local variable pattern. RESULT pattern]
            [COMMAND Return the value stored in pattern. RESULT pattern]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates a static sentence-boundary regex pattern that matches '.', '!', or '?' followed by one or two whitespace characters and an uppercase ASCII letter, then returns that pattern." generate_split_sentences_pattern]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> pattern </REF>: str "Regular expression string that matches a sentence-ending punctuation mark, one or two whitespace characters, and a positive lookahead for an uppercase ASCII letter."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the raw string regex pattern r'[.!?][\s]{1,2}(?=[A-Z])' to the variable pattern. RESULT pattern]
            [COMMAND Return the value of pattern. RESULT pattern]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Split the input text into sentences using a generated sentence split pattern." split_sentences]
    [INPUTS]
        <REF> text </REF>: string "The input text to be split into sentences."
    [END_INPUTS]

    [OUTPUTS]
        <REF> split_result </REF>: list<string> "The result of splitting the input text using the generated pattern."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Generate the sentence split pattern by calling self.generate_split_sentences_pattern() with no arguments. RESULT pattern]
            [COMMAND Call self.split(pattern, text) to split the input text into sentence segments and return the result. RESULT split_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate a phone number pattern from self and validate the provided phone number against that pattern using self.match." validate_phone_number]
    [INPUTS]
        <REF> phone_number </REF>: string "Phone number value to validate against the generated pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: unknown "Return value from self.match(pattern, phone_number), representing the validation result for the phone number."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.generate_phone_number_pattern() and assign the returned pattern to the local variable pattern. RESULT pattern]
            [COMMAND Call self.match with arguments pattern and phone_number, then return the call result as the method return value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts email addresses from the input text by generating an email pattern and returning all matches of that pattern in the text." extract_email]
    [INPUTS]
        <REF> text </REF>: str "The input text in which to search for email addresses."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: list "All matches found by self.findall for the generated email pattern in the input text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call the internal method generate_email_pattern on self with no arguments and assign its returned value to pattern. RESULT pattern]
            [COMMAND Call the internal method findall on self with the generated pattern and the input text, then return the result of that call. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
