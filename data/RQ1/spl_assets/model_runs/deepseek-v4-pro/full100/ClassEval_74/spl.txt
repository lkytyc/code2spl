[DEFINE_WORKER: "Adds an address to the whitelist if it is not already present, returning False if already present and the updated whitelist otherwise." add_white_list]
    [INPUTS]
        <REF> addr </REF>: object "The address or item to add to the whitelist."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or list "False if addr is already in self.white_list; otherwise the updated self.white_list after appending addr."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is already an element of self.white_list. RESULT is_whitelisted]
            [COMMAND If addr is not in self.white_list, append addr to self.white_list. RESULT updated_white_list]
            [COMMAND If addr is not in self.white_list, return self.white_list. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: addr is already an element of self.white_list]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove an address from the whitelist and return the updated whitelist if the address is present; otherwise return False." del_white_list]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is not in self.white_list. RESULT addr_absent]
            [COMMAND When addr_absent is false, remove addr from self.white_list and return self.white_list. RESULT return_value]

[DEFINE_WORKER: "Receives an info dict, validates it, checks addr against white_list, stores addr and content in receive_struct, and returns content for whitelisted addresses." recv]
    [INPUTS]
        <REF> info </REF>: dict "Input mapping expected to contain 'addr' and 'content' keys."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: any "Returns content for whitelisted addr, False for non-whitelisted addr, or -1 for invalid info."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Confirm that info is a dict and contains both 'addr' and 'content' keys; otherwise use the invalid-input alternative flow that returns -1. RESULT input_valid]
            [COMMAND Extract addr from info['addr']. RESULT addr]
            [COMMAND Extract content from info['content']. RESULT content]
            [COMMAND Check whether addr is in self.white_list; if not, use the non-whitelisted alternative flow that returns False. RESULT addr_whitelisted]
            [COMMAND Set self.receive_struct to {'addr': addr, 'content': content}. RESULT self.receive_struct]
            [COMMAND Return self.receive_struct['content']. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dict, or 'addr' not in info, or 'content' not in info (evaluated left-to-right with short-circuiting).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: addr not in self.white_list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that info is a dictionary containing the required keys 'addr' and 'content', then copies those values into self.send_struct; otherwise returns an error string." send]
    [INPUTS]
        <REF> info </REF>: dict "Input dictionary expected to contain the keys 'addr' and 'content'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string or None "Returns the string 'info structure is not correct' when validation fails; otherwise returns None after setting self.send_struct."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate that info is a dictionary and contains both required keys "addr" and "content"; if validation fails, branch to the alternative flow and do not continue. RESULT validation_status]
            [COMMAND Assign self.send_struct to a new dictionary with key "addr" mapped to info["addr"] and key "content" mapped to info["content"]. RESULT self.send_struct]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dict, which short-circuits the remaining key checks; or info is a dict but lacks the key "addr", which short-circuits the final key check; or info is a dict and has "addr" but lacks the key "content".]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the string "info structure is not correct". RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the send_struct when type is 'send', the receive_struct when type is 'receive', or False for any other type." show]
    [INPUTS]
        <REF> type </REF>: any "Selector value used to choose which struct to return; the strings 'send' and 'receive' select the corresponding struct, and any other value falls through to the else branch."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object|boolean "Returns self.send_struct when type is 'send', self.receive_struct when type is 'receive', or False for any other value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the top-level if-elif-else selector, if type equals "send", return self.send_struct. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the same top-level if-elif-else selector, type equals "receive" and the first condition is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return self.receive_struct. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the same top-level if-elif-else selector, type equals neither "send" nor "receive".]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
