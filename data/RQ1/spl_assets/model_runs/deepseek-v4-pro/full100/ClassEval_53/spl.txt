[DEFINE_WORKER: "Normalizes a sentence to lowercase, removes punctuation, splits it on literal spaces, and returns the longest word that is also in self.word_list, defaulting to an empty string when no such word is found." find_longest_word]
    [INPUTS]
        <REF> sentence </REF>: str "The input sentence to normalize, split, and search for the longest known word."
    [END_INPUTS]

    [OUTPUTS]
        <REF> longest_word </REF>: str "The longest word from the normalized sentence that exists in self.word_list; returns an empty string if no matching word is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize longest_word to the empty string as the default sentinel value. RESULT longest_word]
            [COMMAND Convert sentence to lowercase and assign the lowercased string back to sentence. RESULT sentence]
            [COMMAND Remove every punctuation character listed in string.punctuation from sentence by applying re.sub with a regex character class built from re.escape(string.punctuation), and assign the cleaned string back to sentence. RESULT sentence]
            [COMMAND Split the cleaned sentence on each literal space character using re.split(' ', sentence) and assign the resulting list of words to sentence; consecutive spaces can produce empty-string elements, and an empty cleaned sentence yields ['']. RESULT sentence]
            [COMMAND For each word in sentence, iterating sequentially with no break, continue, or early return: if word exists in self.word_list and, due to short-circuit evaluation only when membership holds, len(word) is strictly greater than len(longest_word), set longest_word to word; otherwise leave longest_word unchanged. Strict greater means ties keep the earliest occurrence of the maximum length. RESULT longest_word]
            [COMMAND Return longest_word. RESULT longest_word]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
