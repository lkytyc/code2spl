# CLASS PersonRequest

[DEFINE_WORKER: "Initializes a validated contact record by assigning validated name, sex, and phoneNumber to instance attributes." __init__]
    [INPUTS]
        <REF> name </REF>: str "Raw name input to be validated and assigned."
        <REF> sex </REF>: str "Raw sex input to be validated and assigned."
        <REF> phoneNumber </REF>: str "Raw phone number input to be validated and assigned."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.name to the result returned by self._validate_name(name). RESULT self.name]
            [COMMAND Assign self.sex to the result returned by self._validate_sex(sex). RESULT self.sex]
            [COMMAND Assign self.phoneNumber to the result returned by self._validate_phoneNumber(phoneNumber). RESULT self.phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates a name by returning None when the name is falsy or longer than 33 characters, otherwise returning the original name." _validate_name]
    [INPUTS]
        <REF> name </REF>: str "The candidate name string to validate; may be None or another falsy value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> validated_name </REF>: Optional[str] "The original name if it passes both validation checks, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name is falsy. If it is falsy, such as None or an empty string, return None as the validation result and stop further checks. RESULT validated_name]
            [COMMAND Check whether len(name) is greater than 33. If true, return None as the validation result and stop further checks. RESULT validated_name]
            [COMMAND Return the original name as the validation result because both guard checks passed. RESULT validated_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the sex parameter is one of 'Man', 'Woman', or 'UGM', and returns the original sex if valid or None otherwise." _validate_sex]
    [INPUTS]
        <REF> sex </REF>: str "The sex value to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | None "The validated sex value if it is one of 'Man', 'Woman', or 'UGM'; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether sex is one of the allowed strings "Man", "Woman", or "UGM". RESULT sex_is_allowed]
            [COMMAND If sex_is_allowed is true, return sex unchanged. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: sex is not in ["Man", "Woman", "UGM"] (sex_is_allowed is false).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that phoneNumber is an 11-digit numeric string and returns it, or returns None if the input is missing or invalid." _validate_phoneNumber]
    [INPUTS]
        <REF> phoneNumber </REF>: str "The phone number string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | None "The original phone number if it is exactly 11 numeric digits; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether phoneNumber is truthy. RESULT phone_present]
            [COMMAND If phone_present is true, check whether len(phoneNumber) equals 11. RESULT length_ok]
            [COMMAND If phone_present is true and length_ok is true, check whether phoneNumber.isdigit() is true. RESULT digit_ok]
            [COMMAND If phone_present is true and length_ok is true and digit_ok is true, return phoneNumber. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: phone_present is false because phoneNumber is falsy, such as None or an empty string.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: phone_present is true and length_ok is false because len(phoneNumber) is not equal to 11.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: phone_present is true, length_ok is true, and digit_ok is false because phoneNumber contains at least one non-digit character.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
