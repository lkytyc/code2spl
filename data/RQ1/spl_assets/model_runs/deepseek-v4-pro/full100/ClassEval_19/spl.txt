# CLASS ChandrasekharSieve

[DEFINE_WORKER: "Initializes the instance by storing the provided value n in self.n and setting self.primes to the result of calling generate_primes with no arguments." __init__]
    [INPUTS]
        <REF> n </REF>: any "Input value assigned directly to self.n and used indirectly by generate_primes."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter n to the instance attribute self.n without transformation. RESULT self.n]
            [COMMAND Call self.generate_primes() with no explicit arguments and assign its return value to the instance attribute self.primes. RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate and return all prime numbers from 2 through self.n inclusive using the Sieve of Eratosthenes, returning an empty list immediately when self.n is less than 2." generate_primes]
    [INPUTS]
        <REF> self.n </REF>: integer "Upper bound for prime generation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> primes </REF>: list<integer> "A list of prime numbers from 2 through self.n inclusive, or an empty list when self.n is less than 2."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check the early-return guard: if self.n is less than 2, take the alternative flow and return an empty list immediately; otherwise continue with the sieve setup. RESULT guard_check]
            [COMMAND Initialize sieve as a list of True values with length self.n + 1, so valid indices are 0 through self.n inclusive. RESULT sieve]
            [COMMAND Set sieve[0] and sieve[1] to False because 0 and 1 are not primes; after the guard this requires self.n >= 2, so indices 0 and 1 always exist. RESULT sieve_boundary_flags]
            [COMMAND Set p to the integer literal 2 as the first candidate prime. RESULT candidate_p]
            [COMMAND Enter a while loop that continues while p * p <= self.n, using an inclusive boundary so the loop still runs when p * p equals self.n. RESULT sieve_marking_loop]
            [COMMAND Within the while loop, evaluate the condition sieve[p]; if sieve[p] is True, execute the inner for loop; if false, skip the inner for loop. RESULT sieve_p_check]
            [COMMAND Within the if branch when sieve[p] is True, run an inner for loop over every integer i from p * p to self.n inclusive, stepping by p, and set sieve[i] to False for each visited i. RESULT marked_multiples]
            [COMMAND After the if branch, whether or not it executed, increment p by 1. RESULT p_increment]
            [COMMAND Re-evaluate the while condition p * p <= self.n; exit the loop when p * p becomes greater than self.n. RESULT while_condition_recheck]
            [COMMAND After the while loop terminates, initialize primes as a new empty list. RESULT primes]
            [COMMAND Start a for loop over every integer i from 2 to self.n inclusive, using range(2, self.n + 1). RESULT collection_loop]
            [COMMAND Within the for loop, check whether sieve[i] is True; if true, append i to primes; if false, skip appending. RESULT prime_collection]
            [COMMAND After the for loop completes, return the primes list. RESULT return_primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If self.n is less than 2, the method returns an empty list immediately without performing sieve setup or loop processing.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT return_empty_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the instance attribute primes to the caller." get_primes]
    [INPUTS]
        <REF> self </REF>: object "The current instance from which the primes attribute is read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "The exact object currently bound to self.primes; no transformation or default is applied."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the current value of the instance attribute self.primes to the caller. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
