# CLASS XMLProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided file_name reference in self.file_name and setting self.root to None." __init__]
    [INPUTS]
        <REF> file_name </REF>: object "The input file name or reference to assign to the instance attribute self.file_name; may be None or a mutable object."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter file_name to the instance attribute self.file_name; if file_name is None, self.file_name is also None, and if it references a mutable object, self.file_name aliases it. RESULT self.file_name]
            [COMMAND Initialize the instance attribute self.root to the constant None, independent of other variables. RESULT self.root]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses an XML file from self.file_name, stores its root element in self.root, and returns the root element, or returns None if parsing fails." read_xml]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: Element or None "The root element of the parsed XML file, or None if parsing or root retrieval fails."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the try block, attempt to parse the XML file located at self.file_name using ET.parse and assign the resulting ElementTree object to tree. RESULT tree]
            [COMMAND Still within the try block and after successful parsing, get the root element from tree using tree.getroot() and assign it to self.root. RESULT self.root]
            [COMMAND Return self.root as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If ET.parse(self.file_name) or tree.getroot() raises any exception within the try block.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None as the method result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Serializes the XML element tree rooted at self.root to the specified file and returns a boolean success indicator." write_xml]
    [INPUTS]
        <REF> file_name </REF>: string "The file path where the XML tree will be written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> success </REF>: boolean "True if the XML tree was written successfully; False if any exception occurred"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within a try block, create an ElementTree object from self.root RESULT tree]
            [COMMAND Within the same try block, write tree to file_name RESULT write_result]
            [COMMAND If no exception has occurred, return True RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while creating the ElementTree object or writing it to file_name]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Uppercases the text content of every <item> element under self.root and writes the modified XML to file_name." process_xml_data]
    [INPUTS]
        <REF> file_name </REF>: str "The file name to which the modified XML is written."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each element in self.root.iter('item'): RESULT element]
            [COMMAND Within the loop over element, set text to element.text. RESULT text]
            [COMMAND Within the loop over element, set element.text to text.upper(). RESULT element.text]
            [COMMAND After the loop, return the result of self.write_xml(file_name). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Finds all elements matching the given element name under the root object and returns them." find_element]
    [INPUTS]
        <REF> element_name </REF>: string "Name of the element to find."
    [END_INPUTS]

    [OUTPUTS]
        <REF> elements </REF>: list "All matching elements found under self.root."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call findall(element_name) on self.root and assign the returned elements to elements. RESULT elements]
            [COMMAND Return the elements list. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
