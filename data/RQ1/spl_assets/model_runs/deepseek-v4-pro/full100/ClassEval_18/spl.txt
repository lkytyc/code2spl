[DEFINE_WORKER: "Converts a string key to camel case by calling _to_camel_case and returns non-string keys unchanged." _convert_key]
    [INPUTS]
        <REF> key </REF>: object "The key to convert; if it is a string, it is normalized to camel case."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "The camel-cased string when key is a string; otherwise the original key."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the condition isinstance(key, str) to determine whether key is a string. RESULT is_string]
            [COMMAND Within the if branch where is_string is true, call self._to_camel_case(key) and return its result immediately. RESULT converted_key]
            [COMMAND After the if block, when is_string is false, return key unchanged. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert an underscore-delimited string into camelCase by splitting on underscores, preserving the first segment unchanged, then title-casing and concatenating each subsequent segment without separators." _to_camel_case]
    [INPUTS]
        <REF> key </REF>: string "The input string to convert from underscore-delimited format to camelCase."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The converted camelCase string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign parts to the list created by splitting the input key at every underscore delimiter, preserving exact split behavior: an empty key returns ['']; a key without underscores returns [key]; leading, trailing, or consecutive underscores produce empty-string elements. RESULT parts]
            [COMMAND Build the return value by concatenating the first element parts[0] with the result of joining all elements from parts[1:] after converting each element to title case using part.title(), without any separator, and return that concatenated string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
