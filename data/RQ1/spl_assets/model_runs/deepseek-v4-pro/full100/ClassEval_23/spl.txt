[DEFINE_WORKER: "Computes the binomial coefficient C(n, m) using the factorial formula and returns 1 for the base cases m equals 0 or n equals m." count]
    [INPUTS]
        <REF> n </REF>: int "Total number of items, a non-negative integer."
        <REF> m </REF>: int "Number of selected items, a non-negative integer with m less than or equal to n."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The exact integer binomial coefficient C(n, m)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n_minus_m as n minus m. RESULT n_minus_m]
            [COMMAND Compute fact_n as math.factorial(n). RESULT fact_n]
            [COMMAND Compute fact_n_minus_m as math.factorial(n_minus_m). RESULT fact_n_minus_m]
            [COMMAND Compute fact_m as math.factorial(m). RESULT fact_m]
            [COMMAND Compute denominator as fact_n_minus_m multiplied by fact_m. RESULT denominator]
            [COMMAND Compute result as fact_n integer-divided by denominator. RESULT result]
            [COMMAND Return result as the function output. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m equals 0 or n equals m, evaluated with short-circuit semantics: n equals m is checked only when m equals 0 is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 1 immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes a bitmask-like value for a nonnegative bit count n, with an early False return for n outside [0, 63] and a positive-infinity special case when n equals 63." count_all]
    [INPUTS]
        <REF> n </REF>: int "Number of low-order bits to set to 1; expected to be an integer in the inclusive range [0, 63] for the normal path."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int | float | bool "Returns False if n is outside [0, 63]; returns positive infinity if n equals 63; otherwise returns the integer (1 << n) - 1 with n low-order bits set to 1."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND At top level, evaluate the guard condition n < 0 or n > 63 using short-circuit evaluation, storing the result as invalid_range_check. RESULT invalid_range_check]
            [COMMAND If invalid_range_check is false, evaluate whether n equals 63, storing the result as special_case_check. RESULT special_case_check]
            [COMMAND If special_case_check is false, compute (1 << n) - 1 and return it as the function result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: invalid_range_check is true (n < 0 or n > 63)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately as the function result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: special_case_check is true (n == 63)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return float("inf") as the function result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Initializes an empty result list and calls the recursive helper _select to populate and return all selections of length m." select]
    [INPUTS]
        <REF> m </REF>: int "Length of each selection list to generate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "The list of generated selection lists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list result to collect generated selections. RESULT result]
            [COMMAND Create a list of m None values as the initial candidate buffer. RESULT candidate_buffer]
            [COMMAND Call self._select with start index 0, candidate_buffer, depth 0, and result to populate result with the generated selections. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collects all selection results for sizes from 1 through the number of available data items into a single list of string lists." select_all]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "A list containing all lists of strings returned by self.select(i) for i from 1 to len(self.datas) inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize variable result to an empty list. RESULT result]
            [COMMAND For each integer i from 1 to len(self.datas) inclusive, call self.select(i) and assign the returned list to a temporary variable select_result_i. RESULT select_result_i]
            [COMMAND Within the for loop over i, extend result by appending all elements from select_result_i. RESULT result]
            [COMMAND After the loop completes, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Recursively generates all ordered combinations from self.datas by filling resultList and appending a shallow copy to result when the buffer is full, while pruning branches that cannot build a full-length result." _select]
    [INPUTS]
        <REF> dataIndex </REF>: int "Inclusive starting index in self.datas for the current selection level."
        <REF> resultList </REF>: List[str] "Current buffer used to hold selected elements."
        <REF> resultIndex </REF>: int "Zero-based index in resultList that this call fills next."
        <REF> result </REF>: List[List[str]] "Accumulator list to which completed selection copies are appended."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute resultLen as the current length of resultList. RESULT resultLen]
            [COMMAND Compute resultCount as resultIndex plus 1. RESULT resultCount]
            [COMMAND When resultCount is not strictly greater than resultLen, start a for loop over i in range(dataIndex, len(self.datas) + resultCount - resultLen), where dataIndex is inclusive, the computed stop is exclusive, and the step is 1; if the range is empty, skip the loop body. RESULT i]
            [COMMAND Within the for loop over i, assign resultList[resultIndex] = self.datas[i]. RESULT resultList[resultIndex]]
            [COMMAND Within the for loop over i, call self._select(i + 1, resultList, resultIndex + 1, result) recursively; after that recursive call returns, continue with the next i in the loop. RESULT recursiveCall]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: resultCount is strictly greater than resultLen]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append a shallow copy of resultList to result. RESULT result]
            [COMMAND Return immediately from the current method invocation. RESULT _return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
