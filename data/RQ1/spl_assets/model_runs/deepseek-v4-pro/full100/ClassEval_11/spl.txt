# CLASS BitStatusUtil

[DEFINE_WORKER: "Validates two integer bit status values and returns their bitwise OR as a combined bitmask." add]
    [INPUTS]
        <REF> states </REF>: int "The first bit status or flag mask to merge."
        <REF> stat </REF>: int "The second bit status or flag mask to merge."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The combined bitmask produced by the bitwise OR of states and stat."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check with [states, stat] to validate the input bit statuses before merging. RESULT check_result]
            [COMMAND After validation passes, return the bitwise OR of states and stat as the combined bitmask. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the stat bitmask is a bitwise subset of states after validating the arguments." has]
    [INPUTS]
        <REF> states </REF>: int "Integer bit flags to check against."
        <REF> stat </REF>: int "Integer bitmask to verify as a subset of states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if every bit set in stat is also set in states; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check([states, stat]) to validate the arguments. RESULT check_result]
            [COMMAND Compute the bitwise AND of states and stat. RESULT and_result]
            [COMMAND Return whether and_result equals stat; this is true when stat is a bitwise subset of states, including the stat == 0 case. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the stat status flag from the states bitmask when it is present, and otherwise returns states unchanged." remove]
    [INPUTS]
        <REF> states </REF>: integer "Current bitmask containing status flags."
        <REF> stat </REF>: integer "Status flag bit or bits to remove from states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Updated bitmask after removing stat if it was present; otherwise the original states."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call BitStatusUtil.check with a two-element list containing states and stat to validate the inputs before making changes. RESULT validation_result]
            [COMMAND Evaluate whether the stat flag is present in states by calling BitStatusUtil.has(states, stat). RESULT has_flag]
            [COMMAND When has_flag is false, return the original states unchanged. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: has_flag is true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return states XOR stat to clear the stat flag from states. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that every element in args is a non-negative even integer, raising ValueError for negative or odd values." check]
    [INPUTS]
        <REF> args </REF>: list[int] "Collection of integer values to validate."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each arg in args, validate the value in order. RESULT arg]
            [COMMAND Within the for loop over args, check whether arg is less than 0; if true, the corresponding exception flow is triggered. RESULT negative_check]
            [COMMAND Within the for loop over args, after the negative check, check whether arg modulo 2 is not equal to 0; if true, the corresponding exception flow is triggered. RESULT even_check]
            [COMMAND If neither exception condition is triggered for any arg, the loop completes and the method returns successfully. RESULT validation_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: An arg in args is less than 0.]
        [LOG "<arg> must be greater than or equal to 0"]
        [THROW ValueError "Raised when a negative value is encountered."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: An arg in args satisfies arg % 2 != 0, meaning it is odd.]
        [LOG "<arg> not even"]
        [THROW ValueError "Raised when an odd integer is encountered."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
