# CLASS ComplexCalculator

[DEFINE_WORKER: "Initializes the object instance with no additional behavior." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do nothing (pass). RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds two complex numbers by summing their real and imaginary components and returns the resulting complex number." add]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex number operand; must provide numeric real and imaginary components."
        <REF> c2 </REF>: complex "Second complex number operand; must provide numeric real and imaginary components."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: complex "Newly constructed complex number with real part c1.real + c2.real and imaginary part c1.imag + c2.imag."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute real as the sum of the real components of c1 and c2. RESULT real]
            [COMMAND Compute imaginary as the sum of the imaginary components of c1 and c2. RESULT imaginary]
            [COMMAND Construct answer as a complex number from real and imaginary using complex(real, imaginary). RESULT answer]
            [COMMAND Return answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Subtracts two complex-like values component-wise and returns the resulting complex number." subtract]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex-like operand with real and imaginary attributes."
        <REF> c2 </REF>: complex "Second complex-like operand with real and imaginary attributes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: complex "A complex number whose real part is c1.real minus c2.real and whose imaginary part is c1.imag minus c2.imag."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Subtract the real part of c2 from the real part of c1 and assign the result to real. RESULT real]
            [COMMAND Subtract the imaginary part of c2 from the imaginary part of c1 and assign the result to imaginary. RESULT imaginary]
            [COMMAND Return a complex number constructed from real and imaginary. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiply two complex numbers c1 and c2 by computing the real and imaginary parts of their product and returning the resulting complex number." multiply]
    [INPUTS]
        <REF> c1 </REF>: complex "First complex number operand"
        <REF> c2 </REF>: complex "Second complex number operand"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The complex product of c1 and c2"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the real part of the complex product as c1.real * c2.real - c1.imag * c2.imag. RESULT real]
            [COMMAND Compute the imaginary part of the complex product as c1.real * c2.imag + c1.imag * c2.real. RESULT imaginary]
            [COMMAND Return a complex number constructed from the computed real and imaginary parts. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Divides one complex number by another using the standard complex division formula and returns the resulting complex quotient." divide]
    [INPUTS]
        <REF> c1 </REF>: complex "The dividend complex number to be divided."
        <REF> c2 </REF>: complex "The divisor complex number."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: complex "The quotient of c1 divided by c2 returned as a complex number."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the denominator as the squared magnitude of c2 by adding c2.real squared and c2.imag squared. RESULT denominator]
            [COMMAND Compute the real part by adding the product of c1.real and c2.real to the product of c1.imag and c2.imag, then dividing the sum by denominator. RESULT real]
            [COMMAND Compute the imaginary part by subtracting the product of c1.real and c2.imag from the product of c1.imag and c2.real, then dividing the difference by denominator. RESULT imaginary]
            [COMMAND Return the result of constructing a complex number using the computed real and imaginary parts. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
