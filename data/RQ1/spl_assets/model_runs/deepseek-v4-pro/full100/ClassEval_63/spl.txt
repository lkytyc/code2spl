# CLASS NLPDataProcessor2

[DEFINE_WORKER: "Processes a list of strings by lowercasing each string, stripping non-English letters, splitting each cleaned string into whitespace-separated words, and returning the list of word lists." process_data]
    [INPUTS]
        <REF> string_list </REF>: list[str] "List of input strings to process."
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list[list[str]] "List where each element is the list of words extracted from the corresponding cleaned input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize words_list to an empty list. RESULT words_list]
            [COMMAND For each string in string_list, execute the following nested steps. RESULT string]
            [COMMAND Within the for loop over string, convert the current string to lowercase using string.lower(). RESULT processed_string]
            [COMMAND Within the same for loop, remove all characters that are not ASCII letters or whitespace from processed_string using the regular expression [^a-zA-Z\s]. RESULT processed_string]
            [COMMAND Within the same for loop, split processed_string on whitespace into a list of words using split(). RESULT words]
            [COMMAND Within the same for loop, append the words list to words_list. RESULT words_list]
            [COMMAND Return words_list. RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts word occurrences across the provided words_list and returns a dictionary of up to the five most frequent words sorted by descending frequency." calculate_word_frequency]
    [INPUTS]
        <REF> words_list </REF>: list[Iterable[str]] "A list of iterables, where each iterable contains words to count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> top_5_word_frequency </REF>: dict[str, int] "A dictionary containing up to five words with the highest occurrence counts, sorted from highest to lowest frequency."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize word_frequency as a new empty Counter object. RESULT word_frequency]
            [COMMAND For each words in words_list, update word_frequency by counting each element in words. RESULT word_frequency]
            [COMMAND Sort word_frequency items by frequency value in descending order, with stable ordering preserving the existing order for equal frequencies, and assign the resulting dictionary to sorted_word_frequency. RESULT sorted_word_frequency]
            [COMMAND Take the first 5 items of sorted_word_frequency, or all items if fewer than 5 exist, and assign the resulting dictionary to top_5_word_frequency. RESULT top_5_word_frequency]
            [COMMAND Return top_5_word_frequency. RESULT top_5_word_frequency]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes the input string list into a word frequency dictionary by calling process_data and calculate_word_frequency." Process]
    [INPUTS]
        <REF> string_list </REF>: list "The input list of strings to process."
    [END_INPUTS]

    [OUTPUTS]
        <REF> word_frequency_dict </REF>: dict "The computed dictionary mapping words to their frequencies."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.process_data with string_list and assign the returned value to words_list. RESULT words_list]
            [COMMAND Call self.calculate_word_frequency with words_list and assign the returned value to word_frequency_dict. RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict. RESULT word_frequency_dict]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
