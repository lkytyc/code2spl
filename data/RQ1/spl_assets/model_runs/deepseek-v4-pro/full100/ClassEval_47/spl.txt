# CLASS IPAddress

[DEFINE_WORKER: "Initializes the instance by storing the provided ip_address argument in the self.ip_address attribute." __init__]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of the input parameter ip_address to the instance attribute self.ip_address. RESULT self.ip_address]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates whether the instance's ip_address string is a valid IPv4 address by splitting it on dots, requiring exactly four octets, and ensuring each octet is a digit string between 0 and 255." is_valid]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the IP address is a valid IPv4 address according to the format and range checks; False otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address on the '.' delimiter and assign the returned list of substrings to octets. RESULT octets]
            [COMMAND Check whether len(octets) equals 4; if it does, continue to the octet validation loop. RESULT length_check_passed]
            [COMMAND Within the for loop over octets, for each octet check that octet.isdigit() is true and that int(octet) is between 0 and 255 inclusive using short-circuit evaluation. RESULT octet_validation_result]
            [COMMAND After all octets pass validation, return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(octets) is not equal to 4.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately because the address does not contain exactly four octets. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Any octet fails validation because not octet.isdigit() is true, or after digit validation int(octet) is less than 0, or int(octet) is greater than 255.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately from inside the for loop due to an invalid octet. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the dot-separated octets of a valid IP address as a list of strings, or return an empty list when the address is invalid." get_octets]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> octets </REF>: list<string> "The dot-separated octet strings of the valid IP address, or an empty list when the IP address is invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.is_valid() to determine whether the IP address is valid. RESULT validity_check]
            [COMMAND If validity_check is true, split self.ip_address using '.' as the delimiter and return the resulting list of octet strings. RESULT octets]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If validity_check is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list. RESULT octets]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a dot-separated 8-bit binary representation of each octet when the object is valid, otherwise return an empty string." get_binary]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "A dot-separated binary string representation of each octet when the object is valid; otherwise an empty string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.is_valid() returns a truthy value; if false, follow the alternative flow for invalid input. RESULT validity_check]
            [COMMAND Within the true branch of self.is_valid(), initialize binary_octets to an empty list. RESULT binary_octets]
            [COMMAND Within the true branch of self.is_valid(), for each octet in self.get_octets(), convert octet to an integer, format it as an 8-bit zero-padded binary string, and append it to binary_octets. RESULT binary_octets]
            [COMMAND After completing the loop within the true branch of self.is_valid(), return '.'.join(binary_octets), which is an empty string when there are no octets. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns a false value.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
