    [END_INPUTS]

    [OUTPUTS]
        <REF> stop_word_list </REF>: list[str] "A three-element list of lowercase English stop words: 'a', 'an', and 'the'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a mutable list stop_word_list with exactly three lowercase string elements in order: 'a', 'an', and 'the'. RESULT stop_word_list]

[DEFINE_WORKER: "Splits each string in string_list into whitespace-separated words, attempts to remove stop words from each split list by in-place removal, and returns a list of the resulting word lists." remove_stop_words]
    [INPUTS]
        <REF> string_list </REF>: list[str] "Collection of input strings to split and filter."
        <REF> stop_word_list </REF>: list[str] "Collection of stop words to remove from each split string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: list[list[str]] "A list containing one split and filtered word list for each input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize answer to an empty list. RESULT answer]
            [COMMAND For each string in string_list, split the string on whitespace to produce the string_split list. RESULT string_split]
            [COMMAND Within the outer for loop over string_list, for each word in string_split, if word is present in stop_word_list, remove word from string_split; mutating string_split while iterating may skip subsequent words as in the original method. RESULT string_split]
            [COMMAND Within the outer for loop over string_list, after the inner loop completes, append string_split to answer. RESULT answer]
            [COMMAND Return answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Obtains the stop word list from construct_stop_word_list and removes those stop words from string_list via remove_stop_words." process]
    [INPUTS]
        <REF> string_list </REF>: list "List of strings or tokens to be filtered by removing stop words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list "List of strings or tokens after stop word removal."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.construct_stop_word_list() to obtain the stop word list. RESULT stop_word_list]
            [COMMAND Call self.remove_stop_words(string_list, stop_word_list) to remove stop words from string_list. RESULT words_list]
            [COMMAND Return words_list as the filtered result. RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
