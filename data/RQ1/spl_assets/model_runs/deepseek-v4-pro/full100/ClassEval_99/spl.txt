# CLASS ZipFileProcessor

[DEFINE_WORKER: "Initializes the instance by assigning the provided file_name parameter to the instance attribute file_name." __init__]
    [INPUTS]
        <REF> file_name </REF>: any "The value to store as the instance's file_name attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constructor parameter file_name to the instance attribute self.file_name. RESULT self.file_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Open the ZIP file at self.file_name in read mode and return the ZipFile object, or return None if opening fails." read_zip_file]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> ref </REF>: zipfile.ZipFile or None "The opened ZipFile object if successful, otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open the ZIP file located at self.file_name in read mode using zipfile.ZipFile and assign the resulting ZipFile object to zip_file. RESULT zip_file]
            [COMMAND Return zip_file from the method. RESULT ref]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Opening the ZIP file raises any exception.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None from the method. RESULT ref]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts all contents from the zip file named by self.file_name into the specified output_path and returns True on success or False on any failure." extract_all]
    [INPUTS]
        <REF> output_path </REF>: string "Destination directory path where the zip file contents are extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "Returns True if extraction succeeds, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip file named by self.file_name in read mode as zip_file RESULT zip_file]
            [COMMAND Extract all contents from zip_file into output_path RESULT extraction_result]
            [COMMAND Return True RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip file or extracting its contents]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts a specified file from the zip archive stored at self.file_name into the given output path and returns True on success or False on any exception." extract_file]
    [INPUTS]
        <REF> file_name </REF>: str "Name of the file inside the zip archive to extract."
        <REF> output_path </REF>: str "Destination path where the extracted file is written."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the extraction succeeds, False if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip file referenced by self.file_name in read mode as zip_file. RESULT zip_file]
            [COMMAND Extract the member named file_name from zip_file to output_path. RESULT extraction_result]
            [COMMAND Return True. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception occurs while opening the zip file or extracting the specified file.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create a zip archive at the specified output path by writing each input file into it, returning True on success and False if an exception occurs." create_zip_file]
    [INPUTS]
        <REF> files </REF>: list<string> "File paths to include in the zip archive"
        <REF> output_file_name </REF>: string "Path of the zip archive to create"
    [END_INPUTS]

    [OUTPUTS]
        <REF> success </REF>: boolean "True if zip creation and all file writes succeed; False if any exception is caught"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open output_file_name as a new zip archive in write mode using zipfile.ZipFile RESULT zip_file]
            [COMMAND Within the opened zip archive, for each file in files, write file into zip_file RESULT write_result]
            [COMMAND Return True after all files are successfully added to the zip archive RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception occurs while opening the zip archive or writing any file]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
