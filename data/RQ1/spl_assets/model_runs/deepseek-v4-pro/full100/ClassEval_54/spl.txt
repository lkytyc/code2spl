[DEFINE_WORKER: "Creates and returns a two-dimensional board whose dimensions come from self.BOARD_SIZE and whose cells are randomly selected from self.ICONS." create_board]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> board </REF>: list[list[str]] "The generated two-dimensional board with rows from self.BOARD_SIZE[0] and columns from self.BOARD_SIZE[1], where each cell is an element from self.ICONS randomly chosen by random.choice."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign board to a nested list comprehension: the outer comprehension iterates over range(self.BOARD_SIZE[0]) to determine rows, the inner comprehension iterates over range(self.BOARD_SIZE[1]) to determine columns, and each cell is evaluated as random.choice(self.ICONS). If self.BOARD_SIZE[0] equals 0, board becomes an empty list. If self.BOARD_SIZE[1] equals 0, each row becomes an empty list. If self.ICONS is empty and at least one cell is generated, random.choice raises IndexError. RESULT board]
            [COMMAND Return board. RESULT board]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether moving from pos1 to pos2 is valid by checking board bounds, ensuring the positions differ, verifying both positions hold the same icon, and confirming a path exists between them." is_valid_move]
    [INPUTS]
        <REF> pos1 </REF>: tuple[int, int] or list[int, int] "First board position, unpacked into x1 and y1."
        <REF> pos2 </REF>: tuple[int, int] or list[int, int] "Second board position, unpacked into x2 and y2."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: bool "Returns True when the move is valid; returns False when any guard check fails."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack pos1 into x1 and y1, and pos2 into x2 and y2, where each position must contain exactly two elements. RESULT unpacked_coordinates]
            [COMMAND Check whether all four coordinates are within board bounds: 0 <= x1 < self.BOARD_SIZE[0], 0 <= y1 < self.BOARD_SIZE[1], 0 <= x2 < self.BOARD_SIZE[0], and 0 <= y2 < self.BOARD_SIZE[1]. RESULT bounds_valid]
            [COMMAND Check whether pos1 and pos2 represent different board positions. RESULT positions_different]
            [COMMAND Check whether the icons at self.board[x1][y1] and self.board[x2][y2] are the same. RESULT icons_match]
            [COMMAND Check whether a valid path exists between pos1 and pos2 by calling self.has_path(pos1, pos2). RESULT path_exists]
            [COMMAND Return True because all validation checks passed. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any of the four coordinates is outside the board bounds: not (0 <= x1 < self.BOARD_SIZE[0] and 0 <= y1 < self.BOARD_SIZE[1] and 0 <= x2 < self.BOARD_SIZE[0] and 0 <= y2 < self.BOARD_SIZE[1]).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: pos1 equals pos2.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The icons at the two positions differ: self.board[x1][y1] != self.board[x2][y2].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: There is no valid path between pos1 and pos2: self.has_path(pos1, pos2) returns False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether pos2 is reachable from pos1 by performing an iterative depth-first search over orthogonally adjacent same-valued board cells within board bounds." has_path]
    [INPUTS]
        <REF> pos1 </REF>: tuple[int, int] "Starting board position."
        <REF> pos2 </REF>: tuple[int, int] "Target board position to reach."
    [END_INPUTS]

    [OUTPUTS]
        <REF> has_path_result </REF>: boolean "True if pos2 is reachable from pos1 through a connected same-valued component; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize visited to a new empty set. RESULT visited]
            [COMMAND Initialize stack to a new list containing pos1. RESULT stack]
            [COMMAND While stack is not empty, pop the last element from stack into current_pos. RESULT current_pos]
            [COMMAND Within the while loop, when current_pos is not equal to pos2 and current_pos is not already in visited, add current_pos to visited. RESULT visited]
            [COMMAND Within the same while-loop processing branch, unpack current_pos into x and y. RESULT x, y]
            [COMMAND Within the same while-loop processing branch, for each (dx, dy) in [(0, 1), (0, -1), (1, 0), (-1, 0)], compute new_x = x + dx and new_y = y + dy. RESULT new_x, new_y]
            [COMMAND Within the for loop over directions, check whether 0 <= new_x < self.BOARD_SIZE[0] and 0 <= new_y < self.BOARD_SIZE[1]. RESULT neighbor_in_bounds]
            [COMMAND Within the for loop over directions and only when neighbor_in_bounds is true, check whether (new_x, new_y) is not in visited. RESULT neighbor_unvisited]
            [COMMAND Within the for loop over directions, within the boundary check, and only when neighbor_unvisited is true, check whether self.board[new_x][new_y] equals self.board[x][y]. RESULT neighbor_same_value]
            [COMMAND Within the same for-loop iteration and boundary check, when neighbor_in_bounds, neighbor_unvisited, and neighbor_same_value are all true, append (new_x, new_y) to stack. RESULT stack]
            [COMMAND When the while loop ends without having returned True, return False. RESULT has_path_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the while loop, current_pos equals pos2.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True immediately, terminating the search. RESULT has_path_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the while loop, current_pos is already in visited.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Continue to the next while iteration without processing current_pos. RESULT skipped_pos]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes two icons from the board by assigning blank spaces to the cells at the given positions." remove_icons]
    [INPUTS]
        <REF> pos1 </REF>: tuple "Coordinates (row, column) of the first icon to remove."
        <REF> pos2 </REF>: tuple "Coordinates (row, column) of the second icon to remove."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the first element of pos1 to x1. RESULT x1]
            [COMMAND Assign the second element of pos1 to y1. RESULT y1]
            [COMMAND Assign the first element of pos2 to x2. RESULT x2]
            [COMMAND Assign the second element of pos2 to y2. RESULT y2]
            [COMMAND Set the board cell at row x1 and column y1 to a blank space. RESULT self.board[x1][y1]]
            [COMMAND Set the board cell at row x2 and column y2 to a blank space. RESULT self.board[x2][y2]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the board row by row and cell by cell, returning False immediately if any non-space icon is found and True only if every cell in every row is the space character." is_game_over]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: boolean "True when the board is completely empty or has no rows; False when at least one cell contains a non-space icon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each row in self.board from top to bottom. RESULT current_row]
            [COMMAND Within the outer for loop over each row, evaluate any(icon != ' ' for icon in current_row) from left to right with short-circuiting: stop at the first icon that is not exactly the space character ' '. RESULT row_has_non_space]
            [COMMAND If row_has_non_space is True for the current row, follow the alternative early-return path and do not continue to the next row or final return. RESULT early_return_decision]
            [COMMAND If the outer loop completes without any row triggering the early return, return True because every inspected cell was the space character ' '. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During any row iteration, any(icon != ' ' for icon in row) evaluates to True because at least one cell in that row is not exactly the space character ' '.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Immediately return False, terminating both the inner row check and the outer row loop without examining remaining cells or later rows. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
