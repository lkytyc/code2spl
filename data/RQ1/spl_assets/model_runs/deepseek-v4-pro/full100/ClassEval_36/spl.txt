# CLASS EmailClient

[DEFINE_WORKER: "Initializes a new instance by assigning the provided address and capacity and initializing an empty inbox list." __init__]
    [INPUTS]
        <REF> addr </REF>: any "The address value passed to the constructor and stored as self.addr."
        <REF> capacity </REF>: any "The capacity value passed to the constructor and stored as self.capacity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter addr to the instance attribute self.addr. RESULT self.addr]
            [COMMAND Assign the input parameter capacity to the instance attribute self.capacity. RESULT self.capacity]
            [COMMAND Initialize the instance attribute self.inbox to a new empty list. RESULT self.inbox]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sends an email to the receiver's inbox when capacity allows, otherwise clears the inbox and returns failure." send_to]
    [INPUTS]
        <REF> recv </REF>: object "Receiver object containing inbox storage, address, and capacity-checking behavior."
        <REF> content </REF>: string "Content of the email to be delivered."
        <REF> size </REF>: integer "Size of the email used for capacity checking."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the email was delivered to the inbox; False if the inbox was full and cleared."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the condition by calling recv.is_full_with_one_more_email(size) and negating the result to determine whether the receiver can accept the email. RESULT can_receive]
            [COMMAND If can_receive is true, create a timestamp string from the current date and time in the format "%Y-%m-%d %H:%M:%S". RESULT timestamp]
            [COMMAND If can_receive is true, build an email dictionary with keys sender as self.addr, receiver as recv.addr, content as content, size as size, time as timestamp, and state as "unread". RESULT email]
            [COMMAND If can_receive is true, append the email dictionary to recv.inbox. RESULT inbox_updated]
            [COMMAND If can_receive is true, return True. RESULT delivery_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If can_receive is false because recv.is_full_with_one_more_email(size) is true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.clear_inbox(size) to clear inbox space. RESULT clear_result]
            [COMMAND Return False. RESULT delivery_status]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Fetches the first unread inbox item, marks it as read, and returns it; returns None if the inbox is empty or no unread item exists." fetch]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object or null "The first inbox item whose state was 'unread', after its state is set to 'read'; otherwise null if the inbox is empty or no unread item exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For i from 0 through len(self.inbox) - 1 inclusive, iterate over each index of self.inbox. RESULT current_index]
            [COMMAND Within the for loop over i, if self.inbox[i]['state'] equals "unread", set self.inbox[i]['state'] to "read" and return self.inbox[i]. RESULT returned_item]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If len(self.inbox) equals 0 before the loop starts.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None immediately. RESULT empty_inbox_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If the for loop completes and no item with state exactly equal to "unread" was found.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None after the loop. RESULT no_unread_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether adding one more email of the given size would exceed the current capacity." is_full_with_one_more_email]
    [INPUTS]
        <REF> size </REF>: integer "The size of the additional email to check against the current occupied size and capacity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> is_full </REF>: boolean "True if adding an email of the given size would exceed capacity; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_occupied_size() and assign the returned value to occupied_size. RESULT occupied_size]
            [COMMAND Return True if occupied_size + size is greater than self.capacity; otherwise return False. RESULT is_full]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total occupied size by summing the size of every email in the inbox." get_occupied_size]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> occupied_size </REF>: integer "Total sum of the sizes of all emails in the inbox."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize occupied_size to 0. RESULT occupied_size]
            [COMMAND For each email in self.inbox, add email["size"] to occupied_size. RESULT occupied_size]
            [COMMAND Return the final occupied_size. RESULT occupied_size]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Clears emails from the front of the inbox until the accumulated size reaches or exceeds the requested size or the inbox is empty, returning immediately when the address list is empty." clear_inbox]
    [INPUTS]
        <REF> size </REF>: integer "The amount of space that must be freed by removing emails from the inbox."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check the early-return condition: if `len(self.addr)` is equal to 0, follow the alternative flow and return immediately. RESULT early_return_check]
            [COMMAND Initialize `freed_space` to 0. RESULT freed_space]
            [COMMAND Enter a while loop that continues only while `freed_space` is strictly less than `size` and `self.inbox` is non-empty, using short-circuit evaluation of the `and` operator. RESULT loop_condition]
            [COMMAND Within the while loop, read the first email from `self.inbox` and assign it to `email`. RESULT email]
            [COMMAND Within the while loop, add `email['size']` to `freed_space`. RESULT freed_space]
            [COMMAND Within the while loop, delete the first element from `self.inbox` using `del self.inbox[0]`. RESULT inbox]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: `len(self.addr)` is equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return immediately without freeing any space or removing any emails. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
