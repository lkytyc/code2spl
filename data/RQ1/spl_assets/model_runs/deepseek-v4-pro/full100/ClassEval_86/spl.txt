# CLASS TicTacToe

[DEFINE_WORKER: "Initialize a three-row board with N columns per row, defaulting N to 3, fill every cell with a blank space, and set the current player to X." __init__]
    [INPUTS]
        <REF> self </REF>: object "The current object instance being initialized."
        <REF> N </REF>: int "Number of columns in each of the three board rows; defaults to 3 when omitted."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign to self.board a newly created list of 3 rows, where each row is a list of N cells initialized to the single space character ' '; if N is less than or equal to 0, range(N) is empty and each row is an empty list. RESULT self.board]
            [COMMAND Then assign the constant string 'X' to self.current_player. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to place the current player's mark at the requested board cell if it is empty, switches the current player on success, and returns a boolean indicating whether the move was accepted." make_move]
    [INPUTS]
        <REF> row </REF>: int "Row index of the board cell to update."
        <REF> col </REF>: int "Column index of the board cell to update."
    [END_INPUTS]

    [OUTPUTS]
        <REF> move_succeeded </REF>: bool "True when the cell was empty and the move was applied; False when the cell was already occupied."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.board[row][col] equals exactly the empty space character ' '. RESULT is_empty_cell]
            [COMMAND If is_empty_cell is true, set self.board[row][col] to self.current_player. RESULT updated_board]
            [COMMAND If is_empty_cell is true, toggle self.current_player: if self.current_player equals 'X', set it to 'O'; otherwise set it to 'X'. RESULT updated_current_player]
            [COMMAND If is_empty_cell is true, return True immediately. RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: is_empty_cell is false, meaning self.board[row][col] is any value other than the empty space character ' '.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately from the occupied-cell branch. RESULT failure_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines the winner of a 3x3 Tic-Tac-Toe board by checking rows, columns, and both diagonals for three equal non-space marks and returns the winning mark or None." check_winner]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> winner </REF>: string | null "The winning player mark if a row, column, or diagonal contains three equal non-space cells; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each row in self.board, if row[0], row[1], and row[2] are all equal and not ' ', set winner to row[0] and return immediately. RESULT winner]
            [COMMAND For each column index col from 0 to 2 inclusive, if self.board[0][col], self.board[1][col], and self.board[2][col] are all equal and not ' ', set winner to self.board[0][col] and return immediately. RESULT winner]
            [COMMAND If self.board[0][0], self.board[1][1], and self.board[2][2] are all equal and not ' ', set winner to self.board[0][0] and return immediately. RESULT winner]
            [COMMAND If self.board[0][2], self.board[1][1], and self.board[2][0] are all equal and not ' ', set winner to self.board[0][2] and return immediately. RESULT winner]
            [COMMAND If none of the preceding row, column, or diagonal checks returned a winner, set winner to None and return it. RESULT winner]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the board is full by returning False if any row contains a space, otherwise returning True." is_board_full]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> is_full_result </REF>: bool "True when the board is full; False when any row contains a space."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board. RESULT current_row]
            [COMMAND Within the loop over each row, test whether the current row contains the space character ' '. RESULT row_has_space]
            [COMMAND After the loop completes without finding a row that contains a space, return True. RESULT is_full_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the loop, the current row contains the space character ' '.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately and stop processing any remaining rows. RESULT is_full_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
