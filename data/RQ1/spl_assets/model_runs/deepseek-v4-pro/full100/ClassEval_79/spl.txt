# CLASS SQLGenerator

[DEFINE_WORKER: "Initializes the instance by assigning the provided table_name parameter to the instance attribute self.table_name." __init__]
    [INPUTS]
        <REF> table_name </REF>: str "The name of the table to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the table_name parameter to the instance attribute self.table_name. RESULT self.table_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds a SELECT SQL statement for the table, using '*' when fields is exactly None or a comma-separated field list otherwise, appending a WHERE clause when a condition is provided, and ending the statement with a semicolon." select]
    [INPUTS]
        <REF> fields </REF>: list[str] | None "Optional iterable of field names to select; None selects all fields with '*'."
        <REF> condition </REF>: str | None "Optional WHERE condition to append to the SQL statement; None means no WHERE clause is added."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The generated SELECT SQL statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether fields is exactly None. If it is exactly None, set fields to the literal string "*". Otherwise, join the elements of fields with ", " and assign the resulting comma-separated string to fields. RESULT fields]
            [COMMAND Construct sql as the formatted string "SELECT {fields} FROM {self.table_name}" using the current values of fields and self.table_name. RESULT sql]
            [COMMAND If condition is not None, append " WHERE " followed by the condition value to sql. If condition is None, leave sql unchanged. RESULT sql]
            [COMMAND Append a trailing semicolon to sql and return the resulting SQL statement. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs and returns a SQL INSERT statement for a given dictionary of column names and values, using the instance table name and single-quoted string values." insert]
    [INPUTS]
        <REF> data </REF>: dict "Dictionary whose keys are column names and whose values are the row values to insert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> sql_statement </REF>: str "The constructed SQL INSERT statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build fields as a comma-and-space separated string from the keys of data; if data is empty, fields becomes an empty string. RESULT fields]
            [COMMAND Build values as a comma-and-space separated string where each value from data is converted to a string, wrapped in single quotes, and not escaped; if data is empty, values becomes an empty string. RESULT values]
            [COMMAND Compose sql using the format INSERT INTO {self.table_name} ({fields}) VALUES ({values}). RESULT sql]
            [COMMAND Return sql concatenated with a trailing semicolon. RESULT sql_statement]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Builds and returns a complete SQL UPDATE statement for the provided data mappings and condition using the instance's table name." update]
    [INPUTS]
        <REF> data </REF>: dict "Mapping of column names to values used to build the SET clause."
        <REF> condition </REF>: str "SQL WHERE condition appended to the UPDATE statement."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The generated SQL UPDATE statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the list comprehension over data.items(), for each field and value pair, format it as field = 'value' and collect the formatted strings into a list. RESULT formatted_pairs]
            [COMMAND Join the formatted_pairs list with ", " to create the SET clause; if formatted_pairs is empty, set_clause is an empty string. RESULT set_clause]
            [COMMAND Build sql using an f-string with the pattern UPDATE {self.table_name} SET {set_clause} WHERE {condition}. RESULT sql]
            [COMMAND Return sql with a trailing semicolon appended. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs and returns a SQL DELETE statement string by interpolating the instance's table_name and the provided condition, then appending a semicolon." delete]
    [INPUTS]
        <REF> condition </REF>: string "SQL WHERE condition used to select rows to delete."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "Generated SQL DELETE statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Using Python f-string interpolation semantics, interpolate self.table_name and condition into 'DELETE FROM {self.table_name} WHERE {condition}' and assign the result to sql. RESULT sql]
            [COMMAND Return the value of sql concatenated with the string ';'. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Selects female records whose age is strictly less than the given age threshold by building a textual condition and delegating to self.select." select_female_under_age]
    [INPUTS]
        <REF> age </REF>: number "Exclusive upper age threshold whose current value is interpolated into the condition string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: unknown "The value returned by self.select when called with the generated condition string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a condition string by interpolating the current value of age into the f-string "age < {age} AND gender = 'female'"; the comparison is strict (<), not <=, and the resulting text is a snapshot. RESULT condition]
            [COMMAND Call self.select with the keyword argument condition=condition and return the returned value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs an age BETWEEN condition from the provided min_age and max_age values and returns the result of self.select with that condition." select_by_age_range]
    [INPUTS]
        <REF> min_age </REF>: object "The lower bound value used in the age BETWEEN condition."
        <REF> max_age </REF>: object "The upper bound value used in the age BETWEEN condition."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: object "The value returned by self.select when called with the constructed condition."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Interpolate the provided min_age and max_age values into the string 'age BETWEEN {min_age} AND {max_age}' to construct the condition string. RESULT condition]
            [COMMAND Return the result of calling self.select with the keyword argument condition set to the constructed condition string. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
