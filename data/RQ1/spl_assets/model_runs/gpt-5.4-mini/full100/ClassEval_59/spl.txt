# CLASS MovieBookingSystem

[DEFINE_WORKER: "Initializes the instance by creating an empty movies list." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance attribute self.movies to a new empty list. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create a movie dictionary with the provided name, price, parsed start and end times, and a zero-initialized square seating matrix, then append it to the movie list." add_movie]
    [INPUTS]
        <REF> self </REF>: object "Instance that owns the movies collection."
        <REF> name </REF>: string "Movie name to store in the new dictionary."
        <REF> price </REF>: number "Movie price to store in the new dictionary."
        <REF> start_time </REF>: string "Start time string expected to match the '%H:%M' format."
        <REF> end_time </REF>: string "End time string expected to match the '%H:%M' format."
        <REF> n </REF>: integer "Size of the square seating matrix to allocate."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new movie dictionary with the provided name and price values. RESULT movie]
            [COMMAND Parse the start_time string using the exact '%H:%M' format and store the resulting datetime value in the movie dictionary. RESULT movie.start_time]
            [COMMAND Parse the end_time string using the exact '%H:%M' format and store the resulting datetime value in the movie dictionary. RESULT movie.end_time]
            [COMMAND Allocate an n by n NumPy array filled with zeros and store it as the seats field in the movie dictionary. RESULT movie.seats]
            [COMMAND Append the completed movie dictionary to self.movies. RESULT self.movies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The start_time or end_time string does not match the required '%H:%M' format.]
        [LOG "Datetime parsing fails while constructing the movie dictionary."]
        [THROW ValueError "Raised by datetime.strptime when the input time string is malformed."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search for a movie by name and book each requested seat if it is available, otherwise return a failure message immediately." book_ticket]
    [INPUTS]
        <REF> self </REF>: object "The booking system instance containing the movie list."
        <REF> name </REF>: string "The movie name to search for."
        <REF> seats_to_book </REF>: list "A list of seat coordinates to reserve, where each seat is accessed by row and column indexes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Returns a success message, a failure message if any seat is already booked, or a not-found message if no matching movie exists."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each movie in self.movies. RESULT movie]
            [COMMAND For each movie, check whether movie['name'] equals the requested name. RESULT match_check]
            [COMMAND If the movie name matches, iterate over each seat in seats_to_book. RESULT seat]
            [COMMAND For each requested seat, check whether movie['seats'][seat[0]][seat[1]] is 0, meaning the seat is available. RESULT availability_check]
            [COMMAND If the seat is available, set movie['seats'][seat[0]][seat[1]] to 1 to mark it as booked. RESULT seat_booked]
            [COMMAND If any requested seat is not available, return "Booking failed." immediately. RESULT failure_result]
            [COMMAND After all requested seats are successfully booked for the matching movie, return "Booking success." RESULT success_result]
            [COMMAND If no movie in self.movies matches the requested name, return "Movie not found." RESULT not_found_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A matching movie is found and one requested seat is already occupied.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Stop processing immediately and return "Booking failed." without booking any remaining seats. RESULT failure_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No movie with the requested name exists in self.movies.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Finish scanning all movies and return "Movie not found." RESULT not_found_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses the input time strings, scans all movies, and returns the names of those whose start and end times are fully contained within the inclusive time range." available_movies]
    [INPUTS]
        <REF> start_time </REF>: string "Start time in HH:MM format."
        <REF> end_time </REF>: string "End time in HH:MM format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of string "The names of movies whose start and end times fall within the inclusive time interval."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the input start_time string into a datetime value using the '%H:%M' format. RESULT start_time]
            [COMMAND Parse the input end_time string into a datetime value using the '%H:%M' format. RESULT end_time]
            [COMMAND Initialize an empty list to collect available movie names. RESULT available_movies]
            [COMMAND Within the outer loop over self.movies, examine each movie in turn. RESULT movie]
            [COMMAND Within the outer loop, if the current movie starts at or after the parsed start_time and ends at or before the parsed end_time, append the movie's name to available_movies. RESULT available_movies]
            [COMMAND Return the collected available_movies list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
