[DEFINE_WORKER: "Returns the cosine approximation computed by self.taylor with 50 terms and rounded to 10 decimal places." cos]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the taylor method used to compute the cosine approximation."
        <REF> x </REF>: number "Input value passed into the Taylor-series computation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The rounded cosine approximation returned to 10 decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.taylor with x and the fixed constant 50 to compute the intermediate cosine approximation. RESULT taylor_value]
            [COMMAND Round the intermediate value to 10 decimal places and return the rounded result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the factorial-like product of the input value by initializing an accumulator to 1 and repeatedly multiplying it by the current counter while decrementing the counter until it reaches 1." factorial]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> a </REF>: integer "Positive integer counter to multiply down to 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The accumulated product of all integers from the original value of a down to 1."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the accumulator b to 1. RESULT b]
            [COMMAND While a is not equal to 1, multiply b by the current value of a and then decrement a by 1. RESULT b, a]
            [COMMAND Return b as the final product. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes an alternating Taylor-series style approximation by converting the input angle from degrees to radians, then accumulating power-and-factorial terms with alternating subtraction and addition based on a parity counter." taylor]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the factorial method."
        <REF> x </REF>: number "Input angle value that is converted from degrees to radians before series evaluation."
        <REF> n </REF>: integer "Upper loop boundary; the loop iterates from 1 up to but not including n."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The accumulated series approximation stored in a."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a to 1, convert x from degrees to radians using x = x / 180 * pi, and initialize count to 1. RESULT a, x, count]
            [COMMAND For each k from 1 up to but not including n, compute the term (x ** (2 * k)) / self.factorial(2 * k). RESULT term]
            [COMMAND Within the loop, if count is odd, subtract the term from a; otherwise, add the term to a. RESULT a]
            [COMMAND Within the same loop iteration, increment count by 1. RESULT count]
            [COMMAND After completing all iterations, return a. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert the input angle from degrees to radians, then approximate its sine by summing a series until the term magnitude falls below 1e-15, and return the result rounded to 10 decimal places." sin]
    [INPUTS]
        <REF> self </REF>: object "Instance reference used to call the method."
        <REF> x </REF>: number "Input angle in degrees."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The sine approximation of the input angle, rounded to 10 decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the input angle from degrees to radians by updating x to x / 180 * pi. RESULT x]
            [COMMAND Initialize the accumulator g to 0. RESULT g]
            [COMMAND Initialize the current series term t to the converted x value. RESULT t]
            [COMMAND Initialize the term index n to 1. RESULT n]
            [COMMAND While the absolute value of t is at least 1e-15, add the current term t to g, increment n by 1, and update t to -t * x * x / (2 * n - 1) / (2 * n - 2). RESULT g]
            [COMMAND Return g rounded to 10 decimal places. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the tangent-like ratio sin(x) / cos(x) rounded to 10 decimal places when cosine is nonzero, otherwise return False to avoid division by zero." tan]
    [INPUTS]
        <REF> self </REF>: object "Instance providing cos(x) and sin(x) methods."
        <REF> x </REF>: number "Input angle or value passed to cosine and sine calculations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number|boolean "Rounded sine-to-cosine ratio as a float when cosine is nonzero, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.cos(x) is not equal to 0. RESULT condition]
            [COMMAND When the condition is true, compute result as self.sin(x) divided by self.cos(x). RESULT result]
            [COMMAND Return result rounded to 10 decimal places. RESULT return_value]
            [COMMAND When the condition is false, return False immediately without performing the division. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
