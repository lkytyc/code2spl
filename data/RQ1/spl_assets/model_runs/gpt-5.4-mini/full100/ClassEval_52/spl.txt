[DEFINE_WORKER: "Remove punctuation from a sentence, tokenize and POS-tag the words, then lemmatize each word using a tag-aware part-of-speech choice before returning the list of lemmas." lemmatize_sentence]
    [INPUTS]
        <REF> sentence </REF>: string "Input sentence to clean, tokenize, tag, and lemmatize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[string] "A list of lemmatized words in the same order as the tagged tokens."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect lemmatized words. RESULT lemmatized_words]
            [COMMAND Replace the input sentence with the result of removing punctuation from it by calling self.remove_punctuation(sentence). RESULT sentence]
            [COMMAND Tokenize the cleaned sentence into words using word_tokenize(sentence). RESULT words]
            [COMMAND Assign part-of-speech tags to the token list using pos_tag(words). RESULT tagged_words]
            [COMMAND Within the loop over each (word, tag) pair in tagged_words, if the tag starts with 'V', lemmatize the word as a verb using pos='v'. RESULT lemmatized_word]
            [COMMAND Within the loop over each (word, tag) pair in tagged_words, else if the tag starts with 'J', lemmatize the word as an adjective using pos='a'. RESULT lemmatized_word]
            [COMMAND Within the loop over each (word, tag) pair in tagged_words, else if the tag starts with 'R', lemmatize the word as an adverb using pos='r'. RESULT lemmatized_word]
            [COMMAND Within the loop over each (word, tag) pair in tagged_words, otherwise lemmatize the word with the default lemmatizer behavior and no explicit part of speech. RESULT lemmatized_word]
            [COMMAND Within the loop over each (word, tag) pair in tagged_words, append the computed lemmatized word to lemmatized_words. RESULT lemmatized_words]
            [COMMAND Return the completed list of lemmatized words. RESULT lemmatized_words]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the part-of-speech tags for a sentence after removing punctuation, tokenizing it, and tagging the tokens." get_pos_tag]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the punctuation removal helper."
        <REF> sentence </REF>: string "Input sentence to be cleaned, tokenized, and tagged."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[string] "A list of part-of-speech tags corresponding to the input sentence tokens."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect part-of-speech tags. RESULT pos_tags]
            [COMMAND Remove punctuation from the input sentence by calling the instance punctuation-cleaning helper, and replace the sentence with the cleaned text. RESULT sentence]
            [COMMAND Tokenize the cleaned sentence into words. RESULT words]
            [COMMAND Assign part-of-speech tags to the token list. RESULT tagged_words]
            [COMMAND Iterate over each tagged token in the tagged token list. RESULT tagged_word]
            [COMMAND For each tagged token, append its tag component to the part-of-speech tag list. RESULT pos_tags]
            [COMMAND Return the collected part-of-speech tag list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove all punctuation characters from the input sentence by translating them away with a deletion table." remove_punctuation]
    [INPUTS]
        <REF> sentence </REF>: string "Input text to have punctuation removed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The sentence with all characters in string.punctuation removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a translation table that deletes every character in string.punctuation. RESULT translation_table]
            [COMMAND Apply the translation table to the input sentence using translate() to remove punctuation characters. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
