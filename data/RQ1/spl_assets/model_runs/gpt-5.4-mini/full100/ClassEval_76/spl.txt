# CLASS SignInSystem

[DEFINE_WORKER: "Initializes the instance user registry as an empty dictionary for later storage of user data." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty dictionary to self.users to establish the default user container. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a username to the user collection only if it is not already present, returning false for duplicates and true after insertion." add_user]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the users collection."
        <REF> username </REF>: any "The username to add as a key in the users collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "False if the username already exists; true after adding a new username."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given username is already present as a key in self.users. RESULT exists]
            [COMMAND If the username already exists, return false immediately. RESULT False]
            [COMMAND If the username does not exist, add username to self.users with the value false. RESULT self.users[username]]
            [COMMAND After inserting the new username, return true. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The username is already present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip insertion into self.users and exit the method with false. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username exists in the user registry and, if so, marks it as signed in and returns success; otherwise it returns failure immediately." sign_in]
    [INPUTS]
        <REF> username </REF>: any "The username to look up in self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns True when the username exists and is marked in self.users; otherwise returns False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is not present in self.users. RESULT membership_check]
            [COMMAND If username is not present in self.users, return False immediately. RESULT False]
            [COMMAND Otherwise, set self.users[username] to True. RESULT self.users[username]]
            [COMMAND Return True after recording the username as active. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username exists in the user store and returns True only if the stored value for that username is truthy." check_sign_in]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the users mapping."
        <REF> username </REF>: any "The username key to look up in the users mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False when the username is missing or its stored value is falsy; otherwise True."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is present in self.users. RESULT membership_check]
            [COMMAND If username is not present in self.users, return False immediately. RESULT False]
            [COMMAND If username is present in self.users, evaluate the truthiness of self.users[username]. RESULT user_status_value]
            [COMMAND If self.users[username] is truthy, return True. RESULT True]
            [COMMAND If self.users[username] is falsy, return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True when all stored user values are truthy, including the empty case, and otherwise returns False." all_signed_in]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if all user values are truthy; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether all values in self.users are truthy using all(self.users.values()). RESULT all_users_truthy]
            [COMMAND If all_users_truthy is true, return True immediately. RESULT True]
            [COMMAND Otherwise, return False immediately. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a list of usernames for users whose signed_in status is falsy by scanning all entries in self.users." all_not_signed_in]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of usernames whose signed_in value is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect usernames of users who are not signed in. RESULT not_signed_in_users]
            [COMMAND Iterate over each username and signed_in status pair in self.users.items(). RESULT loop_over_users]
            [COMMAND For each user, if signed_in is falsy, append the username to not_signed_in_users. RESULT not_signed_in_users]
            [COMMAND After processing all users, return the not_signed_in_users list. RESULT not_signed_in_users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
