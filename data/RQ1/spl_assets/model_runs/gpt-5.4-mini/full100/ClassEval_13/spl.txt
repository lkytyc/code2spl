# CLASS BookManagement

[DEFINE_WORKER: "Initializes the instance inventory as an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.inventory with a new empty dictionary. RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a book quantity to the inventory, incrementing an existing entry when the title is already present or creating a new entry otherwise." add_book]
    [INPUTS]
        <REF> self </REF>: object "The instance whose inventory mapping is updated."
        <REF> title </REF>: string "The book title used as the inventory key."
        <REF> quantity </REF>: number "The amount to add for the given title; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the title key exists in the inventory mapping. RESULT _condition_result]
            [COMMAND If the title already exists in the inventory, increase its stored quantity by the given quantity. RESULT self.inventory[title]]
            [COMMAND Otherwise, create a new inventory entry for the title and set its quantity to the given quantity. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove the requested quantity of a book from inventory after validating availability, and delete the entry entirely when its remaining quantity reaches zero." remove_book]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
        <REF> title </REF>: string "Book title used as the inventory key."
        <REF> quantity </REF>: integer "Requested number of copies to remove from inventory."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the requested title is missing from the inventory or the available quantity is less than the requested quantity. RESULT availability_check]
            [COMMAND If the availability check is false, subtract the requested quantity from the stored inventory count for that title. RESULT updated_quantity]
            [COMMAND After updating the count, check whether the remaining quantity is exactly zero. RESULT zero_quantity_check]
            [COMMAND If the remaining quantity is zero, delete the title entry from the inventory. RESULT inventory_cleanup]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The title is not present in the inventory or the stored quantity is less than the requested quantity.]
        [LOG "The method raises False immediately instead of continuing when removal cannot be completed."]
        [THROW False "Raised as an error path to stop execution when the inventory does not contain enough copies."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current inventory stored on the instance." view_inventory]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The inventory value stored on the instance."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the inventory attribute from the current instance. RESULT inventory_value]
            [COMMAND Return the accessed inventory value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the stored quantity for a book title, or 0 when the title is not present in the inventory." view_book_quantity]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
        <REF> title </REF>: string "Book title used as the lookup key."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The quantity for the requested title, or 0 if the title is missing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given title is absent from the inventory mapping. RESULT membership_check]
            [COMMAND If the title is not in the inventory, return 0 immediately as an early exit. RESULT 0]
            [COMMAND If the title is present, return the quantity stored in the inventory for that title. RESULT self.inventory[title]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
