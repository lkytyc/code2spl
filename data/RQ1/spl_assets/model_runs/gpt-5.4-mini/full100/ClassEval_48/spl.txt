# CLASS IpUtil

[DEFINE_WORKER: "Validate whether the given string is a syntactically valid IPv4 address by attempting to parse it with socket.inet_pton and returning True on success or False if parsing raises socket.error." is_valid_ipv4]
    [INPUTS]
        <REF> ip_address </REF>: string "IPv4 address text to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the input is a valid IPv4 address, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to parse the input ip_address as an IPv4 address using socket.inet_pton with socket.AF_INET. RESULT parse_attempt]
            [COMMAND If the IPv4 parsing succeeds, return True immediately. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If socket.inet_pton raises socket.error while parsing the input as IPv4.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Treat the input as not a valid IPv4 address and return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a string is a valid IPv6 address by attempting to parse it with socket.inet_pton and returning True on success or False if parsing fails." is_valid_ipv6]
    [INPUTS]
        <REF> ip_address </REF>: string "The IPv6 address string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the input is a valid IPv6 address, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to parse the input ip_address as an IPv6 address using socket.inet_pton with socket.AF_INET6. RESULT parse_result]
            [COMMAND If parsing succeeds, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If socket.inet_pton raises socket.error while parsing the IPv6 address.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the socket.error exception and treat the input as an invalid IPv6 address. RESULT invalid_ipv6]
            [COMMAND Return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resolves a hostname from an IP address using reverse DNS lookup and returns None when the lookup fails with a socket hostname error." get_hostname]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address to resolve into a hostname."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The resolved hostname, or null if reverse lookup fails with a socket hostname error."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.gethostbyaddr with the provided ip_address and take the first value from the returned tuple as the hostname. RESULT hostname]
            [COMMAND Return the resolved hostname immediately. RESULT hostname]
            [COMMAND If socket.gethostbyaddr raises socket.herror, catch the exception and return null instead of propagating it. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
