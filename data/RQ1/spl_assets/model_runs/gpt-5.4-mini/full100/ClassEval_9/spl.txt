# CLASS BigNumCalculator

[DEFINE_WORKER: "Add two non-negative integer strings by zero-padding them to equal length, summing digits from right to left with carry, and returning the final decimal string." add]
    [INPUTS]
        <REF> num1 </REF>: string "First numeric string operand to add."
        <REF> num2 </REF>: string "Second numeric string operand to add."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decimal string representing the sum of the two inputs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute max_length as the larger of the lengths of num1 and num2. RESULT max_length]
            [COMMAND Pad num1 on the left with zeros to length max_length. RESULT num1]
            [COMMAND Pad num2 on the left with zeros to length max_length. RESULT num2]
            [COMMAND Initialize carry to 0 and initialize result as an empty list. RESULT carry, result]
            [COMMAND Within the loop over i from max_length - 1 down to 0 inclusive, add the digits at num1[i] and num2[i] together with carry to produce digit_sum. RESULT digit_sum]
            [COMMAND Within the same loop, update carry to digit_sum // 10. RESULT carry]
            [COMMAND Within the same loop, compute digit as digit_sum % 10. RESULT digit]
            [COMMAND Within the same loop, prepend the string form of digit to result. RESULT result]
            [COMMAND After the loop, if carry is greater than 0, prepend the string form of carry to result. RESULT result]
            [COMMAND Join all elements of result into a single string and return it. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Subtract two non-negative integer strings by normalizing their order and length, then computing the digit-wise difference with borrowing and trimming leading zeros." subtract]
    [INPUTS]
        <REF> num1 </REF>: string "First numeric string operand."
        <REF> num2 </REF>: string "Second numeric string operand."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The subtraction result as a string, optionally prefixed with '-' when the result is negative."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compare the lengths of num1 and num2, and if num1 is shorter, swap the operands and mark the result as negative. RESULT negative]
            [COMMAND If num1 is longer than num2, keep the operands in place and mark the result as non-negative. RESULT negative]
            [COMMAND If num1 and num2 have equal length, compare the string values; when num1 is smaller, swap the operands and mark the result as negative, otherwise mark it as non-negative. RESULT negative]
            [COMMAND Set max_length to the maximum of the current lengths of num1 and num2, then pad both strings on the left with zeros to that length. RESULT max_length]
            [COMMAND Initialize borrow to 0 and result to an empty list before digit-by-digit subtraction begins. RESULT borrow]
            [COMMAND Within the reverse for loop over i from max_length - 1 down to 0 inclusive, compute digit_diff as int(num1[i]) minus int(num2[i]) minus borrow. RESULT digit_diff]
            [COMMAND Within the reverse for loop, if digit_diff is negative, add 10 to digit_diff and set borrow to 1; otherwise set borrow to 0. RESULT borrow]
            [COMMAND Within the reverse for loop, insert the current digit_diff string at the front of result. RESULT result]
            [COMMAND While result has more than one digit and its first element is '0', remove the leading zero. RESULT result]
            [COMMAND If negative is true, insert '-' at the front of result. RESULT result]
            [COMMAND Join all elements of result into a single string and return it. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiply two non-negative integer strings digit by digit using reverse traversal, accumulate carries in a result array, trim leading zeros, and return the product as a string." multiply]
    [INPUTS]
        <REF> num1 </REF>: string "First non-negative integer represented as a string of digits."
        <REF> num2 </REF>: string "Second non-negative integer represented as a string of digits."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decimal product of num1 and num2, returned without leading zeros."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute len1 and len2 as the lengths of num1 and num2, then allocate result as a zero-filled list of length len1 + len2 to hold the maximum possible product size. RESULT len1, len2, result]
            [COMMAND Within the outer reverse for loop over i from len1 - 1 down to 0 inclusive, and the inner reverse for loop over j from len2 - 1 down to 0 inclusive, multiply the digits int(num1[i]) and int(num2[j]), compute p1 = i + j and p2 = i + j + 1, add the product to result[p2], then write the carry into result[p1] and the ones digit into result[p2]. RESULT result]
            [COMMAND Initialize start to 0 to mark the first candidate index in the result array. RESULT start]
            [COMMAND While start is less than len(result) - 1 and result[start] equals 0, increment start to skip leading zeros without removing the final zero if the entire value is zero. RESULT start]
            [COMMAND Convert the slice result[start:] to strings and join them into the final decimal output string. RESULT final_product_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
