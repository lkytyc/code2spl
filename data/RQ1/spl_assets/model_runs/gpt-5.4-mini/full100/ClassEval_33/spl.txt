# CLASS DiscountStrategy

[DEFINE_WORKER: "Initialize the object by storing the provided customer, cart, and promotion values and computing the cached total from the current object state." __init__]
    [INPUTS]
        <REF> customer </REF>: any "Customer value to store on the instance."
        <REF> cart </REF>: any "Cart value to store on the instance."
        <REF> promotion </REF>: any or null "Optional promotion value to store on the instance; defaults to None."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the customer parameter to the instance attribute customer. RESULT self.customer]
            [COMMAND Assign the cart parameter to the instance attribute cart. RESULT self.cart]
            [COMMAND Assign the promotion parameter to the instance attribute promotion. RESULT self.promotion]
            [COMMAND Call the instance total method using the current object state and assign its return value to the private instance attribute __total. RESULT self.__total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the cart total by summing each item's quantity multiplied by its price and stores the result on the instance before returning it." total]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The computed cart total stored in and returned from self.__total."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each item in self.cart and compute the product of item['quantity'] and item['price'] for every item. RESULT item_product]
            [COMMAND Sum all per-item products to produce the aggregate cart total; if self.cart is empty, the sum evaluates to 0. RESULT total_value]
            [COMMAND Assign the computed total value to self.__total. RESULT self.__total]
            [COMMAND Return self.__total. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the total price minus any computed promotion discount, using zero discount when no promotion is set." due]
    [INPUTS]
        <REF> self </REF>: object "The current instance containing total amount, optional promotion, and the due calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The final amount due after subtracting the applicable discount from the total."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.promotion is None. RESULT promotion_is_missing]
            [COMMAND If self.promotion is None, set discount to 0; otherwise, call self.promotion(self) and assign the returned value to discount. RESULT discount]
            [COMMAND Return self.__total minus discount. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a 5% promotion amount for orders whose customer fidelity is at least 1000, otherwise returns 0." FidelityPromo]
    [INPUTS]
        <REF> order </REF>: Order "The order object containing customer data and a total calculation method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The promotional discount amount, equal to 5% of the order total or 0 when the fidelity threshold is not met."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the customer's fidelity value from order.customer['fidelity'] and compare it to 1000 using a greater-than-or-equal-to threshold. RESULT fidelity_condition]
            [COMMAND If the customer's fidelity is at least 1000, compute order.total() multiplied by 0.05. RESULT discount_amount]
            [COMMAND If the customer's fidelity is below 1000, return 0. RESULT discount_amount]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total promotional discount for an order by summing 10% of each cart item's value when its quantity is at least 20." BulkItemPromo]
    [INPUTS]
        <REF> order </REF>: object "Order object containing a cart collection of items."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "Total discount amount accumulated across qualifying cart items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the discount total to 0 as the starting default value. RESULT discount]
            [COMMAND Within the outer loop over each item in order.cart, inspect each cart item one by one. RESULT item]
            [COMMAND Within the outer loop, for each item, check whether item['quantity'] is greater than or equal to 20. RESULT quantity_check]
            [COMMAND Within the outer loop, if the quantity check is true, add item['quantity'] multiplied by item['price'] multiplied by 0.1 to the running discount total. RESULT discount]
            [COMMAND After all cart items have been processed, return the final discount total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns 7% of the order total when the cart contains at least 10 distinct products, otherwise returns zero." LargeOrderPromo]
    [INPUTS]
        <REF> order </REF>: Order "Order object providing access to the cart items and total amount."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "Either 7% of the order total or 0 depending on the number of distinct products."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the cart items from the order. RESULT cart_items]
            [COMMAND Build a set of distinct product values from the cart items by extracting item['product'] from each item. RESULT distinct_products]
            [COMMAND Count the number of distinct products. RESULT distinct_product_count]
            [COMMAND If the number of distinct products is at least 10, compute 7% of order.total(). RESULT discount_value]
            [COMMAND Otherwise, use 0 as the result. RESULT discount_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cart contains at least 10 distinct products.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 7% of order.total() as the promotion value. RESULT discount_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The cart contains fewer than 10 distinct products.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 as the promotion value. RESULT discount_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
