# CLASS MovieTicketDB

[DEFINE_WORKER: "Initialize a database-backed instance by opening a SQLite connection, creating a cursor from it, and then creating the required table." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "The SQLite database name or path used to open the connection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using db_name and store the resulting connection object in self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection and store the resulting cursor object in self.cursor. RESULT self.cursor]
            [COMMAND Call self.create_table() to initialize the database schema after the connection and cursor have been created. RESULT self.create_table()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates the tickets table if it does not already exist and commits the schema change to the database." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a SQL CREATE TABLE IF NOT EXISTS statement through self.cursor to create the tickets table with columns id as INTEGER PRIMARY KEY, movie_name as TEXT, theater_name as TEXT, seat_number as TEXT, and customer_name as TEXT. RESULT table_creation_statement]
            [COMMAND Commit the database connection through self.connection to persist the table creation. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a ticket record into the tickets table using the provided movie, theater, seat, and customer values, then commits the transaction." insert_ticket]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database cursor and connection."
        <REF> movie_name </REF>: string "Movie name to store in the ticket record."
        <REF> theater_name </REF>: string "Theater name to store in the ticket record."
        <REF> seat_number </REF>: string or integer "Seat number to store in the ticket record."
        <REF> customer_name </REF>: string "Customer name to store in the ticket record."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the current values of movie_name, theater_name, seat_number, and customer_name as bound parameters for an SQL INSERT into the tickets table. RESULT ticket_insert]
            [COMMAND Execute the INSERT statement through self.cursor with the parameters in the exact placeholder order: movie_name, theater_name, seat_number, customer_name. RESULT cursor_result]
            [COMMAND Commit the database transaction through self.connection to make the inserted ticket record persistent. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search for all tickets matching the given customer name and return the full result set from the database cursor." search_tickets_by_customer]
    [INPUTS]
        <REF> customer_name </REF>: string "Customer name used as the SQL filter value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "All ticket rows fetched from the database cursor after executing the query."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL query against the tickets table to select all rows where customer_name equals the provided customer_name. RESULT query_execution]
            [COMMAND Fetch all remaining rows from the cursor after the query completes and store them in tickets. RESULT tickets]
            [COMMAND Return tickets as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes the ticket row matching the given ticket_id and commits the transaction to persist the change." delete_ticket]
    [INPUTS]
        <REF> ticket_id </REF>: any "Identifier of the ticket row to delete."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL DELETE against the tickets table using ticket_id as the bound id value. RESULT cursor_execution]
            [COMMAND Commit the database transaction to make the deletion permanent. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
