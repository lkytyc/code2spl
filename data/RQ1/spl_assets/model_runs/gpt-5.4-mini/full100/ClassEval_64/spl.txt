# CLASS NumberConverter

[DEFINE_WORKER: "Converts an integer decimal number to its binary string representation without the 0b prefix." decimal_to_binary]
    [INPUTS]
        <REF> decimal_num </REF>: integer "The decimal integer to convert to binary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The binary representation of the input integer with the 0b prefix removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the input decimal number to its binary string form using the built-in binary conversion. RESULT binary_num]
            [COMMAND Remove the leading 0b prefix by taking the substring starting at index 2. RESULT binary_num]
            [COMMAND Return the resulting binary string. RESULT binary_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert a binary string into its decimal integer equivalent and return the converted value." binary_to_decimal]
    [INPUTS]
        <REF> binary_num </REF>: str "Binary string to convert from base 2 to a decimal integer."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Decimal integer value obtained by converting the input binary string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the binary string input to an integer using base 2. RESULT decimal_num]
            [COMMAND Return the converted decimal integer value. RESULT decimal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: If the input is not a valid binary string for base-2 conversion.]
        [LOG "The base-2 conversion fails because the input cannot be parsed as binary."]
        [THROW ValueError "Raised by the integer conversion when the input is not valid binary text."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a decimal number to its octal string representation by using Python's oct function and removing the leading 0o prefix." decimal_to_octal]
    [INPUTS]
        <REF> decimal_num </REF>: number "The decimal value to convert to octal."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The octal digits of the input value without the 0o prefix."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the input decimal number to its octal string representation. RESULT oct(decimal_num)]
            [COMMAND Remove the leading 0o prefix from the octal string by slicing from index 2 onward. RESULT octal_num]
            [COMMAND Return the resulting octal digit string. RESULT octal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an octal-formatted input string into its decimal integer value and returns the converted result." octal_to_decimal]
    [INPUTS]
        <REF> octal_num </REF>: string "An octal-formatted number to convert from base 8."
    [END_INPUTS]

    [OUTPUTS]
        <REF> decimal_num </REF>: int "The integer value obtained by converting the octal input to base 10."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the input octal string to an integer using base 8 and store the result in decimal_num. RESULT decimal_num]
            [COMMAND Return decimal_num as the function result. RESULT decimal_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The input string is not a valid base-8 number and int(octal_num, 8) cannot parse it.]
        [LOG "Base-8 conversion failed for the provided octal input."]
        [THROW ValueError "Raised when the octal input is invalid for conversion to an integer."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an integer-like decimal value to a hexadecimal string without the 0x prefix." decimal_to_hex]
    [INPUTS]
        <REF> decimal_num </REF>: integer-like "Input value accepted by Python's hex() function."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Hexadecimal string representation of decimal_num with the 0x prefix removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert decimal_num to a hexadecimal string using hex(decimal_num). RESULT hex_num_with_prefix]
            [COMMAND Remove the first two characters from the hexadecimal string to strip the 0x prefix. RESULT hex_num]
            [COMMAND Return the prefix-stripped hexadecimal string. RESULT hex_num]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a hexadecimal string into its decimal integer equivalent by parsing it with base 16 and returning the computed integer." hex_to_decimal]
    [INPUTS]
        <REF> hex_num </REF>: string "Hexadecimal string to convert to an integer."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Decimal integer value produced from the hexadecimal input."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse hex_num as a base-16 integer and assign the result to decimal_num. RESULT decimal_num]
            [COMMAND Return decimal_num as the decimal conversion result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
