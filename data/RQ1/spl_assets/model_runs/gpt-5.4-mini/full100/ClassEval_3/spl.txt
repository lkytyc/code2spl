[DEFINE_WORKER: "Returns n factorial when m is not provided or equals n, otherwise returns the arrangement-style factorial quotient n! / (n-m)!." count]
    [INPUTS]
        <REF> n </REF>: integer "Total size used to compute the factorial-based result."
        <REF> m </REF>: integer or null "Optional selected size; when omitted or equal to n, the full factorial is returned."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The computed factorial or factorial quotient for the requested arrangement count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether m is not provided or n equals m, using a short-circuit OR condition. RESULT condition_result]
            [COMMAND If the condition is true, return ArrangementCalculator.factorial(n) as the full factorial result. RESULT full_factorial_result]
            [COMMAND Otherwise, return ArrangementCalculator.factorial(n) divided by ArrangementCalculator.factorial(n - m) using integer division. RESULT arrangement_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m is None or n == m evaluates to true]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the full factorial of n without using the quotient branch. RESULT full_factorial_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: m is provided and n != m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n! and divide it by (n - m)! with integer division to produce the arrangement count. RESULT arrangement_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total count by summing ArrangementCalculator.count(n, i) for every i from 1 through n and returning the accumulated result." count_all]
    [INPUTS]
        <REF> n </REF>: integer "The upper bound used to iterate over possible i values and as the first argument to ArrangementCalculator.count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The total sum of ArrangementCalculator.count(n, i) across all i from 1 to n inclusive."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND For each i from 1 through n inclusive, add ArrangementCalculator.count(n, i) to total. RESULT total]
            [COMMAND Return total. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate all permutations of the stored data with length m, defaulting m to the full data length, and return them as a list of lists." select]
    [INPUTS]
        <REF> m </REF>: int or None "Number of elements to include in each permutation; if None, the method uses the length of datas."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list containing each permutation of datas converted to a list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If m is None, set m to the length of self.datas. RESULT m]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND For each permutation produced by itertools.permutations over self.datas with length m, convert the permutation to a list and append it to result. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a combined list by iterating from 1 through the number of data items and extending the result with each per-index selection." select_all]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list containing the concatenation of the selections produced for every index from 1 to len(self.datas)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to accumulate the combined selections. RESULT result]

    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the factorial of a nonnegative integer by initializing an accumulator to 1, multiplying it by each integer from 2 through n, and returning the final product." factorial]
    [INPUTS]
        <REF> n </REF>: integer "The integer whose factorial is to be computed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The factorial value of n."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to 1 as the starting accumulator value. RESULT result]
            [COMMAND For each integer i from 2 through n inclusive, multiply result by i to accumulate the running product. RESULT result]
            [COMMAND Return the final value stored in result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
