# CLASS DocFileHandler

[DEFINE_WORKER: "Initializes the instance by storing the provided file path on the object." __init__]
    [INPUTS]
        <REF> file_path </REF>: text "Path value to be assigned to the instance file_path attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided file_path value directly to the instance attribute self.file_path. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads all paragraphs from the document at self.file_path and returns their text joined together with newline separators." read_text]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the file_path used to open the document."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The combined paragraph text from the document, separated by newline characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Document from self.file_path and store it in doc. RESULT doc]
            [COMMAND Initialize text as an empty list. RESULT text]
            [COMMAND Iterate over each paragraph in doc.paragraphs and append paragraph.text to text. RESULT text]
            [COMMAND Join all strings in text with newline separators and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a new document, writes the provided text with the requested font size and paragraph alignment, saves it to the instance file path, and returns True unless any error occurs." write_text]
    [INPUTS]
        <REF> self </REF>: object "Instance providing file_path and alignment mapping helper."
        <REF> content </REF>: string "Text content to write into the document."
        <REF> font_size </REF>: number "Font size to apply to the written text; defaults to 12."
        <REF> alignment </REF>: string "Paragraph alignment name to convert through the internal alignment helper; defaults to left."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when the document is created, populated, saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new Document object and store it in doc. RESULT doc]
            [COMMAND Add a new paragraph to doc and store it in paragraph. RESULT paragraph]
            [COMMAND Add a run containing content to paragraph and store it in run. RESULT run]
            [COMMAND Access the font of run and store it in font. RESULT font]
            [COMMAND Set font.size to Pt(font_size). RESULT font.size]
            [COMMAND Convert alignment by calling the internal alignment helper with alignment and store the result in alignment_value. RESULT alignment_value]
            [COMMAND Set paragraph.alignment to alignment_value. RESULT paragraph.alignment]
            [COMMAND Save doc to self.file_path. RESULT document_saved]
            [COMMAND Return True after the document is saved successfully. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during document creation, formatting, saving, or alignment conversion, the method catches it and returns a default failure value.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the remaining document writing steps and return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a heading at the requested level to the Word document at the stored file path and saves the document, returning True on success and False if any exception occurs." add_heading]
    [INPUTS]
        <REF> heading </REF>: str "The heading text to insert into the document."
        <REF> level </REF>: int "The heading level to use, defaulting to 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the heading was added and the document was saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Document object from the stored file path. RESULT doc]
            [COMMAND Add the provided heading text to the document at the specified heading level. RESULT heading_added]
            [COMMAND Save the updated document back to the same stored file path. RESULT saved]
            [COMMAND Return True to indicate the operation completed successfully. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception is raised while opening the document, adding the heading, saving the file, or during any other part of the try block.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without propagating the exception. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a Word table from a 2D data array, fills each cell with stringified values, saves the document, and returns success unless any error occurs." add_table]
    [INPUTS]
        <REF> data </REF>: list "Two-dimensional table data used to determine row and column counts and populate cell contents."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the table was added and the document was saved successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open the document at the current file path and create a table with a number of rows equal to the length of data and a number of columns equal to the length of the first row in data. RESULT doc, table]
            [COMMAND Within the nested loops over each row index i and each cell index j, convert each cell value to a string and assign it to the corresponding table cell text. RESULT table]
            [COMMAND Save the document back to the current file path. RESULT saved_doc]
            [COMMAND Return True to indicate successful completion. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during document loading, table creation, cell population, or saving, catch it and return False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without propagating the error. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the corresponding Word paragraph alignment constant for a case-insensitive alignment name, defaulting to left when the name is unknown or missing." _get_alignment_value]
    [INPUTS]
        <REF> alignment </REF>: string "Alignment name to resolve, compared case-insensitively."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: WD_PARAGRAPH_ALIGNMENT "The resolved paragraph alignment constant."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a lookup table that maps 'left' to WD_PARAGRAPH_ALIGNMENT.LEFT, 'center' to WD_PARAGRAPH_ALIGNMENT.CENTER, and 'right' to WD_PARAGRAPH_ALIGNMENT.RIGHT. RESULT alignment_options]
            [COMMAND Convert the input alignment name to lowercase and use it to look up the corresponding alignment constant in the table. RESULT resolved_alignment]
            [COMMAND If the lowercase name is not present in the table, return WD_PARAGRAPH_ALIGNMENT.LEFT as the default value. RESULT resolved_alignment]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
