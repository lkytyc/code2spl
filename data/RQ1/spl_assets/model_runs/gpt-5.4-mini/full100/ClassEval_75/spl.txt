# CLASS ShoppingCart

[DEFINE_WORKER: "Initializes the instance by creating an empty dictionary to store items." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new empty dictionary to self.items to initialize the items storage for the instance. RESULT self.items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Add or replace an item entry in self.items with the given price and quantity, using quantity defaulting to 1." add_item]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the items mapping."
        <REF> item </REF>: any "Item key to add or update in the items mapping."
        <REF> price </REF>: any "Price value to store for the item."
        <REF> quantity </REF>: any "Quantity to store for the item, defaulting to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item exists as a key in self.items. RESULT item_exists]
            [COMMAND Assign self.items[item] to a dictionary containing the provided price and quantity, regardless of whether the item already existed. RESULT self.items[item]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reduce the stored quantity for an existing item in self.items by the requested amount and do nothing when the item is missing." remove_item]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the items collection."
        <REF> item </REF>: any "Item key to remove quantity from if present."
        <REF> quantity </REF>: number "Amount to subtract from the item's stored quantity; defaults to 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item exists as a key in self.items. RESULT item_present]
            [COMMAND If item is present in self.items, subtract quantity from self.items[item]['quantity']. RESULT updated_quantity]
            [COMMAND If item is not present in self.items, take no action. RESULT no_change]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the current value of the instance attribute self.items without modification." view_items]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "The current value stored in self.items, returned without any transformation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of the instance attribute self.items. RESULT items_value]
            [COMMAND Return items_value directly to the caller. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the sum of quantity times price for every item stored in self.items." total_price]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The sum of quantity multiplied by price across all items; returns 0 when self.items is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over all values in self.items. RESULT item]
            [COMMAND For each item, multiply item['quantity'] by item['price'] to produce that item's contribution. RESULT item_total]
            [COMMAND Sum all item contributions together. RESULT total_price]
            [COMMAND Return the computed total price. RESULT total_price]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
