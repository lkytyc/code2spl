# CLASS IPAddress

[DEFINE_WORKER: "Initialize the instance by storing the provided IP address on the object." __init__]
    [INPUTS]
        <REF> ip_address </REF>: any "The IP address value to assign to the instance attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided ip_address value directly to the instance attribute self.ip_address. RESULT self.ip_address]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the instance IP address consists of exactly four numeric octets, each in the inclusive range from 0 to 255." is_valid]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the IP address is exactly four dot-separated numeric octets in the range 0 to 255 inclusive; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address on '.' and store the resulting substrings in octets. RESULT octets]
            [COMMAND If octets does not contain exactly 4 elements, return False immediately. RESULT False]
            [COMMAND For each octet in octets, check whether it contains only digits and its integer value is between 0 and 255 inclusive. RESULT validation_check]
            [COMMAND Within the octet loop, if any octet fails the digit check or falls outside the 0 to 255 inclusive range, return False immediately. RESULT False]
            [COMMAND If all four octets pass validation, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current IP address split into octets when the object is valid, otherwise returns an empty list." get_octets]
    [INPUTS]
        <REF> self </REF>: object "The current instance containing validation logic and the IP address value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of string octets from the IP address, or an empty list if the object is not valid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the current object is valid by calling self.is_valid(). RESULT is_valid]
            [COMMAND If self.is_valid() returns true, split self.ip_address on '.' and return the resulting list of octets immediately. RESULT octets]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately. RESULT []]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a dot-separated 8-bit binary string for each octet when the object is valid, otherwise returns an empty string." get_binary]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A dot-separated string of 8-bit binary octets, or an empty string if the object is invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the object is valid. RESULT is_valid]
            [COMMAND If the object is valid, initialize an empty list to collect binary octets. RESULT binary_octets]
            [COMMAND Within the validity check, iterate over each octet returned by get_octets(). RESULT octet]
            [COMMAND For each octet, convert it to an integer and format it as an 8-bit binary string using base 2 with zero padding to width 8. RESULT binary_octet]
            [COMMAND Append each formatted binary octet to the collection list. RESULT binary_octets]
            [COMMAND After processing all octets, join the collected binary octets with dots and return the joined string. RESULT result]
            [COMMAND If the object is not valid, return an empty string immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The object is not valid.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string without processing any octets. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
