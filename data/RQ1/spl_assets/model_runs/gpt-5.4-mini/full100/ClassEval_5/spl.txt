# CLASS AutomaticGuitarSimulator

[DEFINE_WORKER: "Initializes the instance by storing the provided text in play_text." __init__]
    [INPUTS]
        <REF> text </REF>: unknown "Text value to assign to the instance play_text field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "This initializer does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Receive the text argument and begin initialization of the instance. RESULT text]
            [COMMAND Assign the provided text value to the instance field play_text. RESULT self.play_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "This method parses the stored play text into a list of chord and tune pairs, returning an empty list for blank input and optionally displaying each parsed pair." interpret]
    [INPUTS]
        <REF> display </REF>: bool "If true, display each parsed chord and tune pair as it is produced."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of dict "A list of dictionaries, each containing 'Chord' and 'Tune' keys parsed from the input text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.play_text.strip() is empty; if so, return an empty list immediately. RESULT empty_result]
            [COMMAND Otherwise, initialize an empty list named play_list. RESULT play_list]
            [COMMAND Otherwise, split self.play_text on single spaces into play_segs. RESULT play_segs]
            [COMMAND Within the loop over each play_seg in play_segs, initialize pos to 0. RESULT pos]
            [COMMAND Within the loop over each play_seg in play_segs, scan characters from the start of play_seg; while the current character is alphabetic, increment pos and continue scanning, and stop at the first non-alphabetic character. RESULT pos]
            [COMMAND Within the loop over each play_seg in play_segs, slice play_seg[0:pos] into play_chord and play_seg[pos:] into play_value using the boundary found by the scan. RESULT play_chord_and_play_value]
            [COMMAND Within the loop over each play_seg in play_segs, append a dictionary with Chord set to play_chord and Tune set to play_value to play_list. RESULT play_list]
            [COMMAND Within the loop over each play_seg in play_segs, if display is true, call self.display with play_chord and play_value. RESULT display_call]
            [COMMAND After processing all segments, return play_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If self.play_text.strip() is empty, skip all parsing work and return an empty list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list without creating play_list or splitting input. RESULT empty_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats and returns a string that includes the provided key and value in a fixed guitar-playing message." display]
    [INPUTS]
        <REF> self </REF>: object "Instance receiving the method call."
        <REF> key </REF>: any "Value inserted into the first placeholder of the formatted string."
        <REF> value </REF>: any "Value inserted into the second placeholder of the formatted string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted message containing the key and value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a formatted string using the fixed template "Normal Guitar Playing -- Chord: %s, Play Tune: %s" with key inserted into the first placeholder and value inserted into the second placeholder. RESULT formatted_string]
            [COMMAND Return the formatted string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
