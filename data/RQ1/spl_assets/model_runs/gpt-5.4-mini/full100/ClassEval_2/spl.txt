# CLASS ArgumentParser

[DEFINE_WORKER: "Initializes empty storage for arguments, required parameter names, and parameter types." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an empty dictionary for arguments. RESULT self.arguments]
            [COMMAND Create an empty set for required parameter names. RESULT self.required]
            [COMMAND Create an empty dictionary for parameter types. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses command-line arguments from a command string, storing long and short options as converted values or boolean flags and returning whether all required arguments are present." parse_arguments]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the command string on whitespace, discard the first token, and store the remaining tokens in args. RESULT args]
            [COMMAND Iterate over args by index from 0 to len(args) - 1, and assign the current token to arg. RESULT arg]
            [COMMAND If arg starts with "--", treat it as a long option, remove the leading "--", and split the remainder on '='. RESULT key_value]
            [COMMAND If the long option split produces exactly two parts, store the first part as the key in self.arguments and store the second part converted by _convert_type using the key. RESULT self.arguments[key_value[0]]]
            [COMMAND If the long option split does not produce exactly two parts, store the long option name as a key in self.arguments with the value True. RESULT self.arguments[key_value[0]]]
            [COMMAND Else if arg starts with "-", treat it as a short option and remove the leading "-" to get the key. RESULT key]
            [COMMAND Within the short option branch, if the next token exists and does not start with "-", convert that next token with _convert_type and store it under the short option key. RESULT self.arguments[key]]
            [COMMAND Within the short option branch, if there is no next token or the next token starts with "-", store the short option key in self.arguments with the value True. RESULT self.arguments[key]]
            [COMMAND After processing all tokens, compute missing_args as the set difference between self.required and the keys currently present in self.arguments. RESULT missing_args]
            [COMMAND If missing_args is not empty, return False together with missing_args. RESULT (False, missing_args)]
            [COMMAND If no required arguments are missing, return True together with None. RESULT (True, None)]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the value stored under the given key from the instance argument mapping." get_argument]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the argument mapping."
        <REF> key </REF>: any "The lookup key to retrieve from the argument mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The value associated with key in self.arguments, or None if the key is absent."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve and return the value associated with the given key from self.arguments. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds an argument to the required set when marked required and records its type in the type mapping." add_argument]
    [INPUTS]
        <REF> self </REF>: object "The instance whose argument metadata is being updated."
        <REF> arg </REF>: any "The argument name to register."
        <REF> required </REF>: bool "Whether the argument should be added to the required set."
        <REF> arg_type </REF>: type "The type associated with the argument, defaulting to str."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If required is true, add arg to self.required. RESULT self.required]
            [COMMAND Store arg_type in self.types under the key arg, overwriting any existing entry. RESULT self.types]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to convert a value using the type mapping for the given argument key and falls back to returning the original value when the conversion fails or the key is missing." _convert_type]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the type mapping used for conversion."
        <REF> arg </REF>: any "Key used to select the converter from the type mapping."
        <REF> value </REF>: any "Input value to be converted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The converted value, or the original value if conversion cannot be performed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to convert the input value by looking up the converter for the given argument key in the type mapping and applying it to the value. RESULT converted_value]
            [COMMAND If the conversion succeeds, return the converted result. RESULT converted_value]
            [COMMAND If a ValueError or KeyError occurs during lookup or conversion, return the original input value unchanged. RESULT value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
