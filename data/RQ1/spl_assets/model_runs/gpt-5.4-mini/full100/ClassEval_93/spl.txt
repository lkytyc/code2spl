# CLASS VectorUtil

[DEFINE_WORKER: "Compute the dot product of the normalized input vectors to produce a similarity-like score." similarity]
    [INPUTS]
        <REF> vector_1 </REF>: vector "First input vector to be normalized and compared."
        <REF> vector_2 </REF>: vector "Second input vector to be normalized and compared."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: numeric "The dot product of the two normalized vectors."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read vector_1 and vector_2 as the two input vectors. RESULT inputs]
            [COMMAND Normalize vector_1 by passing it to matutils.unitvec. RESULT normalized_vector_1]
            [COMMAND Normalize vector_2 by passing it to matutils.unitvec. RESULT normalized_vector_2]
            [COMMAND Compute the dot product of normalized_vector_1 and normalized_vector_2. RESULT similarity_score]
            [COMMAND Return similarity_score as the final result. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute cosine similarity scores between a single input vector and each vector in a collection by dividing their dot products by the product of their norms." cosine_similarities]
    [INPUTS]
        <REF> vector_1 </REF>: array-like "The reference vector used as the numerator and norm basis for similarity computation."
        <REF> vectors_all </REF>: array-like "A 2D collection of vectors whose similarities to vector_1 are computed row by row."
    [END_INPUTS]

    [OUTPUTS]
        <REF> similarities </REF>: array-like "An array of cosine-like similarity values, one per vector in vectors_all."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the L2 norm of vector_1. RESULT norm]
            [COMMAND Compute the L2 norm of each vector in vectors_all along axis 1. RESULT all_norms]
            [COMMAND Compute the dot product of every vector in vectors_all with vector_1. RESULT dot_products]
            [COMMAND Divide dot_products by the elementwise product of norm and all_norms to obtain cosine similarity values. RESULT similarities]
            [COMMAND Return similarities. RESULT similarities]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the cosine similarity between the mean vectors of two non-empty vector lists after validating that neither list is empty." n_similarity]
    [INPUTS]
        <REF> vector_list_1 </REF>: list "First list of vectors to average and normalize."
        <REF> vector_list_2 </REF>: list "Second list of vectors to average and normalize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The dot product of the normalized mean vectors, representing similarity."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate that both input lists are non-empty before any similarity computation. RESULT precondition_checked]
            [COMMAND Compute the mean vector of the first input list by converting it to an array and averaging along axis 0, then normalize that mean vector. RESULT normalized_mean_1]
            [COMMAND Compute the mean vector of the second input list by converting it to an array and averaging along axis 0, then normalize that mean vector. RESULT normalized_mean_2]
            [COMMAND Compute the dot product of the two normalized mean vectors and return it as the similarity score. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: At least one of the two input lists is empty.]
        [LOG "Raise ZeroDivisionError because similarity cannot be computed when either input list has no elements."]
        [THROW ZeroDivisionError "Indicates that at least one passed list is empty."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an index-to-key mapping from the input dictionary, compute smoothed inverse document frequency weights from the counts, and return a dictionary that assigns each original key its corresponding weight." compute_idf_weight_dict]
    [INPUTS]
        <REF> total_num </REF>: number "The total item count used as the smoothing numerator for the IDF-style weight calculation."
        <REF> number_dict </REF>: dict "A dictionary mapping keys to counts that will be converted into weighted values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "A dictionary mapping each original key to its computed smoothed logarithmic weight."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty dictionary to map sequential indices back to original keys. RESULT index_2_key_map]
            [COMMAND Initialize the index counter to 0. RESULT index]
            [COMMAND Initialize an empty list to collect counts in iteration order. RESULT count_list]
            [COMMAND For each key and count in number_dict.items(), store the key at the current index in index_2_key_map, append the count to count_list, and increment index by 1. RESULT index_2_key_map, count_list, index]
            [COMMAND Convert count_list into a NumPy array. RESULT a]
            [COMMAND Apply smoothed inverse-frequency weighting by computing log((total_num + 1) / (a + 1)). RESULT a]
            [COMMAND Initialize an empty result dictionary. RESULT result]
            [COMMAND Within the enumeration of the weighted array a, look up each original key by index from index_2_key_map and assign the corresponding weight into result. RESULT result]
            [COMMAND Return the result dictionary. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
