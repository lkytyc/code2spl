# CLASS GomokuGame

[DEFINE_WORKER: "Initialize a square game board of the given size, set every cell to empty space, and set the current player to X." __init__]
    [INPUTS]
        <REF> board_size </REF>: int "The size of both dimensions of the square board to create."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the provided board_size value in the board_size attribute. RESULT self.board_size]
            [COMMAND Create a board_size by board_size two-dimensional list filled with space characters and store it in the board attribute. RESULT self.board]
            [COMMAND Set the current_player attribute to the constant value X. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Places the current player's mark on an empty board cell, toggles the player, and returns True; otherwise it returns False without changing the board." make_move]
    [INPUTS]
        <REF> row </REF>: int "Row index of the target board cell."
        <REF> col </REF>: int "Column index of the target board cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was applied successfully; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the board cell at the given row and column is empty by comparing it to a single space character. RESULT cell_is_empty]
            [COMMAND If the cell is empty, write the current player's mark into that board position. RESULT board_updated]
            [COMMAND If the cell is empty, toggle current_player from X to O or from O to X. RESULT current_player]
            [COMMAND If the cell is empty, return True immediately to indicate the move succeeded. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The target board cell is not empty.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Leave the board unchanged and do not switch the current player. RESULT no_change]
            [COMMAND Return False to indicate the move was rejected. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scan the entire square board for any non-empty cell that forms five in a row in one of four directions and return the winning piece, or None if no winner exists." check_winner]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the four direction vectors used for win detection: horizontal, vertical, diagonal down-right, and diagonal down-left. RESULT directions]
            [COMMAND For each row index from 0 up to but not including self.board_size, iterate across the board. RESULT row]
            [COMMAND Within the outer row loop, for each column index from 0 up to but not including self.board_size, inspect the current board cell. RESULT col]
            [COMMAND Within the row and column loops, if self.board[row][col] is not a space character, treat that cell as a candidate piece. RESULT cell_value]
            [COMMAND Within the non-empty cell branch, for each direction in directions, check whether <SPL>_check_five_in_a_row</SPL> reports a five-in-a-row starting at the current row, column, and direction. RESULT has_five]
            [COMMAND If <SPL>_check_five_in_a_row</SPL> returns true for any direction, return the current board cell value immediately as the winner. RESULT winning_piece]
            [COMMAND If all rows, columns, and directions are exhausted without finding a five-in-a-row, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether there are five consecutive board cells with the same symbol starting from a given position and moving in a specified direction, returning false immediately on out-of-bounds or mismatched cells." _check_five_in_a_row]
    [INPUTS]
        <REF> row </REF>: int "Starting row index on the board."
        <REF> col </REF>: int "Starting column index on the board."
        <REF> direction </REF>: tuple[int, int] "A movement direction as row and column deltas."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True only when the starting cell and the next four cells in the given direction are all within bounds and contain the same symbol; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Destructure the direction into dx and dy, initialize count to 1, and read the starting symbol from self.board[row][col]. RESULT dx_dy_count_symbol]
            [COMMAND Within the for loop over i from 1 to 4 inclusive, compute the next position as new_row = row + dx * i and new_col = col + dy * i. RESULT new_row_new_col]
            [COMMAND Within the same loop, if the next position is outside the board bounds 0 <= new_row < self.board_size or 0 <= new_col < self.board_size fails, return false immediately. RESULT false]
            [COMMAND Within the same loop, if self.board[new_row][new_col] does not equal the starting symbol, return false immediately. RESULT false]
            [COMMAND Within the same loop, when the next cell is in bounds and matches the starting symbol, increment count by 1. RESULT count]
            [COMMAND After the loop finishes all four iterations, return whether count equals 5. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
