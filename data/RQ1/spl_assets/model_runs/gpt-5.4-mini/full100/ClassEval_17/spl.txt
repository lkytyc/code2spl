[DEFINE_WORKER: "Collect all events whose calendar date matches the given date and return them as a list." get_events]
    [INPUTS]
        <REF> date </REF>: date-like object "Target date used to filter events by calendar day."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of events whose event date matches the given date at day resolution."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to store events that match the target date. RESULT events_on_date]
            [COMMAND Iterate through each event in self.events. RESULT event]
            [COMMAND Within the loop over self.events, compare the calendar date of event['date'] with the calendar date of the input date. RESULT date_match]
            [COMMAND If the calendar dates are equal, append the current event to events_on_date. RESULT events_on_date]
            [COMMAND After processing every event, return the collected matching events. RESULT events_on_date]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Check whether a proposed time interval overlaps any existing event and return false immediately on the first overlap, otherwise return true." is_available]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the existing events list."
        <REF> start_time </REF>: number or comparable time value "The proposed interval start time."
        <REF> end_time </REF>: number or comparable time value "The proposed interval end time."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the proposed interval does not overlap any event; false when any overlap is detected."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each event in self.events in sequence. RESULT event]
            [COMMAND Within the loop, check whether the proposed interval overlaps the current event using strict boundaries: start_time < event['end_time'] and end_time > event['start_time']. RESULT overlap_detected]
            [COMMAND If the overlap condition is true for the current event, return false immediately and stop checking further events. RESULT false]
            [COMMAND If the loop completes without finding any overlapping event, return true. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return all available one-hour time slots within the specified date by scanning from midnight through 23:59 and collecting intervals that pass the availability check." get_available_slots]
    [INPUTS]
        <REF> date </REF>: date "Input date whose full day is scanned for hourly availability."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of tuple(datetime, datetime) "A list of available one-hour time intervals for the given date."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to store available time slots. RESULT available_slots]
            [COMMAND Set the start boundary to midnight of the input date using year, month, and day from date. RESULT start_time]
            [COMMAND Set the end boundary to 23:59 of the input date using year, month, and day from date. RESULT end_time]
            [COMMAND While start_time is strictly less than end_time, compute the one-hour slot ending at start_time plus 60 minutes. RESULT slot_end_time]
            [COMMAND Within the while loop, if the interval from start_time to slot_end_time is available, append that tuple to available_slots. RESULT available_slots]
            [COMMAND Within the while loop, advance start_time by 60 minutes to move to the next hourly interval. RESULT start_time]
            [COMMAND Return the collected list of available slots after the loop finishes. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect the first num_events events whose start_time is at or after the current time and return them in encounter order." get_upcoming_events]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the events collection."
        <REF> num_events </REF>: int "Maximum number of upcoming events to collect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of upcoming event objects selected from self.events."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize now to the current datetime and initialize upcoming_events as an empty list. RESULT now, upcoming_events]
            [COMMAND Within the outer loop over self.events, inspect each event in iteration order. RESULT event]
            [COMMAND Within the outer loop, if event['start_time'] is greater than or equal to now, append the event to upcoming_events. RESULT upcoming_events]
            [COMMAND Within the outer loop, if the number of collected upcoming_events is exactly equal to num_events, stop iterating early with break. RESULT loop_exit]
            [COMMAND Return upcoming_events after the loop completes or exits early. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
