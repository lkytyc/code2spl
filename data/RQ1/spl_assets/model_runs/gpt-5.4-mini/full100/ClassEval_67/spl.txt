# CLASS Order

[DEFINE_WORKER: "Initializes the instance state by setting the menu and selected dishes to empty lists and sales to an empty dictionary." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.menu to an empty list to represent a menu with no items initially. RESULT self.menu]
            [COMMAND Set self.selected_dishes to an empty list to represent that no dishes have been selected initially. RESULT self.selected_dishes]
            [COMMAND Set self.sales to an empty dictionary to represent that no sales records exist initially. RESULT self.sales]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks the menu for a matching dish, rejects the request if the available count is too small, otherwise subtracts the requested count from the first matching menu item and records the dish as selected." add_dish]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the menu and selected dishes state."
        <REF> dish </REF>: dict "The requested dish data, including at least a dish name under 'dish' and a requested quantity under 'count'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns False if a matching menu dish has insufficient count; otherwise returns True after updating the menu and appending the dish to selected_dishes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each menu_dish in self.menu, preserving the original order. RESULT menu_dish]
            [COMMAND Within the loop, compare dish["dish"] with menu_dish["dish"] to find the first menu item with the same dish name. RESULT name_match]
            [COMMAND When a matching dish name is found, compare menu_dish["count"] with dish["count"] to determine whether the available count is less than the requested count. RESULT enough_count_check]
            [COMMAND If menu_dish["count"] is less than dish["count"], return False immediately without changing self.menu or self.selected_dishes. RESULT False]
            [COMMAND Otherwise, subtract dish["count"] from menu_dish["count"] to consume the requested quantity from the first matching menu item. RESULT updated_count]
            [COMMAND Break out of the loop after the first successful update so no later menu items with the same name are modified. RESULT loop_exit]
            [COMMAND After the loop finishes without an early failure, append dish to self.selected_dishes. RESULT selected_dishes]
            [COMMAND Return True to indicate the dish was accepted and recorded. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the total sales amount by summing each selected dish's price multiplied by its count and its corresponding sales factor." calculate_total]
    [INPUTS]
        <REF> self </REF>: object "An object containing selected_dishes and sales data."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed total sales amount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize total to 0. RESULT total]
            [COMMAND For each dish in self.selected_dishes, add dish["price"] times dish["count"] times self.sales[dish["dish"]] to total. RESULT total]
            [COMMAND Return the final total. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns False when no dishes are selected; otherwise computes the total, clears the selected dishes, and returns the computed total." checkout]
    [INPUTS]
        <REF> self </REF>: object "The object containing selected dishes and total calculation behavior."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or numeric "Returns False when there are no selected dishes; otherwise returns the computed total."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the selected dishes list is empty. RESULT empty_check]
            [COMMAND If the selected dishes list is empty, return False immediately. RESULT False]
            [COMMAND Otherwise, compute the total by calling the internal total calculation method. RESULT total]
            [COMMAND Reset the selected dishes list to an empty list. RESULT self.selected_dishes]
            [COMMAND Return the computed total. RESULT total]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The selected dishes list has length 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without calculating a total or clearing the selected dishes list. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
