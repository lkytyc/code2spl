# CLASS URLHandler

[DEFINE_WORKER: "Initializes the object by storing the provided URL string on the instance." __init__]
    [INPUTS]
        <REF> url </REF>: any "The value to assign to the instance URL attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self </REF>: object "The initialized instance with its URL attribute set."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided url value directly to the instance attribute self.url. RESULT self.url]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the URL scheme prefix before the first '://' separator, or None when no scheme separator is present." get_scheme]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the url string to inspect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The substring before the first '://' in self.url, or null when the separator is absent."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search self.url for the first occurrence of the substring "://" and store the index in scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not -1, return the substring of self.url from the start up to but not including scheme_end. RESULT scheme]
            [COMMAND If scheme_end is -1, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The substring "://" is not present in self.url.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Keep scheme_end as -1 from the failed search result. RESULT scheme_end]
            [COMMAND Return None instead of a scheme prefix. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the host component from self.url by removing any scheme prefix and trimming any path segment after the first slash, returning None when no scheme delimiter is present." get_host]
    [INPUTS]
        <REF> self.url </REF>: string "The URL string to parse for its host component."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The host portion of the URL, or None when the scheme delimiter is absent."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first occurrence of "://" in self.url and store its index in scheme_end. RESULT scheme_end]
            [COMMAND If scheme_end is not -1, remove the scheme and delimiter by slicing self.url from scheme_end + 3 to the end and store the remainder in url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the scheme-present branch, find the first "/" in url_without_scheme and store its index in host_end. RESULT host_end]
            [COMMAND If host_end is not -1, return the substring of url_without_scheme from the start up to host_end, excluding the slash. RESULT return value]
            [COMMAND If host_end is -1, return url_without_scheme unchanged because there is no path segment after the host. RESULT return value]
            [COMMAND If scheme_end is -1, return None because the URL does not contain the expected scheme delimiter. RESULT return value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The URL contains "://" but no "/" after the scheme delimiter.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove the scheme prefix from self.url and keep the remaining string as url_without_scheme. RESULT url_without_scheme]
            [COMMAND Return the full url_without_scheme because it already consists only of the host portion. RESULT return value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The URL does not contain "://".]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None without attempting further parsing. RESULT return value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the URL path substring after the scheme separator when a scheme and path are present, otherwise return None." get_path]
    [INPUTS]
        <REF> self.url </REF>: string "The URL string to inspect and trim."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or None "The path portion of the URL starting with '/', or None when no scheme or path separator is present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute scheme_end as the index of "://" in self.url using find(). RESULT scheme_end]
            [COMMAND If scheme_end is not -1, slice self.url starting at scheme_end + 3 to remove the scheme and store the remainder in url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the branch where the scheme was found, compute host_end as the index of "/" in url_without_scheme using find(). RESULT host_end]
            [COMMAND If host_end is not -1, return the substring of url_without_scheme starting at host_end, including the leading "/". RESULT return url_without_scheme[host_end:]]
            [COMMAND If scheme_end is -1 or no "/" is found after removing the scheme, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the query portion of self.url, parse valid key=value pairs separated by ampersands, and return them as a dictionary or None when no query exists." get_query_params]
    [INPUTS]
        <REF> self </REF>: object "Instance providing a url string to parse."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict or None "A dictionary of parsed query parameters when a query is present, an empty dictionary when the query exists but contains no valid pairs, or None when no query delimiter is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first occurrence of '?' in self.url and store its index in query_start. RESULT query_start]
            [COMMAND Find the first occurrence of '#' in self.url and store its index in fragment_start. RESULT fragment_start]
            [COMMAND If query_start is not -1, extract the substring from just after '?' up to fragment_start, create an empty params dictionary, and continue parsing that query string. RESULT query_string_and_params]
            [COMMAND Within the branch where query_start is not -1, if query_string length is greater than 0, split it on '&' into param_pairs. RESULT param_pairs]
            [COMMAND Within the branch where query_string length is greater than 0, for each pair in param_pairs, split the pair on '=' into key_value. RESULT key_value]
            [COMMAND Within the loop over param_pairs, if key_value contains exactly 2 parts, unpack them into key and value and assign params[key] = value. RESULT params]
            [COMMAND Within the branch where query_start is not -1, return params after processing all pairs, even if no valid pairs were added. RESULT params]
            [COMMAND If query_start is -1, return None without parsing any query parameters. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the URL fragment substring after the first '#' character, or None if the URL contains no fragment." get_fragment]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a url string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or null "The substring after the first '#' character, or None when no '#' is present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the first occurrence of '#' in self.url and store its zero-based index in fragment_start, using -1 when the character is not found. RESULT fragment_start]
            [COMMAND If fragment_start is not -1, return the substring of self.url starting immediately after fragment_start and continuing to the end. RESULT result]
            [COMMAND If fragment_start is -1, return None. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
