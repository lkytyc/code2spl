# CLASS JSONProcessor

[DEFINE_WORKER: "Read and parse a JSON file from the given path, returning 0 if the path does not exist and -1 if any error occurs during file access or parsing." read_json]
    [INPUTS]
        <REF> file_path </REF>: str "Path to the JSON file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object or int "Parsed JSON data on success, 0 when the file path does not exist, or -1 if an exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the file path exists; if it does not exist, return 0 immediately. RESULT return_0]
            [COMMAND Open the existing file at file_path in read mode, load its JSON content, and store the parsed data. RESULT data]
            [COMMAND Return the parsed JSON data. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The file path does not exist.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 without attempting to open or parse the file. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Any exception is raised while opening the file or loading JSON content.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 as the fallback error value. RESULT return_minus_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Serialize the provided data object to the given file path as JSON and return 1 on success or -1 if any error occurs." write_json]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> data </REF>: any "The data structure to be written as JSON."
        <REF> file_path </REF>: string "Path to the target file to open for writing."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 1 when the JSON write succeeds, otherwise returns -1 if an exception is caught."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Try to open the target file path in write mode and bind the handle to file. RESULT file]
            [COMMAND Serialize the provided data into the open file using json.dump. RESULT written_json]
            [COMMAND Return 1 to indicate the write completed successfully. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs anywhere in the try block.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception and stop normal processing. RESULT exception_handled]
            [COMMAND Return -1 to indicate failure. RESULT -1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Load JSON from a file, return 0 for invalid sentinel values, delete a requested key when present, persist the updated JSON, and return a status code indicating whether removal occurred." process_json]
    [INPUTS]
        <REF> file_path </REF>: path "Path to the JSON file to read and possibly update."
        <REF> remove_key </REF>: any "Key to remove from the loaded JSON object if present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 1 when the key is removed and the file is rewritten; otherwise returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read JSON data from file_path using self.read_json and store the result in data. RESULT data]
            [COMMAND If data is 0 or -1, return 0 immediately without making any changes. RESULT 0]
            [COMMAND If remove_key exists in data, delete data[remove_key], write the updated data back to file_path using self.write_json, and return 1. RESULT 1]
            [COMMAND If remove_key does not exist in data, return 0 without writing the file. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The loaded JSON value is the sentinel 0 or -1, indicating an invalid or unavailable read result.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Stop processing immediately and return 0. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The requested key is not present in the loaded data.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Do not modify data or write the file, and return 0. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
