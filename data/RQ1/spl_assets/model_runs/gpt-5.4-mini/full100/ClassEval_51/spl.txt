# CLASS KappaCalculator

[DEFINE_WORKER: "Computes Cohen's kappa coefficient from a confusion matrix by accumulating diagonal agreement, deriving expected agreement from row and column totals, and returning the normalized agreement score." kappa]
    [INPUTS]
        <REF> testData </REF>: matrix-like data "Input data used to build the contingency matrix."
        <REF> k </REF>: integer "Number of leading diagonal entries to include when summing observed agreement."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The computed Cohen's kappa coefficient."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert testData into a NumPy matrix and initialize P0 to 0.0. RESULT dataMat, P0]
            [COMMAND For each i from 0 up to but not including k, add dataMat[i, i] multiplied by 1.0 to P0. RESULT P0]
            [COMMAND Compute the row-wise sum of dataMat with axis 1. RESULT xsum]
            [COMMAND Compute the column-wise sum of dataMat with axis 0. RESULT ysum]
            [COMMAND Compute the total sum of all elements in dataMat. RESULT sum]
            [COMMAND Compute Pe as float(ysum * xsum) divided by sum and divided by sum again. RESULT Pe]
            [COMMAND Overwrite P0 with float(P0 divided by sum multiplied by 1.0). RESULT P0]
            [COMMAND Compute cohens_coefficient as float((P0 - Pe) divided by (1 - Pe)). RESULT cohens_coefficient]
            [COMMAND Return cohens_coefficient. RESULT cohens_coefficient]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: If sum is 0, the divisions used to compute Pe and P0 are invalid.]
        [LOG "Division by zero would occur while computing the expected agreement or normalized observed agreement."]
        [THROW ZeroDivisionError "Raised when the total matrix sum is zero and division cannot be performed."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: If Pe equals 1, the final kappa normalization denominator becomes zero.]
        [LOG "Division by zero would occur while computing cohens_coefficient because 1 - Pe equals 0."]
        [THROW ZeroDivisionError "Raised when the expected agreement is exactly one."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes Fleiss' kappa from a matrix of categorical ratings by accumulating per-item agreement, normalizing by item count, estimating expected agreement, and returning the final kappa value." fleiss_kappa]
    [INPUTS]
        <REF> testData </REF>: matrix-like data "Input rating table used to build the numeric matrix."
        <REF> N </REF>: int "Number of items (outer-loop iterations and normalization divisor)."
        <REF> k </REF>: int "Number of categories/columns and the size of the one-vector."
        <REF> n </REF>: numeric "Number of ratings per item used in the per-row agreement normalization."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric matrix scalar "The Fleiss' kappa value returned as the single element at position [0, 0] of ans."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert testData into a float NumPy matrix and create a k-by-1 column vector of ones. RESULT dataMat, oneMat]
            [COMMAND Initialize the running total sum and the accumulated agreement value P0 to 0.0. RESULT sum, P0]
            [COMMAND For each i from 0 to N - 1, initialize temp to 0.0 and then, for each j from 0 to k - 1, add dataMat[i, j] to sum and add 1.0 * dataMat[i, j] ** 2 to temp. RESULT sum, temp]
            [COMMAND Within the outer for loop over i, after the inner loop completes, subtract n from temp, divide temp by (n - 1) * n, and add the result to P0. RESULT P0]
            [COMMAND After processing all rows, divide P0 by N using floating-point arithmetic. RESULT P0]
            [COMMAND Compute ysum as the column-wise sum of dataMat along axis 0. RESULT ysum]
            [COMMAND For each i from 0 to k - 1, replace ysum[0, i] with (ysum[0, i] / sum) ** 2. RESULT ysum]
            [COMMAND Compute Pe as ysum multiplied by oneMat and then multiplied by 1.0. RESULT Pe]
            [COMMAND Compute ans as (P0 - Pe) / (1 - Pe). RESULT ans]
            [COMMAND Return ans[0, 0]. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
