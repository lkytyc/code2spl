# CLASS RPGCharacter

[DEFINE_WORKER: "Initialize a new object with the provided name, health, attack power, defense, and level, and set experience to zero." __init__]
    [INPUTS]
        <REF> name </REF>: any "The entity name to store on the instance."
        <REF> hp </REF>: any "The initial health value to store on the instance."
        <REF> attack_power </REF>: any "The initial attack power value to store on the instance."
        <REF> defense </REF>: any "The initial defense value to store on the instance."
        <REF> level </REF>: any "The initial level value to store on the instance; defaults to 1 if not provided."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided name to the instance name attribute. RESULT self.name]
            [COMMAND Assign the provided hp value to the instance hp attribute. RESULT self.hp]
            [COMMAND Assign the provided attack_power value to the instance attack_power attribute. RESULT self.attack_power]
            [COMMAND Assign the provided defense value to the instance defense attribute. RESULT self.defense]
            [COMMAND Assign the provided level value to the instance level attribute, using the default value 1 when no level argument is supplied. RESULT self.level]
            [COMMAND Initialize the instance exp attribute to the constant value 0. RESULT self.exp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate the attack damage against another character using attack power minus defense, with a minimum damage of 1, then subtract that damage from the target's HP." attack]
    [INPUTS]
        <REF> self </REF>: object "The attacker object providing attack_power."
        <REF> other_character </REF>: object "The target character object providing defense and hp."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute damage as the larger of self.attack_power minus other_character.defense and 1, ensuring damage is never below 1. RESULT damage]
            [COMMAND Subtract the computed damage from other_character.hp. RESULT other_character.hp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Heals the object by increasing hp by 10, capping it at 100, and returning the updated hp value." heal]
    [INPUTS]
        <REF> self </REF>: object "The instance whose hp field is updated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The updated hp value after healing and optional capping."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.hp by 10. RESULT self.hp]
            [COMMAND If self.hp is greater than 100, set self.hp to 100. RESULT self.hp]
            [COMMAND Return the current value of self.hp. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Repeatedly applies experience points until the input amount is fully consumed, leveling up whenever the current experience reaches or exceeds the next level threshold." gain_exp]
    [INPUTS]
        <REF> self </REF>: object "The owning object whose experience and level are updated."
        <REF> amount </REF>: integer "The amount of experience to apply, which may be reduced across multiple loop iterations until it becomes zero."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize processing of the input amount for experience gain. RESULT start]
            [COMMAND Within the outer while loop that continues as long as amount is not equal to 0, check whether self.exp plus amount is greater than or equal to self.level multiplied by 100. RESULT threshold_check]
            [COMMAND If self.exp plus amount is greater than or equal to self.level multiplied by 100, subtract the exact experience needed to reach self.level multiplied by 100 from amount, then call level_up on self, and continue the while loop with any remaining amount. RESULT leveled_up]
            [COMMAND Otherwise, add all remaining amount to self.exp and set amount to 0 to terminate the while loop. RESULT xp_applied_and_finished]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Increases the character's level by one when below 100, resets experience, boosts stats, and returns the updated level and attributes." level_up]
    [INPUTS]
        <REF> self </REF>: object "The instance whose level, experience, hit points, attack power, and defense are updated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the updated level, hit points, attack power, and defense."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.level is strictly less than 100. RESULT level_check]
            [COMMAND If self.level is strictly less than 100, increase self.level by 1, reset self.exp to 0, increase self.hp by 20, increase self.attack_power by 5, and increase self.defense by 5. RESULT updated_stats]
            [COMMAND Return the current self.level, self.hp, self.attack_power, and self.defense as a tuple. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns whether the object's hp value is greater than zero." is_alive]
    [INPUTS]
        <REF> self </REF>: object "The current instance containing the hp attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when self.hp is strictly greater than 0; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current instance's hp attribute. RESULT hp_value]
            [COMMAND Compare hp_value against the constant 0 using a strict greater-than check. RESULT is_alive_result]
            [COMMAND Return the boolean result of the comparison. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
