# CLASS PageUtil

[DEFINE_WORKER: "Initializes the object with the provided data and page size, then derives the total item count and the required number of pages using ceiling division." __init__]
    [INPUTS]
        <REF> data </REF>: sequence "The collection of items to store and paginate."
        <REF> page_size </REF>: integer "The number of items allowed per page."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input data to the instance field self.data. RESULT self.data]
            [COMMAND Assign the input page_size to the instance field self.page_size. RESULT self.page_size]
            [COMMAND Compute self.total_items as the length of the input data. RESULT self.total_items]
            [COMMAND Compute self.total_pages using ceiling division: add page_size minus 1 to self.total_items, then perform integer floor division by page_size. RESULT self.total_pages]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a single page of data for a 1-based page number, or an empty list when the requested page is out of range." get_page]
    [INPUTS]
        <REF> self </REF>: object "Instance providing total_pages, page_size, and data."
        <REF> page_number </REF>: integer "Requested 1-based page number."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A slice of self.data for the requested page, or an empty list if the page number is invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether page_number is less than 1 or greater than self.total_pages. RESULT page_number_validity]
            [COMMAND If page_number is out of range, return an empty list immediately. RESULT []]
            [COMMAND Otherwise, compute start_index as (page_number - 1) * self.page_size to convert the 1-based page number into a zero-based offset. RESULT start_index]
            [COMMAND Compute end_index as start_index + self.page_size so the page slice uses an exclusive upper bound. RESULT end_index]
            [COMMAND Return the slice of self.data from start_index up to but not including end_index. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: page_number is less than 1 or greater than self.total_pages]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list without computing any page indices or slicing data. RESULT []]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return pagination metadata and the requested page slice when the page number is within range, otherwise return an empty dictionary." get_page_info]
    [INPUTS]
        <REF> page_number </REF>: int "Requested 1-based page number."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "A dictionary containing page metadata and page data, or an empty dictionary when the page number is out of range."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether page_number is outside the valid range from 1 through self.total_pages inclusive. RESULT range_check]
            [COMMAND Compute start_index as (page_number - 1) * self.page_size using zero-based pagination. RESULT start_index]
            [COMMAND Compute end_index as the smaller of start_index + self.page_size and self.total_items to prevent slicing past the dataset end. RESULT end_index]
            [COMMAND Slice self.data from start_index to end_index to produce the page contents. RESULT page_data]
            [COMMAND Build page_info with current_page, per_page, total_pages, total_items, has_previous, has_next, and data taken from the computed values. RESULT page_info]
            [COMMAND Return page_info. RESULT page_info]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If page_number is less than 1 or greater than self.total_pages.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty dictionary immediately. RESULT {}]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filter the stored data by keyword, count the matches, compute the number of pages using the page size, and return the search details in a dictionary." search]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the data collection and page size."
        <REF> keyword </REF>: string "Substring to search for within each item."
    [END_INPUTS]

    [OUTPUTS]
        <REF> search_info </REF>: dict "Dictionary containing the keyword, total result count, total page count, and matching results."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Filter self.data to collect every item whose string representation contains the keyword, and store the matching items in results. RESULT results]
            [COMMAND Count the number of matching items in results and store the count in num_results. RESULT num_results]
            [COMMAND Compute the total number of pages using ceiling-style integer division with num_results and self.page_size: (num_results + self.page_size - 1) // self.page_size. RESULT num_pages]
            [COMMAND Assemble search_info with the original keyword, the total result count, the total page count, and the filtered results list. RESULT search_info]
            [COMMAND Return search_info. RESULT search_info]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
