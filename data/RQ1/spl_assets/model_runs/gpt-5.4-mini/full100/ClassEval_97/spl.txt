[DEFINE_WORKER: "Convert a hyphen-normalized text string into a mixed output string by parsing recognized number and ordinal words into numeric values while passing through non-number tokens unchanged." text2int]
    [INPUTS]
        <REF> textnum </REF>: string "Input text containing number words, ordinal forms, and ordinary tokens."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The transformed string with recognized numeric phrases converted into digits and other text preserved."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace every hyphen in the input text string with a space, then initialize the numeric accumulators and output buffer to zero, empty string, and false. RESULT textnum, current, result, curstring, onnumber]
            [COMMAND For each word from splitting the normalized input on whitespace, process the token in order and keep parsing state across iterations. RESULT word]
            [COMMAND Within the outer token loop, if the current word is found in the ordinal word mapping, treat it as a direct ordinal value by setting scale to 1, adding the mapped increment into the running current value, and marking that a number is in progress. RESULT current, onnumber]
            [COMMAND Within the outer token loop, if the current word is not an ordinal word, iterate through each ordinal ending replacement pair and, for any matching suffix, rewrite the word by removing the ending and appending the replacement. RESULT word]
            [COMMAND After ordinal suffix normalization, if the current word is not in the number-word mapping, then if a number phrase was active append the accumulated numeric value plus the current value and a trailing space to the output, append the non-number word plus a trailing space, and reset result, current, and the number state. RESULT curstring, result, current, onnumber]
            [COMMAND After ordinal suffix normalization, if the current word is in the number-word mapping, add the mapped scale and increment into the running current value; when the scale is greater than 100, fold the current value into result and reset current to zero, then mark that a number is in progress. RESULT current, result, onnumber]
            [COMMAND After all words have been processed, if a number phrase is still active, append the final accumulated numeric value to the output buffer without a trailing space. RESULT curstring]
            [COMMAND Return the constructed output string. RESULT curstring]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalize hyphenated text into space-separated words, then verify that each token is either an accepted ordinal word or a numeric word after optional ordinal-suffix replacement." is_valid_input]
    [INPUTS]
        <REF> self </REF>: object "Instance providing ordinal_words, ordinal_endings, and numwords lookup collections."
        <REF> textnum </REF>: string "Text containing number-related words to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if every token is recognized as valid, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace every hyphen in the input text with a space and assign the normalized string back to textnum. RESULT textnum]
            [COMMAND Split textnum into words and iterate over each word in order. RESULT word]
            [COMMAND Within the outer loop, if the current word is already present in self.ordinal_words, skip any further processing for that word and continue with the next token. RESULT continue]
            [COMMAND Within the outer loop, for each configured ordinal ending and replacement in self.ordinal_endings, if the current word ends with that ending, rewrite the word by removing the ending and appending the replacement. RESULT word]
            [COMMAND Still within the outer loop, after any suffix normalization, if the current word is not present in self.numwords, return False immediately. RESULT False]
            [COMMAND If every token passes the ordinal-word check or numeric-word membership check, return True after the loop completes. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
