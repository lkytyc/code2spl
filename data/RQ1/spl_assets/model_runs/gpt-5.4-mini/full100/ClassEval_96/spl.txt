# CLASS WeatherSystem

[DEFINE_WORKER: "Initializes the object by clearing temperature and weather, storing the provided city, and creating an empty weather list." __init__]
    [INPUTS]
        <REF> city </REF>: any "The city value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "No return value; the instance is initialized in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.temperature to None. RESULT self.temperature]
            [COMMAND Set self.weather to None. RESULT self.weather]
            [COMMAND Assign the provided city value to self.city. RESULT self.city]
            [COMMAND Initialize self.weather_list as an empty dictionary. RESULT self.weather_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Fetch the current city's weather data from the provided list, returning False if the city is missing and converting the temperature only when the stored unit differs from the requested unit." query]
    [INPUTS]
        <REF> weather_list </REF>: dict "Mapping of city names to weather records containing temperature, weather, and temperature units."
        <REF> tmp_units </REF>: str "Requested temperature unit, defaulting to celsius."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple|bool "Returns False when the city is not present; otherwise returns a temperature value or converted temperature paired with the weather description."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided weather_list to the instance attribute self.weather_list. RESULT self.weather_list]
            [COMMAND Check whether self.city is present in weather_list; if it is not present, return False immediately. RESULT False]
            [COMMAND When self.city is present in weather_list, load self.temperature from self.weather_list[self.city]['temperature'] and load self.weather from self.weather_list[self.city]['weather']. RESULT self.temperature, self.weather]
            [COMMAND Compare self.weather_list[self.city]['temperature units'] with tmp_units; if they are equal, return self.temperature and self.weather without conversion. RESULT self.temperature, self.weather]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When self.weather_list[self.city]['temperature units'] differs from tmp_units and tmp_units is celsius.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the stored temperature using self.fahrenheit_to_celsius() and return the converted value together with self.weather. RESULT converted_temperature, self.weather]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When self.weather_list[self.city]['temperature units'] differs from tmp_units and tmp_units is fahrenheit.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert the stored temperature using self.celsius_to_fahrenheit() and return the converted value together with self.weather. RESULT converted_temperature, self.weather]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sets the instance's city attribute to the provided city value." set_city]
    [INPUTS]
        <REF> self </REF>: object "The instance whose city attribute will be updated."
        <REF> city </REF>: any "The value to assign to the instance city attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input city value directly to the instance attribute self.city. RESULT self.city]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the instance temperature from Celsius to Fahrenheit using the standard linear formula and returns the computed value." celsius_to_fahrenheit]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the temperature value to convert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The temperature converted from Celsius to Fahrenheit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.temperature as the Celsius input value. RESULT celsius_value]
            [COMMAND Multiply the Celsius value by 9 and divide the product by 5. RESULT scaled_value]
            [COMMAND Add 32 to the scaled value to convert it to Fahrenheit. RESULT fahrenheit_value]
            [COMMAND Return the Fahrenheit value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the instance temperature from Fahrenheit to Celsius using the standard linear conversion formula." fahrenheit_to_celsius]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the temperature value in Fahrenheit."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The temperature converted from Fahrenheit to Celsius."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current Fahrenheit temperature from self.temperature. RESULT temperature_fahrenheit]
            [COMMAND Subtract 32 from the Fahrenheit temperature. RESULT adjusted_temperature]
            [COMMAND Multiply the adjusted temperature by 5 and divide the product by 9 to compute the Celsius value. RESULT temperature_celsius]
            [COMMAND Return the computed Celsius temperature. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
