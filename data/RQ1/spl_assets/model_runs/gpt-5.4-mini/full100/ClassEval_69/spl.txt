# CLASS PDFHandler

[DEFINE_WORKER: "Initializes the instance by storing the provided file paths and creating a PyPDF2 PdfReader for each path." __init__]
    [INPUTS]
        <REF> filepaths </REF>: iterable of path-like values "A collection of PDF file paths used to build the reader list."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the provided filepaths argument on self.filepaths. RESULT self.filepaths]
            [COMMAND For each path in filepaths, create a PyPDF2.PdfReader instance and collect the readers into a list. RESULT self.readers]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Merge all pages from every PDF reader in self.readers into a single PDF file saved to the given output path." merge_pdfs]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the readers collection to merge."
        <REF> output_filepath </REF>: str "Destination file path where the merged PDF will be written."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Confirmation message stating where the merged PDFs were saved."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new PyPDF2.PdfWriter instance and store it in pdf_writer. RESULT pdf_writer]
            [COMMAND For each reader in self.readers, iterate through every page index from 0 to len(reader.pages) - 1 using range(len(reader.pages)). RESULT reader,page_num]
            [COMMAND Within the inner loop for the current reader, get the page at reader.pages[page_num] and store it in page. RESULT page]
            [COMMAND Within the nested loops, add the current page to pdf_writer with add_page. RESULT pdf_writer]
            [COMMAND After all readers and pages have been processed, open output_filepath in binary write mode as out. RESULT out]
            [COMMAND Write the accumulated pdf_writer content to out and close the file when the context ends. RESULT written_file]
            [COMMAND Return the message Merged PDFs saved at the output_filepath. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterate through every PDF reader and every page in each reader, extract text from each page, and return the collected texts as a list." extract_text_from_pdfs]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a readers collection, where each reader exposes a pages sequence."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list containing the extracted text from every page of every reader, in traversal order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize pdf_texts as an empty list to collect extracted page text. RESULT pdf_texts]
            [COMMAND For each reader in self.readers, iterate through all pages using page indices from 0 to len(reader.pages) - 1. RESULT reader, page_num]
            [COMMAND Within the outer loop over each reader, fetch the page at the current page_num index from reader.pages. RESULT page]
            [COMMAND Within the nested loops, extract text from the current page and append the extracted text to pdf_texts. RESULT pdf_texts]
            [COMMAND After processing every reader and every page, return pdf_texts. RESULT pdf_texts]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
