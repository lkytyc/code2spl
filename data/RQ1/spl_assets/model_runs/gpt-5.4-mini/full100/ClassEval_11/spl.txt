# CLASS BitStatusUtil

[DEFINE_WORKER: "Checks the provided bitmask inputs and returns their bitwise OR combination." add]
    [INPUTS]
        <REF> states </REF>: number "Current bitmask value to be combined."
        <REF> stat </REF>: number "Bitmask value to merge into states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "Bitwise OR of states and stat."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate the provided values by calling BitStatusUtil.check with states and stat. RESULT validation_status]
            [COMMAND Compute the combined bitmask as states OR stat. RESULT combined_state]
            [COMMAND Return the combined bitmask. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the given bit mask is fully contained in the state bitset by testing that all bits in stat are present in states." has]
    [INPUTS]
        <REF> states </REF>: integer "Bitset value being inspected."
        <REF> stat </REF>: integer "Bit mask to test against states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when every bit set in stat is also set in states; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate the inputs with BitStatusUtil.check before performing the bit test. RESULT validated_inputs]
            [COMMAND Compute the bitwise AND of states and stat, then compare the result to stat to determine whether all required bits are present. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the given status bit from states after validation, toggling it off if present and otherwise leaving states unchanged." remove]
    [INPUTS]
        <REF> states </REF>: bitmask "Current bit status value to update."
        <REF> stat </REF>: bitmask "Bit flag to remove from states."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bitmask "The updated states value with stat cleared if it was set, or the original states value if it was not set."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate states and stat using BitStatusUtil.check([states, stat]). RESULT _validated]
            [COMMAND If BitStatusUtil.has(states, stat) is true, return states XOR stat to toggle the stat bit off. RESULT _toggled_states]
            [COMMAND Otherwise, return states unchanged. RESULT _result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: BitStatusUtil.has(states, stat) is false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the original states value without modification. RESULT _result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that every value in args is non-negative and even, raising a ValueError immediately when any value fails either check." check]
    [INPUTS]
        <REF> args </REF>: iterable of integers "Values to validate for being greater than or equal to 0 and even."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each value in args. RESULT current_arg]
            [COMMAND Within the loop for the current value, check whether the value is less than 0. RESULT negative_check]
            [COMMAND If the current value is less than 0, raise a ValueError stating that the value must be greater than or equal to 0. RESULT exception_path]
            [COMMAND Within the same loop iteration, after the non-negative check passes, check whether the current value modulo 2 is not equal to 0. RESULT evenness_check]
            [COMMAND If the current value is odd, raise a ValueError stating that the value is not even. RESULT exception_path]
            [COMMAND Continue to the next value only when the current value is non-negative and even. RESULT next_iteration]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: A value in args is less than 0.]
        [LOG "Raise ValueError with the message '<value> must be greater than or equal to 0'."]
        [THROW ValueError "Indicates that a value is negative and violates the non-negative requirement."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: A value in args is not evenly divisible by 2.]
        [LOG "Raise ValueError with the message '<value> not even'."]
        [THROW ValueError "Indicates that a value is odd and violates the even-number requirement."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
