# CLASS BookManagementDB

[DEFINE_WORKER: "Initialize the database connection and cursor for the given SQLite database name, then create the table." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "SQLite database name or path used to open the connection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection from db_name and store the resulting connection object in self.connection. RESULT self.connection]
            [COMMAND Use self.connection to create a cursor object and store it in self.cursor. RESULT self.cursor]
            [COMMAND Call self.create_table() to initialize the required table schema. RESULT table_initialized]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates the books table with an id primary key and title, author, and available columns if it does not already exist, then commits the schema change." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a SQL CREATE TABLE IF NOT EXISTS statement to create the books table with columns id INTEGER PRIMARY KEY, title TEXT, author TEXT, and available INTEGER. RESULT table_creation_result]
            [COMMAND Commit the database connection to persist the table creation. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a new book record with the given title and author, marks it available, and commits the database transaction." add_book]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database cursor and connection."
        <REF> title </REF>: string "Title of the book to insert."
        <REF> author </REF>: string "Author of the book to insert."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the provided title and author as bound parameters for a parameterized SQL insert into the books table, setting available to the constant value 1. RESULT insert_statement]
            [COMMAND Execute the insert statement through the database cursor. RESULT execution_result]
            [COMMAND Commit the database transaction to make the inserted book record persistent. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes a book record by its id using a parameterized SQL DELETE statement and commits the transaction." remove_book]
    [INPUTS]
        <REF> self </REF>: object "Instance providing a database cursor and connection."
        <REF> book_id </REF>: int or compatible identifier "Identifier of the book record to delete."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "The method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL DELETE statement on the books table using book_id as the bound id value. RESULT delete_statement_executed]
            [COMMAND Commit the database transaction on the current connection to persist the deletion. RESULT transaction_committed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks the book identified by book_id as unavailable in the database and commits the transaction to persist the update." borrow_book]
    [INPUTS]
        <REF> self </REF>: object "Instance providing database cursor and connection objects."
        <REF> book_id </REF>: integer or string "Identifier of the book row to update."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute an SQL UPDATE statement through self.cursor to set available to 0 for the book row whose id matches the provided book_id parameter. RESULT update_result]
            [COMMAND Commit the database transaction through self.connection so the availability change becomes persistent. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as available again in the database by updating its record by id and committing the change." return_book]
    [INPUTS]
        <REF> self </REF>: object "Instance providing database cursor and connection access."
        <REF> book_id </REF>: integer or string "Identifier of the book record to update."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the database cursor to execute a parameterized UPDATE statement that sets available to 1 for the book row whose id equals the provided book_id. RESULT update_result]
            [COMMAND Commit the database transaction so the availability change is persisted. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Execute a query to retrieve all rows from the books table and return the fetched results." search_books]
    [INPUTS]
        <REF> self </REF>: object "Instance providing access to the database cursor."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "All rows returned by the books query, as fetched from the cursor."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query that selects all columns from the books table using the database cursor. RESULT query_result]
            [COMMAND Fetch all remaining rows from the cursor after the query execution and assign them to books. RESULT books]
            [COMMAND Return the books collection as the method result. RESULT books]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
