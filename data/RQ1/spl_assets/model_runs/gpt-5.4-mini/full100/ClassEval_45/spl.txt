# CLASS ImageProcessor

[DEFINE_WORKER: "Initializes the instance image attribute to None as a default empty state." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constant value None to self.image to initialize the image attribute with a null-like default state. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Loads an image from the given path and stores the resulting image object on the current instance." load_image]
    [INPUTS]
        <REF> self </REF>: object "The current instance whose image attribute will be updated."
        <REF> image_path </REF>: path-like "The file path to the image that should be opened."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the image at image_path using Image.open and obtain the loaded image object. RESULT loaded_image]
            [COMMAND Assign the loaded image object to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: If image_path is invalid, inaccessible, or does not point to a readable image file, Image.open may fail during loading.]
        [LOG "Image loading cannot complete because Image.open cannot open the provided path."]
        [THROW Exception "An exception raised by Image.open prevents self.image from being assigned."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Saves the current image object to the provided path only when an image is present." save_image]
    [INPUTS]
        <REF> save_path </REF>: path-like "Destination path where the image should be saved."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is truthy. RESULT image_present]
            [COMMAND If self.image is truthy, call self.image.save(save_path) to write the image to the given path. RESULT saved_image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Resizes the current image to the given width and height when an image is present, then stores the resized image back on the object." resize_image]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the image attribute to be resized."
        <REF> width </REF>: int "The target width for the resized image."
        <REF> height </REF>: int "The target height for the resized image."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is truthy. RESULT _condition_result]
            [COMMAND If self.image is truthy, resize it to (width, height) and assign the resized image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.image is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the resize operation and leave self.image unchanged. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Rotate the current image by the given degrees and store the rotated image back only when an image is present." rotate_image]
    [INPUTS]
        <REF> degrees </REF>: number "The rotation amount to apply to the image."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: void "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is present and truthy. RESULT image_exists]
            [COMMAND If self.image exists, rotate it by the given degrees and assign the rotated image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "If an image is present, create a brightness enhancer for it and replace the image with the enhanced result using the given factor." adjust_brightness]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the image to adjust."
        <REF> factor </REF>: number "Brightness multiplier passed to the image enhancer."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.image is truthy before applying any brightness change. RESULT image_present]
            [COMMAND If self.image is truthy, create an ImageEnhance.Brightness enhancer for self.image. RESULT enhancer]
            [COMMAND If self.image is truthy, apply enhancer.enhance(factor) to produce the adjusted image. RESULT adjusted_image]
            [COMMAND If self.image is truthy, assign the adjusted image back to self.image. RESULT self.image]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
