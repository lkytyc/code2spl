# CLASS CurrencyConverter

    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]

            [COMMAND Assign self.rates to a dictionary literal containing fixed currency exchange rates. RESULT self.rates]
            [COMMAND Set USD to 1.0 as the baseline reference rate. RESULT self.rates['USD']]

[DEFINE_WORKER: "Converts an amount from one currency to another using stored exchange rates, returning the original amount when the currencies match, false when either rate is missing, and otherwise the rate-adjusted converted amount." convert]
    [INPUTS]
        <REF> amount </REF>: number "The numeric amount to convert."
        <REF> from_currency </REF>: string "The source currency code."
        <REF> to_currency </REF>: string "The target currency code."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or boolean "The converted amount, the original amount if both currencies are the same, or false if either currency rate is unavailable."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether from_currency is equal to to_currency. RESULT currency_match]
            [COMMAND If from_currency equals to_currency, return amount immediately without any conversion. RESULT result]
            [COMMAND Check whether from_currency is missing from self.rates or to_currency is missing from self.rates. RESULT rates_available]
            [COMMAND If either currency is not present in self.rates, return false. RESULT result]
            [COMMAND Look up from_rate from self.rates using from_currency and look up to_rate from self.rates using to_currency. RESULT from_rate_and_to_rate]
            [COMMAND Compute converted_amount as (amount divided by from_rate) multiplied by to_rate. RESULT converted_amount]
            [COMMAND Return converted_amount. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: from_rate is zero during the division in the conversion formula.]
        [LOG "Division by zero occurs while computing the normalized amount from amount / from_rate."]
        [THROW ZeroDivisionError "Raised when the source exchange rate is zero and the conversion formula attempts division by it."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

    [END_INPUTS]

[DEFINE_WORKER: "Add a new currency rate to the mapping only when the currency is not already present, returning False if it exists." add_currency_rate]
    [INPUTS]
        <REF> self </REF>: object "Object containing the rates mapping."
        <REF> currency </REF>: any "Currency key to add if it is not already present."
        <REF> rate </REF>: any "Rate value to store for the currency."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False when the currency already exists; otherwise the method completes after updating self.rates and returns no explicit value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given currency is already present in self.rates. RESULT currency_exists]
            [COMMAND If the currency is already present in self.rates, return False immediately without modifying the mapping. RESULT False]
            [COMMAND Otherwise, store the provided rate in self.rates under the given currency key. RESULT self.rates[currency]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The currency key already exists in self.rates.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without changing self.rates. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Update the stored exchange rate for a currency when the currency already exists, and return False without changes if it does not." update_currency_rate]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the rates mapping."
        <REF> currency </REF>: any "Key identifying the currency rate to update."
        <REF> new_rate </REF>: any "New rate value to store for the given currency."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns False if the currency is not already present; otherwise returns None after updating the mapping."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency is present in self.rates. RESULT currency_exists]
            [COMMAND If currency is not present in self.rates, return False immediately without modifying any stored rate. RESULT false]
            [COMMAND Otherwise, assign new_rate to self.rates[currency] to update the existing currency rate. RESULT updated_rate]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
