# CLASS Classroom

[DEFINE_WORKER: "Initializes the object with the provided id and starts with an empty courses list." __init__]
    [INPUTS]
        <REF> id </REF>: any "Identifier value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming id parameter to the instance field self.id. RESULT self.id]
            [COMMAND Initialize the instance field self.courses to a new empty list. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a course to the course list only when it is not already present." add_course]
    [INPUTS]
        <REF> course </REF>: unknown "The course item to add if it is not already in the collection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether course is already present in self.courses. RESULT membership_check]
            [COMMAND If course is not present in self.courses, append course to self.courses. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes the given course from the stored course list only when it is already present." remove_course]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the courses collection."
        <REF> course </REF>: any "The course item to remove from the collection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given course is present in self.courses. RESULT course_present]
            [COMMAND If the course is present, remove it from self.courses. RESULT self.courses]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a given HH:MM time falls outside all stored course time ranges and returns False immediately if any course overlaps it." is_free_at]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a courses collection with start and end time fields."
        <REF> check_time </REF>: string "A time string in exact 24-hour HH:MM format."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns False if the checked time is within any course interval, otherwise returns True."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the input check_time string using the exact HH:MM format and overwrite check_time with the resulting datetime value. RESULT check_time]
            [COMMAND Within the outer loop over each course in self.courses, parse course['start_time'] and course['end_time'] using the exact HH:MM format and compare them with check_time using inclusive lower and upper bounds. RESULT comparison_result]
            [COMMAND If check_time is greater than or equal to the course start time and less than or equal to the course end time for any course, return False immediately without checking remaining courses. RESULT False]
            [COMMAND If no course interval contains check_time after the loop finishes, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Check whether a proposed course time conflicts with any existing course by comparing the new start and end times against every stored course interval using inclusive boundary checks." check_course_conflict]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the existing courses collection."
        <REF> new_course </REF>: dict "Course data containing 'start_time' and 'end_time' as '%H:%M' formatted strings."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when no conflict is found; False when any existing course overlaps the new course at either endpoint."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse new_course['start_time'] with the '%H:%M' format to produce new_start_time. RESULT new_start_time]
            [COMMAND Parse new_course['end_time'] with the '%H:%M' format to produce new_end_time. RESULT new_end_time]
            [COMMAND Initialize flag to True as the default assumption that no conflict exists yet. RESULT flag]
            [COMMAND For each course in self.courses, parse course['start_time'] and course['end_time'] with the '%H:%M' format to produce start_time and end_time. RESULT start_time, end_time]
            [COMMAND Within the outer for loop over self.courses, if start_time <= new_start_time and end_time >= new_start_time, set flag to False. RESULT flag]
            [COMMAND Within the outer for loop over self.courses, if start_time <= new_end_time and end_time >= new_end_time, set flag to False. RESULT flag]
            [COMMAND After checking all courses, return flag. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Either time string does not match the '%H:%M' format during datetime parsing.]
        [LOG "datetime.strptime fails while parsing course time values."]
        [THROW ValueError "Raised when a provided time string cannot be parsed using the required format."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
