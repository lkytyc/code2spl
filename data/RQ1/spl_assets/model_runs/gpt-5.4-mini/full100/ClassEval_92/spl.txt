# CLASS UserLoginDB

[DEFINE_WORKER: "Initialize a database connection from the given database name and create a cursor from that connection." __init__]
    [INPUTS]
        <REF> db_name </REF>: string "The database name or path used to open a SQLite connection."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a SQLite connection using the provided database name and store it on the instance as self.connection. RESULT self.connection]
            [COMMAND Create a cursor from self.connection and store it on the instance as self.cursor. RESULT self.cursor]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a user record into the users table with parameterized username and password values, then commits the transaction." insert_user]
    [INPUTS]
        <REF> username </REF>: unknown "Username value to insert into the users table."
        <REF> password </REF>: unknown "Password value to insert into the users table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL INSERT statement on the users table using the provided username and password values as bound parameters. RESULT insert_result]
            [COMMAND Commit the database transaction to persist the inserted row. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Execute a parameterized query to find a user record by username and return the first matching row from the database cursor." search_user_by_username]
    [INPUTS]
        <REF> self </REF>: object "Instance providing a database cursor used to execute the query and fetch results."
        <REF> username </REF>: string "Username value used as the query parameter for matching a user record."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: row_or_null "The first matching user row returned by the cursor, or null if no row is available."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL query on self.cursor to select all columns from users where username equals the provided username. RESULT query_result]
            [COMMAND Fetch the next row from the current cursor result set and store it in user. RESULT user]
            [COMMAND Return user as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes the user row matching the given username by executing a parameterized SQL delete statement and committing the transaction." delete_user_by_username]
    [INPUTS]
        <REF> username </REF>: str "The username used as the delete criterion for the users table."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the provided username as the bound parameter for a parameterized SQL DELETE statement targeting the users table. RESULT delete_target]
            [COMMAND Execute the DELETE FROM users WHERE username = ? statement through the database cursor with the username parameter. RESULT execute_result]
            [COMMAND Commit the database transaction to make the deletion permanent. RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a username lookup succeeds and whether the stored password at index 1 matches the supplied password, returning True only when both conditions are satisfied." validate_user_login]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the user lookup method."
        <REF> username </REF>: string "Username used to search for the user record."
        <REF> password </REF>: string "Password value to compare against the stored password field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when a user is found and its password field at index 1 equals the provided password; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the provided username to look up a user record by calling self.search_user_by_username(username), and store the returned value in user. RESULT user]
            [COMMAND If user is not None and the value at user[1] is equal to the provided password, return True immediately. RESULT true]
            [COMMAND Otherwise, return False. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
