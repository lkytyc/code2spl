# CLASS MahjongConnect

[DEFINE_WORKER: "Initialize the instance with the provided board size and icons, then build the board state using the internal board-construction method." __init__]
    [INPUTS]
        <REF> BOARD_SIZE </REF>: data value "Board size used to configure the instance state."
        <REF> ICONS </REF>: data value "Icon collection used to configure the instance state."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided BOARD_SIZE value to the instance attribute self.BOARD_SIZE. RESULT self.BOARD_SIZE]
            [COMMAND Assign the provided ICONS value to the instance attribute self.ICONS. RESULT self.ICONS]
            [COMMAND Call self.create_board() to construct the initial board state, then assign the returned value to self.board. RESULT self.board]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Create and return a two-dimensional board filled with independently random icon selections based on the configured board size." create_board]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[list] "A two-dimensional board whose row count is self.BOARD_SIZE[0] and column count is self.BOARD_SIZE[1], populated with random icons."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a two-dimensional list named board using an outer comprehension that iterates self.BOARD_SIZE[0] times and an inner comprehension that iterates self.BOARD_SIZE[1] times. RESULT board]
            [COMMAND Within the inner comprehension, call random.choice on self.ICONS for each cell so every board entry is independently selected at random. RESULT board]
            [COMMAND Return the completed board. RESULT board]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If either dimension in self.BOARD_SIZE is zero, the corresponding comprehension produces an empty list and the board shape contains empty rows or no rows.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct the board with the same comprehensions, allowing zero iterations in the outer or inner loop where applicable. RESULT board]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: If self.ICONS is empty, random.choice cannot select an element.]
        [LOG "random.choice failed because the icon pool was empty."]
        [THROW IndexError "Raised by random.choice when attempting to choose from an empty sequence."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether two board positions are distinct, within bounds, share the same icon, and are connected by a valid path." is_valid_move]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the board state, board size, and path validation helper."
        <REF> pos1 </REF>: tuple "First board position as a two-element coordinate pair."
        <REF> pos2 </REF>: tuple "Second board position as a two-element coordinate pair."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the move is valid; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack pos1 into x1 and y1, and unpack pos2 into x2 and y2. RESULT x1_y1_x2_y2]
            [COMMAND Check that x1, y1, x2, and y2 are all within the board bounds using inclusive lower bounds of 0 and exclusive upper bounds of self.BOARD_SIZE[0] for x coordinates and self.BOARD_SIZE[1] for y coordinates. RESULT bounds_valid]
            [COMMAND If any coordinate is out of bounds, return False immediately. RESULT false_out_of_bounds]
            [COMMAND Check whether pos1 and pos2 are the same position. RESULT same_position]
            [COMMAND If pos1 equals pos2, return False immediately. RESULT false_same_position]
            [COMMAND Compare the board values at self.board[x1][y1] and self.board[x2][y2] to verify that the two positions have the same icon. RESULT same_icon]
            [COMMAND If the two board values differ, return False immediately. RESULT false_different_icon]
            [COMMAND Call self.has_path(pos1, pos2) to determine whether a valid path exists between the two positions. RESULT path_exists]
            [COMMAND If no valid path exists, return False immediately. RESULT false_no_path]
            [COMMAND Return True when the positions are distinct, in bounds, have the same icon, and are connected by a valid path. RESULT true_valid_move]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determine whether a path exists between two grid positions by performing a stack-based traversal that follows only orthogonally adjacent cells with the same board value and stops early when the target is reached." has_path]
    [INPUTS]
        <REF> pos1 </REF>: tuple[int, int] "Starting grid position."
        <REF> pos2 </REF>: tuple[int, int] "Target grid position to search for."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if a matching-value path exists from pos1 to pos2; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty visited set to track processed positions. RESULT visited]
            [COMMAND Initialize a stack with the starting position pos1. RESULT stack]
            [COMMAND While the stack is not empty, pop the next position into current_pos. RESULT current_pos]
            [COMMAND Within the while loop, if current_pos is equal to pos2, return True immediately. RESULT True]
            [COMMAND Within the while loop, if current_pos has already been visited, skip the rest of this iteration and continue with the next stack item. RESULT continue]
            [COMMAND Within the while loop, add current_pos to visited and unpack it into x and y coordinates. RESULT x_y]
            [COMMAND Within the while loop, iterate over the four orthogonal direction offsets: (0, 1), (0, -1), (1, 0), and (-1, 0). RESULT direction]
            [COMMAND Within the while loop and for each direction, compute the neighbor coordinates new_x and new_y by adding the offset to x and y. RESULT new_x_new_y]
            [COMMAND Within the while loop and for each neighbor, if the neighbor is inside the board bounds 0 <= new_x < self.BOARD_SIZE[0] and 0 <= new_y < self.BOARD_SIZE[1], evaluate it further. RESULT in_bounds]
            [COMMAND Within the while loop and for each in-bounds neighbor, if the neighbor is not in visited and self.board[new_x][new_y] equals self.board[x][y], push the neighbor onto the stack. RESULT stack]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the while loop finishes because the stack becomes empty before pos2 is found, the search fails.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False after exhausting all reachable matching positions. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Clears the two board positions provided in pos1 and pos2 by replacing both cells with a space character." remove_icons]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the board to update."
        <REF> pos1 </REF>: tuple "First board coordinate pair to clear."
        <REF> pos2 </REF>: tuple "Second board coordinate pair to clear."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack pos1 into x1 and y1, and unpack pos2 into x2 and y2. RESULT coordinates]
            [COMMAND Use x1 and y1 to access the first target cell in self.board and assign a space character to it. RESULT self.board[x1][y1]]
            [COMMAND Use x2 and y2 to access the second target cell in self.board and assign a space character to it. RESULT self.board[x2][y2]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks each row of the board and returns False as soon as any non-space cell is found, otherwise returns True after confirming the board is empty." is_game_over]
    [INPUTS]
        <REF> self </REF>: object "The game instance containing the board to inspect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the board contains only space characters in every row; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate through each row in self.board. RESULT row]
            [COMMAND Within the row iteration, check whether any icon in the current row is not a space character using a short-circuit existential test. RESULT has_non_space_icon]
            [COMMAND If the current row contains any non-space icon, return False immediately and stop processing further rows. RESULT False]
            [COMMAND After all rows have been checked and no non-space icons were found, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
