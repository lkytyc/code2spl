[DEFINE_WORKER: "Checks whether a request should be allowed immediately based on its path prefix or, failing that, by validating a JWT user with a level above 2 and logging the current user info." filter]
    [INPUTS]
        <REF> request </REF>: dict "Incoming request containing path and method fields and used for JWT lookup."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the request is allowed; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read request['path'] into request_uri and request['method'] into method. RESULT request_uri, method]
            [COMMAND If request_uri satisfies self.is_start_with(request_uri), return True immediately. RESULT True]
            [COMMAND Otherwise, attempt to extract a JWT user token from the request. RESULT token]
            [COMMAND From the token, read token['user'] into user. RESULT user]
            [COMMAND If user['level'] is greater than 2, call self.set_current_user_info_and_log(user) and return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs while extracting the JWT user, reading the user, or checking the user level, return False.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Handle the exception by stopping authorization processing and returning False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the given request URI starts with any prefix in a fixed list and returns true on the first match, otherwise returns false." is_start_with]
    [INPUTS]
        <REF> self </REF>: object "Instance context used to call the method."
        <REF> request_uri </REF>: string "URI string to test against the configured prefixes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if request_uri starts with '/api' or '/login'; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize start_with as a fixed list containing the two prefixes '/api' and '/login'. RESULT start_with]
            [COMMAND Within the outer for loop over start_with, examine each prefix s one at a time. RESULT s]
            [COMMAND Within the loop, if request_uri starts with the current prefix s, return true immediately and stop processing further prefixes. RESULT true]
            [COMMAND If no prefix in start_with matches request_uri, return false after the loop completes. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extract the authorization token and user from the request, reject JWTs whose embedded date is at least 3 days old when the JWT starts with the user name, and otherwise return the token." get_jwt_user]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> request </REF>: dict "Incoming request containing a headers Authorization object with token data."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object or None "Returns the token object when it is accepted, or None when the embedded JWT date is at least 3 days old after matching the user name prefix."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the Authorization token from request['headers']['Authorization'] and assign it to token. RESULT token]
            [COMMAND Extract token['user'] and assign it to user. RESULT user]
            [COMMAND Check whether token['jwt'] starts with user['name']. RESULT jwt_prefix_matches_user_name]
            [COMMAND If token['jwt'] starts with user['name'], split token['jwt'] by user['name'] and take the substring at index 1 as jwt_str_date. RESULT jwt_str_date]
            [COMMAND If token['jwt'] starts with user['name'], parse jwt_str_date with datetime.datetime.strptime using the format "%Y-%m-%d" to produce jwt_date. RESULT jwt_date]
            [COMMAND Within the branch where token['jwt'] starts with user['name'], compute datetime.datetime.today() - jwt_date and check whether the result is greater than or equal to datetime.timedelta(days=3). RESULT is_jwt_at_least_3_days_old]
            [COMMAND If the JWT date is at least 3 days old, return None immediately. RESULT None]
            [COMMAND If the JWT prefix check fails or the JWT date is less than 3 days old, return token. RESULT token]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "The method reads the user's address into a local variable and logs the user's name, address, and current datetime at level 1." set_current_user_info_and_log]
    [INPUTS]
        <REF> self </REF>: object "Instance on which the method is called."
        <REF> user </REF>: dict "User data containing at least name and address fields."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the user's address from user['address'] and store it in host. RESULT host]
            [COMMAND Log a message composed of user['name'], host, and the string form of datetime.datetime.now() at level 1. RESULT logging.log call]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
