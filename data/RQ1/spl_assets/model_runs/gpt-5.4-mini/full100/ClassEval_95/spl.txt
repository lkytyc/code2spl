# CLASS Warehouse

[DEFINE_WORKER: "Initializes the instance with empty inventory and order mappings." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an empty dictionary and assign it to self.inventory to store products by product ID. RESULT self.inventory]
            [COMMAND Create an empty dictionary and assign it to self.orders to store orders. RESULT self.orders]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a new product to the inventory when the product ID is missing, or increases the stored quantity when the product already exists." add_product]
    [INPUTS]
        <REF> product_id </REF>: any hashable key "Identifier used to look up the product in the inventory."
        <REF> name </REF>: string "Product name stored when a new inventory entry is created."
        <REF> quantity </REF>: number "Amount to initialize or add to the product quantity."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id is absent from self.inventory. RESULT membership_check]
            [COMMAND If product_id is absent from self.inventory, create self.inventory[product_id] with name set to name and quantity set to quantity. RESULT new_inventory_entry]
            [COMMAND Otherwise, increase self.inventory[product_id]['quantity'] by quantity. RESULT updated_inventory_quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the quantity of an existing product in the inventory by adding the provided amount only when the product id is present." update_product_quantity]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
        <REF> product_id </REF>: any "Key used to identify the product in the inventory."
        <REF> quantity </REF>: number "Amount to add to the product's stored quantity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT _condition_met]
            [COMMAND If product_id is present in self.inventory, increase self.inventory[product_id]['quantity'] by quantity. RESULT _updated_inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the stored quantity for a product when its ID exists in the inventory, otherwise returns False." get_product_quantity]
    [INPUTS]
        <REF> product_id </REF>: any "Product identifier used to look up an entry in the inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The product quantity if the product exists in inventory, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether product_id exists as a key in self.inventory. RESULT exists_in_inventory]
            [COMMAND If product_id exists in self.inventory, return self.inventory[product_id]['quantity']. RESULT result]
            [COMMAND If product_id does not exist in self.inventory, return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates an order by checking stock availability, deducting inventory when enough quantity exists, recording the order as shipped, and returning False if stock is insufficient." create_order]
    [INPUTS]
        <REF> order_id </REF>: any "Identifier used as the key for the new order record."
        <REF> product_id </REF>: any "Identifier of the product whose stock is checked and updated."
        <REF> quantity </REF>: number "Requested quantity to order and deduct from inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns False when available stock is less than the requested quantity; otherwise the method completes without an explicit return value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the available quantity for the given product_id is greater than or equal to the requested quantity. RESULT stock_sufficient]
            [COMMAND If stock_sufficient is true, reduce the product inventory by exactly the requested quantity using a negative quantity update. RESULT inventory_updated]
            [COMMAND If stock_sufficient is true, create or overwrite the order entry at order_id with the product_id, quantity, and status set to Shipped. RESULT order_recorded]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Available product quantity is less than the requested quantity.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without updating inventory or creating the order record. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates an existing order's status in self.orders and returns False when the order id is missing." change_order_status]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the orders collection."
        <REF> order_id </REF>: key "The order identifier to look up in self.orders."
        <REF> status </REF>: any "The new status value to assign to the matched order."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean or null "Returns False when the order does not exist; otherwise returns no explicit value after updating the status."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT order_exists]
            [COMMAND If order_id exists in self.orders, update self.orders[order_id]['status'] to status. RESULT updated_order]
            [COMMAND If order_id does not exist in self.orders, return False immediately. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_id is not present in self.orders]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the status update because no matching order was found. RESULT no_update]
            [COMMAND Return False to signal failure to change the order status. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the status of a stored order when the given order ID exists in the orders mapping, otherwise return False." track_order]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the orders collection."
        <REF> order_id </REF>: any "Order identifier to look up in the orders collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The status value for the matching order, or False if the order ID is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether order_id exists as a key in self.orders. RESULT membership_check]
            [COMMAND If order_id is present in self.orders, retrieve self.orders[order_id]['status'] and return that status immediately. RESULT status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: order_id is not present in self.orders.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate that no matching order was found. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
