[DEFINE_WORKER: "Decrypt each character of a ciphertext by shifting alphabetic letters backward by a given amount while preserving case and leaving non-letters unchanged." caesar_decipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> ciphertext </REF>: string "Input text to decrypt."
        <REF> shift </REF>: integer "Number of positions to shift alphabetic characters backward."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decrypted plaintext string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty plaintext string. RESULT plaintext]
            [COMMAND Within the outer loop over each character in ciphertext, examine the current character one by one. RESULT char]
            [COMMAND Within the outer loop, if the current character is alphabetic, determine whether it is uppercase or lowercase. RESULT case_check]
            [COMMAND Within the alphabetic branch, if the current character is uppercase, set the ASCII offset to 65; otherwise set the ASCII offset to 97. RESULT ascii_offset]
            [COMMAND Within the alphabetic branch, compute the shifted character by converting the letter backward by shift positions, wrapping it within 26 letters, and converting it back to a character. RESULT shifted_char]
            [COMMAND Within the alphabetic branch, append the shifted character to plaintext. RESULT plaintext]
            [COMMAND Within the non-alphabetic branch, append the current character to plaintext without modification. RESULT plaintext]
            [COMMAND After processing all characters, return plaintext. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decrypt ciphertext by shifting each alphabetic character backward with a repeating key while preserving case and leaving non-letter characters unchanged." vigenere_decipher]
    [INPUTS]
        <REF> ciphertext </REF>: string "The encrypted text to decrypt character by character."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decrypted text with original non-alphabetic characters preserved."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize decrypted_text as an empty string and key_index as 0. RESULT decrypted_text, key_index]
            [COMMAND For each character char in ciphertext, determine whether char is alphabetic. RESULT char]
            [COMMAND Within the outer loop, if char is alphabetic, compute the repeating key character using key_index modulo the key length and convert it to a shift value from 0 to 25. RESULT shift]
            [COMMAND Within the alphabetic branch, decrypt char by shifting its lowercase form backward by shift positions modulo 26, then convert the result back to uppercase if the original char was uppercase. RESULT decrypted_char]
            [COMMAND Within the alphabetic branch, append the decrypted character to decrypted_text and increment key_index by 1 so only letters consume key characters. RESULT decrypted_text, key_index]
            [COMMAND Within the outer loop, if char is not alphabetic, append char unchanged to decrypted_text and do not change key_index. RESULT decrypted_text]
            [COMMAND After all characters have been processed, return decrypted_text. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decipher the encrypted text by rebuilding a rail-fence grid, marking zigzag positions, filling the empty slots with ciphertext characters, and then reading the grid again in the same zigzag order to produce the plaintext." rail_fence_decipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference used to call the method."
        <REF> encrypted_text </REF>: string "Ciphertext to be decoded using the rail-fence pattern."
        <REF> rails </REF>: integer "Number of rails used to reconstruct the zigzag grid."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The deciphered plaintext returned by the method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a 2D fence grid with rails rows and len(encrypted_text) columns, filling every cell with newline characters, then initialize direction to -1 and row and col to 0. RESULT fence, direction, row, col]
            [COMMAND Within a for loop that runs exactly len(encrypted_text) times, reverse direction whenever row is 0 or rails - 1, mark the current fence cell at [row][col] with an empty string, advance col by 1, and move row by the current direction. RESULT fence]
            [COMMAND Initialize index to 0 before filling the marked fence positions. RESULT index]
            [COMMAND Within nested for loops over i from 0 to rails - 1 and j from 0 to len(encrypted_text) - 1, if fence[i][j] is an empty string, place encrypted_text[index] into that cell and increment index by 1. RESULT fence, index]
            [COMMAND Initialize plain_text to an empty string, then reset direction to -1 and row and col to 0 for the decoding traversal. RESULT plain_text, direction, row, col]
            [COMMAND Within a for loop that runs exactly len(encrypted_text) times, reverse direction whenever row is 0 or rails - 1, append fence[row][col] to plain_text, advance col by 1, and move row by the current direction. RESULT plain_text]
            [COMMAND Return plain_text. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
