# CLASS RegexUtils

[DEFINE_WORKER: "Return True when the regular expression matches the beginning of the input text, and otherwise return False." match]
    [INPUTS]
        <REF> pattern </REF>: string or regex pattern "Regular expression pattern to match against the text."
        <REF> text </REF>: string "Input text to test against the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if re.match(pattern, text) returns a truthy match object; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.match with pattern and text, and store the returned match object in ans. RESULT ans]
            [COMMAND If ans is truthy, return True immediately. RESULT True]
            [COMMAND If ans is falsy, return False immediately. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the list of all non-overlapping regex matches found in the given text using the provided pattern." findall]
    [INPUTS]
        <REF> self </REF>: object "Method receiver, not used by the implementation."
        <REF> pattern </REF>: string "Regular expression pattern used to search the text."
        <REF> text </REF>: string "Input string to search for matches."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of all non-overlapping matches returned by re.findall, or tuples when the pattern contains capturing groups."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.findall with the provided pattern and text to extract all non-overlapping matches. RESULT matches]
            [COMMAND Return the matches produced by re.findall without further transformation. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The regular expression pattern is invalid.]
        [LOG "re.findall may raise a regex-related exception while compiling or evaluating the pattern."]
        [THROW re.error "Raised when the pattern cannot be parsed as a valid regular expression."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Split the input text using the given regular expression pattern and return the resulting list of substrings." split]
    [INPUTS]
        <REF> self </REF>: object "Instance reference; not used by the method logic."
        <REF> pattern </REF>: str "Regular expression pattern used as the split delimiter."
        <REF> text </REF>: str "Input string to be split."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of substrings produced by applying re.split to the pattern and text."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.split with the provided pattern and text to produce the split result. RESULT result]
            [COMMAND Return the split result immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a new string produced by applying a regular-expression substitution with the given pattern, replacement, and input text." sub]
    [INPUTS]
        <REF> self </REF>: object "Instance reference for the method."
        <REF> pattern </REF>: regex pattern "Regular expression used to match substrings in the input text."
        <REF> replacement </REF>: string or function "Value used to replace each regex match."
        <REF> text </REF>: string "Input string to transform."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A new string with all matches replaced according to the pattern and replacement."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call re.sub with the provided pattern, replacement, and input text to perform the substitution. RESULT result]
            [COMMAND Return the substituted string as the method output. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a regular expression pattern string that matches email-like addresses with a top-level domain of at least two alphabetic characters." generate_email_pattern]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: text "The regular expression pattern string for matching email-like addresses."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]

    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a regular expression pattern string that matches U.S.-style phone numbers in the form 123-456-7890 with word boundaries." generate_phone_number_pattern]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Regular expression pattern string for matching phone numbers in the form 123-456-7890."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the regular expression pattern string \b\d{3}-\d{3}-\d{4}\b to the local variable pattern. RESULT pattern]
            [COMMAND Return the pattern string from the method. RESULT pattern]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a regular expression pattern that matches sentence-ending punctuation followed by one or two whitespace characters when the next character is uppercase." generate_split_sentences_pattern]
    [INPUTS]
        <REF> self </REF>: object "Instance of the class containing the method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Regular expression pattern used for splitting sentences."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the raw regular expression string pattern r'[.!?][\s]{1,2}(?=[A-Z])' to the local variable pattern. RESULT pattern]
            [COMMAND Return the pattern string as the method result. RESULT pattern]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate a split pattern and use it to split the input text into sentence segments." split_sentences]
    [INPUTS]
        <REF> text </REF>: string "The text to be split into sentences."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The result returned by splitting the input text with the generated pattern."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Generate the sentence-splitting pattern by calling the internal pattern builder method. RESULT pattern]
            [COMMAND Split the input text using the generated pattern and return the resulting segments. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates a phone number by generating a phone number pattern and matching the input against it." validate_phone_number]
    [INPUTS]
        <REF> phone_number </REF>: unknown "The phone number value to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the phone number matches the generated pattern, otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Generate the phone number pattern by calling self.generate_phone_number_pattern(). RESULT pattern]
            [COMMAND Match the generated pattern against phone_number using self.match(pattern, phone_number). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates an email-matching pattern and returns all matches found in the provided text." extract_email]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the email pattern generator and search capability."
        <REF> text </REF>: string "Input text to scan for email-like matches."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "All matches found in the input text using the generated email pattern."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.generate_email_pattern() to obtain the email pattern used for matching. RESULT pattern]
            [COMMAND Use self.findall(pattern, text) to find all matches of the generated pattern in the provided text and return the matches. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
