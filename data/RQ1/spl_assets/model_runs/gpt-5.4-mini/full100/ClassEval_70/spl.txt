# CLASS PersonRequest

[DEFINE_WORKER: "Initializes the object by validating the provided name, sex, and phone number before storing the validated values on the instance." __init__]
    [INPUTS]
        <REF> name </REF>: str "Raw name value to validate and assign."
        <REF> sex </REF>: str "Raw sex value to validate and assign."
        <REF> phoneNumber </REF>: str "Raw phone number value to validate and assign."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Validate the provided name by calling the internal name validation method and assign the returned value to the instance name attribute. RESULT _validated_name]
            [COMMAND Validate the provided sex by calling the internal sex validation method and assign the returned value to the instance sex attribute. RESULT _validated_sex]
            [COMMAND Validate the provided phone number by calling the internal phone number validation method and assign the returned value to the instance phoneNumber attribute. RESULT _validated_phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that name is present and no longer than 33 characters, returning None for invalid values and the original name otherwise." _validate_name]
    [INPUTS]
        <REF> name </REF>: str "The name value to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The validated name, or None when the value is empty or longer than 33 characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether name is falsy. RESULT falsy_check]
            [COMMAND If name is falsy, return None immediately. RESULT None]
            [COMMAND Check whether the length of name is greater than 33. RESULT length_check]
            [COMMAND If the length of name is greater than 33, return None immediately. RESULT None]
            [COMMAND Return name unchanged. RESULT name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that the input sex value is one of the allowed strings and return it, otherwise return None." _validate_sex]
    [INPUTS]
        <REF> self </REF>: object "Instance reference used to access the method."
        <REF> sex </REF>: str "Text value to validate against the allowed set."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "Returns the original sex value when it is allowed, otherwise returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether sex is not in the allowed values Man, Woman, or UGM. RESULT is_invalid]
            [COMMAND If sex is invalid, return None immediately as an early exit. RESULT none_result]
            [COMMAND If sex is valid, return the original sex value unchanged. RESULT validated_sex]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that the input phone number is present, exactly 11 characters long, and composed only of digits before returning it unchanged; otherwise return None." _validate_phoneNumber]
    [INPUTS]
        <REF> phoneNumber </REF>: str "The phone number string to validate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The original phone number when it passes validation, or None when it is missing or invalid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether phoneNumber is falsy; if it is, return None immediately as an early guard clause. RESULT none_if_missing]
            [COMMAND Check whether phoneNumber has length exactly 11 and consists only of digits; if the length is not 11 or any character is not a digit, return None immediately. RESULT none_if_invalid]
            [COMMAND Return the original phoneNumber unchanged after it passes both validation checks. RESULT validated_phoneNumber]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
