[DEFINE_WORKER: "Compute a mean reciprocal-rank style score for either a single tuple input or a list of tuple inputs by finding the first positive rank-weighted value in each sublist." mrr]
    [INPUTS]
        <REF> data </REF>: list or tuple "Either a single (sub_list, total_num) tuple or a list of such tuples."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A pair containing the computed mean score and either a single-item list or a list of per-item scores."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Verify that data is either a list or a tuple; otherwise raise an exception because the input must be a tuple([0,...,1,...],int) or an iteration of list of tuple. RESULT _validated_input]
            [COMMAND If data is empty, return 0.0 and [0.0] immediately. RESULT _output]
            [COMMAND If data is a tuple, unpack it into sub_list and total_num and convert sub_list to a NumPy array. RESULT _tuple_parts]
            [COMMAND Within the tuple branch, if total_num equals 0, return 0.0 and [0.0] immediately. RESULT _output]
            [COMMAND Within the tuple branch, otherwise build reciprocal rank weights using 1.0 divided by positions 1 through len(sub_list), multiply sub_list elementwise by those weights, and store the result as mr_np. RESULT mr_np]
            [COMMAND Within the tuple branch, scan mr_np in order and keep the first value greater than 0 as mr; if no positive value is found, keep mr as 0.0. RESULT mr]
            [COMMAND Within the tuple branch, return mr and [mr]. RESULT _output]
            [COMMAND If data is a list, initialize separate_result as an empty list. RESULT separate_result]
            [COMMAND Within the list branch, for each pair (sub_list, total_num) in data, convert sub_list to a NumPy array. RESULT _sub_list_array]
            [COMMAND Within the list branch, if total_num equals 0, set mr to 0.0 for that pair. RESULT mr]
            [COMMAND Within the list branch, otherwise build reciprocal rank weights using 1.0 divided by positions 1 through len(sub_list), multiply sub_list elementwise by those weights, and store the result as mr_np. RESULT mr_np]
            [COMMAND Within the list branch, scan mr_np in order and keep the first value greater than 0 as mr; if no positive value is found, keep mr as 0.0. RESULT mr]
            [COMMAND Within the list branch, append mr to separate_result for each pair. RESULT separate_result]
            [COMMAND Within the list branch, after processing all pairs, return the mean of separate_result and separate_result itself. RESULT _output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is neither a list nor a tuple]
        [LOG "Input validation failed because the method only accepts a tuple([0,...,1,...],int) or an iteration of list of tuple."]
        [THROW Exception "Signals that the input type is invalid."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute an average precision-like score from either a single tuple input or a list of tuple inputs by ranking nonzero items and normalizing by the provided total count." map]
    [INPUTS]
        <REF> data </REF>: tuple or list of tuples "A single (sub_list, total_num) tuple or a list of such tuples."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A pair containing the mean AP-like score and either a single-item list with that score or the per-tuple score list."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether data is neither a list nor a tuple; if so, raise an exception stating that the input must be a tuple([0,...,1,...],int) or an iteration of list of tuple. RESULT type_validation]
            [COMMAND If data is an empty sequence, return 0.0 and [0.0] immediately. RESULT empty_input_return]
            [COMMAND If data is a tuple, unpack it into sub_list and total_num, convert sub_list to a NumPy array, and continue with the tuple branch. RESULT tuple_branch]
            [COMMAND Within the tuple branch, if total_num equals 0, return 0.0 and [0.0] immediately to avoid division by zero. RESULT zero_total_tuple_return]
            [COMMAND Within the tuple branch, otherwise build ranking_array as the reciprocal of positions 1 through len(sub_list), assign increasing ranks starting at 1 to each nonzero element in sub_list while assigning 0 to each zero element, compute ap as the weighted sum of the rank list times ranking_array divided by total_num, and return ap with [ap]. RESULT tuple_ap]
            [COMMAND If data is a list, initialize separate_result as an empty list and iterate over each (sub_list, total_num) pair in order. RESULT list_branch]
            [COMMAND Within the list branch, for each pair convert sub_list to a NumPy array, and if total_num equals 0 set ap to 0.0. RESULT zero_total_list_item]
            [COMMAND Within the list branch, for each pair whose total_num is not 0, build ranking_array as the reciprocal of positions 1 through len(sub_list), assign increasing ranks starting at 1 to each nonzero element in sub_list while assigning 0 to each zero element, and compute ap as the weighted sum divided by total_num. RESULT list_item_ap]
            [COMMAND Within the list branch, append each ap value to separate_result, then after processing all pairs return the mean of separate_result and separate_result itself. RESULT list_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is neither a list nor a tuple]
        [LOG "The input type is invalid for map; expected a tuple or a list of tuples."]
        [THROW Exception "Indicates the input must be a tuple([0,...,1,...],int) or an iteration of list of tuple."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
