# CLASS StudentDatabaseProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided database name on the object." __init__]
    [INPUTS]
        <REF> database_name </REF>: object "The database name value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided database_name value to the instance attribute self.database_name. RESULT self.database_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates the students table in the configured SQLite database if it does not already exist, then commits and closes the connection." create_student_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using self.database_name and store the connection in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Define the SQL statement to create the students table with columns id as INTEGER PRIMARY KEY, name as TEXT, age as INTEGER, gender as TEXT, and grade as INTEGER, using IF NOT EXISTS so the operation is idempotent. RESULT create_table_query]
            [COMMAND Execute create_table_query through cursor to create the table if needed. RESULT cursor]
            [COMMAND Commit the transaction on conn to persist the table creation. RESULT conn]
            [COMMAND Close conn to release the database connection. RESULT conn]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert a student record into the students table using a SQLite connection and parameterized query, then commit and close the database connection." insert_student]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database_name attribute used to open the SQLite connection."
        <REF> student_data </REF>: dict "Mapping containing the student's name, age, gender, and grade values."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Define the parameterized SQL INSERT statement for the students table with columns name, age, gender, and grade. RESULT insert_query]
            [COMMAND Execute insert_query on cursor with the values student_data['name'], student_data['age'], student_data['gender'], and student_data['grade'] as the bound parameters. RESULT insert_result]
            [COMMAND Commit the transaction on conn to persist the inserted student row. RESULT commit_result]
            [COMMAND Close conn to release the database connection. RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches the students table for all rows matching the given name and returns the fetched results after closing the database connection." search_student_by_name]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database_name attribute used to open the SQLite connection."
        <REF> name </REF>: string "Student name value bound to the SQL query parameter."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "All rows returned by the query as fetched from the cursor."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Set select_query to the constant SQL statement SELECT * FROM students WHERE name = ?. RESULT select_query]
            [COMMAND Execute select_query on cursor with name bound as the single query parameter. RESULT query_execution]
            [COMMAND Fetch all remaining rows from cursor and store them in result. RESULT result]
            [COMMAND Close the database connection stored in conn. RESULT conn]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Delete the student record matching the provided name from the students table using a parameterized SQL query and then commit and close the database connection." delete_student_by_name]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database_name attribute used to open the SQLite connection."
        <REF> name </REF>: string "Student name used as the deletion criterion in the SQL WHERE clause."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Define the parameterized SQL delete statement for rows in the students table where name equals the supplied parameter. RESULT delete_query]
            [COMMAND Execute delete_query with the provided name bound as a single parameter tuple. RESULT delete_execution]
            [COMMAND Commit the transaction on conn to persist the deletion. RESULT commit_result]
            [COMMAND Close conn to release the database connection. RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
