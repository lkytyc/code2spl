# CLASS TextFileProcessor

[DEFINE_WORKER: "Initialize the instance by storing the provided file path on the object." __init__]
    [INPUTS]
        <REF> file_path </REF>: any "Path value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided file_path argument to the instance attribute self.file_path. RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "This method opens the configured file path, parses the file contents as JSON, and returns the resulting data structure." read_file_as_json]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file_path attribute that points to the JSON file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> data </REF>: object "The Python object produced by json.load from the file contents."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file located at self.file_path in read mode using a context manager. RESULT file]
            [COMMAND Parse the opened file contents as JSON and store the parsed value in data. RESULT data]
            [COMMAND Return data. RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads the file at self.file_path in text mode and returns its full contents as a string." read_file]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the file_path attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The complete text content of the file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file located at self.file_path in read mode using the default text encoding. RESULT file]
            [COMMAND Read the entire contents of the opened file into memory. RESULT content]
            [COMMAND Return the full file contents as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes the provided content string to the file path stored on the instance by opening that path in write mode." write_file]
    [INPUTS]
        <REF> content </REF>: string "Text content to be written into the target file."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.file_path in write mode as file. RESULT file]
            [COMMAND Write the provided content into the open file object. RESULT written_content]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads file content, removes every non-alphabetic character, writes the cleaned text back to the file, and returns the sanitized string." process_file]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current file content into content. RESULT content]
            [COMMAND Filter content so that only alphabetic characters remain and join them into a new string. RESULT content]
            [COMMAND Write the filtered content back to the file. RESULT None]
            [COMMAND Return the filtered content. RESULT content]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
