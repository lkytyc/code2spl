[DEFINE_WORKER: "Decode HTML-style numeric character entities in a string, converting valid decimal or hexadecimal references to characters while copying ordinary text unchanged and returning early on malformed input in certain cases." replace]
    [INPUTS]
        <REF> string </REF>: string "Input text to scan and decode."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Decoded string with numeric entities replaced by their character equivalents."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an output list, set the current position to 0, and store the input string length. RESULT out, pos, length]
            [COMMAND While the current position is less than length minus 2, inspect the current character and the next character to detect the start of a numeric entity marker '&#'. RESULT loop_scan]
            [COMMAND Within the outer scan loop, when the current slice begins with '&#', set the entity start to position plus 2, mark the entity as decimal by default, and read the first character after the marker. RESULT start, is_hex, first_char]
            [COMMAND Within the entity parse path, if the first character is 'x' or 'X', advance the entity start by 1 and mark the entity as hexadecimal. RESULT start, is_hex]
            [COMMAND Within the entity parse path, if the adjusted start equals the string length, return the characters accumulated so far as the final result. RESULT partial_result]
            [COMMAND Within the entity parse path, set the scan end to the entity start, then advance end while it is still inside the string and the current character satisfies the hexadecimal-digit check. RESULT end]
            [COMMAND Within the entity parse path, if end is still inside the string and the character at end is ';', attempt to convert the substring from start to end into an integer using base 16 when hexadecimal was selected, otherwise base 10. RESULT entity_value]
            [COMMAND Within the entity parse path, if integer conversion succeeds, append the Unicode character for that value to the output list, advance the current position to one past the semicolon, and continue the outer loop immediately. RESULT out, pos]
            [COMMAND Within the outer scan loop, when the current position does not begin a valid numeric entity or the entity parse path is not taken, append the current character to the output list and advance the position by 1. RESULT out, pos]
            [COMMAND After the scan loop finishes, join the accumulated output characters into a string and return it. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When the entity marker is incomplete because the adjusted start equals the string length, stop processing immediately and return the partially decoded output.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the characters accumulated so far without processing any remaining input. RESULT partial_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When integer conversion of the entity substring fails inside the valid semicolon-terminated entity path, stop processing immediately and return the partially decoded output.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the characters accumulated so far without processing any remaining input. RESULT partial_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns True when the input character is a decimal digit or a hexadecimal letter from a to f in either lowercase or uppercase, and otherwise returns False." is_hex_char]
    [INPUTS]
        <REF> char </REF>: str "Single character to test for decimal digit or hexadecimal letter membership."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Boolean indicating whether the character is a valid hexadecimal digit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether the input character is a decimal digit using char.isdigit(). RESULT is_digit]
            [COMMAND If the decimal-digit check is true, return True immediately due to short-circuit evaluation. RESULT result]
            [COMMAND Otherwise, convert the character to lowercase and test whether it lies between 'a' and 'f', inclusive. RESULT is_hex_letter]
            [COMMAND Return the result of the lowercase range check as the final boolean value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
