# CLASS PushBoxGame

[DEFINE_WORKER: "Initializes the game state by storing the map reference and resetting the player position, collections, target count, and game-over flag before starting game setup." __init__]
    [INPUTS]
        <REF> map </REF>: any "The map object or reference used to initialize the instance state."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the incoming map reference in self.map. RESULT self.map]
            [COMMAND Initialize self.player_row to 0. RESULT self.player_row]
            [COMMAND Initialize self.player_col to 0. RESULT self.player_col]
            [COMMAND Create a new empty list and assign it to self.targets. RESULT self.targets]
            [COMMAND Create a new empty list and assign it to self.boxes. RESULT self.boxes]
            [COMMAND Initialize self.target_count to 0. RESULT self.target_count]
            [COMMAND Initialize self.is_game_over to False. RESULT self.is_game_over]
            [COMMAND Call self.init_game() to perform further game initialization. RESULT self.init_game()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the entire map to locate the player, targets, and boxes, updating their stored positions and counts." init_game]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row index from 0 to len(self.map) - 1. RESULT row]
            [COMMAND Within the outer row loop, iterate over each column index from 0 to len(self.map[row]) - 1. RESULT col]
            [COMMAND Within the nested row and column loops, inspect the cell at self.map[row][col]. RESULT cell_value]
            [COMMAND If the cell value is "O", store the current row and column as the player position. RESULT self.player_row, self.player_col]
            [COMMAND Else if the cell value is "G", append the current position tuple to self.targets and increment self.target_count by 1. RESULT self.targets, self.target_count]
            [COMMAND Else if the cell value is "X", append the current position tuple to self.boxes. RESULT self.boxes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts how many boxes are on targets, marks the game over when all targets are covered, and returns the game-over state." check_win]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize box_on_target_count to 0. RESULT box_on_target_count]
            [COMMAND Iterate through each box in self.boxes, and within that loop increment box_on_target_count by 1 for every box that is also present in self.targets. RESULT box_on_target_count]
            [COMMAND If box_on_target_count equals self.target_count, set self.is_game_over to True. RESULT self.is_game_over]
            [COMMAND Return self.is_game_over. RESULT self.is_game_over]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Moves the player one step in the chosen direction, optionally pushes a box if the destination is open, and returns whether the win condition is met." move]
    [INPUTS]
        <REF> direction </REF>: str "A movement command: w, s, a, or d."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "The return value of check_win() after any movement attempt."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the tentative player row and column to the current player position. RESULT new_player_row, new_player_col]
            [COMMAND Within the direction check, if direction is "w", decrement the tentative row by 1; else if direction is "s", increment the tentative row by 1; else if direction is "a", decrement the tentative column by 1; else if direction is "d", increment the tentative column by 1. RESULT new_player_row, new_player_col]
            [COMMAND If the map cell at the tentative player position is not "#", continue with movement processing. RESULT movement_allowed]
            [COMMAND Within the non-wall destination path, if the tentative player position contains a box, compute the box's next row and column by extending the same movement delta one more step beyond the current player position. RESULT new_box_row, new_box_col]
            [COMMAND Within the box-push path, if the map cell at the tentative box position is not "#", remove the box from its current position, append it at the new box position, and update the player's row and column to the tentative player position. RESULT self.boxes, self.player_row, self.player_col]
            [COMMAND Within the non-wall destination path, if the tentative player position does not contain a box, update the player's row and column to the tentative player position. RESULT self.player_row, self.player_col]
            [COMMAND Return the result of check_win(). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
