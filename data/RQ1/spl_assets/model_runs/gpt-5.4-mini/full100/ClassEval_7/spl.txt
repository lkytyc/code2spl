# CLASS BalancedBrackets

[DEFINE_WORKER: "Initializes the object with an empty stack, matching bracket lists, and the provided expression string." __init__]
    [INPUTS]
        <REF> expr </REF>: string "The expression to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an empty stack and assign it to the instance state. RESULT self.stack]
            [COMMAND Assign the list of left brackets containing '(', '{', and '[' to the instance state. RESULT self.left_brackets]
            [COMMAND Assign the list of right brackets containing ')', '}', and ']' to the instance state. RESULT self.right_brackets]
            [COMMAND Store the input expression on the instance for later use. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters self.expr so it retains only characters that are present in either self.left_brackets or self.right_brackets." clear_expr]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.expr and iterate through its characters one by one. RESULT character_stream]
            [COMMAND Keep only characters c for which c is contained in self.left_brackets or self.right_brackets, using a short-circuit OR condition. RESULT filtered_characters]
            [COMMAND Join the retained characters into a new string. RESULT new_expr]
            [COMMAND Assign the newly constructed string back to self.expr, replacing the original expression content. RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate whether the expression has properly matched and nested parentheses, braces, and brackets using a stack-based scan." check_balanced_brackets]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True when all brackets are balanced and correctly nested; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Clear the current expression state before validation. RESULT self.clear_expr()]
            [COMMAND Within the full left-to-right scan of self.expr, inspect each character Brkt in order. RESULT loop_over_expression]
            [COMMAND Within the loop, if Brkt is an opening bracket from self.left_brackets, push Brkt onto self.stack. RESULT self.stack]
            [COMMAND Within the loop, if Brkt is not an opening bracket, pop the most recent opening bracket from self.stack into Current_Brkt. RESULT Current_Brkt]
            [COMMAND Within the loop, if Current_Brkt is "(", require Brkt to be ")"; otherwise return False immediately. RESULT false_on_parenthesis_mismatch]
            [COMMAND Within the loop, if Current_Brkt is "{", require Brkt to be "}"; otherwise return False immediately. RESULT false_on_brace_mismatch]
            [COMMAND Within the loop, if Current_Brkt is "[", require Brkt to be "]"; otherwise return False immediately. RESULT false_on_bracket_mismatch]
            [COMMAND After the loop finishes, if self.stack is not empty, return False because unmatched opening brackets remain. RESULT false_on_remaining_stack]
            [COMMAND If no mismatch is found and the stack is empty, return True. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
