# CLASS EmailClient

[DEFINE_WORKER: "Initializes a new instance by storing the provided address and capacity, and creating an empty inbox list." __init__]
    [INPUTS]
        <REF> addr </REF>: any "The address value to store on the instance."
        <REF> capacity </REF>: any "The capacity value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "The constructor does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming addr value directly to the instance field self.addr. RESULT self.addr]
            [COMMAND Assign the incoming capacity value directly to the instance field self.capacity. RESULT self.capacity]
            [COMMAND Initialize self.inbox as a new empty list. RESULT self.inbox]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Send an email to the receiver when there is room in the inbox, otherwise clear space and report failure." send_to]
    [INPUTS]
        <REF> recv </REF>: object "Recipient object with inbox capacity checks and an inbox list."
        <REF> content </REF>: any "Email content to store in the message."
        <REF> size </REF>: number "Size of the email used for capacity checking."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the email was appended to the receiver inbox; false if the inbox was cleared instead."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the receiver inbox is not full enough to accept one more email of the given size. RESULT can_send]
            [COMMAND If the receiver has room, create a timestamp using the current date and time formatted as YYYY-MM-DD HH:MM:SS. RESULT timestamp]
            [COMMAND If the receiver has room, build an email record with sender set to the sender address, receiver set to the receiver address, content, size, time, and state set to unread. RESULT email]
            [COMMAND If the receiver has room, append the email record to the receiver inbox. RESULT recv.inbox]
            [COMMAND If the receiver has room, return true to indicate the send succeeded. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The receiver inbox cannot accept one more email of the given size.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call the inbox clearing routine with the email size to free space. RESULT self.clear_inbox(size)]
            [COMMAND Return false to indicate the email was not delivered. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the first unread inbox item after marking it as read, or return None if the inbox is empty or contains no unread items." fetch]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the inbox list of message records."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object or null "The first unread inbox item after it is marked read, or None when no unread item is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.inbox is empty by testing whether its length is 0. RESULT inbox_is_empty]
            [COMMAND If self.inbox is empty, return None immediately. RESULT None]
            [COMMAND Otherwise, iterate through all valid inbox indices from 0 up to, but not including, len(self.inbox). RESULT i]
            [COMMAND Within the outer for loop over i, inspect self.inbox[i]['state'] and test whether it is exactly "unread". RESULT is_unread]
            [COMMAND Within the outer for loop, if self.inbox[i]['state'] is "unread", set self.inbox[i]['state'] to "read" and return self.inbox[i] immediately. RESULT unread_item]
            [COMMAND If no inbox item has state "unread" after scanning the full range, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether adding one more email of the given size would exceed the current capacity after computing the occupied size." is_full_with_one_more_email]
    [INPUTS]
        <REF> self </REF>: object "Object providing occupied size and capacity information."
        <REF> size </REF>: number "Size of the email to test for addition."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the current occupied size plus the given size is greater than capacity; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_occupied_size() and store the returned value in occupied_size. RESULT occupied_size]
            [COMMAND Compare occupied_size plus size against self.capacity using a strict greater-than check. RESULT comparison_result]
            [COMMAND Return True if occupied_size plus size is greater than self.capacity; otherwise return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates the total occupied size by summing the size of each email in the inbox." get_occupied_size]
    [INPUTS]
        <REF> self </REF>: object "Instance containing an inbox collection of emails."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The summed size of all emails in the inbox."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize occupied_size to 0 as the starting total. RESULT occupied_size]
            [COMMAND Iterate over each email in self.inbox. RESULT email]
            [COMMAND Within the loop, add email["size"] to occupied_size. RESULT occupied_size]
            [COMMAND After processing all emails, return occupied_size. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove emails from the front of the inbox until enough space has been freed or the inbox becomes empty, after first returning immediately when the address list is empty." clear_inbox]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the address list and inbox list."
        <REF> size </REF>: number "Target amount of space to free."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value unless the address list is empty, in which case it returns immediately."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.addr is empty; if it is empty, return immediately without modifying the inbox or freeing space. RESULT early_return]
            [COMMAND Initialize freed_space to 0 as the starting accumulated freed space. RESULT freed_space]
            [COMMAND While freed_space is less than size and self.inbox is non-empty, take the first email from the inbox, add that email's size to freed_space, and remove that first email from self.inbox in FIFO order. RESULT self.inbox]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
