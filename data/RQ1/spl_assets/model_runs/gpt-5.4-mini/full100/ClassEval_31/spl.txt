[DEFINE_WORKER: "Compute the Pearson correlation coefficient from two input sequences by calculating their means, covariance numerator, and standard-deviation-based denominator, then return the ratio or 0 when the denominator is zero." correlation_coefficient]
    [INPUTS]
        <REF> data1 </REF>: sequence of numbers "First input sequence used to compute the mean, deviation sum, and one side of the denominator."
        <REF> data2 </REF>: sequence of numbers "Second input sequence used to compute the mean, deviation sum, and the other side of the denominator."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The Pearson correlation coefficient, or 0 when the denominator is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of data1. RESULT n]
            [COMMAND Compute mean1 as the sum of data1 divided by n. RESULT mean1]
            [COMMAND Compute mean2 as the sum of data2 divided by n. RESULT mean2]
            [COMMAND For each index i from 0 to n - 1, compute the product of data1[i] minus mean1 and data2[i] minus mean2, then sum all products into numerator. RESULT numerator]
            [COMMAND For each index i from 0 to n - 1, compute the squared deviation of data1[i] from mean1 and sum the values; take the square root of that sum. RESULT data1_scale]
            [COMMAND For each index i from 0 to n - 1, compute the squared deviation of data2[i] from mean2 and sum the values; take the square root of that sum. RESULT data2_scale]
            [COMMAND Multiply the two square-root results to form the denominator. RESULT denominator]
            [COMMAND If the denominator is not zero, return numerator divided by denominator; otherwise return 0. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The denominator equals 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 instead of dividing by zero. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the sample skewness of a numeric dataset using the mean, variance, and standard deviation, returning zero when the standard deviation is zero." skewness]
    [INPUTS]
        <REF> data </REF>: sequence of numbers "Input dataset used to compute the skewness statistic."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed skewness value, or 0 when the standard deviation is zero."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute n as the length of data. RESULT n]
            [COMMAND Compute mean as the sum of data divided by n. RESULT mean]
            [COMMAND Compute variance as the average of squared deviations from mean over all values in data. RESULT variance]
            [COMMAND Compute std_deviation as the square root of variance. RESULT std_deviation]
            [COMMAND If std_deviation is not zero, compute skewness as the sum of cubed deviations from mean times n divided by (n - 1) * (n - 2) * std_deviation cubed; otherwise set skewness to 0. RESULT skewness]
            [COMMAND Return skewness. RESULT skewness]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: std_deviation equals zero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set skewness to 0 to avoid division by zero. RESULT skewness]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the population kurtosis of the input data by calculating the mean, standard deviation, and fourth central moment, returning NaN when the standard deviation is zero." kurtosis]
    [INPUTS]
        <REF> data </REF>: sequence of numbers "Input values used to compute kurtosis."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The excess kurtosis of the input data, or NaN when all values are identical."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the number of elements in data and store it as n. RESULT n]
            [COMMAND Compute the arithmetic mean of data by dividing the sum of data by n, and store it as mean. RESULT mean]
            [COMMAND Compute the population standard deviation from the squared deviations of data around mean, dividing by n and taking the square root, then store it as std_dev. RESULT std_dev]
            [COMMAND If std_dev is equal to 0, return math.nan immediately. RESULT math.nan]
            [COMMAND Create centered_data by subtracting mean from each element of data. RESULT centered_data]
            [COMMAND Compute fourth_moment as the average of the fourth powers of the values in centered_data, dividing by n. RESULT fourth_moment]
            [COMMAND Compute kurtosis_value as (fourth_moment divided by std_dev raised to the fourth power) minus 3. RESULT kurtosis_value]
            [COMMAND Return kurtosis_value. RESULT kurtosis_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When std_dev equals 0 after the standard deviation calculation.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return math.nan without computing centered_data, fourth_moment, or kurtosis_value. RESULT math.nan]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the Gaussian probability density value for each input data point using the given mean and standard deviation." pdf]
    [INPUTS]
        <REF> data </REF>: sequence of numbers "Input values to evaluate with the normal density formula."
        <REF> mu </REF>: number "Mean parameter used as the center of the Gaussian distribution."
        <REF> sigma </REF>: number "Standard deviation parameter used as the scale of the Gaussian distribution."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of numbers "A list of Gaussian probability density values corresponding to each element in data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each value x in data, compute the Gaussian density using 1 / (sigma * sqrt(2 * pi)) multiplied by exp(-0.5 * ((x - mu) / sigma) squared). RESULT pdf_values]
            [COMMAND Return the computed list of density values. RESULT pdf_values]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If data is empty, the list comprehension produces an empty list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over no elements and produce an empty result list. RESULT pdf_values]
            [COMMAND Return the empty list. RESULT pdf_values]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: If sigma is zero, the density formula attempts division by zero.]
        [LOG "A division by zero occurs while evaluating the Gaussian density expression."]
        [THROW ZeroDivisionError "Raised when sigma is zero during normalization or inside the exponent term."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
