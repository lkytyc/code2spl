[DEFINE_WORKER: "Add an address to the whitelist only if it is not already present, otherwise reject it by returning False." add_white_list]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the white_list attribute."
        <REF> addr </REF>: any "The address value to check and possibly add."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool_or_list "Returns False when the address is already in the whitelist; otherwise returns the updated whitelist after appending the address."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is already present in self.white_list. RESULT is_duplicate]
            [COMMAND If addr is already present in self.white_list, return False immediately without modifying the whitelist. RESULT false]
            [COMMAND If addr is not present in self.white_list, append addr to self.white_list. RESULT updated_white_list]
            [COMMAND Return the updated self.white_list after appending addr. RESULT updated_white_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove an address from the whitelist when it is present, and return False if the address is not in the whitelist." del_white_list]
    [INPUTS]
        <REF> addr </REF>: any "The address value to check for membership in the whitelist and remove if present."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool or list "Returns False when the address is absent; otherwise returns the updated whitelist after removing the address."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether addr is not present in self.white_list. RESULT membership_check]
            [COMMAND If addr is not present in self.white_list, return False immediately. RESULT False]
            [COMMAND Otherwise, remove addr from self.white_list. RESULT updated_white_list]
            [COMMAND Return the updated self.white_list after removal. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: addr is not in self.white_list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.white_list. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate an input dictionary, reject missing fields or unauthorized addresses, store the allowed address and content, and return the content value." recv]
    [INPUTS]
        <REF> info </REF>: dict "Input dictionary expected to contain the keys 'addr' and 'content'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int or bool or any "Returns -1 if the input is not a valid dictionary with required keys, False if the address is not whitelisted, or the stored content value when accepted."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether info is a dictionary and contains both required keys 'addr' and 'content'; if any of these checks fails, return -1 immediately. RESULT validation_result]
            [COMMAND Read addr from info['addr'] and content from info['content']. RESULT addr, content]
            [COMMAND Check whether addr is not present in self.white_list; if it is not allowed, return False immediately. RESULT whitelist_check]
            [COMMAND When addr is allowed, store a dictionary containing addr and content into self.receive_struct. RESULT self.receive_struct]
            [COMMAND Return self.receive_struct['content'] as the final accepted result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dictionary, or info does not contain 'addr', or info does not contain 'content'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 without reading addr or content. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: addr is not in self.white_list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.receive_struct. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the input is a dictionary containing both required keys and then stores those two fields in self.send_struct without transforming their values." send]
    [INPUTS]
        <REF> info </REF>: dict "Input dictionary expected to contain the keys addr and content."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Returns an error message when validation fails; otherwise no explicit return value is produced."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether info is a dictionary and whether it contains both the addr and content keys. RESULT _validation_ok]
            [COMMAND If validation fails, return the string "info structure is not correct" immediately. RESULT _error_message]
            [COMMAND If validation succeeds, assign self.send_struct to a new dictionary with addr copied from info["addr"] and content copied from info["content"]. RESULT _send_struct]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: info is not a dictionary, or info is missing the addr key, or info is missing the content key.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the validation error message without performing the assignment to self.send_struct. RESULT _error_message]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the matching structure for the given type string, using send for 'send', receive for 'receive', and False for any other value." show]
    [INPUTS]
        <REF> self </REF>: object "Instance providing send_struct and receive_struct attributes."
        <REF> type </REF>: string "Selector value used to choose which structure to return."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object|bool "send_struct when type is 'send', receive_struct when type is 'receive', otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether type is exactly "send". RESULT send_branch]
            [COMMAND If type is "send", return self.send_struct immediately. RESULT self.send_struct]
            [COMMAND Otherwise, check whether type is exactly "receive". RESULT receive_branch]
            [COMMAND If type is "receive", return self.receive_struct immediately. RESULT self.receive_struct]
            [COMMAND For any other value of type, return False as the fallback result. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
