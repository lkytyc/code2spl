# CLASS BankAccount

[DEFINE_WORKER: "Initialize a new instance by setting the balance attribute to the provided balance value, defaulting to 0." __init__]
    [INPUTS]
        <REF> balance </REF>: number "Initial balance value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming balance value directly to the instance attribute self.balance. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validate that the deposit amount is non-negative, add it to the account balance, and return the updated balance." deposit]
    [INPUTS]
        <REF> self </REF>: object "Instance whose balance attribute is updated."
        <REF> amount </REF>: number "Deposit amount; must be greater than or equal to zero."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The updated account balance after applying the deposit."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether amount is less than 0. RESULT validation_check]
            [COMMAND If amount is non-negative, add amount to self.balance. RESULT updated_balance]
            [COMMAND Return self.balance after the update. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount is less than 0]
        [LOG "Reject the request because the deposit amount is invalid."]
        [THROW ValueError "Raised when amount is negative."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the requested withdrawal amount is non-negative and does not exceed the current balance, then subtracts it from the balance and returns the updated balance." withdraw]
    [INPUTS]
        <REF> self </REF>: object "The account object containing the current balance."
        <REF> amount </REF>: number "The withdrawal amount to apply."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The updated account balance after the withdrawal."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the requested amount is less than 0. RESULT negative_amount_check]
            [COMMAND If the amount is less than 0, raise a ValueError with the message "Invalid amount". RESULT invalid_amount_exception]
            [COMMAND Check whether the requested amount is greater than the current self.balance. RESULT sufficient_balance_check]
            [COMMAND If the amount is greater than self.balance, raise a ValueError with the message "Insufficient balance." RESULT insufficient_balance_exception]
            [COMMAND Subtract amount from self.balance in place. RESULT updated_balance]
            [COMMAND Return the updated self.balance. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount < 0]
        [LOG "Invalid withdrawal amount detected; the method raises a ValueError before any balance update occurs."]
        [THROW ValueError "Raised when the amount is negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: amount > self.balance]
        [LOG "Requested amount exceeds the available balance; the method raises a ValueError before any subtraction occurs."]
        [THROW ValueError "Raised when the requested withdrawal is larger than the current balance."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current value stored in self.balance without modification." view_balance]
    [INPUTS]
        <REF> self </REF>: object "The instance whose balance attribute is being accessed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The current value stored in self.balance."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.balance from the instance. RESULT balance_value]
            [COMMAND Return balance_value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Transfers the specified amount from the current account to another account by withdrawing it first and then depositing the same amount into the recipient account." transfer]
    [INPUTS]
        <REF> self </REF>: account "The source account object on which the withdrawal is performed."
        <REF> other_account </REF>: account "The destination account object that receives the deposited amount."
        <REF> amount </REF>: numeric "The amount of value to transfer from the source account to the destination account."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Withdraw the specified amount from the current account. RESULT withdraw_result]
            [COMMAND Deposit the same amount into the other account. RESULT deposit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
