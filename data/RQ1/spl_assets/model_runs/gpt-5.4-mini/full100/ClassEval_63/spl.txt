# CLASS NLPDataProcessor2

[DEFINE_WORKER: "Clean each input string by removing non-letter characters, converting it to lowercase, splitting it into words, and returning a list of word lists." process_data]
    [INPUTS]
        <REF> string_list </REF>: list of strings "Input strings to be normalized and split into words"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of list of strings "A list containing the words extracted from each input string"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect per-string word lists. RESULT words_list]
            [COMMAND For each string in string_list, convert the string to lowercase, remove every character except English letters and whitespace using the pattern [^a-zA-Z\s], split the cleaned string on whitespace into words, and append the resulting word list to words_list. RESULT words_list]
            [COMMAND Return the collected list of word lists. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Count the occurrences of all items across the input lists, then return the five most frequent entries as an ordered dictionary." calculate_word_frequency]
    [INPUTS]
        <REF> words_list </REF>: iterable "A collection of word collections to be counted and aggregated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "A dictionary containing up to the five highest-frequency words and their counts, ordered from highest to lower frequency."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty Counter named word_frequency to accumulate counts. RESULT word_frequency]
            [COMMAND Iterate over each words collection in words_list and update word_frequency with all items from that collection. RESULT word_frequency]
            [COMMAND Sort the items of word_frequency by count in descending order and convert the sorted pairs back into a dictionary named sorted_word_frequency. RESULT sorted_word_frequency]
            [COMMAND Take the first 5 items from sorted_word_frequency and convert them back into a dictionary named top_5_word_frequency. RESULT top_5_word_frequency]
            [COMMAND Return top_5_word_frequency as the final result. RESULT top_5_word_frequency]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If word_frequency contains fewer than 5 entries, the top-five selection includes all available entries.]
        [SEQUENTIAL_BLOCK]
            [COMMAND After sorting word_frequency by descending count, take all existing items if there are fewer than 5 and convert them into top_5_word_frequency. RESULT top_5_word_frequency]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Processes the input string list into words and then computes and returns a word frequency dictionary from those words." process]
    [INPUTS]
        <REF> string_list </REF>: list "List of input strings to be processed into words."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "Dictionary mapping words to their frequencies."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.process_data with string_list and assign the returned value to words_list. RESULT words_list]
            [COMMAND Call self.calculate_word_frequency with words_list and assign the returned value to word_frequency_dict. RESULT word_frequency_dict]
            [COMMAND Return word_frequency_dict. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
