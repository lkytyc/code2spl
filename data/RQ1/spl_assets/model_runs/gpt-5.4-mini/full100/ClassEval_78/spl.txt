# CLASS SplitSentence

[DEFINE_WORKER: "Split an input string into sentences using a regular expression that avoids common abbreviation patterns and splits only after periods or question marks followed by whitespace." split_sentences]
    [INPUTS]
        <REF> self </REF>: object "The instance owning the method."
        <REF> sentences_string </REF>: string "The text to be split into a list of sentence fragments."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of strings "The list of sentence fragments produced by the regular-expression split."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Apply a regular-expression split to sentences_string using the pattern that avoids splitting after dotted abbreviations and short capitalized abbreviations, and splits only when a period or question mark is followed by whitespace. RESULT sentences]
            [COMMAND Return the split sentence list. RESULT sentences]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove all non-letter and non-whitespace characters from the input sentence, split the cleaned text into words, and return the number of resulting tokens." count_words]
    [INPUTS]
        <REF> self </REF>: object "Instance reference used to access the method."
        <REF> sentence </REF>: string "Input text to sanitize and count words from."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The number of whitespace-delimited words remaining after filtering."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove every character from sentence except uppercase letters, lowercase letters, and whitespace by applying the regular expression substitution with pattern [^a-zA-Z\s]. RESULT sentence]
            [COMMAND Split the sanitized sentence on default whitespace to produce the list of words. RESULT words]
            [COMMAND Return the number of words in the list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If the input sentence contains only non-letter and non-whitespace characters, the sanitization step produces an empty string and splitting it yields an empty list.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Treat the cleaned sentence as empty text after substitution. RESULT sentence]
            [COMMAND Split the empty text on whitespace, producing no words. RESULT words]
            [COMMAND Return zero as the word count. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Split the input text into sentences, count the words in each sentence, and return the maximum word count found." process_text_file]
    [INPUTS]
        <REF> sentences_string </REF>: string "Input text to be split into sentences and analyzed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The largest word count among all extracted sentences, or 0 if no sentence produces a larger count."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input text into sentences using self.split_sentences(sentences_string). RESULT sentences]
            [COMMAND Initialize max_count to 0 as the starting boundary value for the running maximum. RESULT max_count]
            [COMMAND For each sentence in sentences, count the words in the current sentence using self.count_words(sentence). RESULT count]
            [COMMAND Within the outer for loop over sentences, if count is strictly greater than max_count, update max_count to count. RESULT max_count]
            [COMMAND After all sentences have been processed, return max_count. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
