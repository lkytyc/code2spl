# CLASS Chat

[DEFINE_WORKER: "Initializes the instance by creating an empty users dictionary as the default user store." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty dictionary to the instance attribute users to initialize the user collection. RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a username to the user registry only if it is not already present, returning False for duplicates and True after creating an empty entry for a new user." add_user]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users collection."
        <REF> username </REF>: str "Username to check and add."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False if the username already exists; otherwise True after insertion."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is already present in self.users. RESULT exists]
            [COMMAND If username is already present, return False immediately without modifying self.users. RESULT False]
            [COMMAND Otherwise, create a new entry in self.users for username with an empty list as its value. RESULT self.users[username]]
            [COMMAND Return True after successfully adding the new username. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a username from the users mapping when present and returns whether the deletion succeeded." remove_user]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users collection."
        <REF> username </REF>: key "Username to check for and delete from the users collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username was found and deleted; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists as a key in self.users. RESULT membership_check]
            [COMMAND If username is present in self.users, delete self.users[username]. RESULT deletion_done]
            [COMMAND After deleting the entry, return True immediately. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not present in self.users.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.users. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Stores a message with a current timestamp for both sender and receiver after verifying that both users exist." send_message]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users collection."
        <REF> sender </REF>: any "Identifier for the user sending the message."
        <REF> receiver </REF>: any "Identifier for the user receiving the message."
        <REF> message </REF>: any "Message content to store."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the message is stored for both users; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether sender and receiver both exist in self.users. RESULT user_existence_check]
            [COMMAND If either sender or receiver is missing from self.users, return False immediately. RESULT False]
            [COMMAND Generate the current local timestamp as a string formatted as YYYY-MM-DD HH:MM:SS. RESULT timestamp]
            [COMMAND Create message_info as a dictionary containing sender, receiver, message, and the generated timestamp. RESULT message_info]
            [COMMAND Append message_info to self.users[sender]. RESULT sender_message_list]
            [COMMAND Append the same message_info to self.users[receiver]. RESULT receiver_message_list]
            [COMMAND Return True to indicate successful storage. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns an empty list for unknown usernames, otherwise returns the stored messages for the given username from self.users." get_messages]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the users mapping."
        <REF> username </REF>: string "The username used to look up messages in self.users."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "An empty list if the username is not found; otherwise the value stored in self.users for that username."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is present in self.users. RESULT membership_check]
            [COMMAND If username is not present in self.users, return an empty list immediately. RESULT empty_list_return]
            [COMMAND Otherwise, retrieve and return self.users[username] directly with no further transformation. RESULT messages]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
