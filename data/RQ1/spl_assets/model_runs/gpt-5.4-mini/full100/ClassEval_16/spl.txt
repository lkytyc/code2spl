# CLASS Calculator

[DEFINE_WORKER: "Initializes an operator lookup table that maps arithmetic symbols to lambda functions implementing addition, subtraction, multiplication, division, and exponentiation." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new dictionary literal to self.operators. RESULT self.operators]
            [COMMAND Populate the dictionary with the '+' key mapped to a lambda that returns x plus y. RESULT self.operators['+']]
            [COMMAND Populate the dictionary with the '-' key mapped to a lambda that returns x minus y. RESULT self.operators['-']]
            [COMMAND Populate the dictionary with the '*' key mapped to a lambda that returns x multiplied by y. RESULT self.operators['*']]
            [COMMAND Populate the dictionary with the '/' key mapped to a lambda that returns x divided by y. RESULT self.operators['/']]
            [COMMAND Populate the dictionary with the '^' key mapped to a lambda that returns x raised to the power of y. RESULT self.operators['^']]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: When the '/' operator lambda is later invoked with a zero divisor, division may fail according to the runtime arithmetic rules.]
        [LOG "Division by zero can occur when the division operator function is called with an invalid denominator."]
        [THROW ArithmeticError "Raised by the underlying division operation when the divisor is not valid."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Evaluate an arithmetic expression character by character using operand and operator stacks, handling numeric buffering, operator precedence, and parentheses, then return the final computed value or None for an empty expression." calculate]
    [INPUTS]
        <REF> self </REF>: object "Instance providing precedence and operator application helpers."
        <REF> expression </REF>: string "Arithmetic expression to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float or null "The final evaluated value, or None if no operands were produced."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty operand stack, an empty operator stack, and an empty numeric buffer. RESULT operand_stack, operator_stack, num_buffer]
            [COMMAND Iterate through each character in the expression from left to right. RESULT char]
            [COMMAND Within the outer loop, if the current character is a digit or a decimal point, append it to the numeric buffer. RESULT num_buffer]
            [COMMAND Within the outer loop, when the current character is not numeric, flush any buffered number by converting it to float, pushing it onto the operand stack, and clearing the numeric buffer. RESULT operand_stack, num_buffer]
            [COMMAND Within the non-numeric branch, if the current character is one of +, -, *, /, or ^, repeatedly apply stacked operators while the operator stack is non-empty, the top is not an opening parenthesis, and the top operator has precedence greater than or equal to the current operator. RESULT operand_stack, operator_stack]
            [COMMAND Within the operator branch after all higher-or-equal precedence operators are applied, push the current operator onto the operator stack. RESULT operator_stack]
            [COMMAND Within the non-numeric branch, if the current character is an opening parenthesis, push it onto the operator stack as a subexpression boundary marker. RESULT operator_stack]
            [COMMAND Within the non-numeric branch, if the current character is a closing parenthesis, repeatedly apply stacked operators until an opening parenthesis is encountered, then remove that opening parenthesis from the operator stack. RESULT operand_stack, operator_stack]
            [COMMAND After the loop, if the numeric buffer still contains digits or a decimal point, convert it to float and push it onto the operand stack. RESULT operand_stack]
            [COMMAND After all characters have been processed, repeatedly apply any remaining operators until the operator stack becomes empty. RESULT operand_stack, operator_stack]
            [COMMAND Return the last value on the operand stack if at least one operand exists; otherwise return None. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the precedence value for a given operator using a fixed lookup table, defaulting to 0 when the operator is not recognized." precedence]
    [INPUTS]
        <REF> operator </REF>: any "The operator key used to look up its precedence value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The precedence of the operator, or 0 if the operator is not found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a dictionary named precedences with the fixed operator precedence mapping: '+' to 1, '-' to 1, '*' to 2, '/' to 2, and '^' to 3. RESULT precedences]
            [COMMAND Return the value from precedences for the given operator, using 0 as the default when the operator key is absent. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Pops one operator from the operator stack, evaluates the top two operands with that operator, pushes the result back onto the operand stack, and returns both stacks." apply_operator]
    [INPUTS]
        <REF> operand_stack </REF>: stack "Stack of operands used for binary evaluation."
        <REF> operator_stack </REF>: stack "Stack of operators whose top element determines the operation to apply."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the updated operand_stack and operator_stack."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop the top operator from operator_stack and store it in operator. RESULT operator]
            [COMMAND Check whether operator equals '^'. RESULT branch_decision]
            [COMMAND Within the operator == '^' branch, pop the top operand from operand_stack into operand2. RESULT operand2]
            [COMMAND Within the operator == '^' branch, pop the next operand from operand_stack into operand1. RESULT operand1]
            [COMMAND Within the operator == '^' branch, apply self.operators[operator] to operand1 and operand2 and store the computed value in result. RESULT result]
            [COMMAND Within the operator == '^' branch, append result to operand_stack. RESULT operand_stack]
            [COMMAND Within the else branch for all operators other than '^', pop the top operand from operand_stack into operand2. RESULT operand2]
            [COMMAND Within the else branch for all operators other than '^', pop the next operand from operand_stack into operand1. RESULT operand1]
            [COMMAND Within the else branch for all operators other than '^', apply self.operators[operator] to operand1 and operand2 and store the computed value in result. RESULT result]
            [COMMAND Within the else branch for all operators other than '^', append result to operand_stack. RESULT operand_stack]
            [COMMAND Return the updated operand_stack and operator_stack. RESULT (operand_stack, operator_stack)]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If operator is '^', follow the dedicated branch before performing the same two operand pops, operator application, and push-back sequence.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop the top operand from operand_stack into operand2. RESULT operand2]
            [COMMAND Pop the next operand from operand_stack into operand1. RESULT operand1]
            [COMMAND Apply self.operators[operator] to operand1 and operand2 and store the computed value in result. RESULT result]
            [COMMAND Append result to operand_stack. RESULT operand_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If operator is any value other than '^', follow the else branch and perform the same two operand pops, operator application, and push-back sequence.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop the top operand from operand_stack into operand2. RESULT operand2]
            [COMMAND Pop the next operand from operand_stack into operand1. RESULT operand1]
            [COMMAND Apply self.operators[operator] to operand1 and operand2 and store the computed value in result. RESULT result]
            [COMMAND Append result to operand_stack. RESULT operand_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
