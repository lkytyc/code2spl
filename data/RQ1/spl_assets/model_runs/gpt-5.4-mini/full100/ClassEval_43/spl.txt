[DEFINE_WORKER: "Adds a new employee record only if the employee ID is not already present, returning false for duplicates and true after inserting the new details." add_employee]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the employee registry."
        <REF> employee_id </REF>: any "The employee identifier used as the dictionary key."
        <REF> name </REF>: any "The employee's name."
        <REF> position </REF>: any "The employee's job position."
        <REF> department </REF>: any "The employee's department."
        <REF> salary </REF>: any "The employee's salary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns false if the employee ID already exists, otherwise returns true after adding the employee record."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id is already present in self.employees. RESULT membership_check]
            [COMMAND If employee_id is already present, return false immediately without modifying self.employees. RESULT false]
            [COMMAND Otherwise, create a new employee record in self.employees keyed by employee_id with name, position, department, and salary. RESULT self.employees[employee_id]]
            [COMMAND Return true after successfully adding the new employee record. RESULT true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Remove an employee by ID from the employee mapping when present and report whether the deletion occurred." remove_employee]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the employees collection."
        <REF> employee_id </REF>: key "Employee identifier to look up and remove."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the employee entry was found and deleted, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id exists in self.employees. RESULT membership_check]
            [COMMAND If employee_id is present in self.employees, delete the entry for that employee_id from self.employees. RESULT employee_removed]
            [COMMAND If the entry was deleted, return True immediately. RESULT true]
            [COMMAND If employee_id is not present in self.employees, return False immediately without modifying the collection. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that an employee record exists and contains all keys from the update data, then copies the provided values into the record and returns whether the update succeeded." update_employee]
    [INPUTS]
        <REF> self </REF>: object "Instance providing access to the employee lookup method."
        <REF> employee_id </REF>: int "Employee identifier used to retrieve the employee record."
        <REF> employee_info </REF>: dict "Mapping of employee fields and values to validate and apply."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns False if the employee is missing or any provided key is not already present in the employee record; otherwise returns True after updating the record."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_employee with employee_id and store the returned employee record in employee. RESULT employee]
            [COMMAND If employee equals False, return False immediately without making any updates. RESULT False]
            [COMMAND Otherwise, for each key-value pair in employee_info, verify that the key already exists in employee. RESULT validation_complete]
            [COMMAND If any key from employee_info is not present in employee, return False immediately before applying any changes. RESULT False]
            [COMMAND After all keys are validated, iterate again over employee_info and assign each value into employee under the matching key. RESULT employee]
            [COMMAND Return True after all updates complete successfully. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The lookup returns False for the requested employee_id.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Stop processing and return False immediately. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Any key in employee_info is missing from the retrieved employee record.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Stop processing during the validation loop and return False before mutating employee. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the employee record for a given employee ID when it exists in the employee collection, otherwise return False." get_employee]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the employees collection."
        <REF> employee_id </REF>: any "The employee identifier to look up in the employees collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: employee record or False "The matching employee value if the ID is present; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id exists in self.employees using membership testing. RESULT membership_check]
            [COMMAND If employee_id is present in self.employees, return self.employees[employee_id] immediately. RESULT employee_record]
            [COMMAND If employee_id is not present in self.employees, return False immediately. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build and return a dictionary of employee records keyed by employee ID, copying each employee's fields into a new normalized detail dictionary when employee data exists." list_employees]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize employee_data as an empty dictionary to hold the final employee records. RESULT employee_data]
            [COMMAND If self.employees is truthy, iterate over each employee_id and employee_info pair in self.employees.items(). RESULT employee_id, employee_info]
            [COMMAND Within the outer loop, create a new empty dictionary for the current employee's details. RESULT employee_details]
            [COMMAND Within the outer loop, store the current employee_id under the key employee_ID in employee_details. RESULT employee_details]
            [COMMAND Within the outer loop, iterate over each key and value pair in employee_info.items() and copy each pair into employee_details. RESULT key, value]
            [COMMAND Within the outer loop, after copying all fields, store employee_details in employee_data under the current employee_id. RESULT employee_data]
            [COMMAND Return employee_data after processing all employees, or return the empty dictionary when self.employees is falsy and no loop body executes. RESULT employee_data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
