# CLASS DataStatistics

[DEFINE_WORKER: "Computes the arithmetic mean of the input data and rounds the result to two decimal places." mean]
    [INPUTS]
        <REF> data </REF>: sequence of numeric values "Collection of numbers to average."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The rounded average of the input values to two decimal places."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sum of all numeric elements in data. RESULT total]
            [COMMAND Compute the number of elements in data. RESULT count]
            [COMMAND Divide total by count to obtain the arithmetic mean. RESULT average]
            [COMMAND Round the average to 2 decimal places and return the rounded value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is empty, so count is 0 and division by zero occurs.]
        [LOG "Division by zero while computing the average from an empty data sequence."]
        [THROW ZeroDivisionError "Raised when attempting to divide the sum of data by zero elements."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sort the input data, then return the median as the middle element for odd-length data or the rounded average of the two middle elements for even-length data." median]
    [INPUTS]
        <REF> data </REF>: sequence of numbers "Input values to be sorted and evaluated for the median."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The median value of the input data, rounded to 2 decimal places only when the length is even."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new sorted sequence from the input data without modifying the original input. RESULT sorted_data]
            [COMMAND Compute the length of the sorted sequence. RESULT n]
            [COMMAND Check whether n is even by testing n % 2 == 0. RESULT parity_check]
            [COMMAND If n is even, set middle to n // 2, then return round((sorted_data[middle - 1] + sorted_data[middle]) / 2, 2) as the median. RESULT result]
            [COMMAND If n is odd, set middle to n // 2, then return sorted_data[middle] as the median. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return all values that occur with the highest frequency in the given data." mode]
    [INPUTS]
        <REF> data </REF>: iterable "Input collection whose element frequencies are counted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of all elements tied for the maximum frequency."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a frequency counter from the input data. RESULT counter]
            [COMMAND Compute the highest occurrence count from the counter values. RESULT mode_count]
            [COMMAND Collect every key in the counter whose count equals the highest occurrence count. RESULT mode]
            [COMMAND Return the list of most frequent elements. RESULT mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
