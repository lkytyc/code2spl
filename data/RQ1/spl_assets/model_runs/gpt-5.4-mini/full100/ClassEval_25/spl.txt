# CLASS CookiesUtil

[DEFINE_WORKER: "Initializes the instance by storing the provided cookies file path and setting the cookies cache to an unset default." __init__]
    [INPUTS]
        <REF> cookies_file </REF>: any "The source value assigned directly to the instance cookies_file field."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming cookies_file value directly to the instance field self.cookies_file. RESULT self.cookies_file]
            [COMMAND Initialize the instance field self.cookies to None as an unset default value. RESULT self.cookies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "This method copies the 'cookies' value from the given response mapping into the instance's cookies attribute and then saves the cookies." get_cookies]
    [INPUTS]
        <REF> reponse </REF>: mapping "A dictionary-like response object expected to contain a 'cookies' key."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the value associated with the 'cookies' key from reponse and assign it to self.cookies. RESULT self.cookies]
            [COMMAND Call self._save_cookies() to persist the updated cookies state. RESULT self._save_cookies()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: reponse does not support key access or does not contain the 'cookies' key when self.cookies is assigned.]
        [LOG "A key access failure may occur while reading reponse['cookies']; the method does not handle this error."]
        [THROW KeyError or TypeError "Raised by the response object when the 'cookies' entry is missing or the object is not subscriptable."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Loads and returns JSON cookie data from the configured cookies file, falling back to an empty dictionary if the file does not exist." load_cookies]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the cookies_file path."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "The parsed cookies data from the file, or an empty dictionary when the file is missing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Enter a try block and open the configured cookies file at self.cookies_file in read mode. RESULT file]
            [COMMAND Parse the file contents as JSON into cookies_data. RESULT cookies_data]
            [COMMAND Return cookies_data immediately after successful parsing. RESULT cookies_data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If opening or reading the cookies file raises FileNotFoundError.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch FileNotFoundError and return an empty dictionary instead of failing. RESULT {}]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Saves the current cookies dictionary to the configured cookies file as JSON and returns true on success, or false if any error occurs." _save_cookies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the cookies are saved successfully; false if any exception occurs."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the cookies file path stored in self.cookies_file in write mode. RESULT file]
            [COMMAND Serialize and write self.cookies to the open file as JSON. RESULT json.dump(self.cookies, file)]
            [COMMAND Return true after successfully saving the cookies. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during file opening or JSON writing, catch it and return false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Handle the exception without re-raising it. RESULT exception]
            [COMMAND Return false to indicate the cookies were not saved. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a cookie header string from self.cookies and assign it to request['cookies']." set_cookies]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the cookies mapping used to build the header string."
        <REF> request </REF>: dict "Request object that will receive the cookies header value."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read all key-value pairs from self.cookies. RESULT cookies_items]
            [COMMAND Format each key-value pair as "key=value" using the current key and value. RESULT formatted_cookies]
            [COMMAND Join the formatted cookie strings with "; " as the separator. RESULT cookies_header]
            [COMMAND Assign the joined cookie header string to request['cookies'], overwriting any existing value. RESULT request]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
