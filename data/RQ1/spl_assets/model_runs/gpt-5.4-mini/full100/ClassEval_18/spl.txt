# CLASS CamelCaseMap

[DEFINE_WORKER: "Initialize the instance data store by creating a new empty dictionary for the _data attribute." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new empty dictionary and assign it to the instance attribute _data. RESULT self._data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the camel-cased version of a string key, otherwise return the key unchanged." _convert_key]
    [INPUTS]
        <REF> key </REF>: object "Value to convert; string values are transformed to camel case while non-string values are passed through unchanged."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: object "The camel-cased key when the input is a string, otherwise the original key."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether key is an instance of str. RESULT is_string]
            [COMMAND If key is a string, convert it with self._to_camel_case(key) and return the converted value immediately. RESULT converted_key]
            [COMMAND If key is not a string, return key unchanged. RESULT key]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an underscore-separated key into lower camel case by keeping the first segment unchanged and capitalizing each subsequent segment before concatenation." _to_camel_case]
    [INPUTS]
        <REF> key </REF>: string "Input string to convert from underscore-separated form to lower camel case."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The camel-cased string produced from the input key."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input key on every underscore character to produce a list of parts. RESULT parts]
            [COMMAND Return the first part concatenated with the title-cased version of every remaining part. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
