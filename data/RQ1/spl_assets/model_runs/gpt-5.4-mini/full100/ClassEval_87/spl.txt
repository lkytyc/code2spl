# CLASS TimeUtils

[DEFINE_WORKER: "Initialize the instance timestamp field with the current local date and time." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call datetime.datetime.now() to obtain the current local date and time. RESULT current_datetime]
            [COMMAND Assign the obtained current local date and time to the instance attribute self.datetime. RESULT self.datetime]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the current time from the datetime object using the fixed hour-minute-second pattern and returns the resulting string." get_current_time]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the datetime attribute used for formatting."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted current time string in HH:MM:SS format."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constant time format string "%H:%M:%S" to the local variable format. RESULT format]
            [COMMAND Return self.datetime.strftime(format) to produce the current time string in hours, minutes, and seconds. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the current datetime using the fixed date pattern '%Y-%m-%d' and returns the resulting string." get_current_date]
    [INPUTS]
        <REF> self </REF>: object "Instance providing access to a datetime object through self.datetime."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The current date formatted as a string in year-month-day form."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the fixed format string "%Y-%m-%d" to the local variable format. RESULT format]
            [COMMAND Return self.datetime.strftime(format) to produce the formatted current date string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a given number of seconds to the stored datetime and returns the result formatted as an HH:MM:SS string." add_seconds]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the current datetime value."
        <REF> seconds </REF>: number "Number of seconds to add to the current datetime."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The updated time formatted as a string in HH:MM:SS format."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute a new datetime by adding a timedelta of the provided seconds to self.datetime. RESULT new_datetime]
            [COMMAND Set the output format string to %H:%M:%S. RESULT format]
            [COMMAND Return new_datetime formatted with the specified format string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parse an input string into a datetime.datetime value using the fixed format %Y-%m-%d %H:%M:%S and return the parsed result." string_to_datetime]
    [INPUTS]
        <REF> string </REF>: str "Datetime text to parse in the exact format year-month-day hour:minute:second."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: datetime.datetime "The datetime object produced by parsing the input string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the input string using datetime.datetime.strptime with the exact format %Y-%m-%d %H:%M:%S. RESULT parsed_datetime]
            [COMMAND Return the parsed datetime value immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats the given datetime value as a string in year-month-day hour:minute:second order and returns it." datetime_to_string]
    [INPUTS]
        <REF> self </REF>: object "Instance context for the method."
        <REF> datetime </REF>: datetime-like object "Value expected to support strftime formatting."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted datetime string using the pattern %Y-%m-%d %H:%M:%S."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call strftime on the input datetime value with the fixed format string %Y-%m-%d %H:%M:%S. RESULT formatted_datetime]
            [COMMAND Return the formatted datetime string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert two input time strings to datetimes, compute the elapsed time between them in minutes, and round the result to the nearest integer." get_minutes]
    [INPUTS]
        <REF> string_time1 </REF>: string "The first time value as a string."
        <REF> string_time2 </REF>: string "The second time value as a string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The rounded number of minutes between the two converted datetimes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert string_time1 to a datetime value using self.string_to_datetime and store it in time1. RESULT time1]
            [COMMAND Convert string_time2 to a datetime value using self.string_to_datetime and store it in time2. RESULT time2]
            [COMMAND Subtract time1 from time2, take the seconds component of the resulting time delta, divide it by 60, round the value, and return the rounded minute count. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a datetime from the provided year, month, day, hour, minute, and second values and return it formatted as a standard timestamp string." get_format_time]
    [INPUTS]
        <REF> self </REF>: object "Instance reference used to call the method."
        <REF> year </REF>: int "Year component of the datetime value."
        <REF> month </REF>: int "Month component of the datetime value."
        <REF> day </REF>: int "Day component of the datetime value."
        <REF> hour </REF>: int "Hour component of the datetime value."
        <REF> minute </REF>: int "Minute component of the datetime value."
        <REF> second </REF>: int "Second component of the datetime value."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The constructed datetime rendered as a string in YYYY-MM-DD HH:MM:SS format."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the local format string to "%Y-%m-%d %H:%M:%S". RESULT format]
            [COMMAND Construct a datetime value from year, month, day, hour, minute, and second using the datetime.datetime constructor. RESULT time_item]
            [COMMAND Return time_item formatted with the configured format string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The provided year, month, day, hour, minute, or second values are not valid for datetime.datetime construction.]
        [LOG "datetime.datetime raises an exception because the input date or time components are out of range or otherwise invalid."]
        [THROW ValueError "Raised when the supplied date or time components do not form a valid datetime."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
