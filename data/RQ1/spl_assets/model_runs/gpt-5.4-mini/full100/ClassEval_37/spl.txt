# CLASS EncryptionUtils

[DEFINE_WORKER: "Initializes the instance by storing the provided key on the object." __init__]
    [INPUTS]
        <REF> key </REF>: any "Value to assign to the instance key attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This initializer does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided key value to the instance key attribute. RESULT self.key]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypts the input string with a Caesar cipher by shifting each alphabetic character by the given amount while preserving case and leaving non-letters unchanged." caesar_cipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> plaintext </REF>: string "Input text to transform."
        <REF> shift </REF>: integer "Number of positions to shift alphabetic characters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The Caesar-ciphered ciphertext."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ciphertext as an empty string. RESULT ciphertext]
            [COMMAND Within the outer loop over each character in plaintext, process each char one by one. RESULT char]
            [COMMAND Within the loop, if char is alphabetic, continue to case handling; otherwise append the character unchanged to ciphertext. RESULT ciphertext]
            [COMMAND Within the alphabetic branch, if char is uppercase, set ascii_offset to 65; otherwise set ascii_offset to 97. RESULT ascii_offset]
            [COMMAND Within the alphabetic branch, compute shifted_char as chr((ord(char) - ascii_offset + shift) % 26 + ascii_offset) to shift the letter within the 26-letter alphabet while preserving case. RESULT shifted_char]
            [COMMAND Append shifted_char to ciphertext. RESULT ciphertext]
            [COMMAND After all characters have been processed, return ciphertext. RESULT ciphertext]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypt the input text with a Vigenere-style shift using the instance key, preserving letter case and leaving non-letter characters unchanged." vigenere_cipher]
    [INPUTS]
        <REF> plain_text </REF>: string "The text to encrypt character by character."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The encrypted text with alphabetic characters shifted by the key and all other characters copied as-is."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the output string as empty and set the key position index to zero. RESULT encrypted_text, key_index]
            [COMMAND Iterate through each character in plain_text from left to right. RESULT char]
            [COMMAND Within the loop, if the current character is alphabetic, compute the shift from the current key character using key_index modulo the key length, then convert that key character to lowercase and map it to a zero-based alphabet shift. RESULT shift]
            [COMMAND Within the alphabetic branch, convert the current input character to lowercase, shift it within the 26-letter alphabet using modulo 26 arithmetic, and build the encrypted lowercase character. RESULT encrypted_char]
            [COMMAND Within the alphabetic branch, append the encrypted character to encrypted_text, preserving uppercase when the original character was uppercase and lowercase otherwise. RESULT encrypted_text]
            [COMMAND Within the alphabetic branch, advance key_index by one so the key only consumes letters. RESULT key_index]
            [COMMAND Within the non-alphabetic branch, append the current character to encrypted_text unchanged. RESULT encrypted_text]
            [COMMAND After all characters are processed, return encrypted_text. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypt the input text using a rail fence zigzag pattern and then read the stored characters row by row to produce the ciphertext." rail_fence_cipher]
    [INPUTS]
        <REF> self </REF>: object "Instance reference for the method."
        <REF> plain_text </REF>: string "The text to be placed into the rail fence grid and encrypted."
        <REF> rails </REF>: integer "The number of rails used to build the zigzag pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The encrypted text formed by reading non-placeholder characters row by row."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a 2D grid named fence with rails rows and len(plain_text) columns, initializing every cell to the newline placeholder character. RESULT fence]
            [COMMAND Initialize direction to -1, then set row and col to 0 to start at the top-left position. RESULT direction,row,col]
            [COMMAND For each character in plain_text, if the current row is at the top rail or the bottom rail, reverse direction by multiplying direction by -1. RESULT direction]
            [COMMAND For each character in plain_text, store the character in fence at the current row and col position. RESULT fence]
            [COMMAND For each character in plain_text, advance col by 1 and move row by the current direction value. RESULT row,col]
            [COMMAND Initialize encrypted_text as the empty string. RESULT encrypted_text]
            [COMMAND For each row index i from 0 to rails - 1, and for each column index j from 0 to len(plain_text) - 1, append fence[i][j] to encrypted_text when the cell is not the newline placeholder. RESULT encrypted_text]
            [COMMAND Return encrypted_text. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
