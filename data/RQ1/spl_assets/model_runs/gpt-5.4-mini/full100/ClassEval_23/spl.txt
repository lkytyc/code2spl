# CLASS CombinationCalculator

[DEFINE_WORKER: "Initializes the instance by storing the provided datas list on the object." __init__]
    [INPUTS]
        <REF> datas </REF>: List[str] "A list of strings to be assigned to the instance field."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided datas list directly to the instance attribute self.datas. RESULT self.datas]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns 1 for the trivial binomial cases where m is 0 or equals n, otherwise computes the binomial coefficient n choose m using factorials." count]
    [INPUTS]
        <REF> n </REF>: int "The total number of items."
        <REF> m </REF>: int "The number of items to choose."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The combinatorial count of ways to choose m items from n items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether m is 0 or n equals m. RESULT condition]
            [COMMAND If the condition is true, return 1 immediately and stop execution. RESULT result]
            [COMMAND Compute factorial(n) and compute factorial(n - m) and factorial(m), then multiply the last two factorial values. RESULT denominator]
            [COMMAND Divide factorial(n) by the multiplied denominator using integer floor division and return the quotient. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m is 0 or n equals m]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 1 as the early-exit binomial result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: n, m, or n - m is not a non-negative integer suitable for factorial computation]
        [LOG "math.factorial raises an error when given invalid input values."]
        [THROW ValueError "Raised by factorial for negative or otherwise invalid inputs."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns False for values outside 0 to 63 inclusive, otherwise computes a bitmask of the lowest n bits set, using infinity as the special result for n equal to 63." count_all]
    [INPUTS]
        <REF> n </REF>: int "Input integer used to determine the range check and bitmask result."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns False when n is out of range; otherwise returns the computed bitmask or positive infinity for n equal to 63."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether n is less than 0 or greater than 63. RESULT range_check]
            [COMMAND If n is outside the inclusive range 0 to 63, return False immediately. RESULT False]
            [COMMAND If n is within the inclusive range 0 to 63, compute the result using the conditional expression based on n. RESULT value]
            [COMMAND When n is not equal to 63, compute (1 << n) - 1 to produce a bitmask with the lowest n bits set. RESULT value]
            [COMMAND When n is equal to 63, return float("inf") instead of the shift-based bitmask. RESULT value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Initializes an empty result list, delegates generation to the internal selection routine with a None-filled working array, and returns the collected combinations." select]
    [INPUTS]
        <REF> m </REF>: int "The length of each generated item in the output structure."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "A list of generated string lists collected by the internal selection process."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Create a working array of length m filled with None values and call the internal selection routine <SPL>_select</SPL> starting at index 0, passing the working array, an initial depth of 0, and the result list to collect generated items. RESULT result]
            [COMMAND Return the result list after the internal selection routine completes. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect and return the concatenation of all selections for every index from 1 through the number of available data items." select_all]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the data source and the select method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "A flattened list containing all lists returned by selecting each valid index."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND For each i from 1 up to and including len(self.datas), call self.select(i) and extend result with the returned items. RESULT result]
            [COMMAND Return result after all indices have been processed. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Select combinations recursively by filling the current result list slot, recording a completed copy when the target length is reached, and pruning the search using the remaining element count." _select]
    [INPUTS]
        <REF> dataIndex </REF>: int "Starting index in self.datas for the current recursive level."
        <REF> resultList </REF>: List[str] "Mutable partial selection being filled in place."
        <REF> resultIndex </REF>: int "Current position to write in resultList."
        <REF> result </REF>: List[List[str]] "Accumulator for completed selections."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "Updated list of completed selections."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute resultLen as the length of resultList and compute resultCount as resultIndex plus 1. RESULT resultLen,resultCount]
            [COMMAND If resultCount is greater than resultLen, append a copy of resultList to result and return immediately. RESULT result]
            [COMMAND Within the for loop over i from dataIndex to len(self.datas) plus resultCount minus resultLen, assign self.datas[i] into resultList at resultIndex and recursively call <SPL>_select</SPL> with i plus 1, resultList, resultIndex plus 1, and result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
