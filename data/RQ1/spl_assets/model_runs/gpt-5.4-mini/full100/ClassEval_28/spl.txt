# CLASS DatabaseProcessor

[DEFINE_WORKER: "Initializes the instance by storing the provided database name on the object." __init__]
    [INPUTS]
        <REF> database_name </REF>: any "The database name value to assign to the instance attribute."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided database_name value directly to the instance attribute self.database_name. RESULT self.database_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a SQLite table if it does not already exist by opening a database connection, building a CREATE TABLE statement from the provided table and column names, executing it, committing the transaction, and closing the connection." create_table]
    [INPUTS]
        <REF> table_name </REF>: string "Name of the table to create if it does not already exist."
        <REF> key1 </REF>: string "Name of the first data column, stored as TEXT."
        <REF> key2 </REF>: string "Name of the second data column, stored as INTEGER."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using self.database_name and store the connection in conn. RESULT conn]
            [COMMAND Create a cursor from the open connection and store it in cursor. RESULT cursor]
            [COMMAND Build a CREATE TABLE IF NOT EXISTS SQL statement using table_name, key1, and key2, with id as an INTEGER PRIMARY KEY, key1 as TEXT, and key2 as INTEGER, and store it in create_table_query. RESULT create_table_query]
            [COMMAND Execute the generated SQL statement through the cursor. RESULT cursor.execute(create_table_query)]
            [COMMAND Commit the transaction on the database connection. RESULT conn.commit()]
            [COMMAND Close the database connection. RESULT conn.close()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Insert each record from the input data into the specified database table using the object's database connection." insert_into_database]
    [INPUTS]
        <REF> self </REF>: object "Object containing the database_name attribute used to open the SQLite connection."
        <REF> table_name </REF>: string "Name of the target database table to receive inserted rows."
        <REF> data </REF>: iterable of dict "Collection of records, where each item provides name and age fields for insertion."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open an SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Within the for loop over each item in data, build an INSERT statement for table_name with placeholders for name and age. RESULT insert_query]
            [COMMAND Within the for loop over each item in data, execute insert_query with item['name'] and item['age'] as the parameter values. RESULT cursor.execute(insert_query, (item['name'], item['age']))]
            [COMMAND Commit the transaction on conn after all items have been processed. RESULT conn.commit()]
            [COMMAND Close the database connection conn. RESULT conn.close()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search a SQLite database table for rows matching a given name and return all matches, or None when no rows are found." search_database]
    [INPUTS]
        <REF> table_name </REF>: str "Name of the database table to query."
        <REF> name </REF>: str "Value to match against the name column."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of matching rows when at least one row is found; otherwise None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection using self.database_name and store it in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Build a parameterized SQL query selecting all columns from the given table_name where name equals a bound parameter, and store it in select_query. RESULT select_query]
            [COMMAND Execute select_query on cursor with name bound as the single parameter. RESULT query_execution]
            [COMMAND Fetch all rows from the cursor after the query execution and store them in result. RESULT result]
            [COMMAND If result is truthy, return result immediately. RESULT result]
            [COMMAND Otherwise, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Connect to the configured SQLite database, build a DELETE statement for the requested table and name, execute it with a parameterized value, then commit and close the connection." delete_from_database]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the database_name attribute used to open the SQLite connection."
        <REF> table_name </REF>: string "Target table name interpolated directly into the DELETE SQL statement."
        <REF> name </REF>: string "Value used as the parameter for the name column filter."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Connect to the SQLite database using self.database_name and store the connection in conn. RESULT conn]
            [COMMAND Create a cursor from conn and store it in cursor. RESULT cursor]
            [COMMAND Build a DELETE SQL string that targets table_name and filters rows where name equals a parameter placeholder. RESULT delete_query]
            [COMMAND Execute delete_query on cursor with name passed as a single-element parameter tuple. RESULT delete_execution]
            [COMMAND Commit the transaction on conn to persist the deletion. RESULT commit_result]
            [COMMAND Close conn to release the database connection. RESULT close_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
