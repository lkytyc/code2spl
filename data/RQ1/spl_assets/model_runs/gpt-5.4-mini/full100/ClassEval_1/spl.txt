# CLASS AreaCalculator

[DEFINE_WORKER: "Initializes the object by storing the provided radius value on the instance." __init__]
    [INPUTS]
        <REF> radius </REF>: unknown "The radius value to assign to the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the provided radius value to the instance attribute self.radius. RESULT self.radius]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the area of a circle by multiplying pi by the square of self.radius." calculate_circle_area]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the radius attribute used in the calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: numeric "The computed circle area based on the current radius value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.radius from the object instance. RESULT radius]
            [COMMAND Compute the square of radius using exponent 2. RESULT radius_squared]
            [COMMAND Multiply math.pi by radius_squared to obtain the circle area. RESULT area]
            [COMMAND Return area as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return the surface area using the formula 4πr² based on the instance radius." calculate_sphere_area]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the radius attribute."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed surface area value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.radius from the instance. RESULT radius]
            [COMMAND Square the radius value. RESULT radius_squared]
            [COMMAND Multiply the squared radius by math.pi. RESULT pi_radius_squared]
            [COMMAND Multiply the result by 4. RESULT area]
            [COMMAND Return the computed area. RESULT area]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the surface area of a cylinder using the instance radius and the provided height." calculate_cylinder_area]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the radius value used in the calculation."
        <REF> height </REF>: number "Cylinder height added to the radius in the area formula."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed cylinder surface area value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance radius from self. RESULT radius]
            [COMMAND Use math.pi as the constant pi value. RESULT pi]
            [COMMAND Add the radius and height to compute the first parenthesized term. RESULT radius_plus_height]
            [COMMAND Multiply 2, pi, radius, and the radius-plus-height term to compute the cylinder area. RESULT result]
            [COMMAND Return the computed cylinder area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute and return the sector area using the instance radius and the given angle." calculate_sector_area]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the radius value."
        <REF> angle </REF>: number "Angle used in the sector area calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The calculated sector area."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the radius value from the current instance as self.radius. RESULT radius]
            [COMMAND Square the radius value. RESULT radius_squared]
            [COMMAND Multiply the squared radius by angle. RESULT scaled_area]
            [COMMAND Divide the product by 2 to compute the sector area. RESULT result]
            [COMMAND Return the computed sector area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculate the area of an annulus by subtracting the inner circle area from the outer circle area and multiplying the difference by pi." calculate_annulus_area]
    [INPUTS]
        <REF> self </REF>: object "Instance reference, unused by this method."
        <REF> inner_radius </REF>: number "Radius of the inner circle."
        <REF> outer_radius </REF>: number "Radius of the outer circle."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The computed annulus area."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the inner_radius and outer_radius input values. RESULT radii]
            [COMMAND Compute outer_radius squared and inner_radius squared. RESULT squared_terms]
            [COMMAND Subtract the inner squared term from the outer squared term to obtain the ring area difference. RESULT area_difference]
            [COMMAND Multiply the area difference by math.pi to produce the final annulus area. RESULT result]
            [COMMAND Return the computed annulus area. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
