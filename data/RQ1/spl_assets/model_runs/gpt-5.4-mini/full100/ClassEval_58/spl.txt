# CLASS MinesweeperGame

[DEFINE_WORKER: "Initialize the object by storing the provided dimensions, generating the minesweeper and player maps, and resetting the score to zero." __init__]
    [INPUTS]
        <REF> n </REF>: any "The first constructor argument, stored directly on the object."
        <REF> k </REF>: any "The second constructor argument, stored directly on the object."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self </REF>: object "The initialized instance with n, k, minesweeper_map, player_map, and score fields set."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the input value n in self.n. RESULT self.n]
            [COMMAND Store the input value k in self.k. RESULT self.k]
            [COMMAND Generate the minesweeper map by calling self.generate_mine_sweeper_map() and store the returned value in self.minesweeper_map. RESULT self.minesweeper_map]
            [COMMAND Generate the player map by calling self.generate_playerMap() and store the returned value in self.player_map. RESULT self.player_map]
            [COMMAND Initialize self.score to the constant value 0. RESULT self.score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate a square Minesweeper board with random mine placements and increment the counts of all valid neighboring cells around each mine." generate_mine_sweeper_map]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: 2D list "A self.n by self.n Minesweeper board containing 'X' mines and integer neighbor counts."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new self.n by self.n 2D list named arr and initialize every cell to 0. RESULT arr]
            [COMMAND Repeat the following block self.k times: randomly choose integer coordinates x and y in the inclusive range 0 to self.n - 1. RESULT x,y]
            [COMMAND Within the same loop iteration, store 'X' in arr[y][x] to place a mine at the chosen cell. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 0 and self.n - 2 inclusive and y is between 0 and self.n - 1 inclusive, then if arr[y][x + 1] is not 'X', increment arr[y][x + 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 1 and self.n - 1 inclusive and y is between 0 and self.n - 1 inclusive, then if arr[y][x - 1] is not 'X', increment arr[y][x - 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 1 and self.n - 1 inclusive and y is between 1 and self.n - 1 inclusive, then if arr[y - 1][x - 1] is not 'X', increment arr[y - 1][x - 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 0 and self.n - 2 inclusive and y is between 1 and self.n - 1 inclusive, then if arr[y - 1][x + 1] is not 'X', increment arr[y - 1][x + 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 0 and self.n - 1 inclusive and y is between 1 and self.n - 1 inclusive, then if arr[y - 1][x] is not 'X', increment arr[y - 1][x] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 0 and self.n - 2 inclusive and y is between 0 and self.n - 2 inclusive, then if arr[y + 1][x + 1] is not 'X', increment arr[y + 1][x + 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 1 and self.n - 1 inclusive and y is between 0 and self.n - 2 inclusive, then if arr[y + 1][x - 1] is not 'X', increment arr[y + 1][x - 1] by 1. RESULT arr]
            [COMMAND Still within the same iteration, if x is between 0 and self.n - 1 inclusive and y is between 0 and self.n - 2 inclusive, then if arr[y + 1][x] is not 'X', increment arr[y + 1][x] by 1. RESULT arr]
            [COMMAND After completing all self.k iterations, return arr. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates and returns a square self.n by self.n matrix filled with the string '-' in every cell." generate_playerMap]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A square matrix of size self.n by self.n where every entry is '-'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a 2D square list with self.n rows and self.n columns, initializing every cell to the string '-'. RESULT arr]
            [COMMAND Return the initialized 2D list. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans every cell in an n by n grid and returns False immediately if any '-' cell in the given map does not correspond to an 'X' cell in the minesweeper map, otherwise returns True." check_won]
    [INPUTS]
        <REF> map </REF>: grid "The current n by n board state being validated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "False if any '-' cell is not matched by 'X' in the minesweeper map; otherwise True."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the outer loop for i from 0 to self.n - 1, iterate through each row of the board. RESULT row_scan_started]
            [COMMAND Within the inner loop for j from 0 to self.n - 1, inspect each cell at position (i, j) in row-major order. RESULT cell_scan_started]
            [COMMAND For each cell (i, j), if map[i][j] is '-' and self.minesweeper_map[i][j] is not 'X', return False immediately. RESULT early_failure]
            [COMMAND After all i and j positions have been checked without finding a mismatching cell, return True. RESULT success]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reveal a selected Minesweeper cell unless it contains an X, update the player board and score, and return either failure, victory, or the updated board." sweep]
    [INPUTS]
        <REF> self </REF>: object "Game instance containing the minesweeper map, player map, score, and win-check logic."
        <REF> x </REF>: int "Row index of the selected cell."
        <REF> y </REF>: int "Column index of the selected cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool_or_board "Returns false if the selected cell contains X, true if the move reveals a winning state, otherwise returns the updated player map."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.minesweeper_map[x][y] is equal to 'X'. RESULT cell_is_mine]
            [COMMAND If the selected cell is not 'X', copy self.minesweeper_map[x][y] into self.player_map[x][y] and increment self.score by 1. RESULT updated_state]
            [COMMAND Within the non-'X' branch, call self.check_won(self.player_map) and compare the result to True. RESULT won]
            [COMMAND If the win check is True, return True immediately. RESULT true]
            [COMMAND If the win check is not True, return self.player_map. RESULT self.player_map]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The selected cell contains 'X'.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.player_map or self.score. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
