# CLASS TwentyFourPointGame

[DEFINE_WORKER: "Initialize the instance by creating an empty list for nums." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new empty list and assign it to the instance attribute nums. RESULT self.nums]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates four random integers from 1 to 9 and appends them to self.nums, then verifies that the list contains exactly four items." _generate_cards]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate exactly 4 times over the range from 0 to 3. RESULT loop_index]
            [COMMAND On each iteration, append one random integer between 1 and 9 inclusive to self.nums. RESULT self.nums]
            [COMMAND Assert that len(self.nums) equals 4 after the loop completes. RESULT validation_passed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Initializes the instance card list, populates it by calling the internal card-generation routine, and returns the resulting list." get_my_cards]
    [INPUTS]
        <REF> self </REF>: object "The current instance whose nums attribute will be initialized and populated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The list stored in self.nums after card generation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.nums to a new empty list. RESULT self.nums]
            [COMMAND Call the internal card-generation routine <SPL>_generate_cards</SPL> to populate self.nums. RESULT _generate_cards_result]
            [COMMAND Return the populated self.nums list. RESULT self.nums]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates an input expression against the required digit counts in self.nums, and if all counts are satisfied and reduced to zero, evaluates the expression; otherwise it returns False, with a special pass case that returns the current cards." answer]
    [INPUTS]
        <REF> expression </REF>: string "Input expression to validate and possibly evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean or evaluated value "Returns self.get_my_cards() when expression is 'pass', returns the evaluated expression when validation succeeds, or returns False when validation fails."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If expression equals 'pass', return self.get_my_cards() immediately. RESULT result]
            [COMMAND Initialize statistic as an empty dictionary. RESULT statistic]
            [COMMAND Within the outer loop over each character c in expression, if c is a digit and int(c) exists in self.nums, increment statistic[c] by 1 using a default of 0. RESULT statistic]
            [COMMAND Copy statistic into nums_used. RESULT nums_used]
            [COMMAND Within the loop over each num in self.nums, if nums_used has the string form of num and its count is greater than 0, decrement that count by 1; otherwise return False immediately. RESULT nums_used]
            [COMMAND After the loop, if all counts in nums_used are exactly 0, return self.evaluate_expression(expression); otherwise return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Evaluates the given expression and returns True only when it computes to 24; otherwise it returns False, including when evaluation raises an exception." evaluate_expression]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> expression </REF>: string "Expression to evaluate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the evaluated expression equals 24, otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the given expression. RESULT evaluated_value]
            [COMMAND If the evaluated value is exactly 24, return True. RESULT True]
            [COMMAND Otherwise, return False. RESULT False]
            [COMMAND If evaluating the expression raises any exception, catch it and return False. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
