# CLASS Snake

[DEFINE_WORKER: "Initializes a snake game instance with default length, screen settings, a centered starting position, zero score, and the provided food position." __init__]
    [INPUTS]
        <REF> SCREEN_WIDTH </REF>: number "Screen width used to store the playfield size and compute the initial centered position."
        <REF> SCREEN_HEIGHT </REF>: number "Screen height used to store the playfield size and compute the initial centered position."
        <REF> BLOCK_SIZE </REF>: number "Block size copied directly into the instance."
        <REF> food_position </REF>: any "Initial food position assigned directly to the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance length to the fixed default value 1. RESULT self.length]
            [COMMAND Copy SCREEN_WIDTH into the instance screen width field. RESULT self.SCREEN_WIDTH]
            [COMMAND Copy SCREEN_HEIGHT into the instance screen height field. RESULT self.SCREEN_HEIGHT]
            [COMMAND Copy BLOCK_SIZE into the instance block size field. RESULT self.BLOCK_SIZE]
            [COMMAND Initialize the instance positions list with one centered coordinate computed as SCREEN_WIDTH / 2 and SCREEN_HEIGHT / 2. RESULT self.positions]
            [COMMAND Set the instance score to the fixed default value 0. RESULT self.score]
            [COMMAND Assign the provided food_position directly to the instance food position field. RESULT self.food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Move the current position by one block in the given direction with screen wraparound, then handle food collection, self-collision reset, and body growth or trimming." move]
    [INPUTS]
        <REF> direction </REF>: tuple[int, int] "A two-element movement vector unpacked into x and y components."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: None "This method updates the object state in place and does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current head position from self.positions[0] and unpack the movement direction into x and y. RESULT cur, x, y]
            [COMMAND Compute the next wrapped position by adding x times self.BLOCK_SIZE to the current x-coordinate modulo self.SCREEN_WIDTH, and y times self.BLOCK_SIZE to the current y-coordinate modulo self.SCREEN_HEIGHT. RESULT new]
            [COMMAND If the computed position equals self.food_position, call self.eat_food() before any body update logic. RESULT food_checked]
            [COMMAND If self.positions has more than 2 elements and the new position appears in self.positions[2:], call self.reset() to handle self-collision. RESULT collision_checked]
            [COMMAND Otherwise, insert the new position at the front of self.positions. RESULT self.positions]
            [COMMAND Within the same non-collision path, if len(self.positions) is greater than self.length, remove the last position with pop() to keep the body within its maximum length. RESULT self.positions]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The new position matches self.food_position.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.eat_food() to apply the food consumption effect before proceeding with movement updates. RESULT self.eat_food()]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len(self.positions) > 2 and the new position is found in self.positions[2:].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.reset() and skip the insertion and trimming branch. RESULT self.reset()]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Repeatedly selects a random grid-aligned food position until it is not already occupied by an existing position." random_food_position]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND While the current food position is present in self.positions, generate a new random grid-aligned coordinate for self.food_position using independent random integers for x and y within the inclusive range from 0 to SCREEN_WIDTH // BLOCK_SIZE - 1 and 0 to SCREEN_HEIGHT // BLOCK_SIZE - 1, then multiply each coordinate by BLOCK_SIZE. RESULT self.food_position]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reset the game state to a one-segment snake positioned at the screen center, clear the score, and place food at a new random location." reset]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.length to 1 to initialize the snake to its default size. RESULT self.length]
            [COMMAND Set self.positions to a single-element list containing the centered starting position computed as self.SCREEN_WIDTH / 2 and self.SCREEN_HEIGHT / 2. RESULT self.positions]
            [COMMAND Set self.score to 0 to reset the score to its default starting value. RESULT self.score]
            [COMMAND Call self.random_food_position() to generate a new food position after resetting the game state. RESULT self.random_food_position()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Increments the snake's length and score, then triggers a new food position to be generated." eat_food]
    [INPUTS]
        <REF> self </REF>: object "The game or snake instance whose length, score, and food position are updated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This method does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increase self.length by 1 to reflect that food was eaten. RESULT self.length]
            [COMMAND Increase self.score by 100 to award points for eating the food. RESULT self.score]
            [COMMAND Call self.random_food_position() to update the food location after consumption. RESULT self.random_food_position()]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
