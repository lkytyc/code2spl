[DEFINE_WORKER: "Reads a CSV file, stores all data rows after the header, and returns the header title with the collected rows." read_csv]
    [INPUTS]
        <REF> file_name </REF>: str "Path to the CSV file to read."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the header row as title and a list of remaining rows as data."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named data to collect CSV rows. RESULT data]
            [COMMAND Open the file named file_name in read mode as file. RESULT file]
            [COMMAND Create a CSV reader over the opened file. RESULT reader]
            [COMMAND Read the first row from the CSV reader and store it as title. RESULT title]
            [COMMAND For each remaining row in the CSV reader, append the row to data. RESULT data]
            [COMMAND Return title and data as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Writes a 2D data table to a CSV file and returns 1 on success or 0 if any exception occurs." write_csv]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> data </REF>: list "Rows of data to write to the CSV file."
        <REF> file_name </REF>: string "Path of the CSV file to create or overwrite."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 1 when writing succeeds, otherwise returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the target file at file_name in write mode with newline set to an empty string. RESULT file]
            [COMMAND Within the open file context, create a csv.writer bound to the file handle. RESULT writer]
            [COMMAND Within the open file context, write all rows from data using writer.writerows(data). RESULT write_result]
            [COMMAND After successfully writing the rows and closing the file context, return 1. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If any exception occurs during file opening, writer creation, or row writing, catch it and return 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Handle the exception by skipping any further file-writing steps. RESULT handled_exception]
            [COMMAND Return 0 to indicate failure. RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a CSV file, extracts the Nth column from each data row, uppercases those values, and writes the processed column back to a new CSV file with a _process suffix." process_csv_data]
    [INPUTS]
        <REF> N </REF>: int "Zero-based column index to extract from each row."
        <REF> save_file_name </REF>: str "Path to the CSV file to read and use when building the output filename."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The return value from writing the processed CSV file."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the CSV file identified by save_file_name and unpack the result into title and data. RESULT title, data]
            [COMMAND Build column_data by selecting the Nth element from each row in data. RESULT column_data]
            [COMMAND Convert each value in column_data to uppercase. RESULT column_data]
            [COMMAND Create new_data as a two-element collection containing title and the transformed column_data. RESULT new_data]
            [COMMAND Write new_data to a CSV file named by removing the extension from save_file_name and appending _process.csv, then return the write result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
