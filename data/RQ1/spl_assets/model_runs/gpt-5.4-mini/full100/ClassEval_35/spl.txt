[DEFINE_WORKER: "Scan a 3x3 grid in row-major order and return the coordinates of the first cell whose value is zero." find_blank]
    [INPUTS]
        <REF> self </REF>: object "Instance reference."
        <REF> state </REF>: list[list[int]] "A 3x3 grid of cell values to search."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple[int, int] "The row and column of the first zero-valued cell found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each i from 0 to 2, iterate over the rows of the 3x3 state grid. RESULT i]
            [COMMAND Within the outer loop for i, for each j from 0 to 2, iterate over the columns of the current row. RESULT j]
            [COMMAND Within the nested loops over i and j, check whether state[i][j] is exactly 0. RESULT cell_is_zero]
            [COMMAND Within the nested loops over i and j, if state[i][j] equals 0, return the coordinates i and j immediately. RESULT (i, j)]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If no cell in the 3x3 grid equals 0, the loops finish without returning a coordinate.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Complete all iterations over i from 0 to 2 and j from 0 to 2 without finding a zero-valued cell. RESULT search_exhausted]
            [COMMAND Continue execution after the loop body with no coordinate returned from this snippet. RESULT no_return_from_snippet]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return a copied board state after swapping the blank tile with its adjacent tile in the specified direction." move]
    [INPUTS]
        <REF> state </REF>: list[list] "Current grid state to be copied and updated."
        <REF> direction </REF>: str "Direction command that selects which adjacent tile to swap with the blank tile."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[list] "A new grid state with the blank tile swapped in the requested direction, or an unchanged copied grid if the direction does not match any branch."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the blank position in the current state and unpack its row and column indices into i and j. RESULT i, j]
            [COMMAND Create a shallow-copied new grid state by copying each row of the input state. RESULT new_state]
            [COMMAND Within the conditional branch for direction equal to 'up', swap the blank at (i, j) with the tile directly above it at (i - 1, j). RESULT new_state]
            [COMMAND Within the conditional branch for direction equal to 'down', swap the blank at (i, j) with the tile directly below it at (i + 1, j). RESULT new_state]
            [COMMAND Within the conditional branch for direction equal to 'left', swap the blank at (i, j) with the tile directly to the left at (i, j - 1). RESULT new_state]
            [COMMAND Within the conditional branch for direction equal to 'right', swap the blank at (i, j) with the tile directly to the right at (i, j + 1). RESULT new_state]
            [COMMAND Return the updated copied grid state. RESULT new_state]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the list of valid directional moves from the blank tile position in the state based on grid boundaries." get_possible_moves]
    [INPUTS]
        <REF> state </REF>: any "The puzzle state used to locate the blank tile."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of strings "A list containing each legal move direction: up, down, left, and/or right."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named moves to collect valid directions. RESULT moves]
            [COMMAND Call self.find_blank(state) to determine the blank tile coordinates and assign the returned pair to i and j. RESULT i, j]
            [COMMAND If i is greater than 0, append 'up' to moves because the blank tile can move upward only when it is not already in the top row. RESULT moves]
            [COMMAND If i is less than 2, append 'down' to moves because the blank tile can move downward only when it is not already in the bottom row. RESULT moves]
            [COMMAND If j is greater than 0, append 'left' to moves because the blank tile can move left only when it is not already in the leftmost column. RESULT moves]
            [COMMAND If j is less than 2, append 'right' to moves because the blank tile can move right only when it is not already in the rightmost column. RESULT moves]
            [COMMAND Return the accumulated moves list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Perform a breadth-first search over states using a queue of state-path pairs, returning the first path that reaches the goal state or None if no solution exists." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the frontier with the initial state paired with an empty path, and initialize the closed list as empty. RESULT open_list, closed_list]
            [COMMAND While the frontier is non-empty, remove the first queued state-path pair, and record the current state in the closed list. RESULT current_state, path]
            [COMMAND Within the while loop, if the current state is exactly equal to the goal state, return the current path immediately. RESULT path]
            [COMMAND Within the while loop, for each possible move from the current state, compute the resulting new state. RESULT new_state]
            [COMMAND Within the while loop and for each move, if the new state is not already in the closed list, append the new state together with the extended path to the frontier. RESULT open_list]
            [COMMAND If the frontier becomes empty without reaching the goal state, return None. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
