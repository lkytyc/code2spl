# CLASS TicTacToe

[DEFINE_WORKER: "Initializes a 3-row by N-column board filled with spaces and sets the current player to X." __init__]
    [INPUTS]
        <REF> N </REF>: int "Number of columns used to create each board row; defaults to 3."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create self.board as a list of 3 rows, where each row is a list of N space characters. RESULT self.board]
            [COMMAND Assign the constant string X to self.current_player. RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Places the current player's mark in an empty board cell, switches to the other player, and returns whether the move succeeded." make_move]
    [INPUTS]
        <REF> self </REF>: object "The game instance containing the board and current player state."
        <REF> row </REF>: int "The row index of the target board cell."
        <REF> col </REF>: int "The column index of the target board cell."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was applied to an empty cell; otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the board cell at row and col is empty by testing self.board[row][col] == ' '. RESULT cell_is_empty]
            [COMMAND If the cell is empty, place self.current_player into self.board[row][col]. RESULT board_updated]
            [COMMAND If the cell is empty, switch self.current_player to 'O' when it is currently 'X', otherwise switch it to 'X'. RESULT current_player_switched]
            [COMMAND If the cell is empty, return True to signal a successful move. RESULT true]
            [COMMAND If the cell is not empty, return False to signal that the move was rejected. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The target board cell already contains a non-space value.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Reject the move without changing the board or the current player. RESULT unchanged_state]
            [COMMAND Return False. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks a 3x3 tic-tac-toe board for any winning row, column, or diagonal and returns the winning symbol, or None if no winner exists." check_winner]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | None "The winning symbol on the board, such as X or O, or None when there is no winner."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Scan each row of self.board, and within that loop, if the first three cells in the row are equal and not a blank space, return that row's symbol immediately. RESULT winner]
            [COMMAND Scan each column index from 0 through 2, and within that loop, if the top, middle, and bottom cells in the column are equal and not a blank space, return that column's symbol immediately. RESULT winner]
            [COMMAND Check the main diagonal from top-left to bottom-right, and if all three cells are equal and not a blank space, return the top-left diagonal symbol immediately. RESULT winner]
            [COMMAND Check the anti-diagonal from top-right to bottom-left, and if all three cells are equal and not a blank space, return the top-right diagonal symbol immediately. RESULT winner]
            [COMMAND If no row, column, or diagonal contains three matching non-blank symbols, return None. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether every row in the board contains no space characters and returns False immediately if any empty cell is found." is_board_full]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the board to inspect."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True when the board has no space characters in any row; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board in order. RESULT row]
            [COMMAND Within the loop over self.board, check whether the current row contains a space character. RESULT row_has_space]
            [COMMAND If the current row contains a space character, return False immediately. RESULT False]
            [COMMAND After all rows have been checked and none contain a space character, return True. RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
