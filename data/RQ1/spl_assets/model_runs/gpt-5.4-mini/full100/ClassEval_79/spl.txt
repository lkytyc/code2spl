# CLASS SQLGenerator

[DEFINE_WORKER: "Initializes the instance by storing the provided table name on the object." __init__]
    [INPUTS]
        <REF> table_name </REF>: any "The table name value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: null "This initializer does not return a value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input parameter table_name directly to the instance field self.table_name. RESULT self.table_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a SQL SELECT statement by normalizing the field list, optionally adding a WHERE clause, and returning the final query with a trailing semicolon." select]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name."
        <REF> fields </REF>: list or null "Field names to select; defaults to all fields when null."
        <REF> condition </REF>: string or null "Optional SQL condition appended as a WHERE clause when provided."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A complete SQL SELECT statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If fields is null, set fields to the wildcard string; otherwise join the field names with comma-space separators. RESULT fields]
            [COMMAND Construct the base SQL string using the normalized fields and the table name from self. RESULT sql]
            [COMMAND If condition is not null, append a WHERE clause containing the condition to sql. RESULT sql]
            [COMMAND Return sql with a trailing semicolon appended. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build an SQL INSERT statement from a mapping by joining its keys as column names and its values as quoted literals, then appending a semicolon." insert]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name through self.table_name."
        <REF> data </REF>: mapping "Mapping whose keys become SQL column names and whose values become inserted field values."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The constructed SQL INSERT statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join the keys from data with comma and space separators to produce the field list. RESULT fields]
            [COMMAND Join the values from data with comma and space separators after wrapping each value in single quotes to produce the value list. RESULT values]
            [COMMAND Format an SQL INSERT statement using self.table_name, fields, and values in the template INSERT INTO {self.table_name} ({fields}) VALUES ({values}). RESULT sql]
            [COMMAND Append a semicolon to the SQL string and return the final statement. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build and return a SQL UPDATE statement by joining field-value assignments from the input data and applying the provided table name and condition." update]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the table_name attribute."
        <REF> data </REF>: mapping "Source of column names and values to convert into SQL assignments."
        <REF> condition </REF>: string "SQL WHERE predicate used to restrict the update."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A complete SQL UPDATE statement terminated with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join all field-value pairs from data.items() into a comma-separated SET clause, formatting each pair as field = 'value'. RESULT set_clause]
            [COMMAND Construct an SQL UPDATE statement using self.table_name as the target table, the generated set_clause as the SET portion, and condition as the WHERE predicate. RESULT sql]
            [COMMAND Append a semicolon to the SQL statement and return the final string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a SQL DELETE statement for the current table using the provided condition and return it with a trailing semicolon." delete]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the target table name through self.table_name."
        <REF> condition </REF>: string "SQL condition used in the WHERE clause to select rows for deletion."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A SQL DELETE statement ending with a semicolon."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a DELETE SQL string using self.table_name in the FROM clause and condition in the WHERE clause. RESULT sql]
            [COMMAND Append a semicolon to the SQL string and return the final statement. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a query condition that filters records to those with age below the given threshold and gender equal to female, then delegates the filtering to the select method." select_female_under_age]
    [INPUTS]
        <REF> self </REF>: object "The current instance that provides the select method."
        <REF> age </REF>: number "The upper age threshold used to build the filter condition."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The return value from self.select applied with the constructed condition."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a condition string that compares age with the provided threshold using a strict less-than boundary and requires gender to equal female. RESULT condition]
            [COMMAND Call self.select with the constructed condition and return its result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a between-range age condition from the provided minimum and maximum ages, then delegate the selection to the underlying select method using that condition." select_by_age_range]
    [INPUTS]
        <REF> self </REF>: object "The current instance used to access the selection method."
        <REF> min_age </REF>: any "The lower bound of the age range used in the condition string."
        <REF> max_age </REF>: any "The upper bound of the age range used in the condition string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The value returned by self.select when called with the constructed condition."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a condition string in the form age BETWEEN min_age AND max_age using the provided minimum and maximum age values. RESULT condition]
            [COMMAND Call self.select with the constructed condition as the condition argument and return its result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
