# CLASS BinaryDataProcessor

[DEFINE_WORKER: "Initializes the object by storing the provided binary string and then cleaning non-binary characters from it." __init__]
    [INPUTS]
        <REF> binary_string </REF>: string "The input string to be stored on the instance and cleaned."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Store the provided binary_string value in the instance attribute self.binary_string. RESULT self.binary_string]
            [COMMAND Call self.clean_non_binary_chars() to remove any non-binary characters from the stored string. RESULT cleaned_binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters the instance attribute binary_string so it contains only the characters '0' and '1', then stores the cleaned result back on the object." clean_non_binary_chars]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the binary_string attribute to be cleaned."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: void "No explicit return value; the method updates self.binary_string in place."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.binary_string. RESULT current_binary_string]
            [COMMAND Filter current_binary_string to keep only characters that are either '0' or '1'. RESULT filtered_binary_chars]
            [COMMAND Join the filtered characters into a new string. RESULT clean_binary_string]
            [COMMAND Assign clean_binary_string back to self.binary_string, replacing the previous value. RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Counts zeroes and ones in the binary string, computes each as a fraction of the total string length, and returns those ratios together with the bit length." calculate_binary_info]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the occurrences of the character '0' in self.binary_string and store the result as zeroes_count. RESULT zeroes_count]
            [COMMAND Count the occurrences of the character '1' in self.binary_string and store the result as ones_count. RESULT ones_count]
            [COMMAND Compute the length of self.binary_string and store the result as total_length. RESULT total_length]
            [COMMAND Compute zeroes_percentage as zeroes_count divided by total_length. RESULT zeroes_percentage]
            [COMMAND Compute ones_percentage as ones_count divided by total_length. RESULT ones_percentage]
            [COMMAND Return a dictionary containing 'Zeroes' mapped to zeroes_percentage, 'Ones' mapped to ones_percentage, and 'Bit length' mapped to total_length. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a binary string into ASCII text by grouping the bits into 8-bit chunks, converting each chunk to a byte value, and decoding the resulting bytearray as ASCII." convert_to_ascii]
    [INPUTS]
        <REF> self.binary_string </REF>: str "A string containing binary digits that is processed in 8-bit segments."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The ASCII-decoded text produced from the accumulated byte values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty mutable bytearray named byte_array. RESULT byte_array]
            [COMMAND For each index i from 0 up to but not including the length of self.binary_string, advancing by 8, extract the 8-character slice self.binary_string[i:i+8], convert that slice from base 2 to an integer, and append the integer to byte_array. RESULT byte_array]
            [COMMAND Decode byte_array using ASCII and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a binary string stored on the object into a UTF-8 decoded byte array by reading 8-bit chunks, converting each chunk to an integer, and appending the bytes before decoding." convert_to_utf8]
    [INPUTS]
        <REF> self.binary_string </REF>: string "A string of binary digits that is processed in 8-bit segments."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bytes "The UTF-8 decoded value produced from the accumulated byte array."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty bytearray named byte_array. RESULT byte_array]
            [COMMAND For each i starting at 0 and increasing by 8 while i is less than the length of self.binary_string, extract the 8-character slice self.binary_string[i:i+8], convert that binary slice to an integer with base 2, and append the integer to byte_array. RESULT byte_array]
            [COMMAND Decode byte_array using UTF-8 and return the decoded result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
