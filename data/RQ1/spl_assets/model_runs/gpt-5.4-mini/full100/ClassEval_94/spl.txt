# CLASS VendingMachine

[DEFINE_WORKER: "Initialize the object state by creating an empty inventory dictionary and setting the balance to zero." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.inventory to an empty dictionary to represent no stored items. RESULT self.inventory]
            [COMMAND Initialize self.balance to 0 to establish the default zero balance state. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds or updates an inventory item by first attempting to restock it, and if restocking fails, stores the item with the provided price and quantity." add_item]
    [INPUTS]
        <REF> self </REF>: object "The owning inventory manager instance."
        <REF> item_name </REF>: string "The name of the item to add or restock."
        <REF> price </REF>: number "The price to store for the item if a new inventory entry is created."
        <REF> quantity </REF>: number "The quantity to restock or store for the item."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.restock_item(item_name, quantity) and evaluate its truthiness. RESULT restock_success]
            [COMMAND If restock_success is false, create or overwrite self.inventory[item_name] with a record containing the given price and quantity. RESULT inventory_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds the deposited amount to the current balance and returns the updated balance." insert_coin]
    [INPUTS]
        <REF> self </REF>: object "The instance whose balance is updated."
        <REF> amount </REF>: number "The amount to add to the current balance."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The updated balance after adding the amount."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current balance from the instance. RESULT current_balance]
            [COMMAND Add the input amount to the current balance and store the updated value back into the instance balance. RESULT updated_balance]
            [COMMAND Return the updated balance. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether an item exists in inventory, verifies stock and balance, then updates the balance and quantity when a purchase succeeds." purchase_item]
    [INPUTS]
        <REF> item_name </REF>: string "Name of the item to buy and look up in the inventory."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number or boolean "Returns the updated balance after a successful purchase, or False if the item is missing, out of stock, or unaffordable."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists in self.inventory. RESULT inventory_membership]
            [COMMAND If item_name exists in self.inventory, retrieve the matching item record from self.inventory[item_name]. RESULT item]
            [COMMAND Within the inventory-hit path, check whether item['quantity'] is greater than 0 and self.balance is greater than or equal to item['price'], using short-circuit and logic. RESULT purchase_eligibility]
            [COMMAND Within the eligible purchase path, subtract item['price'] from self.balance, decrement item['quantity'] by 1, and return the updated self.balance. RESULT self.balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: If item_name is not found in self.inventory, return False immediately.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying balance or inventory. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: If item_name exists but item['quantity'] is not greater than 0 or self.balance is less than item['price'], return False immediately.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying balance or inventory. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Restock an inventory item by increasing its quantity when the item exists, otherwise return false without changing the inventory." restock_item]
    [INPUTS]
        <REF> self </REF>: object "The object containing the inventory mapping."
        <REF> item_name </REF>: string "The inventory key identifying the item to restock."
        <REF> quantity </REF>: number "The amount to add to the existing item quantity."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the item was found and updated, otherwise false."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory. RESULT item_exists]
            [COMMAND If item_name exists, increase self.inventory[item_name]['quantity'] by quantity. RESULT updated_quantity]
            [COMMAND If item_name exists, return true immediately after updating the quantity. RESULT true]
            [COMMAND If item_name does not exist, return false without modifying the inventory. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name is not present in self.inventory.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip any inventory update because the item is missing. RESULT no_change]
            [COMMAND Return false to signal that restocking did not occur. RESULT false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return False when the inventory is empty; otherwise format each inventory item as a price and quantity line joined by newline characters." display_items]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the inventory mapping."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean or string "False when the inventory is empty, otherwise a newline-separated string of formatted inventory items."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.inventory is falsy or empty. RESULT inventory_is_empty]
            [COMMAND If self.inventory is empty, return False immediately. RESULT False]
            [COMMAND Otherwise, create an empty list named items. RESULT items]
            [COMMAND Within the loop over each item_name and item_info in self.inventory.items(), format each entry as item_name, a hyphen, the price prefixed with $, and the quantity in square brackets, then append the formatted string to items. RESULT items]
            [COMMAND After processing all inventory entries, join items with newline characters and return the resulting string. RESULT joined_items]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.inventory is empty or otherwise falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without creating the item list or entering the loop. RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
