# CLASS ClassRegistrationSystem

[DEFINE_WORKER: "Initializes the object with an empty student list and an empty mapping of student registration classes." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set self.students to a new empty list. RESULT self.students]
            [COMMAND Set self.students_registration_classes to a new empty dictionary. RESULT self.students_registration_classes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a student only if they are not already present, returning 0 for duplicates and 1 after adding a new student." register_student]
    [INPUTS]
        <REF> student </REF>: any "The student value to check for membership and possibly add to the students collection."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 0 if the student was already present, or 1 if the student was added successfully."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the given student is already present in self.students. RESULT is_present]
            [COMMAND If the student is already present, return 0 immediately without modifying self.students. RESULT 0]
            [COMMAND If the student is not present, append the student to self.students. RESULT self.students]
            [COMMAND After appending the new student, return 1 to indicate that the collection was changed. RESULT 1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Registers a class for a student by appending it to the student's existing list or creating a new list when the student is not yet registered, then returns that student's updated class list." register_class]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the students_registration_classes mapping."
        <REF> student_name </REF>: string "Student identifier used as the dictionary key."
        <REF> class_name </REF>: string "Class name to register for the student."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The updated list of classes registered for the given student."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether student_name is already present as a key in self.students_registration_classes. RESULT membership_check]
            [COMMAND If student_name is already present, append class_name to the existing list at self.students_registration_classes[student_name]. RESULT updated_existing_list]
            [COMMAND If student_name is not present, create a new list containing class_name and assign it to self.students_registration_classes[student_name]. RESULT initialized_student_list]
            [COMMAND Return self.students_registration_classes[student_name] after the dictionary entry has been updated or created. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Build a list of student names from self.students whose major exactly matches the given major." get_students_by_major]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a students collection."
        <REF> major </REF>: any "Target major used to filter students by exact equality."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of student names for entries whose major equals the requested major."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named student_list to collect matching student names. RESULT student_list]
            [COMMAND Within the loop over each student in self.students, check whether student["major"] exactly equals the target major. RESULT match_check]
            [COMMAND Within the loop over each student in self.students, when student["major"] equals major, append student["name"] to student_list. RESULT student_list]
            [COMMAND After processing all students, return student_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect and return the unique student majors from self.students in the order they are first encountered." get_all_major]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a students collection with records that include a major field."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of distinct major values preserving first-seen order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an empty list named major_list to store unique majors. RESULT major_list]
            [COMMAND For each student in self.students, read the student's major and check whether it is already present in major_list. RESULT student_major_check]
            [COMMAND Within the for loop, if the student's major is not already in major_list, append that major to major_list. RESULT major_list]
            [COMMAND After all students have been processed, return major_list. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Collect all classes taken by students in the given major and return the most frequent class among them." get_most_popular_class_in_major]
    [INPUTS]
        <REF> self </REF>: object "Instance containing student records and registration class mappings."
        <REF> major </REF>: any "Major value used to filter students for aggregation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The class that appears most frequently among students in the specified major."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named class_list to accumulate classes from matching students. RESULT class_list]
            [COMMAND Iterate over each student in self.students. RESULT student]
            [COMMAND Within the loop, check whether student["major"] is equal to major; if not, skip adding classes for that student. RESULT major_match]
            [COMMAND Within the loop, when student["major"] equals major, extend class_list with self.students_registration_classes[student["name"]]. RESULT class_list]
            [COMMAND After processing all students, compute the unique classes from class_list and select the one with the highest occurrence count in the original list. RESULT most_popular_class]
            [COMMAND Return most_popular_class. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: class_list is empty when evaluating the most frequent class.]
        [LOG "max(set(class_list), key=class_list.count) has no elements to evaluate and fails on an empty input."]
        [THROW ValueError "Raised when the method attempts to find a maximum from an empty set of classes."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
