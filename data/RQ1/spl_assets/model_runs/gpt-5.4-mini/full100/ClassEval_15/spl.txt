# CLASS BoyerMooreSearch

[DEFINE_WORKER: "Initializes the object with the given text and pattern and precomputes their lengths for later boundary-aware processing." __init__]
    [INPUTS]
        <REF> text </REF>: string "Input text to store and measure."
        <REF> pattern </REF>: string "Pattern text to store and measure."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input text to the object field self.text and the input pattern to the object field self.pattern. RESULT self.text, self.pattern]
            [COMMAND Compute the length of text and store it in self.textLen, then compute the length of pattern and store it in self.patLen. RESULT self.textLen, self.patLen]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Search backward through the pattern for the rightmost occurrence of the given character and return its index, or return -1 if no match is found." match_in_pattern]
    [INPUTS]
        <REF> self </REF>: object "Object containing the pattern sequence and its length."
        <REF> char </REF>: character "Character to search for in the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The index of the first match found when scanning from the end, or -1 if the character is not present."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the reverse for loop over i from self.patLen - 1 down to 0 inclusive, compare char with self.pattern[i]. RESULT comparison_result]
            [COMMAND If char equals self.pattern[i], return i immediately as the rightmost matching index. RESULT i]
            [COMMAND If the loop completes without finding any match, return -1. RESULT -1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the pattern from end to start and returns the text index of the first mismatching character, or -1 if the pattern fully matches." mismatch_in_text]
    [INPUTS]
        <REF> self </REF>: object "Instance providing pattern, text, and patLen."
        <REF> currentPos </REF>: integer "Starting position in text to compare against the pattern."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The index in text of the first mismatch, or -1 if no mismatch is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a reverse loop over pattern indices from self.patLen - 1 down to 0, inclusive, stepping by -1. RESULT loop_started]
            [COMMAND Within the reverse loop, compare self.pattern[i] with self.text[currentPos + i]. RESULT comparison_result]
            [COMMAND Within the reverse loop, if the two values differ, return currentPos + i immediately as the first mismatch position. RESULT mismatch_index]
            [COMMAND If the reverse loop completes without any mismatch, return -1. RESULT -1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scan each valid text alignment, record positions where no mismatch is found, and otherwise attempt to compute a shifted alignment from the mismatch character." bad_character_heuristic]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named positions to collect matching start indices. RESULT positions]
            [COMMAND For each i from 0 through self.textLen - self.patLen inclusive, call self.mismatch_in_text(i) to find the first mismatch position for the current alignment. RESULT mismatch_index]
            [COMMAND Within the outer for loop, if mismatch_index equals -1, append the current i to positions because the pattern matches at this alignment. RESULT positions]
            [COMMAND Within the outer for loop, if mismatch_index is not -1, look up the current text character with self.text[mismatch_index] and pass it to self.match_in_pattern(...) to obtain a pattern index. RESULT match_index]
            [COMMAND Within the outer for loop, if mismatch_index is not -1, assign i to mismatch_index - match_index as an attempted bad-character shift, even though changing i inside a Python for loop does not affect the next range value. RESULT i]
            [COMMAND After all loop iterations complete, return positions. RESULT positions]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
