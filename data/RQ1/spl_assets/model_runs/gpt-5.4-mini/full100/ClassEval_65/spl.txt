# CLASS NumberWordFormatter

[DEFINE_WORKER: "Initializes several lookup tables on the instance for number words, teen names, tens names, larger scale names, and abbreviated suffix mappings." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the instance attribute NUMBER to a list containing an empty placeholder at index 0 followed by the words ONE through NINE. RESULT self.NUMBER]
            [COMMAND Assign the instance attribute NUMBER_TEEN to a fixed list of teen number words from TEN through NINETEEN. RESULT self.NUMBER_TEEN]
            [COMMAND Assign the instance attribute NUMBER_TEN to a fixed list of tens-place words from TEN through NINETY. RESULT self.NUMBER_TEN]
            [COMMAND Assign the instance attribute NUMBER_MORE to a list containing an empty placeholder followed by THOUSAND, MILLION, and BILLION. RESULT self.NUMBER_MORE]
            [COMMAND Assign the instance attribute NUMBER_SUFFIX to a 16-element suffix mapping list containing abbreviations such as k, w, m, b, t, p, and e, with empty strings used as placeholders in unused positions. RESULT self.NUMBER_SUFFIX]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a formatted string for x when x is not None, otherwise returns an empty string." format]
    [INPUTS]
        <REF> x </REF>: any "Value to format if it is not None."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The formatted string representation of x, or an empty string when x is None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether x is not None using an identity comparison. RESULT condition_result]
            [COMMAND If x is not None, convert x to a string and pass it to self.format_string, then return the formatted result immediately. RESULT formatted_result]
            [COMMAND If x is None, return an empty string immediately. RESULT empty_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats a numeric string into a spoken zero-and-cents phrase by splitting the input around a decimal point, converting the integer part in three-digit groups, optionally converting the cents part, and returning 'ZERO ONLY' when the whole value is zero." format_string]
    [INPUTS]
        <REF> self </REF>: object "Instance providing trans_three, parse_more, and trans_two conversion helpers."
        <REF> x </REF>: string "Input value to format; may contain a decimal point separating whole and fractional parts."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Formatted phrase ending with ONLY, or ZERO ONLY when the whole-number portion is blank after processing."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split the input string on the first decimal point, keep the first two parts only, and assign the whole-number part to lstr and the cents part to rstr, using an empty fallback when the decimal part is missing. RESULT lstr, rstr]
            [COMMAND Reverse the whole-number part into lstrrev so it can be processed in three-digit groups from least significant to most significant digits. RESULT lstrrev]
            [COMMAND Initialize a five-element list a with empty strings as a working buffer for processed three-digit chunks. RESULT a]
            [COMMAND If the length of lstrrev leaves remainder 1 when divided by 3, append "00" so the length becomes divisible by 3. RESULT lstrrev]
            [COMMAND Else if the length of lstrrev leaves remainder 2 when divided by 3, append "0" so the length becomes divisible by 3. RESULT lstrrev]
            [COMMAND Set lm to the empty string as the accumulator for the formatted whole-number phrase. RESULT lm]
            [COMMAND Within the outer loop over i from 0 up to but not including len(lstrrev) // 3, extract the three-character chunk starting at 3 * i, reverse that chunk, and store it in a[i]. RESULT a[i]]
            [COMMAND Within the same loop, if a[i] is not "000", prepend to lm the translated three-digit chunk, a space, the result of parse_more(i), another space, and the previous lm. RESULT lm]
            [COMMAND Within the same loop, if a[i] is "000", append only the translated three-digit chunk to lm. RESULT lm]
            [COMMAND Compute xs as "AND CENTS " followed by trans_two(rstr) and a trailing space when rstr is truthy; otherwise set xs to the empty string. RESULT xs]
            [COMMAND If lm is empty after trimming whitespace, return "ZERO ONLY" immediately. RESULT result]
            [COMMAND Otherwise, return the trimmed whole-number phrase, followed by xs, and then "ONLY". RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Convert a zero-padded two-digit string into its English number-word representation by handling leading zero, teen, exact tens, and mixed tens-and-ones cases." trans_two]
    [INPUTS]
        <REF> s </REF>: str "A numeric string that is left-padded with zeros to at least two characters before conversion."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "The English word representation of the two-digit number."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pad the input string on the left with zeros to ensure it is at least 2 characters long. RESULT s]
            [COMMAND If the first character of the padded string is 0, return the word for the last digit from NUMBER. RESULT result]
            [COMMAND Else if the first character of the padded string is 1, return the teen word from NUMBER_TEEN using int(s) - 10 as the index. RESULT result]
            [COMMAND Else if the second character of the padded string is 0, return the tens word from NUMBER_TEN using int(s[0]) - 1 as the index. RESULT result]
            [COMMAND Otherwise, return the tens word from NUMBER_TEN using int(s[0]) - 1, followed by a space, followed by the ones word from NUMBER using int(s[-1]) as the index. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a three-digit numeric string into words by handling a leading zero, exact hundreds, or hundreds with a remainder." trans_three]
    [INPUTS]
        <REF> self </REF>: object "Instance providing the NUMBER lookup and the trans_two helper."
        <REF> s </REF>: string "Three-character numeric string to convert."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Spelled-out representation of the input as a hundred-based number phrase."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the first character of s is "0"; if it is, return the result of trans_two on the substring s[1:] immediately. RESULT result]
            [COMMAND If the first character of s is not "0", check whether the substring s[1:] is exactly "00"; if it is, return the word for the first digit from NUMBER followed by " HUNDRED". RESULT result]
            [COMMAND If neither previous condition matches, return the word for the first digit from NUMBER followed by " HUNDRED AND " and the result of trans_two on s[1:]. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the element from self.NUMBER_MORE at index i." parse_more]
    [INPUTS]
        <REF> i </REF>: int "Index used to select an element from self.NUMBER_MORE."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: unknown "The value stored in self.NUMBER_MORE at the given index."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access self.NUMBER_MORE using index i. RESULT value]
            [COMMAND Return the retrieved value as the method result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
