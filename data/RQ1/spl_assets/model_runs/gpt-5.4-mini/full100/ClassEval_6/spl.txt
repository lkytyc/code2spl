# CLASS AvgPartition

[DEFINE_WORKER: "Initializes the object by storing the provided list and limit directly on the instance." __init__]
    [INPUTS]
        <REF> lst </REF>: any "The list or list-like value to store on the instance."
        <REF> limit </REF>: any "The limit value to store on the instance."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input list parameter directly to the instance attribute lst. RESULT self.lst]
            [COMMAND Assign the input limit parameter directly to the instance attribute limit. RESULT self.limit]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute the number of full chunks and the leftover remainder by dividing the length of self.lst by self.limit." setNum]
    [INPUTS]
        <REF> self </REF>: object "Instance containing lst and limit attributes."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A two-item tuple containing the quotient size and the remainder."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the length of self.lst and divide it by self.limit using floor division to obtain size. RESULT size]
            [COMMAND Compute the length of self.lst and take the modulus by self.limit to obtain remainder. RESULT remainder]
            [COMMAND Return size and remainder as a tuple. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.limit is zero when evaluating floor division or modulus.]
        [LOG "Division or modulo by zero occurs while computing size and remainder from self.lst length and self.limit."]
        [THROW ZeroDivisionError "Raised because the divisor self.limit is zero."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Compute a slice of the internal list for the given index by deriving the partition start and end positions from the current size and remainder values." get]
    [INPUTS]
        <REF> index </REF>: int "Zero-based partition index used to calculate the slice bounds."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "The sublist of self.lst from the computed start index up to, but not including, the computed end index."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.setNum() and unpack its two returned values into size and remainder. RESULT size_remainder]
            [COMMAND Compute start as index multiplied by size plus the smaller of index and remainder. RESULT start]
            [COMMAND Compute end as start plus size. RESULT end]
            [COMMAND If index plus 1 is less than or equal to remainder, increase end by 1 to include the extra element in the earlier partitions. RESULT end]
            [COMMAND Return the slice of self.lst from start up to, but not including, end. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
