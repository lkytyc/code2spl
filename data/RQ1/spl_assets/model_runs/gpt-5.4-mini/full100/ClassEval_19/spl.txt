# CLASS ChandrasekharSieve

[DEFINE_WORKER: "Initializes the object by storing the provided n value and computing and caching the primes collection." __init__]
    [INPUTS]
        <REF> n </REF>: data "Incoming value used to initialize the instance state and as input to prime generation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> self.n </REF>: data "Instance field set directly from the input n."
        <REF> self.primes </REF>: data "Instance field set to the result returned by generate_primes()."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming parameter n directly to the instance field self.n. RESULT self.n]
            [COMMAND Call self.generate_primes() to compute the primes collection, using the current instance state as needed. RESULT generated_primes]
            [COMMAND Assign the computed generated_primes value to the instance field self.primes. RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generate all prime numbers from 2 through self.n using a sieve, then collect and return the values still marked true." generate_primes]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If self.n is less than 2, return an empty list immediately. RESULT []]
            [COMMAND Create a boolean sieve list of length self.n + 1 with all entries initially set to True, then set sieve[0] and sieve[1] to False, and initialize p to 2. RESULT sieve, p]
            [COMMAND While p * p is less than or equal to self.n, check whether sieve[p] is True; if it is, mark every multiple i from p * p through self.n inclusive in steps of p as False, then increment p by 1. RESULT sieve, p]
            [COMMAND Initialize primes as an empty list. RESULT primes]
            [COMMAND For each i from 2 through self.n inclusive, if sieve[i] is True, append i to primes. RESULT primes]
            [COMMAND Return the primes list. RESULT primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current value of the instance attribute primes without any transformation." get_primes]
    [INPUTS]
        <REF> self </REF>: object "The instance whose primes attribute is being accessed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The current value stored in self.primes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.primes from the object instance. RESULT primes_value]
            [COMMAND Return primes_value directly to the caller. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
