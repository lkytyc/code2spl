[DEFINE_WORKER: "Performs piecewise linear interpolation for each target x value by finding the containing interval in the source x array and computing the corresponding y value." interpolate_1d]
    [INPUTS]
        <REF> x </REF>: sequence "Ordered source x-coordinates that define interpolation intervals."
        <REF> y </REF>: sequence "Source y-coordinates paired with x and used for linear interpolation."
        <REF> x_interp </REF>: sequence "Target x values to interpolate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A list of interpolated y values for the target x values that fall within at least one source interval."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect interpolated y values. RESULT y_interp]
            [COMMAND For each target value xi in x_interp, scan the adjacent index pairs in x from 0 to len(x) - 2. RESULT loop_xi_and_i]
            [COMMAND Within the inner scan for each xi, if x[i] <= xi <= x[i+1], compute yi by linear interpolation using y[i] + (y[i+1] - y[i]) * (xi - x[i]) / (x[i+1] - x[i]). RESULT yi]
            [COMMAND Within the same matched interval for xi, append the computed yi to y_interp. RESULT y_interp]
            [COMMAND Within the same matched interval for xi, break out of the inner loop so only the first matching interval is used. RESULT inner_loop_break]
            [COMMAND After processing all target values, return y_interp. RESULT y_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Perform bilinear interpolation for each paired target coordinate by locating the containing grid cell and computing the interpolated z value from its four corner values." interpolate_2d]
    [INPUTS]
        <REF> x </REF>: list "Ordered grid coordinates along the x axis."
        <REF> y </REF>: list "Ordered grid coordinates along the y axis."
        <REF> z </REF>: list "2D grid of corner values indexed by x and y positions."
        <REF> x_interp </REF>: list "Target x coordinates to interpolate."
        <REF> y_interp </REF>: list "Target y coordinates to interpolate."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "List of interpolated z values, one for each paired target coordinate that falls within a grid cell."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list to collect interpolated z values. RESULT z_interp]
            [COMMAND For each paired target coordinate (xi, yi) produced by zipping x_interp and y_interp, search for the first x interval that contains xi using the inclusive boundary check x[i] <= xi <= x[i+1]. RESULT matching_x_index]
            [COMMAND Within the selected x interval search, for each y interval index j from 0 to len(y) - 2, find the first interval that contains yi using the inclusive boundary check y[j] <= yi <= y[j+1]. RESULT matching_y_index]
            [COMMAND Within the matched x and y intervals, read the four surrounding corner values z00 = z[i][j], z01 = z[i][j+1], z10 = z[i+1][j], and z11 = z[i+1][j+1]. RESULT corner_values]
            [COMMAND Within the matched cell, compute the bilinear interpolation value zi using the weighted corner formula and the denominator (x[i+1] - x[i]) * (y[j+1] - y[j]). RESULT zi]
            [COMMAND Append the computed zi to z_interp and stop searching the current y interval list and the current x interval list for this point using the two breaks. RESULT z_interp]
            [COMMAND After all paired target coordinates have been processed, return z_interp. RESULT z_interp]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
