# CLASS UrlPath

[DEFINE_WORKER: "Initialize the object with an empty segments list and disable the end tag flag by default." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty list to self.segments to initialize the segment collection. RESULT self.segments]
            [COMMAND Assign False to self.with_end_tag to disable end-tag handling by default. RESULT self.with_end_tag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Appends a fixed or normalized version of the given segment to the segments collection." add]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the segments collection and the path-fixing helper."
        <REF> segment </REF>: any "The segment value to be normalized and appended."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.fix_path with the provided segment to compute its normalized form. RESULT fixed_segment]
            [COMMAND Append the normalized segment to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalize a non-empty path, record whether it ends with a slash, split it into segments, URL-decode each segment, and append the decoded values to self.segments." parse]
    [INPUTS]
        <REF> path </REF>: string "Input path to normalize, inspect for a trailing slash, and split into segments."
        <REF> charset </REF>: string "Character encoding used when URL-decoding each path segment."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If path is truthy, continue with path processing; otherwise skip all remaining steps. RESULT path_guard]
            [COMMAND Within the truthy path branch, if path ends with '/', set self.with_end_tag to True. RESULT self.with_end_tag]
            [COMMAND Within the truthy path branch, replace path with the result of self.fix_path(path). RESULT path]
            [COMMAND Within the truthy path branch after normalization, if path is still truthy, split it on '/' and store the segments. RESULT split]
            [COMMAND Within the normalized non-empty path branch, for each seg in split, decode seg with urllib.parse.unquote using encoding=charset. RESULT decoded_seg]
            [COMMAND Within the normalized non-empty path branch, append each decoded segment to self.segments. RESULT self.segments]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Return an empty string for a falsy path, otherwise return the path with leading and trailing slashes removed." fix_path]
    [INPUTS]
        <REF> path </REF>: string "Input path string to normalize."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The normalized path without leading or trailing slash characters, or an empty string when the input path is falsy."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the input path is falsy. RESULT path_truthy_check]
            [COMMAND If the path is falsy, return an empty string immediately. RESULT '']
            [COMMAND Otherwise, remove any leading and trailing '/' characters from the path by calling strip('/'). RESULT segment_str]
            [COMMAND Return the normalized segment string. RESULT segment_str]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The input path is falsy, including an empty string or other false-like value.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty string without performing any stripping operation. RESULT '']
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
