    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any | bool | None "The first track in the playlist, or False if the playlist is non-empty but no current song is set, or None implicitly if the playlist is empty"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.playlist is truthy (non-empty) AND self.current_song is truthy (non-None/non-falsy); both conditions must hold simultaneously RESULT branch_condition_1]
            [COMMAND If branch_condition_1 is true, return self.playlist[0] — the first track; the playlist is confirmed non-empty so the index is safe; exit the method immediately RESULT early_return_first_track]
            [COMMAND If branch_condition_1 is false, evaluate the elif: check whether len(self.playlist) is truthy (equivalent to self.playlist being non-empty); this branch is reached only when self.playlist is empty OR self.current_song is falsy RESULT branch_condition_2]
            [COMMAND If branch_condition_2 is true, the playlist has items but self.current_song must be falsy (the only way to reach here with a non-empty playlist); return False as a sentinel indicating no current song is set despite tracks being available; exit the method immediately RESULT early_return_false_sentinel]
            [COMMAND If both conditions are false (playlist is empty regardless of current_song), no branch executes and the method falls through, returning None implicitly RESULT implicit_return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.playlist is non-empty but self.current_song is falsy — playlist has content yet no active song is set]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the first if-branch because self.current_song fails the truthy check RESULT skip_first_branch]
            [COMMAND Enter the elif branch because len(self.playlist) is truthy; return False to signal the missing current song RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.playlist is empty regardless of the value of self.current_song]
        [SEQUENTIAL_BLOCK]

        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[DEFINE_WORKER: "Advances the playlist to the next track by verifying a current song is set and is not the last entry, updating the current song reference and returning True on success, or returning False if no song is active or the end of the playlist has been reached." switch_song]
    [INPUTS]
        <REF> self </REF>: object "Instance carrying current_song (the active track or a falsy value) and playlist (an ordered list of tracks)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the playlist was successfully advanced to the next track; False if there is no active song or the current song is already the last entry"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.current_song is truthy; if it is falsy, skip to the outer else branch RESULT current_song_exists]
            [COMMAND Call self.playlist.index(self.current_song) to obtain the zero-based position of the current song in the playlist; assign the result to current_index RESULT current_index]
            [COMMAND Compare current_index against len(self.playlist) - 1 using a strict less-than check; this boundary is exclusive of the last element, so the condition is true only when at least one more track follows RESULT has_next_song]
            [COMMAND If has_next_song is true, assign self.playlist[current_index + 1] to self.current_song, mutating instance state to point to the immediately following track RESULT self.current_song]
            [COMMAND Return True to signal that the playlist was successfully advanced RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: current_song is truthy but current_index equals len(self.playlist) - 1, meaning the active song is the last entry in the playlist]
        [SEQUENTIAL_BLOCK]
            [COMMAND Leave self.current_song unchanged; no state mutation occurs RESULT no_state_change]
            [COMMAND Return False to indicate the end of the playlist has been reached RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.current_song is falsy (None, empty string, or any other falsy value), meaning no song is currently active]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the entire inner block without reading or modifying self.playlist; no state mutation occurs RESULT no_state_change]
            [COMMAND Return False to indicate that no current song is set RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.current_song is truthy but is not present in self.playlist when self.playlist.index(self.current_song) is called, which can occur if the playlist is modified externally after current_song was set]
        [LOG "ValueError raised by list.index because current_song is not found in playlist; the method does not catch this exception and it propagates to the caller"]
        [THROW ValueError "Raised by the built-in list.index call when self.current_song cannot be located in self.playlist; no fallback or return value is provided by this method"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

            [COMMAND Within the outer truthy-song branch, check whether current_index is strictly greater than 0 RESULT can_go_previous]
            [COMMAND Within the outer truthy-song branch and the inner current_index > 0 branch, assign self.playlist[current_index - 1] to self.current_song RESULT self.current_song]

    [END_MAIN_FLOW]

        [END_SEQUENTIAL_BLOCK]
