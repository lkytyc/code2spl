[DEFINE_WORKER: "Recursively computes the palindrome radius from a given center index by expanding outward one step at a time, returning 0 when either boundary is exceeded or a character mismatch is found." palindromic_length]
    [INPUTS]
        <REF> center </REF>: int "Index of the palindrome center character in the string"
        <REF> diff </REF>: int "Current expansion offset from the center; starts at 1 for the first outward comparison"
        <REF> string </REF>: str "The input string being examined for palindromic substrings"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The number of additional matching character pairs found from the current diff offset outward; 0 when expansion is invalid or characters do not match"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the guard condition: check whether (center - diff) equals -1 (left index has gone one past the start of the string), OR (center + diff) equals len(string) (right index has gone one past the last valid index), OR the character at position (center - diff) does not equal the character at position (center + diff); short-circuit left-to-right so boundary checks are evaluated before the character access RESULT guard_triggered]
            [COMMAND If guard_triggered is true, return the integer 0 immediately as an early exit, signaling that no further palindrome extension is possible from this invocation RESULT early_return_value]
            [COMMAND If guard_triggered is false, recursively call palindromic_length with the same center, diff incremented by exactly 1, and the same string, to check the next outward character pair RESULT recursive_result]
            [COMMAND Return 1 plus recursive_result, accumulating one valid matching pair for the current diff offset RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Left boundary exceeded: (center - diff) equals -1, meaning expansion has gone one step past index 0; this sub-condition is evaluated first due to short-circuit or-logic and prevents the character access from being reached]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without evaluating the remaining sub-conditions or accessing any string index RESULT boundary_left_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Right boundary exceeded: (center + diff) equals len(string), meaning expansion has reached one past the last valid index; evaluated second if the left boundary check was false]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without evaluating the character comparison sub-condition RESULT boundary_right_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Character mismatch: both boundary checks are false (indices are in bounds), but string[center - diff] does not equal string[center + diff]; palindrome expansion fails at this offset]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately, indicating the palindrome cannot be extended further in this direction RESULT mismatch_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Finds the longest palindromic substring in the instance's input string by inserting pipe delimiters between characters to normalize center detection, scanning every index as a candidate center, then extracting and returning the clean substring with delimiters removed." palindromic_string]
    [INPUTS]
        <REF> self.input_string </REF>: string "The source string stored on the instance from which the longest palindrome is to be found; must have at least one character"
    [END_INPUTS]

    [OUTPUTS]
        <REF> output_string </REF>: string "The longest palindromic substring extracted from the original input string, with no pipe delimiter characters"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize max_length to the integer 0, new_input_string to an empty string, and output_string to an empty string RESULT max_length, new_input_string, output_string]
            [COMMAND Iterate over every character i in self.input_string from index 0 up to but not including the last character (slice [:len-1]), and for each character append that character followed by a pipe '|' to new_input_string; after the loop append the final character of self.input_string (index -1) to new_input_string without a trailing pipe, producing a pipe-delimited version of the original string RESULT new_input_string]
            [COMMAND Iterate over every index i in the range 0 to len(new_input_string) - 1 inclusive; for each index call self.palindromic_length(i, 1, new_input_string) and assign the return value to length; then evaluate the condition max_length < length — if strictly true, update max_length to length and set start to i, recording the center index of the longest palindrome found so far; ties are broken in favor of the first occurrence because the comparison is strict less-than RESULT max_length, start]
            [COMMAND Iterate over each character i in the slice of new_input_string from index start - max_length to start + max_length inclusive (upper bound start + max_length + 1 in Python slice notation, with Python's standard silent clamping applied if either bound is out of range); for each character, if i is not equal to the pipe character '|', append i to output_string RESULT output_string]
            [COMMAND Return output_string as the result of the method RESULT output_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
