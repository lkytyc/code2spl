# CLASS TicTacToe

[DEFINE_WORKER: "Initializes a TicTacToe game instance by creating a 3×N board filled with empty spaces and setting the first player to 'X'." __init__]
    [INPUTS]
        <REF> N </REF>: int "Number of columns in the board; defaults to 3 if not provided"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a 2D list named self.board with exactly 3 rows and N columns, where each cell is initialized to a single space character ' '; each row must be an independent list object constructed via a separate inner list comprehension to avoid shared references RESULT self.board]
            [COMMAND Set self.current_player to the string literal 'X', unconditionally establishing 'X' as the first mover regardless of any other state RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: N is 0 at call time]
        [SEQUENTIAL_BLOCK]
            [COMMAND The inner list comprehension iterates zero times, producing an empty list for each row; self.board is assigned a list of 3 empty row lists with no cells RESULT self.board]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: N is not defined or not passed and no default is resolved, raising NameError]
        [LOG "NameError is raised during board construction because N cannot be resolved in the inner list comprehension scope"]
        [THROW NameError "Raised if N is somehow unresolvable at the time the inner comprehension evaluates range(N)"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates and applies a player's move to the board at the given position, toggling the active player on success and returning a boolean indicating whether the move was accepted." make_move]
    [INPUTS]
        <REF> row </REF>: int "Row index of the target cell on the board"
        <REF> col </REF>: int "Column index of the target cell on the board"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was accepted and applied; False if the cell was already occupied and no state was changed"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the cell at board[row][col] is equal to a single space character ' ', indicating it is unoccupied RESULT cell_is_empty]
            [COMMAND If cell_is_empty is true, assign the current player's marker to board[row][col] RESULT board_updated]
            [COMMAND Toggle current_player: if current_player equals 'X' set it to 'O', otherwise set it to 'X'; this toggle occurs strictly after the board has been updated, never before RESULT player_switched]
            [COMMAND Return True to indicate the move was successfully accepted and applied RESULT move_accepted]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cell at board[row][col] is not a space character, meaning it is already occupied by 'X' or 'O' or any other non-space value]
        [SEQUENTIAL_BLOCK]
            [COMMAND Leave all state unchanged: do not modify board and do not toggle current_player RESULT state_preserved]
            [COMMAND Return False to indicate the move was rejected RESULT move_rejected]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans all eight win lines of a 3x3 tic-tac-toe board (three rows, three columns, two diagonals) and returns the winning player's token if any line is fully occupied by the same non-space mark, or None if no winner exists yet." check_winner]
    [INPUTS]
        <REF> self </REF>: object "Instance of the game class; provides access to self.board, a 3x3 list-of-lists where each cell holds 'X', 'O', or ' '"
    [END_INPUTS]

    [OUTPUTS]
        <REF> winner </REF>: string or None "The winning player's token ('X' or 'O') if a winning line is found; None if no winner exists"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board; for each row evaluate the chained comparison row[0] == row[1] == row[2] != ' ' (equivalent to: row[0] equals row[1], AND row[1] equals row[2], AND row[2] is not a space); if the condition holds, immediately return row[0] as the winning token (early exit) RESULT winner_from_row_or_continue]
            [COMMAND Iterate over column indices 0, 1, 2 via range(3); for each col evaluate the chained comparison self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ' (equivalent to: board[0][col] equals board[1][col], AND board[1][col] equals board[2][col], AND board[2][col] is not a space); if the condition holds, immediately return self.board[0][col] as the winning token (early exit) RESULT winner_from_col_or_continue]
            [COMMAND Check the main diagonal by evaluating self.board[0][0] == self.board[1][1] == self.board[2][2] != ' '; if true, immediately return self.board[0][0] as the winning token (early exit) RESULT winner_from_main_diagonal_or_continue]
            [COMMAND Check the anti-diagonal by evaluating self.board[0][2] == self.board[1][1] == self.board[2][0] != ' '; if true, immediately return self.board[0][2] as the winning token (early exit) RESULT winner_from_anti_diagonal_or_continue]
            [COMMAND If none of the eight win-line checks triggered an early return, return None to indicate no winner has been found RESULT none_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A row win is found before any column or diagonal check is reached]
        [SEQUENTIAL_BLOCK]
            [COMMAND During the row-scanning loop, the first row satisfying row[0] == row[1] == row[2] != ' ' causes an immediate return of row[0]; the column loop, diagonal checks, and final return None are never executed RESULT early_return_row_winner]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A column win is found before the diagonal checks are reached]
        [SEQUENTIAL_BLOCK]
            [COMMAND During the column-scanning loop, the first column index satisfying self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ' causes an immediate return of self.board[0][col]; remaining column indices and both diagonal checks are never evaluated RESULT early_return_col_winner]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans every row of the game board and returns False immediately upon finding any empty cell (space character), or True if no row contains a space, indicating the board is completely filled." is_board_full]
    [INPUTS]
        <REF> self </REF>: object "Instance of the game class providing access to self.board, a 2D iterable where each row is a list or string of cell values"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "False if at least one empty cell (space character) exists anywhere on the board; True if every cell in every row is occupied"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each row in self.board, binding the current row to the variable row on each iteration RESULT current_row]
            [COMMAND Within the for loop, check whether a space character (' ') is present anywhere in current_row using the membership operator, which short-circuits on the first match found RESULT has_empty_cell]
            [COMMAND Within the for loop, if has_empty_cell is True, immediately return False from the method, exiting both the loop and the method without inspecting any remaining rows or cells RESULT early_return_false]
            [COMMAND After the for loop completes without triggering an early return, return True to indicate that no empty cell was found in any row and the board is fully occupied RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The first cell of the first row is a space character, triggering the early-exit path at best-case O(1)]
        [SEQUENTIAL_BLOCK]
            [COMMAND On the very first membership check, has_empty_cell is True; return False immediately without iterating further RESULT early_return_false_best_case]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Every cell in every row is non-space, so the for loop exhausts all rows without ever entering the if branch]
        [SEQUENTIAL_BLOCK]
            [COMMAND The for loop completes all O(m x n) membership checks with no early return triggered; fall through to the post-loop return True statement RESULT return_true_worst_case]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
