# CLASS BookManagementDB

[DEFINE_WORKER: "Initializes a database connection and cursor from the given database name, then delegates table creation to a setup method." __init__]
    [INPUTS]
        <REF> db_name </REF>: str "File path or ':memory:' identifier for the SQLite database to open"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a SQLite connection to the database identified by db_name and assign the resulting Connection object to self.connection RESULT self.connection]
            [COMMAND Create a Cursor object from self.connection and assign it to self.cursor, binding the cursor to the same database context established by self.connection RESULT self.cursor]
            [COMMAND Call self.create_table() to perform initial table setup using the open connection and cursor RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: sqlite3.connect(db_name) raises sqlite3.OperationalError because db_name is invalid or the path is inaccessible]
        [LOG "Connection could not be established for the given db_name; self.connection and self.cursor are never assigned and the instance is left in an unusable state"]
        [THROW sqlite3.OperationalError "Propagates unhandled if the database file cannot be opened or created"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Ensures the 'books' table exists in the SQLite database by executing an idempotent DDL statement and committing the transaction." create_table]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL DDL statement 'CREATE TABLE IF NOT EXISTS books (id INTEGER PRIMARY KEY, title TEXT, author TEXT, available INTEGER)' against the database using the pre-initialized cursor object; discard the return value RESULT _execute_result_discarded]
            [COMMAND Commit the current transaction on the connection object to durably persist the DDL change to disk RESULT _commit_result_discarded]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.cursor is None or has been closed before execute() is called]
        [LOG "AttributeError or sqlite3.ProgrammingError raised because the cursor object is invalid or closed"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.connection is None or has been closed before commit() is called]
        [LOG "AttributeError or sqlite3.ProgrammingError raised because the connection object is invalid or closed"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Inserts a new book record into the books table with the given title and author, setting available to 1, and commits the transaction." add_book]
    [INPUTS]
        <REF> title </REF>: string "The title of the book to insert; passed directly to the SQL driver without validation or transformation"
        <REF> author </REF>: string "The author of the book to insert; passed directly to the SQL driver without validation or transformation"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized INSERT statement against the books table, binding title to the first placeholder and author to the second placeholder, with the available column hardcoded to the integer literal 1 RESULT execute_result]
            [COMMAND Commit the current database transaction unconditionally, persisting the inserted row to disk RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.cursor.execute() raises an exception, for example due to a schema constraint violation, a None value in a NOT NULL column, or an invalid database state]
        [LOG "execute() failed before commit() was reached; the transaction is left in an uncommitted state with no rollback performed, which may leave the connection in a dirty transaction state"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Deletes the record with the given ID from the books table and commits the transaction to persist the change." remove_book]
    [INPUTS]
        <REF> book_id </REF>: integer "Primary key of the book record to delete from the books table"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized DELETE SQL statement against the books table, binding book_id as the value for the WHERE id = ? placeholder via the instance cursor RESULT execute_result]
            [COMMAND Commit the open transaction on the instance connection to durably persist the DELETE operation to the database RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: book_id does not match any row in the books table]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute completes with zero rows affected and no error is raised; proceed to commit RESULT execute_result]
            [COMMAND Commit the transaction; the commit is effectively a no-op for this statement RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.cursor is closed or None when execute is called]
        [LOG "cursor.execute raises an exception; commit is never reached and the transaction remains open or is implicitly rolled back by the DB-API driver"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: An exception occurs after execute but before commit completes]
        [LOG "The DELETE is staged but not persisted; depending on the DB-API 2.0 driver the transaction is left open or implicitly rolled back, and no rollback call is made explicitly"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as unavailable in the database by setting its available flag to 0 for the given book ID and committing the transaction." borrow_book]
    [INPUTS]
        <REF> book_id </REF>: any "The identifier of the book to be borrowed; used as the WHERE clause parameter to match the target row in the books table"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL UPDATE statement against the books table, binding book_id as the sole dynamic parameter to set available = 0 for the row where id equals book_id RESULT sql_execute_result]
            [COMMAND Commit the open database transaction to persist the UPDATE to durable storage RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The database connection is in a failed transaction state when execute is called, book_id type is incompatible with the id column, a concurrent write lock is held on the target row, or an integrity constraint on the available column rejects the value 0]
        [LOG "Database operation failed during borrow_book: either the execute call raised due to a bad transaction state, type mismatch, lock contention, or constraint violation, preventing commit from being reached"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Marks a book as available in the database by setting its available flag to 1 for the given book ID, then commits the transaction." return_book]
    [INPUTS]
        <REF> book_id </REF>: any "The identifier of the book to mark as available; used as the WHERE clause parameter in the UPDATE statement"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute a parameterized SQL UPDATE statement against the books table, binding book_id to the ? placeholder, to set the available column to the integer value 1 for the row where id equals book_id RESULT execute_result]
            [COMMAND Commit the open database transaction on self.connection to durably persist the UPDATE to storage RESULT commit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.cursor or self.connection is closed, stale, or otherwise invalid at call time, causing cursor.execute to raise a database exception]
        [LOG "Exception propagates up from cursor.execute; self.connection.commit is never reached and the transaction remains open for the caller to handle or roll back"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: cursor.execute succeeds but self.connection.commit raises a database exception]
        [LOG "Exception propagates up from connection.commit; the UPDATE may or may not be persisted depending on the database engine state"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Executes a SELECT query to retrieve all rows from the books table and returns them as a list." search_books]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> books </REF>: list "All rows from the books table as a list of tuples; returns an empty list if no rows exist"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Execute the SQL query 'SELECT * FROM books' against the database using the active cursor RESULT _cursor_executed]
            [COMMAND Fetch all rows returned by the executed query from the cursor into the local variable books; result is an empty list if the query matched no rows RESULT books]
            [COMMAND Return books to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The query returns no matching rows]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign an empty list to books as the fetchall result RESULT books]
            [COMMAND Return the empty list to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
