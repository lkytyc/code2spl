[DEFINE_WORKER: "Converts a key to camelCase if it is a string, otherwise returns the key unchanged." ConvertKey]
    [INPUTS]
        <REF> key </REF>: any "The key to convert; may be a string or any other type such as an integer"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "The camelCase-converted string if key was a string, or the original key value for all other types"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether key is an instance of str RESULT is_string]
            [COMMAND If is_string is true, call _to_camel_case with key and return the result immediately as an early exit RESULT camel_case_result]
            [COMMAND If is_string is false, return key unchanged RESULT original_key]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: key is an instance of str (inclusive of str subclasses, exclusive of all other types such as int, None, bytes, or list)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call _to_camel_case passing key as the argument RESULT camel_case_result]
            [COMMAND Return camel_case_result immediately, short-circuiting any remaining logic in the enclosing method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a snake_case string key into camelCase by splitting on underscores, capitalizing the first letter of each subsequent part, and joining all parts together." _to_camel_case]
    [INPUTS]
        <REF> key </REF>: string "A snake_case string to be converted to camelCase; expected to be a non-null string, though boundary cases like empty string or leading/trailing underscores are handled by the split behavior"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The camelCase version of the input string; the first segment is lowercased as-is and each subsequent segment has its first character uppercased"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split key on the underscore character '_' to produce a list of string segments, assigning the result to parts; if key contains no underscore the list has one element equal to key; if key starts or ends with '_' the list includes empty strings at head or tail respectively RESULT parts]
            [COMMAND Take the first element of parts (index 0) as the leading segment, preserving its existing casing without any transformation RESULT head]
            [COMMAND Iterate over all elements of parts starting from index 1 to the end; for each element call its title() method, which uppercases the first character and lowercases the rest, collecting all transformed segments into a sequence RESULT tail_segments]
            [COMMAND Join tail_segments into a single string with an empty separator, producing the concatenated tail RESULT tail]
            [COMMAND Concatenate head and tail into one string and return it as the camelCase result RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: key contains no underscore character, so parts has exactly one element]
        [SEQUENTIAL_BLOCK]
            [COMMAND parts[0] equals key itself; the tail_segments iterable is empty; the joined tail is an empty string; return key unchanged RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: key is an empty string]
        [SEQUENTIAL_BLOCK]
            [COMMAND split('_') on an empty string yields [''], so parts[0] is an empty string, tail is empty, and the return value is an empty string RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: key starts or ends with '_', or contains consecutive underscores, producing empty-string elements in parts]
        [SEQUENTIAL_BLOCK]
            [COMMAND Empty-string segments in the tail position have their title() called, returning an empty string, so those segments contribute nothing to the camelCase output; an empty-string head means the result starts with whatever the first non-empty tail segment's title() produces RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
