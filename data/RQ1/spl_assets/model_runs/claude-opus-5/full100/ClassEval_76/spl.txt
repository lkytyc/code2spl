# CLASS SignInSystem

[DEFINE_WORKER: "Initializes the object by setting up an empty dictionary to store users." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the 'users' attribute on the current instance as an empty dictionary RESULT self.users]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a new username to the users dictionary with an initial value of False if it does not already exist, returning True on success or False if the username is already registered." add_user]
    [INPUTS]
        <REF> username </REF>: string "The username to be registered into the users dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username was successfully added; False if the username already existed and no change was made"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username already exists as a key in self.users RESULT username_exists]
            [COMMAND If username_exists is True, return False immediately as an early exit indicating the username is already registered RESULT early_return_false]
            [COMMAND If username_exists is False, assign self.users[username] the value False to register the username with an initial inactive or unverified state RESULT registration_entry]
            [COMMAND Return True to indicate the username was successfully added RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is already present as a key in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying self.users, signaling that registration was rejected due to a duplicate username RESULT duplicate_rejection]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to sign in a user by verifying their existence in the users dictionary and marking them as active, returning true on success or false if the user is not registered." sign_in]
    [INPUTS]
        <REF> username </REF>: string "The username to look up and activate in the users dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username existed and was successfully marked active; false if the username was not found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is absent from the self.users dictionary RESULT user_exists_check]
            [COMMAND If username is not in self.users, return False immediately to the caller RESULT early_return_false]
            [COMMAND If username is present in self.users, set self.users[username] to True RESULT user_activated]
            [COMMAND Return True to indicate the sign-in operation succeeded RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not a key in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without modifying any state, signalling that the username is unregistered RESULT sign_in_rejected]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a given username is registered and active by verifying its presence in the users dictionary and evaluating its stored value as a boolean flag." check_sign_in]
    [INPUTS]
        <REF> username </REF>: string "The username to look up in the users dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username exists and its stored value is truthy; False if the username is absent or its stored value is falsy"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is absent from the self.users dictionary using a membership test RESULT membership_check]
            [COMMAND If username is not in self.users, return False immediately — the user is unknown and no further evaluation is needed RESULT early_return_unknown]
            [COMMAND If username is present in self.users, evaluate self.users[username] as a boolean via truthiness — any truthy value (non-empty string, non-zero int, any object) yields True; any falsy value (None, 0, empty string, False, empty list) yields False RESULT value_truthiness_check]
            [COMMAND If self.users[username] is truthy, return True — the user exists and is considered active or enabled RESULT return_true]
            [COMMAND If self.users[username] is falsy, return False — the user exists but is considered inactive or disabled RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not a key in self.users]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False without accessing the dictionary value, short-circuiting all further evaluation RESULT short_circuit_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: username is present in self.users but its associated value is falsy]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to signal that the account exists but is not in an active or enabled state RESULT disabled_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns true if every value in the users dictionary is truthy, indicating all users are signed in, and false if any value is falsy or the dictionary is empty." all_signed_in]
    [INPUTS]
        <REF> self </REF>: object "Instance reference providing access to the users dictionary attribute"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if all values in self.users are truthy; false if any value is falsy. Note: returns True vacuously when self.users is empty."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate all(self.users.values()) to check whether every value in the users dictionary is truthy RESULT condition_result]
            [COMMAND If condition_result is True, return True immediately RESULT return_true]
            [COMMAND Otherwise return False immediately RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.users is an empty dictionary, causing all() to return True vacuously]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True without evaluating any user values, as all([]) is vacuously true RESULT vacuous_true]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over all users in the instance's user dictionary and returns a list of usernames whose signed-in status is falsy." all_not_signed_in]
    [INPUTS]
        <REF> self </REF>: object "The instance owning the users dictionary, where keys are usernames and values are boolean sign-in statuses"
    [END_INPUTS]

    [OUTPUTS]
        <REF> not_signed_in_users </REF>: list "A flat list of username strings for every user whose signed-in value is falsy"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named not_signed_in_users to accumulate results RESULT not_signed_in_users]
            [COMMAND Iterate over every key-value pair in self.users using items(), binding each key to username and each value to signed_in; visit all entries in insertion order without early exit RESULT loop_iteration]
            [COMMAND Within the for loop over self.users.items(), for each iteration evaluate whether signed_in is falsy using 'not signed_in'; if the condition is True, append the current username to not_signed_in_users; if the condition is False, skip the current iteration silently with no else branch RESULT not_signed_in_users]
            [COMMAND After the loop completes, return not_signed_in_users as the result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
