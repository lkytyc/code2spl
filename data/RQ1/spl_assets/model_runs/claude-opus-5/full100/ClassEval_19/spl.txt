# CLASS ChandrasekharSieve

[DEFINE_WORKER: "Initializes a prime-number generator instance by storing the requested count and eagerly computing the full list of primes via generate_primes." __init__]
    [INPUTS]
        <REF> n </REF>: int "The number of primes (or upper bound) to generate; must be supplied by the caller — no default is defined"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the constructor argument n to the instance attribute self.n, binding the caller-supplied value to the object for its lifetime RESULT self.n]
            [COMMAND Call self.generate_primes(), which reads self.n to determine how many primes to produce, and assign the returned collection to self.primes, making it available for the lifetime of the instance RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n is 0 or negative]
        [SEQUENTIAL_BLOCK]
            [COMMAND generate_primes() receives a zero or negative self.n and returns an empty collection RESULT self.primes_empty]
            [COMMAND Assign the empty collection to self.primes RESULT self.primes]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates all prime numbers up to and including self.n using the Sieve of Eratosthenes algorithm, returning them as an ordered list." generate_primes]
    [INPUTS]
        <REF> self.n </REF>: int "Upper bound (inclusive) for prime generation; must be >= 2 to produce any results"
    [END_INPUTS]

    [OUTPUTS]
        <REF> primes </REF>: list[int] "Ordered list of all prime numbers from 2 through self.n inclusive, or an empty list if self.n < 2"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a boolean list named sieve of length self.n + 1 with all elements set to True, then immediately set sieve[0] and sieve[1] to False via chained assignment to mark 0 and 1 as non-prime RESULT sieve]
            [COMMAND Set integer variable p to the constant 2 as the seed candidate for the outer sieve loop RESULT p]
            [COMMAND While p * p <= self.n (inclusive boundary, stopping at floor of sqrt(self.n)): if sieve[p] is True, iterate i over range(p * p, self.n + 1, p) and for each i set sieve[i] to False, marking all multiples of p from p-squared upward as composite; unconditionally increment p by 1 after the conditional block and repeat RESULT sieve]
            [COMMAND Initialize an empty list named primes to accumulate confirmed prime values RESULT primes]
            [COMMAND Iterate i over range(2, self.n + 1) inclusive; for each i where sieve[i] is True, append i to primes RESULT primes]
            [COMMAND Return primes RESULT primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.n is less than 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately without constructing the sieve or executing any loop RESULT []]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the list of prime numbers stored in the instance's primes attribute." GetPrimes]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> primes </REF>: list "The list of prime numbers held by the current instance"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value of the primes attribute from the current instance (self.primes) as the result of this method RESULT primes]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
