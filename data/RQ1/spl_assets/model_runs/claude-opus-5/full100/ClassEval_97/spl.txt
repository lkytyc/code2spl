    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.numwords as an empty dict to serve as the central registry mapping word strings to (multiplier, offset) tuples RESULT self.numwords]
            [COMMAND Initialize self.units as a list of 20 strings representing zero through nineteen, index-aligned so units[0] == 'zero' and units[19] == 'nineteen' RESULT self.units]
            [COMMAND Initialize self.tens as a list of 10 strings where indices 0 and 1 are empty string placeholders and meaningful values begin at index 2 with 'twenty' through index 9 with 'ninety' RESULT self.tens]
            [COMMAND Initialize self.scales as a list of 5 strings: ['hundred', 'thousand', 'million', 'billion', 'trillion'], indexed 0 through 4 RESULT self.scales]

[DEFINE_WORKER: "Converts a string containing spelled-out number words (including ordinal forms) into a string where those numeric spans are replaced by their integer equivalents, leaving non-numeric words in place." text2int]
    [INPUTS]
        <REF> textnum </REF>: string "Input string potentially containing spelled-out numbers, ordinal words, ordinal-suffixed tokens (e.g. '21st'), and arbitrary non-numeric text, with hyphenated number forms permitted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> curstring </REF>: string "Reconstructed string in which each contiguous span of recognized number words has been replaced by its integer value as a string literal; non-numeric words are preserved with a trailing space, and the final numeric value (if the string ends mid-number) is appended without a trailing space."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace every hyphen character in textnum with a space so that hyphenated compound number words such as 'twenty-one' are treated as two separate tokens during subsequent splitting. RESULT textnum]
            [COMMAND Initialize numeric accumulators: set current to 0 and result to 0 via chained assignment. Initialize curstring to the empty string as the output buffer. Initialize onnumber to False as the boolean mode flag tracking whether the parser is currently inside a numeric word sequence. RESULT current, result, curstring, onnumber]
            [COMMAND Split textnum on whitespace to produce a list of tokens, then iterate over each token as word. RESULT word]
            [COMMAND Within the outer for loop: check whether word is a key in self.ordinal_words (the table of pure ordinal words such as 'first', 'third'). If it is, retrieve the corresponding value as increment, set scale to 1, update current by computing current * scale + increment (which simplifies to current + increment because scale is always 1 in this branch), and set onnumber to True. Then advance to the next token. RESULT current, onnumber]
            [COMMAND Within the outer for loop, in the else branch (word was not found in self.ordinal_words): iterate over every (ending, replacement) pair in self.ordinal_endings. For each pair, if word ends with that ending, rewrite word by stripping the ending suffix and appending replacement in its place. No break is issued, so all pairs are tested; in practice the endings are mutually exclusive. This normalizes tokens such as '21st' into a form that can be looked up in self.numwords. RESULT word]
            [COMMAND Still within the else branch, after the ordinal-suffix normalization loop: check whether the (possibly rewritten) word is absent from self.numwords. If it is absent, the token is non-numeric text: if onnumber is True, flush the accumulated number by appending repr(result + current) followed by a space to curstring; then append word followed by a space to curstring; reset result to 0, current to 0, and onnumber to False. Then advance to the next token. RESULT curstring, result, current, onnumber]
            [COMMAND Still within the else branch, in the inner else (word was found in self.numwords): retrieve (scale, increment) from self.numwords[word]. Apply the scale-and-increment accumulation: update current by computing current * scale + increment. If scale is greater than 100 (indicating a large-magnitude word such as 'thousand' or 'million'), add current into result and reset current to 0. Set onnumber to True. Then advance to the next token. RESULT current, result, onnumber]
            [COMMAND After the outer for loop ends: if onnumber is True (the string ended while a numeric sequence was still open), append repr(result + current) to curstring with no trailing space. RESULT curstring]
            [COMMAND Return curstring as the final transformed string. RESULT curstring]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates whether a text string consists entirely of recognizable number words by normalizing hyphens to spaces, skipping known ordinal words, attempting to strip ordinal suffixes from remaining tokens, and returning false immediately if any token cannot be resolved to a known number word." is_valid_input]
    [INPUTS]
        <REF> self </REF>: object "Instance providing ordinal_words (set of recognized ordinal word strings), ordinal_endings (list of (ending, replacement) tuples for suffix normalization), and numwords (set or mapping of recognized number word strings)"
        <REF> textnum </REF>: string "The text representation of a number to validate, potentially containing hyphen-separated tokens such as 'one-hundred-twenty'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if every token in the normalized input resolves to a recognized number word or ordinal word; False if any token fails resolution"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace all occurrences of '-' with ' ' in textnum and rebind textnum to the resulting string, unconditionally normalizing hyphen-separated tokens into space-separated tokens RESULT textnum]
            [COMMAND Split textnum on whitespace to produce the list of tokens to validate, then begin iterating over each token as 'word' RESULT word_iterator]
            [COMMAND For each word in the token list: check whether word is a member of self.ordinal_words; if it is, skip all remaining processing for this word and advance to the next token via continue RESULT ordinal_skip_decision]
            [COMMAND For each word not in self.ordinal_words: iterate over every (ending, replacement) pair in self.ordinal_endings; for each pair, check whether word ends with that ending string RESULT suffix_match_decision]
            [COMMAND Within the suffix-matching inner loop, if word ends with the current ending, rewrite word to the concatenation of word[:-len(ending)] and replacement, then continue iterating over the remaining endings without breaking (soft first-match: the rewritten word may still be tested against further endings) RESULT word]
            [COMMAND After the inner endings loop completes for the current word (whether or not any suffix was matched and word was rewritten), check whether the current value of word is absent from self.numwords; if it is absent, immediately return False, terminating the entire function RESULT early_return_false]
            [COMMAND If all tokens have been processed without triggering an early return, return True to indicate the entire input is composed of valid number words RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: textnum contains no hyphens at the replace step]
        [SEQUENTIAL_BLOCK]
            [COMMAND str.replace returns the original string unchanged; textnum is rebound to an equal string and processing continues normally with no effect on subsequent token splitting RESULT textnum]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: textnum is empty or contains only whitespace so that split() produces an empty list]
        [SEQUENTIAL_BLOCK]
            [COMMAND The outer for loop body never executes; control falls through directly to the return True statement at the end of the method RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A word matches an ordinal ending and word is rewritten, but the rewritten word also ends with a later ending in self.ordinal_endings]
        [SEQUENTIAL_BLOCK]
            [COMMAND The inner loop continues past the first match without a break, so the rewritten word is tested against subsequent endings; if another ending matches, word is rewritten again; the final value of word after the loop completes is then checked against self.numwords RESULT word]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
