# CLASS BigNumCalculator

[DEFINE_WORKER: "Adds two non-negative integers represented as decimal strings and returns their sum as a decimal string, handling different lengths and carry propagation." add]
    [INPUTS]
        <REF> num1 </REF>: string "First non-negative integer represented as a decimal string"
        <REF> num2 </REF>: string "Second non-negative integer represented as a decimal string"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The sum of num1 and num2 represented as a decimal string"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute max_length as the maximum of the lengths of num1 and num2 RESULT max_length]
            [COMMAND Left-pad num1 with '0' characters using zfill so its length equals max_length; rebind the name num1 to the padded string, discarding the original RESULT num1]
            [COMMAND Left-pad num2 with '0' characters using zfill so its length equals max_length; rebind the name num2 to the padded string, discarding the original RESULT num2]
            [COMMAND Initialize carry to the integer 0 as the initial carry state for the addition loop RESULT carry]
            [COMMAND Initialize result to an empty list as the accumulator for output digit strings RESULT result]
            [COMMAND For each index i from max_length - 1 down to 0 inclusive (step -1), representing a right-to-left traversal from least significant to most significant digit: compute digit_sum as int(num1[i]) + int(num2[i]) + carry RESULT digit_sum]
            [COMMAND Within the for loop over i: compute the new carry as digit_sum // 10 (integer division; equals 1 if digit_sum >= 10, otherwise 0) RESULT carry]
            [COMMAND Within the for loop over i: compute digit as digit_sum % 10 (the ones place of digit_sum) RESULT digit]
            [COMMAND Within the for loop over i: prepend str(digit) to the front of result at index 0, keeping the list in left-to-right order without a separate reversal RESULT result]
            [COMMAND After the loop, if carry is greater than 0, prepend str(carry) to the front of result at index 0 to account for a final overflow carry RESULT result]
            [COMMAND Join all elements of result into a single string with no separator and return it RESULT _OUTPUT]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Both num1 and num2 are empty strings, making max_length equal to 0; the for loop body never executes and carry remains 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the for loop entirely because range(0 - 1, -1, -1) is empty RESULT result]
            [COMMAND Skip the carry check because carry equals 0 RESULT result]
            [COMMAND Return the empty string produced by joining the empty result list RESULT _OUTPUT]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The digit-by-digit addition produces a carry of 1 after processing the most significant digit position (i = 0), indicating the sum requires one more digit than max_length]
        [SEQUENTIAL_BLOCK]
            [COMMAND After the loop completes, detect that carry > 0 (carry equals 1) RESULT carry]
            [COMMAND Prepend str(carry) — the string '1' — to the front of result at index 0 RESULT result]
            [COMMAND Join result into a string and return it; the output has max_length + 1 digits RESULT _OUTPUT]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Subtracts two non-negative integers represented as digit strings using long subtraction with borrow propagation, automatically handling operand ordering and sign determination to support negative results." subtract]
    [INPUTS]
        <REF> num1 </REF>: string "First operand as a string of decimal digits representing a non-negative integer"
        <REF> num2 </REF>: string "Second operand as a string of decimal digits representing a non-negative integer"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "The decimal string representation of (num1 - num2), prefixed with '-' if the result is negative, with no leading zeros except for the value zero itself"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compare the lengths of num1 and num2; if len(num1) is less than len(num2), swap num1 and num2 and set negative to True; else if len(num1) is greater than len(num2), set negative to False RESULT negative, num1, num2]
            [COMMAND Within the same-length else branch, compare num1 and num2 lexicographically; if num1 is less than num2, swap num1 and num2 and set negative to True; otherwise set negative to False (this branch also silently handles the equal case, leaving negative as False and producing a zero result downstream) RESULT negative, num1, num2]
            [COMMAND Compute max_length as the maximum of len(num1) and len(num2), then left-pad both num1 and num2 with leading zeros via zfill to exactly max_length characters, discarding the original unpadded bindings RESULT max_length, num1, num2]
            [COMMAND Initialize borrow to 0 and result to an empty list RESULT borrow, result]
            [COMMAND Iterate i from max_length minus 1 down to 0 inclusive, stepping by -1 (right-to-left digit walk) RESULT i]
            [COMMAND Within the loop, compute digit_diff as int(num1[i]) minus int(num2[i]) minus borrow RESULT digit_diff]
            [COMMAND Within the loop, if digit_diff is less than 0, add 10 to digit_diff and set borrow to 1; otherwise set borrow to 0 RESULT digit_diff, borrow]
            [COMMAND Within the loop, prepend the string representation of digit_diff to result at index 0 RESULT result]
            [COMMAND After the loop, while result has more than one element and its first element is '0', remove the leading '0' from the front of result RESULT result]
            [COMMAND If negative is True, prepend '-' to result at index 0 RESULT result]
            [COMMAND Join all elements of result into a single string and return it RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Multiplies two non-negative integers represented as digit strings using a grade-school positional algorithm and returns their product as a digit string." multiply]
    [INPUTS]
        <REF> num1 </REF>: string "First non-negative integer represented as a string of decimal digit characters; must be non-empty for defined behavior"
        <REF> num2 </REF>: string "Second non-negative integer represented as a string of decimal digit characters; must be non-empty for defined behavior"
    [END_INPUTS]

    [OUTPUTS]
        <REF> product </REF>: string "The decimal string representation of the product of num1 and num2, with no leading zeros except when the result is zero"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute len1 as the length of num1 and len2 as the length of num2 via simultaneous assignment RESULT len1, len2]
            [COMMAND Allocate result as a list of (len1 + len2) integer zeros to hold all partial-product accumulations RESULT result]
            [COMMAND For each outer index i from len1 - 1 down to 0 inclusive (stepping -1), iterate over every digit of num1 from least-significant to most-significant position RESULT i]
            [COMMAND Within the outer loop over i, for each inner index j from len2 - 1 down to 0 inclusive (stepping -1), iterate over every digit of num2 from least-significant to most-significant position RESULT j]
            [COMMAND Within the inner loop over j, compute mul as the integer product of the digit character num1[i] and the digit character num2[j] RESULT mul]
            [COMMAND Within the inner loop over j, compute positional indices p1 = i + j (carry slot, higher significance) and p2 = i + j + 1 (units slot, lower significance) for the current digit pair RESULT p1, p2]
            [COMMAND Within the inner loop over j, compute total as mul plus the current value already accumulated at result[p2] RESULT total]
            [COMMAND Within the inner loop over j, add the tens digit of total (total // 10) into result[p1] using in-place addition to propagate the carry RESULT result[p1]]
            [COMMAND Within the inner loop over j, overwrite result[p2] with the units digit of total (total % 10) RESULT result[p2]]
            [COMMAND Initialize start to integer 0 to mark the beginning of the leading-zero scan RESULT start]
            [COMMAND While start is strictly less than len(result) - 1 and result[start] equals 0, increment start by 1 to skip leading zero digits; the condition len(result) - 1 ensures at least one digit is always kept RESULT start]
            [COMMAND Convert each integer element of result from index start to the end into its character representation, join them into a single string, and return that string as the product RESULT product]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Either num1 or num2 represents the value zero, causing result to contain all zeros after the nested loops; start advances to len(result) - 1 and the returned string is '0']
        [SEQUENTIAL_BLOCK]
            [COMMAND The leading-zero skip loop halts one position before the end of result, preserving a single '0' digit, and the join returns the string '0' RESULT product]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
