[DEFINE_WORKER: "Registers a new username in the users dictionary with an empty list as its value, returning true on success or false if the username already exists." add_user]
    [INPUTS]
        <REF> username </REF>: string "The username to register; used as a key lookup against the existing users dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the username was successfully registered, false if it was already present in the users dictionary"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username already exists as a key in self.users using an exact membership test RESULT username_exists]
            [COMMAND If username_exists is true, return False immediately without mutating state RESULT early_return_false]
            [COMMAND If username_exists is false, insert username into self.users mapped to an empty list RESULT users_updated]
            [COMMAND Return True to signal successful registration RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is already a key in self.users at the time of the call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip all mutation and return False as an early exit, leaving self.users unchanged RESULT duplicate_rejected]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a user from the users dictionary by username if it exists, returning True on success or False if the username was not found." remove_user]
    [INPUTS]
        <REF> username </REF>: string "The username key to look up and remove from the users dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the user was found and deleted, False if the username did not exist in the users dictionary"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username exists as a key in self.users using a membership test RESULT username_exists]
            [COMMAND If username_exists is True, delete the entry keyed by username from self.users RESULT deletion_performed]
            [COMMAND If username_exists is True, return True immediately to signal successful removal RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not found in self.users (membership test returns False)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately to signal that no deletion occurred because the username did not exist RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sends a message from one registered user to another by validating both users exist, building a timestamped message record, and appending it to both the sender's and receiver's message histories." send_message]
    [INPUTS]
        <REF> sender </REF>: string "Username of the message sender; must exist as a key in self.users"
        <REF> receiver </REF>: string "Username of the intended message recipient; must exist as a key in self.users"
        <REF> message </REF>: string "Message body content to be delivered"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns True if the message was successfully stored for both parties, or False if either sender or receiver is not a registered user"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the current wall-clock time via datetime.now() and format it as a fixed-width string using strftime with the format '%Y-%m-%d %H:%M:%S'; bind the result to timestamp RESULT timestamp]
            [COMMAND Construct a dict named message_info with exactly four keys: 'sender' bound to the sender parameter, 'receiver' bound to the receiver parameter, 'message' bound to the message parameter, and 'timestamp' bound to timestamp; no transformation is applied to any of the three caller-supplied values RESULT message_info]
            [COMMAND Append message_info to the list stored at self.users[sender], adding the record to the sender's message history RESULT sender_append_result]
            [COMMAND Append message_info to the list stored at self.users[receiver], adding the same record object to the receiver's message history RESULT receiver_append_result]
            [COMMAND Return True to indicate the message was successfully stored for both participants RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: sender is not present as a key in self.users, OR receiver is not present as a key in self.users (short-circuit OR: the check fails as soon as the first absent user is found)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without creating a timestamp, constructing message_info, or modifying any user list RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Retrieves the message list for a given username from the internal user store, returning an empty list immediately if the username is not found." get_messages]
    [INPUTS]
        <REF> username </REF>: string "The key used to look up a user's messages in the internal users mapping"
    [END_INPUTS]

    [OUTPUTS]
        <REF> messages </REF>: list "The list of messages associated with the username, or an empty list if the username does not exist in the store"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether username is absent from self.users using an exclusive membership test (not in); if username is not present, immediately return an empty list and terminate the method — this is a guard clause that prevents any downstream lookup RESULT early_exit_or_continue]
            [COMMAND Perform a direct subscript lookup on self.users with username as the key and return the resulting value to the caller RESULT messages]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: username is not present in self.users (guard clause evaluates to True)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty list immediately, bypassing the subscript lookup entirely RESULT empty_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
