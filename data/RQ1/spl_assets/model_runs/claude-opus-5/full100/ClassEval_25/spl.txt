# CLASS CookiesUtil

[DEFINE_WORKER: "Initializes the object by storing the provided cookies file path as an instance attribute and setting the cookies data to None, establishing a lazy-load pattern for deferred cookie reading." __init__]
    [INPUTS]
        <REF> cookies_file </REF>: any "The file path or object identifying the source of cookie data; accepted as-is with no validation, so None or any type is stored directly"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming cookies_file parameter directly to self.cookies_file with no transformation, validation, or default fallback RESULT self.cookies_file]
            [COMMAND Assign None to self.cookies as an explicit sentinel indicating that no cookie data has been loaded yet RESULT self.cookies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts the cookies value from the response dictionary and stores it on the instance, then persists it by calling the internal save method." get_cookies]
    [INPUTS]
        <REF> reponse </REF>: dict "A dictionary-like object expected to contain a 'cookies' key; note: parameter name is misspelled (missing 'r') and will cause a NameError if the caller uses the correctly-spelled 'response'"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Look up the value associated with the key 'cookies' in the parameter 'reponse' using subscript access; assign the result directly to the instance attribute 'self.cookies', creating it if it did not exist or overwriting it if it did; no default, guard, or transformation is applied RESULT self.cookies]
            [COMMAND Call the internal method '_save_cookies' on the current instance to persist the cookies value that was just assigned RESULT _save_cookies_invocation]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The parameter 'reponse' is not defined or is out of scope at call time, most likely due to the misspelled parameter name]
        [LOG "NameError raised because 'reponse' does not match a variable in scope; caller likely passed the argument under the correctly-spelled name 'response'"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The object passed as 'reponse' does not support subscript access or does not contain the key 'cookies']
        [LOG "TypeError raised if the object does not implement __getitem__; KeyError raised if the object is a plain dict and 'cookies' key is absent; no fallback or .get() guard is present"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to open and deserialize a JSON cookies file from the path stored in self.cookies_file, returning the parsed cookie data on success or an empty dictionary if the file does not exist." load_cookies]
    [INPUTS]
        <REF> self </REF>: object "Instance reference providing access to self.cookies_file, the path of the cookies file to read"
    [END_INPUTS]

    [OUTPUTS]
        <REF> cookies_data </REF>: dict "Parsed JSON content of the cookies file, or an empty dictionary if the file was not found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.cookies_file in read mode RESULT file]
            [COMMAND Deserialize the JSON content from file into cookies_data RESULT cookies_data]
            [COMMAND Return cookies_data to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: FileNotFoundError is caught because self.cookies_file does not exist]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return an empty dictionary as the default cookies value RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: FileNotFoundError is raised when opening self.cookies_file]
        [LOG "The cookies file at self.cookies_file does not exist; returning empty dictionary as default"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Serializes the current cookies dictionary to a JSON file at the configured path, returning True on success and False if any exception occurs." _save_cookies]
    [INPUTS]
        <REF> self </REF>: object "Instance reference providing access to cookies_file path and cookies dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the cookies were written successfully, False if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at self.cookies_file in write mode RESULT file_handle]
            [COMMAND Serialize self.cookies as JSON and write it to file_handle using json.dump RESULT write_result]
            [COMMAND Return True to indicate the save operation completed successfully RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is caught by the bare except clause]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to signal that the save operation failed RESULT failure_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during file open or json.dump (bare except catches all exception types)]
        [LOG "No exception details are logged; the exception is silently swallowed"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Serializes all key-value pairs from the instance's cookies dictionary into a single semicolon-delimited HTTP Cookie header string and writes it unconditionally to the 'cookies' key of the given request mapping." set_cookies]
    [INPUTS]
        <REF> request </REF>: dict "A mutable mapping representing the HTTP request; its 'cookies' key will be overwritten with the serialized cookie string"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over all key-value pairs in self.cookies by calling self.cookies.items() RESULT cookie_pairs]
            [COMMAND For each (key, value) pair in cookie_pairs, format it as the string 'key=value' using an f-string, collecting all formatted strings into a list RESULT formatted_cookies]
            [COMMAND Join all strings in formatted_cookies with the separator '; ' to produce a single cookie header string RESULT cookie_header]
            [COMMAND Assign cookie_header to request['cookies'], unconditionally overwriting any prior value at that key RESULT request['cookies']]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.cookies is an empty dictionary]
        [SEQUENTIAL_BLOCK]
            [COMMAND self.cookies.items() yields no pairs; the list comprehension produces an empty list; '; '.join([]) returns an empty string; assign the empty string to request['cookies'] RESULT request['cookies']]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.cookies contains exactly one key-value pair]
        [SEQUENTIAL_BLOCK]
            [COMMAND The list comprehension produces a single-element list; '; '.join produces that element unchanged with no separator; assign the resulting 'key=value' string to request['cookies'] RESULT request['cookies']]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A cookie key or value contains the characters '=' or ';']
        [SEQUENTIAL_BLOCK]
            [COMMAND The f-string embeds the character literally without escaping; the resulting cookie header string may be malformed and could break downstream cookie parsers; the value is still assigned to request['cookies'] without any error RESULT request['cookies']]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.cookies is None or the attribute does not exist on the instance]
        [LOG "AttributeError raised when calling .items() on None or a missing attribute; the assignment to request['cookies'] never executes"]
        [THROW AttributeError "Raised by Python when .items() is called on a None value or when the cookies attribute is absent from self; propagates uncaught from this method"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
