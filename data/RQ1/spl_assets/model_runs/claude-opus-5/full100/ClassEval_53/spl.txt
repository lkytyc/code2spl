[DEFINE_WORKER: "Finds and returns the longest word from a sentence that exists in the instance's word list, after normalizing the input to lowercase and stripping ASCII punctuation." find_longest_word]
    [INPUTS]
        <REF> self </REF>: object "Instance containing word_list, a collection of valid vocabulary words used for membership testing"
        <REF> sentence </REF>: string "Raw input sentence to search; must be a non-None string or .lower() will raise AttributeError"
    [END_INPUTS]

    [OUTPUTS]
        <REF> longest_word </REF>: string "The first vocabulary word in the tokenized sentence with the strictly greatest length; returns empty string if no vocabulary word is found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize longest_word to empty string as the sentinel baseline; any non-empty vocabulary word found later will strictly exceed its length of zero RESULT longest_word]
            [COMMAND Overwrite sentence with its lowercase form by calling sentence.lower(); the original casing is permanently lost after this step RESULT sentence]
            [COMMAND Overwrite sentence by applying re.sub with pattern '[%s]' % re.escape(string.punctuation) and replacement '' to strip all ASCII punctuation characters; Unicode punctuation is not affected RESULT sentence]
            [COMMAND Overwrite sentence by calling re.split(' ', sentence) to tokenize on a single literal space, producing a list of word strings; consecutive spaces will produce empty string tokens in the list, and sentence changes type from string to list RESULT sentence]
            [COMMAND Iterate over each word in sentence (the token list); for each word, evaluate the short-circuit compound condition: first check word in self.word_list (membership test, left clause); only if true, check len(word) > len(longest_word) (strict greater-than, exclusive boundary, right clause) RESULT loop_iteration]
            [COMMAND Within the for loop over sentence, when both clauses of the compound condition are true, overwrite longest_word with the current word; ties are broken in favor of the first qualifying occurrence because the boundary is strict and later equal-length words do not satisfy len(word) > len(longest_word) RESULT longest_word]
            [COMMAND Return longest_word after the loop completes; if no word in sentence passed both conditions, the returned value is the initial empty string RESULT return longest_word]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
