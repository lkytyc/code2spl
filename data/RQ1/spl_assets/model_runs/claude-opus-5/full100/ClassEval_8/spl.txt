[DEFINE_WORKER: "Validates that the given amount is non-negative, then adds it to the current balance and returns the updated balance." deposit]
    [INPUTS]
        <REF> amount </REF>: number "The numeric value to deposit; must be zero or greater"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number "The updated balance after the deposit has been applied"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Add amount to self.balance, storing the result back in self.balance RESULT self.balance]
            [COMMAND Return self.balance as the result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount is strictly less than 0 (exclusive boundary: negative values only; zero is allowed)]
        [LOG "Deposit rejected because amount is negative"]
        [THROW ValueError "Raised with the message 'Invalid amount' when the supplied amount is negative; execution does not reach the balance update"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that the requested withdrawal amount is non-negative and does not exceed the current balance, then deducts the amount from the balance and returns the updated balance." withdraw]
    [INPUTS]
        <REF> amount </REF>: number "The amount to withdraw; must be zero or greater and must not exceed the current balance"
    [END_INPUTS]

    [OUTPUTS]
        <REF> balance </REF>: number "The updated account balance after the withdrawal has been applied"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether amount is strictly less than zero (exclusive lower bound — zero is valid) RESULT amount_negative_check]
            [COMMAND Check whether amount strictly exceeds self.balance (exclusive upper bound — a withdrawal exactly equal to the balance is allowed, reducing the balance to zero) RESULT insufficient_balance_check]
            [COMMAND Subtract amount from self.balance and store the result back in self.balance RESULT updated_balance]
            [COMMAND Return self.balance as the result of the method RESULT balance]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: amount < 0]
        [LOG "Withdrawal rejected because the requested amount is negative; the value must be zero or greater"]
        [THROW ValueError "Raised with message 'Invalid amount' when the supplied amount is strictly less than zero"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: amount > self.balance]
        [LOG "Withdrawal rejected because the requested amount strictly exceeds the current balance"]
        [THROW ValueError "Raised with message 'Insufficient balance.' when the supplied amount is strictly greater than the current balance"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Transfers a specified amount from the current account to another account by withdrawing from self and depositing into the target." transfer]
    [INPUTS]
        <REF> other_account </REF>: Account "The destination account object that will receive the transferred amount"
        <REF> amount </REF>: numeric "The amount to transfer; no validation or default guard is applied at this level — zero, negative, or floating-point values propagate to the called methods unchanged"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call withdraw on the current account object, passing amount, to deduct amount from self.balance; if self.balance is insufficient and withdraw raises an exception, execution halts here and the deposit step never runs RESULT withdraw_result]
            [COMMAND Call deposit on other_account, passing the same amount value, to add amount to other_account.balance; this step executes only if the preceding withdraw call completes without raising RESULT deposit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: withdraw raises an exception, for example due to insufficient funds in self.balance]
        [LOG "Transfer aborted: withdraw on source account failed before deposit was attempted, leaving both account balances unchanged"]
        [THROW WithdrawException "Propagated from self.withdraw when the source account cannot fulfill the requested amount; deposit is never called and the transfer is implicitly rolled back at this point"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: deposit raises an exception after withdraw has already succeeded]
        [LOG "Transfer incomplete: withdraw on source account succeeded and reduced self.balance, but deposit on other_account failed, resulting in an inconsistent state unless the caller handles the exception with a compensating rollback"]
        [THROW DepositException "Propagated from other_account.deposit after the source balance has already been reduced; without external transaction management or a try/except rollback in the caller, the deducted amount is lost"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
