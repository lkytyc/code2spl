# CLASS BookManagement

[DEFINE_WORKER: "Initializes a new instance by setting up an empty inventory dictionary as the sole instance attribute." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the instance attribute 'inventory' to an empty dictionary RESULT self.inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a book title to the inventory by incrementing its quantity if it already exists, or initializing it with the given quantity if it does not." add_book]
    [INPUTS]
        <REF> title </REF>: string "The title of the book to add or update in the inventory"
        <REF> quantity </REF>: integer "The number of copies to add; defaults to 1 if not provided"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether title exists as a key in self.inventory RESULT title_exists]
            [COMMAND If title_exists is true, increment self.inventory[title] by quantity RESULT updated_count]
            [COMMAND If title_exists is false, set self.inventory[title] to quantity RESULT new_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title is already present in self.inventory (membership test returns True)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Retrieve the current value at self.inventory[title] and add quantity to it, storing the result back at self.inventory[title] RESULT updated_count]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: title is not present in self.inventory (membership test returns False)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new key title in self.inventory and assign quantity as its initial value RESULT new_entry]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a specified quantity of a book from the inventory, raising an error if the title is absent or stock is insufficient, decrementing the count otherwise, and deleting the entry entirely if the count reaches exactly zero." remove_book]
    [INPUTS]
        <REF> title </REF>: string "The key identifying the book to remove from inventory"
        <REF> quantity </REF>: integer "The number of copies to remove"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether title is absent from self.inventory OR whether self.inventory[title] is strictly less than quantity; the membership check is evaluated first to short-circuit and prevent a KeyError on the lookup RESULT guard_condition]
            [COMMAND Decrement self.inventory[title] by quantity RESULT updated_count]
            [COMMAND Check whether self.inventory[title] is exactly equal to 0 RESULT zero_check]
            [COMMAND If zero_check is true, delete the key title from self.inventory entirely, removing the zero-count entry so the inventory contains only titles with positive stock RESULT cleanup_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: title is not present in self.inventory, OR self.inventory[title] is less than quantity (short-circuit: membership is tested before the value lookup to avoid a KeyError)]
        [LOG "Attempted to remove quantity copies of title but the item does not exist in inventory or available stock is insufficient"]
        [THROW False "The code raises the literal boolean False, which is a bug — raise expects a BaseException subclass; at runtime this produces a TypeError instead of a meaningful domain exception"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the current inventory of the instance by providing direct access to the internal inventory data structure." view_inventory]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> inventory </REF>: object "The instance's internal inventory data structure"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the value of self.inventory to the caller RESULT inventory]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the quantity of a given book title from the inventory, or 0 if the title does not exist." view_book_quantity]
    [INPUTS]
        <REF> title </REF>: string "The title of the book to look up in the inventory"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The quantity of the book in the inventory, or 0 if the title is not found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether title is absent from self.inventory using a membership test RESULT title_missing]
            [COMMAND If title_missing is true, return the integer constant 0 immediately as a guard-clause early exit RESULT early_return_zero]
            [COMMAND Retrieve and return the value stored at self.inventory[title] RESULT quantity]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: title is not present as a key in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without accessing self.inventory[title] RESULT early_return_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
