# CLASS DataStatistics

[DEFINE_WORKER: "Computes the arithmetic mean of a numeric iterable by dividing the sum of all elements by the count of elements, then rounds the result to two decimal places." mean]
    [INPUTS]
        <REF> data </REF>: iterable of numeric "A non-empty sequence of numeric values whose mean is to be calculated; must not be empty and must contain only numeric types"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The arithmetic mean of the input data rounded to 2 decimal places"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute the sum of all elements in data using sum(data), accumulating a single numeric total RESULT total]
            [COMMAND Compute the count of elements in data using len(data) RESULT count]
            [COMMAND Divide total by count using true division to produce a float quotient RESULT quotient]
            [COMMAND Round quotient to exactly 2 decimal places using round(quotient, 2) RESULT rounded_mean]
            [COMMAND Return rounded_mean as the final result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is an empty sequence, causing len(data) to equal 0 and the division step to divide by zero]
        [LOG "ZeroDivisionError raised at the division step because len(data) is 0; the method has no guard against an empty input"]
        [THROW ZeroDivisionError "Raised by Python's true division operator when the denominator (element count) is zero due to an empty input sequence"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: data contains one or more non-numeric elements, causing sum(data) to fail type checking]
        [LOG "TypeError raised inside sum() because at least one element in data cannot be added as a numeric value"]
        [THROW TypeError "Raised by sum() when data contains elements that are not numeric types and therefore cannot be summed"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the median of a numeric list by sorting it and returning the middle element for odd-length lists or the rounded average of the two middle elements for even-length lists." median]
    [INPUTS]
        <REF> data </REF>: iterable "A non-empty iterable of comparable numeric values whose median is to be computed; mixed incomparable types will raise a TypeError at runtime"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int | float "The median value: the single center element for an odd-length input, or the arithmetic mean of the two center elements rounded to 2 decimal places for an even-length input"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the input iterable 'data' into a new list 'sorted_data' in ascending order, leaving the original 'data' unmodified RESULT sorted_data]
            [COMMAND Compute 'n' as the integer count of elements in 'sorted_data' RESULT n]
            [COMMAND Check whether 'n' is even by evaluating 'n % 2 == 0'; if true, execute the even branch; otherwise execute the odd branch RESULT branch_decision]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n % 2 == 0: the element count is even]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute 'middle' as the upper-center index using integer floor division: middle = n // 2 RESULT middle]
            [COMMAND Return the arithmetic mean of the two center elements sorted_data[middle - 1] and sorted_data[middle], rounded to exactly 2 decimal places RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: n % 2 != 0: the element count is odd]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute 'middle' as the exact center index using integer floor division: middle = n // 2 RESULT middle]
            [COMMAND Return sorted_data[middle] directly without rounding RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: 'data' contains elements of mutually incomparable types (e.g. a mix of integers and strings)]
        [LOG "sorted() raises a TypeError because it cannot establish an ordering between incomparable types; the exception propagates uncaught and the method does not return a value"]
        [THROW TypeError "Raised by the built-in sorted() call when elements in 'data' cannot be compared with one another"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the mode(s) of a data collection by counting element frequencies and returning all elements that share the highest frequency." mode]
    [INPUTS]
        <REF> data </REF>: iterable "A non-empty iterable of elements whose mode(s) are to be found; raises ValueError on empty input"
    [END_INPUTS]

    [OUTPUTS]
        <REF> mode </REF>: list "A list of all elements that appear most frequently; contains multiple elements when the data is multimodal"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the frequency of each element in data using Counter, storing the result in counter RESULT counter]
            [COMMAND Determine the highest frequency value by taking the max of counter.values(), storing it in mode_count RESULT mode_count]
            [COMMAND Collect all elements from counter whose frequency equals mode_count into a list, storing the result in mode RESULT mode]
            [COMMAND Return mode RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: data is empty, causing counter.values() to be empty and max() to receive an empty sequence]
        [LOG "ValueError raised by max() on empty sequence because data contained no elements; no guard exists before this call"]
        [THROW ValueError "Raised by the built-in max() when counter.values() is empty due to empty input data"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
