[DEFINE_WORKER: "Computes the number of full chunks and the remaining item count by performing floor division and modulo of the list length against the configured limit, returning both as a tuple." setNum]
    [INPUTS]
        <REF> self.lst </REF>: list "The instance list whose length determines the total number of items to divide"
        <REF> self.limit </REF>: int "The divisor representing the chunk size or partition boundary; must be non-zero"
    [END_INPUTS]

    [OUTPUTS]
        <REF> size </REF>: int "The integer quotient of len(self.lst) divided by self.limit, representing the number of full chunks"
        <REF> remainder </REF>: int "The modulo of len(self.lst) divided by self.limit, representing the number of leftover items that do not fill a complete chunk"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute size by applying floor division of len(self.lst) by self.limit and assign the integer quotient to local variable size RESULT size]
            [COMMAND Compute remainder by applying modulo of len(self.lst) by self.limit and assign the leftover count to local variable remainder RESULT remainder]
            [COMMAND Return size and remainder together as a two-element tuple RESULT (size, remainder)]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(self.lst) is less than self.limit, meaning the list is shorter than one full chunk]
        [SEQUENTIAL_BLOCK]
            [COMMAND size evaluates to 0 because floor division of a smaller numerator by a larger denominator yields zero RESULT size=0]
            [COMMAND remainder evaluates to len(self.lst) in full, as no items are consumed by a complete chunk RESULT remainder=len(self.lst)]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len(self.lst) is 0, meaning the list is empty]
        [SEQUENTIAL_BLOCK]
            [COMMAND size evaluates to 0 because floor division of zero by any non-zero limit is zero RESULT size=0]
            [COMMAND remainder evaluates to 0 because modulo of zero by any non-zero limit is zero RESULT remainder=0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len(self.lst) is exactly divisible by self.limit with no leftover]
        [SEQUENTIAL_BLOCK]
            [COMMAND size evaluates to the exact quotient with no fractional part discarded RESULT size=len(self.lst)//self.limit]
            [COMMAND remainder evaluates to 0, indicating no trailing partial chunk exists RESULT remainder=0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.limit equals 0 at the time either division expression is evaluated]
        [LOG "ZeroDivisionError raised on the floor division or modulo operation because the divisor self.limit is zero"]
        [THROW ZeroDivisionError "Python raises ZeroDivisionError when self.limit is zero; no guard exists in the method body, so the caller is responsible for ensuring self.limit is non-zero before invoking setNum"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

    [END_OUTPUTS]
