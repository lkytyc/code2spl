[DEFINE_WORKER: "Searches the instance's job listings and returns all entries whose title contains the given criteria as a case-insensitive substring or whose requirements list contains the criteria as a case-insensitive exact match." search_jobs]
    [INPUTS]
        <REF> criteria </REF>: string "The search term used to filter job listings; matched case-insensitively against job titles (substring) and requirements (exact element)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> matching_jobs </REF>: list "A list of job listing dictionaries from self.job_listings that satisfy at least one of the two match conditions."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named matching_jobs to accumulate results. RESULT matching_jobs]
            [COMMAND Normalize criteria once by calling criteria.lower() and store it for reuse. RESULT criteria_lower]
            [COMMAND Begin iterating over every element in self.job_listings, binding each element to job_listing. RESULT job_listing]
            [COMMAND Within the for loop, evaluate the first branch of the OR condition: check whether criteria_lower appears as a substring inside job_listing["job_title"].lower(); this is a case-insensitive inclusive substring test, not an equality test. RESULT title_match]
            [COMMAND Within the for loop, if title_match is False (short-circuit), evaluate the second branch: build a list of lowercased requirement strings via [r.lower() for r in job_listing["requirements"]] and check whether criteria_lower is an exact element of that list; this is a case-insensitive full-value equality test, not a substring test. RESULT requirements_match]
            [COMMAND Within the for loop, if either title_match or requirements_match is True, append job_listing to matching_jobs; otherwise advance to the next iteration with no action. RESULT matching_jobs]
            [COMMAND After the loop completes, return matching_jobs. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.job_listings is empty; the for loop body never executes.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return matching_jobs as an empty list immediately after the loop. RESULT matching_jobs]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The title substring check (first OR branch) is True for a given job_listing; Python short-circuits and skips building the requirements list comprehension for that iteration.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the for loop, append job_listing to matching_jobs without evaluating the requirements branch. RESULT matching_jobs]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters the full list of resumes by checking each one against a job's requirements and returns all matching candidates as a list of applicants." GetJobApplicants]
    [INPUTS]
        <REF> job </REF>: dict "A job object containing at least a 'requirements' key whose value is used to evaluate each resume"
    [END_INPUTS]

    [OUTPUTS]
        <REF> applicants </REF>: list "A list of resume objects that satisfy the job's requirements; empty if no resumes match"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named applicants to accumulate qualifying resumes RESULT applicants]
            [COMMAND Iterate over every resume in self.resumes, visiting each element exactly once in order with no early exit RESULT resume_iteration]
            [COMMAND Within the for loop over self.resumes, for each resume call matches_requirements passing the current resume and job['requirements']; proceed to the append step only when the call returns a truthy value, otherwise skip silently with no else branch RESULT match_check]
            [COMMAND Within the for loop, when the match check passes, append the current resume to applicants RESULT applicants]
            [COMMAND After the loop has completed a full pass over all resumes, return applicants RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether every skill in a resume is present in a requirements collection, returning false immediately upon finding the first missing skill and implicitly returning true if all skills pass." matches_requirements]
    [INPUTS]
        <REF> resume </REF>: dict "A dictionary representing a resume, expected to contain a 'skills' key whose value is an iterable of skill entries"
        <REF> requirements </REF>: collection "A collection (list or set) of required skills against which each resume skill is tested for membership"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "False if any skill from the resume is absent from requirements; True (implicit fall-through) if every skill is present"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each element in resume['skills'], binding the current element to the variable 'skill' RESULT skill]
            [COMMAND Within the for loop, test whether the current 'skill' is NOT present in the 'requirements' collection using a membership-exclusion check RESULT skill_missing]
            [COMMAND Within the for loop, if 'skill_missing' is true, immediately exit the entire function by returning the boolean value False, short-circuiting all remaining iterations (fail-fast early return) RESULT early_return_false]
            [COMMAND After the for loop exhausts all skills without triggering the early return, return the boolean value True, indicating every skill was found in requirements RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: resume['skills'] is empty — the iterable contains no elements]
        [SEQUENTIAL_BLOCK]
            [COMMAND The for loop body never executes; control falls through immediately to the return True statement, so the function returns True for a resume with no skills RESULT return_true_empty]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
