[DEFINE_WORKER: "Encodes a plaintext string using a Caesar cipher by shifting each alphabetic character forward by a given amount within its case boundary, leaving non-alphabetic characters unchanged." CaesarCipher]
    [INPUTS]
        <REF> plaintext </REF>: string "The input string to be encrypted; may contain letters, digits, spaces, and punctuation"
        <REF> shift </REF>: integer "The number of positions to shift each alphabetic character forward in the alphabet"
    [END_INPUTS]

    [OUTPUTS]
        <REF> ciphertext </REF>: string "The resulting encrypted string with the same length and structure as plaintext"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ciphertext as an empty string RESULT ciphertext]
            [COMMAND For each character char in plaintext, determine whether char is alphabetic RESULT is_alpha_check]
            [COMMAND Within the loop, if char is alphabetic and char is uppercase, set ascii_offset to 65; else if char is alphabetic and char is lowercase, set ascii_offset to 97 RESULT ascii_offset]
            [COMMAND Within the loop, if char is alphabetic, compute shifted_char as chr((ord(char) - ascii_offset + shift) % 26 + ascii_offset), normalizing char to a 0-25 index, adding shift, wrapping at alphabet boundary 26, and mapping back to the correct ASCII range RESULT shifted_char]
            [COMMAND Within the loop, if char is alphabetic, append shifted_char to ciphertext RESULT ciphertext]
            [COMMAND Return ciphertext RESULT ciphertext]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: char is not alphabetic (digit, space, punctuation, or any non-letter character)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the loop, append char unchanged to ciphertext, preserving original formatting and spacing RESULT ciphertext]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypts a plaintext string using the Vigenère cipher by shifting each alphabetic character forward in the alphabet by an amount derived from the corresponding character in a repeating key, preserving original case and passing non-alphabetic characters through unchanged." vigenere_cipher]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the encryption key as self.key, a non-empty string of alphabetic characters used cyclically"
        <REF> plain_text </REF>: string "The plaintext string to encrypt; may contain any mix of alphabetic and non-alphabetic characters"
    [END_INPUTS]

    [OUTPUTS]
        <REF> encrypted_text </REF>: string "The Vigenère-encrypted string with the same length and non-alphabetic characters as the input, alphabetic characters shifted by the key"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize encrypted_text to an empty string to accumulate the ciphertext output RESULT encrypted_text]
            [COMMAND Initialize key_index to 0 to track the current position within self.key; this counter advances only on alphabetic characters RESULT key_index]
            [COMMAND For each character char in plain_text, check whether char is alphabetic using char.isalpha() RESULT char]
            [COMMAND Within the for loop, when char.isalpha() is true: compute shift as ord(self.key[key_index % len(self.key)].lower()) minus ord('a'), selecting the key character at position key_index modulo the key length, normalizing it to lowercase, and converting it to a 0-based offset in the range [0, 25] RESULT shift]
            [COMMAND Within the for loop, when char.isalpha() is true: compute encrypted_char as chr((ord(char.lower()) minus ord('a') plus shift) modulo 26 plus ord('a')), converting char to lowercase, applying the shift within the 26-letter alphabet using modular arithmetic to wrap around, and producing a lowercase result character RESULT encrypted_char]
            [COMMAND Within the for loop, when char.isalpha() is true: append encrypted_char.upper() to encrypted_text if char.isupper() is true, otherwise append encrypted_char in its lowercase form, preserving the original case of the input character RESULT encrypted_text]
            [COMMAND Within the for loop, when char.isalpha() is true: increment key_index by 1 so the next alphabetic character uses the next position in the key RESULT key_index]
            [COMMAND Return encrypted_text as the fully encrypted result after all characters in plain_text have been processed RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: char.isalpha() is false, meaning char is a digit, space, punctuation, or any other non-alphabetic character]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the for loop, append char unchanged to encrypted_text without computing a shift and without incrementing key_index, so non-alphabetic characters are transparent to both the ciphertext output and the key schedule RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Encrypts a plaintext string using the rail fence cipher by distributing characters across a zigzag grid of the specified number of rails and then reading them off row by row." rail_fence_cipher]
    [INPUTS]
        <REF> plain_text </REF>: string "The plaintext string to encrypt"
        <REF> rails </REF>: integer "The number of rails (rows) in the fence grid; must be at least 1"
    [END_INPUTS]

    [OUTPUTS]
        <REF> encrypted_text </REF>: string "The ciphertext produced by reading the fence grid row by row, skipping unfilled sentinel cells"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a 2D list named fence with dimensions rails × len(plain_text), filling every cell with the sentinel character '\n' to mark unfilled positions RESULT fence]
            [COMMAND Set direction to -1, and set both row and col to 0 to establish the starting position and initial traversal direction for the zigzag write pass RESULT direction, row, col]
            [COMMAND For each char in plain_text: if row equals 0 or row equals rails-1 (inclusive boundary check on both the top and bottom rails), negate direction to reverse the zigzag; then write char into fence[row][col], increment col by 1, and add direction to row RESULT fence]
            [COMMAND Initialize encrypted_text to an empty string to accumulate the cipher output RESULT encrypted_text]
            [COMMAND For each i from 0 to rails-1, and within that loop for each j from 0 to len(plain_text)-1: if fence[i][j] is not equal to '\n', append fence[i][j] to encrypted_text RESULT encrypted_text]
            [COMMAND Return encrypted_text as the final ciphertext RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: rails equals 1: both boundary conditions (row == 0 and row == rails-1) are simultaneously true on every iteration, so direction flips on every step but row remains 0 throughout, placing all characters on the single rail in their original order]
        [SEQUENTIAL_BLOCK]
            [COMMAND All characters are written to fence[0][col] sequentially; the read pass collects them in order, so encrypted_text equals plain_text RESULT encrypted_text]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
