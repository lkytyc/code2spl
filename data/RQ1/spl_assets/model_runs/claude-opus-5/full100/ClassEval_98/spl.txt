[DEFINE_WORKER: "Parses an XML file from the instance's file path, stores the root element, and returns it, or returns None if parsing fails." read_xml]
    [INPUTS]
        <REF> self </REF>: object "Instance containing file_name (path to the XML file) and root (storage for the parsed root element)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> root </REF>: Element or None "The root element of the parsed XML tree, or None if an exception occurs during parsing"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the XML file located at self.file_name using ET.parse and assign the resulting tree object to a local variable named tree RESULT tree]
            [COMMAND Call getroot() on tree to retrieve the root element and assign it to both self.root and a local result variable RESULT self.root]
            [COMMAND Return self.root as the successful result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is raised during parsing or root extraction]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch all exceptions via a bare except block, bypassing the return of self.root RESULT exception_caught]
            [COMMAND Return None to the caller, leaving self.root in whatever state it held before the call RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during ET.parse or getroot() — including but not limited to file-not-found, permission errors, and malformed XML]
        [LOG "All exceptions are silently caught with a bare except clause; no error is logged or re-raised"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to serialize the current XML element tree to a file and returns True on success or False if any exception occurs." write_xml]
    [INPUTS]
        <REF> file_name </REF>: string "The path to the output file where the XML tree will be written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the XML tree was successfully written to the file, False if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create an ElementTree object by wrapping self.root RESULT tree]
            [COMMAND Write the ElementTree to the file at file_name RESULT write_result]
            [COMMAND Return True to indicate the write operation succeeded RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is raised at any point inside the try block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception without inspecting its type or message RESULT caught_exception]
            [COMMAND Return False to signal that the write operation failed RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during ElementTree construction or the file write operation]
        [LOG "An unspecified exception was caught; the method suppresses it and returns False"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over all 'item' elements in the XML tree, uppercases each element's text content in place, then writes the modified tree to the specified file." process_xml_data]
    [INPUTS]
        <REF> file_name </REF>: string "Path or name of the output file to which the modified XML will be written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any "Return value of write_xml, typically the written file path or a status indicator"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Perform a depth-first traversal of self.root using iter('item'), yielding every element whose tag is 'item' at any depth in document order RESULT item_iterator]
            [COMMAND For each element yielded by item_iterator: read the element's current text content into a local variable named text, then overwrite element.text with text.upper(), mutating the XML tree in place; repeat for every 'item' element with no early exit RESULT mutated_tree]
            [COMMAND Call self.write_xml with file_name and return its result to the caller RESULT write_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Any 'item' element encountered during iteration has text equal to None (element has no text content or contains only child elements)]
        [LOG "AttributeError raised when calling .upper() on None; the loop does not guard against None text values"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Searches the XML tree rooted at self.root for all elements matching the given tag name or XPath expression and returns them as a list." find_element]
    [INPUTS]
        <REF> element_name </REF>: str "An XPath-compatible tag name or path string used to filter elements within self.root; its value fully controls which elements are matched"
    [END_INPUTS]

    [OUTPUTS]
        <REF> elements </REF>: list[Element] "A list of all xml.etree.ElementTree.Element objects found under self.root matching element_name; returns an empty list when no match is found, never None"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.root.findall(element_name) to perform an XPath search over self.root, collecting every matching Element into a list; assign the result to elements — the list is never None, an unmatched query produces [] RESULT elements]
            [COMMAND Return elements to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: element_name does not match any child or descendant of self.root]
        [SEQUENTIAL_BLOCK]
            [COMMAND findall returns an empty list []; assign [] to elements and return it — downstream callers should treat this as a signal that no matching elements exist and guard against assuming at least one result RESULT elements]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: element_name equals "." (a dot)]
        [SEQUENTIAL_BLOCK]
            [COMMAND findall returns a single-element list containing self.root itself; assign [self.root] to elements and return it — callers should be aware this is a common accidental pattern when a dot is passed RESULT elements]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: element_name contains a namespace prefix but no namespaces mapping is supplied]
        [SEQUENTIAL_BLOCK]
            [COMMAND findall silently returns [] because the namespace cannot be resolved without a namespaces argument; assign [] to elements and return it RESULT elements]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.root is None or has not been initialized before this method is called]
        [LOG "AttributeError raised when attempting to call findall on a missing or None self.root attribute"]
        [THROW AttributeError "Raised by the Python runtime when self.root does not exist or is None and findall cannot be dispatched"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: element_name is a malformed XPath expression that the ElementTree parser cannot process]
        [LOG "XPathSyntaxError raised by xml.etree.ElementTree when element_name contains invalid XPath syntax"]
        [THROW XPathSyntaxError "Raised by xml.etree.ElementTree.findall when element_name is not a valid tag name or XPath expression"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
