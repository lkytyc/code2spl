[DEFINE_WORKER: "Evaluates a mathematical expression by transforming it into postfix notation, then iterating the postfix stack to compute the final numeric result using a dedicated evaluation stack." calculate]
    [INPUTS]
        <REF> expression </REF>: string "The infix mathematical expression string to be evaluated, where unary minus may be encoded as a tilde (~) sentinel character"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The numeric result of evaluating the input expression, returned as a Python float"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Transform the input expression and prepare internal state by calling self.prepare(self.transform(expression)) RESULT postfix_stack_ready]
            [COMMAND Initialize result_stack as an empty deque to serve as the evaluation stack RESULT result_stack]
            [COMMAND Reverse self.postfix_stack in place so that subsequent pop() calls consume tokens in original left-to-right postfix order RESULT postfix_stack_reversed]
            [COMMAND While self.postfix_stack is non-empty, pop the next token into current_op RESULT current_op]
            [COMMAND Within the while loop, if current_op is not an operator (i.e. it is an operand), replace any tilde (~) characters in current_op with a minus sign (-), then push the normalized operand string onto result_stack RESULT result_stack]
            [COMMAND Within the while loop, if current_op is an operator, pop second_value from result_stack, then pop first_value from result_stack, preserving left-to-right operand order RESULT first_value, second_value]
            [COMMAND Within the while loop operator branch, normalize first_value and second_value by replacing any tilde (~) characters with a minus sign (-) in each RESULT first_value, second_value]
            [COMMAND Within the while loop operator branch, compute temp_result by calling self._calculate(first_value, second_value, current_op) and push str(temp_result) back onto result_stack RESULT result_stack]
            [COMMAND After the while loop exits, evaluate the expression formed by joining all remaining elements of result_stack with '*' using eval(), convert the result to float, and return it RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The operator branch calls result_stack.pop() twice, but result_stack contains fewer than two elements due to a malformed or invalid postfix expression]
        [LOG "IndexError raised by deque.pop() on an empty result_stack when attempting to retrieve first_value or second_value for an operator token"]
        [THROW IndexError "Raised implicitly when result_stack is exhausted before both operands for an operator can be popped; the postfix expression is structurally invalid"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an infix mathematical expression string into a postfix (Reverse Polish Notation) token sequence using the shunting-yard algorithm, storing results in the instance's postfix_stack." prepare]
    [INPUTS]
        <REF> expression </REF>: str "The infix expression string to be parsed and converted to postfix notation"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize op_stack as a deque pre-seeded with the sentinel character ',' to avoid empty-stack checks; convert expression into a character list arr; set current_index to 0 as the start cursor into arr; set count to 0 as the operand-span accumulator RESULT op_stack, arr, current_index, count]
            [COMMAND Iterate over arr using enumerate, yielding index i and character current_op for each iteration RESULT i, current_op]
            [COMMAND Within the for loop: if current_op is an operator (as determined by self.is_operator), and count is greater than 0, join arr[current_index : current_index + count] into a single string and append it to self.postfix_stack as a complete operand token RESULT self.postfix_stack]
            [COMMAND Within the for loop, after any pending operand flush: assign peek_op to op_stack[-1] (the current top of the operator stack) RESULT peek_op]
            [COMMAND Within the for loop, if current_op equals ')': pop operators from op_stack and append each as a string to self.postfix_stack in a while loop that continues as long as op_stack[-1] is not '('; after the while loop exits, pop and discard the '(' from op_stack RESULT self.postfix_stack, op_stack]
            [COMMAND Within the for loop, else (current_op is any operator other than ')'): enter a while loop that continues only while all three conditions hold — current_op is not '(', peek_op is not ',', and self.compare(current_op, peek_op) returns true (incoming operator has lower or equal precedence); inside the loop, pop op_stack and append the popped operator as a string to self.postfix_stack, then update peek_op to op_stack[-1] RESULT self.postfix_stack, op_stack, peek_op]
            [COMMAND Within the for loop, after the precedence-popping while loop: push current_op onto op_stack RESULT op_stack]
            [COMMAND Within the for loop, after handling any operator branch: reset count to 0 and advance current_index to i + 1 to start a fresh operand span after the current operator position RESULT count, current_index]
            [COMMAND Within the for loop, if current_op is not an operator: increment count by 1 to extend the current operand character span RESULT count]
            [COMMAND After the for loop exits: if count is greater than 1, or count equals 1 and arr[current_index] is not an operator, join arr[current_index : current_index + count] and append it to self.postfix_stack as the final trailing operand RESULT self.postfix_stack]
            [COMMAND After the trailing operand flush: pop remaining operators from op_stack and append each as a string to self.postfix_stack in a while loop that continues as long as op_stack[-1] is not the sentinel ',' RESULT self.postfix_stack, op_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The for loop body reaches a character that is not an operator]
        [SEQUENTIAL_BLOCK]
            [COMMAND Increment count by 1 to extend the accumulated operand span; skip all operator-handling logic for this iteration RESULT count]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the for loop, count is 0 or count equals 1 and arr[current_index] is itself an operator]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the trailing operand append; proceed directly to the final operator-drain while loop RESULT self.postfix_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Normalizes '%' to '/' in both operator characters, then returns true if the incoming operator 'peek' has precedence greater than or equal to the current operator 'cur' according to a priority lookup table." compare]
    [INPUTS]
        <REF> cur </REF>: string "Single character representing the current operator token; '%' is treated as an alias for '/'."
        <REF> peek </REF>: string "Single character representing the next (incoming) operator token; '%' is treated as an alias for '/'."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the precedence of 'peek' is greater than or equal to the precedence of 'cur'; false otherwise."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether 'cur' equals the character '%'; if so, reassign 'cur' to '/' to normalize it before precedence lookup. RESULT cur_normalized]
            [COMMAND Check whether 'peek' equals the character '%'; if so, reassign 'peek' to '/' to normalize it before precedence lookup. RESULT peek_normalized]
            [COMMAND Convert the (possibly normalized) character 'peek' to its Unicode code point via ord(), subtract the fixed offset 40 (ASCII code of '(') to obtain a zero-based index, and retrieve the corresponding precedence integer from self.operat_priority. RESULT peek_precedence]
            [COMMAND Convert the (possibly normalized) character 'cur' to its Unicode code point via ord(), subtract the fixed offset 40 to obtain a zero-based index, and retrieve the corresponding precedence integer from self.operat_priority. RESULT cur_precedence]
            [COMMAND Compare peek_precedence >= cur_precedence and return the resulting boolean as the method's return value. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: 'cur' is '%' on entry]
        [SEQUENTIAL_BLOCK]
            [COMMAND Reassign 'cur' to '/' so that subsequent ord() and index arithmetic use the precedence entry for '/' rather than '%'. RESULT cur_normalized]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: 'peek' is '%' on entry]
        [SEQUENTIAL_BLOCK]
            [COMMAND Reassign 'peek' to '/' so that subsequent ord() and index arithmetic use the precedence entry for '/' rather than '%'. RESULT peek_normalized]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Performs one of five Decimal-precision arithmetic operations on two operands based on an operator token, raising a ValueError for any unrecognized operator." _calculate]
    [INPUTS]
        <REF> first_value </REF>: numeric or string "The left-hand operand; coerced to Decimal at execution time inside each branch"
        <REF> second_value </REF>: numeric or string "The right-hand operand; coerced to Decimal at execution time inside each branch"
        <REF> current_op </REF>: string "Single-character operator token that controls which arithmetic operation is applied; must be one of '+', '-', '*', '/', '%'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: Decimal "The Decimal result of applying the requested arithmetic operation to first_value and second_value"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Test whether current_op equals '+'; if so, return Decimal(first_value) + Decimal(second_value) RESULT result]
            [COMMAND Otherwise test whether current_op equals '-'; if so, return Decimal(first_value) - Decimal(second_value) RESULT result]
            [COMMAND Otherwise test whether current_op equals '*'; if so, return Decimal(first_value) * Decimal(second_value) RESULT result]
            [COMMAND Otherwise test whether current_op equals '/'; if so, return Decimal(first_value) / Decimal(second_value) RESULT result]
            [COMMAND Otherwise test whether current_op equals '%'; if so, return Decimal(first_value) % Decimal(second_value) RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: current_op does not match any of the five recognized operator characters ('+', '-', '*', '/', '%')]
        [LOG "Unexpected operator token encountered: the value of current_op is embedded in the error message via str.format"]
        [THROW ValueError "Raised with the message 'Unexpected operator: <current_op>' to signal that the caller supplied an operator token that the method does not handle"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Preprocesses a mathematical expression string by stripping whitespace and trailing equals signs, then converting unary minus signs to tildes, and finally rewriting a leading negated parenthesis as a subtraction from zero." transform]
    [INPUTS]
        <REF> expression </REF>: string "A mathematical expression string that may contain whitespace, a trailing equals sign, and minus signs that are either unary or binary."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A normalized expression string with whitespace removed, trailing equals stripped, unary minuses replaced by tildes, and any leading negated parenthesized subexpression rewritten as 0-(...)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Remove all whitespace characters (spaces, tabs, newlines) from expression using a regex substitution matching one or more whitespace characters with an empty string. RESULT expression]
            [COMMAND Remove a trailing equals sign from expression, if present, using a regex substitution matching a literal '=' anchored at the end of the string with an empty string; at most one character is removed. RESULT expression]
            [COMMAND Split expression into a list of individual characters. RESULT arr]
            [COMMAND Iterate over arr with index i and character c using enumerate. For each character, if c is not '-', skip it. If c equals '-' and i equals 0, replace arr[0] with '~'. If c equals '-' and i is greater than 0, read the preceding character arr[i-1] into prev_c; if prev_c is one of '+', '-', '*', '/', '(', 'E', or 'e', replace arr[i] with '~'; otherwise leave arr[i] unchanged. RESULT arr]
            [COMMAND After the loop, check whether arr[0] equals '~' AND the length of arr is greater than 1 AND arr[1] equals '('. If this compound condition is true, execute the true branch; otherwise execute the false branch. RESULT branch_selector]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: arr[0] equals '~' and len(arr) is greater than 1 and arr[1] equals '(' — the expression begins with a unary negation applied to a parenthesized subexpression.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Replace arr[0] with '-', converting the tilde back to a minus sign. RESULT arr]
            [COMMAND Join arr into a string and prepend the literal '0', producing a string of the form '0-(...)', then return it immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The compound condition above is false — the expression does not begin with '~(' or arr has only one element.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Join arr into a string without any prefix and return it immediately. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
