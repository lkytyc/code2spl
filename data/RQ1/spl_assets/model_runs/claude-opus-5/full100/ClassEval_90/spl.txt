[DEFINE_WORKER: "Extracts and returns the URL scheme (e.g., 'http', 'https') from the instance's URL string by locating the '://' delimiter, or returns None if no such delimiter is present." get_scheme]
    [INPUTS]
        <REF> self </REF>: object "Instance containing the 'url' attribute, which must be an initialized string representing a URL"
    [END_INPUTS]

    [OUTPUTS]
        <REF> scheme </REF>: string or None "The substring of self.url before '://' if found (e.g., 'http', 'ftp'), or None if '://' is absent"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search self.url for the first occurrence of the substring '://' using str.find, which returns the zero-based index of ':' if found or -1 if not found; assign the result to scheme_end RESULT scheme_end]
            [COMMAND Check whether scheme_end is not equal to -1, meaning '://' was found at a valid position within self.url RESULT found_delimiter]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: scheme_end != -1: '://' was found at a valid non-negative index in self.url]
        [SEQUENTIAL_BLOCK]
            [COMMAND Slice self.url from index 0 up to (but not including) scheme_end to extract the scheme portion, then immediately return that substring as the method result RESULT return self.url[:scheme_end]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: scheme_end == -1: '://' was not found anywhere in self.url, meaning the URL has no recognizable scheme delimiter]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None to signal that no scheme could be identified RESULT return None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts and returns the hostname from a URL string stored in self.url by locating the scheme separator and the first path slash, returning null if no scheme is present." get_host]
    [INPUTS]
        <REF> self.url </REF>: string "The URL string from which the hostname will be extracted; must be a non-null string for safe execution"
    [END_INPUTS]

    [OUTPUTS]
        <REF> host </REF>: string or None "The hostname portion of the URL (e.g. 'example.com'), or None if no scheme separator '://' is found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search for the first occurrence of '://' in self.url using str.find and store the resulting integer index (or -1 if not found) in scheme_end RESULT scheme_end]
            [COMMAND Check whether scheme_end is not equal to -1, meaning a scheme separator was found RESULT scheme_present]
            [COMMAND If scheme_present is true, slice self.url from index scheme_end + 3 to the end to strip the scheme and the '://' separator, storing the remainder in url_without_scheme RESULT url_without_scheme]
            [COMMAND Within the scheme-present branch, search for the first occurrence of '/' in url_without_scheme using str.find and store the result in host_end RESULT host_end]
            [COMMAND Within the scheme-present branch, check whether host_end is not equal to -1, meaning a path separator was found RESULT path_present]
            [COMMAND Within the scheme-present branch and when path_present is true, return the slice url_without_scheme[:host_end], which excludes the '/' and isolates the host RESULT return_value]
            [COMMAND Within the scheme-present branch and when path_present is false (no '/' found), return url_without_scheme in full because the entire remainder is the host with no path component RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: scheme_end equals -1, meaning '://' was not found in self.url (e.g. a relative path or scheme-less string)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None, as no scheme boundary exists and the host cannot be determined RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts the path component from a URL string by locating the scheme separator '://' and then finding the first '/' after the host, returning the path including its leading slash, or null if the URL lacks a scheme or has no path." get_path]
    [INPUTS]
        <REF> self.url </REF>: str "The full URL string stored on the instance from which the path component is to be extracted."
    [END_INPUTS]

    [OUTPUTS]
        <REF> path </REF>: str or None "The path portion of the URL starting from and including the leading '/', or None if no scheme separator or no path separator is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search self.url for the first occurrence of the substring '://' using str.find, which returns the 0-based index of ':' if found or -1 if absent; assign the result to scheme_end. RESULT scheme_end]
            [COMMAND Check whether scheme_end is not equal to -1, meaning a scheme separator was found; if the condition is false, skip all nested steps and proceed to the fallback return. RESULT outer_condition_passed]
            [COMMAND Within the outer guard (scheme_end != -1), slice self.url from index scheme_end + 3 to the end, thereby stripping the scheme and the literal '://' characters; assign the result to url_without_scheme. RESULT url_without_scheme]
            [COMMAND Within the outer guard, search url_without_scheme for the first occurrence of '/' using str.find, which returns the 0-based index if found or -1 if absent; assign the result to host_end. RESULT host_end]
            [COMMAND Within the outer guard, check whether host_end is not equal to -1, meaning a path separator was found in the scheme-stripped string; if the condition is false, skip the inner return and fall through to the fallback return. RESULT inner_condition_passed]
            [COMMAND Within both guards (scheme_end != -1 and host_end != -1), slice url_without_scheme from index host_end to the end to obtain the path including its leading '/', then immediately return that string as an early exit from the method. RESULT return url_without_scheme[host_end:]]
            [COMMAND If execution reaches this point — because either the scheme separator was absent or the path separator was absent — return None as the default result. RESULT return None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.url does not contain '://' (scheme_end equals -1), so no scheme is present in the URL.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the entire nested block; fall through directly to the unconditional return of None. RESULT return None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.url contains '://' but url_without_scheme contains no '/' (host_end equals -1), meaning the URL has a scheme and host but no path component (e.g., 'https://example.com').]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the outer guard, after computing url_without_scheme and finding host_end to be -1, skip the inner return; fall through to the unconditional return of None. RESULT return None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses the query string of a URL stored in self.url into a key-value dictionary, returning the dict (possibly empty) when a '?' delimiter is found, or None when no query string is present." get_query_params]
    [INPUTS]
        <REF> self </REF>: object "The containing object whose 'url' string attribute is read to extract query parameters"
    [END_INPUTS]

    [OUTPUTS]
        <REF> params </REF>: dict or None "A dictionary of query parameter key-value pairs parsed from the URL, an empty dict if the query string is present but empty or contains no well-formed pairs, or None if no '?' delimiter exists in the URL"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Find the index of the first '?' character in self.url using str.find; store the result as query_start, which is -1 if '?' is absent RESULT query_start]
            [COMMAND Find the index of the first '#' character in self.url using str.find; store the result as fragment_start, which is -1 if '#' is absent RESULT fragment_start]
            [COMMAND Check whether query_start is not equal to -1 to determine whether a query string delimiter exists; if the condition is false, skip to the final return RESULT condition_query_exists]
            [COMMAND When query_start != -1, extract the substring of self.url from index query_start + 1 (exclusive of '?') up to fragment_start (exclusive); when fragment_start is -1, Python slice semantics make the slice end one character before the last character rather than at the true end, which is a known boundary concern when '#' is absent RESULT query_string]
            [COMMAND Initialize params as an empty dictionary unconditionally within the query-exists branch RESULT params]
            [COMMAND Check whether the length of query_string is greater than 0; if the query string is empty (e.g., URL ends with '?'), skip the splitting loop and proceed directly to the return of the empty params dict RESULT condition_query_nonempty]
            [COMMAND When query_string is non-empty, split query_string on '&' to produce a list of raw key=value segment strings RESULT param_pairs]
            [COMMAND Iterate over each segment in param_pairs; for each segment, split the segment on '=' with no maxsplit argument, producing a list of parts RESULT key_value]
            [COMMAND Within the loop, check whether the length of key_value equals exactly 2; segments with no '=' (length 1) or with more than one '=' such as base64-encoded values (length > 2) are silently discarded and do not produce an entry in params RESULT condition_well_formed_pair]
            [COMMAND Within the loop, when len(key_value) == 2, unpack key_value into key and value and store the mapping params[key] = value; later duplicate keys overwrite earlier ones RESULT params]
            [COMMAND After the loop completes (all segments have been visited; no break or continue exists), return params, which may be empty or populated RESULT return_params]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: query_start == -1, meaning no '?' character was found anywhere in self.url]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the entire query-parsing block and return None to signal that no query string is present RESULT return_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: query_start != -1 but query_string has length 0, meaning the URL contains '?' but nothing follows it before the fragment marker or end of string]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the split-and-loop body entirely and return the empty params dictionary immediately RESULT return_empty_params]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Extracts the fragment component from the instance's URL string by locating the first '#' character and returning everything after it, or None if no '#' is present." get_fragment]
    [INPUTS]
        <REF> self </REF>: object "Instance containing a 'url' string attribute to be searched for a fragment delimiter"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string or None "The substring of self.url after the first '#' character, or None if no '#' exists in self.url"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Search self.url for the first occurrence of the '#' character using str.find, which returns the zero-based index of the match or -1 if not found, and assign the result to fragment_start RESULT fragment_start]
            [COMMAND Check whether fragment_start is not equal to -1, distinguishing a found '#' (any non-negative index) from the sentinel value indicating absence RESULT has_fragment]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: fragment_start != -1: a '#' character was found at a valid index within self.url (including index 0, which yields an empty fragment body, and the last index, which also yields an empty fragment body)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Slice self.url from the position immediately after fragment_start (fragment_start + 1) to the end of the string to obtain the fragment body, then return it as the method result RESULT fragment_body]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: fragment_start == -1: no '#' character exists in self.url (including the case where self.url is an empty string)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None to signal that the URL contains no fragment component; this early return exits the method without performing any slice operation, explicitly avoiding the Python negative-index pitfall where slicing with -1 would silently drop the last character RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
