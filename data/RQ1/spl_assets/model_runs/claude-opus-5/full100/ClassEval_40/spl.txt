# CLASS FitnessTracker

[DEFINE_WORKER: "Initializes a health-profile instance by binding four caller-supplied parameters to instance attributes and setting a hardcoded BMI normal-range lookup table keyed by sex." __init__]
    [INPUTS]
        <REF> height </REF>: any "The individual's height value, passed through directly with no transformation"
        <REF> weight </REF>: any "The individual's weight value, passed through directly with no transformation"
        <REF> age </REF>: any "The individual's age value, passed through directly with no transformation"
        <REF> sex </REF>: str "The individual's sex; expected to be exactly 'male' or 'female' to match keys in BMI_std; no validation is performed at assignment time"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Bind the parameter height to the instance attribute self.height with no transformation RESULT self.height]
            [COMMAND Bind the parameter weight to the instance attribute self.weight with no transformation RESULT self.weight]
            [COMMAND Bind the parameter age to the instance attribute self.age with no transformation RESULT self.age]
            [COMMAND Bind the parameter sex to the instance attribute self.sex with no transformation; downstream logic implicitly requires this value to be 'male' or 'female' RESULT self.sex]
            [COMMAND Assign the hardcoded literal list [{"male": [20, 25]}, {"female": [19, 24]}] to self.BMI_std, encoding the inclusive normal BMI range [lower, upper] for each sex key; male lower bound is 20, upper bound is 25; female lower bound is 19, upper bound is 24; this value is identical across all instances and has no runtime dependency RESULT self.BMI_std]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes and returns the Body Mass Index by dividing the instance's weight by the square of its height." get_BMI]
    [INPUTS]
        <REF> self.weight </REF>: float | int "The weight of the individual, read from the instance attribute; must be numeric and ideally non-negative"
        <REF> self.height </REF>: float | int "The height of the individual, read from the instance attribute; must be numeric and non-zero to avoid ZeroDivisionError"
    [END_INPUTS]

    [OUTPUTS]
        <REF> bmi </REF>: float "The computed BMI value as a float (Python 3 true division); not stored on the instance, returned directly to the caller"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.height and compute its square using exponentiation (self.height ** 2); exponentiation is evaluated first due to operator precedence RESULT height_squared]
            [COMMAND Divide self.weight by height_squared using Python 3 true division to produce a float BMI value RESULT bmi]
            [COMMAND Return bmi directly to the caller without storing it on the instance or producing any side effects RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.height is 0 at the time of the call]
        [LOG "Division by zero: self.height evaluated to 0, making height_squared equal to 0 and causing the division to fail"]
        [THROW ZeroDivisionError "Raised by Python when self.weight is divided by a height_squared value of 0; not caught here and propagates to the caller"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.weight or self.height is None, a string, or any other non-numeric type]
        [LOG "Type mismatch: the exponentiation or division operator received a non-numeric operand"]
        [THROW TypeError "Raised by Python when the ** or / operator is applied to an incompatible type; propagates to the caller"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.weight or self.height has never been assigned on the instance]
        [LOG "Missing attribute: one of the required instance attributes does not exist on self"]
        [THROW AttributeError "Raised by Python when attribute lookup on self fails because weight or height was never set; propagates to the caller"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Classifies a person's BMI as above normal (1), below normal (-1), or within the normal range (0) by comparing the computed BMI against a sex-appropriate reference range." condition_judge]
    [INPUTS]
        <REF> self </REF>: object "Instance providing get_BMI(), sex (string), and BMI_std (list of two dicts with keys 'male' and 'female', each mapping to a two-element numeric range [lower, upper])"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "1 if BMI strictly exceeds the upper bound, -1 if BMI is strictly below the lower bound, or 0 if BMI falls within the closed interval [lower, upper]"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_BMI() and assign the return value to BMI; no transformation or null guard is applied at the call site RESULT BMI]
            [COMMAND Check whether self.sex equals the string 'male' exactly RESULT sex_is_male]
            [COMMAND If sex_is_male is true, assign self.BMI_std[0]['male'] to BMI_range; otherwise assign self.BMI_std[1]['female'] to BMI_range (any value of self.sex that is not 'male' — including None or unexpected strings — falls to the female branch) RESULT BMI_range]
            [COMMAND If BMI is strictly greater than BMI_range[1] (upper bound exclusive of 'too fat' boundary), return 1 immediately RESULT early_return_1]
            [COMMAND Else if BMI is strictly less than BMI_range[0] (lower bound exclusive of 'too thin' boundary), return -1 immediately RESULT early_return_minus1]
            [COMMAND Otherwise (BMI is within the closed interval [BMI_range[0], BMI_range[1]], inclusive of both boundary values), return 0 RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Calculates daily calorie intake by first computing the Mifflin-St. Jeor BMR based on biological sex, then scaling it by an activity multiplier derived from the result of condition_judge()." calculate_calorie_intake]
    [INPUTS]
        <REF> self.sex </REF>: string "Biological sex of the individual; expected value is exactly 'male' (case-sensitive) or any other string treated as female"
        <REF> self.weight </REF>: number "Body weight in kilograms"
        <REF> self.height </REF>: number "Height in centimeters"
        <REF> self.age </REF>: number "Age in years"
    [END_INPUTS]

    [OUTPUTS]
        <REF> calorie_intake </REF>: number "Estimated daily calorie intake in kilocalories after applying the appropriate activity multiplier to BMR"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether self.sex equals the string 'male' using exact case-sensitive equality RESULT sex_condition]
            [COMMAND If sex_condition is true, compute BMR as (10 * self.weight) + (6.25 * self.height) - (5 * self.age) + 5 (male constant) RESULT BMR]
            [COMMAND If sex_condition is false, compute BMR as (10 * self.weight) + (6.25 * self.height) - (5 * self.age) - 161 (female constant) RESULT BMR]
            [COMMAND Call self.condition_judge() to obtain the activity level indicator; compare the return value to the integer 1 using exact equality RESULT activity_condition_sedentary]
            [COMMAND If activity_condition_sedentary is true, set calorie_intake to BMR * 1.2 (sedentary lifestyle multiplier) RESULT calorie_intake]
            [COMMAND If activity_condition_sedentary is false, call self.condition_judge() a second time and compare the new return value to the integer -1 using exact equality RESULT activity_condition_active]
            [COMMAND If activity_condition_active is true, set calorie_intake to BMR * 1.6 (active lifestyle multiplier) RESULT calorie_intake]
            [COMMAND If activity_condition_active is false (else branch), set calorie_intake to BMR * 1.4 (moderate lifestyle multiplier) RESULT calorie_intake]
            [COMMAND Return calorie_intake RESULT calorie_intake]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.sex is any string other than exactly 'male' (including 'Male', 'MALE', or empty string) — falls to else branch silently]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute BMR using the female Mifflin-St. Jeor formula: (10 * self.weight) + (6.25 * self.height) - (5 * self.age) - 161 RESULT BMR]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.condition_judge() returns a value other than 1 on the first call, and other than -1 on the second call (e.g., 0, 2, None, or any unexpected value)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set calorie_intake to BMR * 1.4, treating the unrecognized activity indicator as a moderate lifestyle by default RESULT calorie_intake]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.condition_judge() is non-deterministic or has side effects: the first call (for the if-branch check) and the second call (for the elif-branch check) may return different values, potentially causing the elif branch to match a different state than was evaluated in the if check]
        [SEQUENTIAL_BLOCK]
            [COMMAND The second call to self.condition_judge() is evaluated independently; if its return value is -1 regardless of what the first call returned, set calorie_intake to BMR * 1.6 RESULT calorie_intake]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
