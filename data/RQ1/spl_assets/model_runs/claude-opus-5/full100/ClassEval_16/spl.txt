# CLASS Calculator

    [END_INPUTS]

    [OUTPUTS]

        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new dictionary and assign it to self.operators, mapping the string key '+' to a lambda that returns x + y, the string key '-' to a lambda that returns x - y, the string key '*' to a lambda that returns x * y, the string key '/' to a lambda that returns x / y (true division, no zero guard), and the string key '^' to a lambda that returns x ** y (exponentiation, not bitwise XOR) RESULT self.operators]

[DEFINE_WORKER: "Evaluates an infix arithmetic expression string containing numbers, operators (+, -, *, /, ^), and parentheses using the Shunting-Yard algorithm with two stacks to return the computed numeric result." calculate]
    [INPUTS]
        <REF> expression </REF>: str "Infix arithmetic expression string containing digits, decimal points, operators (+, -, *, /, ^), and parentheses"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float or None "The computed numeric result of the expression, or None if the operand stack is empty after evaluation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize operand_stack as an empty list, operator_stack as an empty list, and num_buffer as an empty string RESULT operand_stack, operator_stack, num_buffer]
            [COMMAND Iterate over each character char in expression one at a time, consuming all characters without early exit RESULT char]
            [COMMAND Within the for loop: if char is a digit (0-9) or a decimal point '.', append char to num_buffer to accumulate a multi-character numeric token RESULT num_buffer]
            [COMMAND Within the for loop, in the else branch (char is not a digit or dot): if num_buffer is non-empty, convert num_buffer to float and push it onto operand_stack, then reset num_buffer to empty string RESULT operand_stack, num_buffer]
            [COMMAND Within the for loop else branch: if char is one of the operators in '+-*/^', enter the precedence-climbing while loop — while operator_stack is non-empty AND the top of operator_stack is not '(' AND self.precedence(top operator) >= self.precedence(char), call self.apply_operator(operand_stack, operator_stack) and update both stacks with the returned values; after the while loop exits, push char onto operator_stack RESULT operand_stack, operator_stack]
            [COMMAND Within the for loop else branch: if char is '(', push '(' onto operator_stack as a precedence-barrier sentinel RESULT operator_stack]
            [COMMAND Within the for loop else branch: if char is ')', run a while loop — while operator_stack is non-empty AND the top of operator_stack is not '(', call self.apply_operator(operand_stack, operator_stack) and update both stacks; after the while loop exits, pop the '(' sentinel off operator_stack RESULT operand_stack, operator_stack]
            [COMMAND After the for loop completes: if num_buffer is non-empty, convert num_buffer to float and push it onto operand_stack (handles a trailing number at the end of the expression that was never flushed during the loop) RESULT operand_stack]
            [COMMAND Drain any remaining operators: while operator_stack is non-empty, call self.apply_operator(operand_stack, operator_stack) and update both stacks with the returned values RESULT operand_stack, operator_stack]
            [COMMAND Return operand_stack[-1] (the top and only remaining element, which is the final result) if operand_stack is non-empty, otherwise return None RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The expression contains only a single numeric token and no operators or parentheses]
        [SEQUENTIAL_BLOCK]
            [COMMAND All characters are digits or dots and are accumulated entirely into num_buffer during the for loop; num_buffer is flushed to operand_stack in the post-loop trailing-number check; the operator-drain while loop body never executes; operand_stack[-1] is returned directly RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The expression string is empty or produces an empty operand_stack after full evaluation]
        [SEQUENTIAL_BLOCK]
            [COMMAND The for loop iterates zero times or all characters are non-numeric non-operator characters that produce no operands; operand_stack remains empty after all phases; return None RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: A closing parenthesis ')' is encountered with no matching opening '(' in operator_stack, causing operator_stack.pop() to be called on an empty list after the inner while loop exhausts all remaining operators]
        [LOG "Mismatched parenthesis: operator_stack is empty when attempting to pop the expected '(' sentinel after processing a ')' character"]
        [THROW IndexError "Raised implicitly by list.pop() on an empty operator_stack when the expression contains an unmatched closing parenthesis"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.apply_operator is called but operand_stack contains fewer than two operands needed to perform a binary operation]
        [LOG "Insufficient operands on operand_stack when apply_operator attempts to pop two values for a binary operator"]
        [THROW IndexError "Raised implicitly when operand_stack underflows during apply_operator due to a malformed expression"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns the numeric precedence level of a given arithmetic operator, defaulting to 0 for any unrecognized operator." precedence]
    [INPUTS]
        <REF> operator </REF>: string "A single-character string representing an arithmetic operator to look up"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "The precedence level of the operator: 1 for + or -, 2 for * or /, 3 for ^, or 0 if the operator is not recognized"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Define a local dictionary named precedences mapping each of the five arithmetic operator strings to their integer precedence tiers: '+' -> 1, '-' -> 1, '*' -> 2, '/' -> 2, '^' -> 3 RESULT precedences]
            [COMMAND Look up the value of operator in precedences using .get(operator, 0), returning the mapped integer if operator is a key, or 0 if it is not present RESULT lookup_result]
            [COMMAND Return lookup_result as the output of the method RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Pops two operands and an operator from their respective stacks, applies the operator function to the operands in left-to-right order, pushes the result back onto the operand stack, and returns both stacks." apply_operator]
    [INPUTS]
        <REF> operand_stack </REF>: list "Stack of numeric or evaluated operands; top element is the right-hand operand"
        <REF> operator_stack </REF>: list "Stack of operator symbols; top element is the operator to apply"
    [END_INPUTS]

    [OUTPUTS]
        <REF> operand_stack </REF>: list "Operand stack after popping two operands and pushing the computed result"
        <REF> operator_stack </REF>: list "Operator stack after popping the applied operator"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop the top element from operator_stack and assign it to operator RESULT operator]
            [COMMAND If operator equals '^': pop the top element from operand_stack and assign it to operand2 (right-hand operand), then pop the next element and assign it to operand1 (left-hand operand), then call self.operators[operator](operand1, operand2) and assign the return value to result, then push result onto operand_stack RESULT operand_stack]
            [COMMAND Otherwise (operator is not '^'): pop the top element from operand_stack and assign it to operand2 (right-hand operand), then pop the next element and assign it to operand1 (left-hand operand), then call self.operators[operator](operand1, operand2) and assign the return value to result, then push result onto operand_stack RESULT operand_stack]
            [COMMAND Return the tuple (operand_stack, operator_stack) RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: operator equals '^' — the exponentiation branch; note that as currently written the body is identical to the else branch, so no behavioral difference exists; if the intent is right-associative exponentiation the operand order or dispatch function should differ]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pop operand2 (right-hand operand) from operand_stack RESULT operand2]
            [COMMAND Pop operand1 (left-hand operand) from operand_stack RESULT operand1]
            [COMMAND Call self.operators['^'](operand1, operand2) and assign to result RESULT result]
            [COMMAND Push result onto operand_stack RESULT operand_stack]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
