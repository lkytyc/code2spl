# CLASS CurrencyConverter

    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign self.rates to a dictionary literal mapping seven currency code strings to their corresponding float exchange rates relative to USD 1.0: USD→1.0, EUR→0.85, GBP→0.72, JPY→110.15, CAD→1.23, AUD→1.34, CNY→6.40 RESULT self.rates]

[DEFINE_WORKER: "Converts a monetary amount from one currency to another using stored exchange rates, returning the converted float value or short-circuiting early for same-currency conversions and unknown currencies." convert]
    [INPUTS]
        <REF> amount </REF>: number "The numeric amount to convert; may be zero or negative, both of which propagate without error"
        <REF> from_currency </REF>: string "The ISO currency code of the source currency"
        <REF> to_currency </REF>: string "The ISO currency code of the target currency"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: number | boolean "The converted float amount on success; the original amount (unchanged) when from_currency equals to_currency; boolean False when either currency is absent from the rates table"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether from_currency equals to_currency; if they are equal, return amount immediately without any conversion RESULT early_return_same_currency]
            [COMMAND Check whether from_currency is present in self.rates AND to_currency is present in self.rates using a short-circuit OR; if either is absent, return False immediately RESULT early_return_unknown_currency]
            [COMMAND Read self.rates[from_currency] and assign it to from_rate RESULT from_rate]
            [COMMAND Read self.rates[to_currency] and assign it to to_rate RESULT to_rate]
            [COMMAND Compute converted_amount as (amount / from_rate) * to_rate, normalizing amount to a common base by dividing by from_rate then scaling to the target currency by multiplying by to_rate RESULT converted_amount]
            [COMMAND Return converted_amount RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: from_currency equals to_currency at method entry]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return amount unchanged without accessing self.rates or performing any arithmetic RESULT amount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: from_currency is not in self.rates OR to_currency is not in self.rates (short-circuit OR: evaluated left to right, stops at first absent key)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return boolean False to signal an unresolvable currency code RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: from_currency is present in self.rates but its stored value is exactly 0]
        [LOG "ZeroDivisionError raised during (amount / from_rate) because from_rate is 0; execution halts before converted_amount is assigned"]
        [THROW ZeroDivisionError "Raised by Python when dividing amount by a from_rate of zero; no guard exists in this method so it propagates to the caller"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Returns a snapshot list of all currency codes currently held in the instance's rates dictionary." get_supported_currencies]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> currencies </REF>: list "A new list containing the keys of self.rates at the moment of the call; independent of the internal dictionary after creation."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Access the instance attribute self.rates to obtain the internal rates dictionary. RESULT rates_dict]
            [COMMAND Call .keys() on rates_dict to produce a dict_keys view reflecting the current key set. RESULT keys_view]
            [COMMAND Pass keys_view to list() to materialize a new, independent list snapshot of the currency codes at this instant. RESULT currencies]
            [COMMAND Return currencies to the caller. RESULT currencies]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.rates does not exist on the instance at call time]
        [LOG "AttributeError raised because self.rates has not been initialized before this method is called."]
        [THROW AttributeError "Raised when self.rates is absent from the instance, typically because __init__ did not set it."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.rates exists but is not a dict or dict-like object with a .keys() method]
        [LOG "AttributeError raised because .keys() is not available on the actual type bound to self.rates."]
        [THROW AttributeError "Raised when the object bound to self.rates does not expose a .keys() method."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a new currency-to-rate mapping into the instance's rates dictionary, rejecting the operation if the currency key already exists." add_currency_rate]
    [INPUTS]
        <REF> currency </REF>: str "The currency identifier used as the dictionary key; must be a hashable type and must not already exist in self.rates"
        <REF> rate </REF>: float "The exchange rate value to associate with the given currency; stored without transformation or validation"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "Returns False if the currency already exists in self.rates; returns None (implicitly) on successful insertion"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency is already a key in self.rates RESULT duplicate_check]
            [COMMAND Write rate into self.rates at key currency, persisting the value on the instance for the object's lifetime RESULT self.rates[currency]]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency is already present as a key in self.rates at the time of the call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, leaving self.rates unmodified and skipping the dictionary write RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Updates the exchange rate for a given currency in the instance-level rates dictionary, returning False if the currency does not exist and True implicitly after a successful direct write." update_currency_rate]
    [INPUTS]
        <REF> currency </REF>: string "The key identifying the currency whose rate is to be updated; must already exist in self.rates for the update to proceed"
        <REF> new_rate </REF>: numeric "The new exchange rate value to assign to the specified currency; taken as-is with no transformation or validation"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "Returns False explicitly when the currency key is not found in self.rates; returns None (implicit) after a successful update"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether currency is NOT a key in self.rates RESULT currency_exists_check]
            [COMMAND Assign new_rate directly to self.rates[currency], overwriting the existing value for that key; the write is unconditional, applies no transformation to new_rate, resolves the key by the current value of currency, and immediately makes the new rate visible to all callers holding a reference to self RESULT updated_rate]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: currency is not present as a key in self.rates (membership test fails)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately, short-circuiting all remaining logic; no write to self.rates occurs RESULT early_return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
