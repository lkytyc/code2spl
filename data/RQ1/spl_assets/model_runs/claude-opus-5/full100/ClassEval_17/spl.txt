[DEFINE_WORKER: "Filters and returns all events from the instance's event list whose calendar date (time-of-day ignored) matches the calendar date of the given date parameter." get_events]
    [INPUTS]
        <REF> self </REF>: object "The instance containing the events list attribute to be filtered"
        <REF> date </REF>: datetime "The target datetime whose calendar date (date component only) is used as the filter criterion"
    [END_INPUTS]

    [OUTPUTS]
        <REF> events_on_date </REF>: list "A list of all event dicts from self.events whose date field, stripped to its calendar date, exactly equals the calendar date of the date parameter; may be empty if no events match"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named events_on_date to accumulate matching events RESULT events_on_date]
            [COMMAND Iterate over every element in self.events, assigning each element to event; for each event, extract the calendar date from event['date'] by calling .date() and compare it with equality (==) to the calendar date obtained by calling .date() on the input parameter date; if the two calendar dates are equal, append the entire event dict to events_on_date; if they are not equal, take no action and continue to the next element; the loop always completes a full linear scan of self.events with no early exit, yielding O(n) complexity RESULT events_on_date]
            [COMMAND Return events_on_date to the caller RESULT events_on_date]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether a proposed time interval [start_time, end_time) overlaps with any existing event in self.events, returning False immediately upon finding a conflict and True if no conflicts exist." is_available]
    [INPUTS]
        <REF> start_time </REF>: comparable "The inclusive start boundary of the proposed time interval to check for availability."
        <REF> end_time </REF>: comparable "The exclusive end boundary of the proposed time interval to check for availability."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the proposed interval does not overlap any existing event; False if at least one conflict is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each event dict in self.events sequentially. RESULT current_event]
            [COMMAND Within the for loop over self.events, evaluate the interval overlap condition: test whether start_time is strictly less than current_event['end_time'] AND end_time is strictly greater than current_event['start_time']; both comparisons use exclusive boundaries, so intervals that share only a single endpoint are not considered overlapping. RESULT overlap_detected]
            [COMMAND Within the for loop, if overlap_detected is True, immediately return False, exiting both the loop and the method without examining any remaining events. RESULT early_return_false]
            [COMMAND After the for loop completes without triggering an early return, return True to indicate no conflicting event was found. RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.events is empty, so the for loop body never executes.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the loop entirely and fall through to return True, indicating the time slot is available. RESULT return_true_empty]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The very first event in self.events satisfies the overlap condition.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False on the first iteration without examining any subsequent events, achieving O(1) best-case performance. RESULT early_return_first_event]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over all 60-minute windows within a given calendar day (00:00–23:59) and returns those that are available according to the scheduler." get_available_slots]
    [INPUTS]
        <REF> date </REF>: date "The calendar date whose day-long time window is to be scanned for available slots; only its year, month, and day attributes are used."
    [END_INPUTS]

    [OUTPUTS]
        <REF> available_slots </REF>: list of tuple(datetime, datetime) "Ordered list of (start, end) datetime pairs representing each 60-minute window within the day that was reported as available; empty if none found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize available_slots as an empty list to accumulate qualifying time windows. RESULT available_slots]
            [COMMAND Construct start_time as a datetime set to midnight (00:00) on the given date using date.year, date.month, and date.day with hardcoded hour=0 and minute=0. RESULT start_time]
            [COMMAND Construct end_time as a datetime set to 23:59 on the given date using date.year, date.month, and date.day with hardcoded hour=23 and minute=59; note this ceiling is exclusive of any slot whose start falls at or after 23:59. RESULT end_time]
            [COMMAND Begin a while loop that continues as long as start_time is strictly less than end_time (exclusive upper boundary); when start_time equals or exceeds end_time the loop does not execute. RESULT loop_entry_condition]
            [COMMAND Within the while loop, compute slot_end_time as start_time plus a fixed timedelta of exactly 60 minutes, representing the end of the current candidate slot. RESULT slot_end_time]
            [COMMAND Within the while loop, call self.is_available(start_time, slot_end_time) to check whether the current 60-minute window is free; this is a conditional filter and the result determines whether the slot is recorded. RESULT is_available_result]
            [COMMAND Within the while loop, if is_available_result is truthy, append the tuple (start_time, slot_end_time) to available_slots; if falsy, skip without recording. RESULT available_slots]
            [COMMAND Within the while loop, unconditionally advance start_time by exactly 60 minutes using timedelta(minutes=60); this happens regardless of availability and guarantees loop termination. RESULT start_time]
            [COMMAND After the while loop exits, return available_slots containing all collected (start, end) tuples for windows that passed the availability check. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: start_time is not strictly less than end_time on the first evaluation (e.g., the constructed start_time equals or exceeds end_time), meaning the loop body is never entered.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return available_slots as an empty list immediately, since no 60-minute windows were evaluated. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.is_available returns falsy for every candidate slot across all iterations.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Complete all loop iterations without appending any tuple, then return available_slots as an empty list. RESULT available_slots]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans the instance's event list and returns up to num_events events whose start time is at or after the current wall-clock time, preserving the original list order." get_upcoming_events]
    [INPUTS]
        <REF> self </REF>: object "The instance owning the events list; must have a non-null 'events' attribute that is an iterable of dicts, each containing a 'start_time' datetime key"
        <REF> num_events </REF>: int "Maximum number of upcoming events to return; when 0 the result is always an empty list; expected to be a non-negative integer"
    [END_INPUTS]

    [OUTPUTS]
        <REF> upcoming_events </REF>: list "A list of event dicts from self.events whose start_time is greater than or equal to now, containing at most num_events entries in original list order"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Capture the current wall-clock instant by calling datetime.now() and assign the result to 'now'; this value is immutable and will not change during the remainder of the method RESULT now]
            [COMMAND Initialize 'upcoming_events' to a new empty list; this is a fresh allocation with no shared references RESULT upcoming_events]
            [COMMAND Begin iterating over self.events, binding each element to 'event' in turn RESULT event]
            [COMMAND Within the for loop, evaluate whether event['start_time'] >= now using an inclusive lower bound, meaning an event whose start_time equals now exactly also qualifies; if true, append 'event' to 'upcoming_events' RESULT upcoming_events]
            [COMMAND Within the for loop, immediately after the append check and unconditionally, evaluate whether len(upcoming_events) == num_events; if the quota is met, execute break to exit the loop early without scanning the remaining elements of self.events RESULT None]
            [COMMAND Return 'upcoming_events' to the caller RESULT upcoming_events]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.events is empty or no event in self.events has a start_time >= now]
        [SEQUENTIAL_BLOCK]
            [COMMAND The for loop body never appends to 'upcoming_events', so the loop completes without triggering the break; return the empty list RESULT upcoming_events]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: num_events is 0 and upcoming_events is empty at the start of iteration]
        [SEQUENTIAL_BLOCK]
            [COMMAND On the first iteration the len check evaluates len([]) == 0 which is true before any append occurs, so break fires immediately and the method returns an empty list RESULT upcoming_events]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.events contains fewer qualifying events than num_events]
        [SEQUENTIAL_BLOCK]
            [COMMAND The quota condition len(upcoming_events) == num_events is never satisfied, the loop runs to exhaustion, and all qualifying events are returned RESULT upcoming_events]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
