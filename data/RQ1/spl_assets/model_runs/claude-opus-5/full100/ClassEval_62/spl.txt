# CLASS NLPDataProcessor

[DEFINE_WORKER: "Constructs and returns a fixed list containing the English articles 'a', 'an', and 'the' for use as stop words." construct_stop_word_list]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> stop_word_list </REF>: list "A list of three English stop word strings: 'a', 'an', and 'the'"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a list containing the string literals 'a', 'an', and 'the', and assign it to stop_word_list RESULT stop_word_list]
            [COMMAND Return stop_word_list RESULT stop_word_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Tokenizes each string in a list by splitting on whitespace and removes any tokens that appear in the provided stop word list, returning a list of filtered token lists." remove_stop_words]
    [INPUTS]
        <REF> string_list </REF>: list[str] "A list of strings to be tokenized and filtered."
        <REF> stop_word_list </REF>: list[str] "A list of words to be removed from each tokenized string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> answer </REF>: list[list[str]] "A list where each element is a list of tokens from the corresponding input string with stop words removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named answer to accumulate the filtered token lists. RESULT answer]
            [COMMAND For each string in string_list, split it on whitespace to produce string_split, a list of word tokens. RESULT string_split]
            [COMMAND Within the outer for loop over each string, iterate over each word in string_split; if the word is present in stop_word_list, remove the first occurrence of that word from string_split in place. Note: this mutates string_split while iterating over it, which causes consecutive stop words to be skipped. RESULT string_split]
            [COMMAND After the inner loop completes for the current string, append string_split to answer. RESULT answer]
            [COMMAND After all strings have been processed, return answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: string_list is empty, so the outer loop body never executes.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return answer as an empty list without entering any loop. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A given string contains no tokens that appear in stop_word_list, so the if-branch inside the inner loop is never taken.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the outer loop for that string, append string_split unmodified to answer. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters a list of strings by building a stop word list from the instance's internal state and removing all stop words from the input collection." process]
    [INPUTS]
        <REF> string_list </REF>: list "The raw collection of strings to be filtered"
    [END_INPUTS]

    [OUTPUTS]
        <REF> words_list </REF>: list "The filtered list of strings with stop words removed"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.construct_stop_word_list() with no arguments to derive the set of stop words from the instance's internal state; assign the result to stop_word_list RESULT stop_word_list]
            [COMMAND Call self.remove_stop_words(string_list, stop_word_list) to filter string_list against stop_word_list, producing the cleaned collection; assign the result to words_list RESULT words_list]
            [COMMAND Return words_list as the final output of the method RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: construct_stop_word_list returns an empty collection]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pass the empty stop_word_list to remove_stop_words; because no words are filtered, words_list will equal string_list unchanged RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: string_list is empty]
        [SEQUENTIAL_BLOCK]
            [COMMAND Pass the empty string_list to remove_stop_words; regardless of the contents of stop_word_list, words_list will be empty RESULT words_list]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
