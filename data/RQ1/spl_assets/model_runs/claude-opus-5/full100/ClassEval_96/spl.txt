[DEFINE_WORKER: "Looks up the given city in the weather data, stores its temperature and weather condition as instance attributes, and returns the temperature (converted to the requested unit if necessary) along with the weather condition, or False if the city is not found." query]
    [INPUTS]
        <REF> weather_list </REF>: dict "A dictionary mapping city names to weather data objects containing 'temperature', 'weather', and 'temperature units' keys"
        <REF> tmp_units </REF>: str "The desired temperature unit for the return value; accepted values are 'celsius' or 'fahrenheit', defaulting to 'celsius'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple or bool "A tuple of (temperature_value, weather_string) when the city is found, with temperature converted to the requested unit if needed; False when the city is not present in weather_list"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the incoming weather_list parameter to the instance attribute self.weather_list RESULT self.weather_list]
            [COMMAND Check whether self.city is absent from weather_list; if it is absent, return False immediately without mutating any other instance state RESULT early_exit_false]
            [COMMAND Because self.city is present in weather_list, read self.weather_list[self.city]['temperature'] and assign it to self.temperature RESULT self.temperature]
            [COMMAND Read self.weather_list[self.city]['weather'] and assign it to self.weather RESULT self.weather]
            [COMMAND Compare self.weather_list[self.city]['temperature units'] against tmp_units; if they are equal, skip conversion and return (self.temperature, self.weather) directly RESULT temperature_weather_tuple]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The stored temperature unit for self.city differs from tmp_units and tmp_units equals 'celsius']
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.fahrenheit_to_celsius() to convert the stored fahrenheit value to celsius and return the result paired with self.weather as a tuple RESULT celsius_weather_tuple]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The stored temperature unit for self.city differs from tmp_units and tmp_units equals 'fahrenheit']
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.celsius_to_fahrenheit() to convert the stored celsius value to fahrenheit and return the result paired with self.weather as a tuple RESULT fahrenheit_weather_tuple]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The stored temperature unit for self.city differs from tmp_units and tmp_units is neither 'celsius' nor 'fahrenheit']
        [SEQUENTIAL_BLOCK]
            [COMMAND Neither inner branch matches; the method falls through with no explicit return, implicitly returning None RESULT implicit_none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a Celsius temperature stored in the instance attribute to its Fahrenheit equivalent using the standard linear formula (scale by 9/5, then add 32)." celsius_to_fahrenheit]
    [INPUTS]
        <REF> self.temperature </REF>: int | float "The Celsius temperature value read from the instance; its type determines whether the result is an integer or a float"
    [END_INPUTS]

    [OUTPUTS]
        <REF> fahrenheit </REF>: int | float "The computed Fahrenheit value; a float when self.temperature is a float, or when Python 3 float division applies; result type mirrors the input type under Python 2 integer division if self.temperature is an int"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.temperature from the instance; this is the sole input and is not mutated at any point RESULT celsius_value]
            [COMMAND Multiply celsius_value by 9, then divide the product by 5 (left-to-right evaluation: (celsius_value * 9) / 5); note that under Python 2 with an integer input this performs integer division, potentially truncating the result, whereas Python 3 always yields a float RESULT scaled_value]
            [COMMAND Add the constant 32 to scaled_value to apply the freezing-point offset RESULT fahrenheit_value]
            [COMMAND Return fahrenheit_value immediately; no state is written back to self and there are no side effects RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts the instance's Fahrenheit temperature value to Celsius using the standard linear formula (F − 32) × 5/9 and returns the result as a float." fahrenheit_to_celsius]
    [INPUTS]
        <REF> self.temperature </REF>: numeric (int or float) "The Fahrenheit temperature value stored on the instance; must be numeric — None or a non-numeric type will raise a TypeError at the subtraction step."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "The Celsius equivalent of self.temperature, computed as (self.temperature - 32) * 5/9; always a float in Python 3 due to true division."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the instance attribute self.temperature as the sole numeric input. RESULT fahrenheit_value]
            [COMMAND Subtract the constant 32 from fahrenheit_value to shift the Fahrenheit scale to a zero-anchored intermediate value. RESULT shifted_value]
            [COMMAND Multiply shifted_value by the ratio 5/9 (true division yielding 0.555...) to scale it to Celsius. RESULT celsius_value]
            [COMMAND Return celsius_value to the caller; no state is written back to self. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.temperature is None or a non-numeric type, causing the subtraction self.temperature - 32 to fail.]
        [LOG "TypeError raised at the subtraction step because self.temperature does not support numeric subtraction; no guard or default is present in this method."]
        [THROW TypeError "Propagates unhandled when self.temperature is not a numeric type; the caller is responsible for ensuring a valid numeric value before invoking this method."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
