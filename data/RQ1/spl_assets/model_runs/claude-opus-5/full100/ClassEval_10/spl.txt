# CLASS BinaryDataProcessor

[DEFINE_WORKER: "Initializes the object by storing the provided binary string and immediately sanitizing it by removing any non-binary characters." __init__]
    [INPUTS]
        <REF> binary_string </REF>: string "The raw input string intended to represent a binary sequence, potentially containing non-binary characters"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of binary_string parameter to the instance attribute self.binary_string RESULT self.binary_string]
            [COMMAND Call self.clean_non_binary_chars() to remove any characters from self.binary_string that are not valid binary digits (0 or 1) RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sanitizes the instance's binary_string attribute by removing every character that is not '0' or '1', then writes the cleaned result back to the same attribute." clean_non_binary_chars]
    [INPUTS]
        <REF> self </REF>: object "Instance owning the binary_string attribute to be sanitized; binary_string must exist and be iterable (typically a str) before this method is called"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the current value of self.binary_string as the source string for filtering RESULT source_string]
            [COMMAND Apply filter over each character x in source_string, retaining only characters where x is exactly '0' or '1'; all other characters are discarded RESULT filtered_iterator]
            [COMMAND Concatenate all retained characters from filtered_iterator into a new str using empty-string join with no separator RESULT cleaned_string]
            [COMMAND Assign cleaned_string back to self.binary_string, completing the in-place read-modify-write on the instance attribute RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.binary_string is already composed entirely of '0' and '1' characters]
        [SEQUENTIAL_BLOCK]
            [COMMAND All characters pass the filter predicate unchanged; the join produces a string identical to the original value and assigns it back to self.binary_string RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.binary_string is an empty string ""]
        [SEQUENTIAL_BLOCK]
            [COMMAND filter produces an empty iterator; ''.join yields ""; self.binary_string remains "" RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.binary_string contains no '0' or '1' characters (all characters are non-binary)]
        [SEQUENTIAL_BLOCK]
            [COMMAND filter discards every character; ''.join yields ""; self.binary_string is set to "" RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.binary_string contains mixed content such as "a1b0"]
        [SEQUENTIAL_BLOCK]
            [COMMAND filter retains only the '0' and '1' characters in their original order; ''.join concatenates them; the result (e.g. "10" for input "a1b0") is assigned to self.binary_string RESULT self.binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.binary_string is None or any non-iterable type]
        [LOG "filter raises TypeError because None and non-iterables cannot be iterated character by character"]
        [THROW TypeError "Raised by the built-in filter call when self.binary_string is not iterable; no default value is set and no fallback is provided"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.binary_string attribute does not exist on the instance at the time this method executes]
        [LOG "Attribute lookup for self.binary_string raises AttributeError before filter is ever called"]
        [THROW AttributeError "Raised when self.binary_string is undefined; the method performs no initialisation and provides no guard against a missing attribute"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the proportion of zero bits, proportion of one bits, and total bit length of the instance's binary string, returning all three values in a dictionary." calculate_binary_info]
    [INPUTS]
        <REF> self.binary_string </REF>: str "Instance attribute; a string composed of '0' and '1' characters whose statistics are to be computed"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: dict "Dictionary with keys 'Zeroes' (float proportion of zero characters), 'Ones' (float proportion of one characters), and 'Bit length' (int total character count)"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Count the number of '0' characters in self.binary_string using str.count and assign the integer result to zeroes_count RESULT zeroes_count]
            [COMMAND Count the number of '1' characters in self.binary_string using str.count and assign the integer result to ones_count RESULT ones_count]
            [COMMAND Compute the total length of self.binary_string using len and assign the integer result to total_length RESULT total_length]
            [COMMAND Divide zeroes_count by total_length (true division; result is a float proportion in [0.0, 1.0], not a percentage) and assign to zeroes_percentage RESULT zeroes_percentage]
            [COMMAND Divide ones_count by total_length (true division; result is a float proportion in [0.0, 1.0], not a percentage) and assign to ones_percentage RESULT ones_percentage]
            [COMMAND Return a dictionary with key 'Zeroes' mapped to zeroes_percentage, key 'Ones' mapped to ones_percentage, and key 'Bit length' mapped to total_length RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: self.binary_string is an empty string, causing total_length to equal 0 and the division operations for zeroes_percentage and ones_percentage to encounter a zero denominator]
        [LOG "ZeroDivisionError raised at zeroes_percentage or ones_percentage assignment because total_length is 0; no guard exists in the current implementation"]
        [THROW ZeroDivisionError "Raised by Python when dividing by zero; occurs here when self.binary_string is empty and total_length resolves to 0"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts a binary string stored in the instance into a bytearray by parsing it as 8-bit chunks and then decodes the resulting bytes as an ASCII string." convert_to_ascii]
    [INPUTS]
        <REF> self </REF>: object "Instance containing binary_string, a string of '0' and '1' characters whose length is expected to be a multiple of 8"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "ASCII-decoded string produced from the binary representation stored in self.binary_string"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty bytearray and assign it to byte_array RESULT byte_array]
            [COMMAND Iterate over index i stepping through positions 0, 8, 16, ... up to but not including len(self.binary_string), advancing by exactly 8 each iteration; if len(self.binary_string) is not a multiple of 8 the final partial chunk is silently ignored RESULT i]
            [COMMAND Within the for loop, extract the 8-character slice self.binary_string[i:i+8] and assign it to byte RESULT byte]
            [COMMAND Within the for loop, parse byte as a base-2 (binary) integer and assign the resulting decimal value to decimal RESULT decimal]
            [COMMAND Within the for loop, append decimal to byte_array RESULT byte_array]
            [COMMAND After the loop completes, decode byte_array using the ASCII codec and return the resulting string RESULT ascii_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(self.binary_string) is not a multiple of 8]
        [SEQUENTIAL_BLOCK]
            [COMMAND The trailing characters beyond the last complete 8-character group are silently dropped by the range step; no error is raised and no partial byte is processed RESULT byte_array]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: One or more 8-character chunks contain characters other than '0' or '1', causing int(byte, 2) to raise a ValueError]
        [LOG "ValueError raised during base-2 integer conversion of a chunk that is not a valid binary string"]
        [THROW ValueError "Raised implicitly by int(byte, 2) when a chunk contains non-binary characters; not caught by this method"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: A decimal value in byte_array falls outside the valid ASCII range (0-127), causing byte_array.decode('ascii') to raise a UnicodeDecodeError]
        [LOG "UnicodeDecodeError raised when decoding byte values that have no ASCII representation"]
        [THROW UnicodeDecodeError "Raised implicitly by the decode call when any byte value exceeds 127; not caught by this method"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Decodes a binary string stored on the instance by splitting it into 8-character chunks, converting each chunk from base-2 to an integer to build a bytearray, and then decoding that bytearray as UTF-8 text." convert_to_utf8]
    [INPUTS]
        <REF> self </REF>: object "Instance carrying binary_string, a string of '0' and '1' characters whose length should be a multiple of 8 for well-formed input"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "UTF-8 decoded string produced from the binary representation stored in self.binary_string"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty bytearray and assign it to byte_array RESULT byte_array]
            [COMMAND Iterate over integer indices i produced by range(0, len(self.binary_string), 8), stepping by 8 from 0 (inclusive) up to but not including len(self.binary_string); for each i, slice self.binary_string from i to i+8 to obtain a binary digit group of up to 8 characters and assign it to byte RESULT byte]
            [COMMAND Within the for loop, convert byte from a base-2 string to a Python int and assign it to decimal RESULT decimal]
            [COMMAND Within the for loop, append decimal to byte_array RESULT byte_array]
            [COMMAND After the loop completes, decode byte_array using the UTF-8 codec and return the resulting string RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(self.binary_string) is not a multiple of 8, causing the final slice self.binary_string[i:i+8] to contain fewer than 8 characters]
        [SEQUENTIAL_BLOCK]
            [COMMAND int(byte, 2) still parses the short binary fragment, producing an integer smaller than would be produced by a full 8-character byte; that integer is appended to byte_array, making the final byte a partial byte whose validity depends on whether it forms a legal UTF-8 sequence RESULT byte_array]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: byte_array.decode('utf-8') is called and the accumulated bytes do not form valid UTF-8, for example because a partial final byte broke a multi-byte sequence]
        [LOG "UnicodeDecodeError raised by the UTF-8 codec when the bytearray contains an invalid or incomplete byte sequence"]
        [THROW UnicodeDecodeError "Raised by Python's UTF-8 codec if the bytearray built from the binary string cannot be decoded as valid UTF-8 text"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
