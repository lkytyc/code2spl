[DEFINE_WORKER: "Validates that the instance's ip_address string represents a well-formed IPv4 address by confirming it splits into exactly four dot-separated segments each containing a non-negative integer no greater than 255." is_valid]
    [INPUTS]
        <REF> self </REF>: object "Instance whose ip_address attribute holds the string to validate"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if ip_address is a valid IPv4 address; False otherwise"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Split self.ip_address on every literal '.' character using str.split('.') with no limit argument and assign the resulting list of strings to octets RESULT octets]
            [COMMAND Check whether the length of octets is exactly 4; if it is not, return False immediately as an early exit from the method RESULT early_return_if_not_four_segments]
            [COMMAND Iterate over each element octet in octets in order; for each octet, evaluate the compound short-circuit condition: first test whether octet.isdigit() returns False (covering empty strings, signs, decimals, and whitespace); only if isdigit() returns True, then test whether int(octet) is less than 0 or int(octet) is greater than 255 (inclusive boundaries: 0 and 255 are both valid); if any sub-condition is true, return False immediately as an early exit, skipping all remaining octets RESULT early_return_if_invalid_octet]
            [COMMAND After the loop completes without triggering any early return, return True to indicate the address is a valid IPv4 address RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: octets has a length other than 4 after splitting on '.']
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without inspecting any octet values RESULT return_false_wrong_segment_count]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Any octet string fails isdigit() — meaning it is empty, contains a sign character, a decimal point, whitespace, or any non-digit character]
        [SEQUENTIAL_BLOCK]
            [COMMAND Short-circuit the or expression: skip the int() conversion entirely and return False immediately RESULT return_false_non_digit_octet]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: An octet passes isdigit() but its integer value is strictly less than 0 or strictly greater than 255]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately; note that 0 and 255 are inclusive valid boundaries and do not trigger this path RESULT return_false_out_of_range_octet]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

    [END_INPUTS]

            [COMMAND Call self.is_valid() to determine whether the current object is in a valid state RESULT is_valid_result]

        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Converts an IP address object to its dotted binary string representation (e.g., '11000000.10101000.00000001.00000001') by formatting each octet as a zero-padded 8-bit binary value, returning an empty string if the object fails a validity check." get_binary]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Dotted binary representation of the IP address (e.g., '11000000.10101000.00000001.00000001'), or an empty string if the object is not valid"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.is_valid() to check whether the IP address object is in a valid state RESULT is_valid_result]
            [COMMAND If is_valid_result is truthy, initialize an empty list named binary_octets to serve as an accumulator for the per-octet binary strings RESULT binary_octets]
            [COMMAND Call self.get_octets() to retrieve the iterable of IP address octets (e.g., the four decimal segments of an IPv4 address) RESULT octets_iterable]
            [COMMAND For each octet in octets_iterable, convert the octet to an integer via int(octet) to handle both string and numeric input, then format it as a zero-padded 8-character binary string using format(..., '08b') (e.g., 192 becomes '11000000'), and append the result to binary_octets; every octet is processed in order with no skipping, as there are no break, continue, or nested conditionals inside the loop RESULT binary_octets]
            [COMMAND After all octets have been processed, join the elements of binary_octets with '.' as the separator to produce the final dotted binary string, then return it RESULT dotted_binary_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.is_valid() returns a falsy value, indicating the IP address object is not in a valid state]
        [SEQUENTIAL_BLOCK]
            [COMMAND Immediately return an empty string ('') without initializing binary_octets, calling self.get_octets(), or executing any loop iteration; this is a pure early exit that short-circuits all remaining logic RESULT empty_string]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
