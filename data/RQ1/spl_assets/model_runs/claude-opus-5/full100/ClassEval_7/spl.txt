# CLASS BalancedBrackets

[DEFINE_WORKER: "Initializes a bracket-matching instance by storing the input expression and setting up empty stack and bracket lookup lists for subsequent validation." __init__]
    [INPUTS]
        <REF> expr </REF>: string "The expression string to be analyzed for bracket matching"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize self.stack as an empty list to serve as the bracket tracking stack RESULT self.stack]
            [COMMAND Initialize self.left_brackets as the list ['(', '{', '['] containing the three recognized opening bracket characters RESULT self.left_brackets]
            [COMMAND Initialize self.right_brackets as the list [')', '}', ']'] containing the three recognized closing bracket characters, ordered to correspond positionally with self.left_brackets RESULT self.right_brackets]
            [COMMAND Assign the input parameter expr to self.expr for use in later processing RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Filters self.expr in place, retaining only those characters that appear in self.left_brackets or self.right_brackets, discarding all others while preserving original character order." clear_expr]
    [INPUTS]
        <REF> self.expr </REF>: str "The current expression string to be filtered; serves as both the source and the assignment target"
        <REF> self.left_brackets </REF>: iterable "Collection of accepted opening bracket characters; any iterable supporting 'in' membership testing"
        <REF> self.right_brackets </REF>: iterable "Collection of accepted closing bracket characters; any iterable supporting 'in' membership testing"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over every character c in the current value of self.expr, preserving original order RESULT char_sequence]
            [COMMAND For each character c, evaluate the filter predicate: check whether c is a member of self.left_brackets OR c is a member of self.right_brackets; keep c only when at least one condition is true RESULT filtered_chars]
            [COMMAND Join all surviving characters from filtered_chars into a new string using an empty separator, producing a string that contains only bracket characters in their original order RESULT new_expr]
            [COMMAND Assign new_expr back to self.expr, replacing the old value; the right-hand side is fully evaluated before assignment, so there is no aliasing hazard RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.expr is an empty string at the time of the call]
        [SEQUENTIAL_BLOCK]
            [COMMAND The generator produces no characters; join returns an empty string; assign empty string to self.expr RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Both self.left_brackets and self.right_brackets are empty collections]
        [SEQUENTIAL_BLOCK]
            [COMMAND Every character in self.expr fails the membership test; filtered_chars is empty; assign empty string to self.expr RESULT self.expr]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A character in self.expr appears in both self.left_brackets and self.right_brackets]
        [SEQUENTIAL_BLOCK]
            [COMMAND The OR predicate is non-exclusive; the character passes the filter and is retained exactly once in its original position RESULT filtered_chars]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Validates that all bracket characters in self.expr are correctly nested and matched by using a stack, returning False immediately on any mismatch and True only if the stack is empty after processing all characters." check_balanced_brackets]
    [INPUTS]
        <REF> self </REF>: object "Instance providing expr (iterable of characters), left_brackets (set of opening bracket characters), stack (list used as a stack), and clear_expr method"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if all brackets in self.expr are properly nested and matched, False if any mismatch or unclosed opener is detected"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.clear_expr() to reset or prepare the expression state before processing RESULT expr_cleared]
            [COMMAND Iterate over each character Brkt in self.expr one at a time RESULT brkt_iteration]
            [COMMAND Within the for loop, check whether Brkt is a member of self.left_brackets to determine if it is an opening bracket RESULT is_opening_bracket]
            [COMMAND Within the for loop, if Brkt is an opening bracket, push Brkt onto self.stack and continue to the next character RESULT stack_after_push]
            [COMMAND Within the for loop, if Brkt is not an opening bracket, pop the top element from self.stack into Current_Brkt (raises IndexError if stack is empty) RESULT current_brkt]
            [COMMAND Within the for loop, on the closing-bracket path, check whether Current_Brkt equals '(' and if so check whether Brkt is not ')'; if both are true, return False immediately RESULT paren_mismatch_check]
            [COMMAND Within the for loop, on the closing-bracket path, check whether Current_Brkt equals '{' and if so check whether Brkt is not '}'; if both are true, return False immediately RESULT curly_mismatch_check]
            [COMMAND Within the for loop, on the closing-bracket path, check whether Current_Brkt equals '[' and if so check whether Brkt is not ']'; if both are true, return False immediately RESULT square_mismatch_check]
            [COMMAND After the for loop completes, check whether self.stack is non-empty (truthy); if so, return False to indicate unclosed opening brackets remain RESULT unclosed_check]
            [COMMAND If self.stack is empty after the loop, return True to indicate all brackets are balanced RESULT balanced_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.expr contains no characters, so the for loop body never executes]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the for loop entirely and proceed directly to the post-loop stack check RESULT empty_expr_skip]
            [COMMAND Check whether self.stack is non-empty; if empty return True, otherwise return False RESULT empty_expr_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A closing bracket is encountered and the popped Current_Brkt does not match any of the three known opening bracket types '(', '{', or '[']
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the for loop, all three sequential mismatch checks evaluate their outer condition as false, so no return False is triggered and iteration continues without validating the pair RESULT unknown_bracket_passthrough]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: A closing bracket character is encountered in self.expr when self.stack is already empty]
        [LOG "self.stack.pop() raises IndexError because no opener was available to match the current closing bracket; the method does not guard against this case"]
        [THROW IndexError "Raised by list.pop() on an empty stack when a closing bracket appears with no corresponding unmatched opener on the stack"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
