# CLASS PDFHandler

[DEFINE_WORKER: "Initializes the instance by storing the provided file paths and opening a corresponding PyPDF2 PdfReader for each path, creating a positionally aligned pair of attributes." __init__]
    [INPUTS]
        <REF> filepaths </REF>: list "A sequence of file path values (strings or path-like objects) pointing to PDF files to be read"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the parameter filepaths directly to self.filepaths as an alias, with no copying or transformation, so both names reference the same sequence object RESULT self.filepaths]
            [COMMAND Iterate over filepaths and for each element fp construct a PyPDF2.PdfReader instance, collecting all reader objects into a new list assigned to self.readers, maintaining positional alignment so self.readers[i] corresponds to self.filepaths[i] RESULT self.readers]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: filepaths is an empty sequence]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the empty sequence to self.filepaths RESULT self.filepaths]
            [COMMAND Produce an empty list for self.readers without error, as iterating over an empty sequence yields no elements RESULT self.readers]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any fp in filepaths is an invalid path, an unreadable file, or otherwise rejected by PyPDF2.PdfReader during the list comprehension]
        [LOG "PyPDF2.PdfReader raises an exception for the offending fp, aborting the entire list comprehension and leaving self.readers unassigned"]
        [THROW Exception "Any exception raised by PyPDF2.PdfReader (e.g., FileNotFoundError, PermissionError, or a PyPDF2-specific error) propagates uncaught, preventing __init__ from completing"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Merges all pages from every PDF reader in the instance's reader collection into a single output file written to the specified filepath." merge_pdfs]
    [INPUTS]
        <REF> output_filepath </REF>: str "Filesystem path where the merged PDF output file will be written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Confirmation message string indicating the filepath where the merged PDF was saved"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Instantiate a new empty PyPDF2.PdfWriter object with zero pages and assign it to pdf_writer RESULT pdf_writer]
            [COMMAND For each reader in self.readers, iterate over all page indices from 0 through len(reader.pages) - 1 inclusive RESULT reader, page_num]
            [COMMAND Within the outer for loop over reader, for each page_num, retrieve the page object at reader.pages[page_num] and assign it to page RESULT page]
            [COMMAND Within the outer for loop over reader, for each page_num, call pdf_writer.add_page(page) to append the retrieved page to the writer; repeat unconditionally for every page of every reader in order RESULT pdf_writer]
            [COMMAND Open the file at output_filepath in binary write mode ('wb') as out, then call pdf_writer.write(out) to serialize all accumulated pages to disk RESULT out]
            [COMMAND Return the formatted string f"Merged PDFs saved at {output_filepath}" as the method result RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over all PDF readers held by the instance, extracts the text from every page of each reader in document order, and returns a flat list of per-page text strings spanning all loaded PDFs." extract_text_from_pdfs]
    [INPUTS]
        <REF> self </REF>: object "Instance owning self.readers, a collection of PDF reader objects each exposing a pages sequence whose items support extract_text()"
    [END_INPUTS]

    [OUTPUTS]
        <REF> pdf_texts </REF>: list "Flat list of strings (or None for image-only pages) where each element is the extracted text of one page, ordered by reader then by ascending page index"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize pdf_texts as an empty list to accumulate per-page text results RESULT pdf_texts]
            [COMMAND Begin outer for loop: iterate over each reader object in self.readers; for each reader, begin inner for loop iterating page_num from 0 to len(reader.pages) - 1 inclusive (range upper bound is exclusive, so every valid page index is visited exactly once) RESULT reader, page_num]
            [COMMAND Within the outer loop over reader and the inner loop over page_num, retrieve the page object by direct index subscript: page = reader.pages[page_num] RESULT page]
            [COMMAND Within the same nested loop body, call extract_text() on page and append the return value to pdf_texts; the return value may be None for scanned or image-only pages, which is appended without a guard RESULT pdf_texts]
            [COMMAND After both loops have exhausted all readers and all pages, return pdf_texts RESULT pdf_texts]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.readers is empty: the outer loop body never executes]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return pdf_texts as an empty list without entering either loop RESULT pdf_texts]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A particular reader has an empty pages collection: the inner loop body never executes for that reader]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the inner loop for that reader and advance the outer loop to the next reader, leaving pdf_texts unchanged for this iteration RESULT pdf_texts]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
