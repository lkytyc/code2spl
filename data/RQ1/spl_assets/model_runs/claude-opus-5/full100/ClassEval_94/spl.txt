    [END_INPUTS]

[DEFINE_WORKER: "Adds a new item to the inventory by first attempting to restock it, and if restocking fails, directly inserting the item with the given price and quantity." add_item]
    [INPUTS]
        <REF> item_name </REF>: string "The name of the item to add or update in the inventory"
        <REF> price </REF>: number "The unit price of the item"
        <REF> quantity </REF>: integer "The quantity of the item to add"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.restock_item with item_name and quantity to attempt restocking the item RESULT restock_result]
            [COMMAND Evaluate the boolean result of restock_result; if it is truthy, restocking succeeded and no further action is needed — skip the alternative flow RESULT restock_succeeded]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: restock_result is falsy, meaning self.restock_item returned False, None, 0, or another falsy value indicating the item could not be restocked (e.g., it does not yet exist in inventory)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign a new dictionary entry to self.inventory at key item_name, setting its 'price' field to price and its 'quantity' field to quantity, creating or overwriting that entry RESULT self.inventory[item_name]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds the given amount to the current balance and returns the updated balance." insert_coin]

    [END_INPUTS]

    [OUTPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]

    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to purchase a named item by verifying it exists in inventory, confirming sufficient stock and balance, then deducting the price and decrementing quantity before returning the updated balance, or returning False on any failure." purchase_item]
    [INPUTS]
        <REF> item_name </REF>: string "The key used to look up the item in the inventory dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float or bool "The updated balance after a successful purchase, or False if the item does not exist, is out of stock, or the balance is insufficient"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory RESULT item_exists]
            [COMMAND If item_exists is True, retrieve the item dict from self.inventory using item_name as the key and assign it to item RESULT item]
            [COMMAND Within the item-exists branch, evaluate the compound condition: item['quantity'] is strictly greater than 0 AND self.balance is greater than or equal to item['price']; due to short-circuit evaluation, the balance check is skipped entirely when quantity is 0 RESULT can_purchase]
            [COMMAND If can_purchase is True, subtract item['price'] from self.balance and store the result back in self.balance RESULT self.balance]
            [COMMAND Within the same can_purchase branch, decrement item['quantity'] by 1 in place RESULT item['quantity']]
            [COMMAND Return self.balance as the result of a successful transaction RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name exists in self.inventory but item['quantity'] equals 0 OR self.balance is strictly less than item['price']; no state is mutated]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.balance or item['quantity'] RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: item_name is not present as a key in self.inventory; the outer else branch is taken and no state is mutated]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without accessing or modifying any inventory entry or self.balance RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Increments the quantity of an existing inventory item by a given amount, returning true on success or false if the item does not exist in the inventory." restock_item]
    [INPUTS]
        <REF> item_name </REF>: string "The key identifying the item to restock within the inventory dictionary"
        <REF> quantity </REF>: number "The amount to add to the item's existing quantity"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the item was found and its quantity successfully incremented; False if the item does not exist in the inventory"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether item_name exists as a key in self.inventory using a membership test RESULT item_exists]
            [COMMAND If item_exists is true, access self.inventory[item_name] and increment its 'quantity' field in-place by adding quantity to the current value RESULT updated_quantity]
            [COMMAND Return True to signal that the restock operation succeeded RESULT success_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: item_name is not found as a key in self.inventory]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying any state, signalling that the item does not exist in the inventory RESULT failure_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the inventory is populated and, if so, formats each item as a 'name - $price [quantity]' string and returns them joined by newlines; returns false immediately if the inventory is empty or missing." display_items]
    [INPUTS]
        <REF> self.inventory </REF>: dict "Instance-level dictionary mapping item names to dicts containing 'price' and 'quantity' keys; treated as falsy when None or empty"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str | bool "Newline-separated string of formatted inventory entries on success, or boolean False when the inventory is absent or empty"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.inventory is falsy (None, empty dict, or any other falsy value) RESULT inventory_is_empty]
            [COMMAND If inventory_is_empty is True, return False immediately and exit the method RESULT early_exit_false]
            [COMMAND Initialize an empty list named items to accumulate formatted entry strings RESULT items]
            [COMMAND Iterate over every key-value pair in self.inventory using .items(), binding each key to item_name and each value dict to item_info RESULT inventory_iterator]
            [COMMAND Within the for loop, for each iteration format item_name, item_info['price'], and item_info['quantity'] into the string '{item_name} - ${item_info["price"]} [{item_info["quantity"]}]' and append it to items RESULT formatted_entry]
            [COMMAND After the loop completes, join all entries in items with a newline character '\n' as the separator RESULT joined_string]
            [COMMAND Return joined_string as the final result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.inventory is falsy (None, empty dict, or any other falsy value) at the time of the initial check]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the boolean value False immediately without building the items list or iterating RESULT early_exit_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
