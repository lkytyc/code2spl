# CLASS MinesweeperGame

[DEFINE_WORKER: "Initializes a Minesweeper game instance by storing the grid dimension and mine count, generating the mine map and player map from those values, and setting the initial score to zero." __init__]
    [INPUTS]
        <REF> n </REF>: int "Grid dimension defining the n×n size of the Minesweeper board; a zero or negative value may produce an empty or invalid grid"
        <REF> k </REF>: int "Number of mines to place on the board; if k is greater than or equal to n², mines would exceed available cells"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign parameter n to instance attribute self.n RESULT self.n]
            [COMMAND Assign parameter k to instance attribute self.k RESULT self.k]
            [COMMAND Call self.generate_mine_sweeper_map(), which implicitly reads self.n and self.k to produce an n×n grid with k mines placed, and assign the return value to self.minesweeper_map RESULT self.minesweeper_map]
            [COMMAND Call self.generate_playerMap(), which may read self.minesweeper_map to derive dimensions or mine locations, and assign the return value to self.player_map RESULT self.player_map]
            [COMMAND Assign the integer literal 0 to self.score as the canonical starting score with no dependency on any other instance variable RESULT self.score]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Generates an n×n Minesweeper board by randomly placing k mines and incrementing the adjacency counters of all valid non-mine neighboring cells for each placement." generate_mine_sweeper_map]
    [INPUTS]
        <REF> self.n </REF>: int "The dimension of the square board; determines both the row count and column count of the grid. Must be non-negative."
        <REF> self.k </REF>: int "The number of mine placement iterations. Duplicate random positions are possible, so the actual number of distinct mines on the final board may be less than k."
    [END_INPUTS]

    [OUTPUTS]
        <REF> arr </REF>: list[list[int | str]] "An n×n 2D list where each mine cell contains the string 'X' and each non-mine cell contains an integer representing the count of mine placements that landed in an adjacent cell."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize arr as a fresh n×n matrix of integer zeros using a nested list comprehension, where the outer comprehension iterates self.n times and the inner comprehension produces a new independent list of self.n zeros each time, ensuring no row aliasing. RESULT arr]
            [COMMAND Begin a for loop that iterates exactly self.k times, with loop variable num unused in the body; each iteration represents one mine placement attempt. RESULT loop_iteration]
            [COMMAND Within the loop, draw x as a random integer uniformly from the inclusive range [0, self.n-1] using random.randint. RESULT x]
            [COMMAND Within the loop, draw y as a random integer uniformly from the inclusive range [0, self.n-1] using random.randint. RESULT y]
            [COMMAND Within the loop, unconditionally assign the string 'X' to arr[y][x], marking that cell as a mine; if a mine was already present at this coordinate from a prior iteration, this is a silent no-op overwrite. RESULT arr[y][x]]
            [COMMAND Within the loop, check the RIGHT neighbor: if x is in [0, self.n-2] (ensuring x+1 is a valid column index) and y is in [0, self.n-1], and arr[y][x+1] is not 'X', then increment arr[y][x+1] by 1. RESULT arr[y][x+1]]
            [COMMAND Within the loop, check the LEFT neighbor: if x is in [1, self.n-1] (ensuring x-1 is a valid column index) and y is in [0, self.n-1], and arr[y][x-1] is not 'X', then increment arr[y][x-1] by 1. RESULT arr[y][x-1]]
            [COMMAND Within the loop, check the UPPER-LEFT neighbor: if x is in [1, self.n-1] and y is in [1, self.n-1] (ensuring both x-1 and y-1 are valid indices), and arr[y-1][x-1] is not 'X', then increment arr[y-1][x-1] by 1. RESULT arr[y-1][x-1]]
            [COMMAND Within the loop, check the UPPER-RIGHT neighbor: if x is in [0, self.n-2] and y is in [1, self.n-1] (ensuring x+1 and y-1 are valid indices), and arr[y-1][x+1] is not 'X', then increment arr[y-1][x+1] by 1. RESULT arr[y-1][x+1]]
            [COMMAND Within the loop, check the UP neighbor: if x is in [0, self.n-1] and y is in [1, self.n-1] (ensuring y-1 is a valid row index), and arr[y-1][x] is not 'X', then increment arr[y-1][x] by 1. RESULT arr[y-1][x]]
            [COMMAND Within the loop, check the LOWER-RIGHT neighbor: if x is in [0, self.n-2] and y is in [0, self.n-2] (ensuring x+1 and y+1 are valid indices), and arr[y+1][x+1] is not 'X', then increment arr[y+1][x+1] by 1. RESULT arr[y+1][x+1]]
            [COMMAND Within the loop, check the LOWER-LEFT neighbor: if x is in [1, self.n-1] and y is in [0, self.n-2] (ensuring x-1 and y+1 are valid indices), and arr[y+1][x-1] is not 'X', then increment arr[y+1][x-1] by 1. RESULT arr[y+1][x-1]]
            [COMMAND Within the loop, check the DOWN neighbor: if x is in [0, self.n-1] and y is in [0, self.n-2] (ensuring y+1 is a valid row index), and arr[y+1][x] is not 'X', then increment arr[y+1][x] by 1. RESULT arr[y+1][x]]
            [COMMAND After all k iterations are complete, return arr as the final populated board. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.n is 0 or negative, causing range(self.n) to be empty]
        [SEQUENTIAL_BLOCK]
            [COMMAND The nested list comprehension produces an empty list for arr because range(self.n) yields no iterations. RESULT arr]
            [COMMAND The for loop body still executes self.k times, but random.randint(0, self.n-1) will raise a ValueError since the range would be empty or inverted; no mine placement or neighbor update occurs in practice under a valid call. RESULT loop_iteration]
            [COMMAND Return the empty list arr. RESULT arr]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: The same (x, y) coordinate is drawn in two or more separate loop iterations]
        [SEQUENTIAL_BLOCK]
            [COMMAND On the duplicate iteration, arr[y][x] is already 'X'; the assignment overwrites it with 'X' again, which is a no-op for the cell value. RESULT arr[y][x]]
            [COMMAND All 8 neighbor checks still execute for the duplicate coordinate, so each non-mine neighbor of that cell receives an additional +1 increment, potentially double-counting adjacency relative to the intended mine count. RESULT arr neighbor cells]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A cell that previously held a non-zero integer count is later selected as a mine placement target]
        [SEQUENTIAL_BLOCK]
            [COMMAND The cell's accumulated integer count is discarded when arr[y][x] is overwritten with 'X'; all prior adjacency increments recorded in that cell are lost. RESULT arr[y][x]]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs and returns a fresh n×n grid of '-' strings, with each row as an independent list object, using the instance attribute self.n for both dimensions." generate_playerMap]
    [INPUTS]
        <REF> self.n </REF>: int "The dimension size for both rows and columns of the grid; must be accessible on the instance before this method is called"
    [END_INPUTS]

    [OUTPUTS]
        <REF> arr </REF>: list[list[str]] "A self.n × self.n 2D list where every cell is initialized to the string '-' and every row is a distinct list object"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read self.n from the instance to determine the grid dimensions for both axes RESULT grid_size]
            [COMMAND Build a 2D list using a nested list comprehension: for each iteration of the outer loop over self.n columns, evaluate a fresh inner list comprehension that produces a new list of self.n '-' string literals, so that every row is an independent object and mutating one row does not affect any other RESULT arr]
            [COMMAND Return arr as the result of the method RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.n is 0 or any negative integer, causing range(self.n) to be empty]
        [SEQUENTIAL_BLOCK]
            [COMMAND Both the outer and inner range calls produce no iterations, so arr is assigned an empty list [] RESULT arr]
            [COMMAND Return the empty list [] as the result RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Determines whether the Minesweeper game is won by verifying that every non-mine cell in the internal game map has been revealed in the player-facing map." check_won]
    [INPUTS]
        <REF> self </REF>: object "The Minesweeper game instance, providing the board size n and the internal mine map minesweeper_map"
        <REF> map </REF>: 2D list of strings "The player-facing board state where '-' indicates an unrevealed cell"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if all non-mine cells are revealed; False if at least one safe cell remains hidden"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate i over the range 0 to self.n - 1 (inclusive) as the outer row index RESULT outer_loop_i]
            [COMMAND Within the outer loop over i, iterate j over the range 0 to self.n - 1 (inclusive) as the inner column index RESULT inner_loop_j]
            [COMMAND Within the inner loop over j, evaluate whether map[i][j] equals '-' AND self.minesweeper_map[i][j] does not equal 'X'; the AND operator short-circuits so the second condition is only checked when the first is true RESULT cell_is_unrevealed_safe]
            [COMMAND If cell_is_unrevealed_safe is true, immediately return False, exiting both loops and the method without further iteration RESULT early_return_false]
            [COMMAND After both loops complete without triggering the early return, return True to indicate all non-mine cells have been revealed RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The very first cell (0, 0) is unrevealed and not a mine, causing an immediate early return before any further cells are examined]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate map[0][0] == '-' and self.minesweeper_map[0][0] != 'X' at the start of the first iteration RESULT first_cell_check]
            [COMMAND Return False immediately, short-circuiting all remaining O(n²) iterations RESULT early_return_false_best_case]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reveals a cell on the player's map at coordinates (x, y), updates the score, and returns False if the cell is a mine, True if the move wins the game, or the updated player map if the game continues." sweep]
    [INPUTS]
        <REF> x </REF>: int "Row index of the cell to reveal"
        <REF> y </REF>: int "Column index of the cell to reveal"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool | map "False if a mine was hit, True if the win condition is met after revealing the cell, or the updated player_map object if the game continues"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the value at minesweeper_map[x][y] equals the mine marker 'X' RESULT is_mine]
            [COMMAND If is_mine is True, return False immediately without modifying any state RESULT early_return_false]
            [COMMAND If is_mine is False, copy the value at minesweeper_map[x][y] into player_map[x][y] to reveal that cell RESULT cell_revealed]
            [COMMAND Increment score by exactly 1 RESULT score_updated]
            [COMMAND Call check_won passing player_map and compare the return value to True to determine whether the win condition is satisfied RESULT win_check]
            [COMMAND If win_check is True, return True immediately as an early exit indicating the player has won RESULT early_return_true]
            [COMMAND If win_check is not True, return player_map as the fallback indicating a valid move was made but the game is not yet won RESULT return_player_map]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cell at (x, y) contains the mine marker 'X']
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately; no state mutation occurs RESULT mine_hit_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After revealing the cell and incrementing the score, check_won returns True]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return True to signal that the player has won the game RESULT win_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
