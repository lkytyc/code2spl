# CLASS DocFileHandler

[DEFINE_WORKER: "Initializes a file-based object by storing the provided file path as an instance attribute." __init__]
    [INPUTS]
        <REF> file_path </REF>: string "The path to the target file to be associated with this instance"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the input file_path argument to the instance attribute self.file_path RESULT self.file_path]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens a Word document from the instance's file path, extracts the text of every paragraph, and returns them joined as a single newline-delimited string." read_text]
    [INPUTS]
        <REF> self.file_path </REF>: string "File system path to the .docx Word document to be read"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "All paragraph texts from the document concatenated with newline characters as separators"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the Word document located at self.file_path using the Document constructor and assign the resulting document object to doc RESULT doc]
            [COMMAND Initialize an empty list and assign it to text RESULT text]
            [COMMAND Iterate over every paragraph in doc.paragraphs; for each paragraph append its text attribute to the text list RESULT text]
            [COMMAND Join all elements of the text list with a newline character as the separator and return the resulting string RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a new Word document at the configured file path containing a single paragraph with the given text content, font size, and alignment, returning true on success or false if any error occurs." write_text]
    [INPUTS]
        <REF> content </REF>: string "The text string to write into the document paragraph"
        <REF> font_size </REF>: number "Font size in points to apply to the text run; defaults to 12"
        <REF> alignment </REF>: string "Paragraph alignment identifier passed to the internal alignment resolver; defaults to 'left'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the document was created and saved successfully; false if any exception was raised during the process"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new in-memory Document object RESULT doc]
            [COMMAND Add a new empty paragraph to doc RESULT paragraph]
            [COMMAND Add a run containing content to paragraph RESULT run]
            [COMMAND Access the font object of run RESULT font]
            [COMMAND Set font.size to the point value of font_size using Pt(font_size) RESULT font.size]
            [COMMAND Resolve the alignment constant by calling _get_alignment_value with alignment RESULT alignment_value]
            [COMMAND Set paragraph.alignment to alignment_value RESULT paragraph.alignment]
            [COMMAND Save doc to self.file_path RESULT _save_result]
            [COMMAND Return true to indicate success RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is raised at any point within the try block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception silently without logging or re-raising it RESULT _exception_caught]
            [COMMAND Return false to indicate failure RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during document creation, paragraph or run setup, font configuration, alignment resolution, or file save]
        [LOG "Exception caught by bare except block; no error details are recorded"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens the Word document at the stored file path, appends a heading at the specified level, saves the document, and returns True on success or False if any exception occurs." add_heading]
    [INPUTS]
        <REF> heading </REF>: string "The text content of the heading to add to the document"
        <REF> level </REF>: integer "The heading level to apply, defaulting to 1 if not specified"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the heading was added and the document saved successfully; False if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the Word document located at self.file_path by constructing a Document object from that path RESULT doc]
            [COMMAND Call add_heading on doc using the heading text and the level value (default 1) to append the heading to the document RESULT heading_result]
            [COMMAND Save the modified document back to self.file_path RESULT save_result]
            [COMMAND Return True to indicate the operation completed successfully RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception is raised during the try block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to indicate the operation failed due to an exception RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during document opening, heading addition, or file saving]
        [LOG "Exception caught by bare except clause; no error details are recorded"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a table to an existing Word document by populating rows and columns from a 2D data array, then saves the file, returning True on success or False if any exception occurs." add_table]
    [INPUTS]
        <REF> data </REF>: list[list[any]] "A 2D list where the number of outer elements determines row count and the length of the first inner list determines column count; each cell value is converted to string when written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the table was successfully created and the document saved; False if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the existing Word document located at self.file_path and load it into a Document object named doc RESULT doc]
            [COMMAND Create a new table in doc with rows equal to len(data) and cols equal to len(data[0]), and assign it to table RESULT table]
            [COMMAND Iterate over each row index i from 0 to len(data)-1 inclusive; for each i, iterate over each column index j from 0 to len(data[i])-1 inclusive; within that inner loop, set the text of table.cell(i, j) to str(cell_value) where cell_value is data[i][j] RESULT table_populated]
            [COMMAND Save the modified document back to self.file_path by calling doc.save(self.file_path) RESULT save_result]
            [COMMAND Return True to indicate successful completion RESULT True]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is raised anywhere within the try block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Suppress the exception without logging or re-raising it, and return False to the caller RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during document loading, table creation, cell population, or file saving]
        [LOG "An unspecified exception was caught by the bare except clause; no error details are captured or logged"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Looks up and returns the WD_PARAGRAPH_ALIGNMENT enum constant corresponding to a given alignment string, defaulting to LEFT if the string is not recognized." _get_alignment_value]
    [INPUTS]
        <REF> alignment </REF>: str "A string representing the desired paragraph alignment; compared case-insensitively against 'left', 'center', and 'right'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: WD_PARAGRAPH_ALIGNMENT "The WD_PARAGRAPH_ALIGNMENT enum member matching the input string, or WD_PARAGRAPH_ALIGNMENT.LEFT if no match is found"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a dictionary named alignment_options mapping the lowercase string 'left' to WD_PARAGRAPH_ALIGNMENT.LEFT, 'center' to WD_PARAGRAPH_ALIGNMENT.CENTER, and 'right' to WD_PARAGRAPH_ALIGNMENT.RIGHT RESULT alignment_options]
            [COMMAND Call alignment.lower() to normalize the input string to lowercase RESULT normalized_alignment]
            [COMMAND Call alignment_options.get(normalized_alignment, WD_PARAGRAPH_ALIGNMENT.LEFT) to retrieve the matching enum value, using WD_PARAGRAPH_ALIGNMENT.LEFT as the default when the key is absent RESULT alignment_value]
            [COMMAND Return alignment_value to the caller RESULT alignment_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The normalized input string does not match any key in alignment_options (i.e., it is not 'left', 'center', or 'right')]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return WD_PARAGRAPH_ALIGNMENT.LEFT as the fallback default value RESULT alignment_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
