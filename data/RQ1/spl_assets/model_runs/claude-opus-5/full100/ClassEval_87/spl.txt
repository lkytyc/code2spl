    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Current time formatted as HH:MM:SS"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Define the format string as '%H:%M:%S' and assign it to the local variable 'format' RESULT format]

        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats and returns the current datetime as a string in YYYY-MM-DD format." get_current_date]
    [INPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the format string "%Y-%m-%d" to the local variable format RESULT format]

        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Adds a given number of seconds to the instance's datetime and returns the resulting time formatted as HH:MM:SS." add_seconds]
    [INPUTS]
        <REF> self </REF>: object "The instance providing the base datetime via self.datetime attribute."
        <REF> seconds </REF>: numeric "Number of seconds to add to the base datetime; may be zero or negative."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Time string representing the shifted datetime in HH:MM:SS format; the date component is discarded."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct a timedelta of duration 'seconds' using datetime.timedelta(seconds=seconds). RESULT timedelta_value]
            [COMMAND Add timedelta_value to self.datetime to produce a new shifted datetime; self.datetime is not mutated. RESULT new_datetime]
            [COMMAND Assign the string literal "%H:%M:%S" as the strftime format specifier. RESULT format]
            [COMMAND Call new_datetime.strftime(format) to render the time portion of new_datetime as an HH:MM:SS string, discarding the date component. RESULT formatted_time]
            [COMMAND Return formatted_time as the method result. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: seconds is 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND new_datetime equals self.datetime exactly; strftime still runs and returns the current time formatted as HH:MM:SS. RESULT formatted_time]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: seconds is negative]
        [SEQUENTIAL_BLOCK]
            [COMMAND new_datetime precedes self.datetime by the absolute value of seconds; strftime formats the earlier time as HH:MM:SS. RESULT formatted_time]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: self.datetime is near the minimum or maximum representable datetime boundary and the addition overflows]
        [LOG "datetime.timedelta addition on self.datetime raises OverflowError because the resulting datetime falls outside the representable range."]
        [THROW OverflowError "Raised by Python's datetime arithmetic when the shifted datetime exceeds the representable datetime range."]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Parses a date-time string in the format '%Y-%m-%d %H:%M:%S' and returns the corresponding datetime object." string_to_datetime]
    [INPUTS]
        <REF> string </REF>: str "A date-time string expected to conform exactly to the format '%Y-%m-%d %H:%M:%S'; must be a non-None string value"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: datetime.datetime "A datetime object constructed by parsing the input string against the fixed format '%Y-%m-%d %H:%M:%S'"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call datetime.datetime.strptime with 'string' and the fixed format literal "%Y-%m-%d %H:%M:%S" to parse the input into a datetime object RESULT parsed_datetime]
            [COMMAND Return parsed_datetime immediately to the caller without storing it locally RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: string is None or not a string type]
        [LOG "TypeError raised inside datetime.datetime.strptime because the input is not a valid string"]
        [THROW TypeError "Raised by strptime when the 'string' argument is None or a non-string type"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: string is a str but does not match the format "%Y-%m-%d %H:%M:%S", or contains an out-of-range component such as month 13]
        [LOG "ValueError raised inside datetime.datetime.strptime due to format mismatch or an invalid calendar value in the input string"]
        [THROW ValueError "Raised by strptime when the input string's structure or field values are incompatible with the expected format"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Formats a datetime object into a string using the fixed pattern 'YYYY-MM-DD HH:MM:SS'." datetime_to_string]
    [INPUTS]
        <REF> datetime </REF>: datetime "The datetime object to be formatted into a string"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "A string representation of the datetime in the format 'YYYY-MM-DD HH:MM:SS'"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call strftime on the datetime parameter with the format string '%Y-%m-%d %H:%M:%S' to produce a formatted date-time string RESULT formatted_string]
            [COMMAND Return formatted_string as the method result RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the elapsed time in minutes between two time strings by converting each to a datetime, subtracting them, and returning the result rounded to the nearest whole minute." get_minutes]
    [INPUTS]
        <REF> string_time1 </REF>: str "The start time as a string, passed to the internal datetime converter"
        <REF> string_time2 </REF>: str "The end time as a string, passed to the internal datetime converter"
    [END_INPUTS]

    [OUTPUTS]
        <REF> minutes </REF>: int "The elapsed time in whole minutes from time1 to time2, rounded to the nearest integer"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert string_time1 to a datetime object by calling self.string_to_datetime with string_time1 as the argument; assign the result to time1 RESULT time1]
            [COMMAND Convert string_time2 to a datetime object by calling self.string_to_datetime with string_time2 as the argument; assign the result to time2 RESULT time2]
            [COMMAND Subtract time1 from time2 to obtain a timedelta, read the .seconds attribute of that timedelta (a non-negative integer representing the seconds component, not the total duration), divide by 60 to get a fractional minute count, then apply round() to produce the nearest whole integer RESULT minutes]
            [COMMAND Return minutes as the output of the method RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Constructs a naive datetime object from six integer components and returns it as a formatted string in 'YYYY-MM-DD HH:MM:SS' layout." get_format_time]
    [INPUTS]
        <REF> year </REF>: int "Four-digit calendar year (1–9999)"
        <REF> month </REF>: int "Calendar month (1–12)"
        <REF> day </REF>: int "Day of the month, valid for the given year and month"
        <REF> hour </REF>: int "Hour of the day (0–23)"
        <REF> minute </REF>: int "Minute of the hour (0–59)"
        <REF> second </REF>: int "Second of the minute (0–59)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "Formatted datetime string in the pattern '%Y-%m-%d %H:%M:%S', e.g. '2024-03-15 09:05:30'"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the string literal '%Y-%m-%d %H:%M:%S' to the local variable format RESULT format]
            [COMMAND Call datetime.datetime(year, month, day, hour, minute, second) with all six input parameters as positional arguments and assign the resulting naive datetime object to time_item; microsecond defaults to 0 and tzinfo defaults to None RESULT time_item]
            [COMMAND Call time_item.strftime(format) using the format string defined above and return the resulting formatted string to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: Any of the six integer arguments is out of its valid calendar or clock range (e.g. month > 12, day > days-in-month, hour > 23, minute > 59, second > 59), or any argument is not an integer]
        [LOG "datetime.datetime constructor raised ValueError or TypeError due to an invalid component value"]
        [THROW ValueError / TypeError "Propagates unhandled from datetime.datetime(); time_item is never bound and no formatted string is produced"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
