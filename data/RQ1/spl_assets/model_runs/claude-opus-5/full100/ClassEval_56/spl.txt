# CLASS MetricsCalculator

[DEFINE_WORKER: "Initializes all four confusion matrix counters (true positives, false positives, false negatives, true negatives) to zero, establishing the baseline state for a binary classification metrics accumulator." __init__]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the instance variable true_positives to the integer literal 0 RESULT self.true_positives]
            [COMMAND Set the instance variable false_positives to the integer literal 0 RESULT self.false_positives]
            [COMMAND Set the instance variable false_negatives to the integer literal 0 RESULT self.false_negatives]
            [COMMAND Set the instance variable true_negatives to the integer literal 0 RESULT self.true_negatives]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over paired predicted and true binary labels to accumulate the four cells of a confusion matrix — true positives, false positives, false negatives, and true negatives — on the enclosing metrics object." update]
    [INPUTS]
        <REF> predicted_labels </REF>: iterable "Sequence of predicted binary class labels, each expected to be 0 or 1"
        <REF> true_labels </REF>: iterable "Sequence of ground-truth binary class labels, each expected to be 0 or 1, paired positionally with predicted_labels"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Iterate over each (predicted, true) pair produced by zipping predicted_labels with true_labels RESULT current_pair]
            [COMMAND Within the loop, if predicted equals 1 and true equals 1, increment self.true_positives by 1 RESULT self.true_positives]
            [COMMAND Within the loop, else if predicted equals 1 and true equals 0, increment self.false_positives by 1 RESULT self.false_positives]
            [COMMAND Within the loop, else if predicted equals 0 and true equals 1, increment self.false_negatives by 1 RESULT self.false_negatives]
            [COMMAND Within the loop, else if predicted equals 0 and true equals 0, increment self.true_negatives by 1 RESULT self.true_negatives]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Either label in a pair is outside the set {0, 1}]
        [SEQUENTIAL_BLOCK]
            [COMMAND No branch matches; the pair is silently skipped and no counter is incremented for that iteration RESULT no_change]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the precision metric by first updating the internal state with predicted and true labels, then returning the ratio of true positives to all positive predictions, or 0.0 if no positive predictions were made." precision]
    [INPUTS]
        <REF> predicted_labels </REF>: array "Model-predicted labels or scores for the current batch, supplied by the caller at runtime"
        <REF> true_labels </REF>: array "Ground-truth target labels for the current batch, must match predicted_labels in shape and dtype"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "Precision value in the range [0.0, 1.0], or exactly 0.0 when no positive predictions exist"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update with predicted_labels and true_labels to accumulate true-positive and false-positive counts into the object's internal metric state; the return value is discarded RESULT _update_side_effect]
            [COMMAND Read self.true_positives and self.false_positives from the object's accumulated state and compute their sum as the denominator RESULT _denominator]
            [COMMAND Divide self.true_positives by _denominator and return the resulting float as the precision value RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.true_positives + self.false_positives equals exactly 0, meaning the model made no positive predictions of any kind]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 immediately without performing the division, treating undefined precision as zero RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the recall metric (TP / (TP + FN)) for a batch of predictions by first accumulating state via update, then guarding against zero-denominator before returning the ratio." recall]
    [INPUTS]
        <REF> predicted_labels </REF>: array "Model output values (class indices, probabilities, or logits) for the current batch; no default value, must be supplied by the caller"
        <REF> true_labels </REF>: array "Ground-truth target labels for the current batch; no default value, must be supplied by the caller"
    [END_INPUTS]

    [OUTPUTS]
        <REF> recall_score </REF>: float "Recall value in the range [0.0, 1.0], or 0.0 when no actual positive samples exist"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update with predicted_labels and true_labels to accumulate batch statistics into the instance's internal accumulators (self.true_positives, self.false_negatives, and related state); no value is returned from this call RESULT state_updated]
            [COMMAND Read self.true_positives and self.false_negatives from instance state and compute their sum to form the recall denominator RESULT denominator]
            [COMMAND If denominator equals exactly 0 (inclusive equality, fires only when both true_positives and false_negatives are simultaneously zero), return the float literal 0.0 immediately as an early exit; no further statements execute RESULT early_return_zero]
            [COMMAND Divide self.true_positives (numerator) by denominator (self.true_positives + self.false_negatives); self.true_positives feeds both numerator and denominator, so it has a non-linear influence on the result, while self.false_negatives affects only the denominator RESULT recall_score]
            [COMMAND Return recall_score to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.true_positives + self.false_negatives equals exactly 0 at the time of the guard check, meaning no actual positive samples exist in the accumulated state]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the float literal 0.0 immediately, bypassing the division expression entirely RESULT early_return_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the F1 score by updating internal state with the given predictions, then calculating the harmonic mean of precision and recall, returning 0.0 when both are zero." f1_score]
    [INPUTS]
        <REF> predicted_labels </REF>: array "Sequence of labels predicted by the model"
        <REF> true_labels </REF>: array "Sequence of ground-truth labels"
    [END_INPUTS]

    [OUTPUTS]
        <REF> f1 </REF>: float "F1 score in the range [0.0, 1.0], or 0.0 when precision and recall are both zero"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update with predicted_labels and true_labels to update internal tracking state RESULT side_effect]
            [COMMAND Call self.precision with predicted_labels and true_labels and bind the returned float to precision RESULT precision]
            [COMMAND Call self.recall with predicted_labels and true_labels and bind the returned float to recall RESULT recall]
            [COMMAND Compute the harmonic mean as (2 * precision * recall) / (precision + recall) and return the result RESULT f1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: precision + recall equals exactly 0.0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0.0 immediately to avoid division by zero RESULT f1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Computes the accuracy of predictions by delegating to an update method and returning the ratio of correct classifications (true positives plus true negatives) to the total number of samples, or 0.0 if no samples have been recorded." accuracy]
    [INPUTS]
        <REF> predicted_labels </REF>: array "The sequence of predicted class labels to evaluate"
        <REF> true_labels </REF>: array "The sequence of ground-truth class labels corresponding to the predictions"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: float "Accuracy value in the range [0.0, 1.0] representing the proportion of correct predictions; exactly 0.0 when total sample count is zero"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.update with predicted_labels and true_labels to update the confusion-matrix counters on the instance (true_positives, true_negatives, false_positives, false_negatives) RESULT counters_updated]
            [COMMAND Sum self.true_positives, self.true_negatives, self.false_positives, and self.false_negatives and assign the result to local variable total RESULT total]
            [COMMAND Compute the accuracy ratio as (self.true_positives + self.true_negatives) divided by total and return it RESULT accuracy_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: total equals 0 after summing all four confusion-matrix counters, indicating no samples have been processed]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the literal float value 0.0 immediately without performing division RESULT early_return_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
