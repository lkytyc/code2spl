# CLASS JSONProcessor

[DEFINE_WORKER: "Attempts to load and return parsed JSON data from a file at the given path, returning 0 if the file does not exist and -1 if any error occurs during reading or parsing." read_json]
    [INPUTS]
        <REF> file_path </REF>: str "Filesystem path to the JSON file to be read"
    [END_INPUTS]

    [OUTPUTS]
        <REF> data </REF>: dict | list | int "Parsed JSON content on success; 0 if the file does not exist; -1 if an exception is raised during file open or JSON parsing"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the file at file_path exists on the filesystem using os.path.exists RESULT file_exists]
            [COMMAND Open the file at file_path in read mode RESULT file_handle]
            [COMMAND Parse the open file handle as JSON using json.load RESULT data]
            [COMMAND Return data to the caller RESULT data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: file_exists is False, meaning os.path.exists(file_path) returns False]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the integer 0 immediately as an early exit sentinel indicating the file is absent RESULT 0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during the try block covering file open or json.load]
        [LOG "An error occurred while opening or parsing the JSON file at file_path"]
        [THROW caught_silently "The bare except clause suppresses all exceptions and causes the function to return -1 as an error sentinel instead of propagating the exception"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Serializes a data object to a JSON file at the specified path, returning 1 on success or -1 if any exception occurs during the operation." write_json]
    [INPUTS]
        <REF> data </REF>: any "The Python object to serialize as JSON"
        <REF> file_path </REF>: str "The filesystem path of the output file to write"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 1 if the file was written successfully, or -1 if any exception was raised during opening or writing"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the file at file_path in write mode RESULT file_handle]
            [COMMAND Serialize data to JSON and write it to the open file using json.dump RESULT write_result]
            [COMMAND Return 1 to indicate success RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is caught by the bare except block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 to indicate failure RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during file open or json.dump (e.g., IOError, TypeError, PermissionError)]
        [LOG "Exception caught by bare except clause; no exception details are logged or re-raised"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes a specified key from a JSON file's data dictionary and persists the change to disk, returning 1 on successful deletion, or 0 if the file could not be read or the key was not present." process_json]
    [INPUTS]
        <REF> file_path </REF>: string "Path to the JSON file to read and potentially modify"
        <REF> remove_key </REF>: string "The dictionary key to remove from the loaded JSON data"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Returns 1 if the key was found and successfully removed and written back to disk; returns 0 if the file could not be read or the key was not present in the data"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call read_json with file_path and store the result in data RESULT data]
            [COMMAND If data equals 0 or data equals -1, return 0 immediately to signal a read failure RESULT early_return_0]
            [COMMAND Check whether remove_key exists as a key in data RESULT key_present]
            [COMMAND If remove_key is present in data, delete that key from data in place RESULT data_mutated]
            [COMMAND Call write_json with the mutated data and file_path to persist the updated dictionary to disk RESULT write_result]
            [COMMAND Return 1 to signal that the key was found, removed, and the file was written successfully RESULT return_1]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: remove_key is not found as a key in data after a successful read]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 to signal that no deletion occurred and the file was not modified RESULT return_0_no_key]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
