[DEFINE_WORKER: "Tokenizes a sentence after stripping punctuation, assigns each token its correct WordNet part-of-speech based on its Penn Treebank POS tag, lemmatizes it accordingly, and returns the resulting list of lemmatized words." lemmatize_sentence]
    [INPUTS]
        <REF> sentence </REF>: str "Raw input sentence to be lemmatized; punctuation will be removed before processing."
    [END_INPUTS]

    [OUTPUTS]
        <REF> lemmatized_words </REF>: list[str] "List of lemmatized tokens in the same order as the original tokens, with length equal to the number of tokens produced after punctuation removal; empty list if the cleaned sentence yields no tokens."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize accumulator lemmatized_words as an empty list. RESULT lemmatized_words]
            [COMMAND Reassign sentence by calling self.remove_punctuation(sentence), replacing the original value with the punctuation-stripped version; the result feeds all downstream steps. RESULT sentence]
            [COMMAND Tokenize the cleaned sentence by calling word_tokenize(sentence), producing an ordered list of string tokens; if sentence is empty the result is an empty list. RESULT words]
            [COMMAND POS-tag the token list by calling pos_tag(words), producing a list of (token, Penn-Treebank-tag) tuples of the same length; if words is empty the result is an empty list. RESULT tagged_words]
            [COMMAND Iterate over every (word, tag) pair in tagged_words in order; for each pair execute exactly one branch of the POS dispatch below, then unconditionally append lemmatized_word to lemmatized_words; no break, continue, or early return is possible, so the loop always runs to completion and lemmatized_words ends up the same length as tagged_words. RESULT lemmatized_words]
            [COMMAND Return lemmatized_words. RESULT lemmatized_words]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Current iteration: tag starts with 'V' (verb tags such as VB, VBD, VBG, VBN, VBP, VBZ).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.lemmatizer.lemmatize(word, pos='v') and assign the result to lemmatized_word. RESULT lemmatized_word]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Current iteration: tag starts with 'J' (adjective tags such as JJ, JJR, JJS); evaluated only when the 'V' condition is false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.lemmatizer.lemmatize(word, pos='a') and assign the result to lemmatized_word. RESULT lemmatized_word]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Current iteration: tag starts with 'R' (adverb tags such as RB, RBR, RBS); evaluated only when both the 'V' and 'J' conditions are false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.lemmatizer.lemmatize(word, pos='r') and assign the result to lemmatized_word. RESULT lemmatized_word]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Current iteration: tag does not start with 'V', 'J', or 'R' (covers nouns, determiners, pronouns, punctuation tokens, and any unrecognized or malformed tag); this branch is the default and produces noun-mode lemmatization without any warning.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.lemmatizer.lemmatize(word) using the default WordNet POS (noun, 'n') and assign the result to lemmatized_word. RESULT lemmatized_word]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: sentence is empty or becomes empty after self.remove_punctuation(sentence) removes all characters.]
        [SEQUENTIAL_BLOCK]
            [COMMAND word_tokenize returns an empty list, pos_tag returns an empty list, the for loop body never executes, and lemmatized_words remains the empty list it was initialized to. RESULT lemmatized_words]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes punctuation from the input sentence, tokenizes it, retrieves part-of-speech tags for each token, and returns a flat list of those tag strings." get_pos_tag]
    [INPUTS]
        <REF> sentence </REF>: str "Raw input sentence to be POS-tagged"
    [END_INPUTS]

    [OUTPUTS]
        <REF> pos_tags </REF>: list[str] "Ordered list of POS tag strings corresponding to each token in the cleaned sentence"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize pos_tags as an empty list to serve as the accumulator for collected POS tag strings RESULT pos_tags]
            [COMMAND Overwrite sentence by passing the current sentence value into self.remove_punctuation and capturing the returned punctuation-free string RESULT sentence]
            [COMMAND Tokenize the sanitized sentence using word_tokenize and assign the resulting token list to words RESULT words]
            [COMMAND Apply pos_tag to words to obtain a list of (token, tag) tuples and assign it to tagged_words RESULT tagged_words]
            [COMMAND Iterate over every element in tagged_words; for each tagged_word tuple, extract index 1 (the tag string) and append it to pos_tags RESULT pos_tags]
            [COMMAND Return pos_tags as the final result RESULT pos_tags]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.remove_punctuation returns an empty string (e.g., sentence contained only punctuation characters)]
        [SEQUENTIAL_BLOCK]
            [COMMAND word_tokenize receives an empty string and produces an empty list, so words is [] RESULT words]
            [COMMAND pos_tag receives an empty list and returns an empty list, so tagged_words is [] RESULT tagged_words]
            [COMMAND The for loop over tagged_words executes zero iterations, leaving pos_tags as [] RESULT pos_tags]
            [COMMAND Return the empty pos_tags list RESULT pos_tags]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes all punctuation characters from the input string by building a deletion translation table from string.punctuation and applying it via str.translate, returning a new punctuation-free string." remove_punctuation]
    [INPUTS]
        <REF> sentence </REF>: str "The input string from which all punctuation characters will be removed."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "A new string identical to sentence but with every character present in string.punctuation deleted; the original sentence is not mutated."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a translation table by calling str.maketrans with two empty strings and string.punctuation as the third argument, producing a dict that maps the Unicode ordinal of each punctuation character to None; this table is ephemeral and unnamed. RESULT translation_table]
            [COMMAND Apply the translation table to sentence by calling sentence.translate(translation_table), which returns a new str with every character whose ordinal maps to None deleted and all other characters preserved unchanged. RESULT punctuation_free_string]
            [COMMAND Return punctuation_free_string as the result of the method. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: sentence is not a str instance at call time]
        [LOG "Calling .translate on a non-str object raises AttributeError because translate is a str method; str.maketrans itself succeeds since it only processes the constant string.punctuation."]
        [THROW AttributeError "Raised by the Python runtime when sentence does not have a translate attribute, i.e., when it is not a str."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
