# CLASS ZipFileProcessor

[DEFINE_WORKER: "Initializes a new instance by storing the provided file name as an instance attribute." __init__]
    [INPUTS]
        <REF> file_name </REF>: string "The name of the file to associate with this instance"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the value of parameter file_name to the instance attribute self.file_name RESULT self.file_name]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to open a ZIP archive from the instance's file path in read mode, returning the ZipFile object on success or None if any error occurs." read_zip_file]
    [INPUTS]
        <REF> self </REF>: object "Instance containing file_name, the path to the ZIP archive to open"
    [END_INPUTS]

    [OUTPUTS]
        <REF> zip_file </REF>: ZipFile | None "An open ZipFile object in read mode if successful, or None if any exception is raised"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to open the file at self.file_name as a ZIP archive in read mode ('r') using zipfile.ZipFile RESULT zip_file]
            [COMMAND Return zip_file to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception is raised during the ZipFile open call (including but not limited to file not found, bad ZIP file, or permission errors)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Catch the exception silently without logging or re-raising RESULT exception_suppressed]
            [COMMAND Return None to the caller RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Opens the zip archive referenced by the instance's file_name attribute and extracts all its contents to the specified output path, returning True on success or False if any exception occurs." extract_all]
    [INPUTS]
        <REF> output_path </REF>: string "Filesystem path to the directory where all archive contents will be extracted"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if extraction completed without error; False if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip archive located at self.file_name in read mode using zipfile.ZipFile as zip_file RESULT zip_file_context]
            [COMMAND Extract all contents of zip_file to output_path by calling zip_file.extractall(output_path) RESULT extraction_result]
            [COMMAND Return True to indicate successful extraction RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Any exception is caught by the bare except block]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to signal that extraction failed RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during the ZipFile open or extractall call]
        [LOG "Exception is silently caught with a bare except clause; no error details are recorded or re-raised"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to extract a single named file from a zip archive to a specified output path, returning true on success and false if any error occurs." extract_file]
    [INPUTS]
        <REF> file_name </REF>: string "The name of the file to extract from the zip archive"
        <REF> output_path </REF>: string "The destination directory path where the extracted file will be written"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the file was extracted successfully; false if any exception was raised during the operation"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open the zip archive located at self.file_name in read mode using zipfile.ZipFile as zip_file RESULT zip_file]
            [COMMAND Call zip_file.extract with arguments file_name and output_path to extract the target file to the specified destination RESULT extract_result]
            [COMMAND Return True to indicate the extraction completed without error RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception is caught by the bare except clause]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False to signal that extraction failed RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised within the try block, including but not limited to zipfile.BadZipFile, KeyError for a missing member, or an OSError during file writing]
        [LOG "Exception is silently caught by a bare except clause with no logging or re-raising"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Creates a ZIP archive at the specified output path by writing all provided file paths into it, returning true on success or false if any exception occurs." create_zip_file]
    [INPUTS]
        <REF> files </REF>: list "An iterable of file paths to be added to the ZIP archive"
        <REF> output_file_name </REF>: string "The destination file path for the ZIP archive to be created"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the ZIP file was created successfully; false if any exception was raised during the process"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a new ZIP file for writing at the path specified by output_file_name using write mode RESULT zip_file]
            [COMMAND For each file path in the files iterable, write that file into zip_file using its original path RESULT zip_contents]
            [COMMAND Return true to indicate the ZIP archive was created successfully RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An exception of any type is caught during ZIP creation or the file-writing loop]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false to indicate the ZIP archive was not created successfully RESULT failure_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: Any exception is raised during ZIP file creation or file writing]
        [LOG "An unspecified exception occurred while creating the ZIP archive; returning false"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
