# CLASS GomokuGame

[DEFINE_WORKER: "Initializes a game board instance by storing the board dimension, constructing a square grid of blank cells, and setting the first player to 'X'." __init__]
    [INPUTS]
        <REF> board_size </REF>: int "The number of rows and columns for the square game board; determines the dimensions of both axes"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign the board_size parameter to self.board_size as a direct identity copy, storing it for later reference RESULT self.board_size]
            [COMMAND Build a board_size × board_size 2D list using a nested list comprehension: iterate board_size times in the outer loop and board_size times in the inner loop, filling every cell with a single space character ' '; assign the result to self.board RESULT self.board]
            [COMMAND Assign the string literal 'X' to self.current_player unconditionally, marking the first player to move RESULT self.current_player]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: board_size is 0 or a negative integer]
        [SEQUENTIAL_BLOCK]
            [COMMAND range(board_size) yields an empty sequence; both the outer and inner comprehensions produce no iterations, so self.board is set to an empty list [] without raising an error; self.board_size stores the non-positive value and self.current_player is still set to 'X' RESULT self.board]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to place the current player's mark on the board at the given cell coordinates, switches the active player if the cell was empty, and returns whether the move succeeded." make_move]
    [INPUTS]
        <REF> row </REF>: int "Zero-based row index of the target cell on the board"
        <REF> col </REF>: int "Zero-based column index of the target cell on the board"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the move was successfully made; implicitly None (falsy) if the cell was already occupied"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the cell at position (row, col) on self.board contains exactly a single space character, indicating it is unoccupied RESULT cell_is_empty]
            [COMMAND If cell_is_empty is true, write self.current_player into self.board[row][col] to record the move RESULT board_updated]
            [COMMAND After recording the move, toggle self.current_player: set it to 'O' if it was 'X', otherwise set it to 'X' RESULT player_switched]
            [COMMAND Return True to signal that the move was valid and successfully applied RESULT success_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cell at (row, col) is not a single space — it is already occupied by 'X', 'O', or any other non-space value]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip all board mutation and player-switching logic entirely RESULT no_op]
            [COMMAND Return False to signal that the move was invalid and no state was changed RESULT failure_flag]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Scans every occupied cell on the board in four directions and returns the winning player's piece if any five-in-a-row sequence is found, or null if no winner exists." check_winner]
    [INPUTS]
        <REF> self </REF>: object "The game instance, providing board (2D array of player marks and spaces) and board_size (integer dimension of the square board)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> winner </REF>: string or null "The mark of the winning player (e.g. 'X' or 'O') if a five-in-a-row is found; null if no winner is present on the board"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize directions as a list of four 2D offset vectors: (0,1) for horizontal-right, (1,0) for vertical-down, (1,1) for diagonal down-right, and (1,-1) for diagonal down-left RESULT directions]
            [COMMAND Begin outer loop: iterate row from 0 to board_size - 1 (inclusive) RESULT row]
            [COMMAND Within the outer row loop, begin inner loop: iterate col from 0 to board_size - 1 (inclusive) RESULT col]
            [COMMAND Within the row and col loops, check whether the cell at board[row][col] is not equal to a single space character ' '; if the cell is empty, skip all further processing for this cell and continue to the next col RESULT cell_occupied]
            [COMMAND Within the row and col loops, given that the cell is occupied, iterate over each vector in directions RESULT direction]
            [COMMAND Within the row, col, and direction loops, call _check_five_in_a_row with arguments (row, col, direction) to test whether five consecutive identical marks exist starting at (row, col) in the given direction RESULT five_found]
            [COMMAND Within the row, col, and direction loops, if five_found is true, immediately return the value of board[row][col] as the winning player's mark, exiting all loops and the method RESULT winner]
            [COMMAND After all rows, columns, and directions have been checked with no early return, return null to indicate no winner was found RESULT winner]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The cell at board[row][col] equals ' ' (empty space) for the current (row, col) position]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the direction loop entirely for this cell and advance col to the next value RESULT skip_empty_cell]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether exactly five consecutive board cells match the symbol at the origin cell by stepping outward from (row, col) in the given direction and returning true only when all four subsequent cells are in-bounds and hold the same symbol." _check_five_in_a_row]
    [INPUTS]
        <REF> row </REF>: int "Row index of the origin cell on the board"
        <REF> col </REF>: int "Column index of the origin cell on the board"
        <REF> direction </REF>: tuple[int, int] "Two-element iterable (dx, dy) representing the step offsets for row and column per iteration"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if exactly five consecutive cells starting at (row, col) in the given direction all hold the same symbol; False otherwise"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Unpack direction into dx and dy as the row and column step offsets RESULT dx, dy]
            [COMMAND Initialize count to 1, pre-crediting the origin cell at (row, col) as the first matching cell RESULT count]
            [COMMAND Read self.board[row][col] and store it as symbol, the match target for all subsequent comparisons RESULT symbol]
            [COMMAND Iterate i over the integer range [1, 4] inclusive (four iterations: i = 1, 2, 3, 4) RESULT loop_range]
            [COMMAND Within the for loop, compute new_row as row + dx * i RESULT new_row]
            [COMMAND Within the for loop, compute new_col as col + dy * i RESULT new_col]
            [COMMAND Within the for loop, check whether new_row is in [0, self.board_size) AND new_col is in [0, self.board_size); if either is out of bounds, return False immediately RESULT bounds_check]
            [COMMAND Within the for loop, if self.board[new_row][new_col] does not equal symbol, return False immediately RESULT symbol_check]
            [COMMAND Within the for loop, increment count by 1 if both checks passed for this iteration RESULT count]
            [COMMAND After the loop completes all four iterations without an early return, return the boolean result of count == 5 RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: new_row or new_col falls outside [0, self.board_size) on any iteration i in [1, 4]]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without examining the board cell or incrementing count RESULT return_false_oob]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: self.board[new_row][new_col] does not equal symbol on any iteration i in [1, 4] after the bounds check passes]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without incrementing count RESULT return_false_mismatch]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: direction does not contain exactly two elements when unpacked]
        [LOG "Tuple unpacking of direction into dx, dy raises ValueError because direction has an unexpected number of elements"]
        [THROW ValueError "Raised by Python when the right-hand side of a tuple unpacking does not match the expected element count of two"]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: self.board is not initialized or row/col are out of range when reading the origin cell before the loop]
        [LOG "Accessing self.board[row][col] raises AttributeError if board does not exist or IndexError if row or col is out of range"]
        [THROW IndexError "Raised when row or col is outside the valid index range of self.board before any loop iteration begins"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
