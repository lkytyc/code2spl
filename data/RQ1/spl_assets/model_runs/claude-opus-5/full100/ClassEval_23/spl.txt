[DEFINE_WORKER: "Recursively generates all combinations of a fixed size from self.datas by placing candidates into result slots and collecting completed selections as snapshots in result." _select]
    [INPUTS]

        <REF> resultList </REF>: List[str] "Shared mutable buffer of fixed length used to build the current candidate combination slot by slot"
        <REF> resultIndex </REF>: int "Zero-based index of the next slot in resultList to fill in the current recursive call"
        <REF> result </REF>: List[List[str]] "Accumulator that collects shallow copies of completed combinations"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List[List[str]] "Populated in-place with all valid combinations; the function returns None implicitly"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute resultLen as the length of resultList, capturing a point-in-time snapshot of the total number of slots to fill RESULT resultLen]
            [COMMAND Compute resultCount as resultIndex + 1, converting the zero-based slot index to a one-based count of how many slots have been filled so far including the current one RESULT resultCount]
            [COMMAND Iterate i from dataIndex (inclusive) to len(self.datas) + resultCount - resultLen (exclusive), where the upper bound is a pruning constraint ensuring at least (resultLen - resultCount) elements remain in self.datas after position i to complete the combination RESULT loop_variable_i]
            [COMMAND Within the for loop over i, assign self.datas[i] into resultList at position resultIndex, filling the current slot with the candidate element RESULT resultList[resultIndex]]
            [COMMAND Within the for loop over i, after filling the current slot, recursively call self._select with dataIndex=i+1, the updated resultList, resultIndex=resultIndex+1, and result, advancing both the data window and the slot pointer for the next level RESULT recursive_call]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: resultCount is strictly greater than resultLen, meaning all slots have been filled and the candidate combination is complete]

            [COMMAND Append a shallow copy of resultList to result, preserving the completed combination as an immutable snapshot independent of future mutations to resultList RESULT result]
            [COMMAND Return immediately, terminating this recursive call without entering the for loop RESULT early_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
