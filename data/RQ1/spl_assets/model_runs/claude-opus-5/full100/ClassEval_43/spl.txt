[DEFINE_WORKER: "Adds a new employee record to the employees dictionary if the given ID does not already exist, returning true on success or false if the ID is already registered." AddEmployee]
    [INPUTS]
        <REF> employee_id </REF>: any "Unique identifier for the employee to be added; used as the dictionary key"
        <REF> name </REF>: string "Full name of the employee"
        <REF> position </REF>: string "Job title or role of the employee"
        <REF> department </REF>: string "Department the employee belongs to"
        <REF> salary </REF>: number "Salary of the employee"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the employee was successfully inserted; false if the employee_id already existed and no change was made"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id is already present as a key in the employees dictionary RESULT id_exists]
            [COMMAND If employee_id is found, return False immediately without modifying the employees dictionary RESULT early_return_false]
            [COMMAND If employee_id is not found, insert a new entry into the employees dictionary under the key employee_id, setting its value to a dictionary with keys name, position, department, and salary mapped to the corresponding input parameters RESULT insert_record]
            [COMMAND Return True to indicate the employee record was successfully added RESULT return_true]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_id already exists as a key in the employees dictionary]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the insertion entirely and return False to the caller, signaling that the employee already exists and no mutation occurred RESULT duplicate_exit]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Removes an employee entry from the employees dictionary by ID if it exists, returning true on success and false if the ID was not found." remove_employee]
    [INPUTS]
        <REF> employee_id </REF>: any "The key identifying the employee to remove from the employees dictionary"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the employee was found and deleted, False if the employee ID was not present in the dictionary"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id exists as a key in self.employees using a membership test RESULT membership_check]
            [COMMAND If membership_check is true, delete the entry keyed by employee_id from self.employees in place using del RESULT deletion_performed]
            [COMMAND Return True to signal that the deletion was carried out successfully RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_id is not found as a key in self.employees]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately without modifying self.employees, signalling that no deletion occurred RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Looks up an employee by ID and, if found and all keys in the update dict already exist on the employee record, overwrites those fields with the new values and returns True; returns False if the employee is not found or any key is absent." update_employee]
    [INPUTS]
        <REF> employee_id </REF>: int "Unique identifier used to look up the target employee record"
        <REF> employee_info </REF>: dict "Dictionary of field names and their new values to apply to the employee record"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if the employee was found and all fields were successfully updated; False if the employee was not found or any key in employee_info is absent from the employee record"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call self.get_employee with employee_id and bind the return value to employee RESULT employee]
            [COMMAND Check whether employee equals False (weak equality, also matches None, 0, or empty containers); if so, return False immediately RESULT early_return_false_on_missing_employee]
            [COMMAND Iterate over every key-value pair in employee_info; for each key, check whether that key is absent from employee; if any key is not present, return False immediately before any write occurs RESULT validation_pass_result]
            [COMMAND Iterate over every key-value pair in employee_info a second time; for each pair, overwrite employee[key] with value, updating only keys already confirmed to exist RESULT write_pass_result]
            [COMMAND Return True to signal that all fields passed validation and were successfully written RESULT success_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee equals False (or any falsy value) after the get_employee call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately; no validation or mutation is attempted RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Any key in employee_info is not present in the employee record during the validation pass]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return False immediately upon the first missing key; the employee record is left entirely unmodified because no writes have occurred yet RESULT return_false]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Looks up an employee record by ID from the internal employees collection, returning the associated record if found or False if the ID does not exist." GetEmployee]
    [INPUTS]
        <REF> employee_id </REF>: any "The identifier used to look up an employee in the employees collection"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: any | bool "The employee record associated with the given ID if found, or False if the ID is not present in the collection"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether employee_id exists as a key in self.employees RESULT membership_check]
            [COMMAND If membership_check is true, retrieve and return the value stored at self.employees[employee_id] RESULT employee_record]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: employee_id is not found in self.employees]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the literal boolean value False to indicate the employee was not found RESULT not_found_sentinel]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Iterates over all employees in the instance's employee collection and returns a dictionary mapping each employee ID to a details dict containing the employee ID under the key 'employee_ID' plus all original fields from that employee's record." list_employees]
    [INPUTS]
        <REF> self </REF>: object "Instance owning the employees dict; self.employees maps employee IDs to their field dictionaries"
    [END_INPUTS]

    [OUTPUTS]
        <REF> employee_data </REF>: dict "Dictionary keyed by employee ID, where each value is a dict containing 'employee_ID' plus all fields copied from the original employee record; returns an empty dict when self.employees is falsy"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize employee_data as an empty dictionary RESULT employee_data]
            [COMMAND Check whether self.employees is truthy (non-None, non-empty); if falsy, skip the entire loop and proceed to the return step RESULT employees_truthy]
            [COMMAND For each (employee_id, employee_info) pair in self.employees.items(), initialize a new empty dictionary employee_details RESULT employee_details]
            [COMMAND Within the outer for loop over (employee_id, employee_info), assign employee_details['employee_ID'] = employee_id RESULT employee_details]
            [COMMAND Within the outer for loop over (employee_id, employee_info), for each (key, value) pair in employee_info.items(), assign employee_details[key] = value; note that if employee_info contains a key named 'employee_ID' it will overwrite the value set in the previous step RESULT employee_details]
            [COMMAND Within the outer for loop over (employee_id, employee_info), store the completed employee_details dict in employee_data under the key employee_id RESULT employee_data]
            [COMMAND Return employee_data RESULT employee_data]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.employees is falsy (None, empty dict, or any other falsy value)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the iteration block entirely and return the empty employee_data dict initialized at the start RESULT employee_data]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
