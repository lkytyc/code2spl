[DEFINE_WORKER: "Validates that the input phone number is non-empty, exactly 11 characters long, and composed entirely of digits, returning the original string if valid or None if any check fails." _validate_phoneNumber]

        <REF> phoneNumber </REF>: str "The phone number string to validate"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str or None "The original phoneNumber string if all validations pass, or None if any validation fails"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether phoneNumber is falsy (None, empty string, or any other falsy value); if so, return None immediately and exit the function RESULT early_exit_falsy]

            [COMMAND Return the original phoneNumber string unchanged, indicating the input passed all validation checks RESULT validated_phone_number]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: phoneNumber is falsy at the time of the first guard (None, empty string, zero, False, or any falsy value)]

            [COMMAND Return None to signal that no valid phone number was provided RESULT early_exit_falsy]

        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]
