# CLASS IpUtil

[DEFINE_WORKER: "Attempts to parse the given string as an IPv4 address using the socket library and returns True if it is valid, or False if a socket error is raised." is_valid_ipv4]
    [INPUTS]
        <REF> ip_address </REF>: string "The string to be validated as an IPv4 address"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if ip_address is a valid IPv4 address, False otherwise"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.inet_pton with socket.AF_INET and ip_address to attempt parsing the input as an IPv4 address RESULT _parse_attempt]
            [COMMAND Return True indicating the input is a valid IPv4 address RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: socket.error is raised during the inet_pton call, meaning ip_address is not a valid IPv4 address]
        [LOG "socket.error caught; ip_address failed IPv4 validation"]
        [THROW ReturnFalse "Return False to indicate the input is not a valid IPv4 address"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to parse the given string as an IPv6 address using the system socket library and returns true if parsing succeeds or false if a socket error is raised." is_valid_ipv6]
    [INPUTS]
        <REF> ip_address </REF>: string "The string to validate as an IPv6 address"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if ip_address is a valid IPv6 address, false otherwise"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.inet_pton with address family AF_INET6 and ip_address to attempt IPv6 address parsing RESULT parse_result]
            [COMMAND Return true indicating the address is a valid IPv6 address RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.error is caught after the failed inet_pton call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return false indicating the address is not a valid IPv6 address RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: socket.error is raised by socket.inet_pton during IPv6 address parsing]
        [LOG "socket.error caught; ip_address is not a valid IPv6 address"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Attempts to resolve the hostname for a given IP address using a reverse DNS lookup, returning the hostname string on success or None if the lookup fails." get_hostname]
    [INPUTS]
        <REF> ip_address </REF>: string "The IP address to resolve to a hostname via reverse DNS lookup"
    [END_INPUTS]

    [OUTPUTS]
        <REF> hostname </REF>: string or null "The resolved hostname string if the reverse DNS lookup succeeds, or None if a host error occurs"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call socket.gethostbyaddr with ip_address and retrieve the first element (index 0) of the returned tuple, storing it as hostname RESULT hostname]
            [COMMAND Return hostname RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: socket.herror is caught after the failed gethostbyaddr call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return None to the caller, signaling that no hostname could be resolved for the given IP address RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: socket.herror is raised during the gethostbyaddr call, indicating a host address error such as no reverse DNS record found]
        [LOG "Reverse DNS lookup failed for the given IP address with a host error; returning None"]
        [THROW ExceptionName "Exception description"]
    [END_EXCEPTION_FLOW]

[END_WORKER]
