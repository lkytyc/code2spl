[DEFINE_WORKER: "Validates the given mode string against the accepted values 'heat' and 'cool', assigns it to the instance attribute if valid, or returns False immediately if the value is not recognized." set_mode]
    [INPUTS]
        <REF> mode </REF>: string "The mode value to validate and store; accepted values are exactly 'heat' and 'cool' (case-sensitive)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean or None "Returns False when mode is not in the accepted set; returns None implicitly when mode is valid and the assignment succeeds"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether the input parameter mode is a member of the set ['heat', 'cool'] using a strict case-sensitive equality test against each element RESULT condition_result]
            [COMMAND If condition_result is True, assign the value of mode to the instance attribute self.mode and allow execution to continue past the conditional block RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: mode is not equal to 'heat' and not equal to 'cool' (i.e., the membership test evaluates to False)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Immediately exit the method by returning the boolean value False, signalling to the caller that the provided mode is unsupported and self.mode was not modified RESULT False]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Sets the thermostat operating mode to 'heat' if the current temperature is strictly below the target, or to 'cool' if the current temperature equals or exceeds the target." auto_set_mode]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compare self.current_temperature against self.target_temperature using a strict less-than operator RESULT comparison_result]
            [COMMAND If comparison_result is True (current_temperature < target_temperature), assign the string 'heat' to self.mode RESULT self.mode]
            [COMMAND Otherwise (current_temperature >= target_temperature, including the boundary case where both are equal), assign the string 'cool' to self.mode RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: current_temperature is strictly less than target_temperature]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign 'heat' to self.mode RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_temperature equals or exceeds target_temperature (inclusive boundary: equal-to falls here, not in the heat branch)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign 'cool' to self.mode RESULT self.mode]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Checks whether the thermostat's current operating mode matches what the temperature situation requires, returning true if modes align and calling auto_set_mode() then returning false if a mismatch is corrected." auto_check_conflict]
    [INPUTS]
        <REF> self.current_temperature </REF>: numeric "The current measured temperature, compared strictly (greater than) against the target"
        <REF> self.target_temperature </REF>: numeric "The desired target temperature used as the comparison threshold"
        <REF> self.mode </REF>: string "The current operating mode of the thermostat, expected to be 'cool' or 'heat'"
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if the current mode matches the temperature situation and conditioning should proceed; false if a mode mismatch was detected and corrected via auto_set_mode(), signalling the caller to wait"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate whether self.current_temperature is strictly greater than self.target_temperature (boundary: exactly equal is false and falls to the else branch) RESULT outer_condition]
            [COMMAND If outer_condition is true, check whether self.mode equals the string 'cool' exactly RESULT inner_condition_hot]
            [COMMAND If outer_condition is true and inner_condition_hot is true, return True immediately — the system is already in cool mode and conditioning should proceed RESULT return_value]
            [COMMAND If outer_condition is false (current_temperature <= target_temperature), check whether self.mode equals the string 'heat' exactly RESULT inner_condition_cold]
            [COMMAND If outer_condition is false and inner_condition_cold is true, return True immediately — the system is already in heat mode and conditioning should proceed RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: current_temperature > target_temperature but mode is not 'cool']
        [SEQUENTIAL_BLOCK]
            [COMMAND Call auto_set_mode() as a corrective side effect to switch the thermostat to the appropriate mode RESULT mode_corrected]
            [COMMAND Return False to signal that the mode was mismatched and has been corrected, but the caller should not proceed yet RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: current_temperature <= target_temperature but mode is not 'heat']
        [SEQUENTIAL_BLOCK]
            [COMMAND Call auto_set_mode() as a corrective side effect to switch the thermostat to the appropriate mode RESULT mode_corrected]
            [COMMAND Return False to signal that the mode was mismatched and has been corrected, but the caller should not proceed yet RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Simulates a thermostat reaching its target temperature by stepping one degree at a time in the appropriate direction based on the current mode, returning the total number of steps taken." simulate_operation]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> use_time </REF>: int "The number of one-degree steps taken to move current_temperature to target_temperature, equal to the absolute difference between the initial current temperature and the target temperature."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call auto_set_mode on self to determine and assign the operating mode before any temperature logic runs. RESULT mode_assigned]
            [COMMAND Initialize use_time to 0 to begin counting elapsed steps. RESULT use_time]
            [COMMAND Check whether self.mode equals the string 'heat' exactly; if true, enter the heat branch, otherwise enter the else branch regardless of the actual mode value. RESULT branch_selected]
            [COMMAND Within the heat branch, enter a while loop that continues strictly while self.current_temperature is less than self.target_temperature (exclusive upper bound: the loop exits the moment they are equal). On each iteration, increment self.current_temperature by exactly 1 and increment use_time by exactly 1. RESULT use_time]
            [COMMAND Within the else branch, enter a while loop that continues strictly while self.current_temperature is greater than self.target_temperature (exclusive lower bound: the loop exits the moment they are equal). On each iteration, decrement self.current_temperature by exactly 1 and increment use_time by exactly 1. RESULT use_time]
            [COMMAND Return use_time as the total number of degree-steps taken; at this point self.current_temperature equals self.target_temperature. RESULT use_time]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: self.current_temperature already equals self.target_temperature when the appropriate while loop is first evaluated; the loop body never executes.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the while loop body entirely because the entry condition is false from the start. RESULT loop_skipped]
            [COMMAND Return use_time, which remains 0, indicating no steps were needed. RESULT use_time]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
