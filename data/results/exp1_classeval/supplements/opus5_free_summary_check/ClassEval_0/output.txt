import logging
from datetime import datetime, timedelta

logger = logging.getLogger(__name__)


class AccessGatewayFilter:
    WHITELIST_PREFIXES = ["/api", "/login"]

    def filter(self, request):
        """
        Main entry point. Returns True if the request is allowed through,
        False otherwise.
        """
        uri = request.get("path", "")

        if self.is_start_with(uri):
            return True

        try:
            token = self.get_jwt_user(request)
            user = token["user"]
            if user["level"] > 2:
                self.set_current_user_info_and_log(user)
                return True
            return False
        except:
            return False

    def is_start_with(self, uri):
        """
        Returns True if the URI starts with any whitelisted prefix.
        """
        for prefix in self.WHITELIST_PREFIXES:
            if uri.startswith(prefix):
                return True
        return False

    def get_jwt_user(self, request):
        """
        Extracts and validates the JWT token from the Authorization header.

        The Authorization header is expected to be a dict with:
          - 'user': a user object (dict with at least a 'name' key)
          - 'jwt': a JWT string

        If the JWT starts with the user's name, the trailing portion is
        treated as a date string. If that date is 3 or more days in the
        past, returns None (expired token). Otherwise returns the token
        as-is. If the JWT does not start with the user's name, also
        returns the token as-is.
        """
        token = request["headers"]["Authorization"]
        user = token["user"]
        jwt = token["jwt"]
        user_name = user["name"]

        if jwt.startswith(user_name):
            date_str = jwt[len(user_name):]
            token_date = datetime.strptime(date_str, "%Y-%m-%d")
            if datetime.now() - token_date >= timedelta(days=3):
                return None

        return token

    def set_current_user_info_and_log(self, user):
        """
        Logs the user's name, address, and current timestamp at level 1.
        """
        log_message = "{} {} {}".format(
            user["name"],
            user["address"],
            datetime.now().isoformat()
        )
        logger.log(1, log_message)
