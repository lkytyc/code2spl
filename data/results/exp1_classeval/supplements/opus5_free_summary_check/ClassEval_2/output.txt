class ArgumentParser:
    def __init__(self):
        self.arguments = {}
        self.required = set()
        self.types = {}

    def add_argument(self, arg, required=False, arg_type=str):
        if required:
            self.required.add(arg)
        self.types[arg] = arg_type

    def parse_arguments(self, command_string):
        self.arguments = {}
        tokens = command_string.split()
        tokens = tokens[1:]  # skip program name

        i = 0
        while i < len(tokens):
            token = tokens[i]

            if token.startswith("--"):
                key_value = token[2:]
                if "=" in key_value:
                    key, value = key_value.split("=", 1)
                    self.arguments[key] = self._convert_type(key, value)
                else:
                    self.arguments[key_value] = True
                i += 1

            elif token.startswith("-"):
                key = token[1:]
                if i + 1 < len(tokens) and not tokens[i + 1].startswith("-"):
                    value = tokens[i + 1]
                    self.arguments[key] = self._convert_type(key, value)
                    i += 2  # advance past both the flag and its value
                else:
                    self.arguments[key] = True
                    i += 1

            else:
                i += 1

        missing = self.required - self.arguments.keys()
        if missing:
            return (False, missing)

        return (True, None)

    def _convert_type(self, arg, value):
        arg_type = self.types.get(arg)
        if arg_type is None:
            return value
        try:
            return arg_type(value)
        except ValueError:
            return value

    def get_argument(self, key):
        return self.arguments.get(key, None)
