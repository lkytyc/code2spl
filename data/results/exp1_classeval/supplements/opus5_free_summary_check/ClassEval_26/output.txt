```python
import csv


class CSVProcessor:

    def read_csv(self, file_name):
        with open(file_name, newline='') as f:
            reader = csv.reader(f)
            rows = list(reader)
        title = rows[0] if rows else []
        data = rows[1:] if len(rows) > 1 else []
        return (title, data)

    def write_csv(self, rows, file_name):
        try:
            with open(file_name, 'w', newline='') as f:
                writer = csv.writer(f)
                for row in rows:
                    writer.writerow(row)
            return 1
        except Exception:
            return 0

    def process_csv_data(self, file_name, N):
        title, data = self.read_csv(file_name)
        column_values = [row[N].upper() for row in data]
        parts = file_name.split('.')
        output_file = parts[0] + '_process.' + parts[1]
        return self.write_csv([title, column_values], output_file)
```
