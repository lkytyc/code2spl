class FitnessTracker:
    def __init__(self, height: any, weight: any, age: any, sex: any) -> None:
        self.height = height
        self.weight = weight
        self.age = age
        self.sex = sex
        self.BMI_std = [{"male": [20, 25]}, {"female": [19, 24]}]

    def calculate_calorie_intake(self) -> float:
        if self.sex == "male":
            BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age + 5
        else:
            BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age - 161

        condition_value = self.condition_judge()
        if condition_value == 1:
            calorie_intake = BMR * 1.2
        else:
            condition_value = self.condition_judge()
            if condition_value == -1:
                calorie_intake = BMR * 1.6
            else:
                calorie_intake = BMR * 1.4

        return calorie_intake

    def condition_judge(self) -> int:
        BMI = self.get_BMI()
        if self.sex == "male":
            BMI_range = self.BMI_std[0]["male"]
        else:
            BMI_range = self.BMI_std[1]["female"]

        if BMI > BMI_range[1]:
            return 1
        if BMI < BMI_range[0]:
            return -1
        return 0

    def get_BMI(self) -> float:
        weight = self.weight
        height_squared = self.height ** 2
        return weight / height_squared
