class LongestWord:
    def __init__(self):
        self.word_list = []

    def add_word(self, word: object):
        self.word_list.append(word)

    def find_longest_word(self, sentence: str) -> str:
        import re

        longest_word = ""
        sentence = sentence.lower()
        sentence = re.sub(r"[^\w\s]", "", sentence)
        sentence = sentence.split(" ")

        for word in sentence:
            condition_check = word in self.word_list and len(word) > len(longest_word)
            if condition_check:
                longest_word = word

        return longest_word
