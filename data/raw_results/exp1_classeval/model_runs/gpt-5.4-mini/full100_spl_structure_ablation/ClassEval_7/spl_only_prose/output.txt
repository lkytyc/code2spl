class BalancedBrackets:
    def __init__(self, expr: str):
        self.stack = []
        self.left_brackets = ['(', '{', '[']
        self.right_brackets = [')', '}', ']']
        self.expr = expr

    def check_balanced_brackets(self) -> bool:
        self.clear_expr()
        self.stack = []

        for Brkt in self.expr:
            if Brkt in self.left_brackets:
                self.stack.append(Brkt)
            else:
                if not self.stack:
                    return False
                Current_Brkt = self.stack.pop()
                if Current_Brkt == "(":
                    if Brkt != ")":
                        return False
                elif Current_Brkt == "{":
                    if Brkt != "}":
                        return False
                elif Current_Brkt == "[":
                    if Brkt != "]":
                        return False

        if self.stack:
            return False
        return True

    def clear_expr(self):
        self.expr = "".join(
            c for c in self.expr
            if c in self.left_brackets or c in self.right_brackets
        )
