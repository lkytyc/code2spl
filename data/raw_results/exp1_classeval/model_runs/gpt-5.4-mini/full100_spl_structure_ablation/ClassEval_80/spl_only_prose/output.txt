class SQLQueryBuilder:
    def delete(self, table: str, where: dict | None) -> str:
        query = "DELETE FROM " + str(table)
        if where:
            query += " WHERE " + " AND ".join(
                f"{key}='{value}'" for key, value in where.items()
            )
        return query

    def insert(self, table: str, data: dict) -> str:
        keys = ",".join(data.keys())
        values = ",".join(f"'{value}'" for value in data.values())
        return f"INSERT INTO {table} ({keys}) VALUES ({values})"

    def select(self, table, columns, where) -> str:
        if columns != "*":
            columns = ", ".join(columns)
        query = f"SELECT {columns} FROM {table}"
        if where:
            query += " WHERE " + " AND ".join(
                f"{key}='{value}'" for key, value in where.items()
            )
        return query

    def update(self, table, data: dict, where: dict | None) -> str:
        update_str = ", ".join(f"{key}='{value}'" for key, value in data.items())
        query = f"UPDATE {table} SET {update_str}"
        if where:
            query += " WHERE " + " AND ".join(
                f"{key}='{value}'" for key, value in where.items()
            )
        return query
