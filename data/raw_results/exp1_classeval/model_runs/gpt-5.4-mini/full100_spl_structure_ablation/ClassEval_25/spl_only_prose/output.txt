class CookiesUtil:
    def __init__(self, cookies_file: any):
        self.cookies_file = cookies_file
        self.cookies = None

    def _save_cookies(self) -> bool:
        try:
            with open(self.cookies_file, "w") as file:
                import json
                json.dump(self.cookies, file)
            return True
        except Exception:
            return False

    def get_cookies(self, reponse: mapping):
        self.cookies = reponse['cookies']
        self._save_cookies()

    def load_cookies(self) -> dict:
        try:
            with open(self.cookies_file, "r") as file:
                import json
                cookies_data = json.load(file)
            return cookies_data
        except FileNotFoundError:
            return {}

    def set_cookies(self, request: dict):
        cookies_items = self.cookies.items()
        formatted_cookies = [f"{key}={value}" for key, value in cookies_items]
        cookies_header = "; ".join(formatted_cookies)
        request['cookies'] = cookies_header
