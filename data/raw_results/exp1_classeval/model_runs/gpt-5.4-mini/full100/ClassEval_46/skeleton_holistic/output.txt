class Interpolation:
    """
    This is a class that implements the Linear interpolation operation of one-dimensional and two-dimensional data
    """

    def __init__(self):
        pass

    @staticmethod
    def interpolate_1d(x, y, x_interp):
        """
        Linear interpolation of one-dimensional data
        :param x: The x-coordinate of the data point, list.
        :param y: The y-coordinate of the data point, list.
        :param x_interp: The x-coordinate of the interpolation point, list.
        :return: The y-coordinate of the interpolation point, list.
        >>> interpolation = Interpolation()
        >>> interpolation.interpolate_1d([1, 2, 3], [1, 2, 3], [1.5, 2.5])
        [1.5, 2.5]

        """
        if len(x) != len(y):
            raise ValueError("x and y must have the same length")
        if len(x) == 0:
            return []
        paired = sorted(zip(x, y), key=lambda p: p[0])
        xs = [p[0] for p in paired]
        ys = [p[1] for p in paired]

        def interp_one(xi):
            if xi < xs[0] or xi > xs[-1]:
                raise ValueError("x_interp is out of bounds")
            for i in range(len(xs)):
                if xi == xs[i]:
                    return ys[i]
                if xi < xs[i]:
                    x0, y0 = xs[i - 1], ys[i - 1]
                    x1, y1 = xs[i], ys[i]
                    if x1 == x0:
                        return y0
                    return y0 + (y1 - y0) * (xi - x0) / (x1 - x0)
            return ys[-1]

        return [interp_one(xi) for xi in x_interp]

    @staticmethod
    def interpolate_2d(x, y, z, x_interp, y_interp):
        """
        Linear interpolation of two-dimensional data
        :param x: The x-coordinate of the data point, list.
        :param y: The y-coordinate of the data point, list.
        :param z: The z-coordinate of the data point, list.
        :param x_interp: The x-coordinate of the interpolation point, list.
        :param y_interp: The y-coordinate of the interpolation point, list.
        :return: The z-coordinate of the interpolation point, list.
        >>> interpolation = Interpolation()
        >>> interpolation.interpolate_2d([1, 2, 3], [1, 2, 3], [[1, 2, 3], [4, 5, 6], [7, 8, 9]], [1.5, 2.5], [1.5, 2.5])
        [3.0, 7.0]

        """
        if len(x) != len(z) or len(y) != len(z):
            raise ValueError("x, y, and z must have the same length")
        if len(x) == 0:
            return []
        if len(x_interp) != len(y_interp):
            raise ValueError("x_interp and y_interp must have the same length")

        points = sorted(zip(x, y, z), key=lambda p: (p[0], p[1]))
        xs = sorted(set(p[0] for p in points))
        ys = sorted(set(p[1] for p in points))

        grid = {}
        for xi, yi, zi in points:
            grid[(xi, yi)] = zi

        def find_bounds(vals, v):
            if v < vals[0] or v > vals[-1]:
                raise ValueError("Interpolation point is out of bounds")
            for i, cur in enumerate(vals):
                if v == cur:
                    return cur, cur
                if v < cur:
                    return vals[i - 1], cur
            return vals[-1], vals[-1]

        def interp_bilinear(xq, yq):
            x0, x1 = find_bounds(xs, xq)
            y0, y1 = find_bounds(ys, yq)

            if x0 == x1 and y0 == y1:
                return float(grid[(x0, y0)])
            if x0 == x1:
                z0 = grid[(x0, y0)]
                z1 = grid[(x0, y1)]
                return float(z0 + (z1 - z0) * (yq - y0) / (y1 - y0))
            if y0 == y1:
                z0 = grid[(x0, y0)]
                z1 = grid[(x1, y0)]
                return float(z0 + (z1 - z0) * (xq - x0) / (x1 - x0))

            z00 = grid[(x0, y0)]
            z10 = grid[(x1, y0)]
            z01 = grid[(x0, y1)]
            z11 = grid[(x1, y1)]

            tx = (xq - x0) / (x1 - x0)
            ty = (yq - y0) / (y1 - y0)

            return float(
                z00 * (1 - tx) * (1 - ty)
                + z10 * tx * (1 - ty)
                + z01 * (1 - tx) * ty
                + z11 * tx * ty
            )

        return [interp_bilinear(xq, yq) for xq, yq in zip(x_interp, y_interp)]
