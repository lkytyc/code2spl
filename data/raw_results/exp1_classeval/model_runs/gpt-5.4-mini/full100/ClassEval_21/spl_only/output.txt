class Classroom:
    def __init__(self, id):
        self.id = id
        self.courses = []

    def add_course(self, course):
        membership_check = course in self.courses
        if not membership_check:
            self.courses.append(course)

    def remove_course(self, course):
        course_present = course in self.courses
        if course_present:
            self.courses.remove(course)

    def is_free_at(self, check_time):
        check_time = __import__("datetime").datetime.strptime(check_time, "%H:%M")
        for course in self.courses:
            start_time = __import__("datetime").datetime.strptime(course["start_time"], "%H:%M")
            end_time = __import__("datetime").datetime.strptime(course["end_time"], "%H:%M")
            comparison_result = start_time <= check_time <= end_time
            if comparison_result:
                return False
        return True

    def check_course_conflict(self, new_course):
        datetime_module = __import__("datetime")
        new_start_time = datetime_module.datetime.strptime(new_course["start_time"], "%H:%M")
        new_end_time = datetime_module.datetime.strptime(new_course["end_time"], "%H:%M")
        flag = True
        for course in self.courses:
            start_time = datetime_module.datetime.strptime(course["start_time"], "%H:%M")
            end_time = datetime_module.datetime.strptime(course["end_time"], "%H:%M")
            if start_time <= new_start_time and end_time >= new_start_time:
                flag = False
            if start_time <= new_end_time and end_time >= new_end_time:
                flag = False
        return flag
