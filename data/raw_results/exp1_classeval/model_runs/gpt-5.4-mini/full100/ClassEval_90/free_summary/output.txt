class URLHandler:
    def __init__(self, url):
        self.url = url

    def get_scheme(self):
        idx = self.url.find("://")
        if idx == -1:
            return None
        return self.url[:idx]

    def get_host(self):
        idx = self.url.find("://")
        if idx == -1:
            return None
        remainder = self.url[idx + 3:]
        slash_idx = remainder.find("/")
        if slash_idx == -1:
            return remainder
        return remainder[:slash_idx]

    def get_path(self):
        idx = self.url.find("://")
        if idx == -1:
            return None
        remainder = self.url[idx + 3:]
        slash_idx = remainder.find("/")
        if slash_idx == -1:
            return None
        return remainder[slash_idx:]

    def get_query_params(self):
        q_idx = self.url.find("?")
        if q_idx == -1:
            return None
        hash_idx = self.url.find("#", q_idx + 1)
        if hash_idx == -1:
            query_string = self.url[q_idx + 1:]
        else:
            query_string = self.url[q_idx + 1:hash_idx]

        params = {}
        if query_string == "":
            return params

        for pair in query_string.split("&"):
            if pair.count("=") == 1:
                key, value = pair.split("=")
                params[key] = value
        return params

    def get_fragment(self):
        idx = self.url.find("#")
        if idx == -1:
            return None
        return self.url[idx + 1:]
