class BitStatusUtil:
    @staticmethod
    def check(args):
        for value in args:
            if value < 0 or value % 2 != 0:
                raise ValueError("All values must be non-negative even integers.")

    @staticmethod
    def add(states, stat):
        BitStatusUtil.check([states, stat])
        return states | stat

    @staticmethod
    def has(states, stat):
        BitStatusUtil.check([states, stat])
        return (states & stat) == stat

    @staticmethod
    def remove(states, stat):
        BitStatusUtil.check([states, stat])
        if BitStatusUtil.has(states, stat):
            return states ^ stat
        return states
