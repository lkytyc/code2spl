class ExpressionCalculator:
    def __init__(self):
        self.postfix_stack = []

    def is_operator(self, ch):
        return ch in {"+", "-", "*", "/", "%", "~"}

    def compare(self, op1, op2):
        precedence = {
            "+": 1,
            "-": 1,
            "*": 2,
            "/": 2,
            "%": 2,
            "~": 3,
        }
        return precedence.get(op1, 0) - precedence.get(op2, 0)

    def transform(self, expression):
        if expression is None:
            expression = ""
        expression = "".join(expression.split())
        if expression.endswith("="):
            expression = expression[:-1]

        if not expression:
            return expression

        chars = list(expression)
        result = []

        for i, ch in enumerate(chars):
            if ch == "-":
                prev = chars[i - 1] if i > 0 else ""
                if i == 0 or prev in "+-*/%(":
                    if i + 1 < len(chars) and chars[i + 1] == "(":
                        result.append("0")
                    else:
                        result.append("~")
                    continue
            result.append(ch)

        return "".join(result)

    def prepare(self, expression):
        self.postfix_stack = []
        op_stack = []
        i = 0
        while i < len(expression):
            ch = expression[i]
            if ch.isdigit() or ch == ".":
                num = [ch]
                i += 1
                while i < len(expression) and (expression[i].isdigit() or expression[i] == "."):
                    num.append(expression[i])
                    i += 1
                self.postfix_stack.append("".join(num))
                continue
            elif ch == "(":
                op_stack.append(ch)
            elif ch == ")":
                while op_stack and op_stack[-1] != "(":
                    self.postfix_stack.append(op_stack.pop())
                if op_stack and op_stack[-1] == "(":
                    op_stack.pop()
            elif self.is_operator(ch):
                while op_stack and op_stack[-1] != "(" and self.compare(op_stack[-1], ch) >= 0:
                    self.postfix_stack.append(op_stack.pop())
                op_stack.append(ch)
            i += 1

        while op_stack:
            self.postfix_stack.append(op_stack.pop())

    def _calculate(self, a, b, op):
        from decimal import Decimal
        a = Decimal(str(a))
        b = Decimal(str(b))

        if op == "+":
            return a + b
        if op == "-":
            return a - b
        if op == "*":
            return a * b
        if op == "/":
            return a / b
        if op == "%":
            return a % b
        raise ValueError(f"Unexpected operator: {op}")

    def calculate(self, expression):
        expression = self.transform(expression)
        self.prepare(expression)

        stack = []
        for token in self.postfix_stack:
            if token == "~":
                stack.append("-")
            elif token and (token[0].isdigit() or token[0] == ".") or (token.startswith("-") and len(token) > 1):
                stack.append(token)
            elif self.is_operator(token):
                b = stack.pop()
                a = stack.pop()
                if a == "-":
                    a = 0
                if b == "-":
                    b = 0
                res = self._calculate(a, b, token)
                stack.append(res)
            else:
                stack.append(token)

        return float(stack[-1])
