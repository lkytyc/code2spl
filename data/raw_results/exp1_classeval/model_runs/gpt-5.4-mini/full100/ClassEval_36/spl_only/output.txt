class EmailClient:
    def __init__(self, addr, capacity):
        self.addr = addr
        self.capacity = capacity
        self.inbox = []

    def send_to(self, recv, content, size):
        can_send = not recv.is_full_with_one_more_email(size)
        if can_send:
            timestamp = __import__("datetime").datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            email = {
                "sender": self.addr,
                "receiver": recv.addr,
                "content": content,
                "size": size,
                "time": timestamp,
                "state": "unread",
            }
            recv.inbox.append(email)
            return True
        else:
            self.clear_inbox(size)
            return False

    def fetch(self):
        inbox_is_empty = len(self.inbox) == 0
        if inbox_is_empty:
            return None
        for i in range(len(self.inbox)):
            is_unread = self.inbox[i]["state"] == "unread"
            if is_unread:
                self.inbox[i]["state"] = "read"
                return self.inbox[i]
        return None

    def is_full_with_one_more_email(self, size):
        occupied_size = self.get_occupied_size()
        comparison_result = occupied_size + size > self.capacity
        return comparison_result

    def get_occupied_size(self):
        occupied_size = 0
        for email in self.inbox:
            occupied_size += email["size"]
        return occupied_size

    def clear_inbox(self, size):
        if not self.addr:
            return
        freed_space = 0
        while freed_space < size and self.inbox:
            email = self.inbox[0]
            freed_space += email["size"]
            self.inbox.pop(0)
