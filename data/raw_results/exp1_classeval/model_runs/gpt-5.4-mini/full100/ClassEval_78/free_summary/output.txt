class SplitSentence:
    def split_sentences(self, sentences_string):
        import re
        pattern = r'(?<!\b(?:Mr|Mrs|Ms|Dr|Prof|Sr|Jr|St|vs|etc|e\.g|i\.e|U\.S|U\.K|A\.M|P\.M))(?<=[.?])\s+'
        try:
            return re.split(pattern, sentences_string)
        except re.error:
            return re.split(r'(?<=[.?])\s+', sentences_string)

    def count_words(self, sentence):
        import re
        cleaned = re.sub(r'[^A-Za-z\s]', '', sentence)
        words = cleaned.split()
        return len(words)

    def process_text_file(self, sentences_string):
        sentences = self.split_sentences(sentences_string)
        if not sentences:
            return 0
        return max(self.count_words(sentence) for sentence in sentences)
