class CSVProcessor:
    def __init__(self):
        pass

    def read_csv(self, file_name: str):
        import csv
        data = []
        with open(file_name, "r") as file:
            reader = csv.reader(file)
            title = next(reader)
            for row in reader:
                data.append(row)
        return title, data

    def write_csv(self, data, file_name: str):
        import csv
        try:
            with open(file_name, "w", newline="") as file:
                writer = csv.writer(file)
                writer.writerows(data)
            return 1
        except Exception:
            return 0

    def process_csv_data(self, N: int, save_file_name: str):
        import os
        title, data = self.read_csv(save_file_name)
        column_data = [row[N] for row in data]
        column_data = [value.upper() for value in column_data]
        new_data = [title, column_data]
        output_file_name = os.path.splitext(save_file_name)[0] + "_process.csv"
        return self.write_csv(new_data, output_file_name)
