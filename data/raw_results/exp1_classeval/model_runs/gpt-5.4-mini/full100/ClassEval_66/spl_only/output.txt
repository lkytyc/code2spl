class NumericEntityUnescaper:
    def __init__(self):
        pass

    def replace(self, string):
        out = []
        pos = 0
        length = len(string)

        while pos < length - 2:
            if string[pos] == '&' and string[pos + 1] == '#':
                start = pos + 2
                is_hex = False
                first_char = string[start] if start < length else None

                if first_char in ('x', 'X'):
                    start += 1
                    is_hex = True

                if start == length:
                    return ''.join(out)

                end = start
                while end < length and self.is_hex_char(string[end]):
                    end += 1

                if end < length and string[end] == ';':
                    try:
                        entity_value = int(string[start:end], 16 if is_hex else 10)
                    except ValueError:
                        return ''.join(out)
                    out.append(chr(entity_value))
                    pos = end + 1
                    continue

            out.append(string[pos])
            pos += 1

        while pos < length:
            out.append(string[pos])
            pos += 1

        return ''.join(out)

    def is_hex_char(self, char):
        is_digit = char.isdigit()
        if is_digit:
            return True
        is_hex_letter = char.lower() >= 'a' and char.lower() <= 'f'
        return is_hex_letter
