class NumberConverter:

    def binary_to_decimal(self, binary_num: str) -> int:
        decimal_num = int(binary_num, 2)
        return decimal_num

    def decimal_to_binary(self, decimal_num: int) -> str:
        bin_string = bin(decimal_num)
        binary_num = bin_string[2:]
        return binary_num

    def decimal_to_hex(self, decimal_num: int) -> str:
        hex_string_with_prefix = hex(decimal_num)
        hex_num = hex_string_with_prefix[2:]
        return hex_num

    def decimal_to_octal(self, decimal_num: int) -> str:
        oct_string = oct(decimal_num)
        octal_num = oct_string[2:]
        return octal_num

    def hex_to_decimal(self, hex_num: str) -> int:
        decimal_num = int(hex_num, 16)
        return decimal_num

    def octal_to_decimal(self, octal_num: str) -> int:
        decimal_num = int(octal_num, 8)
        return decimal_num
