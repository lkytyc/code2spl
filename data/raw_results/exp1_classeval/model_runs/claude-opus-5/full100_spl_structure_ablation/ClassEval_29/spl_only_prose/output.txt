from collections import Counter


class DataStatistics:

    def mean(self, data):
        total = sum(data)
        count = len(data)
        quotient = total / count
        rounded_mean = round(quotient, 2)
        return rounded_mean

    def median(self, data):
        sorted_data = sorted(data)
        n = len(sorted_data)
        if n % 2 == 0:
            middle = n // 2
            return round((sorted_data[middle - 1] + sorted_data[middle]) / 2, 2)
        else:
            middle = n // 2
            return sorted_data[middle]

    def mode(self, data):
        counter = Counter(data)
        mode_count = max(counter.values())
        mode = [k for k, v in counter.items() if v == mode_count]
        return mode
