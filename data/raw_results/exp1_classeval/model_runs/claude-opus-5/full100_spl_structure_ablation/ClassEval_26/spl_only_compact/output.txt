import csv


class CSVProcessor:
    def __init__(self):
        pass

    def process_csv_data(self, N: int, save_file_name: str):
        title, data = self.read_csv(save_file_name)

        try:
            column_data = [row[N] for row in data]
        except IndexError:
            raise IndexError("Column index N does not exist in one or more rows of the CSV data")

        try:
            column_data = [value.upper() for value in column_data]
        except AttributeError:
            raise AttributeError("Non-string value found in column N; .upper() is not defined for that type")

        new_data = [title, column_data]

        output_file_name = save_file_name.split('.')[0] + '_process.csv'
        return self.write_csv(new_data, output_file_name)

    def read_csv(self, file_name: str):
        data = []
        with open(file_name, 'r') as file:
            reader = csv.reader(file)
            title = next(reader)
            for row in reader:
                data.append(row)
        return (title, data)

    def write_csv(self, data, file_name: str) -> int:
        try:
            with open(file_name, 'w', newline='') as file_handle:
                writer = csv.writer(file_handle)
                writer.writerows(data)
            return 1
        except:
            return 0
