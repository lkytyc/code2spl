from PIL import Image, ImageEnhance


class ImageProcessor:
    def __init__(self):
        self.image = None

    def load_image(self, image_path):
        try:
            pil_image = Image.open(image_path)
            self.image = pil_image
        except (AttributeError, TypeError) as e:
            print("Image.open received None as the path argument")
            raise
        except FileNotFoundError as e:
            print("Image.open could not locate the file at the provided path")
            raise
        except Exception as e:
            print("Image.open identified the file but could not match it to a supported format")
            raise

    def adjust_brightness(self, factor: float):
        if self.image:
            enhancer = ImageEnhance.Brightness(self.image)
            self.image = enhancer.enhance(factor)

    def resize_image(self, width: int, height: int):
        if self.image:
            self.image = self.image.resize((width, height))

    def rotate_image(self, degrees):
        if self.image:
            self.image = self.image.rotate(degrees)

    def save_image(self, save_path: str):
        if self.image:
            self.image.save(save_path)
