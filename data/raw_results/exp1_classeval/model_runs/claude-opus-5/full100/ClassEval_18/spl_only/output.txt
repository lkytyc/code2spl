class CamelCaseMap:
    def __init__(self):
        self._data = {}

    def _to_camel_case(self, key):
        parts = key.split('_')
        head = parts[0]
        tail = ''.join(part.title() for part in parts[1:])
        return head + tail

    def _convert_key(self, key):
        if isinstance(key, str):
            return self._to_camel_case(key)
        return key
