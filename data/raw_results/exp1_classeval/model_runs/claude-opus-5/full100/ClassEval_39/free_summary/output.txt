from decimal import Decimal, InvalidOperation


class ExpressionCalculator:
    def __init__(self):
        # Priority table indexed by ASCII offset from '('
        # '(' = 0, ')' = 1, '*' = 10, '+' = 3, ',' = 4, '-' = 5, '.' = 6, '/' = 9, '%' = 5 (same as '-')
        self.operat_priority = [0, 3, 3, 4, 4, -1, 1, 1, 0, 5, 5, 2, 2, 2, 2, 2, 2]
        self.postfix_stack = []

    def calculate(self, expression):
        expression = self.transform(expression)
        self.postfix_stack = []
        self.prepare(expression)

        result_stack = []
        for token in self.postfix_stack:
            if token not in ('+', '-', '*', '/', '%'):
                result_stack.append(token)
            else:
                second = result_stack.pop()
                first = result_stack.pop()
                result_stack.append(str(self._calculate(first, second, token)))

        return float(eval("*".join(result_stack)))

    def transform(self, expression):
        expression = expression.replace(' ', '')
        if expression.endswith('='):
            expression = expression[:-1]

        # Replace unary minus with '~'
        result = []
        for i, ch in enumerate(expression):
            if ch == '-':
                if i == 0:
                    # Leading minus: check if followed by '('
                    if i + 1 < len(expression) and expression[i + 1] == '(':
                        result.append('0')
                        result.append('-')
                    else:
                        result.append('~')
                elif expression[i - 1] in ('+', '-', '*', '/', '%', '(', 'E', 'e'):
                    if expression[i - 1] == '(' :
                        if i + 1 < len(expression) and expression[i + 1] == '(':
                            result.append('0')
                            result.append('-')
                        else:
                            result.append('~')
                    else:
                        result.append('~')
                else:
                    result.append(ch)
            else:
                result.append(ch)

        expression = ''.join(result)

        # Convert leading '~(' to '0-('
        if expression.startswith('~('):
            expression = '0-(' + expression[2:]

        return expression

    def prepare(self, expression):
        op_stack = [',']  # sentinel
        i = 0
        token_len = 0

        while i < len(expression):
            ch = expression[i]

            if self._is_number(ch):
                token_len += 1
                i += 1
            else:
                if token_len > 0:
                    self.postfix_stack.append(expression[i - token_len:i])
                    token_len = 0

                if ch == '(':
                    op_stack.append(ch)
                    i += 1
                elif ch == ')':
                    while op_stack[-1] != '(':
                        self.postfix_stack.append(op_stack.pop())
                    op_stack.pop()  # discard '('
                    i += 1
                else:
                    # operator: pop while top has >= precedence
                    while self.compare(ch, op_stack[-1]):
                        self.postfix_stack.append(op_stack.pop())
                    op_stack.append(ch)
                    i += 1

        # flush remaining number token
        if token_len > 0:
            self.postfix_stack.append(expression[len(expression) - token_len:])

        # flush remaining operators
        while len(op_stack) > 1:
            self.postfix_stack.append(op_stack.pop())

    def compare(self, cur, peek):
        if peek == ',':
            return False
        if peek == '(':
            return False
        return self._priority(cur) <= self._priority(peek)

    def _priority(self, op):
        if op == '%':
            op = '/'
        idx = ord(op) - ord('(')
        if 0 <= idx < len(self.operat_priority):
            return self.operat_priority[idx]
        return -1

    def _is_number(self, ch):
        return ch.isdigit() or ch == '.' or ch == '~' or ch in ('E', 'e')

    def _calculate(self, first, second, op):
        # Replace '~' prefix (unary minus representation) with actual negative sign
        first = first.replace('~', '-')
        second = second.replace('~', '-')

        try:
            a = Decimal(first)
            b = Decimal(second)
        except InvalidOperation:
            raise ValueError(f"Invalid operands: {first!r}, {second!r}")

        if op == '+':
            return a + b
        elif op == '-':
            return a - b
        elif op == '*':
            return a * b
        elif op == '/':
            return a / b
        elif op == '%':
            return a % b
        else:
            raise ValueError(f"Unrecognized operator: {op!r}")
