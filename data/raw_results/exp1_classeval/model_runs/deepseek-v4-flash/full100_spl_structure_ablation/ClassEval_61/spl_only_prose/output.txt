class MusicPlayer:
    def __init__(self):
        self.playlist = []
        self.current_song = None
        self.volume = 50

    def add_song(self, song):
        self.playlist.append(song)

    def play(self):
        if self.playlist and self.current_song:
            return self.playlist[0]
        elif self.playlist:
            return False
        # implicit None return

    def previous_song(self):
        if self.current_song:
            try:
                current_index = self.playlist.index(self.current_song)
            except ValueError:
                raise ValueError("self.playlist.index(self.current_song) raises ValueError because the current song is not present in self.playlist.")
            if current_index > 0:
                self.current_song = self.playlist[current_index - 1]
                return True
            else:
                return False
        else:
            return False

    def remove_song(self, song):
        if song in self.playlist:
            self.playlist.remove(song)
            if self.current_song == song:
                self.stop()

    def set_volume(self, volume):
        if 0 <= volume <= 100:
            self.volume = volume
            return None
        else:
            return False

    def shuffle(self):
        if self.playlist:
            import random
            random.shuffle(self.playlist)
            return True
        else:
            return False

    def stop(self):
        if self.current_song:
            self.current_song = None
            return True
        else:
            return False

    def switch_song(self):
        if self.current_song:
            try:
                current_index = self.playlist.index(self.current_song)
            except ValueError:
                raise ValueError("`self.playlist.index(self.current_song)` raises ValueError because the current song cannot be found in the playlist.")
            if current_index < len(self.playlist) - 1:
                self.current_song = self.playlist[current_index + 1]
                return True
            else:
                return False
        else:
            return False
