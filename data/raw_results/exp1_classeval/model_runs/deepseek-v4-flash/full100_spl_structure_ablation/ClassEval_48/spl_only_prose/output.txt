class IpUtil:
    @staticmethod
    def get_hostname(ip_address: str) -> str or None:
        try:
            lookup_result = socket.gethostbyaddr(ip_address)
            hostname = lookup_result[0]
            return hostname
        except socket.herror:
            return None

            return is_valid
