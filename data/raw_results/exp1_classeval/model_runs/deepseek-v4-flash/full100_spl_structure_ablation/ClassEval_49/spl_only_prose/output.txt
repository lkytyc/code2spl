class JobMarketplace:
    def __init__(self):
        self.job_listings = []
        self.resumes = []

    def get_job_applicants(self, job: dict) -> list:
        applicants = []
        for resume in self.resumes:
            if self.matches_requirements(resume, job['requirements']):
                applicants.append(resume)
        return applicants

    def matches_requirements(self, resume, requirements: collection) -> boolean:
        skills = resume["skills"]
        for current_skill in skills:
            skill_is_missing = current_skill not in requirements
            if skill_is_missing:
                return False
        return True

    def post_job(self, job_title: string, company: string, requirements: list of strings):
        job = {
            "job_title": job_title,
            "company": company,
            "requirements": requirements
        }
        self.job_listings.append(job)

    def remove_job(self, job: object):
        try:
            self.job_listings.remove(job)
        except ValueError:
            raise ValueError("job was not found in self.job_listings")

    def search_jobs(self, criteria: string) -> list:
        matching_jobs = []
        for job_listing in self.job_listings:
            if criteria.lower() in job_listing["job_title"].lower() or criteria.lower() in [req.lower() for req in job_listing["requirements"]]:
                matching_jobs.append(job_listing)
        return matching_jobs

    def submit_resume(self, name: string, skills: array, experience: string):
        resume = {
            'name': name,
            'skills': skills,
            'experience': experience
        }
        self.resumes.append(resume)

    def withdraw_resume(self, resume: object) -> None:
        try:
            self.resumes.remove(resume)
        except ValueError:
            raise ValueError("resume is not in the list")
        except AttributeError:
            raise AttributeError("self.resumes is not a list or is None")
        return None
