class AssessmentSystem:
    def __init__(self):
        self.students = {}

    def add_course_score(self, name: str, course: str, score: float) -> None:
        if name in self.students:
            self.students[name]['courses'][course] = score

    def add_student(self, name: str, grade: object, major: object) -> None:
        self.students[name] = {
            'name': name,
            'grade': grade,
            'major': major,
            'courses': {}
        }

    def get_all_students_with_fail_course(self) -> list:
        students = []
        for name, student in self.students.items():
            for course, score in student['courses'].items():
                if score < 60:
                    students.append(name)
                    break
        return students

    def get_course_average(self, course: object) -> float | None:
        total = 0
        count = 0
        for student in self.students.values():
            if course in student['courses']:
                score = student['courses'][course]
                if score is not None:
                    total += score
                    count += 1
        if count > 0:
            return total / count
        return None

    def get_gpa(self, name: str) -> float | None:
        if name in self.students and self.students[name]['courses']:
            courses = self.students[name]['courses']
            return sum(courses.values()) / len(courses)
        return None

    def get_top_student(self) -> str | None:
        top_student = None
        top_gpa = 0
        for name, student in self.students.items():
            gpa = self.get_gpa(name)
            if gpa is not None and gpa > top_gpa:
                top_gpa = gpa
                top_student = name
        return top_student
