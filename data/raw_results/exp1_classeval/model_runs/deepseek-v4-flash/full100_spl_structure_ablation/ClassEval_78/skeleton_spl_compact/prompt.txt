import re

class SplitSentence:
    """
    The class allows to split sentences, count words in a sentence, and process a text file to find the maximum word count.
    """


    def split_sentences(self, sentences_string):
        """
        Split a string into a list of sentences. Sentences end with . or ? and with a space after that. Please note that Mr. also end with . but are not sentences.
        :param sentences_string: string, string to split
        :return:list, split sentence list
        >>> ss = SplitSentence()
        >>> ss.split_sentences("aaa aaaa. bb bbbb bbb? cccc cccc. dd ddd?")
        ['aaa aaaa.', 'bb bbbb bbb?', 'cccc cccc.', 'dd ddd?']
        """

    def count_words(self, sentence):
        """
        Count the number of words in a sentence. Note that words are separated by spaces and that punctuation marks and numbers are not counted as words.
        :param sentence:string, sentence to be counted, where words are separated by spaces
        :return:int, number of words in the sentence
        >>> ss.count_words("abc def")
        2
        """

    def process_text_file(self, sentences_string):
        """
        Given a text, return the number of words in the longest sentence
        :param sentences_string: string, undivided long sentence
        :return:int, the number of words in the longest sentence
        >>> ss.process_text_file("aaa aaaa. bb bbbb bbb? cccc ccccccc cc ccc. dd ddd?")
        4
        """

Class specification:
# Class: SplitSentence
# Calls: process_text_file→count_words

## count_words(sentence: str) → int
Summary: Counts the number of words in a sentence after removing all characters except ASCII letters and whitespace.
#   sentence: str — The input text to be sanitized and split into words.
#   → return_count: int — The number of words remaining after sanitization.
1. Sanitize sentence by replacing every character that is not an ASCII letter or whitespace with an empty string, using the regex pattern [^a-zA-Z\s]. → sanitized_sentence
2. Split sanitized_sentence on any sequence of whitespace characters to produce a list of words. → words
3. Set return_count to the number of elements in words. → return_count

## process_text_file(sentences_string: string) → integer
Summary: Returns the maximum word count among all sentences in the input string by splitting the text into sentences, counting the words in each sentence, and keeping the largest count.
#   sentences_string: string — The input text to be split into sentences and scanned for the maximum word count.
#   → max_count: integer — The largest number of words found in any single sentence; 0 if the input contains no sentences.
1. Split the input sentences_string into a list of sentences using self.split_sentences. → sentences
2. Initialize max_count to 0 as the starting maximum word count. → max_count
3. Begin a loop over the sentences list, binding the current element to the variable sentence. → sentence
4. Within the loop over each sentence, compute count by calling self.count_words(sentence). → count
5. Within the loop over each sentence, if count is greater than max_count, set max_count to count; otherwise leave max_count unchanged. → max_count
6. After the loop finishes, return max_count. → max_count

## split_sentences(sentences_string: string) → list of strings
Summary: Splits a text string into sentence substrings using a regex that treats periods and question marks followed by whitespace as sentence boundaries while avoiding abbreviation-like patterns.
#   sentences_string: string — The input text to be split into sentences.
#   → result: list of strings — The list of sentence substrings produced by the regex split.
1. Split `sentences_string` with `re.split` using the pattern `(?<!\w\.\w.)(?<![A-Z][a-z]\.)(?<=\.|\?)\s`, so that whitespace preceded by '.' or '?' is a delimiter except after abbreviation-like sequences such as 'e.g.' or 'Mr.'; assign the resulting list to `sentences`. → sentences
2. Return `sentences`. → result
