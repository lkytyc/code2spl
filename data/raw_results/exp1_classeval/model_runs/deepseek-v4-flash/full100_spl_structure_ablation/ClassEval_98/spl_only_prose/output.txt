import xml.etree.ElementTree as ET

class XMLProcessor:
    def __init__(self, file_name: str):
        self.file_name = file_name
        self.root = None

    def find_element(self, element_name: str) -> list:
        elements = self.root.findall(element_name)
        return elements

    def process_xml_data(self, file_name: str) -> object:
        for element in self.root.iter('item'):
            text = element.text
            element.text = text.upper()
        return_value = self.write_xml(file_name)
        return return_value

    def read_xml(self) -> ET.Element or None:
        try:
            tree = ET.parse(self.file_name)
            root = tree.getroot()
            self.root = root
            return self.root
        except Exception:
            return None

    def write_xml(self, file_name: str) -> bool:
        try:
            tree = ET.ElementTree(self.root)
            tree.write(file_name)
            return True
        except Exception:
            return False
