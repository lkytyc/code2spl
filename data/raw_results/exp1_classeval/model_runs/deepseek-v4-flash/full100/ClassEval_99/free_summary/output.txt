import os
import zipfile

class ZipFileProcessor:
    def __init__(self, file_name):
        self.file_name = file_name

    def read_zip_file(self):
        try:
            return zipfile.ZipFile(self.file_name, "r")
        except Exception:
            return None

    def extract_all(self, output_directory):
        try:
            if output_directory:
                os.makedirs(output_directory, exist_ok=True)
            with zipfile.ZipFile(self.file_name, "r") as zf:
                zf.extractall(output_directory)
            return True
        except Exception:
            return False

    def extract_file(self, file_name, output_directory):
        try:
            if output_directory:
                os.makedirs(output_directory, exist_ok=True)
            with zipfile.ZipFile(self.file_name, "r") as zf:
                zf.extract(file_name, output_directory)
            return True
        except Exception:
            return False

    def create_zip_file(self, file_paths, output_zip_file_name):
        try:
            with zipfile.ZipFile(output_zip_file_name, "w", zipfile.ZIP_DEFLATED) as zf:
                for file_path in file_paths:
                    zf.write(file_path)
            return True
        except Exception:
            return False
