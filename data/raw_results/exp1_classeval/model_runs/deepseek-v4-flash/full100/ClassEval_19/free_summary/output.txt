class ChandrasekharSieve:
    def __init__(self, n):
        self.n = n
        self.primes = []
        if n >= 2:
            sieve = [True] * (n + 1)
            sieve[0] = sieve[1] = False
            for i in range(2, int(n ** 0.5) + 1):
                if sieve[i]:
                    for j in range(i * i, n + 1, i):
                        sieve[j] = False
            self.primes = [i for i in range(2, n + 1) if sieve[i]]

    def get_primes(self):
        return self.primes
