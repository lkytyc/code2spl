class IPAddress:
    def __init__(self, address):
        self.address = address
        parts = address.split(".")
        self.octets = []
        self._valid = False

        if len(parts) == 4:
            valid = True
            for part in parts:
                if not part.isdecimal() or not 0 <= int(part) <= 255:
                    valid = False
                    break
            if valid:
                self._valid = True
                self.octets = parts

    def get_octets(self):
        return list(self.octets)

    def get_binary(self):
        if not self._valid:
            return ""
        return ".".join(format(int(octet), "08b") for octet in self.octets)
