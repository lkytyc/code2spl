```python
import os
import json

class JSONProcessor:
    def process_json(self, file_path: str, remove_key: str) -> int:
        data = self.read_json(file_path)
        if data == 0 or data == -1:
            return 0
        if remove_key in data:
            del data[remove_key]
            self.write_json(data, file_path)
            return 1
        return 0

    def read_json(self, file_path: str):
        if os.path.exists(file_path):
            try:
                with open(file_path, 'r') as file:
                    data = json.load(file)
                return data
            except Exception:
                return -1
        return 0

    def write_json(self, data: object, file_path: str) -> int:
        try:
            with open(file_path, 'w') as file:
                json.dump(data, file)
            return 1
        except Exception:
            return -1
```
