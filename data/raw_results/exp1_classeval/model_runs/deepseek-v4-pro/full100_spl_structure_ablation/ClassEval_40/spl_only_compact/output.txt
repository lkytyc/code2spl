class FitnessTracker:
    def __init__(self, height: float, weight: float, age: int, sex: str):
        self.height = height
        self.weight = weight
        self.age = age
        self.sex = sex
        self.BMI_std = [
            {"male": [20, 25]},
            {"female": [19, 24]}
        ]

    def calculate_calorie_intake(self) -> float:
        if self.sex == "male":
            BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age + 5
        else:
            BMR = 10 * self.weight + 6.25 * self.height - 5 * self.age - 161

        first_judge = self.condition_judge()
        if first_judge == 1:
            calorie_intake = BMR * 1.2
        else:
            second_judge = self.condition_judge()
            if second_judge == -1:
                calorie_intake = BMR * 1.6
            else:
                calorie_intake = BMR * 1.4

        return calorie_intake

    def condition_judge(self) -> int:
        BMI = self.get_BMI()
        if self.sex == "male":
            BMI_range = self.BMI_std[0]["male"]
        else:
            BMI_range = self.BMI_std[1]["female"]

        if BMI > BMI_range[1]:
            return 1
        elif BMI < BMI_range[0]:
            return -1
        else:
            return 0

    def get_BMI(self) -> float:
        return self.weight / (self.height ** 2)
