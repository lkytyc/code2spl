import random


class TwentyFourPointGame:
    def __init__(self):
        self.nums = []

    def _generate_cards(self):
        for i in range(4):
            self.nums.append(random.randint(1, 9))
        assert len(self.nums) == 4, "The generated card list does not contain exactly four cards."

    def answer(self, expression: str):
        if expression == 'pass':
            return self.get_my_cards()
        statistic = {}
        for c in expression:
            if c.isdigit() and int(c) in self.nums:
                statistic[c] = statistic.get(c, 0) + 1
        nums_used = statistic.copy()
        for num in self.nums:
            if nums_used.get(str(num), -100) == -100 or nums_used[str(num)] <= 0:
                return False
            nums_used[str(num)] -= 1
        all_zero = all(count == 0 for count in nums_used.values())
        if all_zero:
            return self.evaluate_expression(expression)
        return False

    def evaluate_expression(self, expression: str) -> bool:
        try:
            evaluated_value = eval(expression)
        except Exception:
            return False
        return evaluated_value == 24

    def get_my_cards(self) -> list:
        self.nums = []
        self._generate_cards()
        return self.nums
