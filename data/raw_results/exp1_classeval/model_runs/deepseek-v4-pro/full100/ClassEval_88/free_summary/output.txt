class TriCalculator:
    def cos(self, x):
        return round(self.taylor(x, 50), 10)

    def factorial(self, a):
        result = 1
        for i in range(1, a + 1):
            result *= i
        return result

    def taylor(self, x, n):
        x = x * 3.141592653589793 / 180
        result = 0
        for i in range(n):
            term = (x ** (2 * i)) / self.factorial(2 * i)
            if i % 2 == 0:
                result += term
            else:
                result -= term
        return result

    def sin(self, x):
        x = x * 3.141592653589793 / 180
        result = 0
        term = x
        n = 1
        while abs(term) > 1e-15:
            result += term
            term = -term * x * x / ((2 * n) * (2 * n + 1))
            n += 1
        return round(result, 10)

    def tan(self, x):
        c = self.cos(x)
        if c == 0:
            return False
        return round(self.sin(x) / c, 10)
