class BracketChecker:
    def __init__(self, expression):
        self.expression = expression

    def is_balanced(self):
        allowed = set("(){}[]")
        filtered = ''.join(ch for ch in self.expression if ch in allowed)
        stack = []
        matching = {')': '(', '}': '{', ']': '['}
        for ch in filtered:
            if ch in "({[":
                stack.append(ch)
            else:
                if not stack or stack.pop() != matching[ch]:
                    return False
        return not stack
