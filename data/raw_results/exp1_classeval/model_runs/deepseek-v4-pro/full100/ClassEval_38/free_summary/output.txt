class ExcelProcessor:
    def __init__(self):
        try:
            import openpyxl
            self.openpyxl = openpyxl
        except ImportError:
            self.openpyxl = None

    def read_excel(self, file_name):
        if self.openpyxl is None:
            return None
        try:
            wb = self.openpyxl.load_workbook(file_name)
            ws = wb.active
            rows = []
            for row in ws.iter_rows(values_only=True):
                rows.append(tuple(row))
            wb.close()
            return rows
        except Exception:
            return None

    def write_excel(self, data, file_name):
        if self.openpyxl is None:
            return 0
        try:
            wb = self.openpyxl.Workbook()
            ws = wb.active
            for row in data:
                ws.append(list(row))
            wb.save(file_name)
            wb.close()
            return 1
        except Exception:
            return 0

    def process_excel_data(self, N, save_file_name):
        rows = self.read_excel(save_file_name)
        if rows is None:
            return 0
        if not rows:
            return 0
        if N < 0 or N >= len(rows[0]):
            return 0
        processed = []
        for row in rows:
            row = list(row)
            value = row[N]
            if value is not None and str(value).isdigit():
                row.append(value)
            else:
                row.append(str(value).upper())
            processed.append(row)
        new_file_name = save_file_name.replace('.xlsx', '_process.xlsx')
        if new_file_name == save_file_name:
            new_file_name = save_file_name + '_process'
        result = self.write_excel(processed, new_file_name)
        return (result, new_file_name)
