import numpy as np


class MetricsCalculator2:
    def __init__(self):
        pass

    def mrr(self, data):
        if type(data) is not list and type(data) is not tuple:
            raise Exception("the input must be a tuple of sub-list and total number, or an iteration of list of tuples")
        if len(data) == 0:
            return (0.0, [0.0])
        if type(data) is tuple:
            sub_list, total_num = data
            sub_list = np.array(sub_list)
            if total_num == 0:
                return (0.0, [0.0])
            ranking_array = 1.0 / np.arange(1, len(sub_list) + 1)
            mr_np = sub_list * ranking_array
            mr = 0.0
            for team in mr_np:
                if team > 0:
                    mr = team
                    break
            return (mr, [mr])
        if type(data) is list:
            separate_result = []
            for sub_list, total_num in data:
                sub_list = np.array(sub_list)
                if total_num == 0:
                    mr = 0.0
                else:
                    ranking_array = 1.0 / np.arange(1, len(sub_list) + 1)
                    mr_np = sub_list * ranking_array
                    mr = 0.0
                    for team in mr_np:
                        if team > 0:
                            mr = team
                            break
                separate_result.append(mr)
            return (np.mean(separate_result), separate_result)

    def map(self, data):
        if type(data) is not list and type(data) is not tuple:
            raise Exception("the input must be a tuple([0,...,1,...],int) or a iteration of list of tuple")
        if len(data) == 0:
            return (0.0, [0.0])
        if type(data) is tuple:
            sub_list, total_num = data
            sub_list = np.array(sub_list)
            if total_num == 0:
                return (0.0, [0.0])
            ranking_array = 1.0 / (np.arange(len(sub_list)) + 1)
            right_ranking_list = []
            count = 1
            for t in sub_list:
                if t == 0:
                    right_ranking_list.append(0)
                else:
                    right_ranking_list.append(count)
                    count += 1
            ap = np.sum(np.array(right_ranking_list) * ranking_array) / total_num
            return (ap, [ap])
        if type(data) is list:
            separate_result = []
            for sub_list, total_num in data:
                sub_list = np.array(sub_list)
                if total_num == 0:
                    ap = 0.0
                else:
                    ranking_array = 1.0 / (np.arange(len(sub_list)) + 1)
                    right_ranking_list = []
                    count = 1
                    for t in sub_list:
                        if t == 0:
                            right_ranking_list.append(0)
                        else:
                            right_ranking_list.append(count)
                            count += 1
                    ap = np.sum(np.array(right_ranking_list) * ranking_array) / total_num
                separate_result.append(ap)
            return (np.mean(separate_result), separate_result)
