class IPAddress:
    def __init__(self, ip):
        self.ip = ip

    def is_valid(self):
        parts = self.ip.split('.')
        if len(parts) != 4:
            return False
        for part in parts:
            if not part.isdigit():
                return False
            num = int(part)
            if num < 0 or num > 255:
                return False
        return True

    def get_octets(self):
        if not self.is_valid():
            return []
        return self.ip.split('.')

    def get_binary(self):
        if not self.is_valid():
            return ""
        octets = self.get_octets()
        binary_octets = [format(int(octet), '08b') for octet in octets]
        return '.'.join(binary_octets)
