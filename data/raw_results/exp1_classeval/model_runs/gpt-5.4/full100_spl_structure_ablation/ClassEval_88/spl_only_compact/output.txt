class TriCalculator:
    def __init__(self):
        pass

    def cos(self, x):
        taylor_value = self.taylor(x, 50)
        result = round(taylor_value, 10)
        return result

    def factorial(self, a):
        b = 1
        while a != 1:
            b *= a
            a -= 1
        return b

    def sin(self, x):
        import math

        x = x / 180 * math.pi
        g = 0
        t = x
        n = 1

        while math.fabs(t) >= 1e-15:
            g += t
            n += 1
            t = -t * x * x / (2 * n - 1) / (2 * n - 2)

        return round(g, 10)

    def tan(self, x):
        cos_value = self.cos(x)
        if cos_value != 0:
            result = self.sin(x) / self.cos(x)
            return round(result, 10)
        return False

    def taylor(self, x, n):

        x = x / 180 * math.pi
        count = 1
