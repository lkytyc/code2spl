import numpy as np


class MetricsCalculator2:
    def __init__(self):
        pass

    def map(self, data):
        if not isinstance(data, (list, tuple)):
            raise Exception("the input must be a tuple([0,...,1,...],int) or a iteration of list of tuple")

        if len(data) == 0:
            return 0.0, [0.0]

        if isinstance(data, tuple):
            sub_list, total_num = data
            sub_list = np.array(sub_list)
            if total_num == 0:
                return 0.0, [0.0]

            ranking_array = 1.0 / np.arange(1, len(sub_list) + 1)
            rank_count = 0
            ranked_hits = []
            for v in sub_list:
                if v != 0:
                    rank_count += 1
                    ranked_hits.append(rank_count)
                else:
                    ranked_hits.append(0)
            ranked_hits = np.array(ranked_hits)
            ap = float(np.sum(ranked_hits * ranking_array) / total_num)
            return ap, [ap]

        separate_result = []
        for sub_list, total_num in data:
            sub_list = np.array(sub_list)
            if total_num == 0:
                ap = 0.0
            else:
                ranking_array = 1.0 / np.arange(1, len(sub_list) + 1)
                rank_count = 0
                ranked_hits = []
                for v in sub_list:
                    if v != 0:
                        rank_count += 1
                        ranked_hits.append(rank_count)
                    else:
                        ranked_hits.append(0)
                ranked_hits = np.array(ranked_hits)
                ap = float(np.sum(ranked_hits * ranking_array) / total_num)
            separate_result.append(ap)

        return float(np.mean(separate_result)), separate_result

    def mrr(self, data):
        if type(data) is not list and type(data) is not tuple:
            raise Exception("the input must be a tuple([0,...,1,...],int) or a iteration of list of tuple")

        if len(data) == 0:
            return 0.0, [0.0]

        if type(data) is tuple:
            sub_list, total_num = data
            sub_list = np.array(sub_list)

            if total_num == 0:
                return 0.0, [0.0]

            ranking_array = 1.0 / np.array(range(1, len(sub_list) + 1))
            mr_np = sub_list * ranking_array

            mr = 0.0
            for v in mr_np:
                if v > 0:
                    mr = float(v)
                    break
            return mr, [mr]

        separate_result = []
        for sub_list, total_num in data:
            sub_list = np.array(sub_list)

            if total_num == 0:
                mr = 0.0
            else:
                ranking_array = 1.0 / np.array(range(1, len(sub_list) + 1))
                mr_np = sub_list * ranking_array
                mr = 0.0
                for v in mr_np:
                    if v > 0:
                        mr = float(v)
                        break

            separate_result.append(mr)

        return float(np.mean(separate_result)), separate_result
