from urllib.parse import unquote


class UrlPath:
    def __init__(self):
        self.segments = []
        self.with_end_tag = False

    @staticmethod
    def fix_path(path):
        if path is None:
            return ""
        return path.strip("/")

    def add(self, segment):
        self.segments.append(self.fix_path(segment))

    def parse(self, path, charset):
        if path and path.endswith("/"):
            self.with_end_tag = True
        fixed = self.fix_path(path)
        if fixed:
            for segment in fixed.split("/"):
                self.add(unquote(segment, encoding=charset))
