class BitFlags:
    @staticmethod
    def add(states, stat):
        BitFlags.check((states, stat))
        return states | stat

    @staticmethod
    def has(states, stat):
        BitFlags.check((states, stat))
        return (states & stat) == stat

    @staticmethod
    def remove(states, stat):
        BitFlags.check((states, stat))
        if BitFlags.has(states, stat):
            return states ^ stat
        return states

    @staticmethod
    def check(args):
        for value in args:
            if value < 0:
                raise ValueError("Value must be non-negative")
            if value % 2 != 0:
                raise ValueError("Value must be even")
