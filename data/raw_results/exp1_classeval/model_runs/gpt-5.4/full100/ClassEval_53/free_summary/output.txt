import string

class LongestWord:
    def __init__(self):
        self.word_list = []

    def add_word(self, word):
        self.word_list.append(word)

    def find_longest_word(self, sentence):
        result = ""
        cleaned_sentence = sentence.lower().translate(str.maketrans("", "", string.punctuation))
        pieces = cleaned_sentence.split("")

        for piece in pieces:
            if piece in self.word_list and len(piece) > len(result):
                result = piece

        return result
