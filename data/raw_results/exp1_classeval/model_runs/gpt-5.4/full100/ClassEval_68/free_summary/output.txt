class PageUtil:
    def __init__(self, data, page_size):
        self.data = data
        self.page_size = page_size
        self.total_items = len(data)
        self.total_pages = (self.total_items + page_size - 1) // page_size if page_size > 0 else 0

    def get_page(self, page_number):
        if page_number < 1 or page_number > self.total_pages:
            return []
        start = (page_number - 1) * self.page_size
        end = min(start + self.page_size, self.total_items)
        return list(self.data[start:end])

    def get_page_info(self, page_number):
        if page_number < 1 or page_number > self.total_pages:
            return {}
        start = (page_number - 1) * self.page_size
        end = min(start + self.page_size, self.total_items)
        items = list(self.data[start:end])
        return {
            "page_number": page_number,
            "page_size": self.page_size,
            "total_pages": self.total_pages,
            "total_items": self.total_items,
            "has_previous": page_number > 1,
            "has_next": page_number < self.total_pages,
            "items": items,
        }

    def search(self, keyword):
        matches = [item for item in self.data if keyword in str(item)]
        total_matches = len(matches)
        pages = (total_matches + self.page_size - 1) // self.page_size if self.page_size > 0 else 0
        return {
            "keyword": keyword,
            "total_matches": total_matches,
            "total_pages": pages,
            "results": matches,
        }
