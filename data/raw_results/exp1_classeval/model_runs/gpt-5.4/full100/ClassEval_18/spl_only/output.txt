class CamelCaseMap:
    def __init__(self):
        self._data = {}

    def _convert_key(self, key):
        if isinstance(key, str):
            return self._to_camel_case(key)
        return key

    def _to_camel_case(self, key):
        parts = key.split('_')
        suffix = ''.join(part.title() for part in parts[1:])
        return parts[0] + suffix
