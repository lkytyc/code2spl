[DEFINE_WORKER: "Given coordinate-weight pairs and threshold s, merges weights inward from both ends and returns the accumulated traversal distance, with direct boundary shortcuts when the entire range lies to one side of s." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> r </REF>: int "The accumulated result returned by the method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer n and integer s from input using LI(). RESULT n_and_s]
            [COMMAND Read n integer pairs into list a by calling LI() once for each pair. RESULT a]
            [COMMAND Set x to the list of first elements of each pair in a, and set p to the list of second elements of each pair in a. RESULT x_and_p]
            [COMMAND If x[0] > s, return x[-1] - s immediately. RESULT return_value]
            [COMMAND If x[-1] < s, return s - x[0] immediately. RESULT return_value]
            [COMMAND Initialize r = 0, i = 0, j = n - 1, d as a shallow copy of p, create e using collections.defaultdict(list), and initialize sk as an empty set. RESULT initialized_state]
            [COMMAND While i < j: if x[i] > s, break; if x[j] < s, break; if d[i] >= d[j], then add d[j] to d[i], decrement j, and if j >= 0 and d[j] < d[j+1], then set d[j] = d[j+1] and add i to sk; otherwise, add d[i] to d[j], increment i, and if i < n and d[i] < d[i-1], then set d[i] = d[i-1] and add i to sk. RESULT state_after_first_while]
            [COMMAND Set t = s. RESULT t]
            [COMMAND If i == j, add abs(x[i] - s) to r, set t = x[i], decrement i, and increment j; otherwise, if x[i] > s, add abs(x[j] - s) to r, set t = x[j], decrement i, and increment j; otherwise, add abs(x[i] - s) to r, set t = x[i], decrement i, and increment j. RESULT state_after_midpoint]
            [COMMAND While i >= 0 or j < n: if i < 0, add abs(x[-1] - t) to r and break; if j >= n, add abs(x[0] - t) to r and break; if i is in sk, set d[i-1] = d[i], decrement i, and continue; if j is in sk, set d[j+1] = d[j], increment j, and continue; if d[i] >= d[j], add abs(x[i] - t) to r, set t = x[i], and decrement i; otherwise, add abs(x[j] - t) to r, set t = x[j], and increment j. RESULT r]
            [COMMAND Return r. RESULT r]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
