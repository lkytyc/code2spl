[DEFINE_WORKER: "Finds a lexicographically missing word of the same length as the first sorted word by scanning each character position and prefix groups, padding the first missing prefix to full length, or returns '-' if no such word exists." FindNewWord]
    [INPUTS]
        <REF> w_list </REF>: list of strings "List of words to search; it is sorted in place."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "A newly formed missing word, or '-' if no missing word is found or the input has only one word/first word length 1."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the input list w_list in place using its natural ordering. RESULT w_list]
            [COMMAND Set L to the length of the first element w_list[0]. RESULT L]
            [COMMAND Check whether len(w_list) == 1 or L == 1; if true, use the early-return alternative. RESULT early_condition]
            [COMMAND Initialize prefix_index to [0]. RESULT prefix_index]
            [COMMAND Set N to len(w_list). RESULT N]
            [COMMAND Set last_prefix to w_list[0][0]. RESULT last_prefix]
            [COMMAND For i from 1 to N-1 inclusive, if w_list[i][0] differs from current last_prefix, update last_prefix to w_list[i][0] and append i to prefix_index. RESULT prefix_index]
            [COMMAND Append N to prefix_index as the sentinel boundary. RESULT prefix_index]
            [COMMAND Initialize curr_word to an empty string. RESULT curr_word]
            [COMMAND For each position p from 1 to L-1 inclusive, build available_letters as a set initially containing w_list[0][p]; then, for k from 1 to N-1 inclusive, add w_list[k][p] to available_letters. RESULT available_letters]
            [COMMAND In the same p iteration, set n_prefixes to len(prefix_index), isFound to False, and new_prefix_index to an empty list. RESULT group_scan_state]
            [COMMAND For n_prefix from 0 to n_prefixes-2 inclusive, set start_index to prefix_index[n_prefix] and end_index to prefix_index[n_prefix+1]. RESULT start_index, end_index]
            [COMMAND Within the current prefix group, initialize letters_in_prefix and last_letter with w_list[start_index][p], and append start_index to new_prefix_index. RESULT letters_in_prefix]
            [COMMAND Within the group, for item from start_index+1 to end_index-1 inclusive, if w_list[item][p] differs from last_letter, update last_letter to w_list[item][p], add w_list[item][p] to letters_in_prefix, and append item to new_prefix_index. RESULT letters_in_prefix]
            [COMMAND After scanning the group, compute remaining_letters = available_letters.difference(letters_in_prefix). RESULT remaining_letters]
            [COMMAND If remaining_letters is non-empty, choose one letter from remaining_letters, set curr_word to w_list[start_index][:p] + letter, set isFound to True, and break out of the n_prefix loop. RESULT curr_word]
            [COMMAND After the n_prefix loop, if isFound is True, break out of the p loop. RESULT isFound]
            [COMMAND If isFound is still False, append N to new_prefix_index and set prefix_index to deepcopy(new_prefix_index). RESULT prefix_index]
            [COMMAND After the p loop completes, if isFound is False, follow the not-found alternative return path. RESULT isFound]
            [COMMAND Set size_p to len(curr_word). RESULT size_p]
            [COMMAND For p from size_p to L-1 inclusive, append w_list[0][p] to curr_word. RESULT curr_word]
            [COMMAND Return curr_word. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(w_list) == 1 or L == 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the constant string '-' immediately and stop. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: isFound is False after the outer p loop]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the constant string '-' immediately and stop. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
