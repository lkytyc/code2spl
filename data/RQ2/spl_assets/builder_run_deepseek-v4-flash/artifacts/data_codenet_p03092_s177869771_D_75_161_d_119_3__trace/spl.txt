[DEFINE_WORKER: "Computes and prints the minimal total cost to transform the input permutation pf into sorted order by repeatedly moving misplaced maximal consecutive-rank blocks to their target positions; it performs two identical passes and keeps the better cost." D]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read n, a, and b from a single line of space-separated integers using LI(). RESULT n,a,b]
            [COMMAND Read pf as a list of integers from another LI() call. RESULT pf]
            [COMMAND Initialize f_ans to positive infinity. RESULT f_ans]
            [COMMAND Initialize l as an empty list. RESULT l]
            [COMMAND For i from 0 to n-1, append pf[i] to l. RESULT l]
            [COMMAND Sort l in ascending order. RESULT l]
            [COMMAND Initialize f as a defaultdict with int default factory. RESULT f]
            [COMMAND For each element i in pf, set f[i] to the first index of i in l using l.index(i). RESULT f]
            [COMMAND For c in range(2), execute the following two-pass process. RESULT c]
            [COMMAND Set p to a copy of pf. RESULT p]
            [COMMAND Set ans to 0. RESULT ans]
            [COMMAND While l is not equal to p, execute the transformation loop. RESULT p]
            [COMMAND Inside the transformation loop, set d to an empty list and i to 0. RESULT d,i]
            [COMMAND While i < n, start j = i and advance j repeatedly while j < n-1 and f[p[j+1]] == f[p[j]] + 1; then append the slice p[i:j+1] to d and set i = j+1, partitioning p into maximal consecutive-rank blocks. RESULT d]
            [COMMAND Set m to len(d). RESULT m]
            [COMMAND Set ld to a copy of d, then sort ld in ascending order. RESULT ld]
            [COMMAND Initialize fd as a defaultdict(int). RESULT fd]
            [COMMAND For each block i in d, set fd[str(i)] to the first index of that block in ld. RESULT fd]
            [COMMAND For the current c value, execute the block-movement scan; the c == 0 branch and the else branch contain identical logic, so the same scan is performed either way. RESULT c]
            [COMMAND Scan block indices i from m-1 down to 0 to find the rightmost block d[i] that is not equal to ld[i]. RESULT i]
            [COMMAND If such a mismatch is found, set j = fd[str(d[i])] and x = d[i]. RESULT j,x]
            [COMMAND If j > i, initialize k to 0 and sum the lengths of blocks d[i+1] through d[j] inclusive into k; then temporarily set p to len(d[i]), add min(b*k, a*p) to ans, pop d[i], insert x at position j, and break out of the scan. RESULT ans,d]
            [COMMAND If such a mismatch was found and j is not greater than i, initialize k to 0 and sum the lengths of blocks d[j] through d[i-1] inclusive into k; then temporarily set p to len(d[i]), add min(a*k, b*p) to ans, pop d[i], insert x at position j, and break out of the scan. RESULT ans,d]
            [COMMAND After the block movement, set p to a list of n None entries. RESULT p]
            [COMMAND Set s to 0. RESULT s]
            [COMMAND Reconstruct p by copying each block in d in order: for block index i and element index j within len(d[i]), set p[s+j] = d[i][j]; then increment s by len(d[i]). RESULT p]
            [COMMAND When l == p, update f_ans to min(f_ans, ans). RESULT f_ans]
            [COMMAND After both c iterations complete, print f_ans. RESULT f_ans]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
