[DEFINE_WORKER: "Reads n and array a, then repeatedly marks the current target value found by scanning from the right while maintaining seen boundaries and a lower limit l; on an invalid state it prints -1, otherwise it prints all collected answer values in order." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next integer from ni() and store it as n. RESULT n]
            [COMMAND Allocate an integer array a of length n, with all elements initially 0. RESULT a]
            [COMMAND For i from 0 to n-1 inclusive, read the next integer from ni() and assign it to a[i]. RESULT a]
            [COMMAND Allocate a boolean array seen of length n, with every element initialized to false. RESULT seen]
            [COMMAND Create an empty ArrayList of Integer named ans. RESULT ans]
            [COMMAND Set t to 1. RESULT t]
            [COMMAND Set l to 0. RESULT l]
            [COMMAND Set r to n-1. RESULT r]
            [COMMAND Enter an infinite while(true) loop; it will be exited only by an explicit break or return. RESULT result]
            [COMMAND Within the while loop, start a backward scan over i from n-1 down to 0 inclusive. The scan is labeled outer, but no labeled break or continue uses that label. RESULT result]
            [COMMAND Within the backward scan, if seen[i] is true, continue with the next i. RESULT result]
            [COMMAND Within the backward scan, if seen[i] is false and i < l, print -1 to out and return from solve. RESULT result]
            [COMMAND Within the backward scan, if seen[i] is false, i >= l, and a[i] != t, do nothing and continue with the next i. RESULT result]
            [COMMAND Within the backward scan, if seen[i] is false, i >= l, and a[i] == t, set seen[i] to true and append a[i] to ans. RESULT result]
            [COMMAND After the match in the previous step, set flag to true; then for j from i+1 to n-1 inclusive, if seen[j] is false, set flag to false. RESULT flag]
            [COMMAND If flag is true after that inner j scan, break out of the backward scan. RESULT result]
            [COMMAND If flag is false after that inner j scan, increment t, set i to n, and continue the backward scan; the for-loop's i-- update causes the next iteration to resume at i = n-1. RESULT t]
            [COMMAND After the backward scan exits, if seen[n-1] is false, print -1 to out and return from solve. RESULT result]
            [COMMAND Set flag to true. RESULT flag]
            [COMMAND Within the while loop, start a second backward scan over i from n-1 down to 0 inclusive. RESULT result]
            [COMMAND Within the second backward scan, find the rightmost i for which seen[i] is false. If no such i exists, leave flag true and continue after the scan. RESULT result]
            [COMMAND If such an i exists, set t to a[i+1], set flag to false, and set r to i. RESULT result]
            [COMMAND Still within that selected i, run an inner backward scan over j from i down to 0 inclusive; if seen[j] is true, break out of the inner scan; otherwise set l to j. RESULT l]
            [COMMAND After handling that selected i, break out of the second backward scan. RESULT result]
            [COMMAND After the second backward scan, if flag is true, break out of the while loop. RESULT result]
            [COMMAND If flag is false after the second backward scan, return to the top of the while loop and execute the while-loop body again. RESULT result]
            [COMMAND After the while loop exits, iterate over ans in iteration order and print each element i to out with a newline. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
