[DEFINE_WORKER: "Reads a grid of h rows and w columns, marks '#' cells as mines with -1, replaces all non-mine cells with their eight-neighbor mine count, and prints the resulting minesweeper grid." calc]
    [INPUTS]
        <REF> h </REF>: int "Number of rows in the panel grid."
        <REF> w </REF>: int "Number of columns in the panel grid."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return </REF>: int "Always returns 0 after printing the computed grid."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare a two-dimensional integer array panel of size h by w and a one-dimensional character array row of length w; then, for each row index i from 0 to h-1, read a whitespace-delimited token into row using scanf("%s", row). RESULT grid_input_read]
            [COMMAND For each row index i from 0 to h-1 and each column index j from 0 to w-1, assign panel[i][j] = -1 if row[j] == '#', otherwise assign panel[i][j] = 0. RESULT panel_initialized]
            [COMMAND For each row index i from 0 to h-1 and each column index j from 0 to w-1, check whether panel[i][j] == -1; if it is -1, continue to the next cell without modifying it, otherwise initialize count = 0 for this cell. RESULT cell_mine_status_checked]
            [COMMAND For each non-mine cell (i,j), inspect all eight neighboring cells in the surrounding 3x3 area excluding (i,j); for each neighbor whose row is between 0 and h-1 inclusive and column is between 0 and w-1 inclusive, increment count if that neighbor's panel value equals -1. RESULT neighbor_mine_count]
            [COMMAND Assign panel[i][j] = count for the current non-mine cell. RESULT panel_with_counts]
            [COMMAND For each row index i from 0 to h-1 and each column index j from 0 to w-1, print '#' if panel[i][j] == -1, otherwise print the integer value of panel[i][j] using printf("%d", panel[i][j]); after finishing each row, print a newline using printf("\n"). RESULT grid_printed]
            [COMMAND Return 0 from the calc function. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
