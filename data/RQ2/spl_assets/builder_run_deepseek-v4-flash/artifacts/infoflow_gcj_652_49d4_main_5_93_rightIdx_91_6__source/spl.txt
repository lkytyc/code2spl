[DEFINE_WORKER: "Simulates the Bathroom Stalls problem for each test case by repeatedly selecting the empty stall with the largest minimum distance to occupied stalls, using max-side and leftmost tie-breakers, and prints the last selected stall's larger and smaller side distances." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the number of test cases from standard input as an integer and store it in T. RESULT T]
            [COMMAND For each test case index i from 1 to T inclusive, read the next line from standard input, split it into tokens, convert the tokens to integers, store the first as N and the second as K. RESULT i]
            [COMMAND For each test case index i from 1 to T inclusive, inside the test-case loop, initialize stall_occupied as [True]; for each j from 0 to N-1 inclusive, append False; then append True, so index 0 and index N+1 are occupied sentinels and indices 1..N are empty stalls. RESULT stall_occupied]
            [COMMAND For each test case index i from 1 to T inclusive, inside the test-case loop, initialize Ls, Rs, Min_s, and Max_s as empty lists; for each index j from 0 to N+1 inclusive, if stall_occupied[j] is True append -1 to Ls and Rs, otherwise append j-1 to Ls and N-j to Rs; after appending each pair, append min(Ls[-1], Rs[-1]) to Min_s and max(Ls[-1], Rs[-1]) to Max_s. RESULT Ls]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, compute maximal = max(Min_s). RESULT maximal]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, select the stall index idx by maximizing Min_s, then maximizing Max_s on ties, and then choosing the leftmost remaining tied index; if the maximal Min_s is unique, choose the first index with that value. The branch-specific selection steps are recorded in alternative_flows. RESULT idx]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, after idx is selected set y = max(Ls[idx], Rs[idx]) and z = min(Ls[idx], Rs[idx]). RESULT y]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, mark stall_occupied[idx] as True and set Ls[idx], Rs[idx], Min_s[idx], Max_s[idx] to -1. RESULT stall_occupied]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, update left distances: set leftIdx = idx-1; while leftIdx > 0 and stall_occupied[leftIdx] is False, set Rs[leftIdx] = idx-leftIdx-1, Min_s[leftIdx] = min(Rs[leftIdx], Ls[leftIdx]), Max_s[leftIdx] = max(Rs[leftIdx], Ls[leftIdx]), and decrement leftIdx; stop at an occupied stall or boundary index 0. RESULT Rs]
            [COMMAND For each test case index i from 1 to T inclusive, for each k from 0 to K-1 inclusive inside the test-case loop, update right distances: set rightIdx = idx+1; while rightIdx < N+2 and stall_occupied[rightIdx] is False, set Ls[rightIdx] = rightIdx-idx-1, Min_s[rightIdx] = min(Rs[rightIdx], Ls[rightIdx]), Max_s[rightIdx] = max(Rs[rightIdx], Ls[rightIdx]), and increment rightIdx; stop at an occupied stall or boundary index N+1. RESULT Ls]
            [COMMAND For each test case index i from 1 to T inclusive, after the K-person loop finishes, print Case #i: y z to standard output and flush stdout. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When Min_s.count(maximal) > 1 inside the customer loop, meaning multiple empty stalls have the maximum minimum-side distance]
        [SEQUENTIAL_BLOCK]
            [COMMAND Convert Min_s to a numpy array and compute idxs as the indices where Min_s equals maximal. RESULT idxs]
            [COMMAND For each index j in idxs, append Max_s[j] to max_of_idxs. RESULT max_of_idxs]
            [COMMAND Compute max_max = max(max_of_idxs). RESULT max_max]
            [COMMAND If max_of_idxs.count(max_max) > 1, set idx = idxs[first position in max_of_idxs whose value equals max_max]; otherwise set idx = idxs[the single position in max_of_idxs whose value equals max_max]. RESULT idx]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When Min_s.count(maximal) == 1 inside the customer loop, meaning exactly one empty stall has the maximum minimum-side distance]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set idx to the first index in Min_s whose value equals maximal. RESULT idx]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
