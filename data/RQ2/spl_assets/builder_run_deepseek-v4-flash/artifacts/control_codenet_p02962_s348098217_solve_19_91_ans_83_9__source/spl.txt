[DEFINE_WORKER: "Reads two strings and, based on their length relation and aligned substring matches, prints -1, 0, or Math.max(orig, ans) to the PrintWriter." solve]
    [INPUTS]
        <REF> testNumber </REF>: int "Test-case number; not used by the method."
        <REF> br </REF>: FastScanner "Scanner from which the two strings are read."
        <REF> pw </REF>: PrintWriter "Writer to which the result or -1/0 is printed."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next string from the FastScanner and store it as in1. RESULT in1]
            [COMMAND Read the next string from the FastScanner and store it as in2. RESULT in2]
            [COMMAND Convert in1 to a new char array t. RESULT t]
            [COMMAND Convert in2 to a new char array s. RESULT s]
            [COMMAND Compare t.length and s.length and dispatch to the matching alternative branch. RESULT length_relation]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: t.length > s.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig = 0. RESULT orig]
            [COMMAND Initialize poss = true. RESULT poss]
            [COMMAND Run a half-open loop: for i = 0; i < t.length - s.length; i += s.length, compare in1.substring(i, i + s.length) to in2. If they are not equal, set poss = false; otherwise increment orig. RESULT orig_and_poss]
            [COMMAND If poss is true, print -1 to pw and close pw, then continue execution without returning. RESULT pw]
            [COMMAND Initialize ans = 0. RESULT ans]
            [COMMAND Run a half-open forward loop: for i = 0; i < t.length - s.length; i += s.length, if in1.substring(i, i + s.length) equals in2, increment ans; otherwise break. RESULT ans]
            [COMMAND Run an inclusive backward loop: for i = t.length - s.length; i >= 0; i -= s.length, if in1.substring(i, i + s.length) equals in2, increment ans; otherwise break. This loop includes i = 0 and may count the first matching substring again after the forward loop already counted it. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: s.length == t.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate s.equals(t), where s and t are char arrays, so Java uses identity equality and this is false for the two distinct arrays created by toCharArray. RESULT array_equal]
            [COMMAND If array_equal is true, print -1 to pw; otherwise print 0 to pw, and do not close pw in this branch. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: s.length > t.length (the final else case)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig = 0. RESULT orig]
            [COMMAND Initialize poss = true. RESULT poss]
            [COMMAND Run a half-open loop: for i = 0; i < s.length - t.length; i += s.length, compare in2.substring(i, i + t.length) to in1. If they are not equal, set poss = false; otherwise increment orig. Because the increment is s.length and the bound is s.length - t.length, this loop executes at most once, at i = 0. RESULT orig_and_poss]
            [COMMAND If poss is true, print -1 to pw and close pw, then continue execution without returning. RESULT pw]
            [COMMAND Initialize ans = 0. RESULT ans]
            [COMMAND Run a half-open forward loop: for i = 0; i < s.length - t.length; i += t.length, if in2.substring(i, i + t.length) equals in1, increment ans; otherwise break. RESULT ans]
            [COMMAND Run an inclusive backward loop: for i = s.length - t.length; i >= 0; i -= t.length, if in2.substring(i, i + t.length) equals in1, increment ans; otherwise break. This loop includes i = 0 and may count the first matching substring again after the forward loop already counted it. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
