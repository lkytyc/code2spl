[DEFINE_WORKER: "Computes the minimum possible difference between the largest and smallest areas among three rectangles partitioning an h-by-w rectangle using combinations of cuts, and prints 0 immediately when either dimension is divisible by 3." solve]
    [INPUTS]
        <REF> testNumber </REF>: int "Unused test case number passed to solve."
        <REF> in </REF>: InputReader "Input reader used to read integer h and then integer w."
        <REF> out </REF>: PrintWriter "Output writer receiving the printed answer."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers from the input reader in sequence; assign the first to h and the second to w. RESULT h, w]
            [COMMAND Assuming the early-return guard did not trigger, initialize offset to the constant 2000 and minSum to Integer.MAX_VALUE. RESULT offset, minSum]
            [COMMAND Run the first T-shaped partition search on the original orientation: for int a from h / 3 - offset to less than h / 3 + offset, incrementing a by 1, and skipping any a with a <= 0 or a >= h; for each valid a, for int b from w / 2 - offset to less than w / 2 + offset, incrementing b by 1, and skipping any b with b <= 0 or b >= w; compute s1 = a * w, s2 = b * (h - a), s3 = (h - a) * (w - b), max = Math.max(Math.max(s1, s2), s3), min = Math.min(Math.min(s1, s2), s3), and update minSum = Math.min(minSum, max - min). RESULT minSum]
            [COMMAND Swap h and w using a temporary variable x: x = h, h = w, w = x. RESULT h, w]
            [COMMAND Run the second T-shaped partition search on the swapped orientation: with the current h and w, for int a from h / 3 - offset to less than h / 3 + offset, incrementing a by 1, and skipping any a with a <= 0 or a >= h; for each valid a, for int b from w / 2 - offset to less than w / 2 + offset, incrementing b by 1, and skipping any b with b <= 0 or b >= w; compute s1 = a * w, s2 = b * (h - a), s3 = (h - a) * (w - b), max = Math.max(Math.max(s1, s2), s3), min = Math.min(Math.min(s1, s2), s3), and update minSum = Math.min(minSum, max - min). RESULT minSum]
            [COMMAND Run the first three-strip partition search on the swapped orientation: with the current h and w, for int a from h / 3 - offset to less than h / 3 + offset, incrementing a by 1, and skipping any a with a <= 0 or a >= h; for each valid a, for int b from h / 3 - offset to less than h / 3 + offset, incrementing b by 1, skipping any b with b <= 0 or b >= h, and skipping a + b >= h; compute s1 = a * w, s2 = b * w, s3 = (h - a - b) * w, max = Math.max(Math.max(s1, s2), s3), min = Math.min(Math.min(s1, s2), s3), and update minSum = Math.min(minSum, max - min). RESULT minSum]
            [COMMAND Swap h and w again using the existing temporary variable x: x = h, h = w, w = x, restoring the original orientation. RESULT h, w]
            [COMMAND Run the second three-strip partition search on the original orientation: with the current h and w, for int a from h / 3 - offset to less than h / 3 + offset, incrementing a by 1, and skipping any a with a <= 0 or a >= h; for each valid a, for int b from h / 3 - offset to less than h / 3 + offset, incrementing b by 1, skipping any b with b <= 0 or b >= h, and skipping a + b >= h; compute s1 = a * w, s2 = b * w, s3 = (h - a - b) * w, max = Math.max(Math.max(s1, s2), s3), min = Math.min(Math.min(s1, s2), s3), and update minSum = Math.min(minSum, max - min). RESULT minSum]
            [COMMAND Print the final minSum to out. RESULT out]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After reading h and w, the short-circuit OR expression h % 3 == 0 || w % 3 == 0 is true.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the integer 0 to out. RESULT out]
            [COMMAND Return immediately from solve, skipping all remaining computation. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
