[DEFINE_WORKER: "Reads a number of test cases and, for each integer, replaces every decimal digit 4 with 3 and prints the transformed number and difference if the number contains a 4, otherwise prints the original number." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Always returns 0 on successful execution."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize N to 0 and read the number of test cases N from standard input, where N becomes the input value only if scanf succeeds; initialize j and k to 0. RESULT N]
            [COMMAND For each j from 1 to N inclusive, execute the outer loop body. RESULT j]
            [COMMAND Within the outer loop for j, read the current test-case integer num from standard input, where num becomes the input value only if scanf succeeds, and initialize flag to 0, temp to num, and four to 0. RESULT num]
            [COMMAND Within the outer loop for j, inspect the decimal digits of num by repeatedly setting four=temp%10 and dividing temp by 10 until temp is 0; if any digit equals 4, set flag to 1 and break. RESULT flag]
            [COMMAND Within the outer loop for j, when flag is true, count the digits of num using a do-while loop that starts count at 0 and divides temp by 10 until temp becomes 0. RESULT count]
            [COMMAND Within the outer loop for j, when flag is true, store the decimal digits of num in reverse order into array a by repeatedly writing a[i++]=temp%10 and dividing temp by 10 until temp becomes 0. RESULT a]
            [COMMAND Within the outer loop for j, when flag is true, build array b with the same length as a, setting b[k]=3 when a[k]==4 and b[k]=a[k] otherwise, for k from 0 to count-1 inclusive. RESULT b]
            [COMMAND Within the outer loop for j, when flag is true, reconstruct integer temp1 by starting temp1 at 0 and iterating i from count-1 down to 0 inclusive, performing temp1=temp1*10+b[i]. RESULT temp1]
            [COMMAND Within the outer loop for j, when flag is true, print the case line using j, temp1, and num-temp1 with the format Case #%d: %d %d. RESULT printed_output]
            [COMMAND After the outer loop completes, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: flag is false, meaning num contains no digit 4]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the outer loop for j, print the case line using j and num with the format Case #%d: %d. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
