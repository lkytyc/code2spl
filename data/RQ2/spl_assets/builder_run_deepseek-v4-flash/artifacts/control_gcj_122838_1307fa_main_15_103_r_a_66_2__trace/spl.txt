[DEFINE_WORKER: "Reads a test count and, for each test case, reads arrays a and b, computes nearest-greater boundary arrays, and counts subarrays whose maximum-value positions in a and b differ by at most k, printing one 'Case #m: total' line per test case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Implicit return value 0 after main completes all test cases."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer t from standard input into variable t. RESULT t]
            [COMMAND For each test case m from 1 to t inclusive, run the remaining test-case steps. RESULT test_case_loop]
            [COMMAND Within the test case loop m, read integers n and k from standard input into variables n and k. RESULT n_and_k]
            [COMMAND Within the test case loop m, declare arrays a, b, l_a, l_b, r_a, and r_b, each of length n. RESULT arrays_declared]
            [COMMAND Within the test case loop m, for i from 0 to n-1 inclusive, set l_a[i], l_b[i], r_a[i], and r_b[i] all to i. RESULT boundary_arrays_initialized]
            [COMMAND Within the test case loop m, read n integers from standard input into a[0] through a[n-1]. RESULT array_a]
            [COMMAND Within the test case loop m, read n integers from standard input into b[0] through b[n-1]. RESULT array_b]
            [COMMAND Within the test case loop m, for each i from 0 to n-1 inclusive, compute l_a[i] as the nearest index to the left of i whose a value is strictly greater than a[i], or -1 if no such index exists. RESULT l_a]
            [COMMAND Within the test case loop m, for each i from 0 to n-1 inclusive, compute l_b[i] as the nearest index to the left of i whose b value is strictly greater than b[i], or -1 if no such index exists. RESULT l_b]
            [COMMAND Within the test case loop m, for each i from 0 to n-1 inclusive, compute r_a[i] as the nearest index to the right of i whose a value is strictly greater than a[i], or n if no such index exists. RESULT r_a]
            [COMMAND Within the test case loop m, for each i from 0 to n-1 inclusive, compute r_b[i] as the nearest index to the right of i whose b value is strictly greater than b[i], or n if no such index exists. RESULT r_b]
            [COMMAND Within the test case loop m, set long long integer tot to 0. RESULT tot]
            [COMMAND Within the test case loop m, for i from 0 to n-1 inclusive and for j from i to r_a[i]-1 inclusive, if l_b[j] < i and a[i]-b[j] is between -k and k inclusive, add (i - max(l_a[i], l_b[j])) * (min(r_a[i], r_b[j]) - j) to tot. RESULT tot]
            [COMMAND Within the test case loop m, for i from 0 to n-1 inclusive and for j from i+1 to r_b[i]-1 inclusive, if l_a[j] < i and a[j]-b[i] is between -k and k inclusive, add (i - max(l_b[i], l_a[j])) * (min(r_b[i], r_a[j]) - j) to tot. RESULT tot]
            [COMMAND Within the test case loop m, print "Case #%d: %lld\n" using m and tot. RESULT stdout]
            [COMMAND After the test case loop m finishes, return 0 implicitly from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
