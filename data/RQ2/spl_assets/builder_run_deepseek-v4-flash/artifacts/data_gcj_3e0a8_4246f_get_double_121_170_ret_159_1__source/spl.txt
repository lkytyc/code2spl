[DEFINE_WORKER: "Reads characters from standard input, skips non-digit characters except when a '-' is immediately followed by a digit, and parses a signed double with an optional fractional part, returning EOF if no digit is found before end-of-file." get_double]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: double "The parsed signed double, or EOF converted to double when end-of-file is reached before any digit is found."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize double ret to 0, declare int sgn without initialization, and read the first character from standard input into char c using getchar(), with implicit int-to-char conversion. RESULT ret, sgn, c]
            [COMMAND Enter the first scanning loop. Skip over non-digit characters until a digit is found, determining the sign from a '-' immediately followed by a digit. If c equals EOF before any digit is found, return EOF. RESULT c, sgn]
            [COMMAND Enter the second infinite loop. Accumulate the integer part by assigning ret = ret * 10 + (c - '0'), then read the next character into c. RESULT ret, c]
            [COMMAND Within the second infinite loop, after reading the next character, if it is '.', parse the fractional part and return sgn * ret. RESULT return_value]
            [COMMAND Within the second infinite loop, after reading the next character, if it is neither '.' nor a digit, return sgn * ret. RESULT return_value]
            [COMMAND Within the second infinite loop, after reading the next character, if it is a digit, continue the second loop from the integer accumulation assignment. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: At the top of the first scanning loop, c equals EOF before a digit is found]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return EOF immediately. RESULT EOF]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: At the top of the first scanning loop, c is a digit from '0' to '9']
        [SEQUENTIAL_BLOCK]
            [COMMAND Set sgn to 1. RESULT sgn]
            [COMMAND Break out of the first scanning loop, leaving c as the first digit. RESULT c]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: At the top of the first scanning loop, c == '-', and after reading the next character into c, the new c is not a digit]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next character into c. RESULT c]
            [COMMAND Continue the first scanning loop without setting sgn; the '-' is discarded as a non-digit and the newly read c is checked again at the top of the loop. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: At the top of the first scanning loop, c == '-', and after reading the next character into c, the new c is a digit]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next character into c. RESULT c]
            [COMMAND Set sgn to -1. RESULT sgn]
            [COMMAND Break out of the first scanning loop, leaving c as the digit after the minus sign. RESULT c]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: At the top of the first scanning loop, c is not EOF, not a digit, and not '-']
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next character into c and repeat the first scanning loop. RESULT c]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the second infinite loop, after reading the next character, c == '.']
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize double pos to 0.1 and read the next character into c. RESULT pos, c]
            [COMMAND While c is a digit from '0' to '9', add pos * (c - '0') to ret, divide pos by 10, and read the next character into c. RESULT ret, pos, c]
            [COMMAND Return sgn * ret. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the second infinite loop, after reading the next character, c is neither '.' nor a digit]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return sgn * ret. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the second infinite loop, after reading the next character, c is a digit]
        [SEQUENTIAL_BLOCK]
            [COMMAND Continue the second loop so the new digit is processed as part of the integer part. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
