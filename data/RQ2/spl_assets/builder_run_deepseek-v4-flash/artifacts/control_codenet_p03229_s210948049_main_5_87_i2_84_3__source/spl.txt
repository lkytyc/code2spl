[DEFINE_WORKER: "Reads N integers, sorts them, performs two alternating two-pointer selection passes that accumulate two scores, and prints the larger score." main]
    [INPUTS]
        <REF> N </REF>: integer "Number of array elements read from input via Z(); determines array length and index boundaries."
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_result </REF>: integer "Maximum of output1 and output2, printed to standard output."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer N from standard input using Z() and assign it to N. RESULT N]
            [COMMAND Initialize A as an empty list. For each index from 0 to N-1, read the next integer using Z(), append it to A, and after the loop sort A in ascending order and assign the sorted list back to A. RESULT A]
            [COMMAND Initialize the first pass: set used to a list of N zeros; set output1 = 0 and output2 = 0; set i1 = 0 and i2 = 0; set j1 = N-1 and j2 = N-2; set f1 = True and f2 = True. Then perform the tuple assignment used[i1], used[i2], used[j1], used[j2] = 1, 1, 1, 1, so used[0] is set twice, used[N-1] is set once, and used[N-2] is set once, with overlaps when N is small. RESULT first_pass_state]
            [COMMAND Inside the first outer while loop, while f1 or f2 is true, if f1 is true add A[j1] - A[i1] to output1, and if f2 is true add A[j2] - A[i2] to output1. RESULT output1]
            [COMMAND Inside the first outer while loop, advance the two left pointers: while used[i1] is true, if i1 == N-1 set f1 = False and break, otherwise increment i1; after that inner loop, unconditionally set used[i1] = 1. Perform the same iteration logic for i2 using f2 and used[i2], with boundary i2 == N-1 setting f2 = False. RESULT left_pointers_first_pass]
            [COMMAND Inside the first outer while loop, after advancing the left pointers, if f1 is true add A[j1] - A[i1] to output1, and if f2 is true add A[j2] - A[i2] to output1. RESULT output1]
            [COMMAND Inside the first outer while loop, retreat the two right pointers: while used[j1] is true, if j1 == 0 set f1 = False and break, otherwise decrement j1; after that inner loop, unconditionally set used[j1] = 1. Perform the same iteration logic for j2 using f2 and used[j2], with boundary j2 == 0 setting f2 = False. RESULT right_pointers_first_pass]
            [COMMAND Continue the first outer while loop until both f1 and f2 are false. RESULT first_loop_complete]
            [COMMAND Reset for the second pass: set i1 = 0 and i2 = 1; set j1 = N-1 and j2 = N-1; set f1 = True and f2 = True; set used = [0] * N; keep output2 at its current accumulated value. Then perform the tuple assignment used[i1], used[i2], used[j1], used[j2] = 1, 1, 1, 1, so used[0] is set once, used[1] is set once, and used[N-1] is set twice, with overlaps when N is small. RESULT second_pass_state]
            [COMMAND Inside the second outer while loop, while f1 or f2 is true, if f1 is true add A[j1] - A[i1] to output2, and if f2 is true add A[j2] - A[i2] to output2. RESULT output2]
            [COMMAND Inside the second outer while loop, retreat the two right pointers first: while used[j1] is true, if j1 == 0 set f1 = False and break, otherwise decrement j1; after that inner loop, unconditionally set used[j1] = 1. Perform the same iteration logic for j2 using f2 and used[j2], with boundary j2 == 0 setting f2 = False. RESULT right_pointers_second_pass]
            [COMMAND Inside the second outer while loop, after retreating the right pointers, if f1 is true add A[j1] - A[i1] to output2, and if f2 is true add A[j2] - A[i2] to output2. RESULT output2]
            [COMMAND Inside the second outer while loop, advance the two left pointers: while used[i1] is true, if i1 == N-1 set f1 = False and break, otherwise increment i1; after that inner loop, unconditionally set used[i1] = 1. Perform the same iteration logic for i2 using f2 and used[i2], with boundary i2 == N-1 setting f2 = False. RESULT left_pointers_second_pass]
            [COMMAND Continue the second outer while loop until both f1 and f2 are false. RESULT second_loop_complete]
            [COMMAND Compute max(output1, output2) and print the result to standard output. RESULT printed_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
