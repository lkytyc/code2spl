[DEFINE_WORKER: "Checks the first and last ball counts, builds a C by C dot grid, simulates left and right falling-ball paths with backslash and slash characters while mutating B and shadowed state, then returns the trimmed grid with a leading all-dot row, or ('IMPOSSIBLE', None) if an edge column has zero balls." solve]
    [INPUTS]
        <REF> B </REF>: list[int] "List of non-negative integer ball counts per column; the list is mutated during the simulation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple[int, list[list[str]]] or tuple[str, None] "On success, returns (len(ans), ans) where ans is a list of dot/slash/backslash rows with an all-dot row prepended. If an edge column has zero balls, returns ('IMPOSSIBLE', None)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate the edge-column guard B[0] == 0 or B[-1] == 0; continue only when both edge values are nonzero. RESULT edge_guard]
            [COMMAND Initialize grid as an empty list; this first assignment is dead because grid is overwritten before being read. RESULT grid]
            [COMMAND Set C = len(B). RESULT C]
            [COMMAND Reinitialize grid as an empty list, overwriting the previous dead grid value. RESULT grid]
            [COMMAND Build the C by C dot grid: for each j from 0 to C-1 create a new row, append '.' for each i from 0 to C-1 to that row, and append the completed row to grid. RESULT grid]
            [COMMAND Initialize shadowed as [False] * len(C), then set shadowed[0] = True. RESULT shadowed]
            [COMMAND Mutate the input list B by decrementing B[0] by 1 and B[-1] by 1. RESULT B]
            [COMMAND Set i = 0 as an initializer; the following for loop overwrites i. RESULT i]
            [COMMAND For each i from 0 to C-1, if B[i] == 0 and shadowed[i] is True, continue to the next i without further action in that iteration. RESULT loop_skip]
            [COMMAND Within the same for loop, if B[i] > 0 and shadowed[i] is False, execute the left pass: set unshadowed_balls = 0, j = i - 1, and l = 1; while j >= 1 and not shadowed[j], place '\' at grid[l][j], set shadowed[j] = True, increment unshadowed_balls, increment l, and decrement j. RESULT left_pass_state]
            [COMMAND After the left pass, if B[i] <= unshadowed_balls, compute excess_balls = unshadowed_balls - B[i] + 1, set j = i, and while excess_balls > 1 decrement j, decrement excess_balls, and set shadowed[j] = False. Otherwise, in the else case, subtract unshadowed_balls + 1 from B[i] and set shadowed[i] = True. RESULT left_balance_state]
            [COMMAND Still in the same loop iteration, evaluate B[i] > 0 and shadowed[i] using the updated values; if true, execute the right pass: set r = B[i], j = i + 1, and l = 1; while j < C - 1 and shadowed[j], place '/' at grid[l][j], increment l, and increment j; then while j < C - 1 and r > 0, place '/' at grid[l][j], set shadowed[j] = True, decrement r, increment l, and increment j. RESULT grid_B_shadowed_state]
            [COMMAND After the for loop over i finishes, initialize ans as an empty list. RESULT ans]
            [COMMAND For index from 1 to C-1, set row = grid[index] and include_row = False; scan each cell in row, and if any cell is '\' or '/', set include_row = True and break out of the cell scan. If include_row is True, append row to ans; otherwise break out of the index loop, collecting only rows with a slash or backslash until the first empty row. RESULT ans]
            [COMMAND Prepend a new all-dot row to ans: build ['.' for i in range(C)] and set ans = [that row] + ans. RESULT ans]
            [COMMAND Return the tuple (len(ans), ans). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: B[0] == 0 or B[-1] == 0, using short-circuit OR.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the tuple ('IMPOSSIBLE', None) and stop all further work. RESULT impossible_result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: B is empty before the edge-column guard accesses B[0] or B[-1].]
        [LOG "IndexError: list index out of range"]
        [THROW IndexError "Accessing B[0] or B[-1] on an empty input list raises IndexError."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: During shadowed initialization, len(C) is evaluated after C was assigned len(B); because C is an integer, len(C) raises TypeError.]
        [LOG "TypeError: object of type 'int' has no len()"]
        [THROW TypeError "The expression len(C) is invalid because C is the integer result of len(B), preventing shadowed from being created."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
