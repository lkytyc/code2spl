[DEFINE_WORKER: "Computes edge latencies by converting signed input times into absolute event times and using sorted receipt order to assign time differences across the given edges." solve_case]
    [INPUTS]
        <REF> in_times </REF>: list of int "Signed input times for each component; positive values are used directly as absolute times and negative values represent relative ordering to be resolved by the receipt loop."
        <REF> edges </REF>: list of tuple of int "Original 1-based directed edge pairs; each tuple is (a, b) and the result contains one latency per edge in this order."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Space-separated string of latency values, one per edge in original edge order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize latencies as a list of 1000000 values with length equal to len(edges). RESULT latencies]
            [COMMAND Replace edges with a new list containing (a-1, b-1) for each original edge tuple (a,b). RESULT edges]
            [COMMAND Initialize abs_times as a list of 1000000 values with length equal to len(in_times). RESULT abs_times]
            [COMMAND For every index, value pair from enumerate(in_times), if value is greater than 0, assign abs_times[index] = value. RESULT abs_times]
            [COMMAND Initialize pos_times as an empty list. RESULT pos_times]
            [COMMAND For every index, value pair from enumerate(in_times), if value is less than 0, append (abs(value), index) to pos_times. RESULT pos_times]
            [COMMAND Sort pos_times in ascending tuple order. RESULT pos_times]
            [COMMAND Set current_time to 0. RESULT current_time]
            [COMMAND Set received to 1. RESULT received]
            [COMMAND While pos_times is non-empty, repeat the following behavior: if pos_times[0][0] equals received, increment current_time by 1, set new_received to 0, consume all leading pos_times tuples whose first element equals received, and for each consumed tuple (a,b) set abs_times[b] to current_time, remove that tuple from the front of pos_times, then add new_received to received; if pos_times[0][0] does not equal received, collect every abs_times value greater than current_time into next_times, set current_time to the minimum value of next_times, set received_count to the number of abs_times values less than or equal to current_time, and leave pos_times and received unchanged. RESULT pos_times]
            [COMMAND Replace abs_times with a new list created by prepending 0 to the current abs_times. RESULT abs_times]
            [COMMAND Build sorted_receipt as [(value, index) for index, value in enumerate(abs_times)]. RESULT sorted_receipt]
            [COMMAND Sort sorted_receipt in ascending order. RESULT sorted_receipt]
            [COMMAND For each element (time, comp) in sorted_receipt after the first element, scan edges in order with their indices; if the current edge's A endpoint equals comp, compare abs_times[A] and abs_times[B], and if abs_times[A] is strictly greater than abs_times[B], set latencies[index] to abs_times[A] - abs_times[B] and break out of the inner edge scan; if the current edge's B endpoint equals comp, compare abs_times[B] and abs_times[A], and if abs_times[B] is strictly greater than abs_times[A], set latencies[index] to abs_times[B] - abs_times[A] and break out of the inner edge scan. RESULT latencies]
            [COMMAND Return a string formed by joining the string form of every latency value in latencies with a single space. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
