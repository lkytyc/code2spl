    [END_INPUTS]

    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer T from standard input. RESULT T]
            [COMMAND For each test case i from 1 to T inclusive, read the next line, split it into integers, and assign N as the first integer and K as the second integer. RESULT i_N_K]
            [COMMAND Inside the test-case loop for i, initialize stall_occupied as [True], append False N times, then append True to form the N+2 stall list with occupied sentinels. RESULT stall_occupied]
            [COMMAND Inside the test-case loop for i, initialize Ls, Rs, Min_s, and Max_s as empty lists. RESULT distance_arrays]
            [COMMAND Inside the test-case loop for i, for each stall index j from 0 to N+1 inclusive, initialize Ls[j] and Rs[j]: if stall_occupied[j] is True set both to -1, otherwise set Ls[j]=j-1 and Rs[j]=N-j; then append Min_s[j]=min(Ls[j],Rs[j]) and Max_s[j]=max(Ls[j],Rs[j]). RESULT distance_arrays]
            [COMMAND Inside the test-case loop for i, for each person k from 0 to K-1 inclusive, compute maximal = max(Min_s). RESULT maximal]
            [COMMAND Inside the test-case loop for i and the placement loop for k, select a stall index idx using the tie-breaking branches described in alternative_flows. RESULT idx]
            [COMMAND Inside the test-case loop for i and the placement loop for k, set y = max(Ls[idx], Rs[idx]) and z = min(Ls[idx], Rs[idx]). RESULT y_z]
            [COMMAND Inside the test-case loop for i and the placement loop for k, set stall_occupied[idx] to True and set Ls[idx], Rs[idx], Min_s[idx], and Max_s[idx] to -1. RESULT stall_occupied_distance_arrays]
            [COMMAND Inside the test-case loop for i and the placement loop for k, set leftIdx = idx - 1; while leftIdx > 0 and stall_occupied[leftIdx] is False, update Rs[leftIdx] = idx - leftIdx - 1, Min_s[leftIdx] = min(Rs[leftIdx], Ls[leftIdx]), Max_s[leftIdx] = max(Rs[leftIdx], Ls[leftIdx]), and decrement leftIdx. RESULT distance_arrays]
            [COMMAND Inside the test-case loop for i and the placement loop for k, set rightIdx = idx + 1; while rightIdx < N + 2 and stall_occupied[rightIdx] is False, update Ls[rightIdx] = rightIdx - idx - 1, Min_s[rightIdx] = min(Rs[rightIdx], Ls[rightIdx]), Max_s[rightIdx] = max(Rs[rightIdx], Ls[rightIdx]), and increment rightIdx. RESULT distance_arrays]
            [COMMAND Inside the test-case loop for i, after placing all K people, print 'Case #i: y z' and flush standard output. RESULT printed_line]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

            [COMMAND Within the test-case loop for i and the placement loop for k, if max_of_idxs contains max_max more than once, create values1 = np.array(max_of_idxs), compute idxs1 = np.where(values1 == max_max)[0], and set idx = idxs[idxs1[0]]; otherwise set idx = idxs[max_of_idxs.index(max_max)]. RESULT idx]
        [END_SEQUENTIAL_BLOCK]
