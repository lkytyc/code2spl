[DEFINE_WORKER: "Reads the number of test cases and, for each, simulates the Bathroom Stalls process using a negative-priority max-heap and a multiplicity map to determine the last person's maximum and minimum distances when K people occupy N stalls." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Open a Scanner named cin bound to System.in. RESULT cin]
            [COMMAND Read the next integer from cin into int T, representing the number of test cases. RESULT T]
            [COMMAND Initialize the outer for loop counter C to 1; loop while C <= T; increment C by 1 after each iteration. RESULT C]
            [COMMAND Within the outer for loop for current C, read the next long from cin into long N, the number of stalls. RESULT N]
            [COMMAND Within the outer for loop for current C, read the next long from cin into long K, the number of people. RESULT K]
            [COMMAND Within the outer for loop for current C, create an empty PriorityQueue<Long> queue. RESULT queue]
            [COMMAND Within the outer for loop for current C, create an empty Map<Long,Long> count. RESULT count]
            [COMMAND Within the outer for loop for current C, add -N to queue so that polling returns the largest positive interval length. RESULT queue]
            [COMMAND Within the outer for loop for current C, put count[N]=1L to record one interval of length N. RESULT count]
            [COMMAND Within the outer for loop for current C, initialize long ans=0 as a fallback. RESULT ans]
            [COMMAND Within the outer for loop for current C, enter an infinite while loop that processes interval lengths in decreasing order; it will break only when the remaining people fit into the currently polled interval length. RESULT while_loop]
            [COMMAND Within the while loop, poll the head of queue, negate it to get the largest interval length cur, and read rest=count.get(cur), the number of intervals of length cur. RESULT cur]
            [COMMAND Within the while loop, if K <= 1, set ans=cur and break out of the while loop. RESULT ans]
            [COMMAND Within the while loop, subtract rest from K. RESULT K]
            [COMMAND Within the while loop, if K <= 0 after the subtraction, set ans=cur and break out of the while loop. RESULT ans]
            [COMMAND Within the while loop, compute large=cur/2 and small=(cur-1)/2 using integer division; these are the larger and smaller intervals created by splitting one interval of length cur. RESULT large]
            [COMMAND Within the while loop, if count already contains large, add rest to its stored multiplicity; otherwise add -large to queue and set count[large]=rest. RESULT count]
            [COMMAND Within the while loop, if count already contains small, add rest to its stored multiplicity; otherwise add -small to queue and set count[small]=rest. RESULT count]
            [COMMAND After the while loop terminates for the current test case C, print to System.out the line "Case #" + C + ": " + (ans/2) + " " + ((ans-1)/2). RESULT output]
            [COMMAND After the outer for loop has processed all T test cases, close cin. RESULT cin]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: cin.nextInt() or cin.nextLong() is called when no more input tokens are available]
        [LOG "NoSuchElementException: No line found"]
        [THROW NoSuchElementException "Thrown by Scanner when input is exhausted while reading T, N, or K."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The next input token is not a valid int for T or not a valid long for N/K]
        [LOG "InputMismatchException: Input token is not a valid integer type"]
        [THROW InputMismatchException "Thrown by Scanner when a token cannot be parsed as the requested primitive type."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
