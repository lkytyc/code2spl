[DEFINE_WORKER: "Computes the maximum number of tasks completable within time t by separating zero-start tasks from positive-start tasks, sorting positive-start tasks by a descending ratio, performing dynamic programming over up to 29 positive-start tasks, and combining those results with cumulative zero-start task costs." solve]
    [INPUTS]
        <REF> n </REF>: integer "Number of tasks in ab_list."
        <REF> t </REF>: integer "Time limit for completing tasks."
        <REF> ab_list </REF>: list of pairs "List of [a, b] pairs where a is the task start-cost multiplier and b is the task additional cost."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: integer "Maximum number of tasks that can be completed within time t."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a0b_list as the list of b values from each pair in ab_list whose first element a equals 0; initialize apb_list as an empty list. RESULT a0b_list, apb_list]
            [COMMAND For each pair ab in ab_list, if ab[0] > 0, append the whole pair ab to apb_list. RESULT apb_list]
            [COMMAND Sort a0b_list in ascending numeric order. RESULT a0b_list]
            [COMMAND Sort apb_list in descending order of a / (b + 1) by using the key -a/(b+1), where a = ab[0] and b = ab[1]. RESULT apb_list]
            [COMMAND Set m to len(apb_list) and initialize dp as a list of m + 1 rows, each containing 30 entries initialized to t + 1. RESULT m, dp]
            [COMMAND For each row i from 0 through m inclusive, set dp[i][0] = 0. RESULT dp]
            [COMMAND For each i from 0 through m - 1, and for each k from 1 through 29 inclusive, compute ti = dp[i][k - 1] + 1, unpack a, b = apb_list[i], and set dp[i + 1][k] = min(dp[i][k], ti + a * ti + b). RESULT dp]
            [COMMAND Initialize a0b_list_cum_sum as [0] and s as 0. RESULT a0b_list_cum_sum, s]
            [COMMAND For each b in a0b_list, increment s by b + 1 and append s to a0b_list_cum_sum. RESULT a0b_list_cum_sum]
            [COMMAND Initialize res to 0. RESULT res]
            [COMMAND For each k from 0 through len(a0b_list_cum_sum) - 1 inclusive, and for each p from 0 through 29 inclusive, if dp[m][p] + a0b_list_cum_sum[k] <= t, update res = max(res, k + p). RESULT res]
            [COMMAND Return the final value of res. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
