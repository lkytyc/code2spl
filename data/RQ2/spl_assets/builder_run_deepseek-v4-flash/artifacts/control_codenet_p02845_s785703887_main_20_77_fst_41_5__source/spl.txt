[DEFINE_WORKER: "Reads N and N integers from standard input, maintains three sorted counters for nonzero values, accumulates a modular multiplier count for repeated values, and prints the final count after applying an unused-slot multiplier." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not used by this method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Enter a try-with-resources block creating a Scanner sc reading from System.in. RESULT sc]
            [COMMAND Read integer N from sc as the number of loop iterations. RESULT N]
            [COMMAND Set MOD to 1000000007, initialize fst, snd, thd to -1, and answer to 1. RESULT MOD, fst, snd, thd, answer]
            [COMMAND For i from 0 to N-1, read the next integer val from sc. RESULT i, val]
            [COMMAND Process the current val using the zero and nonzero branches described in alternative_flows. RESULT fst, snd, thd, answer]
            [COMMAND After the loop, apply the final unused-slot multiplier described in alternative_flows. RESULT answer]
            [COMMAND Print answer to System.out. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Inside the for loop, val == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If fst == -1, set fst to 1. RESULT fst]
            [COMMAND Else if fst != -1 and snd == -1, set snd to 1. RESULT snd]
            [COMMAND Else if fst != -1 and snd != -1 and thd == -1, set thd to 1. RESULT thd]
            [COMMAND If none of the preceding conditions apply, leave fst, snd, and thd unchanged. RESULT fst, snd, thd]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the for loop, val != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If fst, snd, and thd all equal val, increment fst, multiply answer by 3, and reduce answer modulo MOD. RESULT fst, answer]
            [COMMAND Else if fst and snd both equal val, increment fst, multiply answer by 2, and reduce answer modulo MOD. RESULT fst, answer]
            [COMMAND Else if snd and thd both equal val, increment snd, multiply answer by 2, and reduce answer modulo MOD. RESULT snd, answer]
            [COMMAND Else if thd and fst both equal val, increment fst, multiply answer by 2, and reduce answer modulo MOD. RESULT fst, answer]
            [COMMAND Else if fst equals val, increment fst and leave answer unchanged. RESULT fst, answer]
            [COMMAND Else if snd equals val, increment snd and leave answer unchanged. RESULT snd, answer]
            [COMMAND Else if thd equals val, increment thd and leave answer unchanged. RESULT thd, answer]
            [COMMAND If none of the equality cases match, leave fst, snd, thd, and answer unchanged. RESULT fst, snd, thd, answer]
            [COMMAND After equality handling, snapshot old_fst, old_snd, old_thd; set fst to the maximum of the old values, thd to the minimum, and snd to (old_fst + old_snd + old_thd) - fst - thd. RESULT fst, snd, thd]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the loop, snd < 0 and thd < 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Multiply answer by 3 and reduce answer modulo MOD. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the loop, not (snd < 0 and thd < 0)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Multiply answer by 6 and reduce answer modulo MOD. RESULT answer]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
