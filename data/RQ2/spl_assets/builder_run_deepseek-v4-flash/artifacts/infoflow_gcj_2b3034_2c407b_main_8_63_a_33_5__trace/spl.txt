[DEFINE_WORKER: "Reads multiple test cases, each containing a set of patterns with wildcards, merges them into a single consistent string by aligning them in a 2D grid, and outputs the merged string or an asterisk if a conflict is detected." main]
    [INPUTS]
        <REF> stdin </REF>: input stream "The input stream containing the number of test cases, and for each test case, the number of patterns and the patterns themselves."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: output stream "The output stream where the result for each test case is printed in the format 'Case #X: result'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integer variables t, n, i, j, k and initialize t and n to 0. Declare character array s of size 10001 and character array str of size 101. RESULT variables]
            [COMMAND Read an integer from standard input into variable t using scanf. RESULT t]
            [COMMAND Start an outer for loop with i initialized to 1, continuing while i is less than or equal to t, incrementing i by 1 after each iteration. RESULT outer_loop]
            [COMMAND Within the outer for loop, read an integer from standard input into variable n using scanf. RESULT n]
            [COMMAND Within the outer for loop, declare a 2D character array a of size n by 10001 and a character variable dd. RESULT array_a]
            [COMMAND Within the outer for loop, initialize all elements of the 2D array a to the character '#' using nested for loops: for j from 0 to n-1, for k from 0 to 10000, set a[j][k] to '#'. RESULT array_a_initialized]
            [COMMAND Within the outer for loop, declare an integer variable flag and initialize it to 0. RESULT flag]
            [COMMAND Within the outer for loop, start a for loop with j initialized to 0, continuing while j is less than n, incrementing j by 1 after each iteration. RESULT input_loop]
            [COMMAND Within the input loop, read a single character into variable dd using scanf to consume the newline character. RESULT dd]
            [COMMAND Within the input loop, read a string into character array str using scanf. RESULT str]
            [COMMAND Within the input loop, check if the first character of str is '*'. If true, declare an integer variable x and initialize it to 10000, then start a for loop with k initialized to len(str), continuing while k is greater than or equal to 0, decrementing k by 1 after each iteration. Inside this loop, assign the character str[k] to a[j][x] and then decrement x by 1. RESULT right_aligned_pattern]
            [COMMAND Within the input loop, if the first character of str is not '*', check if the last character of str (at index len(str)-1) is '*'. If true, start a for loop with k initialized to 0, continuing while str[k] is not the null character '\0', incrementing k by 1 after each iteration. Inside this loop, assign the character str[k] to a[j][k]. RESULT left_aligned_pattern]
            [COMMAND After the input loop completes, declare an integer variable x and initialize it to 0. RESULT x]
            [COMMAND Within the outer for loop, start a for loop with j initialized to 0, continuing while j is less than 10000, incrementing j by 1 after each iteration. RESULT merge_loop]
            [COMMAND Within the merge loop, declare an integer variable f and initialize it to 0. RESULT f]
            [COMMAND Within the merge loop, start a nested for loop with k initialized to 0, continuing while k is less than n, incrementing k by 1 after each iteration. RESULT column_check_loop]
            [COMMAND Within the column check loop, check if f is equal to 1 and a[k][j] is not equal to s[x-1] and a[k][j] is not equal to '*' and a[k][j] is not equal to '#'. If true, set flag to 1 and break out of the column check loop. RESULT mismatch_detected]
            [COMMAND Within the column check loop, if the previous condition is false, check if a[k][j] is not equal to '*' and a[k][j] is not equal to '#'. If true, check if f is equal to 0. If true, assign a[k][j] to s[x], increment x by 1, and set f to 1. RESULT character_added]
            [COMMAND After the column check loop completes, check if flag is equal to 1. If true, break out of the merge loop. RESULT merge_loop_break]
            [COMMAND After the merge loop completes, assign the null character '\0' to s[x]. RESULT s_terminated]
            [COMMAND Check if flag is equal to 1. If true, print the string 'Case #%d: *\n' with the current value of i using printf. RESULT error_output]
            [COMMAND If flag is not equal to 1, print the string 'Case #%d: %s\n' with the current value of i and the character array s using printf. RESULT success_output]
            [COMMAND After the outer for loop completes, return 0 from the main function. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
