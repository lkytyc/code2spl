[DEFINE_WORKER: "Reads a graph from standard input, builds vertex adjacency lists and edge metadata, then attempts up to three route augmentations from vertex 0 to V-1, printing the accumulated flow on an early terminal route or an error after three successful augmentations." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Returns 0 both on early termination after printing res and after printing the error string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Attempt to read two integers from standard input into V and E using scanf, without checking the scanf return value. RESULT V,E]
            [COMMAND For i starting at 0 and continuing while i < V, incrementing i by 1 after each iteration, set v[i].num to 0. RESULT v[*].num]
            [COMMAND Set cmax to 0. RESULT cmax]
            [COMMAND For i starting at 0 and continuing while i < E, incrementing i by 1 after each iteration, attempt to read three integers from standard input into a, b, and c using scanf, without checking the scanf return value. RESULT a,b,c]
            [COMMAND Within the edge-processing loop over i, set e[i].n[0] to a and e[i].n[1] to b. RESULT e[i].n]
            [COMMAND Within the edge-processing loop over i, append edge index i to v[a].e at position v[a].num and then increment v[a].num. RESULT v[a].e,v[a].num]
            [COMMAND Within the edge-processing loop over i, append edge index i to v[b].e at position v[b].num and then increment v[b].num. RESULT v[b].e,v[b].num]
            [COMMAND Within the edge-processing loop over i, set e[i].fmax to c and e[i].flow to 0. RESULT e[i].fmax,e[i].flow]
            [COMMAND Within the edge-processing loop over i, if b equals V-1, move v[a].e[0] to v[a].e[v[a].num-1], set v[a].e[0] to i, and add c to cmax. RESULT v[a].e,cmax]
            [COMMAND After the edge-processing loop, set res to 0. RESULT res]
            [COMMAND For i starting at 0 and continuing while i < 100, incrementing i by 1 after each iteration, set et[i] to 0. RESULT et[*]]
            [COMMAND For i starting at 0 and continuing while i < 3, incrementing i by 1 after each iteration, declare int x and assign it the return value of route(0, V-1, cmax). RESULT x]
            [COMMAND Within the routing loop over i, if x is greater than 0, add x to res. RESULT res]
            [COMMAND After the routing loop completes normally, print the string "er" followed by a newline. RESULT stdout]
            [COMMAND Return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: In the routing loop over i from 0 while i < 3, route(0, V-1, cmax) returns a value x less than or equal to 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the current value of res followed by a newline. RESULT stdout]
            [COMMAND Return 0 from main immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
