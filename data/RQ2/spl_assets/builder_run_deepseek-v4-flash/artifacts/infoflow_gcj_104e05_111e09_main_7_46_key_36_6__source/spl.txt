[DEFINE_WORKER: "Reads a number of test cases from standard input, stores each test case's words in a sorted TreeSet, and attempts to find the longest common substring between the first word and every word using two iterators, but no computed result is ever output and the unreset inner iterator limits the comparison to the first word." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to main; unused by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a BufferedReader named br that reads from System.in. RESULT br]
            [COMMAND Read the first line from br and parse it as an int into tc. RESULT tc]
            [COMMAND For each test case index i from 0 while i < tc, execute the body of the test case loop. RESULT i]
            [COMMAND Within test case i, read one line from br and parse it as an int into noOfWords. RESULT noOfWords]
            [COMMAND Within test case i, create an empty TreeSet<String> named set, using the natural ordering of String. RESULT set]
            [COMMAND Within test case i, for each word index j from 0 while j < noOfWords, read one line from br as word and add word to set; duplicate words are ignored by TreeSet. RESULT set]
            [COMMAND Within test case i, create two independent iterators over set: wrd1 and wrd2. RESULT wrd1, wrd2]
            [COMMAND Within test case i, set c1 to 0. RESULT c1]
            [COMMAND While wrd1.hasNext(), within the outer iterator loop, increment c1, set c2 to 0, set wordOne to wrd1.next(), and set result to 0. RESULT wordOne, result]
            [COMMAND While wrd2.hasNext(), within the outer iterator loop, increment c2; if c1 equals c2, do not call wrd2.next(), leaving wrd2 in place; if c1 is not equal to c2, set wordTwo to wrd2.next() and proceed with the substring search. RESULT wordTwo]
            [COMMAND Within the c1 != c2 branch for the current wordTwo, initialize max to 0 and key to an empty string. RESULT max, key]
            [COMMAND Within the c1 != c2 branch, for each l from 0 while l < wordTwo.length() - 1 and each k from l + 2 while k <= wordTwo.length(), form the end-exclusive substring wordTwo.substring(l, k); if wordOne contains that substring and, only when true, k - l is greater than max, update max to k - l and key to that substring; otherwise break out of the k loop for the current l. RESULT key]
            [COMMAND Do not reset wrd2 between outer while iterations; therefore after the first wrd1 iteration exhausts wrd2, later wrd1 iterations execute the inner while zero times. Do not use or print result, max, or key. RESULT none]
            [COMMAND After the loops finish for all test cases, return from main without printing any output. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: BufferedReader.readLine fails while reading input from System.in.]
        [LOG "IOException is propagated because the method declares throws IOException."]
        [THROW IOException "Indicates an I/O error occurred during input reading."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: Integer.parseInt is given a line that is null, empty, not a valid decimal integer, or outside the int range.]
        [LOG "NumberFormatException is propagated from Integer.parseInt."]
        [THROW NumberFormatException "Indicates the input line cannot be parsed as an int."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
