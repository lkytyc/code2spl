[DEFINE_WORKER: "Performs a depth-limited traversal from s to collect nodes at distance exactly 3, then assigns values from two queues into ans for s and those distance-3 nodes based on residue counts of existing ans values, terminating when the required assignment is impossible." length3]
    [INPUTS]
        <REF> s </REF>: int "Source node index where traversal and assignment start."
        <REF> n </REF>: int "Number of nodes; sizes the visited and distance arrays."
        <REF> g </REF>: list[list[int]] "Adjacency list of the graph."
        <REF> ans </REF>: list[int] "Array of assigned values; zero entries are unassigned and may be filled."
        <REF> que_1 </REF>: list[int] "First source of values available for assignment; modified by pop operations."
        <REF> que_2 </REF>: list[int] "Second source of values available for assignment; modified by pop operations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> ans </REF>: list[int] "Updated assignment array after filling source and distance-3 nodes."
        <REF> que_1 </REF>: list[int] "First queue after pop operations."
        <REF> que_2 </REF>: list[int] "Second queue after pop operations."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize went = [False]*n, dis = [0]*n, see = [], and que = deque([]). RESULT went, dis, see, que]
            [COMMAND Append s to que and set went[s] = True. RESULT que, went]
            [COMMAND While que is nonempty, pop go from que; for each neighbor index i in range(len(g[go])): if went[g[go][i]] is True, continue; otherwise set dis[g[go][i]] = dis[go] + 1, set went[g[go][i]] = True, then if dis[g[go][i]] < 3 append g[go][i] to que, else if dis[g[go][i]] == 3 append g[go][i] to see. RESULT dis, went, que, see]
            [COMMAND Set check1 = 0 and check2 = 0. RESULT check1, check2]
            [COMMAND For each e in see, skip e if ans[e] == 0; otherwise increment check1 if ans[e] % 3 == 1 and increment check2 if ans[e] % 3 == 2. RESULT check1, check2]
            [COMMAND If check1 > 0 and check2 > 0, print -1 and exit. RESULT program_terminated]
            [COMMAND Assign ans[s] and zero entries of see from que_1/que_2 according to the branch chosen by check1, check2, and queue lengths as detailed in alternative_flows. RESULT ans, que_1, que_2]
            [COMMAND Return ans, que_1, and que_2. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: check1 > 0 and check2 <= 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_2 is nonempty, set ans[s] = que_2.pop(); otherwise print -1 and exit. RESULT ans[s], que_2]
            [COMMAND For each e in see, if ans[e] == 0 and que_1 is nonempty, set ans[e] = que_1.pop(); if ans[e] == 0 and que_1 is empty, print -1 and exit. RESULT ans, que_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 > 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_1 is nonempty, set ans[s] = que_1.pop(); otherwise print -1 and exit. RESULT ans[s], que_1]
            [COMMAND For each e in see, if ans[e] == 0 and que_2 is nonempty, set ans[e] = que_2.pop(); if ans[e] == 0 and que_2 is empty, print -1 and exit. RESULT ans, que_2]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) > len(que_2)]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_2 is nonempty, set ans[s] = que_2.pop(); otherwise print -1 and exit. RESULT ans[s], que_2]
            [COMMAND For each e in see, if ans[e] == 0 and que_1 is nonempty, set ans[e] = que_1.pop(); if ans[e] == 0 and que_1 is empty, print -1 and exit. RESULT ans, que_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) <= len(que_2)]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_1 is nonempty, set ans[s] = que_1.pop(); otherwise print -1 and exit. RESULT ans[s], que_1]
            [COMMAND For each e in see, if ans[e] == 0 and que_2 is nonempty, set ans[e] = que_2.pop(); if ans[e] == 0 and que_2 is empty, print -1 and exit. RESULT ans, que_2]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: check1 > 0 and check2 > 0]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because both residue counters are positive."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 > 0 and que_2 is empty before assigning ans[s]]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_2 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 > 0 and ans[e] == 0 and que_1 is empty while filling see entries]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_1 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check2 > 0 and que_1 is empty before assigning ans[s]]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_1 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check2 > 0 and ans[e] == 0 and que_2 is empty while filling see entries]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_2 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) > len(que_2) and que_2 is empty before assigning ans[s]]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_2 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) > len(que_2) and ans[e] == 0 and que_1 is empty while filling see entries]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_1 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) <= len(que_2) and que_1 is empty before assigning ans[s]]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_1 is exhausted."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) <= len(que_2) and ans[e] == 0 and que_2 is empty while filling see entries]
        [LOG "Print -1 and exit immediately."]
        [THROW SystemExit "Program terminates because que_2 is exhausted."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
