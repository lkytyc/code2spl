[DEFINE_WORKER: "Simulates a one-dimensional token redistribution process to determine whether the initial r array can be transformed into the required b array and prints the sequence of dot/slash/backslash states or IMPOSSIBLE." solve]
    [INPUTS]
        <REF> sc </REF>: Scanner "Scanner source for reading integer c and the c required values of b."
        <REF> num </REF>: int "Case number used in the output prefix."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer c from sc.nextInt(); initialize result to false. RESULT c]
            [COMMAND Create int arrays b and r, each with length c; both arrays are initially zero-filled. RESULT b_and_r]
            [COMMAND For i from 0 to c-1 inclusive, read b[i] from sc.nextInt() and set r[i] = 1. RESULT initialized_b_r]
            [COMMAND Create int array d of length c, empty List<String> l, boolean good = false, and empty String s. RESULT state]
            [COMMAND Enter an infinite while(true) loop. At the beginning of each iteration set result = true, then for i from 0 to c-1 inclusive, if b[i] != r[i] then set result = false without breaking early. After that scan, if result is true, break out of the while loop. RESULT result]
            [COMMAND When result is false, begin one redistribution pass: set last = 0, create int arrays z and rr of length c, set good = false and set s = "". RESULT redistribute_state]
            [COMMAND For each i from 0 to c-1 inclusive where b[i] > 0, set want = b[i]. For j from last to c-1 inclusive: if r[j] == 0, append '.' to s and continue; otherwise subtract r[j] from want. If j > i and j > 0 and j < c-1, append '/' to s, set z[j] = -1, add r[j] to rr[j-1], and set good = true; else if j < i and j > 0 and j < c-1, append '\' to s, set z[j] = 1, add r[j] to rr[j+1], and set good = true; else append '.' to s and add r[j] to rr[j]. If want <= 0, set last = j+1 and break out of the j loop. RESULT updated_rr_and_s]
            [COMMAND After the redistribution loops, assign r = rr, append s to l, and if good is false, break out of the while loop. RESULT updated_r_and_l]
            [COMMAND After the while loop, branch on result: false leads to the impossible-output path, true leads to the possible-output path. RESULT branch_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After the while loop, result == false (the target b was not reached and no further redistribution movement is possible)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print to standard output: "Case #" + num + ": IMPOSSIBLE". RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the while loop, result == true (the target b was reached)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set s = ""; for i from 0 to c-1 inclusive, append '.' to s; then add s to l. RESULT s]
            [COMMAND Print to standard output: "Case #" + num + ": " + l.size(). RESULT _]
            [COMMAND For each String string in l, print string on its own line. RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: c is negative]
        [LOG "Creating an array of negative length c fails"]
        [THROW NegativeArraySizeException "new int[c] is evaluated with c < 0 and throws NegativeArraySizeException."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
