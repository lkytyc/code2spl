[DEFINE_WORKER: "Given positions x with weights p, this method merges adjacent entries by weight until s is inside the remaining interval, then accumulates the total absolute distance needed to visit the remaining positions starting from s." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The accumulated total cost r returned by the method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the first input line and assign the two values to n and s. RESULT n, s]
            [COMMAND Read n input lines, each containing a pair, and store the list of pairs in a. RESULT a]
            [COMMAND Set x to the first element of each pair in a. RESULT x]
            [COMMAND Set p to the second element of each pair in a. RESULT p]
            [COMMAND If x[0] > s, return the value x[-1] - s immediately. RESULT return_value]
            [COMMAND If x[-1] < s, return the value s - x[0] immediately. RESULT return_value]
            [COMMAND Initialize r to 0, i to 0, and j to n - 1. RESULT r, i, j]
            [COMMAND Copy p into d, initialize e as an empty collections.defaultdict(list), and initialize sk as an empty set. RESULT d, e, sk]
            [COMMAND While i < j, repeatedly perform the following merge loop. RESULT i, j, d, sk]
            [COMMAND Inside the merge loop, if x[i] > s, break out of the while loop. RESULT i, j, d, sk]
            [COMMAND Inside the merge loop, if x[j] < s, break out of the while loop. RESULT i, j, d, sk]
            [COMMAND Inside the merge loop, if d[i] >= d[j], add d[j] to d[i], decrement j by 1, and then if j >= 0 and d[j] < d[j+1], copy d[j+1] into d[j] and add i to sk. RESULT i, j, d, sk]
            [COMMAND Inside the merge loop, otherwise when d[i] < d[j], add d[i] to d[j], increment i by 1, and then if i < n and d[i] < d[i-1], copy d[i-1] into d[i] and add i to sk. RESULT i, j, d, sk]
            [COMMAND After the merge loop, set t equal to s. RESULT t]
            [COMMAND After setting t, if i equals j, add abs(x[i] - s) to r, set t to x[i], decrement i by 1, and increment j by 1. RESULT r, t, i, j]
            [COMMAND After setting t, if i does not equal j and x[i] > s, add abs(x[j] - s) to r, set t to x[j], decrement i by 1, and increment j by 1. RESULT r, t, i, j]
            [COMMAND After setting t, if i does not equal j and x[i] <= s, add abs(x[i] - s) to r, set t to x[i], decrement i by 1, and increment j by 1. RESULT r, t, i, j]
            [COMMAND While i >= 0 or j < n, repeatedly perform the following outward expansion loop. RESULT i, j, r, t, d]
            [COMMAND Inside the outward expansion loop, if i < 0, add abs(x[-1] - t) to r and break out of the loop. RESULT r]
            [COMMAND Inside the outward expansion loop, if j >= n, add abs(x[0] - t) to r and break out of the loop. RESULT r]
            [COMMAND Inside the outward expansion loop, if i is in sk, copy d[i] to d[i-1], decrement i by 1, and continue to the next iteration. RESULT d, i]
            [COMMAND Inside the outward expansion loop, if j is in sk, copy d[j] to d[j+1], increment j by 1, and continue to the next iteration. RESULT d, j]
            [COMMAND Inside the outward expansion loop, if d[i] >= d[j], add abs(x[i] - t) to r, set t to x[i], and decrement i by 1. RESULT r, t, i]
            [COMMAND Inside the outward expansion loop, otherwise when d[i] < d[j], add abs(x[j] - t) to r, set t to x[j], and increment j by 1. RESULT r, t, j]
            [COMMAND After the outward expansion loop, return the final value of r. RESULT r]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
