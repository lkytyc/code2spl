[DEFINE_WORKER: "Reads an n by n matrix from standard input, computes the main diagonal sum and row/column duplicate counts, then prints the formatted case output with x+1, sum, rmax, and cmax." Ans]
    [INPUTS]
        <REF> x </REF>: long long int "Zero-based case number used as x+1 in the printed output."
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_output </REF>: stdout "Format: Case #%lld: %lld %lld %lld\n, printed with x+1, sum, rmax, and cmax."
        <REF> return_value </REF>: int "Always returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare local variables: long long int n, i, j, sum initialized to 0, r, c, k. RESULT sum]
            [COMMAND Read a long long integer from standard input using scanf and store it in n. RESULT n]
            [COMMAND Declare Mat as a long long matrix of size n by n. RESULT Mat]
            [COMMAND Declare count as a long long array of length n. RESULT count]
            [COMMAND For i from 0 to n-1, for j from 0 to n-1, read a long long integer from standard input into Mat[i][j]. RESULT Mat]
            [COMMAND For i from 0 to n-1, for j from 0 to n-1, if i equals j then add Mat[i][j] to sum. RESULT sum]
            [COMMAND For each i from 0 to n-1: set count[i] to 0, then for each j from 0 to n-1 and for each k from j+1 to n-1, if Mat[i][j] equals Mat[i][k], then set count[i] to 2 if count[i] is 0, otherwise increment count[i], and break only the k loop. RESULT count]
            [COMMAND Initialize rmax to count[0]. This is valid only when n is positive. RESULT rmax]
            [COMMAND Within an outer loop over i from 0 to n-1, for j from 0 to n-1, if count[j] is greater than or equal to count[j+1], overwrite rmax with count[j]. Note that when j is n-1, count[j+1] reads count[n], which is out of bounds for an array of length n. RESULT rmax]
            [COMMAND Declare ccount as a long long array of length n. RESULT ccount]
            [COMMAND For each j from 0 to n-1: set ccount[j] to 0, then for each i from 0 to n-1 and for each k from i+1 to n-1, if Mat[i][j] equals Mat[k][j], then set ccount[j] to 2 if ccount[j] is 0, otherwise increment ccount[j], and break only the k loop. RESULT ccount]
            [COMMAND Initialize cmax to ccount[0]. This is valid only when n is positive. RESULT cmax]
            [COMMAND Within an outer loop over i from 0 to n-1, for j from 0 to n-1, if ccount[j] is greater than or equal to ccount[j+1], overwrite cmax with ccount[j]. Note that when j is n-1, ccount[j+1] reads ccount[n], which is out of bounds for an array of length n. RESULT cmax]
            [COMMAND Print the formatted string Case #%lld: %lld %lld %lld\n with x+1, sum, rmax, and cmax. RESULT printed_output]
            [COMMAND Return 0 from the Ans function. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
