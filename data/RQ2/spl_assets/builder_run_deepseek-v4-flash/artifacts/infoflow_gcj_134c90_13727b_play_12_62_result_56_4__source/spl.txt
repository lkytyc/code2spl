[DEFINE_WORKER: "Reads Rock-Paper-Scissors input lines and either triples the single input with a countermove or eliminates possible moves from the multi-input list to select one response, then prints 'Case #<i>: <result>'." play]
    [INPUTS]
        <REF> i </REF>: int "Case number used in the output prefix."
        <REF> n </REF>: int "Number of lines to process in the multi-input branch; also selects the single-input branch when n equals 1."
        <REF> in </REF>: Scanner "Scanner used to read the input lines."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to an empty string, R, P, and S to true, and j to 1. RESULT state]
            [COMMAND If n equals 1, execute the alternative flow for the single-input case; otherwise execute the alternative flow for the multi-input case. RESULT computed result]
            [COMMAND Print 'Case #' + i + ': ' + result to standard output. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: n == 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next line from the Scanner into input. RESULT input]
            [COMMAND Append input to result three times, so result becomes input + input + input. RESULT result]
            [COMMAND If the first character of input is 'P', append 'S' to result. RESULT result]
            [COMMAND If the first character of input is 'R', append 'P' to result. RESULT result]
            [COMMAND If the first character of input is 'S', append 'R' to result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: n != 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND While j is less than or equal to n, iterate the loop body: first, if R or P or S is true, read the next line into input; if its first character is 'P', set R and P to false; if it is 'R', set R and S to false; if it is 'S', set S and R to false. After the guarded block, increment j by 1 regardless of whether a line was read. RESULT R, P, S, j]
            [COMMAND After the loop, if R is true, append 'P' to result; else if P is true, append 'S' to result; else if S is true, append 'R' to result; else set result to 'IMPOSSIBLE'. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
