[DEFINE_WORKER: "Reads test cases containing arrays a and b, computes nearest-greater-element boundary arrays for both arrays, and totals weighted contributions from index pairs whose values differ by at most k, printing the total for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer t from standard input. RESULT t]
            [COMMAND For each test case m from 1 to t inclusive, process one case. RESULT m]
            [COMMAND Within the test-case loop over m, read integers n and k from standard input. RESULT n_k]
            [COMMAND Within the test-case loop over m, declare arrays a, b, l_a, l_b, r_a, and r_b of length n. RESULT arrays]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, initialize l_a[i], l_b[i], r_a[i], and r_b[i] to i. RESULT boundary_arrays_initialized]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, read an integer into a[i]. RESULT a]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, read an integer into b[i]. RESULT b]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, compute l_a[i] as the nearest index to the left of i whose a value is greater than a[i], or -1 if none exists, by moving left while a[index] <= a[i]. RESULT l_a]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, compute l_b[i] as the nearest index to the left of i whose b value is greater than b[i], or -1 if none exists, by moving left while b[index] <= b[i]. RESULT l_b]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, compute r_a[i] as the nearest index to the right of i whose a value is greater than a[i], or n if none exists, by moving right while a[index] <= a[i]. RESULT r_a]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1, compute r_b[i] as the nearest index to the right of i whose b value is greater than b[i], or n if none exists, by moving right while b[index] <= b[i]. RESULT r_b]
            [COMMAND Within the test-case loop over m, initialize long long int tot to 0. RESULT tot]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1 and for each j from i to r_a[i]-1 inclusive, if l_b[j] < i and a[i]-b[j] is between -k and k inclusive, add (i - max(l_a[i], l_b[j])) * (min(r_a[i], r_b[j]) - j) to tot. RESULT tot]
            [COMMAND Within the test-case loop over m, for each i from 0 to n-1 and for each j from i+1 to r_b[i]-1 inclusive, if l_a[j] < i and a[j]-b[i] is between -k and k inclusive, add (i - max(l_b[i], l_a[j])) * (min(r_b[i], r_a[j]) - j) to tot. RESULT tot]
            [COMMAND Within the test-case loop over m, print "Case #%d: %lld\n" with m and tot. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
