[DEFINE_WORKER: "Reads q records, initializes a weighted-median tracking structure from the first record, then processes each remaining record as either a weighted insertion or a median-cost query and returns newline-separated query answers." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Newline-separated query answers; each answer is the current median x and the computed cost xl + xr + bt."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read q as an integer from input using I(). RESULT q]
            [COMMAND Read aa as a list of q integer lists from input by calling LI() once for each index in range(q). RESULT aa]
            [COMMAND Initialize bt = aa[0][2], m = 2 ** 20, c = collections.defaultdict(int), lq = [inf], rq = [inf], c[-inf] = 1, c[inf] = 1, x = aa[0][1], c[x] = 1, xl = 0, lc = 0, xr = 0, rc = 0, and r = []. RESULT initial_state]
            [COMMAND For each record t in aa[1:], begin a loop iteration. RESULT t]
            [COMMAND In the loop over t, if t[0] == 2, append '{} {}'.format(x, xl + xr + bt) to r and proceed to the next iteration. RESULT r]
            [COMMAND In the loop over t, when t[0] != 2, unpack the record as _, a, b = t. RESULT a_b]
            [COMMAND In the update branch (t[0] != 2), increase bt by b. RESULT bt]
            [COMMAND In the update branch, if a == x, increment c[x] by 1 and skip to the next loop iteration. RESULT c]
            [COMMAND In the update branch, only when a != x, increment c[a] by 1. RESULT c]
            [COMMAND In the update branch, only when a != x, if c[a] == 1 after the increment, use heapq.heappush to push -a onto lq when a < x, otherwise push a onto rq. RESULT lq_rq]
            [COMMAND In the update branch, only when a != x, set xc = c[x]. RESULT xc]
            [COMMAND In the update branch, only when a != x and a < x, update xl += x - a and lc += 1. RESULT xl_lc]
            [COMMAND In the update branch, only when a != x and a < x, compute t = xl + xr (reusing the local loop variable t without affecting the loop iteration), nx = -lq[0], nc = c[nx], s = x - nx, and nt = xl - lc * s + xr + (rc + xc) * s. RESULT nt_left]
            [COMMAND In the update branch, only when a != x and a < x, if nt <= t, perform a left median shift: update xl -= lc * s, xr += (rc + xc) * s, lc -= nc, rc += xc, heapq.heappush(rq, x), set x = nx, and heapq.heappop(lq). RESULT state_after_left_shift]
            [COMMAND In the update branch, only when a != x and a > x, update xr += a - x and rc += 1. RESULT xr_rc]
            [COMMAND In the update branch, only when a != x and a > x, compute t = xl + xr (reusing the local loop variable t without affecting the loop iteration), nx = rq[0], nc = c[nx], s = nx - x, and nt = xl + (lc + xc) * s + xr - rc * s. RESULT nt_right]
            [COMMAND In the update branch, only when a != x and a > x, if nt < t, perform a right median shift: update xl += (lc + xc) * s, xr -= rc * s, lc += xc, rc -= nc, heapq.heappush(lq, -x), set x = nx, and heapq.heappop(rq). RESULT state_after_right_shift]
            [COMMAND After the loop completes, return '\n'.join(r). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
