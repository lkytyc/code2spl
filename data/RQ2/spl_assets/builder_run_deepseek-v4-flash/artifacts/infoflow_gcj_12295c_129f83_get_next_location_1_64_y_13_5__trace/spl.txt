[DEFINE_WORKER: "Reads a given number of starting coordinate and direction pairs, moves each coordinate one unit in the specified direction, counts the resulting next-location coordinates, and returns the most frequent next location with lexicographically smallest tie-breaking." get_next_location]
    [INPUTS]
        <REF> num_of_people </REF>: int "Number of people/locations to process. The loop runs exactly num_of_people times because people_count starts at 1 and the condition is inclusive (people_count <= num_of_people)."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[int] "The most frequent next-location coordinate after applying the movement. If multiple coordinates tie, returns the lexicographically smallest coordinate. Each coordinate is a two-element list [x, y]."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize coordinates_list as an empty list and people_count to 1. RESULT coordinates_list]
            [COMMAND While people_count <= num_of_people, read one line from stdin, split it into tokens, parse x = int(token[0]), y = int(token[1]), and dir = token[2]; update the coordinate by incrementing y for 'N', decrementing y for 'S', incrementing x for 'E', and decrementing x for 'W' (if dir is any other value, the coordinate remains unchanged); append the resulting [x, y] list to coordinates_list; then increment people_count by 1. RESULT coordinates_list]
            [COMMAND Initialize next_dest as an empty list and next_dest_count as an empty list. RESULT next_dest_count]
            [COMMAND For each coordinate in coordinates_list, search next_dest in order. If coordinate equals an existing destination, increment the corresponding next_dest_count value and stop the inner search; if no existing destination matches, append coordinate to next_dest and append a count of 1 to next_dest_count. RESULT next_dest_count]
            [COMMAND Initialize highest_count to next_dest_count[0], dest_count_idx to 0, and popular_next_dest to a list containing next_dest[0]. RESULT popular_next_dest]
            [COMMAND For each dest_count in next_dest_count, using dest_count_idx as the parallel index, skip the first element by incrementing dest_count_idx and continuing. For each later element, if dest_count equals the current highest_count, append next_dest[dest_count_idx] to popular_next_dest; if dest_count is greater than the current highest_count, reset popular_next_dest to an empty list, set highest_count to dest_count, and append next_dest[dest_count_idx]. Increment dest_count_idx at the end of each non-skipped iteration. RESULT popular_next_dest]
            [COMMAND If len(popular_next_dest) == 1, return popular_next_dest[0]. Otherwise, set smallest_number to popular_next_dest[0], initialize dest_idx to 0, iterate through popular_next_dest skipping the first element, update smallest_number whenever dest < smallest_number, and after the loop return smallest_number. RESULT returned_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
