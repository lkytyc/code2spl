[DEFINE_WORKER: "Computes the trace of the input square matrix, sorts its rows and the rows of its transpose to count adjacent duplicate pairs, and prints one formatted Case # line whose last two displayed values depend on whether the duplicate counts are zero." vestigium]
    [INPUTS]
        <REF> a </REF>: int[][] "The square input matrix of size n x n."
        <REF> n </REF>: int "The number of rows and columns of a; used for loop bounds and for allocating b. It must be non-negative for successful allocation of b."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize k to 0. RESULT k]
            [COMMAND Initialize r to 0. RESULT r]
            [COMMAND Initialize c to 0. RESULT c]
            [COMMAND Allocate b as a new n by n int array. RESULT b]
            [COMMAND For i from 0 to n-1 inclusive, add a[i][i] to k. RESULT k]
            [COMMAND For i from 0 to n-1 inclusive and for j from 0 to n-1 inclusive, assign b[i][j] = a[j][i] to store the transpose of a in b. RESULT b]
            [COMMAND For i from 0 to n-1 inclusive, sort a[i] in ascending order using Arrays.sort. RESULT a]
            [COMMAND Initialize repeat to 0. RESULT repeat]
            [COMMAND Initialize count to 0. RESULT count]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive, if a[i][j] equals a[i][j+1] then increment repeat; after every such comparison set count = Math.max(count, repeat); after finishing the row, if repeat > 0 reset repeat to 0. RESULT count]
            [COMMAND For i from 0 to n-1 inclusive, sort b[i] in ascending order using Arrays.sort. RESULT b]
            [COMMAND Initialize count1 to 0. RESULT count1]
            [COMMAND Initialize repeat1 to 0. RESULT repeat1]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive, if b[i][j] equals b[i][j+1] then increment repeat1; after every such comparison set count1 = Math.max(count1, repeat1); after finishing the row, if repeat1 > 0 reset repeat1 to 0. RESULT count1]
            [COMMAND Set r = count + 1. RESULT r]
            [COMMAND Set c = count1 + 1. RESULT c]
            [COMMAND Set counter = 1. RESULT counter]
            [COMMAND Select and execute exactly one output branch based on whether count and count1 are zero or non-zero, increment counter in that branch, and then end the method. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 && count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the line formed by 'Case #' + counter + ': ' + ' ' + k + ' ' + count + ' ' + count1. RESULT output_line]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 && count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the line formed by 'Case #' + counter + ': ' + ' ' + k + ' ' + r + ' ' + c. RESULT output_line]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 && count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the line formed by 'Case #' + counter + ': ' + ' ' + k + ' ' + r + ' ' + count1. RESULT output_line]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 && count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the line formed by 'Case #' + counter + ': ' + ' ' + k + ' ' + count + ' ' + c. RESULT output_line]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
