[DEFINE_WORKER: "Repeatedly updates an initially all-ones row r by moving unit stones one step toward positive target counts b, recording each pass as a string, and prints either IMPOSSIBLE or the number of pass rows followed by the rows." solve]
    [INPUTS]
        <REF> sc </REF>: Scanner "Scanner used to read c and then the c integers that form the target array b."
        <REF> num </REF>: int "Case number used in the 'Case #num:' output prefix."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to false, read integer c from sc, create arrays b and r of length c, then for each index i from 0 through c-1 read b[i] from sc and set r[i] to 1. RESULT initial_state]
            [COMMAND Initialize unused array d of length c, list l as empty, boolean good as false, and string s as empty. RESULT loop_state]
            [COMMAND Enter while(true). At the start of each iteration set result to true; then for each index i from 0 through c-1, if b[i] differs from r[i] set result to false. RESULT comparison_result]
            [COMMAND If result is true, break out of the while loop and skip the update pass. RESULT success_break]
            [COMMAND If the loop was not broken, reset last to 0, set good to false, reset s to empty, and create a new zero-filled array rr of length c. RESULT update_reset]
            [COMMAND For each index i from 0 through c-1 where b[i] is greater than 0, set want to b[i] and scan j from last through c-1. If r[j] is 0 append '.' to s and continue. Otherwise subtract r[j] from want; if j>i and 0<j<c-1 append '/' to s and add r[j] to rr[j-1]; else if j<i and 0<j<c-1 append '\' to s and add r[j] to rr[j+1]; else append '.' to s and add r[j] to rr[j]. Set good to true when either move branch is taken. When want is less than or equal to 0, set last to j+1 and stop the j scan for this i. RESULT updated_rr_and_s]
            [COMMAND After the update-pass loop assigns all moved values, set r to rr, append s to l, and if good is false break out of the while loop. RESULT updated_loop_state]
            [COMMAND After the while loop exits, if result is false print 'Case #' + num + ': IMPOSSIBLE'; otherwise set s to a string consisting of exactly c periods, append s to l, print 'Case #' + num + ': ' + l.size(), and print every string in l on its own line. RESULT final_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
