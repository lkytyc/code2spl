[DEFINE_WORKER: "Scans every unordered pair of points in M to maintain extreme positive-slope ratios, then returns a scaling pair [x, y] or None depending on whether an integer lies between those ratio bounds." solve]
    [INPUTS]
        <REF> N </REF>: integer "Number of point rows in M."
        <REF> M </REF>: array of two-integer coordinate pairs "List of points; each point is M[i] = [x, y]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of two integers or null "Scaling pair [x, y], or None when an early-return condition is met or execution falls off the end of the supplied method."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize topHigh, botHigh, topLow, and botLow to None. RESULT initial_state]
            [COMMAND For i from 0 to N-1 and j from i+1 to N-1, read a=M[i][0], b=M[i][1], c=M[j][0], d=M[j][1]; if a==c or b==d, continue to the next j. RESULT scanned_pairs]
            [COMMAND Within the nested loop, compute top = a-c, bot = d-b, wasNeg = (top * bot < 0), then replace top with abs(top) and bot with abs(bot). RESULT signed_slope_components]
            [COMMAND Within the nested loop, if b-d > 0: if wasNeg is true, return None; otherwise update topHigh and botHigh to current top and bot when topHigh is None or top * botHigh < topHigh * bot. RESULT high_ratio_state]
            [COMMAND Within the nested loop, for the else case b-d <= 0, which after the earlier equality skip means b < d: if wasNeg is true, continue to the next pair; otherwise update topLow and botLow to current top and bot when topLow is None or top * botHigh > topHigh * bot. RESULT low_ratio_state]
            [COMMAND After all pairs have been scanned, if topHigh is None and topLow is None, return [1, 1]. RESULT no_extremes_return]
            [COMMAND If topHigh is None: if botLow == topLow, return [1, 2]; elif topLow > botLow, return [1, int(math.ceil(topLow / botLow))]; else, return [int(math.ceil(botLow / topLow)), 1]. RESULT high_missing_return]
            [COMMAND If topLow is None: if botHigh == topHigh, return [2, 1]; elif botHigh > topHigh, return [int(math.ceil(botHigh / topHigh)), 1]; else, return [1, int(math.ceil(topHigh / botHigh))]. RESULT low_missing_return]
            [COMMAND Compute a = topHigh / float(botHigh) and b = topLow / float(botLow). RESULT ratio_bounds]
            [COMMAND If a <= b, return None. RESULT invalid_bounds_return]
            [COMMAND Set xMin = 1 and xMax = 1e20. RESULT search_bounds]
            [COMMAND While xMax > xMin: set mid = int((xMax + xMin) / 2.), low = b * mid, and high = a * mid. If low == int(low), set xMax = mid when high > low + 1, otherwise set xMin = mid + 1. If low is not integer, set xMax = mid when math.ceil(low) < high, otherwise set xMin = mid + 1. RESULT binary_search_state]
            [COMMAND When the while loop exits, the supplied method has no explicit return statement, so return None. RESULT final_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]
