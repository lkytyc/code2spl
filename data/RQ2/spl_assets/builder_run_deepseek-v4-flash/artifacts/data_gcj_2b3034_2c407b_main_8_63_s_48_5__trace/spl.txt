[DEFINE_WORKER: "Reads a series of test cases containing patterns that start or end with '*', merges the fixed characters of each test case's patterns into one string, and prints either the merged string or '*' when conflicting fixed characters appear." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Returns 0 after all test cases have been processed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integer variables t, n, i, j, k and character arrays s of size 10001 and str of size 101; initialize t and n to 0. RESULT variables]
            [COMMAND Read the first integer from standard input into t using scanf("%d", &t); this value is the number of test cases. RESULT t]
            [COMMAND For each test case index i from 1 through t inclusive, process one test case. RESULT i]
            [COMMAND Within the outer loop for test case i, read the integer n from standard input using scanf("%d", &n). RESULT n]
            [COMMAND Within the outer loop for test case i, declare a two-dimensional variable-length array a of size n by 10001 and a character variable dd. RESULT a]
            [COMMAND Within the outer loop for test case i, initialize every cell a[j][k] to '#' for all rows j from 0 to n-1 and all columns k from 0 to 10000 inclusive. RESULT a]
            [COMMAND Within the outer loop for test case i, set the conflict flag to 0. RESULT flag]
            [COMMAND Within the outer loop for test case i, for each pattern row j from 0 to n-1, read one character into dd using scanf("%c", &dd) to consume the separator, then read a pattern string into str using scanf("%s", str). RESULT str]
            [COMMAND Within the current pattern row j, if str[0] equals '*', right-align the pattern into row a[j]: set x to 10000 and for k from len(str) down to 0 inclusive assign a[j][x] = str[k] and decrement x; this stores the null terminator into a[j][10000], which is not scanned by the later merge loop. RESULT a]
            [COMMAND Within the current pattern row j, if str[0] does not equal '*' but str[len(str)-1] equals '*', left-align the pattern into row a[j] by copying str[k] to a[j][k] for k from 0 while str[k] is not the null terminator. RESULT a]
            [COMMAND Within the current pattern row j, if str does not start with '*' and does not end with '*', take no action and leave row a[j] initialized to all '#'. RESULT a]
            [COMMAND Within the outer loop for test case i, after all pattern rows have been stored, set x to 0 as the output buffer index. RESULT x]
            [COMMAND Within the outer loop for test case i, scan output columns j from 0 to 9999 inclusive; for each column set f to 0, then for each row k from 0 to n-1 inspect a[k][j]. RESULT column_state]
            [COMMAND During the column scan, if f is 1 (so x is at least 1) and a[k][j] differs from s[x-1] and is neither '*' nor '#', set flag to 1 and break out of the row loop. RESULT flag]
            [COMMAND During the column scan, if a[k][j] is not '*' and not '#', and f is 0, assign s[x] = a[k][j], increment x, and set f to 1. RESULT s]
            [COMMAND After the row loop for column j, if flag is 1, break out of the column loop early. RESULT flag]
            [COMMAND After the column loop terminates, terminate the output string by setting s[x] to the null character. RESULT s]
            [COMMAND If flag is 0, print "Case #%d: %s\n" using the current test case index i and the merged string s. RESULT stdout]
            [COMMAND After the outer loop over all test cases finishes, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: flag == 1 after the column merge scan for the current test case]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #%d: *\n" using the current test case index i. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
