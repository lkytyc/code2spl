[DEFINE_WORKER: "Reads waffle test cases from standard input and, for each case, determines whether exactly one horizontal cut and one vertical cut can divide the chocolate chips into four equal quadrants, printing POSSIBLE or IMPOSSIBLE for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read a line from standard input into str; do not check whether readline returned NULL. RESULT str]
            [COMMAND Parse an integer from str with sscanf into T, ignoring sscanf's return value, so T is not guaranteed to be initialized if parsing fails. RESULT T]
            [COMMAND Free the memory pointed to by str. RESULT none]
            [COMMAND Initialize u_w to 1. RESULT u_w]
            [COMMAND Execute the test-case loop while the pre-decrement value of T is not zero; after each condition test, T is decremented. For an initial positive T this runs T times and leaves T as -1. RESULT none]
            [COMMAND At the start of each test case, declare R, C, H, V; loop counters i, j, k, l, m; Waffle as an array of 10 char pointers; and ChocChipCount as an array of 10 ints. RESULT none]
            [COMMAND Initialize ChocChipCount[0] through ChocChipCount[9] to 0 using a loop from i = 0 to i < 10. RESULT ChocChipCount]
            [COMMAND Read a line into str, parse R, C, H, and V from it with sscanf, ignoring the return value, then free str. H and V are read but never used. RESULT R,C,H,V]
            [COMMAND For each row i from 0 to R-1, read a line into Waffle[i]; for each column j from 0 to C-1, increment ChocChipCount[i] when Waffle[i][j] equals '@'. No bounds check is performed for R exceeding 10. RESULT Waffle,ChocChipCount]
            [COMMAND Set retStr to "IMPOSSIBLE". RESULT retStr]
            [COMMAND For each horizontal cut i from 1 to R-1 inclusive, continue only while retStr is still "IMPOSSIBLE"; compute topCount as the sum of ChocChipCount[0] through ChocChipCount[i-1] and bottomCount as the sum of ChocChipCount[i] through ChocChipCount[R-1]. RESULT topCount,bottomCount]
            [COMMAND If topCount is not equal to bottomCount, continue to the next horizontal cut without searching vertical cuts for this i. RESULT none]
            [COMMAND When topCount equals bottomCount, for each vertical cut k from 1 to C-1 inclusive, compute topleftCount by counting '@' in rows 0 through i-1 and columns 0 through k-1; set toprightCount to the total chips in rows 0 through i-1 minus topleftCount. RESULT topleftCount,toprightCount]
            [COMMAND If topleftCount equals toprightCount, compute bottomleftCount by counting '@' in rows i through R-1 and columns 0 through k-1; set bottomrightCount to the total chips in rows i through R-1 minus bottomleftCount. RESULT bottomleftCount,bottomrightCount]
            [COMMAND If bottomleftCount equals bottomrightCount, set retStr to "POSSIBLE" and break out of the vertical-cut loop; the outer horizontal-cut loop terminates on its next condition check because retStr is no longer "IMPOSSIBLE". RESULT retStr]
            [COMMAND After the search loops, print "Case #%d: %s" without adding a trailing newline using u_w and retStr, then increment u_w by 1. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: topCount != bottomCount]
        [SEQUENTIAL_BLOCK]
            [COMMAND Abandon the current horizontal cut i and proceed to the next iteration of the horizontal-cut loop. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: topleftCount == toprightCount]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute bottomleftCount and bottomrightCount for the current horizontal cut i and vertical cut k. RESULT bottomleftCount,bottomrightCount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: bottomleftCount == bottomrightCount]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set retStr to "POSSIBLE" and break out of the vertical-cut loop. RESULT retStr]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
