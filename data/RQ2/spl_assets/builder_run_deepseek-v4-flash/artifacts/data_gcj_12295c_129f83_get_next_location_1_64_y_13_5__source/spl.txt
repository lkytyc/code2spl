[DEFINE_WORKER: "Reads num_of_people direction commands, computes each person's next coordinate, counts the resulting destinations, and returns the single most frequent destination or the lexicographically smallest coordinate among ties." get_next_location]
    [INPUTS]
        <REF> num_of_people </REF>: int "Number of people/input lines to process; expected to be at least 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of two ints "Selected destination coordinate [x, y]: the unique most frequent next location, or the lexicographically smallest coordinate when multiple destinations tie for highest frequency."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize coordinates_list to an empty list and people_count to 1. RESULT coordinates_list, people_count]
            [COMMAND While people_count <= num_of_people, continue processing the next person. RESULT people_count]
            [COMMAND Within the while loop, read a line from standard input, split it into tokens, and set x = int(token[0]), y = int(token[1]), dir = token[2]. RESULT x, y, dir]
            [COMMAND Within the while loop, if dir is 'N' increment y by 1; if dir is 'S' decrement y by 1; if dir is 'E' increment x by 1; if dir is 'W' decrement x by 1. RESULT x, y]
            [COMMAND Within the while loop, create coordinates = [x, y], append coordinates to coordinates_list, then increment people_count by 1. RESULT coordinates_list, people_count]
            [COMMAND After the while loop, initialize next_dest = [] and next_dest_count = []. RESULT next_dest, next_dest_count]
            [COMMAND For each coordinates in coordinates_list, set found = 0 and dest_idx = 0. RESULT found, dest_idx]
            [COMMAND Within the outer loop over coordinates_list, for each dest in next_dest, if coordinates == dest, increment next_dest_count[dest_idx] by 1, set found = 1, and break; otherwise increment dest_idx. RESULT next_dest_count, found]
            [COMMAND Within the outer loop over coordinates_list, after the inner search, if found == 0, append coordinates to next_dest and append 1 to next_dest_count. RESULT next_dest, next_dest_count]
            [COMMAND Initialize popular_next_dest = [], append next_dest[0] to it, set highest_count = next_dest_count[0], and set dest_count_idx = 0. RESULT popular_next_dest, highest_count, dest_count_idx]
            [COMMAND For each dest_count in next_dest_count, if dest_count_idx == 0, increment dest_count_idx and continue; otherwise compare dest_count with highest_count and update popular_next_dest and highest_count as appropriate, then increment dest_count_idx. RESULT popular_next_dest, highest_count, dest_count_idx]
            [COMMAND If len(popular_next_dest) == 1, return popular_next_dest[0]. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len(popular_next_dest) != 1, meaning multiple destinations tie for the highest frequency.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set smallest_number = popular_next_dest[0] and dest_idx = 0. RESULT smallest_number, dest_idx]
            [COMMAND For each dest in popular_next_dest, if dest_idx == 0, increment dest_idx and continue; otherwise if dest < smallest_number, set smallest_number = dest; then increment dest_idx. RESULT smallest_number]
            [COMMAND Return smallest_number. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
