[DEFINE_WORKER: "Reads N integers from standard input, maintains three occurrence counters in descending order, multiplies the answer when the current value matches two or three counters, applies a final 3 or 6 multiplier based on initialization status, and prints the answer modulo 1000000007." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not used by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND In a try-with-resources block, create a Scanner over System.in; read integer N; set MOD = 1000000007; set fst = -1, snd = -1, thd = -1; set answer = 1. RESULT initial_state]
            [COMMAND Enter a for loop with i from 0 to N-1 inclusive; at each iteration read val = sc.nextInt() from the Scanner. RESULT val]
            [COMMAND Within the loop, if val == 0, set the first counter among fst, snd, thd that is still -1 to 1; if none is -1, leave the counters unchanged; do not perform the sorting step on this path. RESULT counters_after_zero]
            [COMMAND Within the loop, if val != 0, apply this ordered matching: if fst, snd, and thd all equal val, increment fst and set answer = (answer * 3) % MOD; else if fst == val and snd == val, increment fst and set answer = (answer * 2) % MOD; else if snd == val and thd == val, increment snd and set answer = (answer * 2) % MOD; else if fst == val and thd == val, increment fst and set answer = (answer * 2) % MOD; else if fst == val, increment fst; else if snd == val, increment snd; else if thd == val, increment thd; if none matches, make no counter changes and leave answer unchanged. RESULT counters_and_answer_after_match]
            [COMMAND Within the loop, after the nonzero-val update, snapshot old_fst, old_snd, and old_thd; reassign fst = max(old_fst, old_snd, old_thd), thd = min(old_fst, old_snd, old_thd), and snd = (old_fst + old_snd + old_thd) - fst - thd so that fst >= snd >= thd before the next iteration. RESULT sorted_counters]
            [COMMAND After the for loop ends, if snd < 0 and thd < 0, set answer = (answer * 3) % MOD; otherwise, including the original case where only thd < 0, set answer = (answer * 6) % MOD. RESULT final_answer]
            [COMMAND Print final_answer to standard output using System.out.println. RESULT printed_answer]
            [COMMAND Exit the try-with-resources block and automatically close the Scanner. RESULT scanner_closed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
