[DEFINE_WORKER: "Reads a space-separated sequence from standard input, separates even- and odd-indexed elements into two halves, and computes the minimum number of element changes required to make each half uniform while keeping their uniform values different." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Enter a try-with-resources block using a BufferedReader created from InputStreamReader(System.in). RESULT in]
            [COMMAND Read the first line from System.in and parse it as integer n. RESULT n]
            [COMMAND Read the second line from System.in as string line containing the space-separated sequence. RESULT line]
            [COMMAND Create int arrays seqA and seqB with length n/2. RESULT seqA, seqB]
            [COMMAND Initialize offset=0, next=0, regx=' ', empty ArrayLists listA and listB for Integer, and empty HashMaps mapA and mapB from Integer to Integer. RESULT state]
            [COMMAND For i from 0 to n-1, with j = i >> 1, set next to line.indexOf(regx, offset), or to line.length() when no space is found; if i is even, parse the token into seqA[j] and update mapA/listA; otherwise parse the token into seqB[j] and update mapB/listB; then set offset = next + 1. RESULT parsed_data]
            [COMMAND Find the key xa in listA with the highest frequency vma in mapA, using a strict greater-than comparison. RESULT vma, xa]
            [COMMAND Find the key xb in listB with the highest frequency vmb in mapB, using a strict greater-than comparison. RESULT vmb, xb]
            [COMMAND Count cca as the number of elements in seqA that are not equal to xa. RESULT cca]
            [COMMAND Count ccb as the number of elements in seqB that are not equal to xb. RESULT ccb]
            [COMMAND If xa != xb, set ans = cca + ccb; otherwise, when xa == xb, find the most frequent key xa2 in mapA excluding xa and the most frequent key xb2 in mapB excluding xb, compute cca2 as the count of seqA elements not equal to xa2 and ccb2 as the count of seqB elements not equal to xb2, then set ans = cca + ccb2 if cca2 > ccb2, otherwise set ans = cca2 + ccb. RESULT ans]
            [COMMAND Print the value of ans to System.out. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: An IOException occurs while reading from System.in or while closing the BufferedReader]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the caught IOException to System.err and terminate the method without rethrowing it. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
