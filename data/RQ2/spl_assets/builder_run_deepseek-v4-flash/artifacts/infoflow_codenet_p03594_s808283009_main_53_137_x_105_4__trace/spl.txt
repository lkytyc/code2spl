[DEFINE_WORKER: "Reads h, w, and d, fills an h-by-w grid row-major by assigning each cell the smallest color 0..3 not used by any previously filled cell at Manhattan distance exactly d, then prints the grid as R/Y/G/B characters, using R for cells left uncolored." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; unused by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the first integer from the input stream in and assign it to h. RESULT h]
            [COMMAND Read the second integer from the input stream in and assign it to w. RESULT w]
            [COMMAND Read the third integer from the input stream in and assign it to d. RESULT d]
            [COMMAND Allocate int[][] ans = new int[h][w]; all entries initially default to 0. RESULT ans]
            [COMMAND For each row index i from 0 to h-1, call Arrays.fill(ans[i], -1) so every cell is initialized to -1. RESULT ans]
            [COMMAND For i from 0 to h-1 and for j from 0 to w-1, create a fresh boolean array not of length 4 with all values false to track unavailable colors. RESULT not]
            [COMMAND Within the same (i,j) processing, start x = j + d and y = i; while y <= i + d, update x-- and y++; if x < w and y < h and ans[y][x] != -1, set not[ans[y][x]] = true. RESULT not]
            [COMMAND Within the same (i,j) processing, start x = j + d and y = i; while y >= i - d, update x-- and y--; if x < w and y >= 0 and ans[y][x] != -1, set not[ans[y][x]] = true. RESULT not]
            [COMMAND Within the same (i,j) processing, start x = j - d and y = i; while y <= i + d, update x++ and y++; if x >= 0 and y < h and ans[y][x] != -1, set not[ans[y][x]] = true. RESULT not]
            [COMMAND Within the same (i,j) processing, start x = j - d and y = i; while y >= i - d, update x++ and y--; if x >= 0 and y >= 0 and ans[y][x] != -1, set not[ans[y][x]] = true. RESULT not]
            [COMMAND Within the same (i,j) processing, scan z from 0 to 3 in increasing order; if not[z] is false, set ans[i][j] = z and break out of the z loop; if no such z exists, leave ans[i][j] as -1. RESULT ans]
            [COMMAND Define the constant character array c as {'R', 'Y', 'G', 'B'}. RESULT c]
            [COMMAND For i from 0 to h-1 and j from 0 to w-1, call sop(c[ans[i][j] != -1 ? ans[i][j] : 0]); after finishing each row, call sop with a newline string. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: For a cell (i,j), all four colors 0..3 are already marked in not, so the z loop finds no available color.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Leave ans[i][j] as -1; the later output expression treats -1 as index 0, printing 'R'. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: The input stream in throws IOException while reading h, w, or d.]
        [LOG "IOException occurred while reading integer input from in."]
        [THROW IOException "Input reading failed."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: When allocating ans, either h or w is negative.]
        [LOG "NegativeArraySizeException: array dimension is negative."]
        [THROW NegativeArraySizeException "A negative array dimension was supplied to new int[h][w]."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
