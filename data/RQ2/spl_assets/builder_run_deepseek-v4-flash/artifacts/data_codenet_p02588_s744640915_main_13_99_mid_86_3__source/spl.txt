[DEFINE_WORKER: "Reads decimal strings as fixed-point integers scaled by 10^9, validates divisibility by 100000, builds the sorted divisor list of 100000000, and accumulates the number of prior valid divisors that divide the current value using a binary-search-indexed count array, then prints the total." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read a long long integer n from standard input with scanf("%lld", &n). RESULT n]
            [COMMAND For each i from 0 to n-1, read a string s into a char buffer of size 100 with scanf("%s", s), and initialize a[i] to 0 and k to 0. RESULT a]
            [COMMAND Within the same outer loop over i, for each character index j from 0 while s[j] != '\0': if s[j] != '.', set a[i] = 10 * a[i] + (s[j] - '0') and if k > 0 then increment k; if s[j] == '.', set k = 1. RESULT a]
            [COMMAND Within the same outer loop over i, after the character loop, if k == 0 set k = 1; then while k < 10, multiply a[i] by 10 and increment k. RESULT a]
            [COMMAND Set q = 100000, then for each i from 0 to n-1: if a[i] % q > 0, set a[i] = -1; otherwise set a[i] = a[i] / q. RESULT a]
            [COMMAND Set q = 100000000 and reset j to 0. RESULT q]
            [COMMAND Build the sorted divisor list b for q: for i from 1 while i * i <= q, if q % i == 0, set b[j] = i and increment j. RESULT b]
            [COMMAND After the ascending divisor loop, decrement i by 1; if i * i == q, decrement i by 1 again; then for i down to 1, if q % i == 0, set b[j] = q / i and increment j. RESULT b]
            [COMMAND For each i from 0 to j-1, set c[i] = 0. RESULT c]
            [COMMAND Set ans = 0. RESULT ans]
            [COMMAND For each i from 0 to n-1, if a[i] < 0, continue to the next iteration of the outer loop. RESULT i]
            [COMMAND Within the outer loop over i for non-negative a[i], compute l = q / gcd(q, a[i]) using the gcd function. RESULT l]
            [COMMAND Within the same outer loop, perform a lower-bound binary search in b: set min = -1 and max = j; while max - min > 1, set mid = (max + min) / 2; if b[mid] < l, set min = mid, otherwise set max = mid. RESULT max]
            [COMMAND Within the same outer loop, add c[max] to ans. RESULT ans]
            [COMMAND Within the same outer loop, after updating ans, for each k from 0 to j-1, if a[i] % b[k] == 0, increment c[k]. RESULT c]
            [COMMAND Print ans to standard output using printf("%lld\n", ans). RESULT stdout]
            [COMMAND Return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
