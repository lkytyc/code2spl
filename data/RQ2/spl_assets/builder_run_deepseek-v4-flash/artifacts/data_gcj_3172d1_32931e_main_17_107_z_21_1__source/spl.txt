[DEFINE_WORKER: "Processes multiple test cases to determine the minimum number of slicing operations needed to achieve at least d slices of equal size from a set of pancake sizes, using frequency analysis and doubling checks." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command line arguments (unused)"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a BufferedReader to read from standard input, read the first line as an integer t representing the number of test cases, and initialize a StringBuffer to accumulate output. RESULT br, t, sb]
            [COMMAND For each test case z from 1 to t, append the case label 'Case #z: ' to the output buffer. RESULT sb]
            [COMMAND Read the first line of the test case and tokenize it to extract integer n (number of pancake sizes) and integer d (target number of equal slices). RESULT n, d]
            [COMMAND If n equals 1 and d equals 3, append '2' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If n equals 1 and d equals 2, append '1' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND Read the second line of the test case, tokenize it, and parse each token as a long value to populate an array of size n. RESULT array]
            [COMMAND Sort the array in ascending order. RESULT array]
            [COMMAND Create an array of Pair objects to store unique values and their frequencies, then iterate through the sorted array to count consecutive duplicates and create Pair entries for each unique value. RESULT ps, w]
            [COMMAND Sort the Pair array by frequency in ascending order. RESULT ps]
            [COMMAND If the frequency of the first Pair is greater than or equal to d, append '0' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 2 and the frequency of the first Pair is 1, append '1' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the frequency of the first Pair is 2, and its value is not the maximum value in the array, append '1' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the frequency of the first Pair is 2, and its value is the maximum value in the array, check if the frequency of the second Pair is 2; if so, append '1' to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the frequency of the first Pair is 2, and its value is the maximum value in the array, and the second Pair's frequency is not 2, perform a nested loop over all pairs of array elements to check if any element is exactly twice another; if found, append '1' to the output buffer and set a flag, otherwise append '2' to the output buffer, then continue to the next test case. RESULT sb]
            [COMMAND If d equals 3 and the frequency of the first Pair is 1, perform a nested loop over all pairs of array elements to check if any element is exactly twice another; if found, append '1' to the output buffer and set a flag, otherwise append '2' to the output buffer, then continue to the next test case. RESULT sb]
            [COMMAND After processing all test cases, close the BufferedReader and print the accumulated output buffer to standard output. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When n equals 1 and d equals 3]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '2' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When n equals 1 and d equals 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When the frequency of the first Pair is greater than or equal to d]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '0' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 2 and the frequency of the first Pair is 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 2, and its value is not the maximum value in the array]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 2, its value is the maximum value in the array, and the frequency of the second Pair is 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 2, its value is the maximum value in the array, the second Pair's frequency is not 2, and a doubling relationship is found in the array]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 2, its value is the maximum value in the array, the second Pair's frequency is not 2, and no doubling relationship is found in the array]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '2' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 1, and a doubling relationship is found in the array]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the frequency of the first Pair is 1, and no doubling relationship is found in the array]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '2' to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
