[DEFINE_WORKER: "Reads test cases from standard input containing rectangle dimensions and a target perimeter, then for each case appends either the exact target when the original perimeter equals it, the largest achievable perimeter not exceeding the target when all rectangles are identical, or a blank line otherwise, with seven-decimal formatting." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize Scanner in by wrapping System.in with InputStreamReader and BufferedReader. RESULT in]
            [COMMAND Read the next integer token from in into numCases. RESULT numCases]
            [COMMAND Initialize build as an empty StringBuilder. RESULT build]
            [COMMAND Start the outer for loop with int c = 1; continue while c <= numCases; increment c by 1 after each iteration. RESULT _]
            [COMMAND Inside the outer for loop over c, append the formatted case header String.format("Case #%d: ", c) to build. RESULT build]
            [COMMAND Inside the outer for loop over c, read the next integer n from in. RESULT n]
            [COMMAND Inside the outer for loop over c, read the next integer value from in and store it in double p. RESULT p]
            [COMMAND Inside the outer for loop over c, create dims as a double array with n rows and 2 columns. RESULT dims]
            [COMMAND Inside the outer for loop over c, initialize minP to 0.0. RESULT minP]
            [COMMAND Inside the outer for loop over c, start the inner rectangle-reading for loop with int i = 0; continue while i < n; increment i by 1 after each iteration. RESULT _]
            [COMMAND Inside the inner rectangle-reading loop over i, read the next integer from in into first. RESULT first]
            [COMMAND Inside the inner rectangle-reading loop over i, read the next integer from in into second. RESULT second]
            [COMMAND Inside the inner rectangle-reading loop over i, set dims[i][0] = Math.min(first, second) and dims[i][1] = Math.max(first, second). RESULT dims]
            [COMMAND Inside the inner rectangle-reading loop over i, add 2 * (first + second) to minP. RESULT minP]
            [COMMAND After the rectangle-reading loop, initialize boolean small to true. RESULT small]
            [COMMAND Start the labeled smallCheck for loop with int i = 1; continue while i < n; increment i by 1 after each iteration. RESULT _]
            [COMMAND Inside the labeled smallCheck loop, start an inner for loop with int j = 0; continue while j < 2; increment j by 1 after each iteration. RESULT _]
            [COMMAND Inside the inner j loop, if dims[i][j] != dims[0][j], set small to false and break out of the labeled smallCheck loop. RESULT small]
            [COMMAND Inside the outer for loop over c, after the smallCheck loop, if minP == p, append String.format("%.7f\n", p) to build. RESULT build]
            [COMMAND Inside the outer for loop over c, when minP != p, set maxRem = p - minP. RESULT maxRem]
            [COMMAND Inside the outer for loop over c, when minP != p and small is true, begin the identical-rectangles calculation branch. RESULT _]
            [COMMAND In the identical-rectangles branch, initialize minHalf to 0. RESULT minHalf]
            [COMMAND In the identical-rectangles branch, while minHalf < n and maxRem >= 2 * dims[0][0], subtract 2 * dims[0][0] from maxRem and increment minHalf by 1. RESULT maxRem]
            [COMMAND In the identical-rectangles branch, after the minHalf loop, if maxRem == 0, append String.format("%.7f\n", p) to build. RESULT build]
            [COMMAND In the identical-rectangles branch, when maxRem != 0 after the minHalf loop, initialize maxHalf to 0 and compute maxAdd = Math.sqrt(dims[0][0] * dims[0][0] + dims[0][1] * dims[0][1]) - dims[0][0]. RESULT maxHalf]
            [COMMAND In the identical-rectangles branch, while maxHalf < minHalf and maxRem >= 2 * maxAdd, subtract 2 * maxAdd from maxRem and increment maxHalf by 1. RESULT maxRem]
            [COMMAND In the identical-rectangles branch, if maxHalf < minHalf, set maxRem = Math.max(0.0, maxRem - 2 * maxAdd). RESULT maxRem]
            [COMMAND In the identical-rectangles branch, append String.format("%.7f\n", p - maxRem) to build. RESULT build]
            [COMMAND Inside the outer for loop over c, when minP != p and small is false, append a newline character to build using StringBuilder.append("\n"). RESULT build]
            [COMMAND Continue the outer for loop until c exceeds numCases. RESULT _]
            [COMMAND After the outer for loop completes, print the full contents of build to standard output using System.out.printf("%s", build). RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
