[DEFINE_WORKER: "Reads test cases from standard input into sorted TreeSets of words and attempts pairwise longest common substring searches, but the iterator/counter logic limits comparisons and discards every computed result." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not referenced by the method body."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Wrap System.in in an InputStreamReader and BufferedReader, and store the reader in br. RESULT br]
            [COMMAND Read the first line from br and parse it with Integer.parseInt to obtain tc, the number of test cases. RESULT tc]
            [COMMAND Start a for loop with i initialized to 0; continue while i is less than tc; increment i after each test case. RESULT i]
            [COMMAND Read the current test case's word count from br and parse it into noOfWords. RESULT noOfWords]
            [COMMAND Create an empty TreeSet<String> named set for the current test case; this stores words in lexicographic order and removes duplicates. RESULT set]
            [COMMAND For each j from 0 to noOfWords-1, read one word from br and add it to set. RESULT set]
            [COMMAND Create two independent iterators over set, naming them wrd1 and wrd2. RESULT wrd1_wrd2]
            [COMMAND Initialize integer c1 to 0 before entering the outer while loop. RESULT c1]
            [COMMAND While wrd1.hasNext(), increment c1, reset c2 to 0, set wordOne to wrd1.next(), and set result to 0. RESULT wordOne]
            [COMMAND While wrd2.hasNext(), increment c2. If c1 is not equal to c2, call wrd2.next() to get wordTwo and process it; if c1 equals c2, do not advance wrd2 on that iteration, so the intended skip is delayed and self-comparison can still occur. RESULT wordTwo]
            [COMMAND For each wordTwo processed, set max to 0 and key to an empty string. RESULT max_key]
            [COMMAND Within the l loop, set l to every value from 0 through wordTwo.length()-2 inclusive; for each l, set k to every value from l+2 through wordTwo.length() inclusive. RESULT l_k_loops]
            [COMMAND In the innermost loop, if wordOne contains wordTwo.substring(l,k) and k-l is greater than max, update max to k-l and key to the substring; otherwise break out of the k loop immediately. RESULT max_key]
            [COMMAND Because wrd2 is never recreated, the inner wrd2 loop is exhausted during the first outer iteration and is skipped on all subsequent outer iterations. RESULT wrd2]
            [COMMAND Complete the outer test-case loop; computed max and key values are local and never printed, returned, or accumulated, and result remains unused. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: br.readLine() fails because the input stream is closed or an I/O error occurs.]
        [LOG "The IOException is not caught inside main and propagates as declared in the method signature."]
        [THROW IOException "Thrown when reading from System.in fails."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: br.readLine() returns null at end of input, or returns a line that Integer.parseInt cannot parse into an int for tc or noOfWords.]
        [LOG "Integer.parseInt throws NumberFormatException, which is unchecked and propagates from main."]
        [THROW NumberFormatException "Thrown when tc or noOfWords is not a valid integer."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
