[DEFINE_WORKER: "Reads T test cases containing coordinates X, Y and movement string M, builds the time-indexed tourist path on a 2001x2001 grid centered at (1000,1000), uses BFS from (1000-X, 1000+Y) to find the nearest path cell, and prints the earliest interception time for each case or IMPOSSIBLE." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize constant movement arrays dx = {0, 1, 0, -1} and dy = {1, 0, -1, 0}. RESULT dx_dy]
            [COMMAND Create Scanner sc from System.in, create StringBuilder ans, and read integer T = sc.nextInt(). RESULT T_and_ans]
            [COMMAND Loop t from 1 to T inclusive; for each iteration process one test case. RESULT test_case_loop]
            [COMMAND Within the test case loop, read X = sc.nextInt(), Y = sc.nextInt(), and M = sc.next(). RESULT X_Y_M]
            [COMMAND Within the test case loop, allocate int[2001][2001] grid g, set px = 1000 and py = 1000, create empty List<List<Integer>> tour, create List<Integer> te containing [px, py], add te to tour, and set g[py][px] = 1. RESULT path_start]
            [COMMAND Within the test case loop, for i from 0 to M.length() - 1 inclusive, get d = M.charAt(i); if d is 'N' decrement py, if 'S' increment py, if 'E' increment px, if 'W' decrement px; then set g[py][px] = 1 and append a new list containing [px, py] to tour. RESULT path_route]
            [COMMAND Within the test case loop, allocate boolean[2001][2001] visited, compute myx = 1000 - X and myy = 1000 + Y, create LinkedList<Pair> q, offer new Pair(myx, myy, 0), set tmp = -1, nearX = -1, and nearY = -1. RESULT bfs_setup]
            [COMMAND Within the test case loop, while q is not empty, poll Pair now; if g[now.y][now.x] == 1, set tmp = now.dist, nearX = now.x, nearY = now.y and break; otherwise mark visited[now.y][now.x] = true and for v from 0 to 3 compute nx = now.x + dx[v], ny = now.y + dy[v]; if nx < 0 or nx > 2000 or ny < 0 or ny > 2000 or visited[ny][nx] is true, skip; otherwise add new Pair(nx, ny, now.dist + 1) to q. RESULT bfs_result]
            [COMMAND Within the test case loop, append to ans the string formed by "Case #", the current t, and ": ". RESULT case_prefix]
            [COMMAND After the BFS, if tmp is -1, follow the tmp == -1 alternative flow and continue to the next test case; otherwise set ti = 0 and scan tour from index 0 to tour.size() - 1 until the stored coordinates equal nearX and nearY, assign the matching index to ti, and break. RESULT first_path_index]
            [COMMAND Within the test case loop after ti is found, allocate boolean[2001][2001] visited2. RESULT visited2_setup]
            [COMMAND Within the test case loop after allocating visited2, if tmp <= ti follow the tmp <= ti alternative flow; if tmp > ti follow the tmp > ti alternative flow; then continue to the next test case. RESULT branch_after_tmp]
            [COMMAND After the test case loop completes, print the accumulated contents of ans to standard output. RESULT print_result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: tmp == -1 after the BFS, meaning no point on the tourist path is reachable within the bounded grid]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the current test case, append "IMPOSSIBLE" followed by a newline to ans, then continue to the next test case. RESULT impossible_after_bfs]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: tmp <= ti]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the current test case, append tmp + (ti - tmp + 1) / 2 to ans followed by a newline. RESULT early_answer]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: tmp > ti]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the current test case, set success = false; for i from ti while i < tour.size() and tmp - ti + i < tour.size(), get mfx = tour.get(tmp + i - ti).get(0) and mfy = tour.get(tmp + i - ti).get(1), set visited2[mfy][mfx] = true, then get ffx = tour.get(i).get(0) and ffy = tour.get(i).get(1); if visited2[ffy][ffx] is true, append i to ans followed by a newline, set success = true, and break. RESULT late_intercept_loop]
            [COMMAND Within the current test case, after the late interception loop, if success is false, append "IMPOSSIBLE" followed by a newline to ans. RESULT late_impossible]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
