[DEFINE_WORKER: "Reads an n x n board size and n*n cell labels, zeroes the selected board cells, and for each interior cell adds the minimum of the counts of nonzero cells above, below, left, and right, printing the total." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer n from standard input. RESULT n]
            [COMMAND Declare a 501 by 501 integer array a, and declare long variables p, k, and cnt; initialize k to 1 and cnt to 0. RESULT a]
            [COMMAND For each row i from 0 to n-1 and each column j from 0 to n-1, set a[i][j] = k and increment k by 1, filling the board in row-major order with consecutive values starting at 1. RESULT a]
            [COMMAND For each iteration ii from 0 to n*n-1, read a long integer p from standard input. RESULT p]
            [COMMAND Within that loop over ii, if p is between 1 and n inclusive, set a[0][p-1] = 0. RESULT a]
            [COMMAND Within that loop over ii, else if p is between n*n-n+1 and n*n inclusive, set a[n-1][n-n*n+p-1] = 0. RESULT a]
            [COMMAND Within that loop over ii, else if p is divisible by n (p % n == 0), set a[(p/n)-1][n-1] = 0. RESULT a]
            [COMMAND Within that loop over ii, else if p % n == 1, set a[p/n][0] = 0. RESULT a]
            [COMMAND Within that loop over ii, in the final otherwise branch, set a[p/n][(p%n)-1] = 0, let i = p/n, let j = (p%n)-1, and initialize long m[5] = {i, n-1-i, j, n-1-j}. RESULT m]
            [COMMAND Within the final otherwise branch, for jj from 0 to i-1, decrement m[0] whenever a[jj][j] == 0, so m[0] becomes the number of nonzero cells above (i,j). RESULT m]
            [COMMAND Within the final otherwise branch, for jj from n-1 down to i+1, decrement m[1] whenever a[jj][j] == 0, so m[1] becomes the number of nonzero cells below (i,j). RESULT m]
            [COMMAND Within the final otherwise branch, for jj from 0 to j-1, decrement m[2] whenever a[i][jj] == 0, so m[2] becomes the number of nonzero cells to the left of (i,j). RESULT m]
            [COMMAND Within the final otherwise branch, for jj from n-1 down to j+1, decrement m[3] whenever a[i][jj] == 0, so m[3] becomes the number of nonzero cells to the right of (i,j). RESULT m]
            [COMMAND Within the final otherwise branch, sort the first four elements of m in ascending order with qsort(m, 4, sizeof(long), compare_int). RESULT m]
            [COMMAND Within the final otherwise branch, assign mm = m[0] and add mm to cnt. RESULT cnt]
            [COMMAND After completing the loop over ii, print cnt as a long integer followed by a newline. RESULT standard_output]
            [COMMAND Return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
