[DEFINE_WORKER: "Reads graph input, initializes vertex and edge arrays, calls route up to three times to accumulate positive flow values, prints the accumulated flow on early failure, and prints an error marker if all three calls succeed." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_code </REF>: int "Always 0; either the accumulated result is printed before early return, or 'er' is printed after three successful route calls."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers from standard input into V and E using scanf. RESULT V_and_E]
            [COMMAND For i from 0 to V-1 inclusive, set v[i].num to 0. RESULT v]
            [COMMAND Set cmax to 0. RESULT cmax]
            [COMMAND For each edge index i from 0 to E-1 inclusive, read three integers a, b, c from standard input. RESULT a_b_c]
            [COMMAND For each edge i, store the edge endpoints by setting e[i].n[0]=a and e[i].n[1]=b. RESULT e]
            [COMMAND For each edge i, append edge index i to v[a].e at position v[a].num and increment v[a].num, then append edge index i to v[b].e at position v[b].num and increment v[b].num. RESULT v]
            [COMMAND For each edge i, set e[i].fmax=c and e[i].flow=0. RESULT e]
            [COMMAND For each edge i, when b equals V-1, swap the newly appended edge index at v[a].e[v[a].num-1] with v[a].e[0], then add c to cmax. RESULT cmax]
            [COMMAND Set res to 0. RESULT res]
            [COMMAND For i from 0 to 99 inclusive, set et[i] to 0. RESULT et]
            [COMMAND For iteration i from 0 to 2 inclusive, call route(0, V-1, cmax) and store the return value in x. RESULT x]
            [COMMAND In each route iteration, if x is less than or equal to 0, follow the early-return alternative; otherwise add x to res. RESULT res]
            [COMMAND After all three route iterations complete, print the text 'er' followed by a newline. RESULT printed_output]
            [COMMAND Return 0 from main. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: In any of the three route iterations, route returns a value x that is less than or equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the current value of res followed by a newline. RESULT printed_output]
            [COMMAND Return 0 from main immediately. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
