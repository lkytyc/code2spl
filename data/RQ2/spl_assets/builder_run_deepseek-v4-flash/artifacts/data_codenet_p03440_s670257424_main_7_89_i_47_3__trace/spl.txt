[DEFINE_WORKER: "Reads an array and a series of union operations, counts the resulting union-find components, and either prints an early result (0, Impossible, or total sum) or computes the minimal sum of component-minimum values plus the smallest remaining array values after the required boundary condition checks." main]
    [INPUTS]
        <REF> stdin </REF>: standard input "First provides two integers n and m, then n integers for array a, then m pairs of integers x and y for union operations."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: text "Prints either 0, Impossible, an integer sum, or a long long result depending on the computed number of components and the subsequent cost calculation."
        <REF> return_value </REF>: int "Returns 0 in every code path."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers n and m from stdin using scanf("%d%d", &n, &m). RESULT n_m_values]
            [COMMAND Declare integer arrays uni and a of length n, plus integer variables x and y for the union operations. RESULT arrays_ready]
            [COMMAND For i from 0 to n-1 inclusive, read an integer from stdin into a[i] using scanf("%d", &a[i]). RESULT a_loaded]
            [COMMAND For i from 0 to n-1 inclusive, set uni[i] = i. RESULT uni_initialized]
            [COMMAND For operation_index from 0 to m-1 inclusive, read x and y from stdin, then perform the union operation: if uni[x] != x, save saved_parent = uni[y], set uni[y] = x, follow parent pointers starting with j = y while uni[j] != j by setting j = uni[j], set uni[y] = j, then for every j from 0 to n-1 except j == y, if uni[j] == saved_parent, set uni[j] = uni[y]; otherwise, when uni[x] == x, save saved_parent = uni[x], set uni[x] = y, follow parent pointers starting with j = x while uni[j] != j by setting j = uni[j], set uni[x] = j, then for every j from 0 to n-1 except j == x, if uni[j] == saved_parent, set uni[j] = uni[x]. RESULT uni_after_operations]
            [COMMAND Count the number of roots num by iterating i from 0 to n-1 and incrementing num whenever uni[i] == i. RESULT num_roots]
            [COMMAND Evaluate the early-exit conditions in order: if num == 1, then if num * 2 - 2 > n, then if num * 2 - 2 == n; the corresponding alternative_flows handle those returning cases, and only if none of them is true continue with the general computation. RESULT general_case_reached]
            [COMMAND Declare integer arrays min and minn of length n. RESULT min_arrays_ready]
            [COMMAND For i from 0 to n-1 inclusive, set min[i] = INF, where INF is a large sentinel value. RESULT min_initialized]
            [COMMAND For i from 0 to n-1 inclusive, if min[uni[i]] > a[i], set minn[uni[i]] = i and min[uni[i]] = a[i]. RESULT component_mins_recorded]
            [COMMAND Initialize long long ssum = 0 and int tnum = num - 2. RESULT ssum_tnum_initialized]
            [COMMAND For i from 0 to n-1 inclusive, set local variable y = INF; if min[i] != y, add (long long)min[i] to ssum and set a[minn[i]] = INF. RESULT ssum_base_cost_and_inf_markers]
            [COMMAND Sort array a in place using qsort(a, n, sizeof(int), compare) with the comparator function compare. RESULT a_sorted]
            [COMMAND For i from 0 to tnum-1 inclusive, add a[i] to ssum. RESULT ssum_final_cost]
            [COMMAND Print ssum as a long long using printf("%lld", ssum) and return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: num == 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the single character 0 to stdout. RESULT early_return_0]
            [COMMAND Return 0. RESULT return_0_alt1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: num * 2 - 2 > n]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the exact text Impossible to stdout. RESULT early_return_impossible]
            [COMMAND Return 0. RESULT return_0_alt2]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: num * 2 - 2 == n]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize int t = 0; for i from 0 to n-1 inclusive, add a[i] to t. RESULT total_sum_a]
            [COMMAND Print t as an integer using printf("%d", t). RESULT early_return_sum]
            [COMMAND Return 0. RESULT return_0_alt3]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
