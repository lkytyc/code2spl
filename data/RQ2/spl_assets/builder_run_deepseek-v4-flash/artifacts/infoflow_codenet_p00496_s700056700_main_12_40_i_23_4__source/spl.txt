[DEFINE_WORKER: "Reads n, t, s and n item pairs into a and b, then computes a dynamic-programming maximum over the dp table using a guarded recurrence and prints the resulting maximum." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Call fgets(p=buf, 30, stdin), assigning p to buf and reading at most 29 characters from stdin into buf; ignore the return value. RESULT buf]
            [COMMAND Call getint() to parse the next integer, assign it to n, then increment p. RESULT n]
            [COMMAND Call getint() to parse the next integer, assign it to t, then increment p. RESULT t]
            [COMMAND Call getint() to parse the next integer and assign it to s. RESULT s]
            [COMMAND For each i from 1 to n inclusive, call fgets(p=buf, 30, stdin) so p is reset to buf and at most 29 characters are read into buf; ignore the return value, set a[i] to getint(), increment p, then set b[i] to getint(). RESULT a,b]
            [COMMAND Set ans to 0. RESULT ans]
            [COMMAND Using the existing boundary values in dp[0][*] and dp[*][0], for each i from 1 to n inclusive, for each j from 1 to t inclusive, set dp[i][j] to max(dp[i-1][j], dp[i][j-1]). RESULT dp]
            [COMMAND While still inside the outer i loop and inner j loop, set k to j minus b[i]. RESULT k]
            [COMMAND While still inside the outer i loop and inner j loop, if k is greater than or equal to 0 and (s is less than or equal to k or j is less than or equal to s), using short-circuit evaluation, set dp[i][j] to max(dp[i][j], dp[i-1][k] + a[i]). RESULT dp]
            [COMMAND While still inside the outer i loop and inner j loop, set ans to max(ans, dp[i][j]). RESULT ans]
            [COMMAND Print ans as a decimal integer followed by a newline. RESULT stdout]
            [COMMAND Return 0 from main. RESULT exit_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
