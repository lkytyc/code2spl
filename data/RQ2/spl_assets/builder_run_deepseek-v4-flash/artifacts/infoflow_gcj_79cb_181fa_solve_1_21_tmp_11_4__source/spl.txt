[DEFINE_WORKER: "Sorts the even-indexed and odd-indexed subsequences of L separately via repeated same-parity swaps, then checks only the final index i from the sorting loop against the fully sorted list and returns that mismatched index, or -1 if no mismatch is encountered." solve]
    [INPUTS]
        <REF> L </REF>: list "Input list of orderable elements; it is modified in place by the same-parity sorting loop."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns the index i when L[i] differs from sorted_list[i] during the final loop, or -1 if that loop completes without a mismatch."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute sorted_list as the ascending fully sorted copy of L. RESULT sorted_list]
            [COMMAND Initialize done to False. RESULT done]
            [COMMAND Initialize lengthList to len(L). RESULT lengthList]
            [COMMAND While done is False, set done to True and i to 0; then while i is less than lengthList - 2, if L[i] is greater than L[i+2], set done to False and swap L[i] with L[i+2]. Increment i by 1 after every comparison. RESULT L]
            [COMMAND After the sorting loop ends, initialize c to 0. RESULT c]
            [COMMAND While c is less than lengthList, assign expected = sorted_list[i] using the same i left from the sorting loop; if L[i] does not equal expected, return i immediately; otherwise increment c by 1. RESULT return_value]
            [COMMAND If the second while loop completes without returning, return -1. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
