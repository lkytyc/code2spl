[DEFINE_WORKER: "Processes q operations that either insert a value with a weight into a sorted array or output the lower median and the sum of absolute deviations from that median plus the total inserted weight." main]
    [INPUTS]
        <REF> stdin </REF>: standard input "First an integer q, followed by q operations. Each operation begins with an integer x. If x is 1, the operation also contains y and z."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Always returns 0 to indicate successful execution."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare all local variables: q (int), n initialized to 0 (int), a (long long int array of 200005 elements), b initialized to 0 (long long int), x, y, z, m, ans (long long int), and i, j (int). RESULT local_state]
            [COMMAND Read the number of operations q from standard input using scanf("%d", &q). RESULT q]
            [COMMAND For each i from 0 to q-1, read the next operation code into x and dispatch to the insertion branch when x equals 1, otherwise dispatch to the query branch. RESULT branch_selected]
            [COMMAND After the loop finishes, return 0 from main. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When x is 1, the insertion branch is selected.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the x == 1 insertion branch and the outer loop over i, read y and z from standard input using scanf("%lld %lld", &y, &z). RESULT y,z]
            [COMMAND If n is 0, set a[0] to y. RESULT a[0]]
            [COMMAND Otherwise, insert y into a while preserving ascending order: scan j downward from n-1 to 0; if a[j] > y, shift a[j] to a[j+1] and if j is 0 set a[0] to y; if a[j] <= y, set a[j+1] to y and break out of the scan. RESULT a]
            [COMMAND Increment n by 1 to reflect the newly inserted element. RESULT n]
            [COMMAND Add z to b, the total accumulated weight. RESULT b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When x is not 1, the query branch is selected.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Within the x != 1 query branch and the outer loop over i, compute m as the lower median of the sorted array: m = a[(n - 1) / 2]. RESULT m]
            [COMMAND Set ans to the current accumulated weight b. RESULT ans]
            [COMMAND For each j from 0 to n-1, add the absolute difference between m and a[j] to ans: if m - a[j] < 0, add -(m - a[j]); otherwise add m - a[j]. RESULT ans]
            [COMMAND Print m and ans to standard output on one line separated by a space using printf("%lld %lld\n", m, ans). RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
