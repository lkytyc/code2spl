[DEFINE_WORKER: "Solves a multi-player rock-paper-scissors elimination puzzle by repeatedly scanning alive players' cyclic gestures, appending a move that either immediately wins or eliminates all players currently showing one beaten gesture, and printing the final move sequence or 'IMPOSSIBLE' when no valid sequence of at most 500 moves exists." solve]
    [INPUTS]
        <REF> A </REF>: int "Number of players; used to create the isAlive list of length A."
        <REF> C </REF>: list[str] "List of gesture strings; C[i] is the cyclic gesture pattern for player i, using characters R, S, and P."
        <REF> L </REF>: list[int] "List of cycle lengths; L[i] is the modulus used to select C[i][pos % L[i]] at position pos."
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_output </REF>: string "The function prints to stdout either the accumulated ans move sequence or the literal string 'IMPOSSIBLE'; it returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ans to an empty string. RESULT ans]
            [COMMAND Initialize pos to 0. RESULT pos]
            [COMMAND Initialize isAlive to a list of length A where every element is True. RESULT isAlive]
            [COMMAND Enter an infinite while loop that repeats the following steps until a return is executed. RESULT _]
            [COMMAND At the start of each loop iteration, reset possibilities to {'R': 0, 'S': 0, 'P': 0} and reset predR, predS, and predP to empty lists. RESULT possibilities]
            [COMMAND If debug is truthy, print the current isAlive list and ans. RESULT _]
            [COMMAND If no element of isAlive is True, print ans and return from the function. RESULT _]
            [COMMAND For each player index i from 0 to A-1 inclusive, if isAlive[i] is True, execute the following player scan steps. RESULT _]
            [COMMAND Within the player scan, compute gesture = C[i][pos % L[i]] and set possibilities[gesture] to 1. RESULT gesture]
            [COMMAND Within the player scan, if gesture is 'R', append i to predR; if gesture is 'S', append i to predS; if gesture is 'P', append i to predP. RESULT predR]
            [COMMAND Within the player scan, if possibilities['R'] plus possibilities['S'] plus possibilities['P'] equals 3, print 'IMPOSSIBLE' and return from the function. RESULT _]
            [COMMAND After the player scan completes, assign r = possibilities['R'], s = possibilities['S'], and p = possibilities['P']. RESULT r]
            [COMMAND If debug is truthy, print the tuple (r, s, p). RESULT _]
            [COMMAND If (r, s, p) equals (1, 0, 0), append 'P' to ans, print ans, and return from the function. RESULT ans]
            [COMMAND If (r, s, p) equals (0, 1, 0), append 'R' to ans, print ans, and return from the function. RESULT ans]
            [COMMAND If (r, s, p) equals (0, 0, 1), append 'S' to ans, print ans, and return from the function. RESULT ans]
            [COMMAND If len(ans) equals 500, print 'IMPOSSIBLE'; if debug is truthy, print 'LEN' and ans; then return from the function. RESULT _]
            [COMMAND If (r, s, p) equals (1, 1, 0), set isAlive[i] to False for every i in predS, then append 'R' to ans. RESULT ans]
            [COMMAND If (r, s, p) equals (1, 0, 1), set isAlive[i] to False for every i in predR, then append 'P' to ans. RESULT ans]
            [COMMAND If (r, s, p) equals (0, 1, 1), set isAlive[i] to False for every i in predP, then append 'S' to ans. RESULT ans]
            [COMMAND Increment pos by 1 and continue to the next iteration of the infinite while loop. RESULT pos]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
