[DEFINE_WORKER: "Reads t chocolate-chip grid test cases and prints whether each grid can be divided into (h+1)*(v+1) equal-chip portions, using prefix sums and cut validation." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Returns 0 on successful completion after processing all test cases."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer t from standard input into variable t. RESULT t]
            [COMMAND Set test-case counter tc to 1. RESULT tc]
            [COMMAND Enter the outer test-case loop that runs while tc <= t (inclusive). RESULT test_case_loop]
            [COMMAND Inside the current test case, read integers r, c, h, and v from standard input into variables r, c, h, and v. RESULT dimensions]
            [COMMAND Compute guests = (h + 1) * (v + 1) and set chips = 0. RESULT guests_chips]
            [COMMAND For each row i from 0 to r inclusive, if i != 0 read the next row string into str; then for each column j from 0 to c inclusive, if i == 0 or j == 0 set A[i][j] = B[i][j] = 0, otherwise set A[i][j] = 1 when str[j-1] == '@' (else 0) and increment chips whenever A[i][j] is true. RESULT A_B_grid]
            [COMMAND If chips % guests != 0, print 'Case #tc: IMPOSSIBLE' with tc substituted for %d and do not execute the prefix-sum/search else branch. RESULT divisibility_impossible]
            [COMMAND Otherwise, build the 2D prefix-sum matrix B: for i from 1 to r and for j from 1 to c, set B[i][j] = A[i][j] + B[i-1][j] + B[i][j-1] - B[i-1][j-1]. RESULT prefix_sum_B]
            [COMMAND Set share = chips / guests and possible = 0. RESULT share_possible]
            [COMMAND Search candidate cut intersections with short-circuiting: for i from 1 to r-1 while possible is false, and for j from 1 to c-1 while possible is false, if B[i][j] == share then set possible = 1; if fr(h-1, i, j) returns false set possible = 0; if fc(v-1, i, j) returns false set possible = 0; if possible is still true print 'Case #tc: POSSIBLE' with tc substituted for %d. RESULT candidate_search]
            [COMMAND After the candidate search, if possible is false print 'Case #tc: IMPOSSIBLE' with tc substituted for %d. RESULT impossible_after_search]
            [COMMAND At the end of the current test-case iteration, increment tc by 1. RESULT tc_increment]
            [COMMAND After the outer test-case loop ends, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
