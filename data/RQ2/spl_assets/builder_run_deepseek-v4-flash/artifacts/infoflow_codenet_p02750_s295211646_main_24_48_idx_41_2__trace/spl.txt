[DEFINE_WORKER: "Reads N and T, loads N time objects into an array, sorts them, then processes the sorted array while T is positive and idx is in range, counting iterations with one extra count at the last index, then decrements and prints the count." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner sc bound to System.in. RESULT sc]
            [COMMAND Read the next int from sc into N. RESULT N]
            [COMMAND Read the next long from sc into T. RESULT T]
            [COMMAND Allocate a time array ROOT of length N. RESULT ROOT]
            [COMMAND For each i from 0 to N-1 inclusive, create a new time object tm, read sc.nextLong() into tm.a, read sc.nextLong() into tm.b, and assign ROOT[i] = tm. RESULT ROOT]
            [COMMAND Sort the ROOT array in place using time's natural ordering. RESULT ROOT]
            [COMMAND Initialize idx to 0 and ans to 0. RESULT idx, ans]
            [COMMAND Evaluate the while condition T > 0 && idx < N using short-circuit evaluation; continue looping only while both are true. RESULT while_condition]
            [COMMAND In each while iteration, update T = (T - ROOT[idx].b) / (ROOT[idx].a + 1) using long division, then increment ans by 1 and increment idx by 1. RESULT T, ans, idx]
            [COMMAND In each while iteration, after idx has been incremented, if idx == N - 1, increment ans by 1. RESULT ans]
            [COMMAND After the while loop exits, decrement ans by 1. RESULT ans]
            [COMMAND Print the final value of ans to System.out. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
