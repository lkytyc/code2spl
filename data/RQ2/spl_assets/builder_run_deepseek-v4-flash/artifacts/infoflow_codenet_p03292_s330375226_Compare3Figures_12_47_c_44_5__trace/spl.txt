[DEFINE_WORKER: "Arranges a, b, and c into ascending order via strict chained comparisons when possible, then calls MakeCostMinimum on the resulting values." Compare3Figures]
    [INPUTS]
        <REF> a </REF>: int "First integer value to be arranged; sorted in place."
        <REF> b </REF>: int "Second integer value to be arranged; sorted in place."
        <REF> c </REF>: int "Third integer value to be arranged; sorted in place."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Sort the current values of a, b, and c into ascending order by evaluating the chained strict-comparison branches and performing the selected swap or rotation; if no branch condition matches, leave a, b, and c unchanged. RESULT sorted_a_b_c]
            [COMMAND Call MakeCostMinimum with the resulting a, b, and c values. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b < a && a < c]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set tmp to a, assign b to a, and assign tmp to b. RESULT a_b_swapped]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No earlier branch matched and b < a && c < b]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set tmptwo to a, assign c to a, and assign tmptwo to c. RESULT a_c_swapped]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No earlier branch matched and a < b && c < b && a < c]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set tmpthree to b, assign c to b, and assign tmpthree to c. RESULT b_c_swapped]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No earlier branch matched and a < b && c < a]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set tmpfour to a, assign c to a, assign b to c, and assign tmpfour to b. RESULT a_b_c_rotated]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No earlier branch matched and b < a && b < c && c < a]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set tmpfive to a, assign b to a, assign c to b, and assign tmpfive to c. RESULT a_b_c_rotated]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
