[DEFINE_WORKER: "Returns a direction string ('E'/'W'/'N'/'S') representing a valid sequence of jumps from the origin to target (x, y), using parity and conflict-free power-of-two masks to choose the path, or returns 'IMPOSSIBLE' when no such sequence exists." solve]
    [INPUTS]
        <REF> x </REF>: integer "Target x-coordinate; may be negative."
        <REF> y </REF>: integer "Target y-coordinate; may be negative."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Concatenated direction string using 'E', 'W', 'N', and 'S', or the literal string 'IMPOSSIBLE'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Save the original coordinates as origX = x and origY = y, then set x = abs(origX) and y = abs(origY) using max(value, -value). RESULT origX, origY, x, y]
            [COMMAND If x and y have the same parity (x % 2 == y % 2), return the literal string 'IMPOSSIBLE' immediately; otherwise continue. RESULT early_return]
            [COMMAND Set x1 = 1, then double x1 while x1 < x, so x1 becomes the smallest power of two that is at least x. Compute x2 = x1 - x. RESULT x1, x2]
            [COMMAND Set y1 = 1, then double y1 while y1 < y, so y1 becomes the smallest power of two that is at least y. Compute y2 = y1 - y. RESULT y1, y2]
            [COMMAND Evaluate the four branch alternatives in order (see alternative_flows) and populate the result list using the first branch whose conflict checks all pass. If no branch passes, return the literal string 'IMPOSSIBLE'. RESULT result]
            [COMMAND If x != origX, iterate over every index from 0 to len(result)-1 and swap each 'W' to 'E' and each 'E' to 'W'. If y != origY, similarly swap each 'N' to 'S' and each 'S' to 'N'. RESULT result]
            [COMMAND Return the string produced by concatenating all characters in result with no separators. RESULT final_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: First branch: not areInConflict(x, y).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create bx as the reversed binary string of x and by as the reversed binary string of y. RESULT bx, by]
            [COMMAND Create result as a list of None values with length max(len(bx), len(by)). RESULT result]
            [COMMAND For each index i from 0 to len(bx)-1, if bx[i] == '1', set result[i] = 'E'. RESULT result]
            [COMMAND For each index i from 0 to len(by)-1, if by[i] == '1', set result[i] = 'N'. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Second branch: not areInConflict(x2, x1) and not areInConflict(x2, y) and not areInConflict(x1, y).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create bx1, bx2, and by as the reversed binary strings of x1, x2, and y respectively. RESULT bx1, bx2, by]
            [COMMAND Create result as a list of None values with length max(len(bx1), len(bx2), len(by)). RESULT result]
            [COMMAND For each index i from 0 to len(bx1)-1, if bx1[i] == '1', set result[i] = 'E'. RESULT result]
            [COMMAND For each index i from 0 to len(bx2)-1, if bx2[i] == '1', set result[i] = 'W'. RESULT result]
            [COMMAND For each index i from 0 to len(by)-1, if by[i] == '1', set result[i] = 'N'. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Third branch: not areInConflict(y2, y1) and not areInConflict(y2, x) and not areInConflict(y1, x).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create by1, by2, and bx as the reversed binary strings of y1, y2, and x respectively. RESULT by1, by2, bx]
            [COMMAND Create result as a list of None values with length max(len(by1), len(by2), len(bx)). RESULT result]
            [COMMAND For each index i from 0 to len(bx)-1, if bx[i] == '1', set result[i] = 'E'. RESULT result]
            [COMMAND For each index i from 0 to len(by1)-1, if by1[i] == '1', set result[i] = 'N'. RESULT result]
            [COMMAND For each index i from 0 to len(by2)-1, if by2[i] == '1', set result[i] = 'S'. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Fourth branch: not areInConflict(y2, y1) and not areInConflict(y2, x1) and not areInConflict(y2, x2) and not areInConflict(y1, x1) and not areInConflict(y1, x2) and not areInConflict(x2, x1).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create bx1, bx2, by1, and by2 as the reversed binary strings of x1, x2, y1, and y2 respectively. RESULT bx1, bx2, by1, by2]
            [COMMAND Attempt to create result as a list of None values with length max(len(by1), len(by2), len(bx)); because bx is undefined in this branch, this raises NameError and execution stops before the original assignment loops run. RESULT NameError]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: No branch matches: all four if/elif conditions evaluate to false.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the literal string 'IMPOSSIBLE'. RESULT early_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
