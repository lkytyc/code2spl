[DEFINE_WORKER: "Binary-searches a repetition parameter m using double modular rolling hashes to determine whether a hash derived from the mapped input t matches a hash derived from a cyclic bucketed copy of mapped input s, then prints -1 when the sentinel l == ns + 1 is reached." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_value </REF>: integer "The integer printed by the method: l when l != ns + 1, otherwise -1."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set mod = 1000000007, read two input sequences s and t using S(), compute ns = len(s) and nt = len(t), and initialize f as an empty dictionary. RESULT mod, s, t, ns, nt, f]
            [COMMAND For i from 0 to 25 inclusive, set f[chr(i + ord('a'))] = i + 1. Then for each index i from 0 to ns - 1 replace s[i] with f[s[i]], and for each index i from 0 to nt - 1 replace t[i] with f[t[i]]. RESULT f, s, t]
            [COMMAND Set b = 10, mod1 = 1000000007, mod2 = 1000000009. Compute pownt1 = pow(b, nt, mod), invnt1 = pow(pownt1, mod1 - 2, mod1), powns1 = pow(b, ns, mod1), invns1 = pow(powns1, mod1 - 2, mod1), pownt2 = pow(b, nt, mod2), invnt2 = pow(pownt2, mod2 - 2, mod2), powns2 = pow(b, ns, mod2), invns2 = pow(powns2, mod2 - 2, mod2). Initialize l = 0 and r = ns + 2. RESULT b, mod1, mod2, pownt1, invnt1, powns1, invns1, pownt2, invnt2, powns2, invns2, l, r]
            [COMMAND While r > l + 1, compute m = (l + r) >> 1, n = m * nt, st = [m] * nt, ms = n // ns, rs = n - ms * ns, and build ss by appending ms + 1 for the first rs positions and ms for the remaining positions. Compute p1 = pow(b, n, mod1) and p2 = pow(b, n, mod2), and initialize h1 = 0 and h2 = 0. RESULT m, n, st, ms, rs, ss, p1, p2, h1, h2]
            [COMMAND Inside the while loop, compute the target hash h1, h2 from t: for each i from 0 to nt - 1, set ti = t[i], pow1 = pow(b, (n - i) * mod1, mod1), pows1 = pow(pownt1, st[i], mod1), update h1 = (h1 + ti * pow1 * (pows1 - 1) * pow(pows1 * (1 - invnt1), mod1 - 2, mod1)) % mod1; then set pow2 = pow(b, (n - i) * mod2, mod2), pows2 = pow(pownt2, st[i], mod2), update h2 = (h2 + ti * pow2 * (pows2 - 1) * pow(pows2 * (1 - invnt2), mod2 - 2, mod2)) % mod2. RESULT h1, h2]
            [COMMAND Inside the while loop, compute the candidate hash k1, k2 from s: for each i from 0 to ns - 1, break early if ss[i] is 0; otherwise set si = s[i], pow1 = pow(b, (n - i) * mod1, mod1), pows1 = pow(powns1, ss[i], mod1), update k1 = (k1 + si * pow1 * (pows1 - 1) * pow(pows1 * (1 - invns1), mod1 - 2, mod1)) % mod1; then set pow2 = pow(b, (n - i) * mod2, mod2), pows2 = pow(powns2, ss[i], mod2), update k2 = (k2 + si * pow2 * (pows2 - 1) * pow(pows2 * (1 - invns2), mod2 - 2, mod2)) % mod2. RESULT k1, k2]
            [COMMAND Inside the while loop, set k = 0. For j from rs to ns + rs - 1, check if (h1, h2) equals (k1, k2); if it does, set l = m and break out of this for loop. Otherwise set i = j % ns, si = s[i], sk = s[k], update k1 = ((k1 - (sk * p1 % mod1)) + si) * b % mod1 and k2 = ((k2 - (sk * p2 % mod2)) + si) * b % mod2, and increment k. If this for loop completes without breaking, set r = m. RESULT l, r]
            [COMMAND After the while loop exits, if l == ns + 1 then print -1; otherwise print l. RESULT printed_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
