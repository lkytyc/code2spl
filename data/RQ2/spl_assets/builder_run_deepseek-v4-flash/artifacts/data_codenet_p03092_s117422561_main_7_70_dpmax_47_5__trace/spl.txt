[DEFINE_WORKER: "Reads N, A, B, and the array p, then uses dynamic programming over increasing interval lengths with min/max and second-min/max summaries to compute and print the minimum cost dpc[0][N-1]." main]
    [INPUTS]
        <REF> sc </REF>: Scanner "Scanner bound to the input source; reads N as int, A and B as longs, then N int values into p."
        <REF> args </REF>: String[] "Command-line arguments passed to main; unused by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read N as an int from sc.nextInt(), read A and B as longs from sc.nextLong(), and allocate int[] p of length N with default zero elements. RESULT input_vars]
            [COMMAND For each i from 0 while i < N with i++ (i = 0..N-1 when N > 0), read the next int from sc and store it in p[i]. RESULT p]
            [COMMAND Allocate long[][] dpc and int[][] dpmin, dpmin2, dpmax, dpmax2, all N by N with default zero values. RESULT dp_arrays]
            [COMMAND For each i from 0 while i < N with i++, initialize diagonal cells: dpc[i][i]=0, dpmin[i][i]=p[i], dpmax[i][i]=p[i], dpmin2[i][i]=Integer.MAX_VALUE, dpmax2[i][i]=-1. RESULT diagonal_state]
            [COMMAND Within the outer loop for d=1; d<N; d++ and inner loop for i=0; i<N-d; i++, update the two smallest summaries for interval [i,i+d] by inserting p[i+d] into the two-min summary of [i,i+d-1]: if p[i+d] is strictly less than dpmin[i][i+d-1], set dpmin2[i][i+d]=dpmin[i][i+d-1] and dpmin[i][i+d]=p[i+d]; else if p[i+d] is strictly less than dpmin2[i][i+d-1], set dpmin2[i][i+d]=p[i+d] and dpmin[i][i+d]=dpmin[i][i+d-1]; otherwise set dpmin2[i][i+d]=dpmin2[i][i+d-1] and dpmin[i][i+d]=dpmin[i][i+d-1]. RESULT min_summaries]
            [COMMAND Within the same nested loops, update the two largest summaries for interval [i,i+d] by inserting p[i+d] into the two-max summary of [i,i+d-1]: if p[i+d] is strictly greater than dpmax[i][i+d-1], set dpmax2[i][i+d]=dpmax[i][i+d-1] and dpmax[i][i+d]=p[i+d]; else if p[i+d] is strictly greater than dpmax2[i][i+d-1], set dpmax2[i][i+d]=p[i+d] and dpmax[i][i+d]=dpmax[i][i+d-1]; otherwise set dpmax2[i][i+d]=dpmax2[i][i+d-1] and dpmax[i][i+d]=dpmax[i][i+d-1]. RESULT max_summaries]
            [COMMAND Within the same nested loops, compute long rightcost initialized to 0 from the old right interval [i,i+d-1]: if p[i+d] > dpmax[i][i+d-1], set rightcost=dpc[i][i+d-1]; else if p[i+d] > dpmax2[i][i+d-1], set rightcost=dpc[i][i+d-1]+Math.min(A,B); else set rightcost=dpc[i][i+d-1]+B. RESULT rightcost]
            [COMMAND Within the same nested loops, compute long leftcost initialized to 0 from the adjacent left interval [i+1,i+d]: if p[i] < dpmin[i+1][i+d], set leftcost=dpc[i+1][i+d]; else if p[i] < dpmin2[i+1][i+d], set leftcost=dpc[i+1][i+d]+Math.min(A,B); else set leftcost=dpc[i+1][i+d]+A. RESULT leftcost]
            [COMMAND Within the same nested loops, set dpc[i][i+d]=Math.min(leftcost,rightcost), then call debug(i, i+d, leftcost, rightcost, dpmin[i][i+d], dpmin2[i][i+d], dpmax[i][i+d], dpmax2[i][i+d]). RESULT dpc]
            [COMMAND After all interval lengths have been processed, print dpc[0][N-1] with System.out.println. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: N < 0 when new int[N] is evaluated]
        [LOG "Cannot create an array with a negative size."]
        [THROW NegativeArraySizeException "Thrown because N is negative and the array length cannot be negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: N == 0 when System.out.println(dpc[0][N-1]) is reached]
        [LOG "dpc has zero rows because N is zero, so dpc[0] is out of bounds."]
        [THROW ArrayIndexOutOfBoundsException "Thrown because dpc[0] does not exist or index -1 is invalid when N=0."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
