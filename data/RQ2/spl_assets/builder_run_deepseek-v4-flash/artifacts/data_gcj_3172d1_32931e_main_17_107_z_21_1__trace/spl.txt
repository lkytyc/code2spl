[DEFINE_WORKER: "Reads multiple test cases, each containing a list of pancake diameters and a target number of equal slices, and determines the minimum number of cuts required to obtain at least that many equal-sized slices." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments (unused)"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a BufferedReader wrapping System.in to read input. RESULT br]
            [COMMAND Read the first line from the buffered reader, parse it as an integer, and store it in variable t representing the number of test cases. RESULT t]
            [COMMAND Create a StringBuffer named sb to accumulate the output for all test cases. RESULT sb]
            [COMMAND Start an outer for loop with integer z initialized to 1, continuing while z is less than or equal to t, incrementing z by 1 after each iteration. RESULT loop_z]
            [COMMAND Within the outer for loop over z, append the string "Case #" concatenated with the value of z and ": " to the StringBuffer sb. RESULT sb]
            [COMMAND Within the outer for loop over z, read the next line from the buffered reader and create a StringTokenizer named st from it. RESULT st]
            [COMMAND Within the outer for loop over z, read the next line from the buffered reader and create a StringTokenizer named st1 from it. RESULT st1]
            [COMMAND Within the outer for loop over z, parse the next token from st as an integer and store it in variable n. RESULT n]
            [COMMAND Within the outer for loop over z, parse the next token from st as an integer and store it in variable d. RESULT d]
            [COMMAND Within the outer for loop over z, check if n equals 1 and d equals 3. If true, append "2\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, check if n equals 1 and d equals 2. If true, append "1\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, create a long array named array of size n. RESULT array]
            [COMMAND Within the outer for loop over z, start an inner for loop with integer i initialized to 0, continuing while i is less than n, incrementing i by 1. Inside this loop, parse the next token from st1 as a long and assign it to array[i]. RESULT array]
            [COMMAND Within the outer for loop over z, sort the array in ascending order using Arrays.sort. RESULT array]
            [COMMAND Within the outer for loop over z, create a Pair array named ps of size n. RESULT ps]
            [COMMAND Within the outer for loop over z, initialize integer w to 0. RESULT w]
            [COMMAND Within the outer for loop over z, initialize integer count to 1. RESULT count]
            [COMMAND Within the outer for loop over z, start an inner for loop with integer i initialized to 1, continuing while i is less than n, incrementing i by 1. Inside this loop, if array[i] equals array[i-1], increment count; otherwise, create a new Pair with value array[i-1] and count, assign it to ps[w], increment w, and reset count to 1. RESULT ps]
            [COMMAND Within the outer for loop over z, after the inner loop completes, create a new Pair with value array[n-1] and count, assign it to ps[w], and increment w. RESULT ps]
            [COMMAND Within the outer for loop over z, sort the ps array from index 0 (inclusive) to index w (exclusive) using Arrays.sort. RESULT ps]
            [COMMAND Within the outer for loop over z, check if ps[0].count is greater than or equal to d. If true, append "0\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, if the previous condition is false, check if d equals 2 and ps[0].count equals 1. If true, append "1\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, check if d equals 3, ps[0].count equals 2, and ps[0].value is not equal to array[n-1]. If true, append "1\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, check if d equals 3, ps[0].count equals 2, and ps[0].value equals array[n-1]. If true, check if ps[1].count equals 2. If true, append "1\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, if the previous condition is true but ps[1].count is not 2, initialize boolean flag to false. Then, start a labeled outer for loop with integer i from 0 to n-1. Inside it, start an inner for loop with integer j from 0 to n-1. If array[j] equals 2*array[i] or array[i] equals 2*array[j], append "1\n" to sb, set flag to true, and break out of the labeled outer loop. RESULT flag]
            [COMMAND Within the outer for loop over z, if flag is false after the nested loops, append "2\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND Within the outer for loop over z, check if d equals 3 and ps[0].count equals 1. If true, initialize boolean flag to false. Then, start a labeled outer for loop with integer i from 0 to n-1. Inside it, start an inner for loop with integer j from 0 to n-1. If array[j] equals 2*array[i] or array[i] equals 2*array[j], append "1\n" to sb, set flag to true, and break out of the labeled outer loop. RESULT flag]
            [COMMAND Within the outer for loop over z, if flag is false after the nested loops, append "2\n" to sb and continue to the next iteration of the outer for loop. RESULT sb]
            [COMMAND After the outer for loop over z completes, close the BufferedReader br. RESULT br]
            [COMMAND Print the contents of the StringBuffer sb to the standard output. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
