1   from collections import defaultdict
2   def solve(n, k, a_list):
3       b_list = [a - 1 for a in a_list]
4       res_dict = defaultdict(list)
5       res_dict[0].append(0)
6       r = 0
7       for i in range(n):
8           r += b_list[i]
9           r %= k
10          res_dict[r].append(i + 1)
11      res = 0
12      for r in res_dict.keys():
13          r_list = res_dict[r]
14          if len(r_list) == 1:
15              continue
16          i = 0
17          j = 0
18          while i < len(r_list) - 1:
19              if j < len(r_list) - 1:
20                  while r_list[j + 1] - r_list[i] < k:
21                      j += 1
22                      if j == len(r_list) - 1:
23                          break
24              res += j - i
25              i += 1
26      return res
