1   import java.io.ByteArrayInputStream;
2   import java.io.IOException;
3   import java.io.InputStream;

5   import java.util.Arrays;
6   import java.util.InputMismatchException;
7   public class Main {
8   	static InputStream is;
9   	static PrintWriter out;
10  	static String INPUT = "";
11  	static void solve()
12  	{
13  		int n = ni(), A = ni(), B = ni();
14  		int[] P = new int[n];
15  		for(int i = 0;i < n;i++)
16  			P[i] = ni();
17  		for(int i = 0;i < n;i++)
18  			P[i]--;
19  		long[][] dp = new long[n+1][n+1];
20  		for(int i = 0;i <= n;i++)
21  			Arrays.fill(dp[i], Long.MAX_VALUE / 2);
22  		for(int i = 0;i <= n;i++){
23  			for(int j = 0;j <= n;j++){
24  				if(i == 0 && j == 0){
25  					dp[i][j] = 0;
26  				}else if(i-1 >= 0 && j-1 >= 0 && P[i-1] == j-1){
27  					dp[i][j] = dp[i-1][j-1];
28  				}else{
29  					if(i-1 >= 0)
30  						dp[i][j] = Math.min(dp[i][j], dp[i-1][j] + (P[i-1] <= j ? B : A));
31  					if(j-1 >= 0)
32  						dp[i][j] = Math.min(dp[i][j], dp[i][j-1]);
33  				}
34  			}
35  		}
36  		out.println(dp[n][n]);
37  	}
38  	public static void main(String[] args) throws Exception
39  	{
40  		long S = System.currentTimeMillis();
41  		is = INPUT.isEmpty() ? System.in : new ByteArrayInputStream(INPUT.getBytes());
42  		out = new PrintWriter(System.out);
43  		solve();
44  		out.flush();
45  		long G = System.currentTimeMillis();
46  		tr(G-S+"ms");
47  	}
48  	private static boolean eof()
49  	{
50  		if(lenbuf == -1)return true;
51  		int lptr = ptrbuf;
52  		while(lptr < lenbuf)if(!isSpaceChar(inbuf[lptr++]))return false;
53  		try {
54  			is.mark(1000);
55  			while(true){
56  				int b = is.read();
57  				if(b == -1){
58  					is.reset();
59  					return true;
60  				}else if(!isSpaceChar(b)){
61  					is.reset();
62  					return false;
63  				}
64  			}
65  		} catch (IOException e) {
66  			return true;
67  		}
68  	}
69  	private static byte[] inbuf = new byte[1024];
70  	static int lenbuf = 0, ptrbuf = 0;
71  	private static int readByte()
72  	{
73  		if(lenbuf == -1)throw new InputMismatchException();
74  		if(ptrbuf >= lenbuf){
75  			ptrbuf = 0;
76  			try { lenbuf = is.read(inbuf); } catch (IOException e) { throw new InputMismatchException(); }
77  			if(lenbuf <= 0)return -1;
78  		}
79  		return inbuf[ptrbuf++];
80  	}
81  	private static boolean isSpaceChar(int c) { return !(c >= 33 && c <= 126); }
82  	private static int skip() { int b; while((b = readByte()) != -1 && isSpaceChar(b)); return b; }
83  	private static double nd() { return Double.parseDouble(ns()); }
84  	private static char nc() { return (char)skip(); }
85  	private static String ns()
86  	{
87  		int b = skip();
88  		StringBuilder sb = new StringBuilder();
89  		while(!(isSpaceChar(b))){
90  			sb.appendCodePoint(b);
91  			b = readByte();
92  		}
93  		return sb.toString();
94  	}
95  	private static char[] ns(int n)
96  	{
97  		char[] buf = new char[n];
98  		int b = skip(), p = 0;
99  		while(p < n && !(isSpaceChar(b))){
100  			buf[p++] = (char)b;
101  			b = readByte();
102  		}
103  		return n == p ? buf : Arrays.copyOf(buf, p);
104  	}
105  	private static char[][] nm(int n, int m)
106  	{
107  		char[][] map = new char[n][];
108  		for(int i = 0;i < n;i++)map[i] = ns(m);
109  		return map;
110  	}
111  	private static int[] na(int n)
112  	{
113  		int[] a = new int[n];
114  		for(int i = 0;i < n;i++)a[i] = ni();
115  		return a;
116  	}
117  	private static int ni()
118  	{
119  		int num = 0, b;
120  		boolean minus = false;
121  		while((b = readByte()) != -1 && !((b >= '0' && b <= '9') || b == '-'));
122  		if(b == '-'){
123  			minus = true;
124  			b = readByte();
125  		}
126  		while(true){
127  			if(b >= '0' && b <= '9'){
128  				num = num * 10 + (b - '0');
129  			}else{
130  				return minus ? -num : num;
131  			}
132  			b = readByte();
133  		}
134  	}
135  	private static long nl()
136  	{
137  		long num = 0;
138  		int b;
139  		boolean minus = false;
140  		while((b = readByte()) != -1 && !((b >= '0' && b <= '9') || b == '-'));
141  		if(b == '-'){
142  			minus = true;
143  			b = readByte();
144  		}
145  		while(true){
146  			if(b >= '0' && b <= '9'){
147  				num = num * 10 + (b - '0');
