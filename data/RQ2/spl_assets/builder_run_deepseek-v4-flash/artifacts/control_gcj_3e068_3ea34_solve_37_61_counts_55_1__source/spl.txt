[DEFINE_WORKER: "Reads n groups of candidate indices, maintains usage counts, and for each nonempty shuffled group prints the least-used candidate while marking it as used, or prints -1 when the group is empty or all candidates have reached the USED state." solve]
    [INPUTS]
        <REF> case_index </REF>: int "Unused parameter representing the current test case index."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set USED to the constant 0xFFFFFFF. RESULT USED]
            [COMMAND Call next_line to consume the next line of input before reading n. RESULT void]
            [COMMAND Read the integer n from input. RESULT n]
            [COMMAND Initialize counts as a list of n zeros. RESULT counts]
            [COMMAND For each i from 0 to n-1, call next_line to consume the line before reading the current group. RESULT void]
            [COMMAND For each i from 0 to n-1, read the integer d for the current group. RESULT d]
            [COMMAND For each i from 0 to n-1, when d > 0, initialize cur as an empty list. RESULT cur]
            [COMMAND For each i from 0 to n-1, when d > 0, for each j from 0 to d-1, read an integer itemaa and append it to cur. RESULT cur]
            [COMMAND For each i from 0 to n-1, when d > 0, shuffle cur randomly. RESULT cur]
            [COMMAND For each i from 0 to n-1, when d > 0, for each occurrence index in cur, if counts[index] is not equal to USED, increment counts[index] by 1. RESULT counts]
            [COMMAND For each i from 0 to n-1, when d > 0, choose chosen_index as the element in cur with the smallest counts value; the prior shuffle makes ties random. RESULT chosen_index]
            [COMMAND For each i from 0 to n-1, when d > 0 and counts[chosen_index] is not equal to USED, print chosen_index, then set counts[chosen_index] to USED. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Inside the outer loop, after reading d, d is equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1 to the output. RESULT void]
            [COMMAND Continue to the next iteration of the outer loop. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the outer loop, after choosing chosen_index, counts[chosen_index] is equal to USED.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1 to the output. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
