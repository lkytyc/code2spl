[DEFINE_WORKER: "Simulates the Google Code Jam Bathroom Stalls process for each test case by placing K people into N stalls using maximum-minimum-distance selection with max-distance and leftmost tie-breaking, then prints the last selected stall's max and min distances." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer T from standard input using raw_input() and int(); T is the number of test cases. RESULT T]
            [COMMAND For each test case i from 1 to T inclusive, read a line tmp from standard input, split it on whitespace, map every token to an integer, and store the resulting list as tmpList. RESULT i, tmpList]
            [COMMAND Within the outer loop over i, set N = int(tmpList[0]) and K = int(tmpList[1]); N is the number of stalls and K is the number of people to place. RESULT N, K]
            [COMMAND Within the outer loop over i, initialize stall_occupied as [True] followed by N False entries and then append True, producing N+2 elements with occupied sentinels at indices 0 and N+1. RESULT stall_occupied]
            [COMMAND Within the outer loop over i, initialize Ls, Rs, Min_s, and Max_s as empty lists. RESULT Ls, Rs, Min_s, Max_s]
            [COMMAND Within the outer loop over i, for j from 0 to N+1 inclusive, if stall_occupied[j] is True append -1 to Ls and -1 to Rs; otherwise append j-1 to Ls and N-j to Rs. Then append min(Ls[-1], Rs[-1]) to Min_s and max(Ls[-1], Rs[-1]) to Max_s. RESULT Ls, Rs, Min_s, Max_s]
            [COMMAND Within the outer loop over i, for each placement k from 0 to K-1 inclusive, compute maximal = max(Min_s). RESULT maximal]
            [COMMAND Within the placement loop over k, if Min_s.count(maximal) is greater than 1, gather all indices idxs in Min_s for which Min_s equals maximal. RESULT idxs]
            [COMMAND Within the tie-handling path, build max_of_idxs as the list of Max_s[j] for every index j in idxs, then compute max_max = max(max_of_idxs). RESULT max_of_idxs, max_max]
            [COMMAND Within the tie-handling path, if max_of_idxs.count(max_max) is greater than 1, select idx as the first index in idxs whose Max_s value is max_max; otherwise select idx as the index in idxs corresponding to the unique max_max value. RESULT idx]
            [COMMAND Within the placement loop over k, if Min_s.count(maximal) equals 1, select idx as the first index in Min_s whose value equals maximal, using Min_s.index(maximal). RESULT idx]
            [COMMAND After selecting idx in the placement loop over k, set y = max(Ls[idx], Rs[idx]) and z = min(Ls[idx], Rs[idx]), then set stall_occupied[idx] = True and set Ls[idx], Rs[idx], Min_s[idx], and Max_s[idx] all to -1. RESULT y, z, stall_occupied, Ls, Rs, Min_s, Max_s]
            [COMMAND After occupying idx, update left-side distances: start leftIdx = idx - 1 and while leftIdx > 0, stop if stall_occupied[leftIdx] is True; otherwise assign Rs[leftIdx] = idx - leftIdx - 1, Min_s[leftIdx] = min(Rs[leftIdx], Ls[leftIdx]), Max_s[leftIdx] = max(Rs[leftIdx], Ls[leftIdx]), and decrement leftIdx by 1. RESULT Ls, Rs, Min_s, Max_s]
            [COMMAND After occupying idx, update right-side distances: start rightIdx = idx + 1 and while rightIdx < N + 2, stop if stall_occupied[rightIdx] is True; otherwise assign Ls[rightIdx] = rightIdx - idx - 1, Min_s[rightIdx] = min(Rs[rightIdx], Ls[rightIdx]), Max_s[rightIdx] = max(Rs[rightIdx], Ls[rightIdx]), and increment rightIdx by 1. RESULT Ls, Rs, Min_s, Max_s]
            [COMMAND After all K placements for test case i, print the line formatted as Case #i: y z using the last selected y and z, then flush standard output. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
