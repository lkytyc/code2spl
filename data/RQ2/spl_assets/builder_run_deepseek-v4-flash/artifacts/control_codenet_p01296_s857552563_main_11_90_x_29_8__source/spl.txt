[DEFINE_WORKER: "Reads grid domino placements until a zero sentinel and determines for each case whether a consistent alternating two-coloring exists by propagating through same-piece and adjacent-piece constraints, returning 'Yes' or 'No' per case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: str "A single string with each test case answer ('Yes' or 'No') on its own line, in input order."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize inf = 10**20, eps = 1.0 / 10**10, mod = 10**9 + 7, ddn as the eight compass directions, dd as the four cardinal directions, and rr = [] as the result collector. RESULT rr]
            [COMMAND Enter the outer while True loop and read n = I(). RESULT n]
            [COMMAND If n == 0, break out of the outer loop and terminate further test-case processing. RESULT n]
            [COMMAND Initialize a = [] and append n lines read by LS() to it, so each a[i] contains the x, y, direction fields for one domino. RESULT a]
            [COMMAND Initialize d = collections.defaultdict(int) for cell-to-piece mapping and ed = collections.defaultdict(lambda: None) for cell-to-same-piece-partner mapping. RESULT d, ed]
            [COMMAND Within the outer loop, for each i from 0 to n-1, unpack a[i] as x, y, di; convert x and y to int; set d[(x,y)] = i+1; if di == 'x', additionally set d[(x+1,y)] = i+1 and set ed[(x,y)] = (x+1,y) and ed[(x+1,y)] = (x,y); otherwise, additionally set d[(x,y+1)] = i+1 and set ed[(x,y)] = (x,y+1) and ed[(x,y+1)] = (x,y). RESULT d, ed]
            [COMMAND Initialize ee = collections.defaultdict(set) to store grid-neighbor adjacencies between cells of different dominoes, and snapshot dka = list(d.keys()). RESULT ee, dka]
            [COMMAND Within the outer loop, for each occupied cell (x,y) in list(d.keys()), let dt = d[(x,y)]; for each cardinal offset (di,dj) in dd, compute neighbor (ni,nj) = (x+di, y+dj) and if d[(ni,nj)] > 0 and d[(ni,nj)] != dt, add (ni,nj) to ee[(x,y)]. RESULT ee]
            [COMMAND Initialize v = collections.defaultdict(bool) to mark visited cells and set f = True as the validity flag for the current test case. RESULT v, f]
            [COMMAND Within the outer loop, iterate over each unvisited cell k in dka; skip cells where v[k] is true; otherwise start a new component search with s1 = set(), s2 = set(), ns1 = {k}, ns2 = set(). RESULT k]
            [COMMAND While ns1 or ns2 is nonempty, propagate the alternating coloring: absorb ns1 into s1, then for each cell in the old ns1 add all ee neighbors to ns1 and add ed[k] to ns2; remove cells already in s2 from ns2, absorb ns2 into s2, then for each cell in the old ns2 add all ee neighbors to ns2 and add ed[k] to ns1; remove already settled cells from the candidate sets, then absorb the remaining ns1 into s1 and ns2 into s2. RESULT s1, s2, ns1, ns2]
            [COMMAND If s1 and s2 intersect after propagation, set f = False and break out of the propagation while loop. RESULT f]
            [COMMAND If f is still true after processing a component, mark every cell in s1 and s2 as visited in v; otherwise break out of the component iteration loop. RESULT v]
            [COMMAND After the component iteration completes, if f is true append 'Yes' to rr; otherwise append 'No' to rr. RESULT rr]
            [COMMAND After the outer while True loop terminates, return '\n'.join(map(str, rr)). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
