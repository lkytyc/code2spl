[DEFINE_WORKER: "Reads test cases from standard input, separates the values at even and odd positions into two arrays, sorts each array, then checks the alternating sorted order and prints the first mismatch position or OK for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a Scanner named reader on standard input, read the first integer into numberOfTests, and initialize a and b to 0. RESULT reader, numberOfTests, a, b]
            [COMMAND For each test case m from 0 to numberOfTests - 1, execute the test-case body. RESULT m]
            [COMMAND Within test case m, read the current array length n from reader. RESULT n]
            [COMMAND Within test case m, set a and b based on n: if n % 2 == 1, set a = n / 2 + 1 and b = n / 2; otherwise set a = n / 2 and b = n / 2. RESULT a, b]
            [COMMAND Within test case m, allocate oddArray of length a and evenArray of length a, and initialize c and d to 0. RESULT oddArray, evenArray, c, d]
            [COMMAND Within test case m, for each i from 0 to n - 1, read the next integer from reader; if i % 2 == 0, store it in oddArray[c] and increment c, otherwise store it in evenArray[d] and increment d. RESULT oddArray, evenArray, c, d]
            [COMMAND Within test case m, unconditionally set evenArray[evenArray.length - 1] to Integer.MAX_VALUE. RESULT evenArray]
            [COMMAND Within test case m, create a Solution instance named mms and call mms.sort(oddArray) and mms.sort(evenArray). RESULT mms, oddArray, evenArray]
            [COMMAND Within test case m, initialize x to 0. RESULT x]
            [COMMAND Within test case m, for each h from 0 to oddArray.length - 1, perform the interleaving validation inside a try block with an empty finally block and apply the applicable alternative flow for that comparison. RESULT h]
            [COMMAND After the validation loop completes for test case m, if x == 0, print "Case #" + (m + 1) + ": OK". RESULT output]
            [COMMAND After all test cases are processed, close reader. RESULT reader]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the validation loop in test case m, oddArray[h] <= evenArray[h].]
        [SEQUENTIAL_BLOCK]
            [COMMAND If h == oddArray.length - 1, break out of the validation loop without printing a failure. RESULT h]
            [COMMAND If evenArray[h] <= oddArray[h + 1], continue to the next h; otherwise proceed to the next step. RESULT h]
            [COMMAND Print "Case #" + (m + 1) + ": " + (2 * h + 1), increment x by 1, and break out of the validation loop. RESULT x]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the validation loop in test case m, oddArray[h] > evenArray[h].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + (m + 1) + ": " + (2 * h), increment x by 1, and break out of the validation loop. RESULT x]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the validation loop in test case m completes, x == 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + (m + 1) + ": OK". RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
