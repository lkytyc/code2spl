[DEFINE_WORKER: "Reads N integer coordinate pairs, sorts them, groups every ordered point pair by its gcd-normalized direction vector, and for each direction graph constructs source-rooted chains to compute a maximum chain-statistic m, then prints the case result." do_one_case]
    [INPUTS]
        <REF> cnum </REF>: int "Case number used in the formatted output line."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: string "Printed line formatted as 'Case #<cnum>: <m>'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read one integer N from input using rd1int(). RESULT N]
            [COMMAND Initialize XY as an empty list. RESULT XY]
            [COMMAND For i in range(N), read two integers x and y using rdints(), then append the tuple (x, y) to XY. RESULT XY]
            [COMMAND Sort XY lexicographically by coordinate pair. RESULT XY]
            [COMMAND Initialize A as an empty dictionary mapping normalized direction tuples to lists of directed index pairs. RESULT A]
            [COMMAND For j from 1 to N-1 inclusive, and for i from 0 to j-1 inclusive, compute the direction vector from XY[i] to XY[j] as a = x2-x1 and b = y2-y1, divide both a and b by gcd(a,b), and store the directed pair (i,j) under key (a,b) and the reverse pair (j,i) under key (-a,-b), creating new lists when a key is absent and appending when it is present. RESULT A]
            [COMMAND Initialize m to 1. RESULT m]
            [COMMAND For each key (a,b) in A, retrieve the edge list E = A[(a,b)]. RESULT E]
            [COMMAND Within the loop over keys (a,b) in A, build incoming adjacency map I and outgoing adjacency map O from E by appending j to O[i] and i to I[j] for every directed edge (i,j), initializing missing lists as needed. RESULT I_O]
            [COMMAND Within the loop over keys (a,b) in A, compute NI as set(range(N)) minus set(I), representing all node indices with no incoming edge. RESULT NI]
            [COMMAND Within the loop over keys (a,b) in A, build the list of chains C: for each source node x in NI, if x appears in O then form Q = [(0,x)] plus [(len(I[y]), y) for y in O[x]], sort Q, and append the node IDs [y for _, y in Q] as a chain; otherwise append the singleton [x]. RESULT C]
            [COMMAND Within the loop over keys (a,b) in A, partition C into C1 = chains of length 1, C2 = even-length chains, and C3 = odd-length chains of length greater than 1. RESULT C1_C2_C3]
            [COMMAND Within the loop over keys (a,b) in A, while len(C3) > 1, merge the last two chains in C3 by concatenation, append the merged chain to C2, and delete those last two entries from C3. RESULT C2_C3]
            [COMMAND Within the loop over keys (a,b) in A, replace C3 with (C3 + C1)[:2] so that at most two chains remain, preferring any remaining odd chains from C3 and then singleton chains from C1. RESULT C3]
            [COMMAND Within the loop over keys (a,b) in A, compute l2 as the sum of lengths of all chains in C2 and l3 as the sum of lengths of all chains in C3. RESULT l2_l3]
            [COMMAND Within the loop over keys (a,b) in A, compute l: if l3 is nonzero then l = l2 + l3; else if len(C2) > 1 then l = l2 - 1; else l = l2. RESULT l]
            [COMMAND Within the loop over keys (a,b) in A, update m to max(m, l). RESULT m]
            [COMMAND After all keys in A have been processed, print the string 'Case #%d: %d' formatted with cnum and m. RESULT printed_line]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
