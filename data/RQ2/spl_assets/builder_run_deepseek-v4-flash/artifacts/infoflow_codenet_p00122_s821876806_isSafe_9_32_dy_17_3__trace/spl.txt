[DEFINE_WORKER: "Expands each dequeued safe-position entry into a fixed 12-cell stage-neighborhood on a 10x10 grid and enqueues matching stage-indexed entries until either a stage-9 entry is reached (returning True) or the worklist is exhausted (returning False)." isSafe]
    [INPUTS]
        <REF> wfs </REF>: worklist/queue supporting empty(), get(), and put() "Worklist of pending items; each item is [position, stage] where position is [x, y] and stage is an integer."
        <REF> safe </REF>: stage-indexed collection "safe[spn] is a list of candidate items shaped [position, stage] for stage spn."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: boolean "True if a dequeued expansion reaches a matching item at stage 9; False if the outer while loop ends without reaching stage 9."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Use the initial contents of wfs as the pending worklist and keep processing while wfs is not empty. RESULT wfs]
            [COMMAND While wfs is not empty, dequeue one item as flog. RESULT flog]
            [COMMAND Set spn to flog[1] + 1 and reset flogArea to an empty list for this iteration. RESULT spn]
            [COMMAND Build the horizontal-distance-2 part of the neighborhood: for each i in {-1, 0, 1} and each j in {-2, 2}, compute dx = flog[0][0] + j and dy = flog[0][1] + i; if 0 <= dx < 10 and 0 <= dy < 10, append [[dx, dy], spn] to flogArea. RESULT flogArea]
            [COMMAND Build the vertical-distance-2 part: for each j in {-1, 0, 1} and each i in {-2, 2}, compute dx = flog[0][0] + j and dy = flog[0][1] + i; if 0 <= dx < 10 and 0 <= dy < 10, append [[dx, dy], spn] to flogArea. RESULT flogArea]
            [COMMAND For each index i in range(len(safe[spn])), test whether safe[spn][i] is exactly equal to one of the entries in flogArea. RESULT matching_indices]
            [COMMAND For each matched index i, if spn < 9, put safe[spn][i] back into wfs without deduplicating; if spn > 9, ignore that match and continue scanning. RESULT wfs_updated]
            [COMMAND When wfs becomes empty and no stage-9 match has returned, return False. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the safe[spn] scan, a matched index i has spn exactly equal to 9.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Immediately return True, ending the function and skipping all remaining iterations. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
