[DEFINE_WORKER: "Reads n and a, validates prefix conditions, scans a[3:] with moving counters x, y, z, computes a count modulo 1000000007, and prints 0 for invalid inputs." main]
    [INPUTS]
        <REF> given </REF>: callable "Zero-argument callable that returns one input line; defaults to sys.stdin.readline."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read one integer n from the input stream using II(). RESULT n]
            [COMMAND Read the next input line as a list of integers a using LMIIS(). RESULT a]
            [COMMAND Validate that a[0] equals 0; if a[0] is not 0, print 0 and terminate immediately. RESULT None]
            [COMMAND If n equals 1, then: if a[0] equals 0 print 3, otherwise print 0; then terminate. RESULT None]
            [COMMAND Else if n equals 2, run two independent checks on a[1]: if a[1] equals 0 print 3; then if a[1] equals 1 print 6, otherwise print 0; then terminate. RESULT None]
            [COMMAND For n greater than or equal to 3, initialize x, y, z, and res from the pair (a[1], a[2]): if (0,0) set x=1, y=1, z=1, res=1; if (0,1) set x=2, y=1, z=0, res=2; if (1,2) set x=3, y=0, z=0, res=1; otherwise print 0 and terminate. RESULT x, y, z, res]
            [COMMAND Set MOD to 1000000007. RESULT MOD]
            [COMMAND For each i from 3 through n-1, evaluate a[i] against the ordered conditions: if a[i] equals x and does not equal y, increment x; else if a[i] equals x and y and does not equal z, set res = (res * 2) % MOD and increment x; else if a[i] equals x, y, and z, set res = (res * 3) % MOD and increment x; else if a[i] equals y and does not equal z, increment y; else if a[i] equals y and z, set res = (res * 2) % MOD and increment y; else if a[i] equals z, increment z; else print 0 and terminate. RESULT x, y, z, res]
            [COMMAND After the loop, if y equals 0, set res = (res * 3) % MOD; otherwise set res = (res * 6) % MOD. RESULT res]
            [COMMAND Print the final value of res. RESULT None]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
