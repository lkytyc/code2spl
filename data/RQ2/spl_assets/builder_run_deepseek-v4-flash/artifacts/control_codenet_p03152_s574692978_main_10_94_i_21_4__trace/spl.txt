[DEFINE_WORKER: "Reads two sequences, reorders each into strictly descending order while rejecting duplicates, then calculates a modular count from the sorted sequences and prints the resulting integer or 0." main]
    [INPUTS]
        <REF> stdin </REF>: standard input "First line supplies two integers n and m; the next n integers form a[0..n-1] and the following m integers form b[0..m-1]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: text "Prints '0\n' if an equal adjacent pair is found during either reorder pass; otherwise prints the final rounded ans value with no decimals."
        <REF> return_code </REF>: int "Returns 0 in all normal and duplicate-detection paths."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers n and m from standard input using scanf("%d %d", &n, &m) without checking scanf's return value. RESULT n_m_values]
            [COMMAND Declare integer arrays a and b with capacity 1003, and declare integer index i and integer temp x for later use. RESULT arrays_declared]
            [COMMAND For i from 0 to n-1 inclusive, read an integer into a[i] using scanf("%d", &a[i]). RESULT array_a_loaded]
            [COMMAND For i from 0 to m-1 inclusive, read an integer into b[i] using scanf("%d", &b[i]). RESULT array_b_loaded]
            [COMMAND Reorder a into strictly descending order: for i from 0 to n-2 inclusive, if a[i] <= a[i+1], then if a[i] == a[i+1] take the equality alternative flow; otherwise swap a[i] and a[i+1] using x, and if i > 0 decrement i by 2 before the normal loop increment. RESULT array_a_descending]
            [COMMAND Reorder b into strictly descending order: for i from 0 to m-2 inclusive, if b[i] <= b[i+1], then if b[i] == b[i+1] take the equality alternative flow; otherwise swap b[i] and b[i+1] using x, and if i > 0 decrement i by 2 before the normal loop increment. RESULT array_b_descending]
            [COMMAND Set p = 1000000007, ans = 1, k = 0, and prepare variables y, j, and q for the following loop. RESULT state_initialized]
            [COMMAND Main counting loop: for i from n*m down to min(a[n-1], b[m-1]) inclusive using i--, set x = y = q = 0 and j = 0; while a[j] > i increment j and x; if a[j] == i set x = 1 and increment q; reset j = 0; while b[j] > i increment j and y; if b[j] == i set y = 1 and increment q; if q == 0 multiply ans by x*y - k, otherwise multiply ans by x*y; then set ans = round(fmod(ans, p)) and increment k. If n*m < min(a[n-1], b[m-1]), skip this loop and leave i at n*m. RESULT ans_after_main_loop]
            [COMMAND Tail factorial-style loop: for j from 0 to i-1 inclusive, multiply ans by j+1 and then set ans = round(fmod(ans, p)), where i is the final value after the main loop exits. RESULT ans_final]
            [COMMAND Print round(ans) using printf("%.0lf\n", round(ans)). RESULT stdout]
            [COMMAND Return 0. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the descending reorder pass on a, an adjacent pair satisfies a[i] == a[i+1].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "0\n" using printf. RESULT stdout]
            [COMMAND Return 0. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the descending reorder pass on b, an adjacent pair satisfies b[i] == b[i+1].]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "0\n" using printf. RESULT stdout]
            [COMMAND Return 0. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
