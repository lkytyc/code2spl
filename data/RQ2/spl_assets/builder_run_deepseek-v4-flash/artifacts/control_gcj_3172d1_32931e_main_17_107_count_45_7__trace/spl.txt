[DEFINE_WORKER: "Reads multiple test cases, each containing an array of long integers and a target divisor d, then determines the minimum number of slices needed to obtain at least d equal-sized pieces by either using existing equal elements or splitting elements into halves." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments (unused)"
    [END_INPUTS]

    [OUTPUTS]
        <REF> sb </REF>: StringBuffer "Accumulated output string containing the result for each test case"
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a BufferedReader to read from standard input, read the first line as an integer t representing the number of test cases, and initialize a StringBuffer to accumulate output. RESULT br, t, sb]
            [COMMAND For each test case z from 1 to t, append the case label 'Case #z: ' to the output buffer. RESULT sb]
            [COMMAND Read two lines of input and tokenize them; parse the first token from the first line as integer n (array size) and the first token from the second line as integer d (target divisor). RESULT n, d]
            [COMMAND If n equals 1 and d equals 3, append '2' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If n equals 1 and d equals 2, append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND Create a long array of size n and populate it by parsing each token from the second input line as a long value. RESULT array]
            [COMMAND Sort the array in ascending order. RESULT array]
            [COMMAND Create an array of Pair objects of size n, then iterate through the sorted array to count consecutive equal values, storing each unique value and its frequency as a Pair in the ps array; track the number of unique pairs with variable w. RESULT ps, w]
            [COMMAND Sort the ps array from index 0 to w (exclusive) in ascending order of frequency. RESULT ps]
            [COMMAND If the frequency of the most frequent element (ps[0].count) is greater than or equal to d, append '0' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 2 and the maximum frequency is 1, append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the maximum frequency is 2, and the most frequent value is not the largest element in the array, append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the maximum frequency is 2, and the most frequent value is the largest element in the array, check if the second most frequent element also has a frequency of 2; if so, append '1' followed by a newline and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3, the maximum frequency is 2, and the most frequent value is the largest element, and the second most frequent element does not have frequency 2, search for any pair of elements in the array where one is exactly double the other; if found, append '1' followed by a newline and continue to the next test case. RESULT sb, flag]
            [COMMAND If d equals 3, the maximum frequency is 2, and the most frequent value is the largest element, and no doubling relationship is found, append '2' followed by a newline and continue to the next test case. RESULT sb]
            [COMMAND If d equals 3 and the maximum frequency is 1, search for any pair of elements in the array where one is exactly double the other; if found, append '1' followed by a newline and continue to the next test case. RESULT sb, flag]
            [COMMAND If d equals 3 and the maximum frequency is 1, and no doubling relationship is found, append '2' followed by a newline and continue to the next test case. RESULT sb]
            [COMMAND After processing all test cases, close the BufferedReader and print the accumulated output buffer to standard output. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the maximum frequency is 2, the most frequent value is the largest element, and the second most frequent element has a frequency of 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the maximum frequency is 2, the most frequent value is the largest element, and a doubling relationship exists between any two elements]

            [COMMAND Append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3, the maximum frequency is 2, the most frequent value is the largest element, and no doubling relationship exists]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '2' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3 and the maximum frequency is 1, and a doubling relationship exists between any two elements]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '1' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When d equals 3 and the maximum frequency is 1, and no doubling relationship exists]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append '2' followed by a newline to the output buffer and continue to the next test case. RESULT sb]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: If an IOException occurs while reading from the input stream]
        [LOG "IOException thrown by BufferedReader.readLine() or StringTokenizer.nextToken()"]
        [THROW IOException "Thrown when an I/O error occurs during input reading"]
    [END_EXCEPTION_FLOW]
