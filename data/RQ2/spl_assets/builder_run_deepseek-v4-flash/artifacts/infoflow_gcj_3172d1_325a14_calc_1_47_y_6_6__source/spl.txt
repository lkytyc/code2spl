[DEFINE_WORKER: "Computes the minimum candidate score ans by iterating over each starting index, accumulating cut and cur_d counters through a forward sweep and an optional reverse suffix sweep, and returns ans together with the first element of a." calc]
    [INPUTS]
        <REF> n </REF>: integer "Input parameter n; it is immediately overwritten by len(a) and does not affect the rest of the computation."
        <REF> d </REF>: integer "Bound used to initialize ans, control loop ranges, and compute cost updates."
        <REF> a </REF>: list "List of integer values whose first element is returned as ret and whose length determines the outer loop count."
        <REF> b </REF>: list "List of integer values aligned with a; b[i] provides y for each outer iteration."
        <REF> ct </REF>: dict "Dictionary mapping p values to counts tmp used in the forward sweep's inner loop count."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "Tuple (ans, ret), where ans is the best minimum value found and ret is a[0]."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Overwrite input parameter n with len(a), then set ans to d - 1 and ret to the first element of a. RESULT initial_state]
            [COMMAND For each index i from 0 to n - 1 inclusive, set x to a[i], y to b[i], cur_d to 0, p to 0, and cut to 0. RESULT outer_loop_state]
            [COMMAND Within the outer loop over i, run the forward sweep: for j from 0 to d + 1 inclusive, add x to p, set tmp to ct.get(p, 0), set per_d to p / x; then for k from 0 to (tmp * per_d) - 1 inclusive, increment cur_d, increment cut when (k + 1) modulo per_d is not 0, set xx to d / cur_d and decrement xx by 1 when d modulo cur_d is 0, update ans to min(ans, cut + xx * cur_d), and break out of this k-loop if cur_d > d; after the k-loop, break out of the j-loop if cur_d > d. RESULT forward_sweep_state]
            [COMMAND Within the outer loop over i, if cur_d is less than d, run the reverse suffix sweep: for j from n - 1 down to i + 1 inclusive, if a[j] is greater than p, or, only when that is false, if a[j] modulo x is not 0, set tmp to (a[j] / x) * y; then for k from 0 to tmp - 1 inclusive, increment cur_d, increment cut, set xx to d / cur_d and decrement xx by 1 when d modulo cur_d is 0, update ans to min(ans, cut + xx * cur_d), and break out of this k-loop if cur_d > d; after that, break out of the reverse j-loop if cur_d > d. RESULT reverse_sweep_state]
            [COMMAND Within the outer loop over i, after the conditional reverse sweep block, compute xx as d / cur_d, decrement xx by 1 when d modulo cur_d is 0, and update ans to min(ans, cut + xx * cur_d). RESULT final_ans_update]
            [COMMAND After the outer loop completes, return the tuple (ans, ret). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
