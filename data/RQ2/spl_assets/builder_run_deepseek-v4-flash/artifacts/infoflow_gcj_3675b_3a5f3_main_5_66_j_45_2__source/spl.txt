[DEFINE_WORKER: "Reads a number of test cases; for each, derives M and N from D, A, and B, computes backward-consistent run lengths longm and longn, and prints the maximum value of getmax(longm[i], longn[i]) together with the count of indices that attain it." main]
    [INPUTS]
        <REF> argc </REF>: int "Number of command-line arguments; unused by this main method."
        <REF> argv </REF>: char *[] "Array of command-line argument strings; unused by this main method."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return </REF>: int "Returns 0 to the operating system after all test cases are processed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare all local variables and arrays: int ntests; int test; long int D[MAXS], A[MAXS], B[MAXS], M[MAXS], N[MAXS]; int longm[MAXS], longn[MAXS]; int s, i, j, max, max_no. RESULT variables]
            [COMMAND Read one integer into ntests from standard input using scanf("%d", &ntests). If scanf does not return 1, ntests is not assigned; no validation is performed. RESULT ntests]
            [COMMAND Loop test from 1 to ntests inclusive (for test = 1; test <= ntests; test++). RESULT test]
            [COMMAND Inside the test loop, read one integer s from standard input using scanf("%d", &s). RESULT s]
            [COMMAND Inside the test loop, for i from 0 to s-1 inclusive, read three long integers into D[i], A[i], and B[i] using scanf("%li %li %li", D+i, A+i, B+i). RESULT D,A,B]
            [COMMAND Inside the test loop, for i from 0 to s-1 inclusive, set M[i] = D[i] + A[i] and N[i] = D[i] - B[i]. RESULT M,N]
            [COMMAND Inside the test loop, unconditionally set longm[0] = 1 and longn[0] = 1, even if s is 0. RESULT longm,longn]
            [COMMAND Inside the test loop, for i from 1 to s-1 inclusive, if M[i] == M[i-1] then set longm[i] = longm[i-1] + 1; otherwise set longm[i] = 2, set j = i-2, and while j >= 0 and, using short-circuit OR, (M[j] == M[i] || N[j] == N[i-1]), increment longm[i] and decrement j. RESULT longm]
            [COMMAND Inside the test loop, for i from 1 to s-1 inclusive, if N[i] == N[i-1] then set longn[i] = longn[i-1] + 1; otherwise set longn[i] = 2, set j = i-2, and while j >= 0 and, using short-circuit OR, (N[j] == N[i] || M[j] == M[i-1]), increment longn[i] and decrement j. RESULT longn]
            [COMMAND Inside the test loop, set max = getmax(longm[0], longn[0]) and max_no = 1. RESULT max,max_no]
            [COMMAND Inside the test loop, for i from 1 to s-1 inclusive, if getmax(longm[i], longn[i]) > max then set max_no = 1 and max = getmax(longm[i], longn[i]); else if getmax(longm[i], longn[i]) == max then increment max_no. RESULT max,max_no]
            [COMMAND Inside the test loop, print one line to standard output using printf("Case #%d: %d %d\n", test, max, max_no). RESULT stdout]
            [COMMAND After the test loop completes, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
