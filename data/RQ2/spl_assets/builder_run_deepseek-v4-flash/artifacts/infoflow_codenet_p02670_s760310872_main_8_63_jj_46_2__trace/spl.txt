[DEFINE_WORKER: "Reads an integer n and n*n cell labels, marks each corresponding cell in an n by n matrix as zero, and prints the total over all interior marked cells of the smallest of the four directional distances to the matrix border after subtracting previously zeroed cells in those directions." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return </REF>: int "Always returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer n from standard input into n. RESULT n]
            [COMMAND Declare int a[501][501]; declare long p, long k = 1, and long cnt = 0. RESULT state]
            [COMMAND For i from 0 to n-1 inclusive, and for j from 0 to n-1 inclusive, set a[i][j] = k, then increment k by 1, filling the n by n submatrix with values 1 through n*n in row-major order. RESULT a]
            [COMMAND For ii from 0 to n*n-1 inclusive, read a long value p from standard input. The following branch steps are executed inside this loop. RESULT p]
            [COMMAND If p is on the top row (1 <= p and p <= n), set a[0][p-1] = 0. RESULT a]
            [COMMAND Else if p is on the bottom row ((n*n - n + 1) <= p and p <= n*n), set a[n-1][p - (n*n - n + 1)] = 0, equivalently a[n-1][n - (n*n) + p - 1] = 0. RESULT a]
            [COMMAND Else if p % n == 0, set a[(p/n) - 1][n-1] = 0 for the right-column cell. RESULT a]
            [COMMAND Else if p % n == 1, set a[p/n][0] = 0 for the left-column cell. RESULT a]
            [COMMAND Else, for an interior cell, set a[p/n][(p%n) - 1] = 0, and set i = p/n and j = (p%n) - 1. RESULT state]
            [COMMAND In the same interior-cell branch, initialize long m[5] = {i, n-1-i, j, n-1-j}. RESULT m]
            [COMMAND In the same interior-cell branch, for jj from 0 to i-1 inclusive, if a[jj][j] == 0, decrement m[0]. RESULT m]
            [COMMAND In the same interior-cell branch, for jj from n-1 down to i+1 inclusive, if a[jj][j] == 0, decrement m[1]. RESULT m]
            [COMMAND In the same interior-cell branch, for jj from 0 to j-1 inclusive, if a[i][jj] == 0, decrement m[2]. RESULT m]
            [COMMAND In the same interior-cell branch, for jj from n-1 down to j+1 inclusive, if a[i][jj] == 0, decrement m[3]. RESULT m]
            [COMMAND In the same interior-cell branch, sort the first 4 elements of m in ascending order by calling qsort(m, 4, sizeof(long), compare_int), then set mm to the sorted m[0]. RESULT mm]
            [COMMAND In the same interior-cell branch, add mm to cnt: cnt = cnt + mm. RESULT cnt]
            [COMMAND After the processing loop over ii completes, print cnt as a long integer followed by a newline. RESULT stdout]
            [COMMAND Return 0 from main. RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
