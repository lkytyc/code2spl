[DEFINE_WORKER: "Determines whether an alien pylon placement is possible for dimensions a and b, printing IMPOSSIBLE when |a - b| <= 1 and otherwise printing POSSIBLE followed by a*b generated coordinate pairs whose order depends on whether b >= a." alienpylon]
    [INPUTS]
        <REF> a </REF>: int "First positive integer dimension; also the initial value of temp1."
        <REF> b </REF>: int "Second positive integer dimension; also the initial value of temp2."
        <REF> i </REF>: int "Zero-based test case index used to print the case number as i + 1."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integer k (uninitialized), set temp1 = a, and set temp2 = b. RESULT initialized_temp1_temp2]
            [COMMAND Evaluate outer condition b >= a. If true, execute the following b >= a branch; otherwise execute the alternative flow with condition b < a. RESULT outer_branch_b_ge_a]
            [COMMAND In the b >= a branch, check inner condition: b - a == 1 or a + 1 == b or a - 1 == b or b + 1 == a or b - 1 == a or a == b (equivalently |a - b| <= 1). RESULT inner_condition_b_ge_a]
            [COMMAND In the b >= a branch, if the inner condition is true, call printf("Case #%d: IMPOSSIBLE\n", i + 1). RESULT printed_impossible_b_ge_a]
            [COMMAND In the b >= a branch, if the inner condition is false, call printf("Case #%d: POSSIBLE\n", i + 1). RESULT printed_possible_b_ge_a]
            [COMMAND In the b >= a branch after printing POSSIBLE, run a for loop with k from 1 to a * b inclusive. In each iteration, set temp1 = (temp1 + (((b / a) - 1) % a)) % a, then set temp2 = (temp2 + (((b % a) + 1) % b)) % b, then call printf("%d %d\n", temp1 + 1, temp2 + 1). RESULT printed_coordinates_b_ge_a]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b < a]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the b < a branch, check inner condition: a - b == 1 or a + 1 == b or a - 1 == b or b + 1 == a or b - 1 == a or a == b (equivalently |a - b| <= 1). RESULT inner_condition_b_lt_a]
            [COMMAND In the b < a branch, if the inner condition is true, call printf("Case #%d: IMPOSSIBLE\n", i + 1). RESULT printed_impossible_b_lt_a]
            [COMMAND In the b < a branch, if the inner condition is false, call printf("Case #%d: POSSIBLE\n", i + 1). RESULT printed_possible_b_lt_a]
            [COMMAND In the b < a branch after printing POSSIBLE, run a for loop with k from 1 to a * b inclusive. In each iteration, set temp2 = (temp2 + (((a / b) - 1) % b)) % b, then set temp1 = (temp1 + (((a % b) + 1) % a)) % a, then call printf("%d %d\n", temp2 + 1, temp1 + 1). RESULT printed_coordinates_b_lt_a]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
