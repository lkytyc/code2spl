[DEFINE_WORKER: "Reads an h by w grid, marks '#' cells as mines, computes each non-mine cell's adjacent mine count, and prints the resulting MineSweeper board." calc]
    [INPUTS]
        <REF> h </REF>: int "Number of rows in the grid."
        <REF> w </REF>: int "Number of columns in the grid."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Always returns 0 after printing the board."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Allocate a 2D integer array panel with h rows and w columns. RESULT panel]
            [COMMAND Declare a character array row of length w to hold each input string. RESULT row]
            [COMMAND For each row i from 0 to h-1, read a string from standard input into row using scanf("%s", row). RESULT row]
            [COMMAND Within that outer row loop over i, for each column j from 0 to w-1, if row[j] is '#', set panel[i][j] to -1; otherwise set panel[i][j] to 0. RESULT panel]
            [COMMAND For each row i from 0 to h-1, for each column j from 0 to w-1, if panel[i][j] equals -1, continue to the next cell and leave the mine marker unchanged. RESULT panel]
            [COMMAND For each non-mine cell (i, j) in that nested loop, initialize count to 0, then increment count for each valid neighboring cell whose panel value is -1. The neighbor offsets are (i-1,j-1), (i-1,j), (i-1,j+1), (i,j-1), (i,j+1), (i+1,j-1), (i+1,j), and (i+1,j+1); a neighbor is valid only when its row index is between 0 and h-1 and its column index is between 0 and w-1. RESULT count]
            [COMMAND After counting all valid mine neighbors for that non-mine cell, set panel[i][j] to count. RESULT panel]
            [COMMAND For each row i from 0 to h-1, for each column j from 0 to w-1, if panel[i][j] equals -1, print '#'; else print the integer value of panel[i][j] with no separator between columns. RESULT stdout]
            [COMMAND After printing all columns for row i, print a newline. RESULT stdout]
            [COMMAND Return 0 from the function. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
