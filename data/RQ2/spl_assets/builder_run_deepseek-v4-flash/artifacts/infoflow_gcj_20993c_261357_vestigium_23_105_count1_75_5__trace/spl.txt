[DEFINE_WORKER: "Computes the main diagonal sum of an n x n matrix, counts adjacent equal pairs in each row of the matrix and its transpose after sorting, and prints a Case # line with the diagonal sum and the maximum duplicate counts." vestigium]
    [INPUTS]
        <REF> a </REF>: int[][] "n x n matrix of integers to analyze"
        <REF> n </REF>: int "Size of the square matrix; rows and columns are indexed from 0 to n-1, and adjacent-pair checks use indices from 0 to n-2"
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare and initialize integer k, r, and c to 0; allocate b as a new n x n integer array. RESULT k_r_c_b]
            [COMMAND For i from 0 to n-1 inclusive, add a[i][i] to k, accumulating the main diagonal sum. RESULT k]
            [COMMAND For i from 0 to n-1 inclusive, for j from 0 to n-1 inclusive, set b[i][j] = a[j][i], so b becomes the transpose of a. RESULT b]
            [COMMAND For each row i from 0 to n-1 inclusive, sort a[i] in ascending order using Arrays.sort. RESULT a_sorted]
            [COMMAND Initialize repeat to 0 and count to 0. RESULT repeat_count_init]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive, if a[i][j] equals a[i][j+1], increment repeat; then set count = Math.max(count, repeat); after completing each row, if repeat > 0, reset repeat to 0. RESULT count]
            [COMMAND For each row i from 0 to n-1 inclusive, sort b[i] in ascending order using Arrays.sort. RESULT b_sorted]
            [COMMAND Initialize count1 to 0 and repeat1 to 0. RESULT count1_repeat1_init]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive, if b[i][j] equals b[i][j+1], increment repeat1; then set count1 = Math.max(count1, repeat1); after completing each row, if repeat1 > 0, reset repeat1 to 0. RESULT count1]
            [COMMAND Set r = count + 1, c = count1 + 1, and counter = 1. RESULT r_c_counter]
            [COMMAND Evaluate the exhaustive four-way condition on count and count1; select the corresponding output branch and increment counter in every branch. RESULT case_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 and count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print to standard output: "Case #" + counter + ": " + " " + k + " " + count + " " + count1 RESULT case_output_text]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 and count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print to standard output: "Case #" + counter + ": " + " " + k + " " + r + " " + c RESULT case_output_text]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 and count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print to standard output: "Case #" + counter + ": " + " " + k + " " + r + " " + count1 RESULT case_output_text]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 and count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print to standard output: "Case #" + counter + ": " + " " + k + " " + count + " " + c RESULT case_output_text]
            [COMMAND Increment counter by 1. RESULT counter]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
