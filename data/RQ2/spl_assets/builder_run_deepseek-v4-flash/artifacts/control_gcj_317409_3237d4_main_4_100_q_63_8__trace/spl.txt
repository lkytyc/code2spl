[DEFINE_WORKER: "Reads a tourist path and a relative starting offset for each test case, performs a bounded BFS to find the nearest tourist path cell, and computes the earliest meeting time or reports IMPOSSIBLE." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to the main method; they are not used by the algorithm."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize direction offset arrays dx = {0, 1, 0, -1} and dy = {1, 0, -1, 0}, create Scanner sc = new Scanner(new BufferedReader(new InputStreamReader(System.in))), and create an empty StringBuilder ans. RESULT setup]
            [COMMAND Read the integer T = sc.nextInt() as the number of test cases. RESULT T]
            [COMMAND For each test case t from 1 to T inclusive, read X = sc.nextInt(), Y = sc.nextInt(), and M = sc.next(). Initialize a 2001 by 2001 integer grid g with default values 0, and set the tourist position px = 1000 and py = 1000. RESULT case_inputs]
            [COMMAND Initialize tour as an empty list of coordinate pairs. Create the initial coordinate list [px, py], add it to tour, and mark g[py][px] = 1. RESULT tour_initial]
            [COMMAND For each character d in M at index i from 0 to M.length() - 1 inclusive, update px and py according to direction: N means py--, S means py++, E means px++, W means px--. Then mark g[py][px] = 1 and append the newly created coordinate list [px, py] to tour. RESULT tour_path]
            [COMMAND Create a 2001 by 2001 boolean visited grid initialized to false. Compute myx = 1000 - X and myy = 1000 + Y. Initialize a LinkedList<Pair> queue q containing new Pair(myx, myy, 0) without an explicit bounds check. Set tmp = -1, nearX = -1, and nearY = -1. RESULT bfs_setup]
            [COMMAND Run BFS while q is not empty: poll Pair now; if g[now.y][now.x] == 1, set tmp = now.dist, nearX = now.x, nearY = now.y and break; otherwise mark visited[now.y][now.x] = true and for v from 0 to 3 inclusive, compute nx = now.x + dx[v] and ny = now.y + dy[v]; if nx or ny is outside the inclusive range 0 to 2000 or visited[ny][nx] is true, skip; otherwise enqueue new Pair(nx, ny, now.dist + 1). RESULT bfs_result]
            [COMMAND Append "Case #" + t + ": " to ans. RESULT ans]
            [COMMAND Set ti = 0. If tmp != -1, search tour from index 0 upward for the first coordinate list equal to [nearX, nearY] and set ti to that index. RESULT ti]
            [COMMAND Evaluate the meeting-time condition using tmp and ti and follow the corresponding alternative flow. RESULT ans]
            [COMMAND After the outer test-case loop completes, print ans using System.out.print(ans). RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: tmp == -1 after BFS]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append "IMPOSSIBLE" followed by a newline to ans and continue to the next test case. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: tmp != -1 and tmp <= ti]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append the value tmp + ((ti - tmp + 1) / 2), computed with Java integer division, followed by a newline, to ans, and continue to the next test case. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: tmp != -1 and tmp > ti]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a 2001 by 2001 boolean visited2 grid initialized to false and set success = false. RESULT visited2_success]
            [COMMAND For i from ti upward while i < tour.size() and tmp - ti + i < tour.size(), get mfx = tour.get(tmp + i - ti).get(0), mfy = tour.get(tmp + i - ti).get(1), mark visited2[mfy][mfx] = true, then get ffx = tour.get(i).get(0) and ffy = tour.get(i).get(1). If visited2[ffy][ffx] is true, append i followed by a newline to ans, set success = true, and break. RESULT ans]
            [COMMAND If success is false after the loop, append "IMPOSSIBLE" followed by a newline to ans. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
