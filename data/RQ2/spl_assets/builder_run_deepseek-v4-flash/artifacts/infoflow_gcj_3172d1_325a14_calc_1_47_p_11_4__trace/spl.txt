[DEFINE_WORKER: "Computes a minimal cost ans by repeatedly accumulating prefix totals for each array position, optionally scanning backward when cur_d is still below d, and returns (ans, ret) with ret equal to the first element of a." calc]
    [INPUTS]
        <REF> n </REF>: int "Ignored parameter; immediately overwritten by len(a)."
        <REF> d </REF>: integer "Target count used in loop bounds, break condition cur_d > d, and ans update calculations."
        <REF> a </REF>: list "List of integers providing x = a[i], the first element ret = a[0], and backward-phase values a[j]."
        <REF> b </REF>: list "List of integers providing y = b[i] for each outer iteration."
        <REF> ct </REF>: mapping with get "Mapping used as ct.get(p, 0) to obtain tmp for each accumulated p."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "The pair (ans, ret), where ans is the minimal cost produced by all min updates and ret is the first element of a."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Overwrite n with len(a); set ans = d - 1 and ret = a[0]; if a is empty, a[0] raises IndexError. RESULT initial_state]
            [COMMAND For each i from 0 through n-1, execute the loop body: set x = a[i], y = b[i], cur_d = 0, p = 0, cut = 0. RESULT outer_i_loop]
            [COMMAND Within the outer i loop, run the first phase for j = 0 through d+1, i.e. j in range(d + 2); at each j, add x to p and set tmp = ct.get(p, 0). RESULT first_j_loop]
            [COMMAND Within the first j loop, compute per_d = p / x and enter the inner k loop over range(tmp * per_d). RESULT first_k_loop]
            [COMMAND Within the first k loop, increment cur_d by 1; if (k + 1) % per_d != 0, increment cut by 1. RESULT first_k_state_update]
            [COMMAND Within the first k loop after cur_d and cut are updated, compute xx = d / cur_d; if d % cur_d == 0, decrement xx by 1; update ans = min(ans, cut + xx * cur_d). RESULT ans_update_first_k]
            [COMMAND Within the first k loop, if cur_d > d, break out of the k loop; after the k loop, if cur_d > d, break out of the first j loop. RESULT first_phase_early_exit]
            [COMMAND After the first j loop, only if cur_d < d, run the second phase: for j from n-1 down through i+1, evaluate a[j] > p or a[j] % x != 0 with short-circuit OR; if the condition is true, set tmp = (a[j] / x) * y and enter the inner k loop over range(tmp). RESULT second_j_loop]
            [COMMAND Within the second k loop, increment cur_d by 1 and cut by 1; compute xx = d / cur_d; if d % cur_d == 0, decrement xx by 1; update ans = min(ans, cut + xx * cur_d). RESULT second_k_ans_update]
            [COMMAND Within the second k loop, if cur_d > d, break out of the k loop; after the k loop, if cur_d > d, break out of the backward j loop. RESULT second_phase_early_exit]
            [COMMAND After the optional second phase (or immediately after the first phase when the second is skipped), compute xx = d / cur_d; if d % cur_d == 0, decrement xx by 1; update ans = min(ans, cut + xx * cur_d). RESULT final_ans_update_per_i]
            [COMMAND After the outer i loop completes, return the tuple (ans, ret). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
