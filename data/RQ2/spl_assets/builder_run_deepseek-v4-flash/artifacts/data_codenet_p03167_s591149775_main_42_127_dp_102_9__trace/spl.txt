[DEFINE_WORKER: "Reads a grid of height h and width w, counts the number of paths from the top-left cell to the bottom-right cell while treating '#' cells as blocked, and prints the count modulo BIG." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers from standard input into h and w using scanf. RESULT h, w]
            [COMMAND Initialize i to 0 and j to 0, allocate a as an array of h char* pointers, and allocate dp as an array of w int* pointers. RESULT a, dp, i, j]
            [COMMAND While i is less than w, allocate dp[i] as an array of h int values and increment i. RESULT dp]
            [COMMAND While j is less than h, allocate a[j] as an array of w char values and increment j. RESULT a]
            [COMMAND Reset i to 0 and j to 0. RESULT i, j]
            [COMMAND While i is less than h, reset j to 0, read a string into a[i] using scanf with format string %s, then while j is less than w set dp[j][i] to 0; if a[i][j] equals '#' set dp[j][i] to -1; increment j after each column; increment i after each row. RESULT a, dp, i, j]
            [COMMAND Set dp[0][0] to 1, then set i to 1 and j to 1. RESULT dp, i, j]
            [COMMAND While i is less than h, if dp[0][i] equals -1 set dp[0][i] to 0, otherwise set dp[0][i] to dp[0][i-1]; increment i after each iteration. RESULT dp]
            [COMMAND While j is less than w, if dp[j][0] equals -1 set dp[j][0] to 0, otherwise set dp[j][0] to dp[j-1][0]; increment j after each iteration. RESULT dp]
            [COMMAND Set i to 1 and j to 1. RESULT i, j]
            [COMMAND While i is less than h, reset j to 1; while j is less than w, if dp[j][i] equals -1 set dp[j][i] to 0, otherwise set dp[j][i] to (dp[j-1][i] + dp[j][i-1]) % BIG; increment j after each inner iteration; increment i after each outer iteration. RESULT dp]
            [COMMAND Print dp[w-1][h-1] followed by a newline using printf. RESULT printed_value]
            [COMMAND Return 0 from main. RESULT exit_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
