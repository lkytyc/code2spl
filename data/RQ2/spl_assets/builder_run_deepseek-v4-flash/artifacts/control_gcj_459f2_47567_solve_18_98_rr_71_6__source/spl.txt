[DEFINE_WORKER: "Reads a target array b and initializes one unit at each position, then iteratively moves units one step left or right while recording movement strings until the array matches b or no progress is possible, and prints the movement steps or IMPOSSIBLE." solve]
    [INPUTS]
        <REF> sc </REF>: Scanner "Scanner used to read c and the c target values for b."
        <REF> num </REF>: int "Test case number used in the Case # output prefix."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize result to false, read integer c from sc, allocate integer arrays b and r of length c, create empty list l, set good to false, and set s to an empty string. RESULT initial_state]
            [COMMAND For each i from 0 to c-1 inclusive, read the next integer from sc into b[i] and set r[i] to 1. RESULT arrays_initialized]
            [COMMAND Enter an infinite while loop. At the top of each iteration, set result to true, then compare b[i] with r[i] for every i from 0 to c-1; if any pair differs, set result to false. RESULT equality_checked]
            [COMMAND If result is true, break out of the infinite while loop because the current r array exactly equals the target b array. RESULT equality_exit]
            [COMMAND Within the while loop, reset last to 0, set good to false, set s to an empty string, and create a fresh zero-filled integer array rr of length c. RESULT pass_initialized]
            [COMMAND Within the while loop, for each demand index i from 0 to c-1 where b[i] > 0, set want to b[i] and scan supply index j from last to c-1. If r[j] == 0, append '.' to s and continue. Otherwise subtract r[j] from want; if j > i and j > 0 and j < c-1, add r[j] to rr[j-1], set good to true, and append '/'; else if j < i and j > 0 and j < c-1, add r[j] to rr[j+1], set good to true, and append '\'; else add r[j] to rr[j] and append '.'. After processing a nonzero supply, if want <= 0, set last to j+1 and break the inner j-loop. RESULT movement_pass]
            [COMMAND After the demand sweep, assign rr to r, append the movement string s to list l, and if good is false, break out of the infinite while loop. RESULT state_updated]
            [COMMAND After the infinite while loop exits, branch on the final value of result. RESULT final_branch]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: result is false after the while loop.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print exactly 'Case #' + num + ': IMPOSSIBLE'. RESULT impossible_output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: result is true after the while loop.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build final string s by appending '.' for each i from 0 to c-1, so s contains c dots. RESULT final_dots]
            [COMMAND Append s to list l, print exactly 'Case #' + num + ': ' + l.size(), then print each string in l in its current order. RESULT solution_output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
