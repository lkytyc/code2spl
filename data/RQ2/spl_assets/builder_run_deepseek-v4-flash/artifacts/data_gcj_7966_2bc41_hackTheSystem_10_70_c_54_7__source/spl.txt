[DEFINE_WORKER: "Determines the number of swap movements needed to make the total charged-shot damage from an S/C program fit within a shield, returning IMPOSSIBLE when the number of shots exceeds the shield capacity." hackTheSystem]
    [INPUTS]
        <REF> softwareToHack </REF>: String[] "Array whose first element is the shield value as a string and whose second element is the S/C program string."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: String "Movement count as a string, or IMPOSSIBLE if the number of shots exceeds the shield capacity."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize shootDamage to 1L, shootCounter to 0L, damage to 0L, and movements to 0L. RESULT initial_state]
            [COMMAND Parse shield from softwareToHack[0] using Long.parseLong, and assign code = softwareToHack[1]. RESULT parsed_inputs]
            [COMMAND Set sizeScript = code.length() - 1 and lastIterableElement = sizeScript. RESULT index_state]
            [COMMAND Run the first loop with i initialized to 0, continuing while i < code.length(), and incrementing i after each iteration. In each iteration, if code.charAt(i) == 'S', add shootDamage to damage and increment shootCounter; if code.charAt(i) == 'C', multiply shootDamage by 2. RESULT first_scan_state]
            [COMMAND Immediately return String.valueOf(movements) if damage <= shield. RESULT damage_guard_result]
            [COMMAND Immediately return "IMPOSSIBLE" if shootCounter > shield. RESULT impossible_guard_result]
            [COMMAND Run the outer loop with i initialized to sizeScript, continuing while i >= 0, and decrementing i at the end of each outer iteration. RESULT outer_loop_state]
            [COMMAND Inside the outer loop, run the inner loop with j initialized to sizeScript - 1, continuing while j >= 0, and decrementing j at the end of each inner iteration. RESULT inner_loop_state]
            [COMMAND Within the inner loop, first check, using short-circuit evaluation, whether code.charAt(j) == 'S' && code.charAt(i) == 'S'; if true, decrement i and continue to the next inner-loop iteration. RESULT double_s_branch_state]
            [COMMAND Within the inner loop, if the first check did not match and code.charAt(i) == 'C', divide shootDamage by 2, decrement i, decrement lastIterableElement, and continue to the next inner-loop iteration. RESULT charge_branch_state]
            [COMMAND Within the inner loop, when neither of the first two checks matched, evaluate the short-circuit condition code.charAt(j) == 'C' && code.charAt(i) == 'S'; if true, increment movements, subtract shootDamage / 2 from damage, and immediately return String.valueOf(movements) if damage <= shield. RESULT swap_branch_start_state]
            [COMMAND Within the same swap branch, if the early return is not taken, copy code into a character array, swap c[j] with c[i], and assign code = String.valueOf(c). RESULT swap_executed_state]
            [COMMAND Within the same swap branch after the swap, if i == lastIterableElement, divide shootDamage by 2 and decrement i only when i > 0; otherwise increment i and add 2 to j. RESULT post_swap_index_state]
            [COMMAND After the outer and inner loops complete, return String.valueOf(movements). RESULT final_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: softwareToHack has fewer than two elements.]
        [LOG "ArrayIndexOutOfBoundsException raised while accessing softwareToHack[0] or softwareToHack[1]."]
        [THROW ArrayIndexOutOfBoundsException "The input array must contain at least two elements: shield and code."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: softwareToHack[0] is not a parseable long integer.]
        [LOG "NumberFormatException raised by Long.parseLong(softwareToHack[0])."]
        [THROW NumberFormatException "The shield value must be a valid long integer string."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
