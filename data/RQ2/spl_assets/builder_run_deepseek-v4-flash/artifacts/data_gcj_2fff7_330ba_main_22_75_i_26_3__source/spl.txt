[DEFINE_WORKER: "Reads T test cases, each with N rectangles and a budget, and prints the total rectangle perimeter plus the largest feasible additional amount not exceeding the remaining budget for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer T from stdin using scanf("%d\n", &T); scanf's return value is ignored, so if the read fails T is not guaranteed to be assigned. RESULT T]
            [COMMAND For each test case index i satisfying 0 <= i < T, execute the test-case body below. RESULT test_case_context]
            [COMMAND Within the test-case body, read N and P from stdin using scanf("%d %d\n", &N, &P) and ignore scanf's return value; set P_orig = P; allocate int arrays w, h, min_add and double arrays max_add, best_min, best_max, each of length N; initialize total = 0.0. RESULT test_data]
            [COMMAND Within the test-case body, for each j satisfying 0 <= j < N, read w[j] and h[j] using scanf("%d %d\n", w + j, h + j) and ignore scanf's return value; compute circ = 2 * (w[j] + h[j]); subtract circ from P; add circ to total; set min_add[j] = 2 * (w[j] < h[j] ? w[j] : h[j]); set max_add[j] = 2 * sqrt(w[j] * w[j] + h[j] * h[j]). RESULT rectangle_loop_done]
            [COMMAND Within the test-case body, after the rectangle-reading loop, call dpr("looking for %d\n", P); initialize best_valid = 0.0. RESULT best_valid_initialized]
            [COMMAND Within the best-prefix loop, for each j satisfying 0 <= j < N, initialize min1 = 0, max1 = 0, k1 = -1; then for each k satisfying 0 <= k < j, if best_min[k] + min_add[j] < P and best_max[k] > max1, select that k by setting k1 = k, min1 = best_min[k], max1 = best_max[k]. RESULT dp_selected_previous]
            [COMMAND Within the best-prefix loop, after the k scan for the current j, set best_min[j] = min1 + min_add[j] and best_max[j] = max1 + max_add[j]; log with dpr("cookie %d: using prev %d [%f..%f]\n", j, k1, best_min[j], best_max[j]); if best_min[j] <= P and best_max[j] > best_valid, set best_valid = best_max[j]. RESULT best_prefix_updated]
            [COMMAND After all best-prefix iterations for the test case, if best_valid > P, clamp best_valid = P. RESULT best_valid_clamped]
            [COMMAND Within the test-case body, call printf("Case #%d: %.6f\n", i + 1, total + best_valid). RESULT printed_case]
            [COMMAND Within the test-case body, free max_add, then min_add, then h, then w. RESULT arrays_freed]
            [COMMAND After all test cases complete, call exit(0). RESULT process_exited]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
