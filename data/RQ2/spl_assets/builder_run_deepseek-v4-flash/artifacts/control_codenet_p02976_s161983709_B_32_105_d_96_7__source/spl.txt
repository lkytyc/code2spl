[DEFINE_WORKER: "Builds an undirected graph from edge-pair input using adjacency lists and edge flags, runs a BFS traversal from vertex 0 that orients and consumes incident edges according to parity and per-vertex flags, then prints -1 if any flag remains set, otherwise prints the selected oriented edges in 1-based indices." B]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read n and m from input using LI(), unpacking the returned sequence into two integer variables. RESULT n, m]
            [COMMAND Initialize v as an empty list, then for each index from 0 to n-1 append a new empty list to v so that v becomes an adjacency list with n vertex lists. RESULT v]
            [COMMAND Initialize f as a list of n zeros and dic as a defaultdict whose default factory returns 0. RESULT f, dic]
            [COMMAND For i from 0 to m-1, read a and b using LI(), subtract 1 from both to convert them to 0-based indices, append b to v[a] and a to v[b], increment f[a] and f[b], and set dic[(a,b)] = 1 and dic[(b,a)] = 1. RESULT v, f, dic]
            [COMMAND Initialize ans to an empty list, d to a list of n zeros, q to deque([0]), and bfs to a list of n ones; then set bfs[0] = 0. RESULT ans, d, q, bfs]
            [COMMAND While q is not empty, pop the leftmost element x from q and set s to an empty list. For every neighbor y in v[x], if dic[(x,y)] is truthy append y to s, and if bfs[y] is truthy append y to q and set bfs[y] = 0. Then, if s is nonempty, consume s as follows: if len(s) is odd and d[x] is nonzero, for every y in s append (x,y) to ans, clear dic[(x,y)] and dic[(y,x)], and set d[x] = 0; if len(s) is odd and d[x] is zero, for every y in s[:-1] append (x,y) to ans and clear both dic entries, set d[x] = 0, let y = s[-1], toggle d[y] with d[y] ^= 1, append (y,x) to ans, and clear dic[(x,y)] and dic[(y,x)]; if len(s) is even and d[x] is zero, for every y in s append (x,y) to ans, clear both dic entries, and set d[x] = 0; if len(s) is even and d[x] is nonzero, for every y in s[:-1] append (x,y) to ans and clear both dic entries, set d[x] = 0, let y = s[-1], toggle d[y] with d[y] ^= 1, append (y,x) to ans, and clear dic[(x,y)] and dic[(y,x)]. RESULT ans, d, dic]
            [COMMAND After the BFS loop, if sum(d) is zero, print i+1 and j+1 for each pair (i,j) in ans. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After the BFS loop, sum(d) is nonzero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
