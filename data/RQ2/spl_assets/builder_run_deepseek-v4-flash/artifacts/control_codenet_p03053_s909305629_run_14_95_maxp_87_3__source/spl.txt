[DEFINE_WORKER: "Reads a grid of '.' and '#' cells, performs a level-order BFS that blackens adjacent '.' cells each round, and prints the number of expansion rounds needed for all cells to become black (or does not terminate if some '.' cells are unreachable from the initial black cells)." run]
    [INPUTS]
        <REF> stdin </REF>: System.in "The standard input provides h and w as integers, followed by h strings of length w consisting of '.' and '#' characters."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: int "The computed integer ans is printed on a line by PrintWriter and flushed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create Scanner sc for System.in and read the first two tokens as integers h and w. RESULT sc, h, w]
            [COMMAND Allocate char array a with dimensions h by w, int array atmp of length h, and int array blackar with dimensions h*w by 2; set max=0 and p=0. RESULT a, atmp, blackar, max, p]
            [COMMAND For each row index i from 0 to h-1, read the next token from sc, convert it to a char array, and assign it to a[i]. RESULT a]
            [COMMAND For each row index i from 0 to h-1 and for each column counter j of type char starting at 0 while j is less than w, if a[i][j] equals '#', set blackar[p][0]=i, blackar[p][1]=j, and increment p. RESULT blackar, p]
            [COMMAND After the scanning loops, set maxp=p and p2=p, then set p=0. RESULT maxp, p2, p]
            [COMMAND While p is less than h*w, start one outer flood-fill layer: set hap=0 (this variable is never used), then run the inner for loop with no initialization, condition p is less than maxp, and p incrementing by 1 after each iteration; the condition uses the maxp value from before this layer, so cells appended to blackar during this layer are not processed until the next layer. RESULT p, maxp, p2, max]
            [COMMAND Within the inner for loop, read the queued black cell by setting i=blackar[p][0] and j=blackar[p][1]. RESULT i, j]
            [COMMAND Within the inner for loop for the current black cell (i,j), if i-1 is greater than or equal to 0 and a[i-1][j] equals '.', set a[i-1][j]='#', set blackar[p2][0]=i-1, blackar[p2][1]=j, and increment p2. RESULT a, blackar, p2]
            [COMMAND Within the inner for loop for the current black cell (i,j), if j-1 is greater than or equal to 0 and a[i][j-1] equals '.', set a[i][j-1]='#', set blackar[p2][0]=i, blackar[p2][1]=j-1, and increment p2. RESULT a, blackar, p2]
            [COMMAND Within the inner for loop for the current black cell (i,j), if i+1 is less than h and a[i+1][j] equals '.', set a[i+1][j]='#', set blackar[p2][0]=i+1, blackar[p2][1]=j, and increment p2. RESULT a, blackar, p2]
            [COMMAND Within the inner for loop for the current black cell (i,j), if j+1 is less than w and a[i][j+1] equals '.', set a[i][j+1]='#', set blackar[p2][0]=i, blackar[p2][1]=j+1, and increment p2. RESULT a, blackar, p2]
            [COMMAND Within the outer while loop, after the inner for loop finishes, set maxp=p2 and increment max by 1. RESULT maxp, max]
            [COMMAND After the outer while loop exits, decrement max by 1 and assign ans=max. RESULT max, ans]
            [COMMAND Create PrintWriter out for System.out, write ans to out using println, and flush out. RESULT out]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The grid has h*w greater than zero but contains no initial '#' cells, or it contains at least one '.' cell that is not reachable from any initial '#' cell, so after some layer the inner for loop adds no new cells while p remains less than h*w.]
        [SEQUENTIAL_BLOCK]
            [COMMAND The outer while loop never terminates because p is less than h*w and maxp stops increasing; no answer is printed. RESULT infinite_loop]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: sc.next() is called after the input is exhausted while reading h, w, or a grid row.]
        [LOG "NoSuchElementException from Scanner.next() because no token is available."]
        [THROW NoSuchElementException "Thrown when an input token is required but the input stream has no more tokens."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The h token or w token is not a valid integer.]
        [LOG "NumberFormatException from Integer.parseInt() while parsing h or w."]
        [THROW NumberFormatException "Thrown when a non-numeric token is supplied for h or w."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: h or w is negative, or h*w overflows to a negative int before the array allocation.]
        [LOG "NegativeArraySizeException while allocating a, atmp, or blackar with a negative computed length."]
        [THROW NegativeArraySizeException "Thrown when an array is allocated with a negative size."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: A grid row token contains fewer than w characters.]
        [LOG "ArrayIndexOutOfBoundsException in the initial scan when a[i][j] is accessed beyond the converted row length."]
        [THROW ArrayIndexOutOfBoundsException "Thrown when the row string is shorter than the declared width w."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
