[DEFINE_WORKER: "Returns the minimum number of adjacent swaps that move 'S' characters left over 'C' characters to reduce the cumulative damage from 'S' positions to at most d, or -1 if impossible." minswaps]
    [INPUTS]
        <REF> p </REF>: char* "Array of characters 'S' and 'C' representing the sequence; it is modified in place when swaps are performed."
        <REF> d </REF>: int "Maximum allowed damage threshold."
        <REF> len </REF>: int "Length of the array p."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Minimum number of swaps required so that damage <= d; 0 when damage is already at most d or no swap is needed; -1 when damage cannot be reduced to at most d."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If len == 2, return according to the two-character special-case rules and do not continue with the general algorithm. RESULT special_case_result]
            [COMMAND Declare integer variables i, strength=1, damage=0, temp, k=0, swaps=0, flag=0, and allocate integer array a with malloc(sizeof(int)*len) for tracking strength values. RESULT local_state]
            [COMMAND For each i from 0 to len-1 inclusive, scan p[i]: if p[i]=='S', add strength to damage and store strength into a[k++]; otherwise multiply strength by 2 and store the doubled strength into a[k++]. RESULT initial_damage_and_a]
            [COMMAND If damage <= d, return 0 immediately without performing any swaps. RESULT return_0_early]
            [COMMAND Set i = len - 1 and begin the outer while(!flag) loop. RESULT outer_loop_started]
            [COMMAND At the start of each outer-loop pass, set flag = 1 and reset i = len - 1; then, while both damage-d > 0 and i > 0, enter the inner loop. RESULT inner_loop_started]
            [COMMAND Inside the inner loop, while p[i] != 'S', decrement i to find the rightmost 'S' at or before the current index. RESULT rightmost_s_index]
            [COMMAND Inside the inner loop, if p[i-1] == 'S', decrement i by 1; otherwise, because p[i-1] is 'C', swap the CS pair to SC by setting p[i]='C' and p[i-1]='S', set flag=0, subtract a[i] from damage, update a[i-1] and a[i] using the rule a[i-1]=1 and a[i]=2 when i-1==0, or a[i-1]=a[i-2] and a[i]=2*a[i-1] otherwise, add the updated a[i-1] to damage, and increment swaps. RESULT updated_p_damage_a_swaps]
            [COMMAND If after this processing i == 1 and p[0] == 'S', break out of the inner while loop only. RESULT inner_loop_break]
            [COMMAND Repeat the outer while loop until flag remains 1 after an entire pass, meaning no swap was made in that pass; a swap sets flag=0 and causes another pass. RESULT final_damage_and_swaps]
            [COMMAND Free the allocated array a with free(a). RESULT memory_freed]
            [COMMAND If damage-d <= 0, return swaps; otherwise return -1. RESULT final_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: len == 2 and p[0] == 'S' and p[1] == 'C']
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len == 2 and p[0] == 'S' and p[1] == 'S']
        [SEQUENTIAL_BLOCK]
            [COMMAND If d < 2, return -1. RESULT return_minus_1]
            [COMMAND If d >= 2, return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len == 2 and p[0] == 'C' and p[1] == 'S']
        [SEQUENTIAL_BLOCK]
            [COMMAND If d < 2, return 1. RESULT return_1]
            [COMMAND If d >= 2, return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: len == 2 and none of the SC, SS, or CS patterns match]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: In the general algorithm, after the initial scan, damage <= d]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 0 immediately without performing swap passes. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: In the general algorithm, after the swap passes and free(a), damage - d <= 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return the accumulated swaps value. RESULT return_swaps]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: In the general algorithm, after the swap passes and free(a), damage - d > 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return -1 to indicate the damage threshold cannot be met. RESULT return_minus_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
