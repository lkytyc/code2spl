[DEFINE_WORKER: "Reads a number of test cases; for each test case, reads an integer sequence, separates values by original position parity into two arrays, sorts both arrays, and checks whether the sorted parity arrays can be interleaved to form a non-decreasing sequence, printing the first violation position or OK." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to the main method; not used by the program."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner reader reading from System.in. RESULT reader]
            [COMMAND Read the next integer from reader into numberOfTests. RESULT numberOfTests]
            [COMMAND Initialize a to 0 and b to 0. RESULT a, b]
            [COMMAND For m initialized to 0, continue while m < numberOfTests, and increment m by 1 after each iteration, execute the following test-case body. RESULT m]
            [COMMAND Within the outer loop over m, read the next integer from reader into n. RESULT n]
            [COMMAND Within the outer loop over m, branch on the value of n % 2: if n % 2 == 1, set a = n / 2 + 1 and b = n / 2; otherwise set a = n / 2 and b = n / 2. RESULT a, b]
            [COMMAND Within the outer loop over m, allocate an int array oddArray of length a and an int array evenArray of length a. RESULT oddArray, evenArray]
            [COMMAND Within the outer loop over m, initialize c to 0 and d to 0. RESULT c, d]
            [COMMAND Within the outer loop over m, for i initialized to 0, continue while i < n, and increment i by 1 after each iteration, read one integer from reader; if i % 2 == 0, store it in oddArray[c] and increment c; otherwise store it in evenArray[d] and increment d. RESULT oddArray, evenArray, c, d]
            [COMMAND Within the outer loop over m, unconditionally assign Integer.MAX_VALUE to evenArray[evenArray.length - 1], overwriting whatever value is currently stored in that slot. RESULT evenArray]
            [COMMAND Within the outer loop over m, create a Solution instance mms. RESULT mms]
            [COMMAND Within the outer loop over m, call mms.sort(oddArray) in place and mms.sort(evenArray) in place. RESULT oddArray, evenArray]
            [COMMAND Within the outer loop over m, initialize x to 0. RESULT x]
            [COMMAND Within the outer loop over m, for h initialized to 0, continue while h < oddArray.length, and increment h by 1 after each iteration, perform the following checks inside a try-finally whose finally block is empty. RESULT h]
            [COMMAND Within the outer loop over m and the comparison loop over h, if oddArray[h] > evenArray[h], print "Case #" + (m + 1) + ": " + (2 * h), increment x, and break out of the comparison loop. RESULT x]
            [COMMAND Within the outer loop over m and the comparison loop over h, if oddArray[h] <= evenArray[h] and h == oddArray.length - 1, break out of the comparison loop. RESULT comparison_loop]
            [COMMAND Within the outer loop over m and the comparison loop over h, if oddArray[h] <= evenArray[h], h is not the last index, and evenArray[h] > oddArray[h + 1], print "Case #" + (m + 1) + ": " + (2 * h + 1), increment x, and break out of the comparison loop. RESULT x]
            [COMMAND Within the outer loop over m and the comparison loop over h, if oddArray[h] <= evenArray[h], h is not the last index, and evenArray[h] <= oddArray[h + 1], continue to the next iteration of the comparison loop. RESULT comparison_loop]
            [COMMAND Within the outer loop over m, after the comparison loop, if x == 0, print "Case #" + (m + 1) + ": OK". RESULT printed_output]
            [COMMAND After the outer loop over m finishes, close reader. RESULT reader]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
