[DEFINE_WORKER: "Reads n and an integer sequence a from line-based input, rejects invalid prefixes by printing 0, and for allowed inputs computes the number of valid continuations using a state-machine scan over a[3..n-1], printing the result modulo 1000000007, with special n=1 and n=2 output behavior." main]
    [INPUTS]
        <REF> given </REF>: callable "Callable that returns one line of input as a string; defaults to sys.stdin.readline."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set up local input helpers: input() calls given().rstrip(), LMIIS() reads a whitespace-separated line as a list of ints, II() reads a single int, and XLMIIS(x) reads x such lines as a list of lists of ints. RESULT input_helpers]
            [COMMAND Read n using II(); read the sequence a using LMIIS(). RESULT n_a]
            [COMMAND If a[0] != 0, print 0 and terminate the program immediately. RESULT guard_a0]
            [COMMAND If n == 1: if a[0] == 0 print 3, otherwise print 0; then terminate the program immediately. RESULT case_n_1]
            [COMMAND Else if n == 2: perform two independent checks: if a[1] == 0, print 3; if a[1] == 1, print 6, otherwise print 0; then terminate the program immediately. RESULT case_n_2]
            [COMMAND If (a[1], a[2]) == (0, 0), set x=1, y=1, z=1, res=1; else if (a[1], a[2]) == (0, 1), set x=2, y=1, z=0, res=2; else if (a[1], a[2]) == (1, 2), set x=3, y=0, z=0, res=1; otherwise print 0 and terminate the program immediately. RESULT initial_state]
            [COMMAND Set MOD = 1000000007 (10**9 + 7). RESULT MOD]
            [COMMAND For each i from 3 through n-1 inclusive, inspect a[i] against the current x, y, z in the following priority order: if a[i] == x and a[i] != y, increment x; else if a[i] == x and a[i] == y and a[i] != z, set res = res*2 % MOD and increment x; else if a[i] == x and a[i] == y and a[i] == z, set res = res*3 % MOD and increment x; else if a[i] == y and a[i] != z, increment y; else if a[i] == y and a[i] == z, set res = res*2 % MOD and increment y; else if a[i] == z, increment z; otherwise print 0 and terminate the program immediately. RESULT validation_loop]
            [COMMAND After the loop, if y == 0, set res = res*3 % MOD; otherwise set res = res*6 % MOD. RESULT final_res]
            [COMMAND Print res. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
