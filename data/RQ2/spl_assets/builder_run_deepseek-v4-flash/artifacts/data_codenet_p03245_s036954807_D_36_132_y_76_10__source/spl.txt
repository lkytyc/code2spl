[DEFINE_WORKER: "Reads point lists, checks that every pairwise Manhattan distance is even (printing -1 and returning if any odd pair is found), otherwise constructs and prints edge lengths and per-point direction strings from padded reversed binary coordinates." D]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer n using I(). RESULT n]
            [COMMAND Initialize p as an empty list. RESULT p]
            [COMMAND For i from 0 to n-1 inclusive, read one list of integers using LI() and append it to p. RESULT p]
            [COMMAND Initialize b as an empty list. RESULT b]
            [COMMAND Validate all point pairs: for i from 0 to n-1 inclusive, unpack x and y from p[i]; for j from 0 to i-1 inclusive, unpack s and t from p[j]; compute (abs(x-s)+abs(y-t)) % 2; if the result is nonzero, follow the invalid-parity alternative flow. RESULT validation_result]
            [COMMAND Initialize ans as an empty list. RESULT ans]
            [COMMAND Preprocess each point i from 0 to n-1 inclusive: set f=0; unpack x,y=p[i]; if (x+y)%2 is 0, decrement x by 1 and if ans is empty append 1 to ans; if x is now negative, negate x and set f |= 1; if y is negative, negate y and set f |= 2; convert x to a reversed list of binary digits bx from bin(x)[2:], convert y similarly to by; pad bx and by with zeros to length 39 when shorter, leaving longer lists unchanged; append [bx, by, f] to b. RESULT b]
            [COMMAND For i from 0 to 38 inclusive, append 1 << i to ans. RESULT ans]
            [COMMAND Initialize v = [[] for _ in range(n)] as a list of n independent empty lists. RESULT v]
            [COMMAND For each i from 0 to n-1 inclusive, unpack x,y,f = b[i]; if x[0] is truthy, set p=1 and append 0 to v[i]; otherwise set p=0 and append 1 to v[i]. RESULT v]
            [COMMAND For each i from 0 to n-1 inclusive and each j from 1 to 38 inclusive, apply the carry-state rules using current p, x[j], y[j], and v[i]: if p is 1 and x[j] and y[j] are both 1, toggle v[i][-1] with XOR 3, set x[j+1]=1, append 1 to v[i], and set p=0; if p is 1, x[j]=1, y[j]=0, append 0 to v[i] and keep p=1; if p is 1, x[j]=0, y[j]=1, append 1 to v[i] and set p=0; if p is 1 and x[j] and y[j] are both 0, toggle v[i][-1] with XOR 3, append 0 to v[i], and keep p=1; if p is 0 and y[j] and x[j] are both 1, toggle v[i][-1] with XOR 3, set y[j+1] = y[j+1] XOR 1, append 0 to v[i], and set p=1; if p is 0, y[j]=1, x[j]=0, append 1 to v[i] and keep p=0; if p is 0, y[j]=0, x[j]=1, append 0 to v[i] and set p=1; if p is 0 and y[j] and x[j] are both 0, toggle v[i][-1] with XOR 3, append 1 to v[i], and keep p=0. RESULT v]
            [COMMAND For each i from 0 to n-1 inclusive and each j from 0 to 38 inclusive, apply the final flag correction to v[i][j]: if v[i][j] is 0 or 3, then if f & 1 is nonzero set v[i][j] = v[i][j] XOR 3; otherwise if v[i][j] is 1 or 2, then if f & 2 is nonzero set v[i][j] = v[i][j] XOR 3. RESULT v]
            [COMMAND Set p to the constant mapping list ['R', 'U', 'D', 'L']. RESULT p]
            [COMMAND Print len(ans) on its own line, then print all elements of ans separated by spaces on the next line. RESULT printed_ans]
            [COMMAND For each direction sequence i in v: if len(ans) equals 40, print R without a newline; then for each j in i, print p[j] without a newline; after finishing the sequence, print a newline. RESULT printed_paths]
            [COMMAND Return from the function with no value. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During pair validation, there exist i and j with 0 <= j < i < n such that (abs(p[i][0]-p[j][0]) + abs(p[i][1]-p[j][1])) % 2 is nonzero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1. RESULT printed_invalid_marker]
            [COMMAND Return immediately from the function, skipping all remaining steps. RESULT early_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
