[DEFINE_WORKER: "Reads an array, finds the first non-zero element to set the active sign, then scans the rest of the array counting zero-sum skips and sign-correction operations, and prints the total count." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> ans </REF>: int "The total number of zero-sum skips and sign-correction operations, printed to standard output."
        <REF> return_code </REF>: int "Always 0, returned by main."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integer n, array a of size 100000, integers i and j, and character sign; initialize sum to 0 and ans to 0. RESULT none]
            [COMMAND Read one integer from standard input into n, ignoring scanf's return value. RESULT n]
            [COMMAND For i starting at 0 while i < n, read one integer from standard input into a[i], ignoring scanf's return value. RESULT a]
            [COMMAND Scan a with i starting at 0 while i < n to find the first non-zero element: if a[i] > 0, set sign to 'p' and break; if a[i] < 0, set sign to 'm' and break; if a[i] == 0, continue to the next i. If no non-zero element is found, sign remains uninitialized and the loop ends with i equal to n. RESULT sign]
            [COMMAND Start the second loop with the current i; continue for (; i < n; i++). At each iteration add a[i] to sum. If i is n because all elements were zero, this loop is skipped. RESULT sum]
            [COMMAND Within the second loop, if sum equals 0, scan j starting at i+1 while j < n: if a[j] equals sum, continue; if sign is 'p', increment sum, increment ans, set i to j, and break; if sign is 'm', decrement sum, increment ans, set i to j, and break; if sign is neither 'p' nor 'm', continue scanning. After this scan, set i to the final value of j and increment ans; this zero-sum branch skips the sign-flip checks below. RESULT ans]
            [COMMAND If the sum==0 branch was not taken, evaluate the sign-flip checks in order: if sum > 0 and sign == 'p', set sign to 'm'; else if sum < 0 and sign == 'm', set sign to 'p'; else, if sign == 'p', add abs(sum)+1 to ans, add abs(sum)+1 to sum, and set sign to 'm'; if sign == 'm', add abs(sum)+1 to ans, subtract abs(sum)+1 from sum, and set sign to 'p'; if sign is neither 'p' nor 'm', take no action. RESULT sign]
            [COMMAND After the second loop completes, print ans followed by a newline to standard output. RESULT ans]
            [COMMAND Return 0 from main. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
