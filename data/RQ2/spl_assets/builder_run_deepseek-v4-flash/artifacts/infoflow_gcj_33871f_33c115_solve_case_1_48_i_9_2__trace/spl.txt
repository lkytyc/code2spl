[DEFINE_WORKER: "Reconciles positive and negative receipt times, simulates a propagation ordering to assign absolute times, and computes edge latencies returned as a space-separated string." solve_case]
    [INPUTS]
        <REF> in_times </REF>: list of integers "Per-node receipt times; positive values are absolute times and negative values represent delayed receipts using their absolute value."
        <REF> edges </REF>: list of integer pairs "Edge endpoint pairs as 1-based node indices; they are converted to zero-based indices before use."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Space-separated latency values, one per edge in input order; edges not assigned a latency retain the sentinel value 1000000."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize latencies to a list containing 1000000 repeated len(edges) times. RESULT latencies]
            [COMMAND Convert every edge pair (a,b) in edges to (a-1,b-1) and reassign edges to the converted zero-based list. RESULT edges]
            [COMMAND Initialize abs_times to a list containing 1000000 repeated len(in_times) times. RESULT abs_times]
            [COMMAND For each index and value i in in_times, if i is strictly greater than 0, set abs_times[index] to i. RESULT abs_times]
            [COMMAND Create pos_times as an empty list. RESULT pos_times]
            [COMMAND For each index and value i in in_times, if i is strictly less than 0, append (abs(i), index) to pos_times. RESULT pos_times]
            [COMMAND Sort pos_times in ascending tuple order. RESULT pos_times]
            [COMMAND Initialize current_time to 0 and received to 1. RESULT simulation_state]
            [COMMAND While pos_times is non-empty, if the first tuple's key equals received: increment current_time by 1, then while pos_times is non-empty and the first tuple's key equals received, increment new_received by 1, set abs_times at the tuple's index to current_time, and remove that tuple from the front of pos_times; after consuming the group, add new_received to received. Otherwise: build next_times from all abs_times values strictly greater than current_time, set current_time to min(next_times), compute received_count as the number of abs_times values less than or equal to current_time, and leave pos_times unchanged. RESULT simulation_state]
            [COMMAND Prepend 0 to abs_times by reassigning abs_times to [0] + abs_times. RESULT abs_times]
            [COMMAND Build sorted_receipt as a list of (time, index) pairs from enumerate(abs_times) and sort it ascending. RESULT sorted_receipt]
            [COMMAND For each (time, comp) in sorted_receipt[1:], scan edges in order; for each edge (A,B) with index index, if A == comp and abs_times[A] > abs_times[B], set latencies[index] to abs_times[A] - abs_times[B] and break out of the edge scan; if that condition did not break, then if B == comp and abs_times[B] > abs_times[A], set latencies[index] to abs_times[B] - abs_times[A] and break out of the edge scan. RESULT latencies]
            [COMMAND Join the latencies values with spaces and return the resulting string. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: In the non-matching branch of the outer while loop, no element of abs_times is strictly greater than current_time, so next_times is empty.]
        [LOG "ValueError: min() attempted on an empty next_times sequence"]
        [THROW ValueError "Raised by min(next_times) when next_times is empty."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
