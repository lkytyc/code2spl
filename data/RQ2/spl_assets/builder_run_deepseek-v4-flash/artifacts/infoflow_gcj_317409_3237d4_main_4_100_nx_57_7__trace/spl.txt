[DEFINE_WORKER: "Simulates a tour from each move string and uses BFS to find the earliest time at which a moving character can meet the tour, printing a result or IMPOSSIBLE for each test case." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not used by the program."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize direction arrays dx = {0, 1, 0, -1} and dy = {1, 0, -1, 0}. RESULT dx_dy]
            [COMMAND Create Scanner sc reading from System.in through BufferedReader and InputStreamReader. RESULT sc]
            [COMMAND Create StringBuilder ans to accumulate all output. RESULT ans]
            [COMMAND Read integer T from sc as the number of test cases. RESULT T]
            [COMMAND For each test case t from 1 to T inclusive, execute the following case processing. RESULT case_loop]
            [COMMAND Inside the case loop for t, read integer X, integer Y, and string M from sc. RESULT X_Y_M]
            [COMMAND Inside the case loop for t, create int grid g with dimensions 2001 by 2001, initially all zero. RESULT g]
            [COMMAND Inside the case loop for t, set current position px = 1000 and py = 1000. RESULT start_position]
            [COMMAND Inside the case loop for t, create empty tour list, add a first coordinate list [px, py] to tour, and set g[py][px] = 1. RESULT tour_initialized]
            [COMMAND Inside the case loop for t, for each index i from 0 to M.length() - 1 inclusive, read d = M.charAt(i); if d is N decrement py, if d is S increment py, if d is E increment px, if d is W decrement px; then set g[py][px] = 1 and append a new coordinate list [px, py] to tour. RESULT tour_built]
            [COMMAND Inside the case loop for t, create boolean visited array with dimensions 2001 by 2001, initially false. RESULT visited]
            [COMMAND Inside the case loop for t, compute BFS start coordinates myx = 1000 - X and myy = 1000 + Y. RESULT bfs_start]
            [COMMAND Inside the case loop for t, create LinkedList q and add a Pair with coordinates (myx, myy) and distance 0; initialize tmp = -1, nearX = -1, nearY = -1. RESULT bfs_state]
            [COMMAND Inside the case loop for t, while q is not empty, poll Pair now; if g[now.y][now.x] == 1, set tmp = now.dist, nearX = now.x, nearY = now.y and break; otherwise mark visited[now.y][now.x] = true and for each v from 0 to 3 inclusive compute nx = now.x + dx[v] and ny = now.y + dy[v]; if nx < 0, ny < 0, nx >= visited[0].length, or ny >= visited.length, skip this neighbor; if visited[ny][nx] is true, skip this neighbor; otherwise add a Pair with coordinates (nx, ny) and distance now.dist + 1 to q. RESULT bfs_result]
            [COMMAND Inside the case loop for t, append 'Case #' + t + ': ' to ans. RESULT ans_case_header]
            [COMMAND Inside the case loop for t, if tmp == -1, append the text IMPOSSIBLE followed by a newline to ans and continue to the next test case. RESULT impossible_output]
            [COMMAND Inside the case loop for t, search tour from index 0 to tour.size() - 1 for the first coordinate list whose x equals nearX and whose y equals nearY; store the found index as ti and break when found. RESULT ti]
            [COMMAND Inside the case loop for t, create boolean visited2 array with dimensions 2001 by 2001, initially false. RESULT visited2]
            [COMMAND Inside the case loop for t, if tmp <= ti, append the integer result of tmp + (ti - tmp + 1) / 2 followed by a newline to ans. RESULT short_intercept_output]
            [COMMAND Inside the case loop for t, when tmp > ti, set success = false; for i from ti while i < tour.size() and tmp - ti + i < tour.size(), incrementing i, mark visited2 at the future tour position tour.get(tmp + i - ti), then check the current tour position tour.get(i); if that current position is already marked in visited2, append i followed by a newline to ans, set success = true, and break; after the loop, if success is false, append the text IMPOSSIBLE followed by a newline to ans. RESULT catch_up_output]
            [COMMAND After the case loop completes, print the accumulated ans to standard output using System.out.print(ans). RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
