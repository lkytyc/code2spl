[DEFINE_WORKER: "Reads the next signed decimal long value from the underlying byte input, handling an optional leading '-' and rejecting printable non-digit input with NumberFormatException." nextLong]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: long "The parsed signed long value accumulated from the digit sequence, negated when minus is true."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check whether hasNext() returns false. If it does, stop normal execution and throw NoSuchElementException; otherwise continue. RESULT hasNextGuard]
            [COMMAND Initialize long accumulator n to 0. RESULT n]
            [COMMAND Initialize boolean sign flag minus to false. RESULT minus]
            [COMMAND Call readByte() and assign the returned int to b. RESULT b]
            [COMMAND Check whether b is the leading '-' character. If it is, use the alternative flow for a leading minus sign; if it is not, continue with b unchanged and minus false. RESULT signChecked]
            [COMMAND Validate that b is a decimal digit. If b < '0' is true or '9' < b is true, throw NumberFormatException and exit; otherwise continue with b as the first digit. RESULT b]
            [COMMAND Enter the infinite while(true) loop. Within this loop, if b is between '0' and '9' inclusive, multiply n by 10, add (b - '0') to n, call readByte() to assign the next byte to b, and continue the loop. RESULT n, b]
            [COMMAND Inside the same while(true) loop, if b equals -1 or isPrintableChar(b) returns false, exit the loop and return minus ? -n : n. RESULT returnValue]
            [COMMAND Inside the same while(true) loop, if b is a printable character that is not a digit, throw NumberFormatException and exit. RESULT exception]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b == '-' after the initial readByte() call]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set the sign flag minus to true. RESULT minus]
            [COMMAND Call readByte() and assign the returned byte to b, consuming the '-' before digit parsing. RESULT b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: hasNext() returns false]
        [LOG "NoSuchElementException: no next element is available"]
        [THROW NoSuchElementException "Indicates that the iterator/input has no more elements to read."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: After optional sign handling, b is outside the inclusive digit range '0'..'9' (b < '0' || '9' < b)]
        [LOG "NumberFormatException: first character of the number is not a decimal digit"]
        [THROW NumberFormatException "Indicates that the numeric token does not begin with a valid decimal digit."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: Inside the while(true) loop, b is a printable character that is not a digit]
        [LOG "NumberFormatException: printable non-digit encountered inside the number"]
        [THROW NumberFormatException "Indicates that the input contains an invalid printable character in the decimal token."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
