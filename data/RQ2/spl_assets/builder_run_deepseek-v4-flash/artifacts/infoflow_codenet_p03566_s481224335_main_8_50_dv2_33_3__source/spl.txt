[DEFINE_WORKER: "Reads N, time list T, and velocity list V; computes and prints a combined area value by summing acceleration trapezoids, classified rectangle contributions, and the final right-triangle term, with a separate short-circuit special case when N equals 1." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read one integer from standard input using ii() and store it as N. RESULT N]
            [COMMAND Read a line of whitespace-separated integers using mi(), materialize the iterator into a list, and store it as T. RESULT T]
            [COMMAND Read the next line of whitespace-separated integers using mi(), materialize the iterator into a list, and store it as V. RESULT V]
            [COMMAND On the non-N==1 path, initialize acce_areas to 0. RESULT acce_areas]
            [COMMAND On the non-N==1 path, initialize pre_v to 0. RESULT pre_v]
            [COMMAND Loop i from 0 to N-1: read v = V[i]; add ((v + pre_v) * abs(v - pre_v)) / 2 to acce_areas; then set pre_v = v. RESULT acce_areas]
            [COMMAND Append 0 to V. RESULT V]
            [COMMAND Append 0 to T. RESULT T]
            [COMMAND Initialize pre_v to 0. RESULT pre_v]
            [COMMAND Initialize rect_sums to 0. RESULT rect_sums]
            [COMMAND Loop i from 0 to N-1: assign v = V[i] and t = T[i]; assign next_v = V[i+1] from the appended V; compute dv1 = abs(v - pre_v); compute dv2 = abs(next_v - v); set lu = (pre_v < v), ld = not lu, ru = (v < next_v), and rd = not ru; initialize bottom = 0; if lu and rd are true add t - (dv1 + dv2) to bottom; if ld and ru are true add t to bottom; if lu and ru are true add t - dv1 to bottom; if ld and rd are true add t - dv2 to bottom; add bottom * v to rect_sums; then set pre_v = v. RESULT rect_sums]
            [COMMAND Compute ans = acce_areas + rect_sums + ((V[-2] ** 2) / 2), using the appended V list so V[-2] is the last original velocity value. RESULT ans]
            [COMMAND Print ans. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After reading N, T, and V, N equals 1; this special path prints and exits, so the main flow does not run.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign t = T[0]. RESULT t]
            [COMMAND Assign v = V[0]. RESULT v]
            [COMMAND Compute ans = (t - v) * v. RESULT ans]
            [COMMAND If ans is less than 0, recompute ans as (t ** 2) / 4. RESULT ans]
            [COMMAND Print ans. RESULT stdout]
            [COMMAND Exit the program. RESULT program_exit]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
