            [COMMAND Inside the outer loop over i, read integer p (number of people) and integer q (grid dimension) from sc. RESULT p_q]
            [COMMAND Inside the outer loop over i, create an ArrayList<Person> named ppl with initial capacity p. RESULT ppl]
            [COMMAND Within the outer loop over i, start a person-reading loop for j from 0 to p-1 inclusive. RESULT person_read_loop_j]
            [COMMAND Within that person-reading loop, read the first coordinate integer (original variable b) and retain it as person_y. RESULT person_y]
            [COMMAND Within that person-reading loop, read the next integer raw_x and compute person_x = q - raw_x - 1. RESULT person_x]
            [COMMAND Within that person-reading loop, read the direction token c from sc. RESULT direction_c]
            [COMMAND Within that person-reading loop, initialize d=0 and e=0; if c equals "N" set e=-1, else if c equals "S" set e=1, else if c equals "E" set d=1, else set d=-1 (treating any other token as W). RESULT direction_d_e]

            [COMMAND Within the ray loop over each Person a, set x = a.x and y = a.y. RESULT current_x_y]
            [COMMAND Within the ray loop, if a.xDir == 0 (vertical ray), for each row j starting at x + a.yDir and stepping by a.yDir while j >= 0 and j < q, increment b[j][k] for every column k from 0 to q-1 inclusive; the starting cell is excluded. RESULT vertical_coverage]
            [COMMAND Within the ray loop, else (a.xDir != 0, horizontal ray), for every row j from 0 to q-1 inclusive, increment b[j][k] for each column k starting at y + a.xDir and stepping by a.xDir while k >= 0 and k < q; the starting cell is excluded. RESULT horizontal_coverage]

            [COMMAND Scan b in column-major order: for c from 0 to q-1 inclusive, for r from 0 to q-1 inclusive, if b[r][c] equals max then set row[num] = q - r - 1, col[num] = c, and increment num. RESULT max_coord_candidates]

RESULT f_min_sum]
            [COMMAND Initialize m = Integer.MAX_VALUE and ArrayList<Person> g with capacity f.size(). For each Person in f, if m > person.x then update m to person.x, clear g, and add the person; else if m == person.x then add the person to g. RESULT g_min_row]
