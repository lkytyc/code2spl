[DEFINE_WORKER: "Sums the main diagonal of a matrix, counts rows that contain at least one duplicate value, and counts columns that contain at least one duplicate value, returning the three numbers as a list." vestigial]
    [INPUTS]
        <REF> pinax </REF>: list of lists "Input matrix; the method uses the length of pinax[0] as size and assumes the first size rows and columns are accessible."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of integers "Returns [trail, same_row_counter, sumC] where trail is the sum of diagonal elements, same_row_counter is the number of rows with at least one duplicate, and sumC is the number of columns with at least one duplicate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set size to the length of pinax[0]. RESULT size]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Initialize row_digits as an empty list. RESULT row_digits]
            [COMMAND Initialize column_digits as an empty list. RESULT column_digits]
            [COMMAND For each i from 0 to size-1 inclusive, append a new empty list to column_digits so that each column has its own tracking list. RESULT column_digits]
            [COMMAND Set same_row_counter to 0. RESULT same_row_counter]
            [COMMAND Set same_column_counter to a list containing size zeros. RESULT same_column_counter]
            [COMMAND Set trail to 0. RESULT trail]
            [COMMAND For each row from 0 to size-1 inclusive, start processing the row and clear row_digits by deleting all of its elements. RESULT row_digits]
            [COMMAND Within the row loop, for each column from 0 to size-1 inclusive, process the cell pinax[row][column]. RESULT column]
            [COMMAND Within the row loop and column loop, if row equals column, add pinax[row][column] to trail. RESULT trail]
            [COMMAND Within the row loop and column loop, run the row duplicate check only when len(row_digits) equals column. RESULT row_digits]
            [COMMAND Within the row duplicate check, scan row_digits for the current cell value; if the value is found at an index i, increment same_row_counter by 1 and stop scanning. RESULT same_row_counter]
            [COMMAND Within the row duplicate check, if the current cell value is not found in row_digits, append the current cell value to row_digits. RESULT row_digits]
            [COMMAND Within the row loop and column loop, scan column_digits[column] for the current cell value; if the value is found, set same_column_counter[column] to 1 and stop scanning. RESULT same_column_counter]
            [COMMAND Within the row loop and column loop, if the current cell value is not found in column_digits[column], append the current cell value to column_digits[column]. RESULT column_digits]
            [COMMAND After all row and column loops complete, set sumC to 0. RESULT sumC]
            [COMMAND For each i from 0 to size-1 inclusive, add same_column_counter[i] to sumC. RESULT sumC]
            [COMMAND Append trail to result. RESULT result]
            [COMMAND Append same_row_counter to result. RESULT result]
            [COMMAND Append sumC to result. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: pinax is an empty list]
        [LOG "Cannot access pinax[0] because the input matrix has no rows."]
        [THROW IndexError "IndexError is raised when reading pinax[0] from an empty input."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: pinax has fewer than size rows, or a row has fewer than size elements, so pinax[row][column] does not exist during the loops]
        [LOG "An out-of-range row or column access occurs because the input matrix is smaller than the size taken from the first row."]
        [THROW IndexError "IndexError is raised when accessing pinax[row][column] outside the existing matrix bounds."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
