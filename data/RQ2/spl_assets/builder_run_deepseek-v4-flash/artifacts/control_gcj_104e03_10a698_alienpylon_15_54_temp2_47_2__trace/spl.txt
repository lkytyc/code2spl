[DEFINE_WORKER: "Decides whether an alien-pylon layout with dimensions a and b is possible and, if possible, prints a Case # header followed by a*b coordinate pairs generated by a deterministic modular update." alienpylon]
    [INPUTS]
        <REF> a </REF>: int "First dimension parameter used in bounds checks and modular arithmetic."
        <REF> b </REF>: int "Second dimension parameter used in bounds checks and modular arithmetic."
        <REF> i </REF>: int "Zero-based case number; displayed as i+1 in the Case # output."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare int k without initialization, set temp1 = a, and set temp2 = b. RESULT temp1,temp2]
            [COMMAND Evaluate whether b >= a; if false, proceed to the b < a branch. RESULT outer_ge]
            [COMMAND In the b >= a branch, evaluate with short-circuit OR whether b-a==1 or a+1==b or a-1==b or b+1==a or b-1==a or a==b. RESULT inner_impossible_b_ge_a]
            [COMMAND In the b >= a branch, if inner_impossible_b_ge_a is true, print "Case #%d: IMPOSSIBLE\n" with i+1 and skip the coordinate loop. RESULT printed_impossible]
            [COMMAND In the b >= a branch, if inner_impossible_b_ge_a is false, print "Case #%d: POSSIBLE\n" with i+1. RESULT printed_possible]
            [COMMAND In the b >= a branch, when inner_impossible_b_ge_a is false, run a for loop with k initialized to 1, continuing while k <= a*b, and incrementing k by 1 after each iteration. RESULT k_loop_b_ge_a]
            [COMMAND Inside that for loop, update temp1 = temp1 + (((b/a)-1)%a), then set temp1 = temp1 % a. RESULT temp1]
            [COMMAND Inside that for loop, update temp2 = temp2 + (((b%a)+1)%b), then set temp2 = temp2 % b. RESULT temp2]
            [COMMAND Inside that for loop, print "%d %d\n" with temp1+1 and temp2+1. RESULT printed_coordinate]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b < a]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the b < a branch, evaluate with short-circuit OR whether a-b==1 or a+1==b or a-1==b or b+1==a or b-1==a or a==b. RESULT inner_impossible_b_lt_a]
            [COMMAND In the b < a branch, if inner_impossible_b_lt_a is true, print "Case #%d: IMPOSSIBLE\n" with i+1 and skip the coordinate loop. RESULT printed_impossible]
            [COMMAND In the b < a branch, if inner_impossible_b_lt_a is false, print "Case #%d: POSSIBLE\n" with i+1. RESULT printed_possible]
            [COMMAND In the b < a branch, when inner_impossible_b_lt_a is false, run a for loop with k initialized to 1, continuing while k <= a*b, and incrementing k by 1 after each iteration. RESULT k_loop_b_lt_a]
            [COMMAND Inside that for loop, update temp2 = temp2 + (((a/b)-1)%b), then set temp2 = temp2 % b. RESULT temp2]
            [COMMAND Inside that for loop, update temp1 = temp1 + (((a%b)+1)%a), then set temp1 = temp1 % a. RESULT temp1]
            [COMMAND Inside that for loop, print "%d %d\n" with temp2+1 and temp1+1. RESULT printed_coordinate]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
