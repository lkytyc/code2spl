[DEFINE_WORKER: "Reads flavor preferences for each batch, tracks preference counts and sold status, prints the least popular unsold flavor in each batch or -1 if none, marks that flavor sold, and returns with an error if a batch count is negative." solve]
    [INPUTS]
        <REF> caseNum </REF>: int "Unused case number passed to solve."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read an integer nflavors from standard input using scanf without checking its return value. RESULT nflavors]
            [COMMAND Allocate sold as a zero-initialized uint8 array of length nflavors using calloc. RESULT sold]
            [COMMAND Allocate preferred as a zero-initialized int array of length nflavors using calloc. RESULT preferred]
            [COMMAND For i from 0 to nflavors-1, enter each outer-loop batch iteration. RESULT i]
            [COMMAND Within the outer loop, initialize leastPopular to -1 for the current batch. RESULT leastPopular]
            [COMMAND Within the outer loop, read an integer count from standard input using scanf without checking its return value. RESULT count]
            [COMMAND Within the outer loop, if count is negative, branch to the negative-count alternative flow; otherwise continue with normal processing. RESULT count]
            [COMMAND Within the outer loop, for j from 0 to count-1, enter each inner-loop iteration. RESULT j]
            [COMMAND Within the inner loop, read an integer preference from standard input, using it directly as an array index without bounds checking. RESULT preference]
            [COMMAND Within the inner loop, increment preferred[preference] by 1. RESULT preferred]
            [COMMAND Within the inner loop, using short-circuit evaluation, if preference has not been marked sold and either leastPopular is -1 or preferred[preference] is less than preferred[leastPopular], set leastPopular to preference. RESULT leastPopular]
            [COMMAND After the inner loop completes, print leastPopular to standard output as an integer followed by a newline. RESULT none]
            [COMMAND Immediately after printing leastPopular, flush standard output. RESULT none]
            [COMMAND After the inner loop completes, if leastPopular is greater than or equal to 0, set sold[leastPopular] to 1. RESULT sold]
            [COMMAND After all outer-loop iterations complete, free preferred. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within an outer-loop iteration, the just-read count is negative.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print the error message "ERROR: received count of %d\n" to standard output, substituting the negative count value. RESULT none]
            [COMMAND Return immediately from solve without executing remaining outer-loop iterations, the inner loop, the print/flush steps, or free(preferred). RESULT return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
