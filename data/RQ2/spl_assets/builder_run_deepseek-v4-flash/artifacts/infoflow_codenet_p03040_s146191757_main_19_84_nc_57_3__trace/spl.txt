[DEFINE_WORKER: "Reads a sequence of insertion and query operations, maintains the dynamic median of inserted coordinates using count-aware left and right heaps, and returns newline-separated query answers containing the current median and accumulated absolute-distance offset." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Newline-joined query result lines; each line contains the current median x and the value xl + xr + bt."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read q = I(). RESULT q]
            [COMMAND Read aa = [LI() for _ in range(q)]. RESULT aa]
            [COMMAND Set bt = aa[0][2] as the starting accumulated offset. RESULT bt]
            [COMMAND Set m = 2 ** 20 (unused constant). RESULT m]
            [COMMAND Initialize c = collections.defaultdict(int). RESULT c]
            [COMMAND Initialize lq = [inf] and rq = [inf]. RESULT heaps]
            [COMMAND Set c[-inf] = 1 and c[inf] = 1. RESULT c]
            [COMMAND Set x = aa[0][1], then set c[x] = 1. RESULT x]
            [COMMAND Initialize xl = 0, lc = 0, xr = 0, rc = 0. RESULT counters]
            [COMMAND Initialize r = [] as the output accumulator. RESULT r]
            [COMMAND For each operation op in aa[1:], process op in order. RESULT state]
            [COMMAND Within the loop over op, if op[0] == 2, append the string containing x followed by xl + xr + bt to r and continue with the next operation. RESULT r]
            [COMMAND Within the loop over op, for an insertion operation (op[0] != 2), unpack a = op[1] and b = op[2], add b to bt, and if a == x, increment c[x] and continue with the next operation. RESULT state]
            [COMMAND Within the loop over op, after the a == x check, increment c[a]; if c[a] == 1 and a < x, push -a onto lq; if c[a] == 1 and a > x, push a onto rq. RESULT state]
            [COMMAND Within the loop over op, set xc = c[x]. RESULT xc]
            [COMMAND Within the loop over op, if a < x, update xl += x - a and lc += 1; set current_total = xl + xr; set nx = -lq[0], nc = c[nx], s = x - nx; compute candidate_total = xl - lc*s + xr + (rc + xc)*s; if candidate_total <= current_total, move the median left by updating xl -= lc*s, xr += (rc + xc)*s, lc -= nc, rc += xc, pushing x onto rq, setting x = nx, and popping from lq. RESULT state]
            [COMMAND Within the loop over op, if a > x, update xr += a - x and rc += 1; set current_total = xl + xr; set nx = rq[0], nc = c[nx], s = nx - x; compute candidate_total = xl + (lc + xc)*s + xr - rc*s; if candidate_total < current_total, move the median right by updating xl += (lc + xc)*s, xr -= rc*s, lc += xc, rc -= nc, pushing -x onto lq, setting x = nx, and popping from rq. RESULT state]
            [COMMAND After the loop, return the result of joining r with newline characters. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
