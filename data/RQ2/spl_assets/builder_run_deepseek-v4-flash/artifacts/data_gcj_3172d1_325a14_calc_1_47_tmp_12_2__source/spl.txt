[DEFINE_WORKER: "For each index, the method updates ans with the minimum cut + xx * cur_d value while advancing cur_d toward d via prefix and backward list scans, then returns (ans, the first element of a)." calc]
    [INPUTS]
        <REF> n </REF>: int "Initial parameter; immediately overwritten by len(a) and not otherwise used."
        <REF> d </REF>: int "Target value used as a loop bound and in the min updates."
        <REF> a </REF>: list[int] "Input list whose elements are used as x values and whose first element becomes ret."
        <REF> b </REF>: list[int] "Input list whose elements are paired with a in the outer loop as y values."
        <REF> ct </REF>: dict[int, int] "Dictionary mapping accumulated prefix values p to integer counts used in the inner loop bounds."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple[int, int] "Returns the tuple (ans, ret), where ans is the minimum computed value of cut + xx * cur_d and ret is the first element of a."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set n to the length of the input list a. RESULT n]
            [COMMAND Set ans to d - 1 and ret to a[0]. RESULT ans, ret]
            [COMMAND Start the outer loop over i in range(n), so i iterates from 0 through n - 1 inclusive. At the beginning of each iteration, set x = a[i], y = b[i], cur_d = 0, p = 0, and cut = 0. RESULT i]
            [COMMAND Within the outer loop, start the first j loop over j in range(d + 2), so j iterates from 0 through d + 1 inclusive. For each j, update p = p + x, set tmp = ct.get(p, 0), remain_d = d - cur_d, per_d = p / x, and per_cut = per_d - 1. RESULT j]
            [COMMAND Within that j loop, run the inner k loop over k in range(tmp * per_d), so k iterates from 0 up to tmp * per_d - 1. For each k, increment cur_d by 1; if (k + 1) % per_d != 0, increment cut by 1; set xx = d / cur_d; if d % cur_d == 0, set xx = xx - 1; update ans = min(ans, cut + xx * cur_d); and if cur_d > d, break out of the k loop. RESULT cur_d, cut, xx, ans]
            [COMMAND After the inner k loop, if cur_d > d, break out of the first j loop. RESULT first_j_loop_state]
            [COMMAND After the first j loop, if cur_d < d, start the backward j loop over j in range(n - 1, i, -1), so j iterates from n - 1 down to i + 1, excluding i. RESULT backward_j]
            [COMMAND Within the backward j loop, using short-circuit OR, enter the branch if a[j] > p is true, or if it is false and a[j] % x != 0 is true. Then set tmp = (a[j] / x) * y and run an inner k loop over k in range(tmp), so k iterates from 0 up to tmp - 1. For each k, increment cur_d by 1 and cut by 1; set xx = d / cur_d; if d % cur_d == 0, set xx = xx - 1; update ans = min(ans, cut + xx * cur_d); and if cur_d > d, break out of this inner k loop. RESULT cur_d, cut, xx, ans]
            [COMMAND After that inner k loop, if cur_d > d, break out of the backward j loop. RESULT backward_j_loop_state]
            [COMMAND After the optional backward phase completes, unconditionally set xx = d / cur_d; if d % cur_d == 0, set xx = xx - 1; and update ans = min(ans, cut + xx * cur_d). This completes the current iteration of the outer i loop. RESULT cur_outer_iteration_complete]
            [COMMAND After all outer i iterations finish, return the tuple (ans, ret). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
