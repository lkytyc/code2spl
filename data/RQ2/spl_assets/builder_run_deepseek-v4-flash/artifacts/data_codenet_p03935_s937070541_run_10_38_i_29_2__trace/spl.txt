[DEFINE_WORKER: "Reads n and m, returns early if m is greater than 200000, builds Fibonacci-derived prefix sums modulo 998244353, applies repeated prefix-sum transformations, and prints the resulting last value." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> last </REF>: long "The value printed to standard output; it is captured from sum[m - 1] just before the final replacement of sum with next."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new MyScanner instance for reading input. RESULT sc]
            [COMMAND Read the first integer from the scanner and store it in n. RESULT n]
            [COMMAND Read the second integer from the scanner and store it in m. RESULT m]
            [COMMAND Set M to the numeric constant 998244353. RESULT M]
            [COMMAND Allocate int[2][m] array dp and int[m] array sum; assign dp[0][0] = 1, dp[0][1] = 1, sum[0] = 1, and sum[1] = 2. RESULT arrays_initialized]
            [COMMAND For i starting at 2 and continuing while i < m, incrementing i by 1, set dp[0][i] = (dp[0][i - 1] + dp[0][i - 2]) % M and then set sum[i] = (sum[i - 1] + dp[0][i]) % M. RESULT initial_prefix_sums]
            [COMMAND Set last = dp[0][m - 1]. RESULT last]
            [COMMAND For the outer loop with i starting at 1 and continuing while i < n, incrementing i by 1: allocate a fresh int[m] array next; set next[0] = sum[0]; for j starting at 1 and continuing while j < m, incrementing j by 1, compute next[j] = (next[j - 1] + sum[j]) % M; after that inner loop, set last = sum[m - 1] using the current sum before replacement; then assign sum = next. RESULT transformed_sum]
            [COMMAND Print the value of last followed by a newline to standard output. RESULT printed_last]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: After n and m have been read, m is strictly greater than 200000 (m = 200000 does not trigger the guard).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return immediately from run(), skipping all later allocations, loops, and output. RESULT early_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
