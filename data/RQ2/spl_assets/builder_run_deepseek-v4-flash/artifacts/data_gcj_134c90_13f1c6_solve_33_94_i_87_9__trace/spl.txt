[DEFINE_WORKER: "Simulates a rock-paper-scissors elimination game over cycling move sequences, printing the winning move sequence until only one move type remains, all players die, an impossible three-move set appears, or the sequence reaches 500 moves." solve]
    [INPUTS]
        <REF> A </REF>: int "Number of players."
        <REF> C </REF>: list of strings "Move sequence for each player; C[i] is the sequence used by player i."
        <REF> L </REF>: list of int "Length of each player's move sequence; the current move of player i is C[i][pos % L[i]]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_value </REF>: string "Prints the final answer ans or the string IMPOSSIBLE to stdout; the function returns None."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ans to an empty string. RESULT ans]
            [COMMAND Initialize pos to 0. RESULT pos]
            [COMMAND Initialize isAlive to a list of A True values. RESULT isAlive]
            [COMMAND Enter an infinite loop. At the start of each iteration, reset possibilities to {'R':0, 'S':0, 'P':0} and reset predR, predS, and predP to empty lists. RESULT _]
            [COMMAND If debug is enabled, print the current alive status and current ans. RESULT _]
            [COMMAND Check whether any player is alive; if no player is alive, print ans and return from the function. RESULT _]
            [COMMAND For each alive player i in range(A), compute move = C[i][pos % L[i]], set possibilities[move] = 1, and append i to predR if move is 'R', to predS if move is 'S', or to predP if move is 'P'. If all three moves become present during this scan, print "IMPOSSIBLE" and return from the function. RESULT predR, predS, predP, possibilities]
            [COMMAND Read the tuple (r, s, p) from possibilities['R'], possibilities['S'], and possibilities['P']. RESULT rsp_tuple]
            [COMMAND If debug is enabled, print the tuple (r, s, p). RESULT _]
            [COMMAND If exactly one move type is present, append the corresponding winning move to ans ('P' when only R is present, 'R' when only S is present, 'S' when only P is present), print ans, and return from the function. RESULT ans, printed_value]
            [COMMAND If len(ans) is exactly 500, print "IMPOSSIBLE" and return from the function; if debug is enabled, also print "LEN" and ans before returning. RESULT printed_value]
            [COMMAND If exactly two move types are present, eliminate the losing players and append the winning move: when R and S are present, set isAlive[i] = False for all i in predS and append 'R'; when R and P are present, set isAlive[i] = False for all i in predR and append 'P'; when S and P are present, set isAlive[i] = False for all i in predP and append 'S'. RESULT isAlive, ans]
            [COMMAND Increment pos by 1 and continue the infinite loop from the start. RESULT pos]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
