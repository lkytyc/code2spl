[DEFINE_WORKER: "Reads 10,000 pairs of strings, records letter observations and minimum digit constraints to reconstruct a 10-character mapping, and prints it as 'Case #T: <out>'." solve]
    [INPUTS]
        <REF> T </REF>: int "Test case number printed in the output label."
        <REF> stdin </REF>: input stream "Provides an integer U followed by 10,000 pairs of strings M and R."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: formatted text "Prints 'Case #<T>: <out>' followed by a newline, where out is the reconstructed 10-character string."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize integer arrays MC[0..25] to 10 and O[0..25] to 0; declare out as char[10]; read integer U from stdin; set i to 0; declare char arrays R[U] and M[U]; declare integer variables lr and lm. RESULT initialized_state]
            [COMMAND Run a while loop while i < 10000. In each iteration, read strings M and R from stdin using scanf("%s %s", M, R); set lr = strlen(R) and lm = strlen(M); set O[R[lr-1]-'A'] = 1; if lr == lm, set O[R[0]-'A'] = 1 and if MC[R[0]-'A'] > M[0]-'0', set MC[R[0]-'A'] = M[0]-'0'; then increment i. RESULT processed_pairs]
            [COMMAND Run a for loop with i starting at 0 and continuing while i < 26, incrementing i by 1 after each iteration. In each iteration, if O[i] is nonzero, write 'A' + i into out[MC[i] % 10]. RESULT out_built]
            [COMMAND Call printf("Case #%d: %s\n", T, out) to print the result, then return. RESULT completed]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
