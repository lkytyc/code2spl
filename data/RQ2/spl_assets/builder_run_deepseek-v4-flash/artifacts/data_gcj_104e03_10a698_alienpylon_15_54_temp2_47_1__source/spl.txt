[DEFINE_WORKER: "Given integer dimensions a and b and a zero-based case index i, this method prints either an IMPOSSIBLE or POSSIBLE case header and, when possible, prints exactly a*b coordinate pairs generated by two modular counters whose update and print order depend on whether b >= a." alienpylon]
    [INPUTS]
        <REF> a </REF>: int "First integer dimension and modular counter base."
        <REF> b </REF>: int "Second integer dimension and modular counter base."
        <REF> i </REF>: int "Zero-based test-case index used for the Case # label."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare int k without initialization; assign temp1 = a and temp2 = b. RESULT initial_state]
            [COMMAND Evaluate the outer condition b >= a. If it is true, branch into the alternative flows for the 'b >= a is true' case; otherwise branch into the alternative flows for the 'b >= a is false, so a > b' case. RESULT outer_branch_selection]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b >= a is true, and the inner impossible condition is true: b-a==1 || a+1==b || a-1==b || b+1==a || b-1==a || a==b]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print `Case #%d: IMPOSSIBLE` followed by a newline, with i + 1 as the case number. RESULT impossible_output_b_ge_a]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: b >= a is true, and the inner impossible condition is false, i.e. |a - b| > 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print `Case #%d: POSSIBLE` followed by a newline, with i + 1 as the case number. RESULT possible_header_output_b_ge_a]
            [COMMAND Within the b >= a possible branch, initialize k = 1; while k <= a * b, run the loop-body step and then increment k by 1; there is no early exit. RESULT loop_control_b_ge_a]
            [COMMAND Within that b >= a loop iteration, update temp1 += (((b/a)-1)%a), then temp1 = temp1 % a; update temp2 += (((b%a)+1)%b), then temp2 = temp2 % b; then print `%d %d` followed by a newline using temp1 + 1 and temp2 + 1. RESULT loop_body_output_b_ge_a]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: b >= a is false, so a > b, and the inner impossible condition is true: a-b==1 || a+1==b || a-1==b || b+1==a || b-1==a || a==b]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print `Case #%d: IMPOSSIBLE` followed by a newline, with i + 1 as the case number. RESULT impossible_output_a_gt_b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: b >= a is false, so a > b, and the inner impossible condition is false, i.e. |a - b| > 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print `Case #%d: POSSIBLE` followed by a newline, with i + 1 as the case number. RESULT possible_header_output_a_gt_b]
            [COMMAND Within the a > b possible branch, initialize k = 1; while k <= a * b, run the loop-body step and then increment k by 1; there is no early exit. RESULT loop_control_a_gt_b]
            [COMMAND Within that a > b loop iteration, update temp2 += (((a/b)-1)%b), then temp2 = temp2 % b; update temp1 += (((a%b)+1)%a), then temp1 = temp1 % a; then print `%d %d` followed by a newline using temp2 + 1 and temp1 + 1. RESULT loop_body_output_a_gt_b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
