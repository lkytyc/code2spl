[DEFINE_WORKER: "Computes the greatest common divisor of two long integers using the Euclidean algorithm, ordering the operands and returning early when the divisor becomes 1." GCD]
    [INPUTS]
        <REF> a </REF>: long "First long integer input to the GCD calculation."
        <REF> b </REF>: long "Second long integer input to the GCD calculation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: long "The final divisor after the Euclidean algorithm completes, equal to gcd(a, b)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare local long variables dividend and divisor. RESULT dividend, divisor]
            [COMMAND If a >= b, assign dividend = a and divisor = b; otherwise assign dividend = b and divisor = a. RESULT dividend, divisor]
            [COMMAND Declare local long variable remainder. RESULT remainder]
            [COMMAND Enter an infinite while (1) loop. RESULT loop]
            [COMMAND Inside the infinite while loop, if divisor equals 1, return 1 immediately. RESULT return_value]
            [COMMAND Inside the infinite while loop, compute remainder = dividend % divisor. RESULT remainder]
            [COMMAND Inside the infinite while loop, if remainder equals 0, break out of the loop. RESULT loop_exit]
            [COMMAND Inside the infinite while loop, when remainder != 0, assign dividend = divisor and divisor = remainder, then repeat the loop. RESULT dividend, divisor]
            [COMMAND After the loop exits, return divisor. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: a < b]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign dividend = b. RESULT dividend]
            [COMMAND Assign divisor = a. RESULT divisor]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: divisor == 1 at the start of a while iteration]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return 1 immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: remainder == 0 after computing dividend % divisor]
        [SEQUENTIAL_BLOCK]
            [COMMAND Break out of the while loop. RESULT loop_exit]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: remainder != 0 after computing dividend % divisor]
        [SEQUENTIAL_BLOCK]
            [COMMAND Assign dividend = divisor. RESULT dividend]
            [COMMAND Assign divisor = remainder. RESULT divisor]
            [COMMAND Repeat the while loop with the updated values. RESULT loop]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
