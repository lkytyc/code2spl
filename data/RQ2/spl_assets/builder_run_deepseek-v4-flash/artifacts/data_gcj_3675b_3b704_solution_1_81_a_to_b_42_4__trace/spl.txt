[DEFINE_WORKER: "Reads a list of sign tuples, groups consecutive tuples that share at least one common element, and returns the maximum group size and the count of groups achieving that maximum size." solution]
    [INPUTS]
        <REF> number_of_signs </REF>: integer "The number of signs to read from input."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "A tuple containing the maximum set size and the number of sets that achieve that maximum size."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize an empty list named a_to_b. RESULT a_to_b]
            [COMMAND For each i from 0 to number_of_signs - 1, read a line of input, split it into three parts, convert them to integers d, a, and b, and append the tuple (d + a, d - b) to a_to_b. RESULT a_to_b]
            [COMMAND Initialize N to None, M to None, maximum_set_size to 1, number_of_maximum_set_sized to 0, and current_set_size to 1. RESULT state_variables]
            [COMMAND Pop the first tuple from a_to_b and assign its two elements to a_for_comparision and b_for_comparision. RESULT a_for_comparision, b_for_comparision]
            [COMMAND Set is_finish to False. RESULT is_finish]
            [COMMAND While is_finish is False and a_to_b is not empty, iterate over each tuple in a_to_b with its index. For each tuple, if it is the last element, set is_finish to True. Then, based on the current state of N and M, update the comparison values, current_set_size, and potentially reset the state and truncate a_to_b when a new set is detected. RESULT maximum_set_size, number_of_maximum_set_sized]
            [COMMAND Return the tuple (maximum_set_size, number_of_maximum_set_sized). RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When both N and M are not None and a tuple has both a and b different from the comparison values, the current set ends.]
        [SEQUENTIAL_BLOCK]
            [COMMAND If current_set_size is greater than maximum_set_size, update maximum_set_size to current_set_size and set number_of_maximum_set_sized to 1. Else if current_set_size equals maximum_set_size, increment number_of_maximum_set_sized. RESULT maximum_set_size, number_of_maximum_set_sized]
            [COMMAND Reset current_set_size to 1, set N and M to None, truncate a_to_b to start from the current index, and break out of the for loop. RESULT current_set_size, N, M, a_to_b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: When both a and b differ from the comparison values and there is a next element in a_to_b, a look-ahead strategy is used to determine the new set's fixed values.]
        [SEQUENTIAL_BLOCK]
            [COMMAND If the next tuple's a matches the current a, set N to that a, a_for_comparision to that a, and M to the previous b_for_comparision. Otherwise, set M to the next tuple's b, b_for_comparision to that b, and N to the previous a_for_comparision. Increment current_set_size. RESULT N, M, a_for_comparision, b_for_comparision, current_set_size]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
