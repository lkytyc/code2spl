[DEFINE_WORKER: "Reads test cases of wildcard patterns, aligns each pattern by its leading or trailing asterisk, merges non-wildcard characters column by column, and prints either the reconstructed matching string or an asterisk on conflict." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Always returns 0 after all test cases have been processed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integers t, n, i, j, k; declare character array s of size 10001 and character array str of size 101. Initialize t and n to 0. If scanf reads a valid integer from standard input, store it in t; otherwise t remains 0. RESULT t]
            [COMMAND Start the outer loop with i = 1; while i <= t, process one test case and then increment i. RESULT i]
            [COMMAND Within the outer loop, read an integer from standard input into n. RESULT n]
            [COMMAND Within the outer loop, declare a variable-length array a of n rows and 10001 columns plus a character dd; for every row j from 0 to n-1 and every column k from 0 to 10000 inclusive, set a[j][k] to '#'. RESULT a]
            [COMMAND Within the outer loop, initialize flag to 0. RESULT flag]
            [COMMAND Within the outer loop, for each row j from 0 to n-1, read one character into dd to consume the delimiter left from previous input, then read a whitespace-delimited string into str. RESULT str]
            [COMMAND Within the row loop, if the first character of str is '*', right-align the entire string into row a[j]: set x to 10000 and for k from the string length of str down to 0 inclusive assign a[j][x--] = str[k]. RESULT a[j]]
            [COMMAND Within the row loop, otherwise if the last character of str is '*', left-align the string into row a[j]: copy str[k] into a[j][k] while str[k] != '\0'. RESULT a[j]]
            [COMMAND Within the row loop, if str has neither a leading nor a trailing '*', leave row a[j] as all '#' placeholders. RESULT a[j]]
            [COMMAND After all rows have been read, initialize the output index x to 0. RESULT x]
            [COMMAND Within the outer loop, scan columns j from 0 to 9999 inclusive; for each column, initialize f to 0, then scan rows k from 0 to n-1. RESULT s]
            [COMMAND Within the column and row scan, if f == 1 and a[k][j] is not '*', is not '#', and is not equal to s[x-1], set flag to 1 and break out of the row loop. RESULT flag]
            [COMMAND Within the column and row scan, if a[k][j] is neither '*' nor '#': when f == 0, append the character to s as s[x++] = a[k][j] and set f to 1. RESULT s]
            [COMMAND After the row loop for the current column, if flag == 1, break out of the column loop. RESULT flag]
            [COMMAND After the column loop, terminate s by storing '\0' at index x. RESULT s]
            [COMMAND Within the outer loop, if flag == 1, print the line 'Case #%d: *\n' using i; otherwise print the line 'Case #%d: %s\n' using i and s. RESULT stdout]
            [COMMAND After all test cases have been processed, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
