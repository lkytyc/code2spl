[DEFINE_WORKER: "For each test case, simulates placing K people into N stalls by repeatedly splitting the largest remaining interval using a max-heap and multiplicity map, then prints the final larger and smaller gaps." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner named cin connected to System.in. RESULT cin]
            [COMMAND Read the integer T from cin; T is the number of test cases. RESULT T]
            [COMMAND For each test case, start a for loop with int C = 1; continue while C <= T; increment C by 1 after each iteration. RESULT C]
            [COMMAND Inside the for loop over C, read the long N from cin. RESULT N]
            [COMMAND Inside the for loop over C, read the long K from cin. RESULT K]
            [COMMAND Inside the for loop over C, create a PriorityQueue<Long> named queue that stores negative interval lengths so -queue.poll() returns the largest interval length. RESULT queue]
            [COMMAND Inside the for loop over C, create a HashMap<Long, Long> named count mapping each interval length to its multiplicity. RESULT count]
            [COMMAND Inside the for loop over C, seed the data structures by adding -N to queue and putting count[N] = 1L. RESULT queue]
            [COMMAND Inside the for loop over C, initialize long ans = 0. RESULT ans]
            [COMMAND Inside the for loop over C, enter an infinite while loop to process intervals until the K-th person is placed. RESULT while_loop]
            [COMMAND Inside the while loop, set long cur = -queue.poll(), the current largest interval length. RESULT cur]
            [COMMAND Inside the while loop, set long rest = count.get(cur), the number of intervals of length cur. RESULT rest]
            [COMMAND Inside the while loop, if K <= 1, set ans = cur and break out of the while loop. RESULT ans]
            [COMMAND Inside the while loop, subtract rest from K: K -= rest. RESULT K]
            [COMMAND Inside the while loop, if K <= 0 after the subtraction, set ans = cur and break out of the while loop. RESULT ans]
            [COMMAND Inside the while loop, if no break occurred, set long large = cur / 2. RESULT large]
            [COMMAND Inside the while loop, if no break occurred, set long small = (cur - 1) / 2. RESULT small]
            [COMMAND Inside the while loop, if count contains large, add rest to count[large]; otherwise add -large to queue and set count[large] = rest. RESULT queue_and_count]
            [COMMAND Inside the while loop, if count contains small, add rest to count[small]; otherwise add -small to queue and set count[small] = rest. RESULT queue_and_count]
            [COMMAND Inside the for loop over C, after the while loop exits, print "Case #" + C + ": " + (ans / 2) + " " + ((ans - 1) / 2). RESULT printed_output]
            [COMMAND After the for loop over C finishes, close cin. RESULT cin]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
