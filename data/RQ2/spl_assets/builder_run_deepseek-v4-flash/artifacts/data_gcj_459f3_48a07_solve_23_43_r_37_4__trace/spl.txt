[DEFINE_WORKER: "Reads two integers, orders them as r >= b, counts consecutive non-negative integers whose standard binary representations fit the remaining r one-bit and b zero-bit budgets, adds a cntForOne-based count for the leftover budget, and returns the total count." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> cnt </REF>: int "Total count of qualifying integers: successful loop iterations plus the additional cntForOne contribution; in the early-return path, the result is cntForOne(r)."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers R and B from the input using readia() RESULT R,B]
            [COMMAND Set r to max(R,B) and b to min(R,B) RESULT r,b]
            [COMMAND Check if b equals 0; if true, follow the early-return alternative flow and return cntForOne(r); otherwise continue with the next step RESULT b_is_zero]
            [COMMAND Initialize i to 0, cnt to 0, cR to 0, and cB to 1 RESULT i,cnt,cR,cB]
            [COMMAND Repeat the loop body until break: convert i to binary string s via bin(i)[2:], set cR to s.count('1') and cB to s.count('0'); if cR > r or cB > b, break; otherwise subtract cR from r and cB from b, then increment i and cnt RESULT r,b,i,cnt after loop termination]
            [COMMAND Compute remaining_bits as len(bits) - 1 RESULT remaining_bits]
            [COMMAND Add cntForOne(r, remaining_bits) to cnt RESULT cnt]
            [COMMAND Return cnt RESULT cnt]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: b equals 0 after ordering R and B]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return cntForOne(r) and terminate RESULT cntForOne(r)]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
