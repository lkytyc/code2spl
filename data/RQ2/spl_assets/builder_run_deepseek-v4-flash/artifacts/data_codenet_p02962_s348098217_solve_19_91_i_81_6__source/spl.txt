[DEFINE_WORKER: "Reads two strings, compares their lengths and aligned substring blocks, and writes either -1 for full aligned repetition, 0 for unequal equal-length strings, or the maximum aligned end-match count to the provided PrintWriter." solve]
    [INPUTS]
        <REF> testNumber </REF>: int "Unused test case index passed to solve."
        <REF> br </REF>: FastScanner "Scanner used to read the two input strings."
        <REF> pw </REF>: PrintWriter "Writer used for all output and closed on many paths."
    [END_INPUTS]

    [OUTPUTS]
        <REF> pw </REF>: PrintWriter "Writer containing the output produced by the method; may be closed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the first string via br.nextString() and store it as in1. RESULT in1]
            [COMMAND Read the second string via br.nextString() and store it as in2. RESULT in2]
            [COMMAND Convert in1 to a char array t using in1.toCharArray(). RESULT t]
            [COMMAND Convert in2 to a char array s using in2.toCharArray(). RESULT s]
            [COMMAND Evaluate the length relation. If t.length > s.length, execute the following longer-first-string branch; otherwise use the alternative_flows for the equal-length and shorter-first-string cases in that order. RESULT length_branch]
            [COMMAND Inside the t.length > s.length branch, initialize orig to 0 and poss to true. RESULT orig, poss]
            [COMMAND Loop with i from 0 while i < t.length - s.length, incrementing i by s.length. At each iteration, if in1.substring(i, i + s.length) does not equal in2, set poss to false; otherwise increment orig. The loop excludes the starting index t.length - s.length. RESULT orig, poss]
            [COMMAND If poss is true, print -1 to pw and close pw. Execution does not return, but pw is closed so later prints are suppressed. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND Loop with i from 0 while i < t.length - s.length, incrementing i by s.length. If in1.substring(i, i + s.length) equals in2, increment ans; otherwise break. RESULT ans]
            [COMMAND Loop with i from t.length - s.length while i >= 0, decrementing i by s.length. If in1.substring(i, i + s.length) equals in2, increment ans; otherwise break. This loop includes the starting index t.length - s.length and index 0. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: t.length > s.length is false and s.length == t.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND In the equal-length branch, compare s to t using s.equals(t). Since s and t are distinct char arrays created by toCharArray, this reference comparison is always false, so print 0 to pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: t.length > s.length is false and s.length != t.length, so t.length < s.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Inside the t.length < s.length branch, initialize orig to 0 and poss to true. RESULT orig, poss]
            [COMMAND Loop with i from 0 while i < s.length - t.length, incrementing i by s.length. At each iteration, if in2.substring(i, i + t.length) does not equal in1, set poss to false. There is no else clause, so orig is never incremented; the step uses s.length rather than t.length. RESULT poss]
            [COMMAND If poss is true, print -1 to pw and close pw. Execution does not return, but pw is closed so later prints are suppressed. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND Loop with i from 0 while i < s.length - t.length, incrementing i by t.length. If in2.substring(i, i + t.length) equals in1, increment ans; otherwise break. RESULT ans]
            [COMMAND Loop with i from s.length - t.length while i >= 0, decrementing i by t.length. If in2.substring(i, i + t.length) equals in1, increment ans; otherwise break. This loop includes the starting index s.length - t.length and index 0. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
