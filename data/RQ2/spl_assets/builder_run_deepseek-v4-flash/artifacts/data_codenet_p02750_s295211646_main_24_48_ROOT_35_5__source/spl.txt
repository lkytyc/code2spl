[DEFINE_WORKER: "Reads N time records into an array, sorts the array, consumes T by iteratively applying (T - b) / (a + 1), counts processed records, and prints the count after a final decrement." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to main; not used by this method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new Scanner attached to System.in. RESULT sc]
            [COMMAND Read the next integer token from sc and assign it to N. RESULT N]
            [COMMAND Read the next long token from sc and assign it to T. RESULT T]
            [COMMAND Allocate a new time array of length N and assign it to ROOT. RESULT ROOT]
            [COMMAND For i from 0 to N-1 inclusive, create a new time object tm, read a long into tm.a from sc, read a long into tm.b from sc, assign ROOT[i] = tm, and increment i. RESULT ROOT]
            [COMMAND Sort the ROOT array in-place using Arrays.sort. RESULT ROOT]
            [COMMAND Initialize idx to 0. RESULT idx]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND While both T > 0 and idx < N are true, evaluated in that order, update T = (T - ROOT[idx].b) / (ROOT[idx].a + 1), increment ans, increment idx, and if the new idx equals N - 1 then increment ans again; repeat until the condition is false. RESULT T_ans_idx]
            [COMMAND After the while loop, decrement ans by 1. RESULT ans]
            [COMMAND Print the value of ans to standard output. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: The next token from sc is not a valid int for N or not a valid long for T, tm.a, or tm.b.]
        [LOG "Scanner cannot parse the expected numeric input."]
        [THROW InputMismatchException "Scanner.nextInt or Scanner.nextLong received a token of the wrong type."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: N is negative when new time[N] is executed.]
        [LOG "Attempted to create an array with negative length."]
        [THROW NegativeArraySizeException "Array length N must be non-negative."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: ROOT[idx].a equals -1 during the while-loop update of T.]
        [LOG "Division by zero in (T - ROOT[idx].b) / (ROOT[idx].a + 1)."]
        [THROW ArithmeticException "The divisor ROOT[idx].a + 1 is zero."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
