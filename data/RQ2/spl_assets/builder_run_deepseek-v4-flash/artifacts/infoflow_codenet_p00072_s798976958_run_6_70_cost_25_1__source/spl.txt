[DEFINE_WORKER: "Reads undirected graph test cases from standard input, records edge weights in a symmetric cost matrix, selects qualifying edges in two passes using Integer.MAX_VALUE and 0 as sentinels, and prints per-case counts computed from edge weight quotients." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner named scan that reads from System.in. RESULT scan]
            [COMMAND While scan has another token, process the next test case. RESULT outer_loop]
            [COMMAND Read the next integer from scan as n. RESULT n]
            [COMMAND If n equals 0, break out of the outer while loop immediately. RESULT break_exit]
            [COMMAND Read the next integer from scan as m. RESULT m]
            [COMMAND Using nested loops over i from 0 to n-1 and j from 0 to n-1, set every entry of the n by n matrix cost to Integer.MAX_VALUE. RESULT cost]
            [COMMAND For each edge index i from 0 to m-1 inclusive, read the next token as a string, split it on comma into str, parse str[0] as a, str[1] as b, and str[2] as c, then assign cost[a][b] = c and cost[b][a] = c. RESULT cost]
            [COMMAND Initialize count to 0, min to 0, ti to 0, tj to 0, and a boolean array fl of length n with all false entries. RESULT count, min, ti, tj, fl]
            [COMMAND For each i from 0 to n-2 inclusive, before scanning row i, set min to 0. RESULT min]
            [COMMAND Within the same outer first-pass loop over i, for each j from i+1 to n-1 inclusive, if cost[i][j] equals Integer.MAX_VALUE skip this j; otherwise, using short-circuit OR with the sentinel condition min == 0 || min > cost[i][j], update min to cost[i][j], ti to i, and tj to j. RESULT min, ti, tj]
            [COMMAND Still within the outer first-pass loop over i, after the inner j loop, if min is not 0, add the integer quotient of min divided by 100 minus 1 to count, and set fl[ti] to true and fl[tj] to true. RESULT count, fl]
            [COMMAND For each i from 0 to n-1 inclusive in the second pass, if fl[i] is true, skip this vertex and continue with the next i. RESULT fl_skip]
            [COMMAND For each unmarked i in the second pass, set min to 0. RESULT min]
            [COMMAND Within this second-pass loop over i, for each j from 0 to i-1 inclusive, if cost[j][i] equals Integer.MAX_VALUE skip this j; otherwise, using short-circuit OR with the sentinel condition min == 0 || min > cost[j][i], update min to cost[j][i]. RESULT min]
            [COMMAND Still within this second-pass loop over i, after the inner j loop, if min is not 0, add the integer quotient of min divided by 100 minus 1 to count. RESULT count]
            [COMMAND Print the current count to standard output on its own line. RESULT count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The integer read as n equals 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Break out of the outer while loop before reading m or any further input. RESULT break_exit]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the first-pass inner loop, cost[i][j] equals Integer.MAX_VALUE.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the rest of the current inner-loop iteration and continue with the next j. RESULT continue]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the second-pass loop, fl[i] is true for the current vertex.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the rest of this second-pass iteration and continue with the next i. RESULT continue]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the second-pass inner loop, cost[j][i] equals Integer.MAX_VALUE.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Skip the rest of the current inner-loop iteration and continue with the next j. RESULT continue]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
