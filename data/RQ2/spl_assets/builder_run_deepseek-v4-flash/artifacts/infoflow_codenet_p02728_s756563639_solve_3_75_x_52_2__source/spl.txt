[DEFINE_WORKER: "Computes a per-vertex modular combinatorial assignment count for a rooted tree by BFS-ordering the tree, precomputing factorials, and performing bottom-up and top-down dynamic-programming passes." solve]
    [INPUTS]
        <REF> n </REF>: int "Number of vertices in the tree."
        <REF> edge_list </REF>: list[tuple[int, int]] "List of n-1 undirected edges, each a 0-indexed pair of vertex labels."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list[int] "List of length n containing the computed modular value for each vertex."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Precompute factorials modulo the global MOD: set factorial to a list of n ones, then for i from 1 to n-1 inclusive, set factorial[i] = (factorial[i-1] * i) % MOD. RESULT factorial]
            [COMMAND Precompute inverse factorials: set factorial_inv to a list of n ones, set factorial_inv[-1] = pow(factorial[-1], MOD - 2, MOD), then for i from n-2 down to 1 inclusive, set factorial_inv[i] = (factorial_inv[i+1] * (i+1)) % MOD. RESULT factorial_inv]
            [COMMAND Build undirected adjacency list g: create g as a list of n independent empty lists, then for i from 0 to n-2 inclusive, unpack edge_list[i] as a and b, append b to g[a], and append a to g[b]. RESULT g]
            [COMMAND Initialize order as an empty list. RESULT order]
            [COMMAND Initialize parent as a list of n copies of -1 and set parent[0] = 0. RESULT parent]
            [COMMAND Initialize children as a list of n independent empty lists. RESULT children]
            [COMMAND Initialize queue as deque([0]). RESULT queue]
            [COMMAND Run BFS on g from root 0: while queue is not empty, pop p from queue, append p to order, and for each neighbor q in g[p], if parent[q] == -1, set parent[q] = p, append q to children[p], and enqueue q. This fills order, parent, and children. RESULT order]
            [COMMAND Initialize count_children_res as a list of n copies of -1. RESULT count_children_res]
            [COMMAND Compute descendant counts bottom-up: for each x in reversed(order), if children[x] is empty, set count_children_res[x] = 0; otherwise compute total as the sum of count_children_res[y] over all children y of x, set r = len(children[x]) + total, and set count_children_res[x] = r. RESULT count_children_res]
            [COMMAND Initialize assign_children_res as a list of n copies of -1. RESULT assign_children_res]
            [COMMAND Compute bottom-up child assignment values: for each x in reversed(order), if children[x] is empty, set assign_children_res[x] = 1; otherwise set r = factorial[count_children_res[x]], and for each child y of x, multiply r by assign_children_res[y] and by factorial_inv[count_children_res[y] + 1], reducing modulo MOD after each multiplication, then set assign_children_res[x] = r. RESULT assign_children_res]
            [COMMAND Initialize assign_parents_res as a list of n copies of -1 and set assign_parents_res[0] = 1. RESULT assign_parents_res]
            [COMMAND Compute top-down parent assignment values: for each x in order[1:], let p = parent[x], and compute r = assign_children_res[p] * pow(assign_children_res[x], MOD - 2, MOD) * factorial[count_children_res[x] + 1] * factorial_inv[count_children_res[p]] * assign_parents_res[p] * factorial[n - count_children_res[x] - 2] * factorial_inv[n - count_children_res[p] - 1], reducing modulo MOD, then set assign_parents_res[x] = r. RESULT assign_parents_res]
            [COMMAND Initialize res_list as a list of n zeros and set res_list[0] = assign_children_res[0]. RESULT res_list]
            [COMMAND Compute final results for non-root vertices: for i from 1 to n-1 inclusive, set res = factorial[n-1]; for each child j of i, multiply res by assign_children_res[j] and by factorial_inv[count_children_res[j] + 1], reducing modulo MOD after each multiplication; after the child loop, multiply res by assign_parents_res[i] and by factorial_inv[n - 1 - count_children_res[i]], reduce modulo MOD, and set res_list[i] = res. RESULT res_list]
            [COMMAND Return res_list as the final result. RESULT res_list]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
