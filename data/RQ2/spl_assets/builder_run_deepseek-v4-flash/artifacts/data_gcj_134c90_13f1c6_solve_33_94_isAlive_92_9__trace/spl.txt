[DEFINE_WORKER: "Simulates an iterative rock-paper-scissors elimination among A players using cyclic gesture strings, printing the shortest winning counter sequence or IMPOSSIBLE when no valid sequence exists within 500 moves." solve]
    [INPUTS]
        <REF> A </REF>: integer "Number of elements/players to simulate."
        <REF> C </REF>: list of strings "List of cyclic gesture strings; C[i][pos % L[i]] gives the current gesture of element i."
        <REF> L </REF>: list of integers "List of cycle lengths; L[i] is the length of C[i] used for modulo indexing."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ans to an empty string. RESULT ans]
            [COMMAND Initialize pos to 0. RESULT pos]
            [COMMAND Initialize isAlive to a list of True values with length A. RESULT isAlive]
            [COMMAND Start an infinite while True loop. At the top of each iteration, reset possibilities to an empty dict and set possibilities['R'], possibilities['S'], and possibilities['P'] to 0; also reset predR, predS, and predP to empty lists. RESULT possibilities]
            [COMMAND If the debug flag is true, print the current alive list and ans. RESULT _]
            [COMMAND If not any(isAlive), print ans and return from the function. RESULT _]
            [COMMAND For each index i from 0 to A-1 inside the while loop, if isAlive[i] is true: set gesture = C[i][pos % L[i]], mark possibilities[gesture] = 1, append i to predR if gesture == 'R', append i to predS if gesture == 'S', append i to predP if gesture == 'P', and if possibilities['R'] + possibilities['S'] + possibilities['P'] == 3, print 'IMPOSSIBLE' and return. RESULT _]
            [COMMAND After the for loop, assign r, s, and p from possibilities['R'], possibilities['S'], and possibilities['P'] respectively. RESULT r_s_p]
            [COMMAND If the debug flag is true, print the tuple (r, s, p). RESULT _]
            [COMMAND If (r, s, p) == (1, 0, 0), append 'P' to ans, print ans, and return. RESULT _]
            [COMMAND If (r, s, p) == (0, 1, 0), append 'R' to ans, print ans, and return. RESULT _]
            [COMMAND If (r, s, p) == (0, 0, 1), append 'S' to ans, print ans, and return. RESULT _]
            [COMMAND If len(ans) == 500, print 'IMPOSSIBLE', and if debug is true print 'LEN' followed by ans, then return. RESULT _]
            [COMMAND If (r, s, p) == (1, 1, 0), set isAlive[i] = False for every i in predS, then append 'R' to ans. RESULT _]
            [COMMAND If (r, s, p) == (1, 0, 1), set isAlive[i] = False for every i in predR, then append 'P' to ans. RESULT _]
            [COMMAND If (r, s, p) == (0, 1, 1), set isAlive[i] = False for every i in predP, then append 'S' to ans. RESULT _]
            [COMMAND Increment pos by 1. RESULT pos]
            [COMMAND Repeat the while loop from the top using the updated state; continue until a return statement is reached. RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
