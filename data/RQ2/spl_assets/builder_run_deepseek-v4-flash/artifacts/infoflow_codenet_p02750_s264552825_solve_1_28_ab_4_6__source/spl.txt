[DEFINE_WORKER: "Returns the maximum number of items that can be completed within time t by separating zero-first tasks and positive-first tasks, sorting them by their required keys, and using a 30-column dynamic program combined with prefix sums of zero-first task times." solve]
    [INPUTS]
        <REF> n </REF>: int "Number of items; not directly used in the algorithm."
        <REF> t </REF>: int "Available total time threshold for completing items."
        <REF> ab_list </REF>: list[tuple[int,int]] "List of pairs where each pair has a first element a and second element b; a==0 denotes a zero-first task and a>0 denotes a positive-first task."
    [END_INPUTS]

    [OUTPUTS]
        <REF> res </REF>: int "Maximum number of items that can be completed within time t."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Accept input parameters n, t, and ab_list; n is not referenced again in the algorithm. RESULT input_parameters]
            [COMMAND Build a0b_list by filtering ab_list for pairs with ab[0] == 0 and projecting each matching element to ab[1]. RESULT a0b_list]
            [COMMAND Initialize apb_list as an empty list, then iterate over ab_list in original order and append each ab for which ab[0] > 0; elements with ab[0] <= 0 are not appended to apb_list. RESULT apb_list]
            [COMMAND Sort a0b_list in ascending numeric order and reassign it to a0b_list. RESULT a0b_list]
            [COMMAND Sort apb_list by the key -x[0]/(x[1]+1), equivalent to descending x[0]/(x[1]+1), and reassign it to apb_list; evaluating this key requires x[1]+1 != 0. RESULT apb_list]
            [COMMAND Set m = len(apb_list). RESULT m]
            [COMMAND Initialize dp as a list of m+1 rows, each containing 30 entries all equal to t+1, so dp has one row when m is 0. RESULT dp]
            [COMMAND For every i from 0 to m inclusive, set dp[i][0] = 0. RESULT dp]
            [COMMAND For each i from 0 to m-1 inclusive (skipped if m=0) and for each k from 1 to 29 inclusive, compute ti = dp[i][k-1] + 1, unpack a,b = apb_list[i], and set dp[i+1][k] = min(dp[i][k], ti + a*ti + b). RESULT dp]
            [COMMAND Initialize a0b_list_cum_sum = [0] and s = 0. RESULT a0b_list_cum_sum]
            [COMMAND For each b in the sorted a0b_list, update s += b + 1 and append the new s to a0b_list_cum_sum; if a0b_list is empty, a0b_list_cum_sum remains [0]. RESULT a0b_list_cum_sum]
            [COMMAND Initialize res = 0. RESULT res]
            [COMMAND For each k from 0 to len(a0b_list_cum_sum)-1 inclusive and for each p from 0 to 29 inclusive, if dp[m][p] + a0b_list_cum_sum[k] <= t, update res = max(res, k+p). RESULT res]
            [COMMAND Return res as the maximum number of items that can be completed within time t, remaining 0 if no feasible combination is found. RESULT res]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
