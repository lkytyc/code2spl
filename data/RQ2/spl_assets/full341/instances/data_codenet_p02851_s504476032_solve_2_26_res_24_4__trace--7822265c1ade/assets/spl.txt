[DEFINE_WORKER: "Counts the number of subarrays whose adjusted sum (sum of (a_i - 1)) is divisible by k and whose length is strictly less than k, by grouping prefix remainders and counting close equal-remainder positions." solve]
    [INPUTS]
        <REF> n </REF>: int "Number of elements in a_list; range(n) iterates i from 0 to n-1."
        <REF> k </REF>: int "Modulus for prefix remainders and strict upper bound for the distance between counted positions."
        <REF> a_list </REF>: list[int] "List of integer values, each decremented by 1 before building prefix sums."
    [END_INPUTS]

    [OUTPUTS]
        <REF> res </REF>: int "Total number of valid subarrays, equivalently pairs of equal prefix-remainder positions whose index difference is strictly less than k."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute b_list by subtracting 1 from every element in a_list using a list comprehension. RESULT b_list]
            [COMMAND Create res_dict as a collections.defaultdict(list), so missing keys are initialized to empty lists. RESULT res_dict]
            [COMMAND Append initial prefix position 0 to res_dict[0]. RESULT res_dict]
            [COMMAND Initialize r to 0 for the running prefix-sum remainder. RESULT r]
            [COMMAND For each i in range(n) (0 to n-1), update r to (r + b_list[i]) % k. RESULT r]
            [COMMAND In the same for loop iteration over i, append i + 1 to the list at res_dict[r]. RESULT res_dict]
            [COMMAND Initialize res to 0 to accumulate the final count. RESULT res]
            [COMMAND For each key r in res_dict.keys(), assign r_list = res_dict[r]; if len(r_list) == 1, skip this remainder class and continue with the next key. RESULT r_list]
            [COMMAND In the loop over r, after the length-one check, initialize i to 0. RESULT i]
            [COMMAND In the loop over r, after the length-one check, initialize j to 0. RESULT j]
            [COMMAND While i < len(r_list) - 1, execute the pair-counting scan for the current left endpoint i. RESULT i]
            [COMMAND In the while loop over i, if j < len(r_list) - 1, advance j while r_list[j + 1] - r_list[i] < k; if j reaches len(r_list) - 1, break out of the inner advancement loop. RESULT j]
            [COMMAND In the while loop over i, add j - i to res. RESULT res]
            [COMMAND In the while loop over i, increment i by 1. RESULT i]
            [COMMAND Return res as the final result. RESULT res]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
