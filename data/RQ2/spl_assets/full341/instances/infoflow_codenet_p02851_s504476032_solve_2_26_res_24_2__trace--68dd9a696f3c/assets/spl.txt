[DEFINE_WORKER: "Counts the number of pairs of prefix positions with the same cumulative sum modulo k after decrementing each input value by 1, where the distance between the positions is strictly less than k." solve]
    [INPUTS]
        <REF> n </REF>: int "Number of elements in a_list and the exclusive upper bound for the prefix loop."
        <REF> k </REF>: int "Modulus for prefix sums and the strict maximum allowed distance between prefix positions."
        <REF> a_list </REF>: list[int] "Input list of integers, each transformed by subtracting 1 before building prefix sums."
    [END_INPUTS]

    [OUTPUTS]
        <REF> res </REF>: int "Total count of valid prefix-index pairs satisfying the equal-remainder and distance conditions."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Construct b_list by computing a - 1 for each element a in a_list. RESULT b_list]
            [COMMAND Initialize res_dict as a defaultdict with list factory so every missing key maps to an empty list. RESULT res_dict]
            [COMMAND Append prefix index 0 to res_dict under key 0. RESULT res_dict]
            [COMMAND Initialize running prefix remainder r to 0. RESULT r]
            [COMMAND For each i from 0 to n-1, add b_list[i] to r, set r = r modulo k, and append i+1 to res_dict[r]. RESULT res_dict]
            [COMMAND Initialize total result res to 0. RESULT res]
            [COMMAND For each key r in res_dict.keys(), assign r_list = res_dict[r]; if len(r_list) == 1, skip to the next remainder key. RESULT r_list]
            [COMMAND Within the remainder-list processing, initialize i=0 and j=0; while i < len(r_list)-1, if j < len(r_list)-1, advance j while r_list[j+1] - r_list[i] < k, breaking after j reaches len(r_list)-1; then add j - i to res and increment i. RESULT res]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
