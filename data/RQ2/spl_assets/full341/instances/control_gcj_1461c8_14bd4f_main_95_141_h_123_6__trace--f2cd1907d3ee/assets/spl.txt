[DEFINE_WORKER: "Executes a fixed 100-step interactive probing protocol, tracks the row index with the smallest returned count across rows 8..20, and finally prints that chosen row." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare integer variables i, j, n, m, k, a, b, c, h, w, r, l, t, p, and initialize r to 0. RESULT declarations]
            [COMMAND Read an integer into t from standard input using scanf, without checking the return value; t initially controls the outer loop, but is reassigned later. RESULT t]
            [COMMAND Start the outer loop with i = 0; after each iteration increment i, and continue while i is less than the current value of t, which changes inside the loop body. RESULT i]
            [COMMAND At the beginning of each outer iteration, set h = -1. RESULT h]
            [COMMAND At the beginning of each outer iteration, set t = inf, a large sentinel value, overwriting the previous loop count. RESULT t]
            [COMMAND At the beginning of each outer iteration, set p = 7. RESULT p]
            [COMMAND Start the inner loop with j = 1; after each iteration increment j, and continue while j <= 100. RESULT j]
            [COMMAND At the top of every inner iteration, read one integer into w with scanf; w is subsequently unused. RESULT w]
            [COMMAND In the inner loop, if j <= 56, print computed row 1 + ((j - 1) / 8), literal column 1, and a newline using printf, then flush stdout. RESULT stdout]
            [COMMAND In the inner loop, if 57 <= j <= 62, print literal row 1 and column 0 with a newline, flush stdout, read an integer c with scanf, then consume c additional integers into w using a for loop from k = 0 while k < c. RESULT c]
            [COMMAND In the inner loop, if 63 <= j <= 75, print j - 55 as the row and 0 as the column with a newline, flush stdout, read an integer c with scanf, then consume c additional integers into w using a for loop from k = 0 while k < c. RESULT c]
            [COMMAND In the inner loop, for 63 <= j <= 75 after reading and consuming the c integers, if c <= t, update t = c and h = j - 55. RESULT h]
            [COMMAND In the inner loop, if 76 <= j <= 99, increment p when j is even; then while p == h increment p; finally print p and 1 with a newline and flush stdout. RESULT p]
            [COMMAND In the inner loop, if j == 100, print h and literal 100 with a newline using printf, then flush stdout. RESULT stdout]
            [COMMAND After the inner loop for j and the outer loop for i terminate, return 0 from main. RESULT exit]
        [END_SEQUENTIAL_BLOCK]
