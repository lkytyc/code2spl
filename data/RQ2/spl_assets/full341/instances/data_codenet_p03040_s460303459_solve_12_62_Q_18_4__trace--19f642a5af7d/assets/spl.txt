[DEFINE_WORKER: "Processes a sequence of commands using two heaps to maintain the lower and upper halves of inserted values, and for each query prints the current median candidate x with a cumulative B-adjusted sum." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer Q from ir as the number of commands. RESULT Q]
            [COMMAND Initialize max-heap f using new PriorityQueue<Long>(Collections.reverseOrder()), min-heap l using new PriorityQueue<Long>(), long fsum = 0, long lsum = 0, and long B = 0. RESULT f, l, fsum, lsum, B]
            [COMMAND Start while loop with condition Q-- > 0; the body executes while the value of Q before decrement is greater than 0, and Q is decremented by 1 after each condition check. RESULT loop_control]
            [COMMAND Inside the while loop, read integer com from ir. RESULT com]
            [COMMAND If com == 1, read long a from ir. RESULT a]
            [COMMAND If com == 1 and f.size() == l.size() and l.size() == 0, add a to f and add a to fsum. RESULT f, fsum]
            [COMMAND If com == 1 and f.size() == l.size() and l.size() != 0, set x = l.peek(). RESULT x]
            [COMMAND If com == 1 and f.size() == l.size() and l.size() != 0 and x < a, update lsum += a - x, fsum += x, poll x from l, add a to l, and add x to f. RESULT f, l, fsum, lsum]
            [COMMAND If com == 1 and f.size() == l.size() and l.size() != 0 and x >= a, add a to f and add a to fsum. RESULT f, fsum]
            [COMMAND If com == 1 and f.size() != l.size(), set x = f.peek(). RESULT x]
            [COMMAND If com == 1 and f.size() != l.size() and x <= a, add a to l and add a to lsum. RESULT l, lsum]
            [COMMAND If com == 1 and f.size() != l.size() and x > a, update lsum += x, fsum += a - x, poll x from f, add a to f, and add x to l. RESULT f, l, fsum, lsum]
            [COMMAND If com == 1, read the next long from ir and add it to B. RESULT B]
            [COMMAND If com != 1, set x = f.peek(). RESULT x]
            [COMMAND If com != 1 and f.size() == l.size(), print x and the expression lsum - fsum + B. RESULT printed_output]
            [COMMAND If com != 1 and f.size() != l.size(), print x and the expression lsum - fsum + x + B. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
