[DEFINE_WORKER: "Reads a number of test cases, stores each case's words in a TreeSet to sort and de-duplicate them, and performs nested substring comparisons whose computed values are discarded." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not used by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a BufferedReader named br that wraps InputStreamReader(System.in). RESULT br]
            [COMMAND Read the first line from br and parse it as an integer named tc. RESULT tc]
            [COMMAND Start the outer for loop over test cases with i initialized to 0, continuing while i < tc, and incrementing i by 1 after each iteration. RESULT i]
            [COMMAND Within the outer for loop over i, read one line from br and parse it as an integer named noOfWords. RESULT noOfWords]
            [COMMAND Within the outer for loop over i, create an empty TreeSet named set to store strings in sorted order and remove duplicates. RESULT set]
            [COMMAND Within the outer for loop over i, start a for loop with j initialized to 0, continuing while j < noOfWords, and incrementing j by 1 after each iteration. RESULT j]
            [COMMAND Within the j loop, read a line from br as word and add it to set. RESULT set]
            [COMMAND After reading all words for the current test case, create an Iterator named wrd1 over set. RESULT wrd1]
            [COMMAND After reading all words for the current test case, create an Iterator named wrd2 over set; wrd2 is not reset afterward. RESULT wrd2]
            [COMMAND Set c1 to 0. RESULT c1]
            [COMMAND Enter the outer while loop and continue while wrd1.hasNext() is true. RESULT none]
            [COMMAND Within the outer while loop, increment c1 by 1. RESULT c1]
            [COMMAND Within the outer while loop, set c2 to 0. RESULT c2]
            [COMMAND Within the outer while loop, set wordOne to wrd1.next(). RESULT wordOne]
            [COMMAND Within the outer while loop, set result to 0; result is never subsequently read. RESULT result]
            [COMMAND Within the outer while loop, enter the inner while loop and continue while wrd2.hasNext() is true. Because wrd2 is not reset, after the first outer iteration wrd2.hasNext() is normally false and this inner loop is skipped. RESULT none]
            [COMMAND Within the inner while loop, increment c2 by 1. RESULT c2]
            [COMMAND Within the inner while loop, when c1 equals c2, do not execute the if block, so wrd2 is not advanced and no substring comparison is performed for that iteration. RESULT none]
            [COMMAND Within the inner while loop, when c1 is not equal to c2, set wordTwo to wrd2.next(); when the condition is false, wrd2 is not advanced. RESULT wordTwo]
            [COMMAND When c1 is not equal to c2, initialize max to 0 and key to an empty string for the substring search. RESULT max]
            [COMMAND When c1 is not equal to c2, start an l loop with l initialized to 0, continuing while l < wordTwo.length() - 1, and incrementing l by 1 after each iteration. RESULT l]
            [COMMAND Within the l loop, start a k loop with k initialized to l + 2, continuing while k <= wordTwo.length(), and incrementing k by 1 after each iteration. RESULT k]
            [COMMAND Within the k loop, using short-circuit &&, if wordOne contains wordTwo.substring(l, k) and k - l is greater than max, set max to k - l and key to wordTwo.substring(l, k); these values are not used outside the if block. RESULT max]
            [COMMAND Within the k loop, if the preceding condition fails, break out of the k loop only. RESULT none]
            [COMMAND Continue executing all loop levels until the outer for loop over i completes. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: BufferedReader.readLine() encounters an I/O error.]
        [LOG "IOException is propagated from main because there is no catch block."]
        [THROW IOException "Indicates an I/O error occurred while reading from standard input."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: An input line passed to Integer.parseInt is not a valid integer, or readLine() returns null before a required integer is read.]
        [LOG "NumberFormatException is thrown while parsing the input as an int."]
        [THROW NumberFormatException "Indicates that the input string is null or cannot be parsed as an integer."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
