[DEFINE_WORKER: "Reads a test-case count and then for each case reads a waffle grid, counts chocolate chips per row, and determines whether a single horizontal cut and a single vertical cut can divide the chocolate chips into four equal quadrants, printing POSSIBLE or IMPOSSIBLE; H and V are parsed but ignored." main]
    [INPUTS]
        <REF> argc </REF>: int "Number of command-line arguments; ignored by the method."
        <REF> argv </REF>: char ** "Command-line argument vector; ignored by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the first input line into str, parse it as an integer into T using sscanf, free str, and set u_w to 1. RESULT T]
            [COMMAND While the old value of T is nonzero, decrement T and process one waffle test case. RESULT T]
            [COMMAND At the start of each test case, declare R, C, H, V, loop indices i, j, k, l, m, Waffle[10], and ChocChipCount[10]; initialize ChocChipCount[0] through ChocChipCount[9] to 0. RESULT ChocChipCount]
            [COMMAND Read the next input line into str, parse four integers into R, C, H, and V using sscanf, and free str; H and V are not used later. RESULT R, C, H, V]
            [COMMAND Read R waffle rows: for i from 0 to R-1 inclusive, read the next input line into Waffle[i]. RESULT Waffle]
            [COMMAND For each row i from 0 to R-1 inclusive and each column j from 0 to C-1 inclusive, increment ChocChipCount[i] when Waffle[i][j] is '@'. RESULT ChocChipCount]
            [COMMAND Set retStr to "IMPOSSIBLE". RESULT retStr]
            [COMMAND For each horizontal cut candidate i from 1 to R-1 inclusive, while retStr is exactly "IMPOSSIBLE", set topCount to the sum of ChocChipCount[0] through ChocChipCount[i-1] and bottomCount to the sum of ChocChipCount[i] through ChocChipCount[R-1]. RESULT topCount, bottomCount]
            [COMMAND If topCount != bottomCount, continue to the next horizontal cut candidate i without searching any vertical cut. RESULT i]
            [COMMAND When topCount == bottomCount, for each vertical cut candidate k from 1 to C-1 inclusive, initialize topleftCount and toprightCount to 0. RESULT k]
            [COMMAND Within the k-loop, for each top row l from 0 to i-1 inclusive, add ChocChipCount[l] to toprightCount and, for each column m from 0 to k-1 inclusive, increment topleftCount if Waffle[l][m] is '@'. RESULT topleftCount, toprightCount]
            [COMMAND Subtract topleftCount from toprightCount to obtain the chip count in the top-right quadrant. RESULT toprightCount]
            [COMMAND If topleftCount != toprightCount, continue to the next vertical cut candidate k. RESULT k]
            [COMMAND When topleftCount == toprightCount, initialize bottomleftCount and bottomrightCount to 0. RESULT bottomleftCount, bottomrightCount]
            [COMMAND Within the same k candidate, for each bottom row l from i to R-1 inclusive, add ChocChipCount[l] to bottomrightCount and, for each column m from 0 to k-1 inclusive, increment bottomleftCount if Waffle[l][m] is '@'. RESULT bottomleftCount, bottomrightCount]
            [COMMAND Subtract bottomleftCount from bottomrightCount to obtain the chip count in the bottom-right quadrant. RESULT bottomrightCount]
            [COMMAND If bottomleftCount != bottomrightCount, continue to the next vertical cut candidate k. RESULT k]
            [COMMAND If bottomleftCount == bottomrightCount, set retStr to "POSSIBLE" and break out of the k-loop only; the enclosing i-loop stops on its next condition check because retStr is no longer "IMPOSSIBLE". RESULT retStr]
            [COMMAND After the search completes, print Case #%d: %s using u_w and retStr, then increment u_w. RESULT u_w]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
