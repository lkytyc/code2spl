[DEFINE_WORKER: "Reads and validates N and triples A, B, C, applies a dynamic-programming recurrence using max1, and prints the maximum final SUM." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize N to 0, SUM to 0, sumA to 0, sumB to 0, sumC to 0, ref to 0, and N_limit to pow(10,5), i.e. 100000. RESULT N, SUM, sumA, sumB, sumC, ref, N_limit]
            [COMMAND Read an integer from standard input into N using scanf("%d", &N). If scanf returns a value <= 0, terminate with exit code 1. RESULT N]
            [COMMAND Validate N: if N <= 0 or N_limit < N, terminate with exit code 1. RESULT N]
            [COMMAND Declare integer pointers A, B, C and allocate N-element integer arrays via A = malloc(N * sizeof(int)), B = malloc(N * sizeof(int)), C = malloc(N * sizeof(int)). RESULT A, B, C]
            [COMMAND For each i from 0 to N-1, read three integers into A[i], B[i], and C[i] using scanf("%d%d%d", &A[i], &B[i], &C[i]); if fewer than 3 values are successfully read, terminate with exit code 1. RESULT A, B, C]
            [COMMAND Within the same for loop over i from 0 to N-1, validate the read triple with short-circuiting else-if checks: if A[i] <= 0 or A[i] > 10000 terminate; else if B[i] <= 0 or B[i] > 10000 terminate; else if C[i] <= 0 or C[i] > 10000 terminate, each with exit code 1. RESULT A, B, C]
            [COMMAND Initialize ref_A to 0, ref_B to 0, and ref_C to 0. RESULT ref_A, ref_B, ref_C]
            [COMMAND For each i from 0 to N-1, if i - 1 <= 0 (i.e. i is 0 or 1), set sumA = A[i], sumB = B[i], sumC = C[i], then set ref_A = sumA, ref_B = sumB, ref_C = sumC, and continue to the next iteration. RESULT sumA, sumB, sumC, ref_A, ref_B, ref_C]
            [COMMAND Within the same loop, for i - 1 > 0 (i.e. i > 1), set sumA = A[i] + max1(ref_B, ref_C), sumB = B[i] + max1(ref_C, ref_A), and sumC = C[i] + max1(ref_A, ref_B). RESULT sumA, sumB, sumC]
            [COMMAND Within the same loop, after computing sumA, sumB, and sumC for i > 1, set ref_A = sumA, ref_B = sumB, and ref_C = sumC. RESULT ref_A, ref_B, ref_C]
            [COMMAND After the second loop, compute SUM by evaluating the original nested conditional chain: if sumA >= sumB, then if sumA >= sumC set SUM = sumA else SUM = sumC; else if sumB >= sumA, then if sumB >= sumC set SUM = sumB else SUM = sumC; else if sumC >= sumB, then if sumC >= sumA set SUM = sumC else SUM = sumA. RESULT SUM]
            [COMMAND Print SUM to standard output using printf("%d", SUM) without adding a newline. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
