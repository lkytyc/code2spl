[DEFINE_WORKER: "Builds a falling-balls mirror grid matching the target per-column ball counts, returning the grid height and rows, or ('IMPOSSIBLE', None) when an edge column count is zero." solve]
    [INPUTS]
        <REF> B </REF>: list of int "Target ball count for each column; must be non-empty and have non-zero first and last entries for a possible solution. B is mutated during the computation."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: tuple "Returns ('IMPOSSIBLE', None) when B[0] or B[-1] is zero; otherwise returns (len(ans), ans), where ans is a list of rows containing dot, backslash, and forward slash characters."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND If B[0] == 0 or B[-1] == 0, return ('IMPOSSIBLE', None). RESULT early_return]
            [COMMAND Initialize grid to [] and set C = len(B); reinitialize grid to [] (the first assignment is dead). RESULT grid,C]
            [COMMAND For j in range(C), create row = []; for i in range(C), append '.' to row; after the inner loop, append row to grid. RESULT grid]
            [COMMAND Set shadowed = [False] * len(C), then set shadowed[0] = True. RESULT shadowed]
            [COMMAND Decrement B[0] by 1 and decrement B[-1] by 1. RESULT B]
            [COMMAND Set i = 0. RESULT i]
            [COMMAND For i in range(C), if B[i] == 0 and shadowed[i], continue to the next iteration of the loop. RESULT loop_continue]
            [COMMAND Within the same for loop, if B[i] > 0 and not shadowed[i], set unshadowed_balls = 0, j = i - 1, l = 1; while j >= 1 and not shadowed[j], write a backslash at grid[l][j], set shadowed[j] = True, increment unshadowed_balls and l, and decrement j. RESULT grid,shadowed,unshadowed_balls,j,l]
            [COMMAND Still within that if branch, if B[i] <= unshadowed_balls, set excess_balls = unshadowed_balls - B[i] + 1 and j = i; while excess_balls > 1, decrement j and excess_balls, then set shadowed[j] = False. RESULT shadowed]
            [COMMAND In the else of that branch (when B[i] > unshadowed_balls), subtract unshadowed_balls + 1 from B[i] and set shadowed[i] = True. RESULT B,shadowed]
            [COMMAND Within the same for loop, after the leftward unshadowed pass, if B[i] > 0 and shadowed[i], set r = B[i], j = i + 1, l = 1; while j < C - 1 and shadowed[j], write '/' at grid[l][j], then increment l and j; then while j < C - 1 and r > 0, write '/' at grid[l][j], set shadowed[j] = True, decrement r, and increment l and j. RESULT grid,shadowed,r]
            [COMMAND Initialize ans = []. RESULT ans]
            [COMMAND For index in range(1, C), set row = grid[index] and include_row = False; for each cell in row, if cell is a backslash or '/', set include_row = True and break; after scanning, if include_row is True append row to ans, otherwise break out of the outer loop. RESULT ans]
            [COMMAND Prepend a row of C '.' characters to ans by assigning ans = [['.' for i in range(C)]] + ans. RESULT ans]
            [COMMAND Return (len(ans), ans). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
