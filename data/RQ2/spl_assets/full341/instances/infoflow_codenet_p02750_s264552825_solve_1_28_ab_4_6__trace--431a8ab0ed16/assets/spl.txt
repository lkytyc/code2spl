[DEFINE_WORKER: "Computes the maximum number of selectable pairs that fit within time budget t by separating zero-initial pairs, sorting positive-initial pairs, applying a bounded dynamic program, and combining with cumulative costs." solve]
    [INPUTS]
        <REF> n </REF>: int "Total number of pairs; accepted as an argument but not referenced inside the method."
        <REF> t </REF>: int "Time budget threshold used to constrain the final selection."
        <REF> ab_list </REF>: list[tuple[int,int]] "List of pairs (a,b) to be filtered, sorted, and selected."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Maximum number of items that can be selected within the time budget, computed as the largest k + p satisfying the cost constraint."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Filter ab_list into a0b_list by selecting ab[1] for every pair ab with ab[0] == 0. RESULT a0b_list]
            [COMMAND Initialize apb_list as an empty list. RESULT apb_list]
            [COMMAND For each ab in ab_list, if ab[0] > 0, append ab to apb_list. RESULT apb_list]
            [COMMAND Sort a0b_list in ascending natural order. RESULT a0b_list]
            [COMMAND Sort apb_list using the key -x[0] / (x[1] + 1), preserving the exact ordering defined by that key. RESULT apb_list]
            [COMMAND Set m to the length of apb_list. RESULT m]
            [COMMAND Initialize dp as a list of m + 1 rows, each row containing 30 entries all initialized to t + 1. RESULT dp]
            [COMMAND For each i from 0 through m inclusive, set dp[i][0] to 0. RESULT dp]
            [COMMAND For each i from 0 through m - 1 inclusive, and for each k from 1 through 29 inclusive, compute ti = dp[i][k - 1] + 1, unpack a and b from apb_list[i], and set dp[i + 1][k] = min(dp[i][k], ti + a * ti + b). RESULT dp]
            [COMMAND Initialize a0b_list_cum_sum as a list containing the single value 0. RESULT a0b_list_cum_sum]
            [COMMAND Set s to 0. RESULT s]
            [COMMAND For each b in a0b_list, increment s by b + 1 and append the updated s to a0b_list_cum_sum. RESULT a0b_list_cum_sum]
            [COMMAND Set res to 0. RESULT res]
            [COMMAND For each k from 0 through len(a0b_list_cum_sum) - 1 inclusive, and for each p from 0 through 29 inclusive, if dp[m][p] + a0b_list_cum_sum[k] <= t, update res to max(res, k + p). RESULT res]
            [COMMAND Return the value of res. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
