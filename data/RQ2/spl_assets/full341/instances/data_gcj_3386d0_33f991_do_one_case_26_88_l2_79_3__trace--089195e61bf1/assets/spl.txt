[DEFINE_WORKER: "Reads N points, groups every pair by normalized direction, then for each direction computes the maximum chain length from directed adjacency lists and prints the best result as Case #cnum." do_one_case]
    [INPUTS]
        <REF> cnum </REF>: integer "Current test case number used in the output line."
    [END_INPUTS]

    [OUTPUTS]
        <REF> printed_output </REF>: string "Formatted line 'Case #%d: %d' written to standard output, where the first placeholder is cnum and the second is the computed maximum m."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read one integer from input using rd1int() and assign it to N. RESULT N]
            [COMMAND Initialize XY as an empty list. RESULT XY]
            [COMMAND For i from 0 to N-1 inclusive, read two integers via rdints(), unpack them into x and y, and append the tuple (x, y) to XY. RESULT XY]
            [COMMAND Sort XY in ascending lexicographic order. RESULT XY]
            [COMMAND Initialize A as an empty dictionary. RESULT A]
            [COMMAND For each j from 1 to N-1 inclusive, for each i from 0 to j-1 inclusive, read XY[j] as (x2, y2) and XY[i] as (x1, y1), compute dx = x2 - x1, dy = y2 - y1, g = gcd(dx, dy), and normalize the direction to (a, b) = (dx / g, dy / g) using integer division. If (a, b) is already a key in A, append (i, j) to A[(a, b)] and (j, i) to A[(-a, -b)]; otherwise initialize A[(a, b)] with [(i, j)] and A[(-a, -b)] with [(j, i)]. RESULT A]
            [COMMAND Initialize m to 1. RESULT m]
            [COMMAND For each key (a, b) in A in dict iteration order, retrieve E = A[(a, b)] and initialize I and O as empty dictionaries. RESULT E]
            [COMMAND Within the current direction loop, for each edge (i, j) in E, append j to O[i], creating O[i] = [j] if i is not already in O, and append i to I[j], creating I[j] = [i] if j is not already in I. RESULT O_and_I]
            [COMMAND Within the current direction loop, compute NI = set(range(N)) - set(I), the set of all node indices from 0 to N-1 that have no incoming edge. RESULT NI]
            [COMMAND Within the current direction loop, initialize C as an empty list. RESULT C]
            [COMMAND Within the current direction loop, for each x in NI in the iteration order of the Python set produced by set(range(N)) - set(I), if x is present in O, build Q = [(0, x)] + [(len(I[y]), y) for y in O[x]], sort Q, and append the node sequence [y for _, y in Q] to C; otherwise append the singleton [x] to C. RESULT C]
            [COMMAND Within the current direction loop, partition C into C1 = chains of length exactly 1, C2 = chains of even length, and C3 = chains of odd length greater than 1. RESULT C1_C2_C3]
            [COMMAND Within the current direction loop, while len(C3) > 1, append the concatenation of the last two chains in C3 to C2 and delete those two chains from C3. RESULT C2_C3]
            [COMMAND Within the current direction loop, set C3 = (C3 + C1)[:2], keeping at most two chains from the remaining odd chains and singleton chains. RESULT C3]
            [COMMAND Within the current direction loop, compute l2 = sum of the lengths of all chains in C2 and l3 = sum of the lengths of all chains in C3. RESULT l2_l3]
            [COMMAND Within the current direction loop, if l3 is nonzero, set l = l2 + l3; else if len(C2) > 1, set l = l2 - 1; else set l = l2. RESULT l]
            [COMMAND Within the current direction loop, set m = max(m, l). RESULT m]
            [COMMAND Print the line 'Case #%d: %d' formatted with cnum and m. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
