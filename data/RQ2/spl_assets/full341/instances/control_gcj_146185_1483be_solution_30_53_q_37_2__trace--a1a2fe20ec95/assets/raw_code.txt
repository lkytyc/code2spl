1   import java.util.Scanner;
2   import java.util.*;
3   public class Solution{
4       public static void main(String[] args) {
5           Scanner input = new Scanner(System.in);
6           int t = input.nextInt();
7           for(int t_0=0;t_0<t;t_0++) {
8               int numOfElem = input.nextInt();
9               Map<Integer, List<Integer>> graph = new HashMap<>();
10              for(int i = 0;i<numOfElem;i++){
11                  int x = input.nextInt()-1;
12                  int y = input.nextInt()-1;
13                  if(!graph.containsKey(x)) {
14                      graph.put(x, new LinkedList<>());
15                  }
16                  if(!graph.containsKey(y)) {
17                      graph.put(y, new LinkedList<>());
18                  }
19                  graph.get(x).add(i);
20                  graph.get(y).add(i);
21              }
22              int[] startElem = new int[numOfElem];
23              for(int i=0;i<numOfElem;i++) {
24                  startElem[i] = input.nextInt();
25              }
26              int solution = !graph.containsKey(0)?startElem[0]:solution(graph, numOfElem, startElem);
27              System.out.println("Case #" + (t_0+1) + ": " + (solution==-1?"UNBOUNDED":solution));
28          }
29      }
30      private static int solution(Map<Integer, List<Integer>> graph, int numOfElem, int[] startElem) {
31              Set<Integer> visited = new HashSet<>();
32              visited.add(0);
33              Queue<Integer> q = new LinkedList<>();
34              int result = startElem[0];
35              for(int i: graph.get(0)) {
36                  if(i != 0)
37                      q.add(i);
38              }
39              boolean acyclic = false;
40              while(!q.isEmpty()) {
41                  int next = q.poll();
42                  for(int neighbor: graph.get(next)) {
43                      if(visited.contains(neighbor)) {
44                          acyclic = true;
45                      } else {
46                          result += startElem[neighbor];
47                          q.add(neighbor);
48                          visited.add(neighbor);
49                      }
50                  }
51              }
52              return acyclic?(result==0?0:-1):result;
53      }
54  }
