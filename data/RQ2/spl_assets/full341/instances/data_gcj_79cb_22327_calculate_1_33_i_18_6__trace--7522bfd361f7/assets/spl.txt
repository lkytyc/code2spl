[DEFINE_WORKER: "Separates the input list into odd-indexed and even-indexed sublists, sorts both, and checks whether the alternating sequence formed from the sorted sublists is non-decreasing; returns the 0-based position of the earlier element in the first strict decrease as a string, otherwise returns 'OK'." calculate]
    [INPUTS]
        <REF> arr </REF>: list "Input list of values to partition by original index parity and then compare after sorting."
        <REF> size </REF>: integer "Number of elements of arr to process; also the upper bound of the xrange loops."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: string "Returns 'OK' when the alternating sorted sequence is non-decreasing; otherwise returns str(i - 1), the 0-based position of the earlier element in the first strict decrease."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize even_arr as a zero-filled list of length floor(size / 2), odd_arr as a zero-filled list of length size - len(even_arr), and set eveni = 0 and oddi = 0. RESULT partitioning_arrays_initialized]
            [COMMAND For each i from 0 through size - 1, if (i + 1) % 2 == 0 (i is odd), assign arr[i] to even_arr[eveni] and increment eveni; otherwise assign arr[i] to odd_arr[oddi] and increment oddi. RESULT partitioned_arrays]
            [COMMAND Sort even_arr in ascending order, then sort odd_arr in ascending order. RESULT sorted_arrays]
            [COMMAND Reset eveni = 0 and oddi = 0, and set last = -1. RESULT verification_state_initialized]
            [COMMAND For i from 0 through size - 1, if i == 0, set last = odd_arr[oddi], increment oddi, and continue to the next iteration. RESULT last_initialized]
            [COMMAND For i from 1 through size - 1, when i is odd compare even_arr[eveni] with last; when i is even compare odd_arr[oddi] with last. If the compared value is not less than last, assign it to last and increment the corresponding index. If it is less than last, follow the matching alternative flow. RESULT verification_state_updated]
            [COMMAND If the verification loop completes without taking an alternative flow, return the string 'OK'. RESULT ok_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: During the verification loop, for i from 1 through size - 1, i is odd and even_arr[eveni] < last.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return str(i - 1) immediately. RESULT early_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: During the verification loop, for i from 1 through size - 1, i is even and odd_arr[oddi] < last.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return str(i - 1) immediately. RESULT early_return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
