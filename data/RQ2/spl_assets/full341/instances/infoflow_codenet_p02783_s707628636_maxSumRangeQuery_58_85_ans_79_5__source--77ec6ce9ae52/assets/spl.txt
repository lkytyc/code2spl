[DEFINE_WORKER: "Computes the maximum possible sum over all inclusive query ranges after permuting nums by assigning larger values to positions with higher range coverage, and returns the result modulo 1,000,000,007." maxSumRangeQuery]
    [INPUTS]
        <REF> nums </REF>: array of integers "Array of integers to permute for maximizing the range-query sum."
        <REF> req </REF>: array of integer index pairs "List of inclusive ranges [start, end] for which the sums are accumulated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: integer "Total sum over all requested ranges after optimal permutation, modulo 1,000,000,007."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a difference array l as a list of zeros with length len(nums) + 1. RESULT l]
            [COMMAND Sort nums in descending order, in place. RESULT nums]
            [COMMAND For each request i in req, increment l[i[0]] by 1 and decrement l[i[1] + 1] by 1, where i[1] + 1 is the sentinel position just after the inclusive range end. RESULT l]
            [COMMAND Convert l into a prefix-count array by iterating i from 1 through len(l) - 1 inclusive and adding l[i - 1] to l[i]. RESULT l]
            [COMMAND Remove the final sentinel element from l by reassigning l = l[:-1]. RESULT l]
            [COMMAND Create a defaultdict d with list factory that maps each coverage count to the list of indices having that count, by appending i to d[l[i]] for every index i in l. RESULT d]
            [COMMAND Create di as an OrderedDict containing d's items sorted ascending by coverage count. RESULT di]
            [COMMAND Initialize k to 0. RESULT k]
            [COMMAND Initialize ans as a list of zeros with length len(nums). RESULT ans]
            [COMMAND For each coverage-count key i in di in ascending order, for each index j stored in di[i], assign ans[j] = nums[len(nums) - k - 1] and then increment k by 1. RESULT ans]
            [COMMAND Initialize accumulator c to 0. RESULT c]
            [COMMAND For each request i in req, add sum(ans[i[0]:i[1] + 1]), the inclusive slice from start through end, to c. RESULT c]
            [COMMAND Return c modulo 10**9 + 7. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
