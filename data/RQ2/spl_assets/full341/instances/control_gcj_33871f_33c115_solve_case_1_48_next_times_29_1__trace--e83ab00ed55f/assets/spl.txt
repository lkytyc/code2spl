[DEFINE_WORKER: "Resolves signed receipt times into absolute times, computes a latency for each edge as a strict absolute-time difference, and returns all latencies as a space-separated string." solve_case]
    [INPUTS]
        <REF> in_times </REF>: list of ints "Signed receipt times for nodes; positive values are known absolute times and negative values are resolved using their absolute values."
        <REF> edges </REF>: list of pairs of ints "Original 1-based edge endpoint pairs; one latency is stored at the matching edge index."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: string "Space-separated latencies for all edges; unresolved entries remain 1000000."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize latencies as a list of length len(edges) with every element set to 1000000. RESULT latencies]
            [COMMAND Rebind edges to a new list where each original edge tuple (a, b) is replaced by (a - 1, b - 1). RESULT edges]
            [COMMAND Initialize abs_times as a list of length len(in_times) with every element set to 1000000. RESULT abs_times]
            [COMMAND For each index, i in enumerate(in_times), if i is strictly greater than 0, set abs_times[index] to i. RESULT abs_times_after_positives]
            [COMMAND Initialize pos_times to an empty list. RESULT pos_times]
            [COMMAND For each index, i in enumerate(in_times), if i is strictly less than 0, append the tuple (abs(i), index) to pos_times. RESULT pos_times_initial]
            [COMMAND Sort pos_times in ascending order by tuple values, so by absolute time first and then by index. RESULT pos_times_sorted]
            [COMMAND Set current_time to 0 and received to 1. RESULT current_time_and_received]
            [COMMAND In each iteration of the outer while loop while pos_times is non-empty, if pos_times[0][0] equals received, increment current_time by 1, set new_received to 0, then while pos_times is non-empty and pos_times[0][0] equals received, consume each leading tuple by setting abs_times[b] to current_time, removing that tuple from the front of pos_times, and incrementing new_received; after the inner while, add new_received to received. RESULT abs_times_resolved]
            [COMMAND In each iteration of the outer while loop while pos_times is non-empty, if pos_times[0][0] does not equal received, build next_times from all abs_times values strictly greater than current_time, set current_time to min(next_times), and count abs_times values less than or equal to current_time into received_count without using it; this branch leaves pos_times and received unchanged, so the same else branch is reached again unless min(next_times) raises ValueError because next_times is empty. RESULT current_time_after_jump]
            [COMMAND After the outer while loop ends, prepend 0 to abs_times. RESULT abs_times_with_zero]
            [COMMAND Build sorted_receipt as a list of (value, index) pairs for each index, value in enumerate(abs_times_with_zero). RESULT sorted_receipt]
            [COMMAND Sort sorted_receipt in ascending order by value and then by index. RESULT sorted_receipt_sorted]
            [COMMAND For each time, comp in sorted_receipt_sorted, skipping the first element and ignoring time, iterate over edges in order; for each edge index, (A, B), if A equals comp and abs_times[A] is strictly greater than abs_times[B], set latencies[index] to abs_times[A] - abs_times[B] and break the inner edge loop; otherwise if B equals comp and abs_times[B] is strictly greater than abs_times[A], set latencies[index] to abs_times[B] - abs_times[A] and break the inner edge loop; if neither condition is met, continue scanning edges. RESULT latencies_final]
            [COMMAND Return the elements of latencies_final converted to strings and joined by single spaces. RESULT result_string]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: In the outer while loop's else branch, next_times is empty because no abs_times value is strictly greater than current_time.]
        [LOG "min(next_times) is called with an empty list during solve_case while trying to advance current_time"]
        [THROW ValueError "Raised by min() when next_times has no elements."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
