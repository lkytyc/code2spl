[DEFINE_WORKER: "Processes multiple test cases containing a chocolate-chip grid and cut counts, builds a 2D prefix sum, and determines whether the grid can be divided so every resulting piece contains the same number of chips." main]
    [INPUTS]
        <REF> stdin </REF>: stream "First line contains t. For each test case, a line contains r, c, h, and v, followed by r grid lines of c characters where '@' marks a chip."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: stream "For each test case, prints Case #X: POSSIBLE or Case #X: IMPOSSIBLE."
        <REF> return_value </REF>: int "Always returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the number of test cases t from standard input. RESULT t]
            [COMMAND Set tc to 1. RESULT tc]
            [COMMAND While tc is less than or equal to t, process one test case. RESULT test_case_loop]
            [COMMAND Within the test-case loop, read integers r, c, h, and v from standard input. RESULT case_params]
            [COMMAND Within the test-case loop, set guests = (h + 1) * (v + 1). RESULT guests]
            [COMMAND Within the test-case loop, set chips to 0. RESULT chips]
            [COMMAND Within the test-case loop, loop i from 0 to r inclusive. If i is not 0, read a row string str of c characters from standard input. RESULT row_string]
            [COMMAND Within the test-case loop and the outer i loop, for j from 0 to c inclusive, if i is 0 or j is 0, set A[i][j] and B[i][j] to 0. RESULT padding_cells]
            [COMMAND Within the test-case loop and the outer i loop, when i is not 0, for j from 1 to c inclusive, set A[i][j] to 1 if str[j - 1] is '@', otherwise set it to 0, and increment chips when A[i][j] is 1. RESULT A_and_chips]
            [COMMAND Within the test-case loop, if chips % guests is not 0, print 'Case #' followed by tc and ': IMPOSSIBLE' on one line, and skip the following prefix-sum and search steps for this test case. RESULT impossible_printed]
            [COMMAND Within the test-case loop, when chips % guests is 0, for i from 1 to r inclusive and for j from 1 to c inclusive, compute B[i][j] = A[i][j] + B[i - 1][j] + B[i][j - 1] - B[i - 1][j - 1]. RESULT prefix_B]
            [COMMAND Within the test-case loop, when chips % guests is 0, set share = chips / guests using integer division. RESULT share]
            [COMMAND Within the test-case loop, when chips % guests is 0, set possible to 0. RESULT possible]
            [COMMAND Within the test-case loop, when chips % guests is 0, search for a cut point with i from 1 to r - 1 inclusive and j from 1 to c - 1 inclusive while possible is 0; if B[i][j] equals share, set possible to 1, call fr(h - 1, i, j) and set possible to 0 if it returns false, call fc(v - 1, i, j) and set possible to 0 if it returns false, and if possible is still 1 print 'Case #' followed by tc and ': POSSIBLE' on one line. RESULT search_result]
            [COMMAND Within the test-case loop, when chips % guests is 0 and possible is 0 after the search, print 'Case #' followed by tc and ': IMPOSSIBLE' on one line. RESULT final_possible]
            [COMMAND Increment tc by 1 after finishing the current test case. RESULT tc]
            [COMMAND After the while loop exits, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
