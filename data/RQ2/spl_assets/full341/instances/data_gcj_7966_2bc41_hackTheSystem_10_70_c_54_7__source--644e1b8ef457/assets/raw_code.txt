1   import java.util.Scanner;
2   public class Solution {
3       static Long shootDamage;
4       static Long shootCounter;
5       static Long shield;
6       static String code;
7       static Long damage;
8       static Long movements;
9       static int lastIterableElement;
10      public static String hackTheSystem(String[] softwareToHack) {
11          shootDamage = 1L;
12          shootCounter = 0L;
13          shield = Long.parseLong(softwareToHack[0]);
14          code = softwareToHack[1];
15          damage = 0L;
16          movements = 0L;
17          int sizeScript = code.length() - 1;
18          lastIterableElement = sizeScript;
19          for (int i = 0; i < code.length(); i++) {
20              if (code.charAt(i) == 'S') {
21                  damage += shootDamage;
22                  shootCounter++;
23              }
24              if (code.charAt(i) == 'C') {
25                  shootDamage *= 2;
26              }
27          }
28          if (damage <= shield) {
29              return String.valueOf(movements);
30          }
31          if (shootCounter > shield) {
32              return "IMPOSSIBLE";
33          }
34          for (int i = sizeScript; i >= 0; i--) {
35              for (int j = sizeScript - 1; j >= 0; j--) {
36                  if(code.charAt(j) == 'S' && code.charAt(i) == 'S') {
37                      i--;
38                      continue;
39                  }
40                  if(code.charAt(i) == 'C') {
41                      shootDamage /= 2;
42                      i--;
43                      lastIterableElement--;
44                      continue;
45                  }
46                  if (code.charAt(j) == 'C' && code.charAt(i) == 'S') {
47                      movements++;
48                      damage -= (shootDamage / 2);
49                      char[] c = code.toCharArray();
50                      if (damage <= shield) {
51                          return String.valueOf(movements);
52                      }
53                      char temp = c[j];
54                      c[j] = c[i];
55                      c[i] = temp;
56                      code = String.valueOf(c);
57                      if (i == lastIterableElement) {
58                          shootDamage /= 2;
59                          if(i>0) {
60                              i--;
61                          }
62                      } else {
63                          i++;
64                          j += 2;
65                      }
66                  }
67              }
68          }
69          return String.valueOf(movements);
70      }
71      private static final Scanner scan = new Scanner(System.in);
72      public static void main(String[] args) {
73          int numberOfCases = Integer.parseInt(scan.nextLine());
74          String[][] cases = new String[numberOfCases][2];
75          for (int i = 0; i < numberOfCases; i++) {
76              String[] caseNumber = scan.nextLine().split(" ");
77              cases[i][0] = caseNumber[0];
78              cases[i][1] = caseNumber[1];
79          }
80          for (int i = 0; i < numberOfCases; i++) {
81              System.out.println("Case #" + i + ": " + hackTheSystem(cases[i]));
82          }
83      }
