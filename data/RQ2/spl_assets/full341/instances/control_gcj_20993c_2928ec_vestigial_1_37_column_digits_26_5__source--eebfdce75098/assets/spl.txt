[DEFINE_WORKER: "Computes and returns [main diagonal sum, number of rows with at least one duplicate, number of columns with at least one duplicate] for a square matrix." vestigial]
    [INPUTS]
        <REF> pinax </REF>: list of lists "A non-empty square matrix; pinax[0] provides the row length and must support len()."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list "A three-element list [trail, same_row_counter, sumC], where trail is the sum of the main diagonal, same_row_counter is the number of rows with at least one duplicate value, and sumC is the number of columns with at least one duplicate value."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set size to the length of pinax[0], the number of columns in the square matrix. RESULT size]
            [COMMAND Initialize result as an empty list. RESULT result]
            [COMMAND Initialize row_digits as an empty list. RESULT row_digits]
            [COMMAND Initialize column_digits as an empty list. RESULT column_digits]
            [COMMAND For i from 0 to size-1, append an empty list to column_digits, making column_digits a list of size empty sublists. RESULT column_digits]
            [COMMAND Set same_row_counter to 0. RESULT same_row_counter]
            [COMMAND Set same_column_counter to a list of size zeros. RESULT same_column_counter]
            [COMMAND Set trail to 0. RESULT trail]
            [COMMAND For each row from 0 to size-1, clear row_digits in-place with del row_digits[:] at the start of the iteration. RESULT row_digits]
            [COMMAND Within the outer loop over row, for each column from 0 to size-1, if row equals column then add pinax[row][row] to trail. RESULT trail]
            [COMMAND Within the outer loop over row, for each column from 0 to size-1, if len(row_digits) equals column, iterate i from 0 to len(row_digits) inclusive; if i equals len(row_digits) then append pinax[row][column] to row_digits, else if row_digits[i] equals pinax[row][column] then increment same_row_counter by 1 and break out of this inner scan. After a duplicate is found, row_digits stops growing, so len(row_digits) no longer equals subsequent column indexes and the row-duplicate check is skipped for the rest of that row. RESULT same_row_counter]
            [COMMAND Within the outer loop over row, for each column from 0 to size-1, iterate i from 0 to len(column_digits[column]) inclusive; if i equals len(column_digits[column]) then append pinax[row][column] to column_digits[column], else if column_digits[column][i] equals pinax[row][column] then set same_column_counter[column] to 1 and break out of this inner scan. RESULT same_column_counter]
            [COMMAND After the row and column loops complete, set sumC to 0. RESULT sumC]
            [COMMAND For i from 0 to size-1, add same_column_counter[i] to sumC. RESULT sumC]
            [COMMAND Append trail, same_row_counter, and sumC to result in that order. RESULT result]
            [COMMAND Return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
