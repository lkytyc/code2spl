[DEFINE_WORKER: "Simulates the Bathroom Stalls process for every test case by greedily selecting stalls with the largest minimum distance, tie-breaking by largest maximum distance and then leftmost index, and prints the final selected stall's max and min distances." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the number of test cases T from standard input using raw_input and int conversion. RESULT T]
            [COMMAND For each test case i from 1 to T inclusive, read the next input line, split it into integers, and assign N = int(tmpList[0]) and K = int(tmpList[1]). RESULT N,K]
            [COMMAND Within the test case loop, initialize stall_occupied as a list containing True, followed by N False values, followed by True, for a total length of N+2. RESULT stall_occupied]
            [COMMAND Within the test case loop, for each stall index j from 0 to N+1 inclusive, initialize Ls[j] and Rs[j]: if stall_occupied[j] is True set both to -1; otherwise set Ls[j] = j - 1 and Rs[j] = N - j. Then append Min_s[j] = min(Ls[j], Rs[j]) and Max_s[j] = max(Ls[j], Rs[j]). RESULT Ls,Rs,Min_s,Max_s]
            [COMMAND Within the test case loop, for each person k from 0 to K-1, compute maximal = max(Min_s). If Min_s.count(maximal) is exactly 1, select idx = Min_s.index(maximal); if more than one index has this maximal value, use the tie-breaking alternative flow. RESULT idx]
            [COMMAND Within the same for-k loop, after selecting idx, set y = max(Ls[idx], Rs[idx]) and z = min(Ls[idx], Rs[idx]). RESULT y,z]
            [COMMAND Within the same for-k loop, mark stall_occupied[idx] as True and set Ls[idx], Rs[idx], Min_s[idx], and Max_s[idx] to -1. RESULT stall_occupied,Ls,Rs,Min_s,Max_s]
            [COMMAND Within the same for-k loop, update empty stalls to the left of idx: start with leftIdx = idx - 1 and while leftIdx > 0, break if stall_occupied[leftIdx] is True; otherwise set Rs[leftIdx] = idx - leftIdx - 1, update Min_s[leftIdx] = min(Rs[leftIdx], Ls[leftIdx]) and Max_s[leftIdx] = max(Rs[leftIdx], Ls[leftIdx]), then decrement leftIdx. RESULT Ls,Rs,Min_s,Max_s]
            [COMMAND Within the same for-k loop, update empty stalls to the right of idx: start with rightIdx = idx + 1 and while rightIdx < N + 2, break if stall_occupied[rightIdx] is True; otherwise set Ls[rightIdx] = rightIdx - idx - 1, update Min_s[rightIdx] = min(Rs[rightIdx], Ls[rightIdx]) and Max_s[rightIdx] = max(Rs[rightIdx], Ls[rightIdx]), then increment rightIdx. RESULT Ls,Rs,Min_s,Max_s]
            [COMMAND After completing all K people in test case i, print "Case #{}: {} {}".format(i, y, z) and flush standard output. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When more than one stall index has Min_s equal to maximal, select the stall using Max_s as the secondary tie-breaker, preserving the code's leftmost preference if a further tie occurs.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a numpy array from Min_s and obtain idxs as all indices where Min_s == maximal, in ascending index order. RESULT idxs]
            [COMMAND Build max_of_idxs by appending Max_s[j] for each j in idxs. RESULT max_of_idxs]
            [COMMAND Compute max_max = max(max_of_idxs). RESULT max_max]
            [COMMAND If max_of_idxs.count(max_max) > 1, create a numpy array from max_of_idxs, find idxs1 where max_of_idxs == max_max, and set idx = idxs[idxs1[0]]; otherwise set idx = idxs[max_of_idxs.index(max_max)]. RESULT idx]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
