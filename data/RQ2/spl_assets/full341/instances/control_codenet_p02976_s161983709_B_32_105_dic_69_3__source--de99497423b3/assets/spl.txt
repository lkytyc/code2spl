[DEFINE_WORKER: "Reads an undirected graph, orients its edges using a BFS-based greedy parity-correction procedure, and prints the resulting directed edges with 1-based vertex numbers, or -1 when the orientation cannot be completed." B]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read two integers n and m by calling LI() and unpacking the first returned element as n and the second as m. RESULT n_m]
            [COMMAND Initialize v as an empty list. RESULT v]
            [COMMAND For each _ from 0 to n-1, append a new empty list to v so that v contains n adjacency lists. RESULT v]
            [COMMAND Initialize f as a list of n zeros. RESULT f]
            [COMMAND Initialize dic as a defaultdict with default factory lambda: 0, so missing keys are treated as 0. RESULT dic]
            [COMMAND For each i from 0 to m-1, read two integers a and b by calling LI(). RESULT a_b]
            [COMMAND Within the for loop over i, decrement a and b by 1 to convert them from 1-based to 0-based vertex indices. RESULT a_b]
            [COMMAND Within the for loop over i, append b to v[a] and append a to v[b] to add the undirected edge. RESULT v]
            [COMMAND Within the for loop over i, increment f[a] and f[b] by 1. RESULT f]
            [COMMAND Within the for loop over i, set dic[(a,b)] to 1 and dic[(b,a)] to 1. RESULT dic]
            [COMMAND Initialize ans as an empty list, d as a list of n zeros, q as a deque containing 0, and bfs as a list of n ones; then set bfs[0] to 0 to mark vertex 0 as visited. RESULT state]
            [COMMAND While q is non-empty, pop the leftmost vertex from q into x and initialize s as an empty list. RESULT x_s]
            [COMMAND In the while loop, for each neighbor y of the current vertex x, if dic[(x,y)] is truthy append y to s; if bfs[y] is truthy append y to q and set bfs[y] to 0. RESULT state]
            [COMMAND In the while loop, if s is non-empty and len(s) is odd and d[x] is truthy, for each y in s append (x,y) to ans, set dic[(x,y)] and dic[(y,x)] to 0, and set d[x] to 0. RESULT ans]
            [COMMAND In the while loop, if s is non-empty and len(s) is odd and d[x] is false, for each y in s except the last element append (x,y) to ans and set dic[(x,y)] and dic[(y,x)] to 0; then set d[x] to 0, set y to the last element of s, toggle d[y] by XOR with 1, append (y,x) to ans, and set dic[(x,y)] and dic[(y,x)] to 0. RESULT ans]
            [COMMAND In the while loop, if s is non-empty and len(s) is even and d[x] is false, for each y in s append (x,y) to ans, set dic[(x,y)] and dic[(y,x)] to 0, and set d[x] to 0. RESULT ans]
            [COMMAND In the while loop, if s is non-empty and len(s) is even and d[x] is truthy, for each y in s except the last element append (x,y) to ans and set dic[(x,y)] and dic[(y,x)] to 0; then set d[x] to 0, set y to the last element of s, toggle d[y] by XOR with 1, append (y,x) to ans, and set dic[(x,y)] and dic[(y,x)] to 0. RESULT ans]
            [COMMAND After the while loop, if sum(d) is nonzero, print -1. RESULT stdout]
            [COMMAND After the while loop, if sum(d) is zero, for each pair (i,j) in ans in original order print i+1 and j+1 as space-separated 1-based vertex numbers. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]
