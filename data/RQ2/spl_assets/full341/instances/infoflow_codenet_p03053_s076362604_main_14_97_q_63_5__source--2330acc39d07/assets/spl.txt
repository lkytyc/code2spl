[DEFINE_WORKER: "Reads an H by W character grid, computes for each cell the Manhattan distance to the nearest '#' using four directional dynamic-programming passes, and prints the maximum such distance." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare the fixed-size character grid l[1001][1001], integer arrays p[1001][1001], q[1002][1001], r[1001][1002], s[1002][1002], and n[1001][1001], plus scalar variables H, W, c, d, i, j, m, and ans. RESULT declared_memory]
            [COMMAND Read two integers from standard input with scanf("%d", &H) and scanf("%d", &W), storing the grid height in H and width in W. RESULT H, W]
            [COMMAND For each row i from 1 through H, first consume the preceding newline or separator by reading one character into d with scanf("%c", &d); then for each column j from 1 through W, read one character into c with scanf("%c", &c) and store it in l[i][j] = c. RESULT l]
            [COMMAND Set m = H + W to use as the sentinel distance value for boundary initialization. RESULT m]
            [COMMAND Initialize the sentinel boundaries using m: for each i from 1 through H set p[i][0] = m, q[i][W+1] = m, r[i][0] = m, and s[i][W+1] = m; for each j from 1 through W set p[0][j] = m, q[0][j] = m, r[H+1][j] = m, and s[H+1][j] = m. RESULT p, q, r, s]
            [COMMAND Compute the top-left directional distance array p: for i from 1 through H and j from 1 through W, if l[i][j] == '#' set p[i][j] = 0; otherwise set p[i][j] = min(p[i-1][j], p[i][j-1]) + 1. RESULT p]
            [COMMAND Compute the top-right directional distance array q: for i from 1 through H and j from W down through 1, if l[i][j] == '#' set q[i][j] = 0; otherwise set q[i][j] = min(q[i-1][j], q[i][j+1]) + 1. RESULT q]
            [COMMAND Compute the bottom-left directional distance array r: for i from H down through 1 and j from 1 through W, if l[i][j] == '#' set r[i][j] = 0; otherwise set r[i][j] = min(r[i+1][j], r[i][j-1]) + 1. RESULT r]
            [COMMAND Compute the bottom-right directional distance array s: for i from H down through 1 and j from W down through 1, if l[i][j] == '#' set s[i][j] = 0; otherwise set s[i][j] = min(s[i+1][j], s[i][j+1]) + 1. RESULT s]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For each i from 1 through H and each j from 1 through W, set n[i][j] = min(p[i][j], q[i][j]); then set n[i][j] = min(n[i][j], r[i][j]); then set n[i][j] = min(n[i][j], s[i][j]); then update ans = max(ans, n[i][j]). RESULT ans]
            [COMMAND Print ans to standard output with printf("%d", ans), using no trailing newline. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
