[DEFINE_WORKER: "Reads a tree-like structure and flow capacities, then iteratively augments flow until the accumulated flow exceeds total capacity, printing the final augmented amount." calc]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare M, initialize S to 0 and R to 0. RESULT M, S, R]
            [COMMAND Read an integer M from standard input into M. RESULT M]
            [COMMAND Allocate an integer array r1 of length M. RESULT r1]
            [COMMAND Allocate an integer array r2 of length M. RESULT r2]
            [COMMAND For i from 0 to M-1 (i < M), read r1[i] and r2[i] from standard input, then decrement both r1[i] and r2[i] by 1. RESULT r1, r2]
            [COMMAND Allocate a long long array G of length M. RESULT G]
            [COMMAND Allocate a long long array f of length M. RESULT f]
            [COMMAND For i from 0 to M-1 (i < M), read G[i] from standard input, set f[i] to 0, and add G[i] to S. RESULT G, f, S]
            [COMMAND Set f[0] to 1, declare g without an initializer, and set s to 1. RESULT f, s]
            [COMMAND At the start of each while iteration while s <= S, set n to INT_MAX. RESULT n]
            [COMMAND Within the while loop, for j from 0 to M-1 (j < M), if f[j] > 0, set n to the minimum of n and G[j] / f[j] using integer division, then set g to j. RESULT n, g]
            [COMMAND Within the while loop, if n > 0, add n to R. RESULT R]
            [COMMAND Within the while loop, if n > 0, for j from 0 to M-1 (j < M), if f[j] > 0, subtract f[j] * n from both G[j] and S. RESULT G, S]
            [COMMAND Within the while loop, add G[g] to G[r1[g]] and add G[g] to G[r2[g]]. RESULT G]
            [COMMAND Within the while loop, set a to f[g]. RESULT a]
            [COMMAND Within the while loop, add a to s. RESULT s]
            [COMMAND Within the while loop, if s > S, free f, G, r2, and r1, print R followed by a newline, and return immediately. RESULT R]
            [COMMAND Within the while loop, if s <= S, set f[g] to 0, add a to f[r1[g]], and add a to f[r2[g]]. RESULT f]
            [COMMAND After the while loop exits, free f, G, r2, and r1, then print R followed by a newline. RESULT R]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]

[DEFINE_WORKER: "Reads a test-case count T from standard input, prints a case label for each t from 1 through T, delegates the per-case work to calc(), and returns EXIT_SUCCESS." main]
    [INPUTS]
        <REF> stdin </REF>: stream "Standard input stream; the first token is read as a decimal integer into T."
    [END_INPUTS]

    [OUTPUTS]
        <REF> exit_status </REF>: int "The value returned by main; always EXIT_SUCCESS (0) when the documented flow completes."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare an uninitialized integer variable T. RESULT T]
            [COMMAND Read a decimal integer from standard input into T using scanf('%d', &T); do not check the return value. If the input is missing or cannot be parsed, T is left unchanged or undefined. RESULT T]
            [COMMAND Declare loop variable t initialized to 1; execute the loop body for every t from 1 through T inclusive, incrementing t by 1 after each iteration. RESULT loop_control]
            [COMMAND Within the for loop over t from 1 to T inclusive, write the formatted string 'Case #%d: ' to standard output, substituting the current value of t. RESULT void]
            [COMMAND Within the for loop over t from 1 to T inclusive, call calc() with no arguments. RESULT void]
            [COMMAND After the for loop completes, return the constant EXIT_SUCCESS (0) from main. RESULT exit_status]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
