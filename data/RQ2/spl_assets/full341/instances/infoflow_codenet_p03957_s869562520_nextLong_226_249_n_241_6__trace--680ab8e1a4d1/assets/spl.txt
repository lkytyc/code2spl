[DEFINE_WORKER: "Parses and returns the next long integer from the underlying input stream, applying an optional leading minus sign and terminating at end-of-input or a non-printable character." nextLong]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: long "The signed long value parsed from the input, returned when end-of-input or a non-printable delimiter is reached."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Check that an input element is available by calling hasNext(); if it returns false, throw NoSuchElementException. RESULT has_next]
            [COMMAND Set the long accumulator n to 0. RESULT n]
            [COMMAND Set the boolean sign flag minus to false. RESULT minus]
            [COMMAND Read the first input byte b by calling readByte(). RESULT b]
            [COMMAND If b equals the minus sign '-', set minus to true. RESULT minus]
            [COMMAND If b equals the minus sign '-', read the next byte into b by calling readByte() after setting the sign. RESULT b]
            [COMMAND If b is less than '0' or greater than '9', throw NumberFormatException. RESULT b]
            [COMMAND Enter an infinite loop. At the top of the loop, if b is in the inclusive ASCII digit range '0'..'9', update n = n * 10 + (b - '0'). RESULT n]
            [COMMAND Inside the infinite loop, if b is -1 or !isPrintableChar(b), return minus ? -n : n as the parsed long value. RESULT return_value]
            [COMMAND Inside the infinite loop, after a digit branch, read the next byte into b by calling readByte() and continue the loop. RESULT b]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: hasNext() returns false]
        [LOG "Attempted to read the next long when no input element is available."]
        [THROW NoSuchElementException "Indicates that the input has no next element to read."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: After the optional leading minus sign, b is less than '0' or greater than '9'.]
        [LOG "The first character of the token is not a decimal digit."]
        [THROW NumberFormatException "Indicates that the input is not a valid long number."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: Inside the infinite loop, b is a printable character that is not an ASCII digit and is not the end/terminator condition.]
        [LOG "Encountered an invalid printable character while parsing the long number."]
        [THROW NumberFormatException "Indicates that the numeric token contains an invalid character."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
