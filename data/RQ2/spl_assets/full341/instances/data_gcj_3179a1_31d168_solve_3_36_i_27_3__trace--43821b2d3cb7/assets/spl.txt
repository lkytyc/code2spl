[DEFINE_WORKER: "Processes test case T by reading U and 10000 string pairs, recording observed first/last letters and the minimum leading digit for equal-length pairs, then prints Case #T with the reconstructed 10-character output." solve]
    [INPUTS]
        <REF> T </REF>: int "Current test case number, used only in the final printf output."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the integer array MC of size 26 so that MC[0] through MC[25] are all 10. RESULT MC]
            [COMMAND Initialize the integer array O of size 26 so that O[0] through O[25] are all 0. RESULT O]
            [COMMAND Declare the character array out of size 10. RESULT out]
            [COMMAND Read an integer from standard input into U using scanf("%d", &U); then set i to 0, declare character arrays R and M of length U, and declare integers lr and lm. RESULT U]
            [COMMAND While i is strictly less than 10000, read two whitespace-separated strings, storing the first in M and the second in R, and set lr = strlen(R) and lm = strlen(M). RESULT M_R_lr_lm]
            [COMMAND In each while-loop iteration, set O at index R[lr - 1] - 'A' to 1, using the last character of R. RESULT O]
            [COMMAND In each while-loop iteration, if lr == lm, set O at index R[0] - 'A' to 1, using the first character of R. RESULT O]
            [COMMAND Inside the lr == lm branch, if MC[R[0] - 'A'] is strictly greater than M[0] - '0', assign M[0] - '0' to MC[R[0] - 'A'], keeping the minimum seen digit for that first character. RESULT MC]
            [COMMAND Increment i by 1 at the end of each while-loop iteration; because the loop condition is i < 10000, it executes with i = 0 through 9999 and exits when i reaches 10000. RESULT i]
            [COMMAND After the while loop, for i from 0 through 25 inclusive, if O[i] is nonzero, set out[MC[i] % 10] to the character 'A' + i. RESULT out]
            [COMMAND Print the final line using printf("Case #%d: %s\n", T, out), then return. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
