[DEFINE_WORKER: "Reads multiple directed graphs from input and, for each query, performs a priority-queue search with a TTL node limit to find the minimum hop count from the start node to the destination node, printing NA when no valid path exists." AOJ0144]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND While sc.hasNext(), begin reading the next test case and read the number of vertices N. RESULT N]
            [COMMAND Allocate a boolean adjacency matrix R with dimensions N+1 by N+1, initially all false. RESULT R]
            [COMMAND For each source index i from 1 to N inclusive, read source node r and outdegree k; for each neighbor index j from 0 to k-1, read a destination node and set R[r][destination] to true. RESULT R]
            [COMMAND Read the number of queries P. RESULT P]
            [COMMAND For each query index i from 0 to P-1, read start node s, destination node d, and initial TTL v; initialize ans to -1. RESULT query]
            [COMMAND Within the current query, create an empty priority queue open of C0144 states and add the initial state with node s, TTL v, and cost 1. RESULT open]
            [COMMAND Within the current query, create a C0144 array close of length N+1 filled with null entries. RESULT close]
            [COMMAND Within the current query, while open is not empty, poll the next state now; decrement its TTL, and if --now.ttl equals 0, skip expanding this state and continue to the next state. RESULT now]
            [COMMAND Within the current state, for each neighbor j from 1 to N inclusive, check if R[now.now][j] is true; if close[j] is null, or close[j] is non-null and either now.c+1 is less than close[j].c or now.ttl is greater than close[j].ttl, treat j as a qualifying neighbor. RESULT qualifying_neighbor]
            [COMMAND For each qualifying neighbor j, if j equals the destination d, update ans to min(ans, now.c+1), preserving -1 when ans was negative, and do not enqueue j; otherwise add a new C0144 state with node j, TTL now.ttl, and cost now.c+1 to open and store that same state in close[j]. RESULT ans_open_close]
            [COMMAND After the priority-queue search for the current query is exhausted without any early break, write out.println((ans < 0 ? "NA" : ans)). RESULT output_line]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
