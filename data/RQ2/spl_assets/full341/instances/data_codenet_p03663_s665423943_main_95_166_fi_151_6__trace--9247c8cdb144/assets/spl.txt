[DEFINE_WORKER: "Interactively determines a digit-range index from responses to power-of-ten queries, binary-searches candidate values scaled by ten using yes/no responses, and prints the final fi." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare long long variables i, j, w, fl=0, dg, st, fi, te; initialize bool mem[16][2] to all false; declare char buffers s[8] and t[8]. RESULT state]
            [COMMAND For i from 9 down to 0 inclusive, compute w=llpow(10,i), print a query line with w, flush stdout, read string s; if s[0]=='Y' set mem[i][0]=1, otherwise set fl|=2. Then compute w=llpow(10,i+1)-1, print a query line with w, flush stdout, read string t; if t[0]=='Y' set mem[i][1]=1, otherwise set fl|=1. RESULT probe_state]
            [COMMAND Execute in source order the conditional dg-determination blocks: if fl==1 apply alternative flow 'fl == 1'; if fl==2 apply alternative flow 'fl == 2'; if fl==3 apply alternative flow 'fl == 3'. If none applies, dg is not assigned. RESULT dg]
            [COMMAND Set st=1 and fi=1. RESULT bounds_initialized]
            [COMMAND For i from 1 to dg inclusive, multiply st by 10. RESULT st_lower_bound]
            [COMMAND For i from 1 to dg+1 inclusive, multiply fi by 10; after the loop, subtract 1 from fi. RESULT fi_upper_bound]
            [COMMAND While st<=fi, compute te=(st+fi)/2, print a query line with te*10, flush stdout, read string s; if s[0]=='N' set fi=te-1, otherwise set st=te+1. RESULT binary_search_state]
            [COMMAND Print the final answer line with fi, flush stdout, and return 0. RESULT final_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: fl == 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Scan i from 9 down to 0 inclusive. If mem[i][1]==0, set dg=i and break. If i==0 is reached without finding a zero row, print the early answer 1, flush stdout, and return 0. RESULT dg_or_early_return]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: fl == 2]
        [SEQUENTIAL_BLOCK]
            [COMMAND Scan i from 0 to 9 inclusive. If mem[i][0]==0, set dg=i and break; if no zero row exists, leave dg unchanged. RESULT dg_or_unchanged]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: fl == 3]
        [SEQUENTIAL_BLOCK]
            [COMMAND Scan i from 0 to 9 inclusive. If (mem[i][0] & mem[i][1]) == 1, set dg=i and break; if no such row exists, leave dg unchanged. RESULT dg_or_unchanged]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
