[DEFINE_WORKER: "Reads an undirected graph with n vertices and m edges, uses a queue-based traversal to greedily orient and remove available edges while maintaining per-vertex parity toggles, and prints -1 if any parity remains active, otherwise prints the selected oriented edges in 1-based indexing." B]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read n and m by unpacking the result of LI(). RESULT n_m]
            [COMMAND Initialize v as an empty list. RESULT v]
            [COMMAND For each index from 0 to n-1 inclusive, append a new empty list to v. RESULT v]
            [COMMAND Initialize f as a list of n zeros. RESULT f]
            [COMMAND Initialize dic as a defaultdict whose default factory returns 0. RESULT dic]
            [COMMAND Repeat m times: read a and b via LI(), decrement both a and b by 1, append b to v[a], append a to v[b], increment f[a] and f[b], and set dic[(a,b)] and dic[(b,a)] to 1. RESULT v_f_dic]
            [COMMAND Initialize ans as an empty list, d as a list of n zeros, q as a deque containing 0, and bfs as a list of n ones with bfs[0] set to 0. RESULT ans_d_q_bfs]
            [COMMAND While q is nonempty, pop the left element of q as x and initialize s as an empty list. RESULT x_s]
            [COMMAND Within the same while iteration, for each neighbor y in v[x], if dic[(x,y)] is truthy append y to s, and if bfs[y] is truthy append y to q and set bfs[y] to 0. RESULT s_q_bfs]
            [COMMAND Within the same while iteration, if s is nonempty, len(s) is odd, and d[x] is truthy, append (x,y) to ans for every y in s, set dic[(x,y)] and dic[(y,x)] to 0 for every such y, and set d[x] to 0. RESULT ans_dic_d]
            [COMMAND Within the same while iteration, if s is nonempty, len(s) is odd, and d[x] is falsy, append (x,y) to ans for every y in s except the last element, clear dic for those edges, set d[x] to 0, let y be the last element of s, toggle d[y], append (y,x) to ans, and clear dic[(x,y)] and dic[(y,x)]. RESULT ans_dic_d]
            [COMMAND Within the same while iteration, if s is nonempty, len(s) is even, and d[x] is falsy, append (x,y) to ans for every y in s, set dic[(x,y)] and dic[(y,x)] to 0 for every such y, and set d[x] to 0. RESULT ans_dic_d]
            [COMMAND Within the same while iteration, if s is nonempty, len(s) is even, and d[x] is truthy, append (x,y) to ans for every y in s except the last element, clear dic for those edges, set d[x] to 0, let y be the last element of s, toggle d[y], append (y,x) to ans, and clear dic[(x,y)] and dic[(y,x)]. RESULT ans_dic_d]
            [COMMAND After the while loop, if sum(d) is truthy, print -1. RESULT printed_minus_one]
            [COMMAND After the while loop, otherwise, for each pair (i,j) in ans, print i+1 and j+1 separated by a space. RESULT printed_edges]
            [COMMAND Return None. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
