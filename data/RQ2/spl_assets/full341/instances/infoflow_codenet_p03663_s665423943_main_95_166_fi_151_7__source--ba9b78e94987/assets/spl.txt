[DEFINE_WORKER: "Determines the final answer for an interactive guessing problem by querying powers of ten to choose a digit-scale bound, then performing an inclusive ten-scaled binary search and printing the final upper bound fi." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> exit_code </REF>: int "Returns 0 on normal termination, including the early return after printing '! 1'."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare long long variables i, j, w, fl, dg, st, fi, and te; initialize fl to 0 and the two-dimensional bool array mem[16][2] to all false; declare char buffers s[8] and t[8] without initialization. RESULT fl, mem]
            [COMMAND For each i from 9 down to 0 inclusive, set w = llpow(10, i), print '? ' followed by w, flush standard output, and read the response into s. If s[0] equals 'Y', set mem[i][0] to 1; otherwise set fl to fl | 2. RESULT mem, fl]
            [COMMAND In the same iteration of the downward i loop, set w = llpow(10, i + 1) - 1, print '? ' followed by w, flush standard output, and read the response into t. If t[0] equals 'Y', set mem[i][1] to 1; otherwise set fl to fl | 1. RESULT mem, fl]
            [COMMAND If fl equals 1, loop i from 9 down to 0 inclusive. If mem[i][1] equals 0, set dg = i and break; if i equals 0 and the current mem[0][1] is not 0, print '! 1', flush standard output, and return 0 immediately. RESULT dg]
            [COMMAND If fl equals 2, loop i from 0 to 9 inclusive until the first mem[i][0] equal to 0 is found; set dg = i and break. RESULT dg]
            [COMMAND If fl equals 3, loop i from 0 to 9 inclusive until the first i where the bitwise AND of mem[i][0] and mem[i][1] equals 1 is found; set dg = i and break. RESULT dg]
            [COMMAND Set st = 1 and fi = 1. RESULT st, fi]
            [COMMAND For i from 1 to dg inclusive, multiply st by 10. RESULT st]
            [COMMAND For i from 1 to dg + 1 inclusive, multiply fi by 10; after the loop completes, subtract 1 from fi. RESULT fi]
            [COMMAND While st <= fi, compute te = (st + fi) / 2, print '? ' followed by te * 10, flush standard output, and read the response into s. If s[0] equals 'N', set fi = te - 1; otherwise set st = te + 1. RESULT te, st, fi]
            [COMMAND Print '! ' followed by fi, flush standard output, and return 0. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
