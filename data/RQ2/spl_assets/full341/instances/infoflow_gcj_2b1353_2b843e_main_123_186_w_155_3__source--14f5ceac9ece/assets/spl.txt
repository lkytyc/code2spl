[DEFINE_WORKER: "Reads a number of test cases and, for each input n, prints a 'Case #%d:' header followed by a deterministic sequence of (row, column) coordinates representing a binary-based decomposition of n, with traversal direction toggling at selected rows and extra edge cells appended until the printed count reaches the original n." main]
    [INPUTS]
        <REF> stdin </REF>: standard input "An integer t, then t lines each containing an integer n."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: text "For each case, a header line 'Case #<cs>:' followed by coordinate pairs 'row col', one per line."
        <REF> exit_code </REF>: int "Always returns 0."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize local variable r to 0; r is never used later. RESULT r]
            [COMMAND Read integer t from standard input using scanf; t is the number of test cases. RESULT t]
            [COMMAND For cs from 1 to t inclusive, process each test case. RESULT cs]
            [COMMAND Within the cs loop, read integer n from standard input as the current test case value. RESULT n]
            [COMMAND Within the cs loop, save the original target sn = n before modifying n. RESULT sn]
            [COMMAND Within the cs loop, print the case header using printf with format 'Case #%d:\n' and the current cs. RESULT stdout]
            [COMMAND Within the cs loop, set w = 0. RESULT w]
            [COMMAND Within the cs loop, find the smallest l such that the cumulative sum of powers of two reaches n: for i from 1 to 49 inclusive, add (1 << (i-1)) to w and if w >= n then set l = i and break. RESULT l]
            [COMMAND Within the cs loop, for i from l down to 1 inclusive, decompose n: if n >= (1 << (i-1)) + (i-1), set fl[i] = 1 and subtract (1 << (i-1)) from n; otherwise set fl[i] = 0 and decrement n by 1. RESULT fl]
            [COMMAND Within the cs loop, reset a = 0 and w = 0 before the output reconstruction phase. RESULT a]
            [COMMAND Within the cs loop, for i from 1 to l inclusive, reconstruct and print the path: if fl[i] == 1, add (1 << (i-1)) to w, print row i with columns 1 through i when a == 0 or columns i down to 1 when a != 0, then toggle a = 1 - a; if fl[i] == 0, add 1 to w and print only (i, 1) when a == 0 or (i, i) when a != 0. RESULT w]
            [COMMAND Within the cs loop, while w is not equal to sn, increment l by 1, print (l, 1) when a == 0 or (l, l) when a != 0, and increment w by 1. RESULT stdout]
            [COMMAND After the cs loop finishes, return 0 from main. RESULT exit_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
