[DEFINE_WORKER: "Finds the subset XOR ans of A that maximizes ans + (ans ^ xor_of_all_A) using a 62-bit downward greedy feasibility search." solve]
    [INPUTS]
        <REF> N </REF>: int "Number of elements in the array A."
        <REF> A </REF>: list[int] "List of N integers whose subset-XOR pairs are evaluated."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "The maximum evaluated value ans + (ans ^ allxor), where allxor is the XOR of all elements of A."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Compute allxor by XORing every element of A together. RESULT allxor]
            [COMMAND Initialize B as a list of N zeros, set ans to 0, and set bcount to 0. RESULT B, ans, bcount]
            [COMMAND For bit from 61 down to 0 inclusive, set mask = 1 << bit. If allxor & mask is true, take the alternative flow for a set bit in allxor; otherwise continue with the remaining bit-processing steps. RESULT bit, mask]
            [COMMAND In the non-continue path, create an empty set bset. RESULT bset]
            [COMMAND Within the same bit iteration, for each i from 0 to N-1, compute b = (B[i] << 1) | ((A[i] & mask) >> bit), assign B[i] = b, and add b to bset. RESULT B, bset]
            [COMMAND Increment bcount by 1 and set full_bit = (1 << bcount) - 1. RESULT bcount, full_bit]
            [COMMAND Convert bset to list blist and split it into blist1 = blist[:len(blist)//2] and blist2 = blist[len(blist)//2:]. RESULT blist1, blist2]
            [COMMAND Build r1 as the XOR closure of blist1: initialize r1 to a set containing only 0, then for each b in blist1, for each c in a snapshot of r1, add b ^ c to r1. RESULT r1]
            [COMMAND Build r2 as the XOR closure of blist2 in the same way: initialize r2 to a set containing only 0, then expand it by XORing each element of blist2 with each snapshot element of r2. RESULT r2]
            [COMMAND Set best = full_bit - 1. Search r1 for a b1 such that (b1 ^ full_bit) is present in r2; if found, set best = full_bit and break out of the search loop. RESULT best]
            [COMMAND Set bestb = best & 1. Append bestb to ans by shifting ans left by 1 and OR-ing bestb into the lowest bit. RESULT ans, bestb]
            [COMMAND If bestb == 0, take the alternative flow for the failed feasibility rollback; otherwise keep the updated B and bcount and proceed to the next bit. RESULT B, bcount]
            [COMMAND After completing all bit iterations from 61 down to 0, return ans + (ans ^ allxor). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: At the beginning of a bit iteration, allxor has the current bit set (allxor & mask is true).]
        [SEQUENTIAL_BLOCK]
            [COMMAND Shift ans left by 1 and set its lowest bit to 1. RESULT ans]
            [COMMAND Continue to the next bit iteration, skipping all remaining processing for this bit. RESULT next_bit]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After appending bestb, bestb equals 0, meaning no valid split-half XOR combination produced full_bit.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each i from 0 to N-1, shift B[i] right by 1 to undo the earlier left shift. RESULT B]
            [COMMAND Decrement bcount by 1. RESULT bcount]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
