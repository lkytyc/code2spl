[DEFINE_WORKER: "Reads an integer n and long values from stdin, validates that the first long value is zero, then maintains a length-3 counter array to accumulate a modular product of per-input match counts and prints (ans*3) mod 1000000007." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; they are not used by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set mod to the long constant 1000000007 by computing (long) Math.pow(10, 9) + 7. RESULT mod]
            [COMMAND Create FastScanner sc reading from System.in. RESULT scanner]
            [COMMAND Read the int n from sc. RESULT n]
            [COMMAND Initialize long array arr of length 3, with every element defaulting to 0. RESULT arr]
            [COMMAND Read the first long value ff from sc. RESULT first_value]
            [COMMAND Evaluate whether first_value != 0; if it is true, use the alternative flow that prints 0 and returns, otherwise continue with the normal path. RESULT guard_evaluation]
            [COMMAND Initialize ans to 1. RESULT ans]
            [COMMAND Set arr[0] to 1, so arr becomes [1, 0, 0]. RESULT arr]
            [COMMAND Begin the outer for loop with i = 0; continue while i < n - 1; increment i by 1 after each iteration. The following three steps are executed inside this loop. RESULT i]
            [COMMAND Inside the outer loop iteration, reset fi to -1 and cnt to 0, then read the next long value a from sc. RESULT current_a]
            [COMMAND Inside the outer loop iteration, run the inner for loop with j from 0 to 2 inclusive: if arr[j] == a, increment cnt; and if this is the first match (fi == -1), increment arr[j] and set fi = j. RESULT match_count]
            [COMMAND Inside the outer loop iteration, after the inner loop finishes, update ans = (ans * cnt) % mod. RESULT ans]
            [COMMAND After the outer loop completes, print the value (ans * 3) % mod to standard output. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: The first value ff read from sc is not equal to 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print 0 to standard output. RESULT printed_output]
            [COMMAND Return immediately from the main method, skipping the remaining initialization, loop, and final print. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
