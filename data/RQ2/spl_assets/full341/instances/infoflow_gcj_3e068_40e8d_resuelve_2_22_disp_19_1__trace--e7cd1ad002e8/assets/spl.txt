[DEFINE_WORKER: "Processes N lines of flavor requests, accumulating per-flavor counts and repeatedly outputting the least-used available flavor ID, or -1 when no available flavor exists, while marking each selected flavor as unavailable." resuelve]
    [INPUTS]
        <REF> N </REF>: int "Number of iterations to process; also serves as the exclusive upper bound for valid flavor IDs."
        <REF> disp </REF>: list of bool "Availability flags indexed by flavor ID; True means the flavor has not been selected yet."
        <REF> cuentas </REF>: list of int "Counters indexed by flavor ID, incremented for each request that mentions a valid flavor."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize pasados to 0. RESULT pasados]
            [COMMAND While pasados is less than N, at the top of the loop body increment pasados by 1 and reset devolver to "-1". RESULT devolver]
            [COMMAND Within the while loop, read one line from standard input, strip it, split it on single spaces, convert each token to an integer, and store the resulting list in flavours. RESULT flavours]
            [COMMAND Within the while loop, set D to the first element of flavours. RESULT D]
            [COMMAND Within the while loop, if D is not equal to 0, initialize posibles to an empty list. RESULT posibles]
            [COMMAND Within the while loop and the D != 0 branch, iterate over each element el in flavours[1:]; if el is less than N, increment cuentas[el] by 1 and, if disp[el] is True, append the tuple (cuentas[el], el) to posibles. RESULT posibles]
            [COMMAND Within the while loop and the D != 0 branch, after the iteration over flavours[1:], if posibles is not empty, unpack the minimum tuple of posibles into maxi and elmax, set devolver to the string form of elmax, and set disp[elmax] to False. RESULT devolver]
            [COMMAND Within the while loop and the D != 0 branch, if posibles is empty, leave devolver as "-1". RESULT devolver]
            [COMMAND Within the while loop, after the D != 0 branch, write devolver followed by a newline to standard output. RESULT _]
            [COMMAND Within the while loop, after writing the output, flush standard output. RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
