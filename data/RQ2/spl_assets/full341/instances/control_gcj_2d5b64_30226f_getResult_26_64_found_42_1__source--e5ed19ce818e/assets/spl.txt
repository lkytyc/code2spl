[DEFINE_WORKER: "Constructs a list containing the sequence 1..r repeated s times, then performs a backward scan and block-rotation process that records each match as an 'index size' string in the returned list." getResult]
    [INPUTS]
        <REF> input </REF>: String "External string containing two space-separated integers; the first is r and the second is s."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: List<String> "List of formatted strings 'index size' describing each matched block move."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Parse the external string input by splitting on a single space; parse the first token as r and the second token as s. RESULT r, s]
            [COMMAND Initialize result as an empty List<String> and v as an empty List<Integer>. RESULT result, v]
            [COMMAND For each ss from 0 inclusive while ss < s, for each rr from 0 inclusive while rr < r, append the value rr + 1 to v. RESULT v]
            [COMMAND For each rr from 0 inclusive while rr < r - 1, set current = r - rr, count = s - 1, idx = current * s - 2, and start = idx. RESULT current, count, idx, start]
            [COMMAND Within the outer rr loop, while count > 0: set found to false; while found is false, compare v.get(idx) to current; after each comparison, decrement idx by 1. RESULT found, idx]
            [COMMAND Within that inner while loop, when v.get(idx) equals current: decrement count, set found to true, compute sz = start - idx, add String.format("%d %d", idx + 1, sz) to result, build t by repeatedly calling v.remove(idx + 1) and adding the result to t while sz > 0, append all remaining v elements to t, assign v = t, set idx = start, and decrement start. RESULT count, found, sz, result, t, v, idx, start]
            [COMMAND Continue looping until the inner while condition found is true and the outer while count > 0 is false; after finishing all rr values, return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: input is null]
        [LOG "Null input while parsing r and s"]
        [THROW NullPointerException "input.split cannot be invoked on a null string."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: input.split(" ") contains fewer than two tokens]
        [LOG "Malformed input: expected two space-separated integers"]
        [THROW ArrayIndexOutOfBoundsException "Accessing split token 0 or 1 fails when fewer than two tokens exist."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: Either input token cannot be parsed as an integer]
        [LOG "Invalid integer token while parsing r or s"]
        [THROW NumberFormatException "Integer.parseInt fails because the token is not a valid integer."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: During the backward search, idx becomes negative before finding current, or idx + 1 is out of range during removal]
        [LOG "Index out of bounds while accessing or mutating list v"]
        [THROW IndexOutOfBoundsException "v.get(idx) or v.remove(idx + 1) fails because the required index is outside the current list bounds."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
