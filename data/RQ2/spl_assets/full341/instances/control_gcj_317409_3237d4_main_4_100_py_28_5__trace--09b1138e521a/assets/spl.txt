[DEFINE_WORKER: "Reads T test cases from stdin, simulates a moving person's tour on a 2001x2001 grid, BFS-finds the nearest tour cell to the given starting offset, and computes the earliest meeting time or outputs IMPOSSIBLE for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize int[] dx with values {0, 1, 0, -1} and int[] dy with values {1, 0, -1, 0}; create Scanner sc reading from new BufferedReader(new InputStreamReader(System.in)); create StringBuilder ans; read int T = sc.nextInt(). RESULT T]
            [COMMAND For each test case t from 1 to T inclusive: read int X = sc.nextInt(), int Y = sc.nextInt(), String M = sc.next(); allocate int[][] g = new int[2001][2001]; set px = 1000 and py = 1000; create List<List<Integer>> tour; create List<Integer> te containing px and py, add it to tour, and set g[py][px] = 1. RESULT X, Y, M, g, tour]
            [COMMAND Within the current test case t, for each character index i from 0 to M.length() - 1: get char d = M.charAt(i); if d == 'N', decrement py; if d == 'S', increment py; if d == 'E', increment px; if d == 'W', decrement px; set g[py][px] = 1; create a new List<Integer> te containing px and py and add it to tour. RESULT tour, g]
            [COMMAND Within the current test case t, allocate boolean[][] visited = new boolean[2001][2001]; set myx = 1000 - X and myy = 1000 + Y; create LinkedList<Pair> q and offer new Pair(myx, myy, 0); initialize tmp = -1, nearX = -1, and nearY = -1. RESULT visited, q, tmp, nearX, nearY]
            [COMMAND Within the current test case t, run BFS while q is not empty: poll Pair now; if g[now.y][now.x] == 1, set tmp = now.dist, nearX = now.x, nearY = now.y and break; mark visited[now.y][now.x] = true; for v from 0 to 3, compute nx = now.x + dx[v] and ny = now.y + dy[v]; skip this neighbor if nx < 0 or visited[0].length <= nx or ny < 0 or visited.length <= ny or visited[ny][nx] is true; otherwise add new Pair(nx, ny, now.dist + 1) to q. RESULT tmp, nearX, nearY]
            [COMMAND Within the current test case t, append the text Case # concatenated with t and followed by `: ` to ans. RESULT ans]
            [COMMAND Within the current test case t, after appending the case prefix: if tmp == -1, append `IMPOSSIBLE\n` to ans and continue to the next t (see first alternative flow); otherwise continue to locate the matching tour index. RESULT ans]
            [COMMAND Within the current test case t, because tmp != -1, find ti as the first index i from 0 to tour.size() - 1 such that tour[i].get(0) == nearX and tour[i].get(1) == nearY; initialize ti = 0 before the loop and break upon the first match. RESULT ti]
            [COMMAND Within the current test case t, allocate boolean[][] visited2 = new boolean[2001][2001]; then apply the appropriate alternative flow based on whether tmp <= ti or tmp > ti. RESULT visited2]
            [COMMAND After the outer loop over t completes, print the accumulated contents of ans to System.out using System.out.print(ans). RESULT System.out]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Within the current test case t, tmp == -1 after the case prefix has been appended.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append `IMPOSSIBLE\n` to ans. RESULT ans]
            [COMMAND Continue to the next iteration of the outer for loop over t. RESULT next_t]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the current test case t, tmp != -1 and tmp <= ti.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Append the result of tmp + (ti - tmp + 1) / 2 followed by a newline to ans. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Within the current test case t, tmp != -1 and tmp > ti.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set success = false. For i = ti while i < tour.size() and tmp - ti + i < tour.size(): set mfx = tour.get(tmp + i - ti).get(0), mfy = tour.get(tmp + i - ti).get(1), mark visited2[mfy][mfx] = true, set ffx = tour.get(i).get(0), ffy = tour.get(i).get(1), and if visited2[ffy][ffx] is true, append i and a newline to ans, set success = true, and break. RESULT success, ans]
            [COMMAND If success is false after the loop, append `IMPOSSIBLE\n` to ans. RESULT ans]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
