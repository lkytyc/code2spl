[DEFINE_WORKER: "Reads commands that insert values into a two-heap median-maintenance structure and, for queries, outputs the current median along with an accumulated expression based on the two heap sums and B." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer Q from the input and initialize max-heap f using Collections.reverseOrder(), min-heap l, long accumulators fsum=0, lsum=0, and long B=0. RESULT Q]
            [COMMAND Loop exactly Q times using while (Q-- > 0); inside the loop read the next integer command com. RESULT com]
            [COMMAND If com == 1, read the next long a as the value to insert. When f.size() == l.size() and l.size() == 0, add a to f and add a to fsum. RESULT a]
            [COMMAND In the same insertion case with f.size() == l.size() and l.size() > 0, set x = l.peek(); if x < a, remove x from l, add a to l, add x to f, update lsum += a - x and fsum += x; otherwise add a to f and update fsum += a. RESULT x]
            [COMMAND In the insertion case when f.size() != l.size(), set x = f.peek(); if x <= a, add a to l and update lsum += a; otherwise remove x from f, add a to f, add x to l, update fsum += a - x and lsum += x. RESULT x]
            [COMMAND After the insertion logic for com == 1, read an additional long from the input and add it to B. RESULT B]
            [COMMAND If com != 1, set x = f.peek() as the current median; if f.size() == l.size(), print x followed by (lsum - fsum + B); otherwise print x followed by (lsum - fsum + x + B). RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
