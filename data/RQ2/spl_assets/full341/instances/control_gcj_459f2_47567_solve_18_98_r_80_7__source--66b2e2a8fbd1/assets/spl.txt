[DEFINE_WORKER: "Repeatedly compares the required array b with the current carrier array r, moving carriers left or right through interior positions while recording '.', '/', and '\' rows, then prints IMPOSSIBLE if no progress is made or prints the accumulated rows when equality is reached." solve]
    [INPUTS]
        <REF> sc </REF>: Scanner "Scanner containing the test case input; it provides the value of c and the b array values."
        <REF> num </REF>: int "The case number used in the output prefix."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next integer from Scanner sc and store it as c. RESULT c]
            [COMMAND Allocate an integer array b of length c, initialized to zero. RESULT b]
            [COMMAND Allocate an integer array r of length c, initialized to zero. RESULT r]
            [COMMAND For i from 0 to c-1, read the next integer from sc into b[i] and set r[i] to 1. RESULT r]
            [COMMAND Create an empty ArrayList of strings l, and initialize boolean good to false and string s to an empty string. RESULT l]
            [COMMAND Enter the outer while(true) loop. Set result to true, then for every i from 0 to c-1 set result to false if b[i] is not equal to r[i]. If result is still true, break out of the outer while loop. RESULT result]
            [COMMAND When result is false, prepare a movement pass: set last to 0, good to false, s to an empty string, and allocate a new zero-filled integer array rr of length c. RESULT rr]
            [COMMAND For each i from 0 to c-1 for which b[i] is greater than 0, set want equal to b[i] and scan j from last to c-1. RESULT want]
            [COMMAND Within the j scan, if r[j] is 0, append '.' to s and continue to the next j. RESULT s]
            [COMMAND Within the j scan, if r[j] is nonzero, subtract r[j] from want. If j is greater than i, j is greater than 0, and j is less than c-1, add r[j] to rr[j-1], append '/' to s, and set good to true. Otherwise, if j is less than i, j is greater than 0, and j is less than c-1, add r[j] to rr[j+1], append '\' to s, and set good to true. Otherwise, append '.' to s and add r[j] to rr[j]. RESULT rr]
            [COMMAND During the j scan, after updating rr and s, if want is less than or equal to 0, set last to j+1 and break the inner j loop. RESULT last]
            [COMMAND After processing all i, assign r to refer to rr, append s to l, and if good is false break the outer while loop; otherwise repeat the equality check at the start of the outer while loop. RESULT l]
            [COMMAND After the outer while loop ends, check result. If result is false, follow the impossible alternative; otherwise follow the successful-output alternative. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: result is false after the outer while loop.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + num + ": IMPOSSIBLE" to standard output. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: result is true after the outer while loop.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set s to an empty string, then append '.' to s for each i from 0 to c-1 so that s has length c. RESULT s]
            [COMMAND Add the final s to the list l. RESULT l]
            [COMMAND Print "Case #" + num + ": " + l.size() to standard output. RESULT none]
            [COMMAND For each string in l, print the string on its own line. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
