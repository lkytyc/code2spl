[DEFINE_WORKER: "Traverses the graph from start node s with a stack-like traversal, collecting nodes whose computed dis value is exactly 3, then fills ans at s and those collected nodes from que_1 and que_2 based on modulo-3 remainder counts, returning the updated ans and remaining queues." length3]
    [INPUTS]
        <REF> s </REF>: int "Starting node index; must be a valid index for ans and g."
        <REF> n </REF>: int "Number of nodes; determines the lengths of went and dis."
        <REF> g </REF>: list[list[int]] "Adjacency list representation of the graph."
        <REF> ans </REF>: list[int] "Array of node values; 0 represents an unfilled entry that may be assigned."
        <REF> que_1 </REF>: list or deque "Source of values to be popped from the end and assigned to ans entries."
        <REF> que_2 </REF>: list or deque "Source of values to be popped from the end and assigned to ans entries."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: tuple "Tuple (ans, que_1, que_2) with ans updated in place and que_1/que_2 shortened by the popped values."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize went as a list of False of length n, dis as a list of 0 of length n, see as an empty list, and que as an empty deque. RESULT went, dis, see, que]
            [COMMAND Append s to que and set went[s] to True. RESULT que, went]
            [COMMAND While que is not empty, pop go from the end of que; for each neighbor v in g[go], if went[v] is False, set dis[v] = dis[go] + 1 and went[v] = True; if dis[v] < 3 append v to que, else if dis[v] == 3 append v to see. RESULT went, dis, see, que]
            [COMMAND Initialize check1 to 0 and check2 to 0. RESULT check1, check2]
            [COMMAND For each e in see, if ans[e] == 0, continue; otherwise increment check1 when ans[e] % 3 == 1, and increment check2 when ans[e] % 3 == 2. RESULT check1, check2]
            [COMMAND After counting, if both check1 and check2 are positive, follow the guard exception flow; otherwise execute the assignment branch selected by the alternative_flows conditions and then continue to the return step. RESULT ans, que_1, que_2]
            [COMMAND Return the tuple (ans, que_1, que_2). RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: check1 > 0 and check2 <= 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_2 is non-empty, pop one value from the end of que_2 and assign it to ans[s]; otherwise follow the corresponding exception flow. RESULT ans, que_2]
            [COMMAND For each e in see, when ans[e] == 0, if que_1 is non-empty pop one value from the end of que_1 and assign it to ans[e]; otherwise follow the corresponding exception flow. RESULT ans, que_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 > 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_1 is non-empty, pop one value from the end of que_1 and assign it to ans[s]; otherwise follow the corresponding exception flow. RESULT ans, que_1]
            [COMMAND For each e in see, when ans[e] == 0, if que_2 is non-empty pop one value from the end of que_2 and assign it to ans[e]; otherwise follow the corresponding exception flow. RESULT ans, que_2]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) > len(que_2)]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_2 is non-empty, pop one value from the end of que_2 and assign it to ans[s]; otherwise follow the corresponding exception flow. RESULT ans, que_2]
            [COMMAND For each e in see, when ans[e] == 0, if que_1 is non-empty pop one value from the end of que_1 and assign it to ans[e]; otherwise follow the corresponding exception flow. RESULT ans, que_1]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: check1 <= 0 and check2 <= 0 and len(que_1) <= len(que_2)]
        [SEQUENTIAL_BLOCK]
            [COMMAND If que_1 is non-empty, pop one value from the end of que_1 and assign it to ans[s]; otherwise follow the corresponding exception flow. RESULT ans, que_1]
            [COMMAND For each e in see, when ans[e] == 0, if que_2 is non-empty pop one value from the end of que_2 and assign it to ans[e]; otherwise follow the corresponding exception flow. RESULT ans, que_2]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [EXCEPTION_FLOW: check1 > 0 and check2 > 0]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "The method exits because both modulo-3 categories are present among non-zero values in see."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the check1 > 0 branch, que_2 is empty before assigning ans[s]]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot assign a value to ans[s] because que_2 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the check1 > 0 branch, while filling a distance-3 node e with ans[e] == 0, que_1 is empty]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot fill a distance-3 node because que_1 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the check2 > 0 branch, que_1 is empty before assigning ans[s]]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot assign a value to ans[s] because que_1 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the check2 > 0 branch, while filling a distance-3 node e with ans[e] == 0, que_2 is empty]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot fill a distance-3 node because que_2 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the else branch with len(que_1) > len(que_2), que_2 is empty before assigning ans[s]]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot assign a value to ans[s] because que_2 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the else branch with len(que_1) > len(que_2), while filling a distance-3 node e with ans[e] == 0, que_1 is empty]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot fill a distance-3 node because que_1 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the else branch with len(que_1) <= len(que_2), que_1 is empty before assigning ans[s]]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot assign a value to ans[s] because que_1 has no remaining value."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: In the else branch with len(que_1) <= len(que_2), while filling a distance-3 node e with ans[e] == 0, que_2 is empty]
        [LOG "Print -1 to stdout and terminate."]
        [THROW SystemExit "Cannot fill a distance-3 node because que_2 has no remaining value."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
