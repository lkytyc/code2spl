[DEFINE_WORKER: "Reads a grid of black (#) and white (.) cells from standard input, performs a breadth-first spread in which black cells color adjacent white cells black, and prints the number of BFS layers minus one required to make every cell black." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a Scanner sc on System.in, read the first whitespace-separated token and parse it with Integer.parseInt into int h, then read the second token and parse it with Integer.parseInt into int w. RESULT h, w]
            [COMMAND Allocate char array a as char[h][w], allocate int array atmp of length h (unused later), set int max to 0, allocate blackar as int[h*w][2] to store black-cell coordinates, and set int p to 0. RESULT a, atmp, max, blackar, p]
            [COMMAND For each row i from 0 to h-1, read the next token from sc and store its characters in a[i]; then for each column j from 0 to w-1 using a char loop variable j (so correct behavior requires w to be at most 65535), if a[i][j] is '#', store (i,j) in blackar[p] and increment p. RESULT blackar, p]
            [COMMAND After the scanning loop, set maxp to p and p2 to p, then reset p to 0; maxp marks the end of the initial black-cell queue frontier and p2 marks the queue tail. RESULT maxp, p2, p]
            [COMMAND Start the outer while loop that continues while p < h*w; at the beginning of each outer iteration declare int hap = 0, which is not used by any other statement. RESULT hap]
            [COMMAND Within the outer while loop, process the current frontier with the inner for loop using no initialization and condition p < maxp, incrementing p after each iteration; for each iteration retrieve the queue entry blackar[p] as int i and int j. RESULT i, j]
            [COMMAND For the current black cell (i,j), if i-1 >= 0 and a[i-1][j] is '.', set a[i-1][j] to '#', store (i-1,j) in blackar[p2], and increment p2. RESULT blackar, p2]
            [COMMAND For the same current black cell (i,j), if j-1 >= 0 and a[i][j-1] is '.', set a[i][j-1] to '#', store (i,j-1) in blackar[p2], and increment p2. RESULT blackar, p2]
            [COMMAND For the same current black cell (i,j), if i+1 < h and a[i+1][j] is '.', set a[i+1][j] to '#', store (i+1,j) in blackar[p2], and increment p2. RESULT blackar, p2]
            [COMMAND For the same current black cell (i,j), if j+1 < w and a[i][j+1] is '.', set a[i][j+1] to '#', store (i,j+1) in blackar[p2], and increment p2. RESULT blackar, p2]
            [COMMAND After the inner for loop finishes for the current frontier, set maxp to p2 to include all newly added cells in the next frontier, and increment max by 1 to count the BFS layer. RESULT maxp, max]
            [COMMAND After the outer while loop exits, subtract 1 from max to exclude the layer that only processed the initial black cells, and assign the resulting value to int ans. RESULT ans]
            [COMMAND Create PrintWriter out wrapping System.out, write ans using out.println(ans), and flush out with out.flush(). RESULT out]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: sc.next() is called when no more input tokens are available]
        [LOG "NoSuchElementException is raised while reading h, w, or one of the grid rows."]
        [THROW NoSuchElementException "Thrown when required input is missing."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: The token passed to Integer.parseInt is not a valid base-10 integer]
        [LOG "NumberFormatException is raised while parsing h or w."]
        [THROW NumberFormatException "Thrown when an input token cannot be parsed as an integer."]
    [END_EXCEPTION_FLOW]

    [EXCEPTION_FLOW: h or w is negative, or h*w overflows the int range during array allocation]
        [LOG "NegativeArraySizeException is raised while allocating a, atmp, or blackar."]
        [THROW NegativeArraySizeException "Thrown when an array is allocated with a negative size."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
