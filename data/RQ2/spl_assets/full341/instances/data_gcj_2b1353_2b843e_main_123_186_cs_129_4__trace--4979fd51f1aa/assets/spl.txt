[DEFINE_WORKER: "Reads the number of test cases and, for each input n, prints a case header and a sequence of row/column coordinate pairs generated from a decreasing-power-of-two decomposition with alternating row direction." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: int "Always returns 0 after all test cases are processed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set up local state: declare integer variables cs, i, j, n, m, k, a, b, c, h, w, l, t, sn, and r, and integer array fl of size 64; initialize r to 0. RESULT local_state]
            [COMMAND Read an integer t from standard input into variable t. RESULT t]
            [COMMAND For each test case with cs from 1 to t inclusive, read an integer n from standard input into variable n, and copy n to sn. RESULT case_input]
            [COMMAND Print the case header line "Case #%d:\n" using cs. RESULT case_header]
            [COMMAND Set w to 0, then iterate i from 1 to 49 inclusive; add 2^(i-1) to w on each iteration, and when w is greater than or equal to n, set l to i and break out of the loop. RESULT l]
            [COMMAND Iterate i downward from l to 1 inclusive; if n is greater than or equal to 2^(i-1) + (i-1), set fl[i] to 1 and subtract 2^(i-1) from n; otherwise set fl[i] to 0 and decrement n by 1. RESULT fl]
            [COMMAND Set a to 0 and w to 0. RESULT a_and_w]
            [COMMAND Iterate i from 1 to l inclusive; if fl[i] is 1, add 2^(i-1) to w, print coordinate pairs (i,j) with j from 1 to i when a is 0 or with j from i down to 1 when a is 1, then set a to 1 - a; otherwise increment w by 1 and print coordinate pair (i,1) when a is 0 or (i,i) when a is 1. RESULT row_coordinates]
            [COMMAND While w is not equal to sn, increment l by 1 and w by 1, and print coordinate pair (l,1) when a is 0 or (l,l) when a is 1. RESULT remaining_coordinates]
            [COMMAND Return 0 after all test cases have been processed. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
