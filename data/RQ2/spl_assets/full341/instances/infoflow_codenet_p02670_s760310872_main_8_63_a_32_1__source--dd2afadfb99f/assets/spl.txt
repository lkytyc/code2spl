[DEFINE_WORKER: "Reads an integer board size n, fills an n x n grid with consecutive values, processes n*n cell queries by marking selected cells and accumulating the minimum number of unmarked cells above, below, left, and right for interior queries, then prints the accumulated total." main]
    [INPUTS]
        <REF> stdin </REF>: standard input stream "First an integer n, then exactly n*n long integers p, each representing a 1-based cell label in an n by n grid."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: standard output stream "The accumulated total cnt printed as a long integer with a trailing newline."
        <REF> return_code </REF>: int "Always returns 0 from main."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer n from standard input using scanf("%d",&n). RESULT n]
            [COMMAND Declare int a[501][501], long p, k=1, cnt=0; set k to 1 and cnt to 0; p will be assigned inside the loop. RESULT a,p,k,cnt]
            [COMMAND For i from 0 to n-1 inclusive, for j from 0 to n-1 inclusive, assign a[i][j]=k and then increment k, filling the n x n grid in row-major order with consecutive values starting from 1. RESULT a]
            [COMMAND For ii from 0 to n*n-1 inclusive, read a long value p from standard input using scanf("%ld",&p). RESULT p]
            [COMMAND Within the loop over ii, if 1 <= p && n >= p, set a[0][p-1]=0. RESULT a]
            [COMMAND Within the loop over ii, else if n*n >= p && n*n-n+1 <= p, set a[n-1][n-(n*n)+p-1]=0. RESULT a]
            [COMMAND Within the loop over ii, else if p%n==0, set a[(p/n)-1][n-1]=0. RESULT a]
            [COMMAND Within the loop over ii, else if p%n==1, set a[p/n][0]=0. RESULT a]
            [COMMAND Within the loop over ii, for the remaining interior case, set a[p/n][(p%n)-1]=0, then set i=p/n and j=(p%n)-1, and initialize long m[5]={i,n-1-i,j,n-1-j}. RESULT i,j,m]
            [COMMAND Within the interior case, for jj from 0 to i-1 inclusive, if a[jj][j]==0 decrement m[0]; after the scan m[0] counts unmarked cells above (i,j). RESULT m]
            [COMMAND Within the interior case, for jj from n-1 down to i+1 inclusive, if a[jj][j]==0 decrement m[1]; after the scan m[1] counts unmarked cells below (i,j). RESULT m]
            [COMMAND Within the interior case, for jj from 0 to j-1 inclusive, if a[i][jj]==0 decrement m[2]; after the scan m[2] counts unmarked cells left of (i,j). RESULT m]
            [COMMAND Within the interior case, for jj from n-1 down to j+1 inclusive, if a[i][jj]==0 decrement m[3]; after the scan m[3] counts unmarked cells right of (i,j). RESULT m]
            [COMMAND Within the interior case, sort the first four elements of m in ascending order using qsort(m,4,sizeof(long),compare_int), set long mm=m[0], and add mm to cnt. RESULT cnt]
            [COMMAND After the loop over ii completes, print cnt using printf("%ld\n",cnt) and return 0. RESULT return_code]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
