[DEFINE_WORKER: "Processes Q operations that either insert a long a and accumulate another long into B, or query the current maximum of the lower-half heap and a computed expression based on heap sums and B, maintaining the two heaps with running sums." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the initial integer value from the input and store it in Q. RESULT Q]
            [COMMAND Initialize f as an empty max-heap priority queue of Long values using reverse order, initialize l as an empty min-heap priority queue of Long values using natural order, and set fsum, lsum, and B to 0. RESULT _]
            [COMMAND While the current value of Q is greater than 0, decrement Q by 1 after the comparison and execute the loop body; for a nonnegative initial Q this runs exactly the original Q times. RESULT _]
            [COMMAND At the beginning of each loop iteration, read the next integer from the input into com. RESULT com]
            [COMMAND If com equals 1, perform the insertion path; otherwise, perform the query path. RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: com equals 1 (insertion operation)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next long from the input into a. RESULT a]
            [COMMAND If the sizes of f and l are equal and l is empty, add a to f and add a to fsum. RESULT _]
            [COMMAND If the sizes of f and l are equal and l is not empty, get x as the top of l. If x is less than a, add (a - x) to lsum, add x to fsum, remove the top of l, add a to l, and add x to f. If x is not less than a, add a to f and add a to fsum. RESULT _]
            [COMMAND If the sizes of f and l are not equal, get x as the top of f. If x is less than or equal to a, add a to l and add a to lsum. If x is greater than a, add x to lsum, add (a - x) to fsum, remove the top of f, add a to f, and add x to l. RESULT _]
            [COMMAND After the insertion branch completes, read the next long from the input and add it to B. RESULT B]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: com does not equal 1 (query operation)]
        [SEQUENTIAL_BLOCK]
            [COMMAND Get x as the top of f. RESULT x]
            [COMMAND If the sizes of f and l are equal, print x and the value (lsum - fsum + B) separated by a space. RESULT _]
            [COMMAND If the sizes of f and l are not equal, print x and the value (lsum - fsum + x + B) separated by a space. RESULT _]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
