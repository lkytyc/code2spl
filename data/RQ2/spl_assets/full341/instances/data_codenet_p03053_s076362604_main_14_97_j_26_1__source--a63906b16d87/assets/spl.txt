[DEFINE_WORKER: "Reads an H-by-W grid and prints the largest value of the minimum of four directional distances to a '#' wall, computed with dynamic programming and sentinel boundaries." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer H from standard input, then read integer W from standard input. RESULT H, W]
            [COMMAND Read the grid into l[1..H][1..W]: for each row i from 1 to H, consume one character into d as the separator or newline skip, then for each j from 1 to W read one character into c and store it into l[i][j]. RESULT l]
            [COMMAND Set the sentinel value m = H + W. RESULT m]
            [COMMAND For i from 1 to H, set p[i][0], q[i][W+1], r[i][0], and s[i][W+1] to m. RESULT p, q, r, s]
            [COMMAND For j from 1 to W, set p[0][j], q[0][j], r[H+1][j], and s[H+1][j] to m. RESULT p, q, r, s]
            [COMMAND For i from 1 to H and j from 1 to W, if l[i][j] is '#' set p[i][j] to 0; otherwise set p[i][j] to min(p[i-1][j], p[i][j-1]) + 1. RESULT p]
            [COMMAND For i from 1 to H and j from W down to 1, if l[i][j] is '#' set q[i][j] to 0; otherwise set q[i][j] to min(q[i-1][j], q[i][j+1]) + 1. RESULT q]
            [COMMAND For i from H down to 1 and j from 1 to W, if l[i][j] is '#' set r[i][j] to 0; otherwise set r[i][j] to min(r[i+1][j], r[i][j-1]) + 1. RESULT r]
            [COMMAND For i from H down to 1 and j from W down to 1, if l[i][j] is '#' set s[i][j] to 0; otherwise set s[i][j] to min(s[i+1][j], s[i][j+1]) + 1. RESULT s]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For i from 1 to H and j from 1 to W, compute n[i][j] as the minimum of p[i][j], q[i][j], r[i][j], and s[i][j] using the three successive min assignments, then update ans to max(ans, n[i][j]). RESULT n, ans]
            [COMMAND Print the final integer value ans to standard output using printf("%d", ans) without adding a newline. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
