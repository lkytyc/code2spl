[DEFINE_WORKER: "Determines a hidden integer through interactive Y/N queries: after an initial 10^10 test, it either finds the magnitude using repeated 9s or reconstructs every decimal digit with binary-search queries, then prints the answer." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize ans as an empty string. RESULT ans]
            [COMMAND Print the query "? " + "1" + "0"*10 (that is, "? 10000000000"), flush standard output, and read the interactive response into A. RESULT A]
            [COMMAND If A equals "Y", execute the first alternative flow; otherwise execute the second alternative flow. RESULT A]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: A equals "Y"]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set n = 1. RESULT n]
            [COMMAND Print the query "? " + "9"*n, flush standard output, and read the response into A. RESULT A]
            [COMMAND While A == "N", increment n by 1, print "? " + "9"*n, flush standard output, and read the response into A again. RESULT n]
            [COMMAND After A becomes "Y", print the final answer "! " + str(10**(n-1)). RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: A does not equal "Y"]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize l = 0, r = 10, and S as an 11-element array with S[0] = "Y" and S[10] = "N". RESULT l/r/S]
            [COMMAND Determine K by binary search on powers of 10: in an infinite loop, set m = (l+r)//2, print "? " + str(10**m), flush, and read A. If A == "Y" and S[m+1] == "N", set K = m+1 and break; if A == "Y" otherwise, set S[m] = "Y" and l = m. If A == "N" and S[m-1] == "Y", set K = m and break; otherwise set S[m] = "N" and r = m. RESULT K]
            [COMMAND Reinitialize l = 1, r = 9, S as a 10-element array with S[9] = "Y", and set ashi = "9"*10. RESULT l/r/S/ashi]
            [COMMAND Print the query "? " + ans + "1" + ashi, flush standard output, and read A. If A == "Y", append "1" to ans. RESULT ans/A]
            [COMMAND If A != "Y" for the first digit, set S[1] = "N" and perform a binary search on digits: in an infinite loop, set m = (l+r)//2, print "? " + ans + str(m) + ashi, flush, and read A. If A == "Y" and S[m-1] == "N", append str(m) to ans and break; if A == "Y" otherwise, set S[m] = "Y" and r = m. If A == "N" and S[m+1] == "Y", append str(m+1) to ans and break; otherwise set S[m] = "N" and l = m. RESULT ans]
            [COMMAND For i in range(K-1), reconstruct each remaining digit: initialize l = 0, r = 9, S as a 10-element array with S[9] = "Y"; print "? " + ans + "0" + ashi, flush, and read A. If A == "Y", append "0" to ans; otherwise set S[0] = "N" and in an infinite loop set m = (l+r)//2, print "? " + ans + str(m) + ashi, flush, and read A. If A == "Y" and S[m-1] == "N", append str(m) to ans and break; if A == "Y" otherwise, set S[m] = "Y" and r = m. If A == "N" and S[m+1] == "Y", append str(m+1) to ans and break; otherwise set S[m] = "N" and l = m. RESULT ans]
            [COMMAND Print the final answer "! " + ans. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
