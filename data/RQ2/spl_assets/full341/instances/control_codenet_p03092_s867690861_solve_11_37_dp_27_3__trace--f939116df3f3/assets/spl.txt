[DEFINE_WORKER: "Reads n, A, B and an array P, decrements each P element by one, then uses dynamic programming with a large sentinel value to compute and print the optimal cost dp[n][n]." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer n from input using ni(). RESULT n]
            [COMMAND Read integer A from input using ni(). RESULT A]
            [COMMAND Read integer B from input using ni(). RESULT B]
            [COMMAND Allocate an int array P of length n. RESULT P]
            [COMMAND For each index i from 0 to n-1 inclusive, read an integer from input using ni() and assign it to P[i]. RESULT P]
            [COMMAND For each index i from 0 to n-1 inclusive, decrement P[i] by 1. RESULT P]
            [COMMAND Allocate a 2D long array dp with dimensions n+1 by n+1. RESULT dp]
            [COMMAND For each row i from 0 to n inclusive, fill all columns dp[i][j] for j from 0 to n inclusive with Long.MAX_VALUE / 2 using Arrays.fill. RESULT dp]
            [COMMAND For each row i from 0 to n inclusive and each column j from 0 to n inclusive, apply the DP recurrence: if i==0 and j==0 set dp[i][j] to 0; else if i>0 and j>0 and P[i-1]==j-1 set dp[i][j] to dp[i-1][j-1]; else if i>0 update dp[i][j] to the minimum of dp[i][j] and dp[i-1][j] plus (P[i-1] <= j ? B : A), and if j>0 update dp[i][j] to the minimum of dp[i][j] and dp[i][j-1]. RESULT dp]
            [COMMAND Print dp[n][n] to out. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [EXCEPTION_FLOW: n is negative when executing new int[n] for P.]
        [LOG "NegativeArraySizeException: array length is negative."]
        [THROW NegativeArraySizeException "Thrown when an array with negative length is instantiated."]
    [END_EXCEPTION_FLOW]

[END_WORKER]
