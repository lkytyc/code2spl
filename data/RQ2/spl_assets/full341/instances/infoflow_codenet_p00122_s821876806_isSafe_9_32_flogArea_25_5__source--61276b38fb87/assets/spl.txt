[DEFINE_WORKER: "Checks whether a queue-based expansion of positions on a 10x10 grid can reach a layer-9 safe state using two bounded move patterns and a precomputed safe-state table, returning True immediately when such a state is found and False when no candidates remain." isSafe]
    [INPUTS]
        <REF> wfs </REF>: queue "Worklist queue of states; each state is a two-element list [position, layer], where position is [x, y]."
        <REF> safe </REF>: list "Precomputed safe-state table; safe[spn] is a list of states at layer spn, each state having the form [[x, y], spn]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: bool "True if a state at layer 9 is reachable from the worklist using the defined move patterns; otherwise False."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND While the worklist wfs is not empty, continue processing states; if wfs becomes empty, fall through to return False. RESULT loop_active]
            [COMMAND Dequeue the next state flog from wfs with wfs.get() and use flog[0] as the coordinate and flog[1] as the current layer. RESULT flog]
            [COMMAND Create an empty list flogArea and set spn = flog[1] + 1. RESULT flogArea_spn]
            [COMMAND For the first move pattern, with i as the outer loop over {-1, 0, 1} and j as the inner loop over {-2, 2}, compute dx = flog[0][0] + j and dy = flog[0][1] + i. If 0 <= dx < 10 and 0 <= dy < 10, append [[dx, dy], spn] to flogArea. RESULT flogArea]
            [COMMAND For the second move pattern, with j as the outer loop over {-1, 0, 1} and i as the inner loop over {-2, 2}, compute dx = flog[0][0] + j and dy = flog[0][1] + i. If 0 <= dx < 10 and 0 <= dy < 10, append [[dx, dy], spn] to flogArea. RESULT flogArea]
            [COMMAND For each index i from 0 to len(safe[spn])-1 inclusive, take candidate = safe[spn][i] and test whether candidate is contained in flogArea. RESULT candidate_matches]
            [COMMAND If candidate is in flogArea and spn == 9, immediately return True, terminating the search. RESULT true_return]
            [COMMAND If candidate is in flogArea and spn < 9, enqueue candidate back into wfs with wfs.put(candidate) and continue the outer while loop. If spn > 9, take no action for this candidate. RESULT wfs]
            [COMMAND After the while loop exits because wfs is empty, return False. RESULT false_return]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
