[DEFINE_WORKER: "Traverses nodes up to distance 3 from s, counts modulo-3 classes among the discovered boundary nodes, and assigns available values from two queues to ans[s] and to unassigned boundary nodes according to which classes are required." length3]
    [INPUTS]
        <REF> s </REF>: int "Starting node index for the depth-limited graph traversal."
        <REF> n </REF>: int "Total number of graph nodes; determines the lengths of the visited and distance lists."
        <REF> g </REF>: list of lists of int "Adjacency list of the graph; g[node] contains neighbor indices."
        <REF> ans </REF>: list of int "Value array that is mutated; zero entries are treated as unassigned and may be filled."
        <REF> que_1 </REF>: list or collections.deque of int "First pool of candidate values; entries are consumed by pop()."
        <REF> que_2 </REF>: list or collections.deque of int "Second pool of candidate values; entries are consumed by pop()."
    [END_INPUTS]

    [OUTPUTS]
        <REF> ans </REF>: list of int "Updated value array after assigning ans[s] and eligible entries in the boundary list."
        <REF> que_1 </REF>: list or collections.deque of int "First candidate pool after consumed values are removed."
        <REF> que_2 </REF>: list or collections.deque of int "Second candidate pool after consumed values are removed."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize went as [False] * n, dis as [0] * n, see as an empty list, and que as an empty deque. RESULT went, dis, see, que]
            [COMMAND Append s to que and set went[s] to True. RESULT que, went]
            [COMMAND While que is not empty, pop the rightmost/last element go from que. For each neighbor in g[go] in order, if went[neighbor] is True, continue; otherwise set dis[neighbor] = dis[go] + 1, mark went[neighbor] = True, and if dis[neighbor] < 3 append neighbor to que, else if dis[neighbor] == 3 append neighbor to see. RESULT que, went, dis, see]
            [COMMAND Set check1 = 0 and check2 = 0. RESULT check1, check2]
            [COMMAND For each e in see, if ans[e] == 0 continue; if ans[e] % 3 == 1 increment check1; if ans[e] % 3 == 2 increment check2. RESULT check1, check2]
            [COMMAND If check1 > 0 and check2 > 0 (short-circuit conjunction), print -1 and terminate by calling exit(). RESULT none]
            [COMMAND In the branch reached when check1 > 0, if que_2 is empty print -1 and terminate; otherwise pop from que_2 and assign the value to ans[s]. Then for each e in see, only if ans[e] == 0, check que_1; if que_1 is empty print -1 and terminate, otherwise pop from que_1 and assign the value to ans[e]. RESULT ans, que_1, que_2]
            [COMMAND In the mutually exclusive branch reached when check1 <= 0 and check2 > 0, if que_1 is empty print -1 and terminate; otherwise pop from que_1 and assign the value to ans[s]. Then for each e in see, only if ans[e] == 0, check que_2; if que_2 is empty print -1 and terminate, otherwise pop from que_2 and assign the value to ans[e]. RESULT ans, que_1, que_2]
            [COMMAND In the fallback branch reached when check1 <= 0 and check2 <= 0, if len(que_1) > len(que_2), pop from que_2 for ans[s] after checking que_2 is nonempty, otherwise print -1 and terminate; then for each e in see with ans[e] == 0, pop from que_1 for ans[e] after checking que_1 is nonempty, otherwise print -1 and terminate. If len(que_1) <= len(que_2), do the mirror assignment: pop from que_1 for ans[s] after checking que_1 is nonempty, otherwise print -1 and terminate; then for each e in see with ans[e] == 0, pop from que_2 for ans[e] after checking que_2 is nonempty, otherwise print -1 and terminate. RESULT ans, que_1, que_2]
            [COMMAND Return ans, que_1, and que_2. RESULT ans, que_1, que_2]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
