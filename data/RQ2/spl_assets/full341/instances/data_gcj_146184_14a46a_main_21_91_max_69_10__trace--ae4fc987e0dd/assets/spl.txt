[DEFINE_WORKER: "Reads multiple test cases of paired (c, j) constraints, validates consistency by tracking the maximum observed ratio between adjacent conflicting pairs, and prints the resulting positive ans_c and ans_j assignment or IMPOSSIBLE for each case." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the double variable max to -1. RESULT max]
            [COMMAND Read one integer from standard input into T using scanf("%d", &T). RESULT T]
            [COMMAND For each test case t from 1 to T inclusive, execute the following steps using the current value of max, which is initialized to -1 and is not reset between test cases. RESULT t]
            [COMMAND Within the test-case loop over t, set ans to 1. RESULT ans]
            [COMMAND Within the test-case loop over t, read one integer from standard input into N using scanf("%d", &N). RESULT N]
            [COMMAND Within the test-case loop over t, for i from 1 to N inclusive, read two integers into arr[i].c and arr[i].j using scanf("%d%d", &arr[i].c, &arr[i].j). RESULT arr]
            [COMMAND Within the test-case loop over t, set flag to 0. RESULT flag]
            [COMMAND Within the test-case loop over t, set flag_big to -1. RESULT flag_big]
            [COMMAND Within the test-case loop over t, for i from 1 to N-1 inclusive, set j to i+1 and execute the following pair analysis. RESULT i, j]
            [COMMAND Within the inner pair-analysis loop, if both arr[i].c <= arr[j].c and arr[i].j <= arr[j].j, continue to the next pair without changing state. RESULT none]
            [COMMAND Within the inner pair-analysis loop, if both arr[i].c >= arr[j].c and arr[i].j >= arr[j].j, set flag to 1 and break out of the inner loop. RESULT flag]
            [COMMAND Within the inner pair-analysis loop, when neither dominance condition holds, compute double a = arr[j].c - arr[i].c and double b = arr[i].j - arr[j].j. RESULT a, b]
            [COMMAND Within the inner pair-analysis loop, if flag_big is -1, set flag_big to 1 when a > 0, otherwise set flag_big to 0. RESULT flag_big]
            [COMMAND Within the inner pair-analysis loop, if flag_big is 1 and a < 0, set flag to 1 and break out of the inner loop. RESULT flag]
            [COMMAND Within the inner pair-analysis loop, if flag_big is 1 and a >= 0, update max to b/a when max < b/a. RESULT max]
            [COMMAND Within the inner pair-analysis loop, if flag_big is 0 and a > 0, set flag to 1 and break out of the inner loop. RESULT flag]
            [COMMAND Within the inner pair-analysis loop, if flag_big is 0 and a <= 0, update max to b/a when max < b/a. RESULT max]
            [COMMAND After the inner pair-analysis loop, if flag is nonzero, print "Case #%d: IMPOSSIBLE\n" with t. RESULT none]
            [COMMAND After the inner pair-analysis loop, if flag is zero and flag_big is -1, set ans_c to 1 and ans_j to 1. RESULT ans_c, ans_j]
            [COMMAND After the inner pair-analysis loop, if flag is zero and flag_big is 1, set ans_j to 1 and ans_c to floor(max + 1). RESULT ans_c, ans_j]
            [COMMAND After the inner pair-analysis loop, if flag is zero and flag_big is 0, set ans_c to 1 and ans_j to floor(max + 1). RESULT ans_c, ans_j]
            [COMMAND After the inner pair-analysis loop, if flag is zero, print "Case #%d: %d %d\n" with t, ans_c, ans_j. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
