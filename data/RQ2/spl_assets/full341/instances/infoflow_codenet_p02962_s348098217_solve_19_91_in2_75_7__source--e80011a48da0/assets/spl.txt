[DEFINE_WORKER: "Reads two strings, compares overlapping string blocks based on their length relationship, and prints either -1 or the maximum matching block count to the provided PrintWriter, preserving the original early-close and loop-boundary behavior." solve]
    [INPUTS]
        <REF> testNumber </REF>: int "Test case number passed to solve; ignored by the method body."
        <REF> br </REF>: FastScanner "Input scanner used to read the two strings in1 and in2."
        <REF> pw </REF>: PrintWriter "Output writer receiving all printed results."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the next string token from br into in1. RESULT in1]
            [COMMAND Read the next string token from br into in2. RESULT in2]
            [COMMAND Convert in1 to the character array t and in2 to the character array s. RESULT t, s]
            [COMMAND Compare t.length and s.length and follow the branch selected by the length ordering. RESULT length_branch]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: t.length > s.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig to 0 and poss to true. RESULT orig, poss]
            [COMMAND For i from 0 to t.length - s.length - 1 in increments of s.length, test whether in1.substring(i, i + s.length) equals in2; if not, set poss to false; otherwise increment orig. RESULT orig, poss]
            [COMMAND If poss is true, print -1 to pw and close pw; do not return or stop, continue with the remaining steps. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For i from 0 to t.length - s.length - 1 in increments of s.length, while in1.substring(i, i + s.length) equals in2, increment ans, and break on the first mismatch. RESULT ans]
            [COMMAND For i from t.length - s.length down to 0 in decrements of s.length, while in1.substring(i, i + s.length) equals in2, increment ans, and break on the first mismatch; note this can double-count the block at index 0. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw; this close may be a second close after the early poss-true close. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: s.length == t.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Evaluate s.equals(t), where s and t are char arrays, so this is reference equality rather than content equality. RESULT equal_ref]
            [COMMAND If equal_ref is true, print -1 to pw; otherwise print 0 to pw. Do not close pw in this branch. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: s.length > t.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig to 0 and poss to true. RESULT orig, poss]
            [COMMAND For i from 0 while i < s.length - t.length, using increment s.length, test whether in2.substring(i, i + t.length) equals in1; if not, set poss to false. Because s.length > t.length, this loop effectively tests only i = 0. RESULT orig, poss]
            [COMMAND If poss is true, print -1 to pw and close pw; do not return or stop, continue with the remaining steps. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For i from 0 to s.length - t.length - 1 in increments of t.length, while in2.substring(i, i + t.length) equals in1, increment ans, and break on the first mismatch. RESULT ans]
            [COMMAND For i from s.length - t.length down to 0 in decrements of t.length, while in2.substring(i, i + t.length) equals in1, increment ans, and break on the first mismatch; note this can double-count the block at index 0. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw; this close may be a second close after the early poss-true close. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
