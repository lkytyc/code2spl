[DEFINE_WORKER: "Reads n decimal strings, converts each to a fixed-point integer scaled to 4 decimal places while marking values with more than four decimal places as invalid, enumerates the divisors of 100000000, and counts pairs of valid values in input order whose product is an integer." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read a long long integer from standard input into n using %lld format. RESULT n]
            [COMMAND For each i from 0 to n-1, read a whitespace-delimited string token from standard input into s. RESULT s]
            [COMMAND Set a[i] to 0. RESULT a[i]]
            [COMMAND Set k to 0. RESULT k]
            [COMMAND For each character index j in s while s[j] is not the null terminator, process the current character. RESULT j]
            [COMMAND If the current character is not '.', update a[i] to 10 * a[i] plus the digit value of s[j]. RESULT a[i]]
            [COMMAND If the current character is not '.' and k is greater than 0, increment k. RESULT k]
            [COMMAND If the current character is '.', set k to 1. RESULT k]
            [COMMAND After finishing the string, if k is 0, set k to 1. RESULT k]
            [COMMAND Multiply a[i] by 10 while k is less than 10, incrementing k each time. RESULT a[i]]
            [COMMAND Set q to 100000. RESULT q]
            [COMMAND For each i from 0 to n-1, if a[i] modulo 100000 is greater than 0, set a[i] to -1; otherwise set a[i] to a[i] divided by 100000. RESULT a[i]]
            [COMMAND Set q to 100000000. RESULT q]
            [COMMAND Set j to 0. RESULT j]
            [COMMAND For i from 1 while i*i <= 100000000, if 100000000 is divisible by i, store i in b[j] and increment j. RESULT b]
            [COMMAND After the ascending divisor loop, decrement i by 1; if i*i equals 100000000, decrement i by 1 again. RESULT i]
            [COMMAND For i descending while i > 0, if 100000000 is divisible by i, store 100000000/i in b[j] and increment j. RESULT b]
            [COMMAND For i from 0 to j-1, set c[i] to 0. RESULT c]
            [COMMAND Set ans to 0. RESULT ans]
            [COMMAND For each i from 0 to n-1, if a[i] is less than 0, skip the rest of the current iteration and continue with the next i. RESULT i]
            [COMMAND For the current valid a[i], compute l as 100000000 divided by the greatest common divisor of 100000000 and a[i]. RESULT l]
            [COMMAND Set min to -1 and max to j. RESULT max]
            [COMMAND While max minus min is greater than 1, compute mid as (max + min) / 2; if b[mid] is less than l set min to mid, otherwise set max to mid. RESULT max]
            [COMMAND Add c[max] to ans. RESULT ans]
            [COMMAND For k from 0 to j-1, if a[i] is divisible by b[k], increment c[k]. RESULT c]
            [COMMAND Print ans as a long long integer followed by a newline. RESULT printed_ans]
            [COMMAND Return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
