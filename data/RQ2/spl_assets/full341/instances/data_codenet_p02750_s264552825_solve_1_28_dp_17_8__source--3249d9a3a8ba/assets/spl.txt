[DEFINE_WORKER: "Computes the maximum number of tasks that can be completed within time t by separating a==0 tasks, dynamic-programming over positive a tasks, and checking feasible combinations with cumulative zero-task costs." solve]
    [INPUTS]
        <REF> n </REF>: int "Task count parameter; not referenced in the method body."
        <REF> t </REF>: int "Available time budget used both as the DP sentinel and as the final feasibility threshold."
        <REF> ab_list </REF>: list[tuple[int,int]] "List of (a, b) pairs; entries with a == 0 contribute b to the zero-task list, while entries with a > 0 are kept as full pairs."
    [END_INPUTS]

    [OUTPUTS]
        <REF> res </REF>: int "Maximum total number of tasks achievable within time t."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Build a0b_list by comprehension: collect ab[1] for every ab in ab_list for which ab[0] == 0. RESULT a0b_list]
            [COMMAND Initialize apb_list as an empty list. RESULT apb_list]
            [COMMAND For each ab in ab_list, in iteration order, if ab[0] > 0 then append the entire ab pair to apb_list; otherwise skip it. RESULT apb_list]
            [COMMAND Rebind a0b_list to sorted(a0b_list), producing ascending order of its b values. RESULT a0b_list]
            [COMMAND Rebind apb_list to sorted(apb_list, key=lambda x: -x[0]/(x[1]+1)), which sorts positive-task pairs by descending a/(b+1). RESULT apb_list]
            [COMMAND Set m to len(apb_list). RESULT m]
            [COMMAND Create dp as [[t+1]*30 for _ in range(m+1)], giving m+1 rows each containing 30 copies of the sentinel t+1. RESULT dp]
            [COMMAND For each i from 0 through m inclusive, set dp[i][0] = 0. RESULT dp]
            [COMMAND For each i from 0 to m-1 and each k from 1 to 29, compute ti = dp[i][k-1] + 1, unpack a,b = apb_list[i], and set dp[i+1][k] = min(dp[i][k], ti + a*ti + b). RESULT dp]
            [COMMAND Initialize a0b_list_cum_sum with a list containing the single element 0. RESULT a0b_list_cum_sum]
            [COMMAND Set s = 0. RESULT s]
            [COMMAND For each b in a0b_list, update s += b + 1 and append s to a0b_list_cum_sum, forming cumulative sums of zero-task costs. RESULT a0b_list_cum_sum]
            [COMMAND Set res = 0 as the initial maximum count. RESULT res]
            [COMMAND For each k from 0 through len(a0b_list_cum_sum)-1 and each p from 0 through 29, if dp[m][p] + a0b_list_cum_sum[k] <= t, update res = max(res, k + p). RESULT res]
            [COMMAND Return the value of res. RESULT res]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
