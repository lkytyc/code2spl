[DEFINE_WORKER: "Reads a binary grid using a serpentine traversal and outputs coordinate records for every segment traversed while a running coin parity flag is true." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "Always 0 to indicate successful termination."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare a static unsigned short output array out of size 1000000, a char array a of size 501, read h and w by calling gl() for h and gl() for w, initialize pos to 0, set haveCoin to 0, and declare short old_y and old_x. RESULT initial_state]
            [COMMAND For each row i from 1 to h inclusive, for each column j from 1 to w inclusive, read a character with getchar_unlocked(), subtract 48 into t, consume the following character with getchar_unlocked(), and store t into a[j]. RESULT row_buffer]
            [COMMAND In the outer row loop, when i is odd, iterate j from 1 to w inclusive and, at each cell (i, j), if haveCoin is true, store old_y into out[4*pos], old_x into out[4*pos+1], i into out[4*pos+2], and j into out[4*pos+3], then increment pos. RESULT odd_row_output]
            [COMMAND In the same odd-row left-to-right traversal, after the conditional output, update haveCoin by XORing it with a[j] & 1, then set old_y to i and old_x to j. RESULT odd_row_state]
            [COMMAND In the outer row loop, when i is even, iterate j from w down to 1 inclusive and, at each cell (i, j), if haveCoin is true, store old_y into out[4*pos], old_x into out[4*pos+1], i into out[4*pos+2], and j into out[4*pos+3], then increment pos. RESULT even_row_output]
            [COMMAND In the same even-row right-to-left traversal, after the conditional output, update haveCoin by XORing it with a[j] & 1, then set old_y to i and old_x to j. RESULT even_row_state]
            [COMMAND After all rows, print pos using pl(pos) and print a newline using putchar_unlocked. RESULT position_output]
            [COMMAND For each record index i from 0 to pos-1, print the value at out[4*i], a space, the value at out[4*i+1], a space, the value at out[4*i+2], a space, the value at out[4*i+3], and a newline. RESULT records_output]
            [COMMAND Return 0. RESULT return_0]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
