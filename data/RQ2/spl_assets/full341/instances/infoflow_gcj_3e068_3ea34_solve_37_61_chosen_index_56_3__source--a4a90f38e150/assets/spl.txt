[DEFINE_WORKER: "Processes a sequence of candidate sets by shuffling each set, incrementing usage counts for non-used candidates, selecting the least-used candidate, and printing it or -1 when no candidate is available." solve]
    [INPUTS]
        <REF> case_index </REF>: integer "Unused per-case index parameter passed to solve."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize the sentinel constant USED to 0xFFFFFFF (268435455). RESULT USED]
            [COMMAND Advance to the next input line and read integer n, the number of outer iterations. RESULT n]
            [COMMAND Initialize counts as a zero-filled list of length n, where counts[index] tracks usage of each candidate index. RESULT counts]
            [COMMAND For each outer iteration i from 0 to n-1, advance to the next input line and read integer d, the number of candidates in the current iteration. RESULT d]
            [COMMAND If d == 0, execute the d == 0 alternative flow; otherwise continue with candidate processing. RESULT d]
            [COMMAND For the current nonzero-d iteration, create an empty candidate list cur, then for j from 0 to d-1 read integer itemaa and append it to cur. RESULT cur]
            [COMMAND Shuffle cur in place so tie-breaking among equal usage counts is randomized. RESULT cur]
            [COMMAND For each index in the shuffled cur, if counts[index] is not USED, increment counts[index] by 1. RESULT counts]
            [COMMAND Select chosen_index as the element of cur with the minimum counts value, using min(cur, key=lambda i: counts[i]); the prior shuffle randomizes the choice when counts tie. RESULT chosen_index]
            [COMMAND After choosing chosen_index, evaluate counts[chosen_index] and execute the corresponding alternative flow. RESULT chosen_index]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: In the current outer iteration, d == 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1 to stdout. RESULT stdout]
            [COMMAND Continue to the next outer iteration, skipping candidate reading and selection. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After selecting chosen_index, counts[chosen_index] == USED.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1 to stdout. RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After selecting chosen_index, counts[chosen_index] != USED.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print chosen_index to stdout. RESULT stdout]
            [COMMAND Set counts[chosen_index] to USED. RESULT counts]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
