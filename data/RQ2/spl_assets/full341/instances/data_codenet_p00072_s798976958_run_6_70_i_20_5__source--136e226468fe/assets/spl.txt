[DEFINE_WORKER: "Processes multiple graph test cases from standard input until a zero node count terminates input, selects minimum-cost edges from the upper and lower triangles of a symmetric cost matrix, accumulates min/100 - 1 for each selected edge, and prints the resulting count for each test case." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner instance bound to System.in and assign it to scan. RESULT scan]
            [COMMAND While scan.hasNext(), read the next integer from scan and assign it to n; if n is 0, terminate the loop as described in alternative_flows. RESULT n]
            [COMMAND For this non-zero n test case, read the next integer from scan and assign it to m. RESULT m]
            [COMMAND Create an n by n integer matrix cost and initialize every entry cost[i][j] to Integer.MAX_VALUE for i from 0 to n-1 and j from 0 to n-1. RESULT cost]
            [COMMAND For each edge index from 0 to m-1, read the next token from scan, split it by ',', parse the three resulting strings as a, b, and c, then set cost[a][b] = c and cost[b][a] = c. RESULT cost]
            [COMMAND Initialize count to 0, min to 0, ti to 0, tj to 0, and fl to a new boolean array of length n with all false entries. RESULT count, min, ti, tj, fl]
            [COMMAND For i from 0 to n-2 inclusive, reset min to 0; for each j from i+1 to n-1 inclusive, if cost[i][j] is Integer.MAX_VALUE then continue, otherwise if min == 0 || min > cost[i][j] holds (with short-circuit OR), set min = cost[i][j], ti = i, and tj = j. RESULT min, ti, tj]
            [COMMAND After the inner j loop for each i, if min is not 0, add min / 100 - 1 to count and set fl[ti] = true and fl[tj] = true. RESULT count, fl]
            [COMMAND For i from 0 to n-1 inclusive, if fl[i] is true then continue to the next i; otherwise reset min to 0 and for each j from 0 to i-1 inclusive, if cost[j][i] is Integer.MAX_VALUE then continue, otherwise if min == 0 || min > cost[j][i] holds (with short-circuit OR), set min = cost[j][i]. RESULT min]
            [COMMAND After the inner j loop for each processed i, if min is not 0, add min / 100 - 1 to count. RESULT count]
            [COMMAND Print the current value of count using System.out.println. RESULT printed_count]
            [COMMAND Continue the outer while loop until scan.hasNext() is false, then return from the run method. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: In the outer while loop, after reading the first integer n, n equals 0.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Break out of the outer while loop immediately. RESULT loop_exit]
            [COMMAND Return from the run method without reading m or printing a count for the zero test case. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
