[DEFINE_WORKER: "Reads an undirected graph and performs a BFS-based parity-aware edge orientation, printing -1 if the residual parity sum is nonzero and otherwise printing the oriented edges as 1-based vertex pairs." B]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integers from input using LI() and unpack them into integer variables n and m. RESULT n_m]
            [COMMAND Initialize v as an empty list; then for each index from 0 to n-1 append a new empty list to v. RESULT v]
            [COMMAND Initialize f as a list of n zeros and dic as a defaultdict whose default factory returns 0 for missing keys. RESULT f_dic]
            [COMMAND For each edge iteration from 0 to m-1, read a and b using LI(), decrement both by 1, append b to v[a] and a to v[b], increment f[a] and f[b], and set dic[(a,b)] = 1 and dic[(b,a)] = 1. RESULT graph]
            [COMMAND Initialize ans as an empty list, d as a list of n zeros, q as a deque containing vertex 0, and bfs as a list of n ones; then set bfs[0] = 0. RESULT state]
            [COMMAND While q is not empty, pop the left element x from q, initialize s as an empty list, and scan every neighbor y in v[x]: if dic[(x,y)] is truthy, append y to s; if bfs[y] is truthy, append y to q and set bfs[y] = 0. RESULT x_s]
            [COMMAND Inside the while loop, after building s for x, if s is empty take no further action for x; otherwise apply the parity-based edge orientation branch determined by whether len(s) is odd or even and by whether d[x] is truthy or falsy. RESULT orientation_applied]
            [COMMAND After the while loop, branch on whether sum(d) is nonzero or zero and output the corresponding result. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, for the current vertex x, s is non-empty, len(s) is odd, and d[x] is truthy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For every y in s, append (x,y) to ans and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT ans_consumed]
            [COMMAND Set d[x] = 0. RESULT d_cleared]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, for the current vertex x, s is non-empty, len(s) is odd, and d[x] is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For every y in s except the last element, append (x,y) to ans and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT ans_consumed]
            [COMMAND Set d[x] = 0. RESULT d_cleared]
            [COMMAND Let y be the last element of s; toggle d[y] using xor 1, append (y,x) to ans, and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT last_edge_oriented]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, for the current vertex x, s is non-empty, len(s) is even, and d[x] is falsy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For every y in s, append (x,y) to ans and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT ans_consumed]
            [COMMAND Set d[x] = 0. RESULT d_cleared]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: Inside the while loop, for the current vertex x, s is non-empty, len(s) is even, and d[x] is truthy.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For every y in s except the last element, append (x,y) to ans and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT ans_consumed]
            [COMMAND Set d[x] = 0. RESULT d_cleared]
            [COMMAND Let y be the last element of s; toggle d[y] using xor 1, append (y,x) to ans, and set dic[(x,y)] = 0 and dic[(y,x)] = 0. RESULT last_edge_oriented]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the while loop, sum(d) is nonzero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print -1. RESULT printed_invalid]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: After the while loop, sum(d) is zero.]
        [SEQUENTIAL_BLOCK]
            [COMMAND For each pair (i,j) in ans, print i+1 and j+1 on one line separated by a space. RESULT printed_edges]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
