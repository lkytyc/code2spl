[DEFINE_WORKER: "Reads counts n, a, b, and c, then processes n string tokens; for each token, if the relevant paired counts are both zero it prints No and returns, otherwise it transfers one unit between the paired counts, records the recipient character, and after all tokens prints Yes followed by the recorded characters." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to main; not used by the method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a FastScanner reading from System.in and assign it to scanner. RESULT scanner]
            [COMMAND Read the next integer token from scanner into n. RESULT n]
            [COMMAND Read the next integer token from scanner into a. RESULT a]
            [COMMAND Read the next integer token from scanner into b. RESULT b]
            [COMMAND Read the next integer token from scanner into c. RESULT c]
            [COMMAND Create an empty ArrayList of Strings named answer. RESULT answer]
            [COMMAND Start a for loop with i initialized to 0; continue while i < n, incrementing i by 1 after each iteration. RESULT loop]
            [COMMAND Inside the for loop over i, read the next token from scanner into final String s. RESULT s]
            [COMMAND Inside the for loop over i, switch on s: case "AB", case "BC", or default for any other token. RESULT switch_dispatch]
            [COMMAND In the AB case, check whether a == 0 and b == 0; if so take the AB early-return alternative, otherwise continue. RESULT ab_guard]
            [COMMAND In the AB case, if a < b, append "A" to answer, increment a, and decrement b. RESULT ab_less]
            [COMMAND In the AB case, if a >= b, append "B" to answer, decrement a, and increment b. RESULT ab_ge]
            [COMMAND After the AB case, break out of the switch. RESULT ab_break]
            [COMMAND In the BC case, check whether b == 0 and c == 0; if so take the BC early-return alternative, otherwise continue. RESULT bc_guard]
            [COMMAND In the BC case, if b < c, append "B" to answer, increment b, and decrement c. RESULT bc_less]
            [COMMAND In the BC case, if b >= c, append "C" to answer, decrement b, and increment c. RESULT bc_ge]
            [COMMAND After the BC case, break out of the switch. RESULT bc_break]
            [COMMAND In the default case, check whether a == 0 and c == 0; if so take the default early-return alternative, otherwise continue. RESULT ac_guard]
            [COMMAND In the default case, if a < c, append "A" to answer, increment a, and decrement c. RESULT ac_less]
            [COMMAND In the default case, if a >= c, append "C" to answer, decrement a, and increment c. RESULT ac_ge]
            [COMMAND After the switch finishes, continue with the next for-loop iteration. RESULT loop_continue]
            [COMMAND After the for loop completes, print "Yes" to standard output. RESULT yes_output]
            [COMMAND Call answer.forEach(System.out::println) to print each recorded choice on its own line. RESULT answer_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: In the AB case, a == 0 and b == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "No" to standard output. RESULT void]
            [COMMAND Return from the method immediately. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: In the BC case, b == 0 and c == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "No" to standard output. RESULT void]
            [COMMAND Return from the method immediately. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: In the default case, a == 0 and c == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "No" to standard output. RESULT void]
            [COMMAND Return from the method immediately. RESULT void]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
