[DEFINE_WORKER: "Reads n and n*n cell values, builds a grid adjacency graph with border cells marked, processes the queue of removed cells to propagate possible status, and prints the final count." main]
    [INPUTS]
        <REF> args </REF>: String[] "Unused command-line arguments."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new FastScanner instance named fs. RESULT fs]
            [COMMAND Read the integer n from fs. RESULT n]
            [COMMAND Initialize boolean flag to true. RESULT flag]
            [COMMAND Declare m as a two-dimensional int array of size n by n. RESULT m]
            [COMMAND Initialize x to 1. RESULT x]
            [COMMAND Initialize count to 0. RESULT count]
            [COMMAND Initialize graph as an empty ArrayList of ArrayList<Integer>. RESULT graph]
            [COMMAND Declare exist as a boolean array of size n*n. RESULT exist]
            [COMMAND Declare possible as a boolean array of size n*n. RESULT possible]
            [COMMAND Fill exist with true. RESULT exist]
            [COMMAND Fill possible with false. RESULT possible]
            [COMMAND Initialize set as an empty HashSet<Integer>. RESULT set]
            [COMMAND Initialize queue as an empty ArrayDeque<Integer>. RESULT queue]
            [COMMAND For i from 0 to n-1, begin the outer loop. RESULT i]
            [COMMAND Within the outer loop over i, for j from 0 to n-1, begin the inner loop. RESULT j]
            [COMMAND Within the nested i/j loops, create an unused local ArrayList named near. RESULT near]
            [COMMAND Within the nested i/j loops, append a new empty ArrayList to graph. RESULT graph]
            [COMMAND Within the nested i/j loops, if i is 0 or n-1 or j is 0 or n-1, add x to set. RESULT set]
            [COMMAND Within the nested i/j loops, if x - 1 > i*n, add x - 1 to graph.get(x-1). RESULT graph]
            [COMMAND Within the nested i/j loops, if x + 1 <= n*(i+1), add x + 1 to graph.get(x-1). RESULT graph]
            [COMMAND Within the nested i/j loops, if x - n > 0, add x - n to graph.get(x-1). RESULT graph]
            [COMMAND Within the nested i/j loops, if x + n <= n*n, add x + n to graph.get(x-1). RESULT graph]
            [COMMAND Within the nested i/j loops, read an integer p from fs and add p to queue. RESULT queue]
            [COMMAND Within the nested i/j loops, set m[i][j] = x. RESULT m]
            [COMMAND Within the nested i/j loops, increment x by 1. RESULT x]
            [COMMAND While queue.size() > 0, poll an integer a from queue and set exist[a-1] to false. RESULT a, exist]
            [COMMAND In the same while iteration, if set contains a, set possible[a-1] to true. RESULT possible]
            [COMMAND In the same while iteration, if possible[a-1] is true, then for each integer w in graph.get(a-1), set possible[w-1] to true. RESULT possible]
            [COMMAND In the same while iteration, if possible[a-1] is false, then for each integer w in graph.get(a-1), if possible[w-1] is true and exist[w-1] is false, set possible[a-2], possible[a-1-n], possible[a], possible[a-1+n], and possible[a-1] to true, decrement count by 1, and break out of the neighbor loop. RESULT possible, count]
            [COMMAND In the same while iteration, after the neighbor scan that ran when possible[a-1] was false, increment count by 1. RESULT count]
            [COMMAND In the same while iteration, after incrementing count, if at least one of a-2 >= 0, a-1-n >= 0, a <= n*n, or a-1+n <= n*n is true, and at least one of the four paired conditions (exist[a-2] is false and possible[a-2] is true), (exist[a-1-n] is false and possible[a-1-n] is true), (exist[a] is false and possible[a] is true), or (exist[a-1+n] is false and possible[a-1+n] is true) is true, set possible[a-2], possible[a-1-n], possible[a], possible[a-1+n], and possible[a-1] to true. RESULT possible]
            [COMMAND Print the value of count to standard output using System.out.print. RESULT count]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
