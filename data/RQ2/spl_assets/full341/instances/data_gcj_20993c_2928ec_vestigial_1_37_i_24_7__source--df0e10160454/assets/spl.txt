[DEFINE_WORKER: "Computes and returns a list of three integers: the sum of the diagonal entries, the number of rows containing at least one duplicate, and the number of columns containing at least one duplicate, for a square matrix-like input pinax." vestigial]
    [INPUTS]
        <REF> pinax </REF>: list of lists (matrix-like) "The input matrix-like structure; its square size is derived from len(pinax[0])."
    [END_INPUTS]

    [OUTPUTS]
        <REF> result </REF>: list of three integers "Returns [trail, same_row_counter, sumC], where trail is the diagonal sum, same_row_counter is the count of rows with at least one duplicate, and sumC is the count of columns with at least one duplicate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Determine size as len(pinax[0]) and initialize result, row_digits, and column_digits as empty lists. RESULT initial_state]
            [COMMAND For i from 0 through size-1, append a new empty list to column_digits so that column_digits has one distinct-value list per column. RESULT column_digits]
            [COMMAND Set same_row_counter to 0, same_column_counter to a list of size zeros, and trail to 0. RESULT counters]
            [COMMAND For each row from 0 through size-1: clear row_digits by deleting all its elements at the start of the row. For each column from 0 through size-1 in that row: if row equals column, add pinax[row][row] to trail; if len(row_digits) equals column, scan i from 0 through len(row_digits); when i equals len(row_digits), append pinax[row][column] to row_digits and break, and when row_digits[i] equals pinax[row][column], increment same_row_counter and break; then scan i from 0 through len(column_digits[column]); when i equals len(column_digits[column]), append pinax[row][column] to column_digits[column] and break, and when column_digits[column][i] equals pinax[row][column], set same_column_counter[column] to 1 and break. Because a found row duplicate prevents an append, the row-duplicate check is skipped for all later columns in that row. RESULT processed_state]
            [COMMAND Set sumC to 0, then for each i from 0 through size-1 add same_column_counter[i] to sumC. RESULT sumC]
            [COMMAND Append trail, same_row_counter, and sumC to result in that order, then return result. RESULT result]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
