[DEFINE_WORKER: "Computes the trace of a square matrix, transposes it, sorts the rows of both matrices, and prints a formatted Case # output using zero-adjusted row and column duplicate-pair counts." vestigium]
    [INPUTS]
        <REF> a </REF>: int[][] "Square integer matrix of size n x n; its rows are sorted in place during processing."
        <REF> n </REF>: int "Dimension of the square matrix a."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize local variables k, r, and c to 0, and allocate a new n x n integer matrix b with all elements defaulting to 0. RESULT initial_variables]
            [COMMAND For i from 0 to n-1 inclusive, add the diagonal element a[i][i] to k to compute the trace. RESULT k]
            [COMMAND For i from 0 to n-1 inclusive and j from 0 to n-1 inclusive, assign b[i][j] = a[j][i] to transpose matrix a into matrix b. RESULT b]
            [COMMAND For each row i from 0 to n-1 inclusive, sort a[i] in ascending order using Arrays.sort. RESULT a_sorted]
            [COMMAND Initialize repeat to 0 and count to 0 for counting horizontal duplicate pairs in sorted rows of a. RESULT repeat_count_initialized]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive: if a[i][j] equals a[i][j+1], increment repeat; then set count to Math.max(count, repeat). After the inner loop completes for the row, if repeat is greater than 0 reset repeat to 0. RESULT count]
            [COMMAND For each row i from 0 to n-1 inclusive, sort b[i] in ascending order using Arrays.sort. RESULT b_sorted]
            [COMMAND Initialize count1 to 0 and repeat1 to 0 for counting horizontal duplicate pairs in sorted rows of b. RESULT repeat1_count1_initialized]
            [COMMAND For each row i from 0 to n-1 inclusive, for each j from 0 to n-2 inclusive: if b[i][j] equals b[i][j+1], increment repeat1; then set count1 to Math.max(count1, repeat1). After the inner loop completes for the row, if repeat1 is greater than 0 reset repeat1 to 0. RESULT count1]
            [COMMAND Assign r = count + 1, c = count1 + 1, and counter = 1. RESULT r_c_counter]
            [COMMAND Evaluate the zero/non-zero status of count1 and count and execute exactly the matching output branch from alternative_flows, printing the formatted line and incrementing counter by 1. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 and count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + counter + ": " + " " + k + " " + count + " " + count1, then increment counter by 1. RESULT output_both_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 and count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + counter + ": " + " " + k + " " + r + " " + c, then increment counter by 1. RESULT output_both_nonzero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 == 0 and count != 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + counter + ": " + " " + k + " " + r + " " + count1, then increment counter by 1. RESULT output_count1_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: count1 != 0 and count == 0]
        [SEQUENTIAL_BLOCK]
            [COMMAND Print "Case #" + counter + ": " + " " + k + " " + count + " " + c, then increment counter by 1. RESULT output_count_zero]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
