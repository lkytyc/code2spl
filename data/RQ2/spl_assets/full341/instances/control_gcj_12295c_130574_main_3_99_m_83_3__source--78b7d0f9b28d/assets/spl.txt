[DEFINE_WORKER: "Reads test cases describing people with starting positions and directions, scores every grid cell by how many people pass through it, and prints the best cell using tie-breaking by smallest row+col, then smallest x, then smallest y." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments passed to main; unused by the method logic."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize a Scanner reading from System.in and read the next integer as the number of test cases t. RESULT t]
            [COMMAND For each test case index i from 1 through t inclusive, execute the following per-test-case processing. RESULT i]
            [COMMAND Within the outer for loop over i, read p as the number of people and q as the grid dimension from the Scanner. RESULT p_q]
            [COMMAND Within the outer loop, create an empty ArrayList ppl with initial capacity p to hold Person objects. RESULT ppl]
            [COMMAND Within the outer loop, for each j from 0 through p-1, read the raw y coordinate b and the raw x coordinate rawX for one person, compute transformed x = q - rawX - 1, and read the direction token c. RESULT person_input]
            [COMMAND Within the same per-person loop, convert direction token c into horizontal direction d and vertical direction e: N sets e=-1, S sets e=1, E sets d=1, and any other token sets d=-1. RESULT direction_components]
            [COMMAND Within the same per-person loop, construct and append Person(transformed_x, b, d, e) to ppl. RESULT ppl]
            [COMMAND Within the outer loop after all p people have been read, create a q by q integer grid b initialized to all zeros. RESULT grid_b]
            [COMMAND Within the outer loop, for each Person a in ppl, copy a.x to x and a.y to y, then update grid b according to a's direction: if a.xDir is 0, iterate rows from x+a.yDir by step a.yDir while the row index is between 0 and q-1 inclusive and increment every column in those rows; otherwise iterate columns from y+a.xDir by step a.xDir while the column index is between 0 and q-1 inclusive and increment every row in those columns. RESULT grid_b]
            [COMMAND Within the outer loop, scan every cell of grid b and compute the maximum value max, initialized to 0. RESULT max]
            [COMMAND Within the outer loop, allocate integer arrays row and col of length q*q, set num to 0, then scan columns c from 0 through q-1 and rows r from 0 through q-1; when b[r][c] equals max, store row[num] = q - r - 1, col[num] = c, and increment num. RESULT candidate_arrays]
            [COMMAND Within the outer loop, initialize min to Integer.MAX_VALUE and create an empty ArrayList f; for each stored candidate index j from 0 through num-1, if row[j]+col[j] is less than min then update min, clear f, and add Person(row[j], col[j], 0, 0); else if row[j]+col[j] equals min, append Person(row[j], col[j], 0, 0). RESULT min_sum_candidates]
            [COMMAND Within the outer loop, create an empty ArrayList g and set m to Integer.MAX_VALUE; for each Person in f, if that person's x is less than m then update m, clear g, and add the person; else if x equals m, append the person. RESULT min_x_candidates]
            [COMMAND Within the outer loop, set z to g.get(0); then for each Person in g from index 1 onward, replace z with the current person if its y is smaller than z.y. RESULT z]
            [COMMAND Within the outer loop, after z is selected, print the case result in the format Case #<i>: <z.y> <z.x>. RESULT console_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
