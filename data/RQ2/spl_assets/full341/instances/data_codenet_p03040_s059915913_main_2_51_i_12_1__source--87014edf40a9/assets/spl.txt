[DEFINE_WORKER: "Processes q online operations by maintaining a sorted array a and accumulator b; type-1 operations insert y and add z to b, while type-2 operations print the lower median m of a and b plus the sum of absolute deviations from m." main]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: int "The return code of main; always 0 after successful execution."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Declare int q. RESULT q]
            [COMMAND Read q from standard input using scanf("%d", &q). RESULT q]
            [COMMAND Declare int n = 0, long long int a[200005], b = 0; declare long long int x, y, z, m, ans; declare int i, j. RESULT declarations]
            [COMMAND For i starting at 0 while i < q, incrementing i by 1 after each iteration, read x from standard input using scanf("%lld", &x). If x equals 1, perform the insertion flow; otherwise, perform the query flow. RESULT loop_operation]
            [COMMAND After the loop over i finishes, return 0 from main. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: x equals 1]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read y and z from standard input using scanf("%lld %lld", &y, &z). RESULT y,z]
            [COMMAND If n equals 0, set a[0] = y. RESULT a]
            [COMMAND If n is greater than 0, run j from n-1 down to 0; when a[j] > y, shift a[j] to a[j+1] and, if j == 0, set a[0] = y; when a[j] <= y, set a[j+1] = y and break the loop. RESULT a]
            [COMMAND Increment n by 1. RESULT n]
            [COMMAND Add z to b. RESULT b]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: x does not equal 1, so the operation is a query]
        [SEQUENTIAL_BLOCK]
            [COMMAND Set m = a[(n - 1) / 2] as the lower median of the stored values. RESULT m]
            [COMMAND Set ans = b. RESULT ans]
            [COMMAND For j from 0 to n-1, add the absolute difference between m and a[j] to ans: if m - a[j] is less than 0, add -(m - a[j]); otherwise add m - a[j]. RESULT ans]
            [COMMAND Print m and ans on one line using printf("%lld %lld\n", m, ans). RESULT stdout]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
