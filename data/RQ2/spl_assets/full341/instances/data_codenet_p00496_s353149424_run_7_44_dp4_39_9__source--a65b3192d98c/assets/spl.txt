[DEFINE_WORKER: "Reads items with value and time, computes a prefix-value DP for capacity s and a suffix-value DP for capacity t-s via 0/1 knapsack relaxations, then prints the maximum sum over boundary i." run]
    [INPUTS]
        <REF> stdin </REF>: Scanner "System.in supplying n, t, s, followed by n pairs of integers as[i] and bs[i]."
    [END_INPUTS]

    [OUTPUTS]
        <REF> stdout </REF>: int "The computed maximum total value printed by System.out.println."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize constants INF to 1 << 28 (268435456) and EPS to 1e-10. RESULT constants]
            [COMMAND Create a Scanner on System.in and read integers n, t, s from it. RESULT input_bounds]
            [COMMAND Allocate integer arrays as and bs of length n; for i from 0 to n-1, read as[i] and bs[i] from the scanner. RESULT items]
            [COMMAND Allocate dp1 of length s+1 and dp2 of length n+1; fill dp1 with -1, set dp1[0] to 0, and leave dp2 initialized to zeros. RESULT dp1_dp2]
            [COMMAND For each item i from 0 to n-1, run a descending 0/1 knapsack over j from s down to 0; if dp1[j] >= 0 and bs[i] + j <= s and dp1[bs[i] + j] < dp1[j] + as[i], set dp1[bs[i] + j] = dp1[j] + as[i] and update dp2[i+1] to max(dp1[bs[i] + j], dp2[i+1]). RESULT dp1_knapsack]
            [COMMAND For i from 1 to n inclusive, set dp2[i] = max(dp2[i], dp2[i-1]), producing prefix maxima so dp2[i] is the best value achievable from any subset of the first i items within capacity s. RESULT dp2_prefix_max]
            [COMMAND Allocate dp3 of length t-s+1 and dp4 of length n+1; fill dp3 with -1, set dp3[0] to 0, and leave dp4 initialized to zeros. RESULT dp3_dp4]
            [COMMAND For each item i from n-1 down to 0, run a descending 0/1 knapsack over j from t-s down to 0; if dp3[j] >= 0 and bs[i] + j <= t-s and dp3[bs[i] + j] < dp3[j] + as[i], set dp3[bs[i] + j] = dp3[j] + as[i] and update dp4[n-i] to max(dp3[bs[i] + j], dp4[n-i]). RESULT dp3_knapsack]
            [COMMAND For i from 1 to n inclusive, set dp4[i] = max(dp4[i], dp4[i-1]), producing prefix maxima so dp4[i] is the best value achievable from any subset of the last i items within capacity t-s. RESULT dp4_prefix_max]
            [COMMAND Initialize integer variable max to 0. RESULT max]
            [COMMAND For i from 0 to n-1, compute candidate dp2[i] + dp4[n-i] and update max to max(max, candidate). RESULT answer_max]
            [COMMAND Print the final max value to System.out. RESULT printed_output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
