[DEFINE_WORKER: "Reads N points, groups directed point pairs by normalized direction vector, and computes the maximum combined even/odd chain length over those direction groups, printing the case result." do_one_case]
    [INPUTS]
        <REF> cnum </REF>: int "Case number used in the output line."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read integer N from rd1int() and initialize XY as an empty list. RESULT N, XY]
            [COMMAND For i from 0 through N-1 inclusive, read two integers x and y from rdints(), append the tuple (x,y) to XY. RESULT XY]
            [COMMAND Sort XY lexicographically. RESULT XY]
            [COMMAND Initialize A as an empty dictionary. RESULT A]
            [COMMAND For each j from 1 through N-1 inclusive, let x2,y2 = XY[j]; then for each i from 0 through j-1 inclusive, let x1,y1 = XY[i]; compute a = x2-x1, b = y2-y1, and g = gcd(a,b); divide a and b by g; if (a,b) is already a key in A, append (i,j) to A[(a,b)] and append (j,i) to A[(-a,-b)]; otherwise set A[(a,b)] = [(i,j)] and A[(-a,-b)] = [(j,i)]. RESULT A]
            [COMMAND Initialize m to 1. RESULT m]
            [COMMAND For each direction key (a,b) in A, set E = A[(a,b)], initialize I as an empty dictionary and O as an empty dictionary. RESULT E, I, O]
            [COMMAND Within that same direction iteration, for every directed edge (i,j) in E, append j to the outgoing list O[i] (creating O[i] = [j] if needed) and append i to the incoming list I[j] (creating I[j] = [i] if needed). RESULT I, O]
            [COMMAND Within that same direction iteration, compute NI = set(range(N)) - set(I); initialize C as an empty list; for each source node x in NI, if x has outgoing edges in O, build Q = [(0,x)] + [(len(I[y]), y) for y in O[x]], sort Q by its first then second tuple element, and append the node sequence [y for (priority, y) in Q] to C; otherwise append the singleton [x] to C. RESULT NI, C]
            [COMMAND Within that same direction iteration, partition C into C1 = groups of length 1, C2 = groups of even length, and C3 = groups of odd length greater than 1; while len(C3) > 1, concatenate the last two groups of C3 into one group, append that combined group to C2, and remove those two groups from C3. RESULT C1, C2, C3]
            [COMMAND Within that same direction iteration, set C3 = (C3 + C1)[:2], compute l2 as the sum of lengths of all groups in C2 and l3 as the sum of lengths of all groups in C3; if l3 is nonzero, set l = l2 + l3; otherwise if C2 has more than one group, set l = l2 - 1; otherwise set l = l2. RESULT l]
            [COMMAND Within that same direction iteration, update m = max(m, l). RESULT m]
            [COMMAND Print the formatted string 'Case #%d: %d' with cnum and m as the two integer arguments. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
