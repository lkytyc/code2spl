[DEFINE_WORKER: "Reads two strings, compares length-difference-aligned substring blocks, prints -1 if the first non-breaking scan finds all checked blocks equal, and otherwise prints the maximum of the original scan count and the prefix-plus-suffix consecutive match count; the equal-length branch uses char-array reference equality." solve]
    [INPUTS]
        <REF> testNumber </REF>: int "Test case number; not used in the method body."
        <REF> br </REF>: FastScanner "Input scanner used to read the two strings."
        <REF> pw </REF>: PrintWriter "Output writer used to print numeric results."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the first input string into in1 using br.nextString(). RESULT in1]
            [COMMAND Read the second input string into in2 using br.nextString(). RESULT in2]
            [COMMAND Convert in1 to a char array named t using in1.toCharArray(). RESULT t]
            [COMMAND Convert in2 to a char array named s using in2.toCharArray(). RESULT s]
            [COMMAND Compare t.length with s.length to select the corresponding length-based branch. RESULT length_relation]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: t.length > s.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig to 0. RESULT orig]
            [COMMAND Initialize poss to true. RESULT poss]
            [COMMAND For i = 0 while i < t.length - s.length, incrementing by s.length, compare in1.substring(i, i + s.length) with in2; if they are not equal set poss = false, otherwise increment orig; this loop scans the full range without breaking. RESULT orig_and_poss]
            [COMMAND If poss is true, print -1 to pw and close pw; there is no return, so execution continues with the next steps. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For i = 0 while i < t.length - s.length, incrementing by s.length, check whether in1.substring(i, i + s.length) equals in2; if so increment ans, otherwise break. RESULT ans]
            [COMMAND For i = t.length - s.length while i >= 0, decrementing by s.length, check whether in1.substring(i, i + s.length) equals in2; if so increment ans, otherwise break. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: t.length == s.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND If s.equals(t) evaluates to true using reference equality for these char arrays, print -1 to pw; otherwise print 0 to pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

    [ALTERNATIVE_FLOW: s.length > t.length]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize orig to 0. RESULT orig]
            [COMMAND Initialize poss to true. RESULT poss]
            [COMMAND For i = 0 while i < s.length - t.length, incrementing by s.length, compare in2.substring(i, i + t.length) with in1; if they are not equal set poss = false; matches do not increment orig in this loop. RESULT poss]
            [COMMAND If poss is true, print -1 to pw and close pw; there is no return, so execution continues with the next steps. RESULT pw]
            [COMMAND Initialize ans to 0. RESULT ans]
            [COMMAND For i = 0 while i < s.length - t.length, incrementing by t.length, check whether in2.substring(i, i + t.length) equals in1; if so increment ans, otherwise break. RESULT ans]
            [COMMAND For i = s.length - t.length while i >= 0, decrementing by t.length, check whether in2.substring(i, i + t.length) equals in1; if so increment ans, otherwise break. RESULT ans]
            [COMMAND Print Math.max(orig, ans) to pw and close pw. RESULT pw]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
