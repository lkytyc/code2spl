[DEFINE_WORKER: "Reads n and m, builds a Fibonacci-like modulo 998244353 prefix-sum array of length m, repeatedly applies the same prefix-sum transformation n-1 times, and prints the final last element; returns immediately if m > 200000." run]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a new MyScanner instance. RESULT sc]
            [COMMAND Read the next integer from sc into n. RESULT n]
            [COMMAND Read the next integer from sc into m. RESULT m]
            [COMMAND Set M to the constant 998244353. RESULT M]
            [COMMAND Check whether m > 200000; if it is, follow the alternative early-return flow. RESULT none]
            [COMMAND Allocate dp as an int[2][m] array and sum as an int[m] array. RESULT dp, sum]
            [COMMAND Initialize dp[0][0] to 1, dp[0][1] to 1, sum[0] to 1, and sum[1] to 2. RESULT dp, sum]
            [COMMAND For i from 2 to m-1 inclusive, compute dp[0][i] = (dp[0][i-1] + dp[0][i-2]) % M and sum[i] = (sum[i-1] + dp[0][i]) % M. RESULT dp, sum]
            [COMMAND Set last to dp[0][m-1]. RESULT last]
            [COMMAND For each outer iteration i from 1 to n-1 inclusive, allocate next as an int[m] array, set next[0] = sum[0], for j from 1 to m-1 compute next[j] = (next[j-1] + sum[j]) % M, set last = sum[m-1], then replace sum with next. RESULT last, sum]
            [COMMAND Print last to standard output. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: m > 200000]
        [SEQUENTIAL_BLOCK]
            [COMMAND Return from run() immediately without allocating arrays or printing any result. RESULT none]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
