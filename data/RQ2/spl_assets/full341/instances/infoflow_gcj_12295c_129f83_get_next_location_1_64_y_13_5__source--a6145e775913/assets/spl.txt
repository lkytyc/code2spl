[DEFINE_WORKER: "Reads num_of_people starting positions and movement directions from standard input, moves each position one unit in the specified cardinal direction, counts how many times each resulting coordinate occurs, and returns the coordinate with the highest count, using the lexicographically smallest coordinate to break ties." get_next_location]
    [INPUTS]
        <REF> num_of_people </REF>: int "Number of people/location entries to read. The loop reads exactly num_of_people lines because people_count starts at 1 and the inclusive while condition is people_count <= num_of_people. The code assumes num_of_people is at least 1."
    [END_INPUTS]

    [OUTPUTS]
        <REF> return_value </REF>: list of two ints "The [x, y] coordinate of the most frequent next destination; if multiple destinations tie for the highest frequency, returns the lexicographically smallest coordinate."
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize coordinates_list to an empty list and people_count to 1. RESULT coordinates_list, people_count]
            [COMMAND While people_count <= num_of_people, for each person: read a line from standard input with raw_input(""), split it into tokens, parse x as int(curr_location[0]), y as int(curr_location[1]), and dir as curr_location[2]; if dir is "N", increment y by 1; if dir is "S", decrement y by 1; if dir is "E", increment x by 1; if dir is "W", decrement x by 1; if dir is any other value, leave x and y unchanged; append [x, y] to coordinates_list; increment people_count by 1. RESULT coordinates_list, people_count]
            [COMMAND Initialize next_dest to an empty list and next_dest_count to an empty list. RESULT next_dest, next_dest_count]
            [COMMAND For each coordinates in coordinates_list, set found to 0 and dest_idx to 0; search next_dest for an element equal to coordinates; if coordinates equals dest, increment next_dest_count[dest_idx], set found to 1, and break out of the inner loop; after the inner loop, if found is 0, append coordinates to next_dest and append 1 to next_dest_count. RESULT next_dest, next_dest_count]
            [COMMAND Initialize popular_next_dest as an empty list, append next_dest[0] to it, set highest_count to next_dest_count[0], and set dest_count_idx to 0. RESULT popular_next_dest, highest_count, dest_count_idx]
            [COMMAND For each dest_count in next_dest_count: if dest_count_idx is 0, increment dest_count_idx and continue; otherwise, if dest_count equals highest_count, append next_dest[dest_count_idx] to popular_next_dest; if dest_count is greater than highest_count, set popular_next_dest to a new empty list, set highest_count to dest_count, and append next_dest[dest_count_idx] to popular_next_dest; after the equality and greater-than checks, increment dest_count_idx. RESULT popular_next_dest, highest_count]
            [COMMAND If len(popular_next_dest) == 1, return popular_next_dest[0] immediately. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

    [ALTERNATIVE_FLOW: When len(popular_next_dest) is greater than 1, meaning more than one destination has the highest count.]
        [SEQUENTIAL_BLOCK]
            [COMMAND Initialize smallest_number to popular_next_dest[0] and dest_idx to 0. RESULT smallest_number, dest_idx]
            [COMMAND For each dest in popular_next_dest: if dest_idx is 0, increment dest_idx and continue; otherwise, if dest < smallest_number, set smallest_number to dest; increment dest_idx after the comparison. RESULT smallest_number]
            [COMMAND Return smallest_number. RESULT return_value]
        [END_SEQUENTIAL_BLOCK]
    [END_ALTERNATIVE_FLOW]

[END_WORKER]
