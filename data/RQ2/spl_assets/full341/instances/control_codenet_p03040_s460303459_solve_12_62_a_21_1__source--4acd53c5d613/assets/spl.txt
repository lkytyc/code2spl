[DEFINE_WORKER: "Processes update/query commands by maintaining two balanced heaps that split inserted values into lower and upper halves, tracking their sums and an extra bias B, and prints a median-derived adjusted sum for each non-insertion command." solve]
    [INPUTS]
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Read the number of operations Q from ir.nextInt(). RESULT Q]
            [COMMAND Initialize f as an empty max-heap PriorityQueue<Long> using Collections.reverseOrder(), and l as an empty min-heap PriorityQueue<Long>. RESULT f, l]
            [COMMAND Set fsum, lsum, and B all to 0L. RESULT fsum, lsum, B]
            [COMMAND While the expression Q-- is greater than 0, using the current Q then decrementing Q, read the next command integer com from ir.nextInt(). RESULT com]
            [COMMAND Within the while loop, if com equals 1, read a long value a from ir.nextLong() for the insertion command. RESULT a]
            [COMMAND Within the insertion command, if f.size() equals l.size() and l.size() equals 0, add a to f and add a to fsum. RESULT f, fsum]
            [COMMAND Within the insertion command, balanced case with l non-empty, peek x = l.peek(); if x is strictly less than a, update lsum by adding a - x, update fsum by adding x, poll l, add a to l, and add x to f. RESULT l, f, lsum, fsum]
            [COMMAND Within the insertion command, balanced case with l non-empty, if x is not strictly less than a, add a to f and add a to fsum. RESULT f, fsum]
            [COMMAND Within the insertion command, if f.size() is not equal to l.size(), peek x = f.peek(); if x is less than or equal to a, add a to l and add a to lsum. RESULT l, lsum]
            [COMMAND Within the insertion command, unbalanced case, if x is greater than a, update lsum by adding x, update fsum by adding a - x, poll f, add a to f, and add x to l. RESULT f, l, fsum, lsum]
            [COMMAND After the insertion branching completes, read one additional long from ir and add it to B. RESULT B]
            [COMMAND Within the while loop, if com is not equal to 1, peek x = f.peek() for the query command. RESULT x]
            [COMMAND Within the query command, if f.size() equals l.size(), print x, a space, and the value (lsum - fsum + B) to out using println. RESULT out]
            [COMMAND Within the query command, if f.size() is not equal to l.size(), print x, a space, and the value (lsum - fsum + x + B) to out using println. RESULT out]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]
