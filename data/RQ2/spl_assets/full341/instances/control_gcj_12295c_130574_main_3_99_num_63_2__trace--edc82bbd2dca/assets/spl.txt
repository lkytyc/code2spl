[DEFINE_WORKER: "Reads test cases describing people and their walking directions on a QxQ grid, counts how many people's paths pass through each grid cell, and chooses the cell with the maximum count subject to the specified tie-breaks, printing one result per test case." main]
    [INPUTS]
        <REF> args </REF>: String[] "Command-line arguments; not used by this method."
    [END_INPUTS]

    [OUTPUTS]
    [END_OUTPUTS]

    [MAIN_FLOW]
        [SEQUENTIAL_BLOCK]
            [COMMAND Create a Scanner sc bound to System.in. RESULT sc]
            [COMMAND Read the next integer from sc into t, the number of test cases. RESULT t]
            [COMMAND For each test case index i from 1 to t inclusive, read the integer p (number of people) and integer q (grid size) from sc. RESULT p_q]
            [COMMAND Initialize ppl as an empty ArrayList of Person with initial capacity p for the current test case. RESULT ppl]
            [COMMAND For each person index j from 0 to p-1 in the current test case, read integer b as the y-coordinate, read integer raw_x as the raw x-coordinate, compute a = q - raw_x - 1 as the x-coordinate, read direction string c, map c to horizontal/vertical deltas: 'N' -> d=0,e=-1; 'S' -> d=0,e=1; 'E' -> d=1,e=0; any other value (including 'W') -> d=-1,e=0, create new Person(a, b, d, e), and add it to ppl. RESULT ppl]
            [COMMAND Initialize a q by q integer grid grid_b with all cells set to 0 for the current test case. RESULT grid_b]
            [COMMAND For each Person a in ppl in the current test case, extract x = a.x and y = a.y; if a.xDir == 0, increment every cell in each column for rows from x + a.yDir while the row index is between 0 and q-1 inclusive, moving by a.yDir; otherwise increment every cell in each row for columns from y + a.xDir while the column index is between 0 and q-1 inclusive, moving by a.xDir. RESULT grid_b]
            [COMMAND For the current test case, scan all q*q cells of grid_b to compute max, the maximum cell value; initialize max to 0 before scanning. RESULT max]
            [COMMAND For the current test case, initialize row and col arrays of length q*q and num to 0; scan grid_b in column-major order with c from 0 to q-1 and r from 0 to q-1; whenever grid_b[r][c] == max, set row[num] = q - r - 1, set col[num] = c, and increment num. RESULT row_col_arrays]
            [COMMAND For the current test case, initialize min to Integer.MAX_VALUE and f to an empty ArrayList of Person with capacity q*q; for each candidate index j from 0 to num-1, if row[j] + col[j] < min, set min to row[j] + col[j], clear f, and add new Person(row[j], col[j], 0, 0); if row[j] + col[j] equals min, add that candidate to f. RESULT f]
            [COMMAND For the current test case, initialize m to Integer.MAX_VALUE and g to an empty ArrayList of Person with capacity f.size(); for each candidate in f, if candidate.x < m, set m to candidate.x, clear g, and add candidate; if candidate.x equals m, add candidate to g. RESULT g]
            [COMMAND For the current test case, initialize z to g.get(0); for each remaining candidate in g from index 1 to g.size()-1, if candidate.y < z.y, set z to candidate. RESULT z]
            [COMMAND Print to standard output the current test case result with the format 'Case #' + i + ': ' + z.y + ' ' + z.x. RESULT output]
        [END_SEQUENTIAL_BLOCK]
    [END_MAIN_FLOW]

[END_WORKER]
