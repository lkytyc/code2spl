1   import java.io.IOException;
2   import java.io.InputStream;
3   import java.util.NoSuchElementException;
4   public class Main {
5       private static FastScanner sc = new FastScanner();
6       private static boolean DEBUG_FLG = false;
7       public static void main(String[] args) {
8       	int N = sc.nextInt();
9       	long A = sc.nextLong();
10      	long B = sc.nextLong();
11      	int[] p = new int[N];
12      	for(int i=0; i<N; i++) {
13      		p[i] = sc.nextInt();
14      	}
15      	long[][] dpc = new long[N][N];
16      	int[][] dpmin = new int[N][N];
17      	int[][] dpmin2 = new int[N][N];
18      	int[][] dpmax = new int[N][N];
19      	int[][] dpmax2 = new int[N][N];
20      	for(int i=0; i<N; i++) {
21      		dpc[i][i] = 0;
22      		dpmin[i][i] = p[i];
23      		dpmax[i][i] = p[i];
24      		dpmin2[i][i] = Integer.MAX_VALUE;
25      		dpmax2[i][i] = -1;
26      	}
27      	for(int d=1; d<N; d++) {
28      		for(int i=0; i<N-d; i++) {
29      			if(p[i+d] < dpmin[i][i+d-1]) {
30      				dpmin2[i][i+d] = dpmin[i][i+d-1];
31      				dpmin[i][i+d] = p[i+d];
32      			} else if(p[i+d] < dpmin2[i][i+d-1]) {
33      				dpmin2[i][i+d] = p[i+d];
34      				dpmin[i][i+d] = dpmin[i][i+d-1];
35      			} else {
36      				dpmin2[i][i+d] = dpmin2[i][i+d-1];
37      				dpmin[i][i+d] = dpmin[i][i+d-1];
38      			}
39      			if(p[i+d] > dpmax[i][i+d-1]) {
40      				dpmax2[i][i+d] = dpmax[i][i+d-1];
41      				dpmax[i][i+d] = p[i+d];
42      			} else if(p[i+d] > dpmax2[i][i+d-1]) {
43      				dpmax2[i][i+d] = p[i+d];
44      				dpmax[i][i+d] = dpmax[i][i+d-1];
45      			} else {
46      				dpmax2[i][i+d] = dpmax2[i][i+d-1];
47      				dpmax[i][i+d] = dpmax[i][i+d-1];
48      			}
49      			long rightcost = 0;
50      			if(p[i+d] > dpmax[i][i+d-1]) {
51      				rightcost = dpc[i][i+d-1];
52      			} else if(p[i+d] >  dpmax2[i][i+d-1]){
53      				rightcost = dpc[i][i+d-1] + Math.min(A, B);
54      			} else {
55      				rightcost = dpc[i][i+d-1] + B;
56      			}
57      			long leftcost = 0;
58      			if(p[i] < dpmin[i+1][i+d]) {
59      				leftcost = dpc[i+1][i+d];
60      			} else if(p[i] < dpmin2[i+1][i+d]){
61      				leftcost = dpc[i+1][i+d] + Math.min(A, B);
62      			} else {
63      				leftcost = dpc[i+1][i+d] + A;
64      			}
65      			dpc[i][i+d] = Math.min(leftcost, rightcost);
66      			debug(i, i+d, leftcost, rightcost, dpmin[i][i+d], dpmin2[i][i+d], dpmax[i][i+d], dpmax2[i][i+d]);
67      		}
68      	}
69      	System.out.println(dpc[0][N-1]);
70      }
71      static void debug(long... args) {
72      	if(!DEBUG_FLG) return;
73      	boolean flg = false;
74      	for(long s : args) {
75      		if(flg) System.out.print(" ");
76      		flg = true;
77      		System.out.print(s);
78      	}
79      	System.out.println();
80      }
81      static class FastScanner {
82          private final InputStream in = System.in;
83          private final byte[] buffer = new byte[1024];
84          private int ptr = 0;
85          private int buflen = 0;
86          private boolean hasNextByte() {
87              if(ptr < buflen) {
88                  return true;
89              } else {
90                  ptr = 0;
91                  try {
92                      buflen = in.read(buffer);
93                  } catch(IOException e) {
94                      e.printStackTrace();
95                  }
96                  if(buflen <= 0) {
97                      return false;
98                  }
99              }
100              return true;
101          }
102          private int readByte() { if (hasNextByte()) return buffer[ptr++]; else return -1;}
103          private static boolean isPrintableChar(int c) { return 33 <= c && c <= 126;}
104          private void skipUnprintable() { while(hasNextByte() && !isPrintableChar(buffer[ptr])) ptr++;}
105          public boolean hasNext() { skipUnprintable(); return hasNextByte();}
106          public String next() {
107              if (!hasNext()) throw new NoSuchElementException();
108              StringBuilder sb = new StringBuilder();
109              int b = readByte();
110              while(isPrintableChar(b)) {
111                  sb.appendCodePoint(b);
112                  b = readByte();
113              }
114              return sb.toString();
115          }
116          public long nextLong() {
117              return Long.parseLong(next());
118          }
119          public int nextInt(){
120              return Integer.parseInt(next());
121          }
122          public double nextDouble(){
123              return Double.parseDouble(next());
124          }
125      }
