1   import java.io.BufferedReader;
2   import java.io.InputStreamReader;
3   import java.util.Scanner;
4   public class Solution {
5       private static long guessNumber(long lowerBound, long upperBound) {
6           if (lowerBound == upperBound) return lowerBound;
7           return (upperBound - lowerBound) / 2 + lowerBound;
8       }
9       public static void main(String[] args) {
10          private static final String TOO_BIG = "TOO_BIG";
11          private static final String TOO_SMALL = "TOO_SMALL";
12          private static final String CORRECT = "CORRECT";
13          private static final String WRONG_ANSWER = "WRONG_ANSWER";
14          Scanner in = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
15          int t = Integer.parseInt(in.nextLine());
16          for (int i = 1; i <= t; ++i) {
17              String[] splits = in.nextLine().split(" ");
18              long a = Long.parseLong(splits[0]) + 1;
19              long b = Long.parseLong(splits[1]);
20              int n = Integer.parseInt(in.nextLine());
21              while (n-- > 0) {
22                  long cur;
23                  if (a == b)
24                      cur = a;
25                  else
26                      cur = (b - a) / 2 + a;
27                  System.out.println(cur);
28                  String result = in.nextLine().trim();
29                  if (TOO_BIG.equals(result)) {
30                      b = cur - 1;
31                  } else if (TOO_SMALL.equals(result)) {
32                      a = cur + 1;
33                  } else if (CORRECT.equals(result)) {
34                      break;
35                  } else {
36                      return;
37                  }
38              }
39          }
40      }
41  }
