1   #include <stdio.h>
2   #include <stdlib.h>
3   typedef struct st
4   {
5       int x;
6       int y;
7   }arr;
8   arr array[100005];
9   int comp(const void *a, const void *b)
10  {
11      long long int c1 = (*(arr*)a).x;
12      c1 *= (*(arr*)b).y;
13      long long int c2 = (*(arr*)b).x;
14      c2 *= (*(arr*)a).y;
15      if (c1 == c2) return 0;
16      else if (c1 > c2) return 1;
17      else return -1;
18  }
19  int abs(int n)
20  {
21      if (n < 0)
22      {
23          return -1 * n;
24      }
25      return n;
26  }
27  int main()
28  {
29      int input[310][2];
30      int cases, N;
31      scanf("%d", &cases);
32      for (int cas = 1; cas <= cases; cas++)
33      {
34          scanf("%d", &N);
35          for (int i = 1; i <= N; i++)
36          {
37              scanf("%d %d", &input[i][0], &input[i][1]);
38          }
39          int pair = 0;
40          for (int i = 1; i <= N; i++)
41          {
42              for (int j = i + 1; j <= N; j++)
43              {
44                  long long int tmp = input[i][0] - input[j][0];
45                  tmp *= input[i][1] - input[j][1];
46                  if (tmp < 0)
47                  {
48                      array[pair].x = abs(input[i][0] - input[j][0]);
49                      array[pair].y = abs(input[i][1] - input[j][1]);
50                      pair++;
51                  }
52              }
53          }
54          if (pair == 0)
55          {
56              printf("Case #%d: 1\n", cas);
57          }
58          else
59          {
60              qsort(array, pair, sizeof(arr), comp);
61              int ans = 2;
62              for (int i = 1; i < pair; i++)
63              {
64                  long long int c1 = array[i - 1].x * array[i].y;
65                  long long int c2 = array[i - 1].y * array[i].x;
66                  if (c1 != c2)
67                  {
68                      ans++;
69                  }
70              }
71              printf("Case #%d: %d\n", cas, ans);
72          }
73      }
74      return 0;
75  }
