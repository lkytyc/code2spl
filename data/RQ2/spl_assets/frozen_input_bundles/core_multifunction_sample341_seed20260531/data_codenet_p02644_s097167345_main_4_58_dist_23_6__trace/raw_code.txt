5   		int n=sc.nextInt(),m=sc.nextInt(),k=sc.nextInt();
6   		int[]start=new int[2],end=new int[2];
7   		for(int i=0;i<2;i++) {
8   			start[i]=sc.nextInt()-1;
9   		}
10  		for(int i=0;i<2;i++) {
11  			end[i]=sc.nextInt()-1;
12  		}
13  		char[][]in=new char[n][m];
14  		for(int i=0;i<n;i++)
15  			in[i]=sc.nextLine().toCharArray();
16  		int inf=(int)1e9;
17  		int[][][]dist=new int[n][m][4];
18  		for(int[][]i:dist)
19  			for(int[]j:i)
20  				Arrays.fill(j, inf);
21  		LinkedList<int[]>q=new LinkedList<>();
22  		q.add(new int[] {start[0],start[1],k,-1,0});
23  		dist[start[0]][start[1]][0]=0;
24  		dist[start[0]][start[1]][1]=0;
25  		dist[start[0]][start[1]][2]=0;
26  		dist[start[0]][start[1]][3]=0;
27  		int[]dx= {0,1,0,-1},dy= {1,0,-1,0};
28  		while(!q.isEmpty()) {
29  			int cur[]=q.poll();
30  			int x=cur[0],y=cur[1],curK=cur[2],dir=cur[3];
31  			int curDist=cur[4];
32  			for(int i=0;i<4;i++) {
33  				int xx=x+dx[i],yy=y+dy[i];
34  				if(xx<0 || xx>=n || yy<0 || yy>=m || dist[xx][yy][(i)]!=inf || in[xx][yy]=='@')continue;
35  				int[]nxt=new int[] {xx,yy,curK-1,(i),0};
36  				if(nxt[2]<0) {
37  					nxt[2]=k-1;
38  					dist[xx][yy][(i)]=curDist+1;
39  				}
40  				else {
41  					if(dir==(i)) {
42  						dist[xx][yy][(i)]=curDist;
43  					}
44  					else {
45  						dist[xx][yy][(i)]=curDist+1;
46  						nxt[2]=k-1;
47  					}
48  				}
49  				nxt[4]=dist[xx][yy][(i)];
50  				q.add(nxt);
51  			}
52  		}
53  		int ans=inf;
54  		for(int i=0;i<4;i++) {
55  			ans=Math.min(ans, dist[end[0]][end[1]][i]);
56  		}
57  		pw.println(ans==inf?-1:ans);
58  	}
