1   import sys
2   import itertools
3   import math
4   import collections
5   import functools
6   sys.setrecursionlimit(10000)
7   def inputInts():
8       return map(int, raw_input().split())
9   def solve(N, M):
10      topHigh = None
11      botHigh = None
12      topLow = None
13      botLow = None
14      for i in xrange(N):
15          for j in xrange(i+1, N):
16              a = M[i][0]
17              b = M[i][1]
18              c = M[j][0]
19              d = M[j][1]
20              if a == c or b == d:
21                  continue
22              top = a-c
23              bot = d-b
24              wasNeg = (top * bot < 0)
25              top = abs(top)
26              bot = abs(bot)
27              if b-d > 0: 
28                  if wasNeg:
29                      return None
30                  if topHigh is None or top * botHigh < topHigh * bot:
31                      topHigh = top
32                      botHigh = bot
33              else: 
34                  if wasNeg:
35                      continue
36                  if topLow is None or top * botHigh > topHigh * bot:
37                      topLow = top
38                      botLow = bot
39      if topHigh is None and topLow is None:
40          return [1, 1]
41      if topHigh is None:
42          if botLow == topLow:
43              return [1, 2]
44          elif topLow > botLow:
45              return [1, int(math.ceil(topLow/botLow))]
46          else:
47              return [int(math.ceil(botLow/topLow)), 1]
48      if topLow is None:
49          if botHigh == topHigh:
50              return [2, 1]
51          elif botHigh > topHigh:
52              return [int(math.ceil(botHigh/topHigh)), 1]
53          else:
54              return [1, int(math.ceil(topHigh/botHigh))]
55      a = topHigh / float(botHigh)
56      b = topLow / float(botLow)
57      if a <= b:
58          return None
59      xMin = 1
60      xMax = 1e20
61      while xMax > xMin:
62          mid = int((xMax + xMin) / 2.)
63          low = b*mid
64          high = a*mid
65          if low == int(low):
66              if high > low + 1:
67                  xMax = mid
68              else:
69                  xMin = mid+1
70          else:
71              if math.ceil(low) < high:
72                  xMax = mid
73              else:
74                  xMin = mid+1
75  T = int(raw_input())
76  for testId in range(T):
77      N = int(raw_input())
78      M = []
79      for n in xrange(N):
80          M.append(inputInts())
81      res = solve(N, M)
82      if res is None:
83          print "Case #{:d}: IMPOSSIBLE".format(testId+1)
84      else:
85          print "Case #{:d}: {:d} {:d}".format(testId+1, res[0], res[1])
