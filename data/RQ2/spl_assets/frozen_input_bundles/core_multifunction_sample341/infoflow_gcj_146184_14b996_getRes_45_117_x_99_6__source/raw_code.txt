1   import java.lang.*;
2   import java.io.*;
3   import java.util.*;
4   class Solution {
5       public static void main(String[]args) throws IOException {
6           BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
7           PrintWriter pw = new PrintWriter(System.out);
8           int T = Integer.parseInt(br.readLine());
9           for(int i=1;i<=T;i++) {
10              pw.println("Case #" + i + ": " + solve(br));
11          }
12          pw.flush();
13      }
14      private static String solve(BufferedReader br) throws IOException {
15          int N = Integer.parseInt(br.readLine());
16          List<P> paris = new ArrayList<>();
17          for (int i = 0; i < N; i++) {
18              StringTokenizer st = new StringTokenizer(br.readLine());
19              int x = Integer.parseInt(st.nextToken());
20              int y = Integer.parseInt(st.nextToken());
21              paris.add(new P(x, y));
22          }
23          String res = getRes(N, paris);
24          return res;
25      }
26      private static boolean check(String res, List<P> paris) {
27          if (res.equals("IMPOSSIBLE")) return true;
28          StringTokenizer st = new StringTokenizer(res);
29          int x = Integer.parseInt(st.nextToken());
30          int y = Integer.parseInt(st.nextToken());
31          int[] v = new int[paris.size()];
32          for (int i = 0; i < paris.size(); i++) {
33              int z = paris.get(i).a * x + paris.get(i).b * y;
34              v[i] = z;
35          }
36          int[] v2 = new int[v.length];
37          System.arraycopy(v, 0, v2, 0, v.length);
38          Arrays.sort(v);
39          boolean r = Arrays.equals(v, v2);
40          if (!r) {
41              debug(v, v2);
42          }
43          return r;
44      }
45      private static String getRes(int n, List<P> paris) {
46          TreeSet<F> pfracs = new TreeSet<>();
47          TreeSet<F> nfracs = new TreeSet<>();
48          boolean hasDup = false;
49          for (int i = 0; i < n; i++) {
50              P p1 = paris.get(i);
51              for (int j = i + 1; j < n; j++) {
52                  P p2 = paris.get(j);
53                  int d1 = p1.a - p2.a;
54                  int d2 = p1.b - p2.b;
55                  if ((d1 == 0 && d2 == 0) || (d1 > 0 && d2 > 0)) {
56                      hasDup = true;
57                      break;
58                  }
59                  if (d1 != 0 && d2 != 0 && (d1 > 0 || d2 > 0)) {
60                      if (d1 < 0) {
61                          pfracs.add(new F(Math.abs(d1), Math.abs(d2)));
62                      }
63                      if (d2 < 0) {
64                          nfracs.add(new F(Math.abs(d1), Math.abs(d2)));
65                      }
66                  }
67              }
68              if (hasDup) break;
69          }
70          if (hasDup) {
71              return "IMPOSSIBLE";
72          }
73          F above = pfracs.isEmpty() ? null : pfracs.first();
74          F below = nfracs.isEmpty() ? null : nfracs.last();
75          if (above != null && below != null && above.compareTo(below) <= 0) {
76              return "IMPOSSIBLE";
77          }
78          if (above == null && below == null) {
79              return "1 1";
80          }
81          if (above == null) {
82              long z = below.num;
83              long r = below.denum;
84              long y = 0;
85              if (r % z == 0) {
86                  y = r / z;
87              } else {
88                  y = r / z + 1;
89              }
90              return "1 " + (y + 1);
91          }
92          if (below == null) {
93              long z = above.num;
94              long r = above.denum;
95              long x = 0;
96              if (r % z == 0) {
97                  x = r / z;
98              } else {
99                  x = r / z + 1;
100              }
101              return (x + 1) + " 1";
102          }
103          int gcd = gcd(above.denum, below.denum);
104          long b2 = above.denum / gcd;
105          long lcd = below.denum * b2;
106          long a = above.num * (lcd / above.denum);
107          long b = below.num * (lcd / below.denum);
108          long z = a - b;
109          long r = 2 * lcd;
110          long x = 0;
111          if (r % z == 0) {
112              x = r / z;
113          } else {
114              x = r / z + 1;
115          }
116          return x + " " + (x + 1);
117      }
118      private static int gcd(int a, int b) {
119          return (b != 0) ? gcd(b, a % b) : a;
120      }
121      private static final class P {
122          int a;
123          int b;
124          public P(int a, int b) {
125              this.a = a;
126              this.b = b;
127          }
128      }
129      private static final class F implements Comparable<F> {
130          int num;
131          int denum;
132          public F(int num, int denum) {
133              int gcd = gcd(num, denum);
134              this.num = num / gcd;
135              this.denum = denum / gcd;
136          }
137          @Override
138          public int compareTo(F o) {
139              int gcd = gcd(o.denum, denum);
140              long b2 = o.denum / gcd;
141              long lcd = denum * b2;
142              long a = num * (lcd / denum);
143              long b = o.num * (lcd / o.denum);
144              return Long.compare(a, b);
145          }
146      }
147      public static void debug(Object...args) {
148          System.out.println(Arrays.deepToString(args));
149      }
