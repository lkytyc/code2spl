1   #include <stdio.h>
2   #include <stdlib.h>
3   #include <string.h>
4   #include <math.h>
5   int A[101][101];
6   int B[101][101];
7   int r, c, h, v;
8   int guests, chips, share;
9   int fr(int rRemaining, int i, int j) {
10  	int k;
11  	if( (rRemaining + 1)*share != (B[r][j] - B[i][j]) )
12  		return 0;
13  	if(rRemaining == 0) 
14  		return 1;
15  	k = i+1;
16  	while(k < r) {
17  		if(B[k][j] - B[i][j] == share) {
18  			return fr(rRemaining - 1, k, j);
19  		}
20  		k++;
21  	}
22  	return 0;
23  }
24  int fc(int cRemaining, int i, int j) {
25  	int k;
26  	if( (cRemaining + 1)*share != (B[i][c] - B[i][j]) )
27  		return 0;
28  	if(cRemaining == 0) 
29  		return 1;
30  	k = i+1;
31  	while(k < c) {
32  		if(B[i][k] - B[i][j] == share) {
33  			return fr(cRemaining - 1, i, k);
34  		}
35  		k++;
36  	}
37  	return 0;
38  }
39  int main() {
40  	int t, tc;
41  	int i, j, possible;
42  	char str[101];
43  	scanf("%d", &t);
44  	tc = 1;
45  	while(tc <= t) {
46  		scanf("%d", &r);
47  		scanf("%d", &c);
48  		scanf("%d", &h);
49  		scanf("%d", &v);
50  		guests = (h+1)*(v+1);
51  		chips = 0;
52  		for (i = 0; i <= r; ++i) {
53  			if(i != 0) scanf("%s", str);
54  			for (j = 0; j <= c; ++j) {
55  				if(i == 0 || j == 0) 
56  					A[i][j] = B[i][j] = 0;
57  				else {
58  					A[i][j] = (str[j-1] == '@') ? 1 : 0; 
59  					if(A[i][j]) 
60  						chips++;
61  				}
62  			}
63  		}
64  		if(chips % guests != 0) {
65  			printf("Case #%d: IMPOSSIBLE\n", tc);
66  		} else {
67  			for (i = 1; i <= r; ++i) {
68  				for (j = 1; j <= c; ++j) {
69  					B[i][j] = A[i][j] + B[i-1][j] + B[i][j-1] - B[i-1][j-1];
70  				}
71  			}
72  			share = chips / guests;
73  			possible = 0;
74  			for (i = 1; i < r && !possible; ++i) {
75  				for (j = 1; j < c && !possible; ++j) {
76  					if(B[i][j] == share) {
77  						possible = 1;
78  						if(!fr(h - 1, i, j)) {
79  							possible = 0;
80  						}
81  						if(!fc(v - 1, i, j)) {
82  							possible = 0;
83  						}
84  						if(possible) {
85  							printf("Case #%d: POSSIBLE\n", tc);
86  						}
87  					}
88  				}
89  			}
90  			if(!possible) printf("Case #%d: IMPOSSIBLE\n", tc);
91  		}
92  		tc++;
93  	}
94  	return 0;
95  }
