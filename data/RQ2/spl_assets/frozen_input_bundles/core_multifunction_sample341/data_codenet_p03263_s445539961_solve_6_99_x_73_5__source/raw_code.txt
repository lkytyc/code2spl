107  	static class Scanner {
108  		BufferedReader br;
109  		Scanner (InputStream in) {
110  			br = new BufferedReader(new InputStreamReader(in));
111  		}
112  		private boolean isPrintable(int ch) {
113  			return ch >= '!' && ch <= '~';
114  		}
115  		private boolean isCRLF(int ch) {
116  			return ch == '\n' || ch == '\r' || ch == -1;
117  		}
118  		private int nextPrintable() {
119  			try {
120  				int ch;
121  				while (!isPrintable(ch = br.read())) {
122  					if (ch == -1) {
123  						throw new NoSuchElementException();
124  					}
125  				}
126  				return ch;
127  			} catch (IOException e) {
128  				throw new NoSuchElementException();
129  			}
130  		}
131  		String next() {
132  			try {
133  				int ch = nextPrintable();
134  				StringBuilder sb = new StringBuilder();
135  				do {
136  					sb.appendCodePoint(ch);
137  				} while (isPrintable(ch = br.read()));
138  				return sb.toString();
139  			} catch (IOException e) {
140  				throw new NoSuchElementException();
141  			}
142  		}
143  		int nextInt() {
144  			try {
145  				boolean negative = false;
146  				int res = 0;
147  				int limit = -Integer.MAX_VALUE;
148  				int radix = 10;
149  				int fc = nextPrintable();
150  				if (fc < '0') {
151  					if (fc == '-') {
152  						negative = true;
153  						limit = Integer.MIN_VALUE;
154  					} else if (fc != '+') {
155  						throw new NumberFormatException();
156  					}
157  					fc = br.read();
158  				}
159  				int multmin = limit / radix;
160  				int ch = fc;
161  				do {
162  					int digit = ch - '0';
163  					if (digit < 0 || digit >= radix) {
164  						throw new NumberFormatException();
165  					}
166  					if (res < multmin) {
167  						throw new NumberFormatException();
168  					}
169  					res *= radix;
170  					if (res < limit + digit) {
171  						throw new NumberFormatException();
172  					}
173  					res -= digit;
174  				} while (isPrintable(ch = br.read()));
175  				return negative ? res : -res;
176  			} catch (IOException e) {
177  				throw new NoSuchElementException();
178  			}
179  		}
180  		long nextLong() {

192  						throw new NumberFormatException();

197  				int ch = fc;
198  				do {
199  					int digit = ch - '0';
200  					if (digit < 0 || digit >= radix) {
201  						throw new NumberFormatException();
202  					}
203  					if (res < multmin) {
204  						throw new NumberFormatException();
205  					}
206  					res *= radix;
207  					if (res < limit + digit) {
208  						throw new NumberFormatException();
209  					}
210  					res -= digit;
211  				} while (isPrintable(ch = br.read()));
212  				return negative ? res : -res;
213  			} catch (IOException e) {
214  				throw new NoSuchElementException();
215  			}
216  		}
217  		float nextFloat() {
218  			return Float.parseFloat(next());
219  		}
220  		double nextDouble() {
221  			return Double.parseDouble(next());
222  		}
223  		String nextLine() {
224  			try {
225  				int ch;
226  				while (isCRLF(ch = br.read())) {
227  					if (ch == -1) {
228  						throw new NoSuchElementException();
229  					}
230  				}
231  				StringBuilder sb = new StringBuilder();
232  				do {
233  					sb.appendCodePoint(ch);
234  				} while (!isCRLF(ch = br.read()));
235  				return sb.toString();
236  			} catch (IOException e) {
237  				throw new NoSuchElementException();
238  			}
239  		}
240  		void close() {
241  			try {
242  				br.close();
243  			} catch (IOException e) {
244  			}
245  		}
