93      static class FastScanner {
94          private InputStream stream;
95          private byte[] buf = new byte[1024];
96          private int curChar;
97          private int numChars;
98          private FastScanner.SpaceCharFilter filter;
99          public FastScanner(InputStream stream) {
100              this.stream = stream;
101          }
102          public int read() {
103              if (numChars == -1) {
104                  throw new InputMismatchException();
105              }
106              if (curChar >= numChars) {
107                  curChar = 0;
108                  try {
109                      numChars = stream.read(buf);
110                  } catch (IOException e) {
111                      throw new InputMismatchException();
112                  }
113                  if (numChars <= 0) {
114                      return -1;
115                  }
116              }
117              return buf[curChar++];
118          }
119          public String nextString() {
120              int c = read();
121              while (isSpaceChar(c)) {
122                  c = read();
123              }
124              StringBuilder res = new StringBuilder();
125              do {
126                  if (Character.isValidCodePoint(c)) {
127                      res.appendCodePoint(c);
128                  }
129                  c = read();
130              } while (!isSpaceChar(c));
131              return res.toString();
132          }
133          public boolean isSpaceChar(int c) {
134              if (filter != null) {
135                  return filter.isSpaceChar(c);
136              }
137              return isWhitespace(c);
138          }
139          public static boolean isWhitespace(int c) {
140              return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
141          }
142          public interface SpaceCharFilter {
143              public boolean isSpaceChar(int ch);
144          }
145      }
