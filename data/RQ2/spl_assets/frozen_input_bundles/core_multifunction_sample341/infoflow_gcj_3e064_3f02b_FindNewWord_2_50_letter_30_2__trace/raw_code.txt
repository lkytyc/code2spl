2   def FindNewWord(w_list):
3   	w_list.sort()
4   	L = len(w_list[0])
5   	if len(w_list) == 1 or L ==1:
6   		return "-"
7   	prefix_index = [0]
8   	N = len(w_list)
9   	last_prefix = w_list[0][0]
10  	for i in range(1, N):
11  		if w_list[i][0] != last_prefix:
12  			last_prefix = w_list[i][0]
13  			prefix_index.append(i)
14  	prefix_index.append(N)
15  	curr_word = ""
16  	for p in range(1, L):
17  		available_letters = {w_list[0][p]}
18  		for k in range(1, N):
19  			available_letters.add(w_list[k][p])
20  		n_prefixes = len(prefix_index)
21  		isFound=False
22  		new_prefix_index = []
23  		for n_prefix in range(n_prefixes -1):
24  			start_index = prefix_index[n_prefix]
25  			end_index   = prefix_index[n_prefix+1]
26  			letters_in_prefix = {w_list[start_index][p]}
27  			last_letter = w_list[start_index][p]
28  			new_prefix_index.append(start_index)
29  			for item in range(start_index + 1, end_index):
30  				letter = w_list[item][p]
31  				if letter != last_letter:
32  					last_letter =letter
33  					letters_in_prefix.add(w_list[item][p])
34  					new_prefix_index.append(item)
35  			remaining_letters = available_letters.difference(letters_in_prefix)
36  			if len(remaining_letters) > 0:
37  				letter = next(iter(remaining_letters))
38  				curr_word = w_list[start_index][:p] + letter
39  				isFound = True
40  				break
41  		if isFound:
42  			break
43  		new_prefix_index.append(N)
44  		prefix_index = deepcopy(new_prefix_index)
45  	if not isFound:
46  		return "-"
47  	size_p = len(curr_word)
48  	for p in range(size_p, L):
49  		curr_word+= w_list[0][p]
50  	return  curr_word
