3   long long int modpow(long long int a, long long int n, long long int p)
4   {
5   	if (n < 0)return 0;
6   	long long int res = 1;
7   	while (n > 0)
8   	{
9   		if (n % 2 == 1)
10  			res = res * a % p;
11  		a = a * a % p;
12  		n /= 2;
13  	}
14  	return res;
15  }
