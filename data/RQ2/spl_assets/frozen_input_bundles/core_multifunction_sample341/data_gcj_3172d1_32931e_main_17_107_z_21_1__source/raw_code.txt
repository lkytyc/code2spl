1   import java.io.*;
2   import java.lang.*;
3   import java.util.*;
4   public class Solution{
5       static class Pair implements Comparable<Pair>{
6           long value;
7           int count;
8           public Pair(long value,int count){
9               this.value = value;
10              this.count = count;
11          }
12          @Override
13          public int compareTo(Pair o){
14              return o.count - this.count;
15          }
16      }
17      public static void main(String[] args)throws IOException{
18          BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
19          int t = new Integer(br.readLine());
20          StringBuffer sb = new StringBuffer();
21          for(int z=1;z<=t;z++){
22              sb.append("Case #"+z+": ");
23              StringTokenizer st = new StringTokenizer(br.readLine());
24              StringTokenizer st1 = new StringTokenizer(br.readLine());
25              int n = new Integer(st.nextToken());
26              int d = new Integer(st.nextToken());
27              if(n==1&&d==3){
28                  sb.append("2\n");
29                  continue;
30              }
31              else if(n==1&&d==2){
32                  sb.append("1\n");
33                  continue;
34              }
35              long[] array = new long[n];
36              for(int i=0;i<n;i++){
37                  array[i]=new Long(st1.nextToken());
38              }
39              Arrays.sort(array);
40              Pair[] ps = new Pair[n];
41              int w=0;
42              int count = 1;
43              for(int i=1;i<n;i++){
44                  if(array[i]==array[i-1]){
45                      count++;
46                  }
47                  else{
48                      ps[w++]=new Pair(array[i-1],count);
49                      count=1;
50                  }
51              }
52              ps[w++]=new Pair(array[n-1],count);
53              Arrays.sort(ps,0,w);
54              if(ps[0].count>=d){
55                  sb.append("0\n");
56                  continue;
57              }
58              else{
59                  if(d==2&&ps[0].count==1){
60                      sb.append("1\n");
61                      continue;
62                  }
63                  else if(d==3&&ps[0].count==2&&ps[0].value!=array[n-1]){
64                      sb.append("1\n");
65                      continue;
66                  }
67                  else if(d==3&&ps[0].count==2&&ps[0].value==array[n-1]){
68                      if(ps[1].count==2){
69                          sb.append("1\n");
70                          continue;
71                      }
72                      boolean flag = false;
73                      outer:for(int i=0;i<n;i++){
74                          for(int j=0;j<n;j++){
75                              if(array[j]==2*array[i]||array[i]==2*array[j]){
76                                  sb.append("1\n");
77                                  flag = true;
78                                  break outer;
79                              }
80                          }
81                      }
82                      if(!flag){
83                          sb.append("2\n");
84                          continue;
85                      }
86                  }
87                  else if(d==3&&ps[0].count==1){
88                      boolean flag = false;
89                      outer:for(int i=0;i<n;i++){
90                          for(int j=0;j<n;j++){
91                              if(array[j]==2*array[i]||array[i]==2*array[j]){
92                                  sb.append("1\n");
93                                  flag = true;
94                                  break outer;
95                              }
96                          }
97                      }
98                      if(!flag){
99                          sb.append("2\n");
100                          continue;
101                      }
102                  }
103              }
104          }
105          br.close();
106          System.out.println(sb);
107      }
108  }
