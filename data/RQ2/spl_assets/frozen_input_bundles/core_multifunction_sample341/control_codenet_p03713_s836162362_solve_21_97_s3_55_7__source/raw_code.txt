99      static class InputReader {
100          StringTokenizer st;
101          BufferedReader br;
102          public InputReader(InputStream is) {
103              BufferedReader br = new BufferedReader(new InputStreamReader(is));
104              this.br = br;
105          }
106          public String next() {
107              if (st == null || !st.hasMoreTokens()) {
108                  String nextLine = null;
109                  try {
110                      nextLine = br.readLine();
111                  } catch (IOException e) {
112                      throw new RuntimeException(e);
113                  }
114                  if (nextLine == null)
115                      return null;
116                  st = new StringTokenizer(nextLine);
117              }
118              return st.nextToken();
119          }
120          public int nextInt() {
121              return Integer.parseInt(next());
122          }
